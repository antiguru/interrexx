/* An untrapped RAISE NOMETHOD is a 97.1 error whose ADDITIONAL holds the
   receiver and the DESCRIPTION objects themselves, the null string where
   there is no DESCRIPTION. The message names them by default name; a reader
   that concatenates an item runs its MAKESTRING then. */
signal on syntax name s1
raise nomethod additional ('obj') description (.key~new)
s1: o = condition('O'); say o~additional~items o~additional[2]~class~id
say o~message
say '['o~additional[1]']' '['o~additional[2]']'
signal on syntax name s2
a = .array~of('p', 'q')
raise nomethod additional (a) description 'm'
s2: o = condition('O'); say o~additional[1]~class~id (o~additional[1] == a)
say o~message
signal on syntax name s3
raise nomethod additional 'obj'
s3: o = condition('O'); say o~additional~items o~additional[2]~class~id '['o~additional[2]']'
signal on syntax name s4
call sub
s4: o = condition('O'); say o~additional[1]~class~id o~additional[2]~class~id
exit
sub: raise nomethod additional (.key~new) description (.array~new) exit
