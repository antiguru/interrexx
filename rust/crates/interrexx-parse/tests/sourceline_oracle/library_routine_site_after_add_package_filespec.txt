count 8
/* A call site that found FILESPEC, a builtin, keeps it: a public routine of the
   same name that addPackage made callable is not found, on the next pass
   either. */
source = .array~of('::routine filespec public', "  return 'pk' arg(1)")
do i = 1 to 2
  say 'internal' i filespec('N', '/a/b.c')
  if i = 1 then .context~package~addPackage(.package~new('filespec.cls', source))
end
