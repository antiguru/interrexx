   flag. */

::class sink
::method init
  expose seen
  seen = .array~new
::method seen
  expose seen
  return seen
::method lineout unguarded
  expose seen
  use arg v
  tl = v~traceline
  if wordpos(word(tl, 1), '>I> <I<') > 0 then tl = word(tl, 1)
  else tl = subword(tl, 1, 3)
  if v~hasEntry('ISGUARDED') then
    seen~append(tl '| pool' v~attributepool 'guarded' v~isguarded 'count' v~scopelockcount 'lock' v~hasscopelock)
  return 0
