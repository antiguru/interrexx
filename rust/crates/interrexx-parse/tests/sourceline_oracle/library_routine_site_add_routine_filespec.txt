/* A call site that found FILESPEC, a builtin, keeps it: the routine ~addRoutine
   adds under the same name is not found, on the next pass either. */
p = .context~package
do i = 1 to 2
  say 'internal' i filespec('N', '/a/b.c')
  if i = 1 then p~addRoutine('FILESPEC', .routine~new('x', 'return "added" arg(1)'))
end
