7/// Below this working size division uses stack buffers and can never fail an
8/// allocation, so the reservation probe in `div` is skipped the same way.
9const FAST_BUFFER: u64 = 48;
10
11impl Number {
12    pub fn mul(&self, other: &Number, digits: u64) -> Result<Number, ArithError> {
13        self.mul_checked(other, digits, Number::check_range)
14    }
15
16    /// The product `**` builds its power from. Only a first digit past the
17    /// exponent range is refused, which is `Overflow`; a last digit below the
18    /// range is not, because the power decides on its final result alone.
19    pub(crate) fn mul_intermediate(
20        &self,
21        other: &Number,
22        digits: u64,
23    ) -> Result<Number, ArithError> {
24        self.mul_checked(other, digits, Number::check_overflow)
25    }
26
27    fn mul_checked(
28        &self,
29        other: &Number,
30        digits: u64,
31        check: fn(&Number) -> Result<(), ArithError>,
32    ) -> Result<Number, ArithError> {
33        // checkNumber truncates an over-long operand to DIGITS + 1 and does
34        // NOT round it. Rounding operands here instead would turn
35        // `2 * 1.5` at DIGITS 1 into `2 * 2` = 4, where the answer is 3.
36        let left = self.truncated_to(crate::working_length(digits));
37        let right = other.truncated_to(crate::working_length(digits));
38
39        if left.is_zero() || right.is_zero() {
40            return Ok(Number::zero());
41        }
42
43        if let Some(exact) = exact_integer_product(&left, &right, digits) {
44            return Ok(exact);
45        }
46        Number::mul_long(&left, &right, digits, check)
47    }
48
49    /// `left * right` by long multiplication over the digit vectors, which is
50    /// what [`Number::mul`] runs wherever [`exact_integer_product`] declines.
51    #[inline(always)]
52    fn mul_long(
53        left: &Number,
54        right: &Number,
55        digits: u64,
56        check: fn(&Number) -> Result<(), ArithError>,
57    ) -> Result<Number, ArithError> {
58        let product = mul_magnitudes(&left.digits, &right.digits);
59        // Saturated: a bare `digits` past usize means "keep everything",
60        // which the length test below then decides.
61        let digits_usize = usize::try_from(digits).unwrap_or(usize::MAX);
62
63        // Anything beyond DIGITS + 1 digits is dropped from the low end and
64        // its count folded into the exponent; the extra digit is what the
65        // final rounding then looks at.
66        let (kept, extra) = if product.len() > digits_usize {
67            let keep = digits_usize + 1;
68            (
69                Digits::from_slice(&product[..keep.min(product.len())]),
70                product.len() - keep,
71            )
72        } else {
73            (product, 0)
74        };
75
76        // Checked, not wrapping: two operands near the exponent limit
77        // multiply to something well outside i32. See
78        // `ArithError::ExponentComputationOverflow`'s doc comment for why
79        // this is believed unreachable, and what it falls back to if it
80        // ever isn't.
81        let exponent = left
82            .exponent
83            .checked_add(right.exponent)
84            .and_then(|e| e.checked_add(i32::try_from(extra).ok()?))
85            .ok_or(ArithError::ExponentComputationOverflow)?;
86        let negative = left.negative != right.negative;
87        let raw = Number {
88            negative,
89            digits: kept,
90            exponent,
91        };
92        let rounded = raw.into_round(digits);
93        let result = Number::assemble(rounded.negative, rounded.digits, rounded.exponent);
94        check(&result)?;
95        Ok(result)
96    }
97}

99/// The exact product, when both operands are plain integers whose digits and
100/// product an `i64` holds, and the product needs no rounding at `digits`.
101fn exact_integer_product(left: &Number, right: &Number, digits: u64) -> Option<Number> {
102    if left.exponent != 0 || right.exponent != 0 {
103        return None;
104    }
105    let width = left.digits.len() + right.digits.len();
106    // Eighteen digits is the widest product an `i64` holds whatever its
107    // digits are, so this bounds the folds below as well as the multiply.
108    if width > 18 || u64::cast_from(width) > digits {
109        return None;
110    }
111    let fold = |number: &Number| {
112        number
113            .digits
114            .iter()
115            .fold(0i64, |acc, digit| acc * 10 + i64::from(*digit))
116    };
117    let mut product = Number::from_i64(fold(left) * fold(right));
118    product.negative = left.negative != right.negative;
119    Some(product)
120}

122/// The widest digit string a `u64` holds whatever its digits are.
123const WORD_DIGITS: usize = 19;
124
125/// The whole number a digit string of at most [`WORD_DIGITS`] spells.
126fn fold_digits(digits: &[u8]) -> u64 {
127    digits
128        .iter()
129        .fold(0u64, |acc, digit| acc * 10 + u64::from(*digit))
130}

132/// Exact product of two digit vectors, most significant first.
133fn mul_magnitudes(a: &[u8], b: &[u8]) -> Digits {
134    if a.len() <= WORD_DIGITS && b.len() <= WORD_DIGITS {
135        return word_product(a, b);
136    }
137    let product = crate::limbs::mul(
138        &crate::limbs::from_digits(a, 0),
139        &crate::limbs::from_digits(b, 0),
140    );
141    Digits::from_vec(crate::limbs::to_digits(&product))
142}

144/// [`mul_magnitudes`] for operands a `u64` holds each, whose product a
145/// `u128` holds.
146fn word_product(a: &[u8], b: &[u8]) -> Digits {
147    let product = u128::from(fold_digits(a)) * u128::from(fold_digits(b));
148    let mut buf = [0u8; 2 * WORD_DIGITS];
149    // Most products fit one word, and converting one word needs no `u128`
150    // division at all.
151    let (mut word, mut end) = match u64::try_from(product) {
152        Ok(word) => (word, buf.len()),
153        Err(_) => {
154            // Two words of nineteen digits each, so each half converts in
155            // `u64`: the low half to all its digits, the high half below.
156            const SPLIT: u128 = 10_000_000_000_000_000_000;
157            // The product is below `SPLIT^2`, so both halves are below `SPLIT`.
158            let half = |value: u128| u64::try_from(value).expect("below 10^19");
159            let mut low = half(product % SPLIT);
160            for slot in buf[WORD_DIGITS..].iter_mut().rev() {
161                *slot = last_digit(low);
162                low /= 10;
163            }
164            (half(product / SPLIT), WORD_DIGITS)
165        }
166    };
167    loop {
168        end -= 1;
169        buf[end] = last_digit(word);
170        word /= 10;
171        if word == 0 {
172            break;
173        }
174    }
175    Digits::from_slice(&buf[end..])
176}

178/// Which division the caller wants. All three share one algorithm in the
179/// interpreter, differing in when they stop and what they return.
180#[derive(Copy, Clone, PartialEq, Eq, Debug)]
181pub enum DivOp {
182    /// `/`
183    Divide,
184    /// `%` -- the integer part of the quotient.
185    IntegerDivide,
186    /// `//` -- the residue left after an integer divide.
187    Remainder,
188}
189
190impl Number {
191    pub fn div(&self, other: &Number, digits: u64, op: DivOp) -> Result<Number, ArithError> {
192        if other.is_zero() {
193            return Err(ArithError::DivideByZero);
194        }
195        if self.is_zero() {
196            return Ok(Number::zero());
197        }
198
199        let left = self.truncated_to(crate::working_length(digits));
200        let right = other.truncated_to(crate::working_length(digits));
201        let negative = left.negative != right.negative;
202
203        // The interpreter's estimate of where the quotient's first digit
204        // lands, from the operand exponents and lengths. Same believed-
205        // unreachable guard as `mul`'s above.
206        let calc_exp = left
207            .exponent
208            .checked_sub(right.exponent)
209            .and_then(|e| {
210                e.checked_add(signed_count(left.digits.len()) - signed_count(right.digits.len()))
211            })
212            .ok_or(ArithError::ExponentComputationOverflow)?;
213
214        // A quotient below 1 has no integer part, so % is zero and // is the
215        // left operand unchanged.
216        if calc_exp < 0 && op != DivOp::Divide {
217            return Ok(match op {
218                DivOp::IntegerDivide => Number::zero(),
219                _ => {
220                    let mut r = left.into_owned();
221                    r.negative = self.negative && !r.is_zero();
222                    r
223                }
224            });
225        }
226
227        // Division's working storage is sized by DIGITS itself rather than by
228        // its operands: division allocates `3 * ((digits + 1) * 2 + 1)` bytes up
229        // front, before the first quotient digit, and a failed allocation is
230        // error 5, "System resources exhausted". That allocation *is* the whole
231        // bound -- there is no fixed threshold. So `4.0 / 2` and `123456.0 % 2`
232        // both fail with 5.0 at DIGITS 999999999999999999 even though their
233        // results are tiny, because the buffer is sized before anyone looks at
234        // the operands; `1 / 7` computes at DIGITS 1e10 but is 5.0 at 1e11 on
235        // the same machine (the boundary is what malloc will grant, so it moves
236        // with the machine). Mirrored here with a fallible reservation of the
237        // same request, so the same boundary falls out of the same cause -- and
238        // only past a small-size cutoff, so ordinary divisions never pay for an
239        // allocation probe. This sits after the `calc_exp < 0` early returns
240        // above because the allocation comes after the equivalent ones: a
241        // `%`/`//` whose quotient has no integer part never reaches the
242        // allocation at all.
243        let total_digits = digits.saturating_add(1).saturating_mul(2).saturating_add(1);
244        if total_digits > FAST_BUFFER {
245            let request = usize::try_from(total_digits.saturating_mul(3)).unwrap_or(usize::MAX);
246            let mut probe: Vec<u8> = Vec::new();
247            probe
248                .try_reserve_exact(request)
249                .map_err(|_| ArithError::SystemResources)?;
250            // The reservation is the whole point of the buffer, and nothing
251            // reads it afterwards, so the optimiser is entitled to delete the
252            // allocation and with it the failure this arm exists to report.
253            // It does: under `lto = "fat"` and without this line, `numeric
254            // digits 999999999999999999; r = 4.0 / 2` answers `2` instead of
255            // raising error 5, while the same build without LTO raises it.
256            // `black_box` forces the vector to exist, which forces the
257            // allocation to have been attempted.
258            std::hint::black_box(&probe);
259        }
260
261        // Long-divide the digit strings, generating one more digit than
262        // DIGITS so the final rounding has something to look at. `%` and `//`
263        // keep only the integer part of the quotient, so their division stops
264        // at the units digit: the digits it would generate past that are the
265        // ones the truncation below drops. `calc_exp >= 0` here, so the units
266        // digit comes no earlier than the divisor's length.
267        let want = crate::working_length(digits);
268        let units = if op == DivOp::Divide {
269            usize::MAX
270        } else {
271            usize::try_from(calc_exp)
272                .ok()
273                .and_then(|e| e.checked_add(right.digits.len()))
274                .unwrap_or(usize::MAX)
275        };
276        let (mut q, consumed) = long_divide(&left.digits, &right.digits, want, units);
277        debug_assert!(
278            op != DivOp::Divide || consumed >= left.digits.len(),
279            "an operand truncated to the working length is consumed whole"
280        );
281
282        // Each numerator digit consumed past the units digit scales the
283        // quotient down one place: value = q * 10^(left.exponent -
284        // right.exponent + left length - consumed). Same believed-
285        // unreachable guard as the two above.
286        let q_exp = i64::from(left.exponent) - i64::from(right.exponent)
287            + signed_len(left.digits.len())
288            - i64::try_from(consumed).unwrap_or(i64::MAX);
289        let q_exp = i32::try_from(q_exp).map_err(|_| ArithError::ExponentComputationOverflow)?;
290
291        if op == DivOp::Divide {
292            let raw = Number {
293                negative,
294                digits: q,
295                exponent: q_exp,
296            };
297            // The range check goes after rounding but BEFORE the trailing
298            // zeros come off. Those zeros are significant to the check: a
299            // quotient of 1.0120 sits one power of ten lower than the 1.012
300            // it prints as, and that is the difference between representable
301            // and not.
302            let mut rounded = raw.into_round(digits);
303            rounded.check_range()?;
304            // Division strips trailing zeros; `1 / 7.7` at DIGITS 3 is 0.13,
305            // not 0.130. Addition and the remainder operators do the
306            // opposite and keep them -- `1.50 + 0.50` is 2.00 and
307            // `100 // 6.66666665` is 0.010 -- because there the zeros come
308            // from the operands rather than from a generated quotient.
309            while rounded.digits.len() > 1 && *rounded.digits.last().unwrap() == 0 {
310                rounded.digits.pop();
311                rounded.exponent += 1;
312            }
313            let result = Number::assemble(rounded.negative, rounded.digits, rounded.exponent);
314            result.check_range()?;
315            return Ok(result);
316        }
317
318        // For % and //, keep only the integer part of the quotient. A
319        // division stopped at the units digit before any significant one has
320        // an empty quotient, which is the zero integer part.
321        if q.is_empty() {
322            q = Digits::single(0);
323        }
324        if q_exp < 0 {
325            let drop = usize::cast_from(q_exp.unsigned_abs());
326            if drop >= q.len() {
327                q = Digits::single(0);
328            } else {
329                q.truncate(q.len() - drop);
330            }
331        }
332        let int_digits = Number::assemble(negative, q, q_exp.max(0));
333        // Compared in i64, with `digits` saturated into it rather than
334        // narrowed: `digits` is a bare unbounded u64 (and even the
335        // `Settings`-bounded value legitimately reaches 10^18 - 1 now), so
336        // an `as i32`/`as i64` narrowing wraps negative and makes every
337        // non-zero quotient look too wide -- the same silent-narrowing
338        // defect `format` and `as_whole` each fixed before this. Saturation
339        // is exact: the left side is a digit count plus an exponent,
340        // nowhere near i64::MAX.
341        let digits_i64 = i64::try_from(digits).unwrap_or(i64::MAX);
342        if !int_digits.is_zero()
343            && signed_len(int_digits.digits.len()) + i64::from(int_digits.exponent) > digits_i64
344        {
345            // Only `%`/`//` reach here (`Divide` already returned above),
346            // and the interpreter reports each with its own, substitution-
347            // free text -- confirmed with `123456 % 2` and `123456 // 2`
348            // at DIGITS 3.
349            return Err(if op == DivOp::IntegerDivide {
350                ArithError::IntegerDivideNotWhole
351            } else {
352                ArithError::RemainderNotWhole
353            });
354        }
355
356        Ok(match op {
357            DivOp::IntegerDivide => int_digits,
358            _ => {
359                // remainder = left - (left % right) * right. The intermediate
360                // must be computed at enough precision to be exact, or a large
361                // dividend loses the low digits that ARE the remainder.
362                // Carried in u64, saturating: the old `as u32` narrowing of
363                // this sum truncated the working precision for large
364                // `digits` -- the same class as the comparison fix above,
365                // on the arithmetic side.
366                let exact_extra = u64::cast_from(
367                    left.digits.len() + right.digits.len() + int_digits.digits.len() + 10,
368                );
369                let exact = digits.saturating_add(exact_extra);
370                let product = int_digits.mul(&right, exact)?;
371                let mut r = left.sub(&product, exact)?;
372                r.negative = self.negative && !r.is_zero();
373                r.into_round(digits)
374            }
375        })
376    }
377}

379/// The widest divisor [`short_divide`] can carry in a machine word.
380const SHORT_DIVISOR_DIGITS: usize = 18;
381
382/// Divides two digit strings, returning up to `want` significant quotient
383/// digits and how many digits of the numerator (padded with zeros past its
384/// end) were consumed to produce them. The division also stops once it has
385/// consumed `max_consumed` digits, or as soon as it comes out even once the
386/// numerator's own digits are used up.
387fn long_divide(n: &[u8], d: &[u8], want: usize, max_consumed: usize) -> (Digits, usize) {
388    // Quotient digit `k` comes from consuming numerator digit `first + k`, so
389    // a bound on the digits consumed is a bound on the digits wanted. Only
390    // `%` and `//` set one; `/` divides exactly as it always did.
391    let want = if max_consumed == usize::MAX {
392        want
393    } else {
394        let first = first_quotient_digit(n, d);
395        if max_consumed < first {
396            return (Digits::new(), max_consumed);
397        }
398        want.min(max_consumed - first + 1)
399    };
400    if d.len() <= SHORT_DIVISOR_DIGITS {
401        let divisor = fold_digits(d);
402        // A zero divisor never arrives -- `Number::div` answers `DivideByZero`
403        // before this -- and neither route terminates on one, so the test
404        // routes it to the behaviour that was already here rather than to a
405        // new division by zero.
406        if divisor != 0 {
407            if want < LIMB_SHORT_QUOTIENT || !short_quotient_is_long(n, divisor, want) {
408                return short_divide(n, divisor, d.len(), want);
409            }
410            return limb_divide(n, d, want);
411        }
412    }
413    if want.saturating_mul(d.len()) >= LIMB_QUOTIENT && want <= LIMB_QUOTIENT_MAX {
414        return limb_divide(n, d, want);
415    }
416    wide_divide(n, d, want)
417}

419/// Whether a quotient by a divisor a word holds runs long enough that the
420/// limb route pays: not when it comes out even a few digits past a numerator
421/// a word holds, which is where the short division stops early and the limb
422/// route would not.
423#[inline(never)]
424fn short_quotient_is_long(n: &[u8], divisor: u64, want: usize) -> bool {
425    want <= LIMB_QUOTIENT_MAX
426        && !(n.len() <= SHORT_DIVISOR_DIGITS
427            && terminating_zeros(fold_digits(n) % divisor, divisor).is_some())
428}

430/// How many zeros after a numerator leaving `residue` modulo `divisor` it
431/// takes for the division to come out even, if it ever does. It does within
432/// as many zeros as the divisor has factors of 2 or 5, which is at most 59
433/// for a divisor below 10^18; and below 10^18 the residue times ten still
434/// fits a `u64`.
435fn terminating_zeros(mut residue: u64, divisor: u64) -> Option<usize> {
436    debug_assert!(divisor < 1_000_000_000_000_000_000, "a short divisor");
437    for zeros in 0..64 {
438        if residue == 0 {
439            return Some(zeros);
440        }
441        residue = residue * 10 % divisor;
442    }
443    None
444}

446/// The numerator digit whose consumption produces the first significant
447/// quotient digit: the divisor's length if the numerator's first that many
448/// digits are at least the divisor, and one more otherwise.
449fn first_quotient_digit(n: &[u8], d: &[u8]) -> usize {
450    let head = n.iter().copied().chain(std::iter::repeat(0)).take(d.len());
451    if head.cmp(d.iter().copied()).is_ge() {
452        d.len()
453    } else {
454        d.len() + 1
455    }
456}

458/// From this many quotient digits on, even a divisor [`short_divide`] takes
459/// divides faster nine digits at a time.
460const LIMB_SHORT_QUOTIENT: usize = 64;
461
462/// Below this many quotient-digit-by-divisor-digit steps, [`wide_divide`]
463/// is cheaper than converting to limbs and back.
464const LIMB_QUOTIENT: usize = 100;
465
466/// Above this many quotient digits [`limb_divide`] would build a numerator
467/// of that length up front, where [`wide_divide`] can still stop early on a
468/// division that comes out even: a quotient this long comes only from a
469/// DIGITS setting no oracle comparison reaches.
470const LIMB_QUOTIENT_MAX: usize = 1 << 24;
471
472/// [`long_divide`] as whole-number division over limbs. The digits the
473/// digit-at-a-time division produces after consuming `i` numerator digits
474/// are the quotient of those `i` digits (zero-padded past the numerator's
475/// end) by the divisor, so its whole quotient is one division at the digit
476/// where it would stop. That is the `want`th quotient digit, or the first
477/// digit past the numerator's own where the division comes out even.
478#[inline(never)]
479fn limb_divide(n: &[u8], d: &[u8], want: usize) -> (Digits, usize) {
480    use crate::limbs::{divide, from_digits, to_digits};
481    debug_assert!(
482        n.first().is_some_and(|lead| *lead != 0),
483        "a canonical numerator"
484    );
485    debug_assert!(
486        d.first().is_some_and(|lead| *lead != 0),
487        "a canonical divisor"
488    );
489    if want == 0 {
490        return (Digits::new(), 0);
491    }
492    let first = first_quotient_digit(n, d);
493    let mut stop = first.saturating_add(want - 1);
494    if stop > n.len() && d.len() <= SHORT_DIVISOR_DIGITS {
495        // For a divisor a word holds, the zero past the numerator at which
496        // the division first comes out even is found from the numerator's
497        // residue, without dividing.
498        let divisor = fold_digits(d);
499        let residue = from_digits(n, 0).iter().rev().fold(0u64, |acc, limb| {
500            u64::try_from(
501                (u128::from(acc) * 1_000_000_000 + u128::from(*limb)) % u128::from(divisor),
502            )
503            .expect("below the divisor")
504        });
505        if let Some(zeros) = terminating_zeros(residue, divisor) {
506            stop = stop.min(first.max(n.len() + zeros));
507        }
508    }
509    let numerator = if stop <= n.len() {
510        from_digits(&n[..stop], 0)
511    } else {
512        from_digits(n, stop - n.len())
513    };
514    let (quotient, rem) = divide(&numerator, &from_digits(d, 0));
515    let mut q = to_digits(&quotient);
516    debug_assert_eq!(q.len(), stop + 1 - first, "one digit per consumed digit");
517    // The digit-at-a-time division stops at the first digit past the
518    // numerator's own where it comes out even. Coming out even at an
519    // earlier digit than `stop` leaves the quotient with a trailing zero for
520    // each digit past that one, which is how a divisor too wide for the
521    // search above finds where it stops.
522    if rem.is_empty() && stop >= n.len() {
523        let earliest = first.max(n.len());
524        while stop > earliest && q.last() == Some(&0) {
525            q.pop();
526            stop -= 1;
527        }
528    }
529    (Digits::from_vec(q), stop)
530}

532/// [`long_divide`] for a divisor a `u64` holds, which is the schoolbook short
533/// division: one hardware divide per quotient digit, and no working storage
534/// at all.
535fn short_divide(n: &[u8], divisor: u64, divisor_len: usize, want: usize) -> (Digits, usize) {
536    // **A quotient of a working length that fits the stack is written there
537    // and built once**, where pushing digit by digit re-tests the storage
538    // arm and the capacity per digit. The loop is the one below.
539    const STACK_QUOTIENT: usize = 64;
540    if want <= STACK_QUOTIENT {
541        let mut quotient = [0u8; STACK_QUOTIENT];
542        let mut length = 0usize;
543        let mut remainder = 0u64;
544        let mut i = 0usize;
545        let exhausted = n.len().saturating_add(want).saturating_add(divisor_len);
546        while length < want {
547            let next = if i < n.len() { n[i] } else { 0 };
548            i += 1;
549            remainder = remainder * 10 + u64::from(next);
550            let digit = u8::try_from(remainder / divisor).expect("a remainder below 10 * divisor");
551            remainder %= divisor;
552            if length == 0 && digit == 0 {
553                if i > exhausted {
554                    break;
555                }
556                continue;
557            }
558            quotient[length] = digit;
559            length += 1;
560            if remainder == 0 && i >= n.len() {
561                break;
562            }
563        }
564        return (Digits::from_slice(&quotient[..length]), i);
565    }
566    let mut remainder = 0u64;
567    let mut q = Digits::new();
568    let mut i = 0usize;
569    // Saturating, for the reason the wide path's own copy of this bound
570    // gives: `want` is itself saturated by `working_length`.
571    let exhausted = n.len().saturating_add(want).saturating_add(divisor_len);
572
573    while q.len() < want {
574        let next = if i < n.len() { n[i] } else { 0 };
575        i += 1;
576        remainder = remainder * 10 + u64::from(next);
577        let digit = u8::try_from(remainder / divisor).expect("a remainder below 10 * divisor");
578        remainder %= divisor;
579        if q.is_empty() && digit == 0 {
580            // Not yet reached the first significant quotient digit.
581            if i > exhausted {
582                break;
583            }
584            continue;
585        }
586        q.push(digit);
587        if remainder == 0 && i >= n.len() {
588            break;
589        }
590    }
591    (q, i)
592}

594/// [`long_divide`] for a divisor too wide for a machine word: guess a
595/// multiple from the divisor's leading digits, subtract it out, and repeat.
596fn wide_divide(n: &[u8], d: &[u8], want: usize) -> (Digits, usize) {
597    // The live remainder is `rem[start..]`: leading zeros are skipped by
598    // advancing `start` instead of draining them out, which cost a memmove
599    // on every subtraction pass. The dead prefix stays zero, so slicing from
600    // `start` is always the whole value.
601    let mut rem: Vec<u8> = Vec::new();
602    let mut start = 0usize;
603    // The quotient becomes the result's digits, so it is built in the same
604    // representation they are; the working remainder above is scratch that
605    // never leaves this function and stays a plain vector.
606    let mut q = Digits::new();
607    let mut i = 0usize;
608
609    // Digit guesses divide by the divisor's first two digits plus one, so a
610    // guess is either correct or errs low, never high -- the estimate the
611    // interpreter's `Division` uses, shared with `divide_power`.
612    let mut div_char = i32::from(d[0]) * 10;
613    if d.len() > 1 {
614        div_char += i32::from(d[1]);
615    }
616    div_char += 1;
617
618    // Feed digits of the numerator, then zeros, taking one quotient digit
619    // per step once the quotient has started.
620    while q.len() < want {
621        rem.push(if i < n.len() { n[i] } else { 0 });
622        i += 1;
623        while rem.len() - start > 1 && rem[start] == 0 {
624            start += 1;
625        }
626        // The digit accumulates from under-guesses until the remainder drops
627        // below the divisor. The remainder enters each step below ten times
628        // the divisor, so the total never exceeds 9.
629        let mut count = 0u8;
630        loop {
631            let cur = &rem[start..];
632            let multiplier = if cur.len() == d.len() {
633                match cur.cmp(d) {
634                    // The remainder is smaller: this digit is complete.
635                    std::cmp::Ordering::Less => break,
636                    // Exactly equal: one last subtraction empties it.
637                    std::cmp::Ordering::Equal => {
638                        count += 1;
639                        rem.clear();
640                        rem.push(0);
641                        start = 0;
642                        break;
643                    }
644                    std::cmp::Ordering::Greater => i32::from(cur[0]),
645                }
646            } else if cur.len() > d.len() {
647                // The remainder is longer, so it has at least two digits.
648                i32::from(cur[0]) * 10 + i32::from(cur[1])
649            } else {
650                break;
651            };
652            // A zero guess gets wrapped to 1.
653            let m = (multiplier * 10 / div_char).max(1);
654            // At most `99 * 10 / 11`: the divisor leads with a nonzero digit.
655            count += u8::try_from(m).expect("a guess below 91");
656            subtract_multiple(&mut rem[start..], d, m);
657            while rem.len() - start > 1 && rem[start] == 0 {
658                start += 1;
659            }
660        }
661        if q.is_empty() && count == 0 {
662            // Not yet reached the first significant quotient digit.
663            // Saturating: `want` is itself saturated (`working_length`), so
664            // this bound must stay huge rather than wrap -- producer-side
665            // saturation needs matching consumer-side arithmetic, or the
666            // sum overflows in debug at exactly the values the saturation
667            // was added for.
668            if i > n.len().saturating_add(want).saturating_add(d.len()) {
669                break;
670            }
671            continue;
672        }
673        q.push(count);
674        // The interpreter stops as soon as the division comes out even
675        // rather than padding to the full width, so 1 / 1 is `1` and not
676        // `1.00000000`.
677        if rem[start..].iter().all(|x| *x == 0) && i >= n.len() {
678            break;
679        }
680    }
681    // The residue is not returned: its one caller discarded it, because a
682    // remainder good enough to report has to be recomputed at exact
683    // precision rather than read off the division (see `DivOp::Remainder`
684    // above). Building it cost a `split_off`, which allocates.
685    (q, i)
686}

688pub(crate) fn strip_leading(v: &mut Vec<u8>) {
689    let lead = v.iter().take_while(|d| **d == 0).count();
690    let lead = lead.min(v.len().saturating_sub(1));
691    v.drain(..lead);
692}
693
694/// `left -= m * divisor`, aligned at the low-order ends. The guess `m` is
695/// correct or low, never high, so the result cannot go negative. Shared by
696/// `long_divide` and `pow`'s `divide_power`.
697pub(crate) fn subtract_multiple(left: &mut [u8], divisor: &[u8], m: i32) {
698    let mut carry: i32 = 0;
699    for i in 0..left.len() {
700        let pos = left.len() - 1 - i;
701        let sub = divisor
702            .len()
703            .checked_sub(i + 1)
704            .map_or(0, |k| i32::from(divisor[k]) * m);
705        let mut v = carry + i32::from(left[pos]) - sub;
706        if v < 0 {
707            // A single digit product can leave a deficit as large as 81, so
708            // the borrow out can span two positions.
709            v += 100;
710            carry = v / 10 - 10;
711            v %= 10;
712        } else {
713            carry = 0;
714        }
715        left[pos] = u8::try_from(v).expect("a digit");
716    }
717    debug_assert_eq!(
718        carry, 0,
719        "an under-guess never drives the remainder negative"
720    );
721}
