13impl Number {
14    /// Extends the digit vector downward so both operands share an exponent.
15    #[inline(always)]
16    fn aligned_to(&self, exponent: i32) -> Digits {
17        let pad = usize::cast_from((self.exponent - exponent).max(0).unsigned_abs());
18        let mut digits = self.digits.clone();
19        digits.extend_zeros(pad);
20        digits
21    }

23    #[inline]
24    pub fn add(&self, other: &Number, digits: u64) -> Result<Number, ArithError> {
25        self.add_signed(other, false, digits)
26    }

28    /// Prefix `+`: `0 + self`. A value no longer than `digits` is its own
29    /// answer, range check aside, and is handed back without being copied.
30    #[inline]
31    pub fn plus(self, digits: u64) -> Result<Number, ArithError> {
32        if u64::cast_from(self.digits.len()) <= digits {
33            self.check_range()?;
34            return Ok(self);
35        }
36        Number::zero().add(&self, digits)
37    }
38
39    #[inline]
40    pub fn sub(&self, other: &Number, digits: u64) -> Result<Number, ArithError> {
41        self.add_signed(other, true, digits)
42    }

44    fn add_signed(
45        &self,
46        other: &Number,
47        negate_other: bool,
48        digits: u64,
49    ) -> Result<Number, ArithError> {
50        let left_negative = self.negative;
51        let right_negative = other.negative != negate_other;
52
53        // Operands longer than DIGITS are truncated to DIGITS + 1 working
54        // digits first. (The interpreter also raises LOSTDIGITS here.)
55        let max_length = crate::working_length(digits);
56        let left = self.truncated_to(max_length);
57        let right = other.truncated_to(max_length);
58
59        let min_exp = left.exponent.min(right.exponent);
60        // Both exponents are within the exponent range, so neither
61        // difference from their minimum leaves an `i32`.
62        let left_gap = (left.exponent - min_exp).unsigned_abs();
63        let right_gap = (right.exponent - min_exp).unsigned_abs();
64        let adjusted_left_exp = usize::cast_from(left_gap);
65        let adjusted_right_exp = usize::cast_from(right_gap);
66        let left_len = left.digits.len();
67        let right_len = right.digits.len();
68        // Saturated for the fast-path sums below: `digits` is a bare u64,
69        // and `len + digits_usize` must compare large, not wrap.
70        let digits_usize = usize::try_from(digits).unwrap_or(usize::MAX);
71
72        // Fast paths. Each returns one operand essentially untouched, and they
73        // are not an optimisation: going through the general path produces a
74        // leading zero that rounding then turns into a different number. `0 +
75        // 123456789` is 123456789, but computing it yields `0123456789`, which
76        // rounds to 123456790.
77        let fast = if left.is_zero() {
78            Some((&right, right_negative))
79        } else if right.is_zero() {
80            Some((&left, left_negative))
81        } else if adjusted_left_exp + left_len > right_len.saturating_add(digits_usize) {
82            // The right number is too far below the left to reach any digit
83            // the result will keep. (`saturating_add`: the saturated
84            // `digits_usize` must stay "huge", not wrap back around small.)
85            Some((&left, left_negative))
86        } else if adjusted_right_exp + right_len > left_len.saturating_add(digits_usize) {
87            Some((&right, right_negative))
88        } else {
89            None
90        };
91        if let Some((value, negative)) = fast {
92            let mut result = value.round_to(digits).into_owned();
93            result.negative = negative && !result.is_zero();
94            result.check_range()?;
95            return Ok(result);
96        }
97
98        if let Some(sum) = exact_small_sum(&left, &right, left_negative, right_negative, digits) {
99            sum.check_range()?;
100            return Ok(sum);
101        }
102        if let Some(sum) = aligned_wide_sum(&left, &right, left_negative, right_negative, digits) {
103            sum.check_range()?;
104            return Ok(sum);
105        }
106        // Alignment adjustment, ported from addSub. When the two operands
107        // together span more than the working precision, the less significant
108        // one is shortened from its low end so the pair fits. Skipping this
109        // is not merely imprecise -- it changes which digit the rounding
110        // decision sees, so `12.3400 - 9.999999995` at DIGITS 3 comes out as
111        // 2.3 rather than 2.4.
112        let mut left_digits = left.digits.clone();
113        let mut right_digits = right.digits.clone();
114        let mut left_exp = left.exponent;
115        let mut right_exp = right.exponent;
116        let adjusted_left_exp = i64::from(left_gap);
117        let adjusted_right_exp = i64::from(right_gap);
118        // Saturated: `max_length` came from the bare `digits` and may sit at
119        // usize::MAX, which an `as i64` would fold to -1. i64::MAX keeps
120        // both `adjusted_*_digits` differences negative, exactly as a
121        // precision that large should -- nothing ever needs shortening.
122        let max_len = i64::try_from(max_length).unwrap_or(i64::MAX);
123
124        let adjusted_left_digits = signed_len(left_digits.len()) + adjusted_left_exp - max_len;
125        let adjusted_right_digits = signed_len(right_digits.len()) + adjusted_right_exp - max_len;
126        if adjusted_left_digits > 0 || adjusted_right_digits > 0 {
127            // Neither operand is longer than `max_len`, so each count here is
128            // at most an exponent gap, which fits an `i32`.
129            let gap = |count: i64| i32::try_from(count).expect("at most an exponent gap");
130            let mut adjust = adjusted_left_digits.max(adjusted_right_digits);
131
132            // Exactly one of the adjusted exponents is non-zero: the more
133            // significant operand's. The other operand is the one shortened.
134            if adjusted_left_exp != 0 {
135                let taken = adjust.min(adjusted_left_exp);
136                drop_low_digits(&mut right_digits, taken);
137                right_exp = right_exp.saturating_add(gap(taken));
138                adjust -= taken;
139            } else if adjusted_right_exp != 0 {
140                let taken = adjust.min(adjusted_right_exp);
141                drop_low_digits(&mut left_digits, taken);
142                left_exp = left_exp.saturating_add(gap(taken));
143                adjust -= taken;
144            }
145
146            if adjust != 0 {
147                drop_low_digits(&mut left_digits, adjust);
148                left_exp = left_exp.saturating_add(gap(adjust));
149                drop_low_digits(&mut right_digits, adjust);
150                right_exp = right_exp.saturating_add(gap(adjust));
151            }
152        }
153
154        let left = Number {
155            negative: left_negative,
156            digits: left_digits,
157            exponent: left_exp,
158        };
159        let right = Number {
160            negative: right_negative,
161            digits: right_digits,
162            exponent: right_exp,
163        };
164        let min_exp = left.exponent.min(right.exponent);
165        let a = left.aligned_to(min_exp);
166        let b = right.aligned_to(min_exp);
167
168        let (raw_digits, negative) = if left_negative == right_negative {
169            (add_magnitudes(&a, &b), left_negative)
170        } else {
171            match compare_magnitudes(&a, &b) {
172                std::cmp::Ordering::Equal => return Ok(Number::zero()),
173                std::cmp::Ordering::Greater => (sub_magnitudes(&a, &b), left_negative),
174                std::cmp::Ordering::Less => (sub_magnitudes(&b, &a), right_negative),
175            }
176        };
177
178        // Deliberately not `assemble` here: that strips leading zeros, and a
179        // zero left by a borrow or an absent carry has to still be present
180        // for `round_to` to count it. Normalising happens inside `round_to`,
181        // after the rounding decision has been made.
182        let raw = Number {
183            negative,
184            digits: raw_digits,
185            exponent: min_exp,
186        };
187        let rounded = raw.into_round(digits);
188        let result = Number::assemble(rounded.negative, rounded.digits, rounded.exponent);
189        result.check_range()?;
190        Ok(result)
191    }
192
193    /// Shortens an over-long operand to the working precision, as the addition
194    /// and its operand check both do. Truncation, not rounding.
195    pub(crate) fn truncated_to(&self, max_length: usize) -> Cow<'_, Number> {
196        if self.digits.len() <= max_length {
197            return Cow::Borrowed(self);
198        }
199        let dropped = self.digits.len() - max_length;
200        Cow::Owned(Number {
201            negative: self.negative,
202            digits: Digits::from_slice(&self.digits[..max_length]),
203            exponent: self.exponent + signed_count(dropped),
204        })
205    }
206}

219/// The exact sum, when both operands align into an `i64` mantissa and the
220/// result needs no rounding at `digits`.
221fn exact_small_sum(
222    left: &Number,
223    right: &Number,
224    left_negative: bool,
225    right_negative: bool,
226    digits: u64,
227) -> Option<Number> {
228    #[cfg(test)]
229    if SHORTCUT_OFF.with(std::cell::Cell::get) {
230        return None;
231    }
232    // **The lengths alone decide most calls, and they decide them first.**
233    // The width below is never less than the longer operand plus one, so an
234    // operand already that long settles it without the exponents being read
235    // at all -- which is the answer for every operand a program has widened
236    // to its own precision, and for every one `div` hands back at the
237    // inflated precision it works in.
238    // A `digits` past `i64` declines, as a negative one would.
239    let longest = signed_len(left.digits.len().max(right.digits.len()));
240    if longest >= 18 || i64::try_from(digits).map_or(true, |digits| longest >= digits) {
241        return None;
242    }
243    let min_exp = i64::from(left.exponent.min(right.exponent));
244    let left_shift = i64::from(left.exponent) - min_exp;
245    let right_shift = i64::from(right.exponent) - min_exp;
246    let width = (signed_len(left.digits.len()) + left_shift)
247        .max(signed_len(right.digits.len()) + right_shift)
248        + 1;
249    // Eighteen digits is the widest an `i64` holds whatever they are, and it
250    // bounds the scaling loop below as well as the sum.
251    if width > 18 || i64::try_from(digits).map_or(true, |digits| width > digits) {
252        return None;
253    }
254    let mantissa = |number: &Number, shift: i64, negative: bool| {
255        let mut value = number
256            .digits
257            .iter()
258            .fold(0i64, |acc, digit| acc * 10 + i64::from(*digit));
259        for _ in 0..shift {
260            value *= 10;
261        }
262        if negative { -value } else { value }
263    };
264    let sum =
265        mantissa(left, left_shift, left_negative) + mantissa(right, right_shift, right_negative);
266    if sum == 0 {
267        return Some(Number::zero());
268    }
269    let mut result = Number::from_i64(sum);
270    result.exponent = left.exponent.min(right.exponent);
271    Some(result)
272}

274/// The widest aligned operand [`aligned_wide_sum`] takes. The sum of two is
275/// below 2·10^37, so its digits above the nineteenth (the sum divided by
276/// 10^19) stay below 2·10^18 and fit the `u64` they are written out of; at
277/// thirty-eight digits the sum reaches 2·10^38 and that half does not.
278const WIDE_DIGITS: usize = 37;
279
280/// The general addition below, on `u128` magnitudes, for operands that
281/// after its alignment adjustment fit [`WIDE_DIGITS`] aligned to the lower
282/// exponent. Within that bound the digit
283/// vectors the general path builds are these integers written out, with
284/// the raw length that path's rounding counts: the wider aligned operand,
285/// plus one for a carry out of an addition, and with the leading zeros a
286/// subtraction leaves.
287fn aligned_wide_sum(
288    left: &Number,
289    right: &Number,
290    left_negative: bool,
291    right_negative: bool,
292    digits: u64,
293) -> Option<Number> {
294    #[cfg(test)]
295    if SHORTCUT_OFF.with(std::cell::Cell::get) {
296        return None;
297    }
298    // The general path's alignment adjustment, on slices: where the pair
299    // spans more than the working length, the less significant operand's
300    // low digits go first and then both operands' -- truncated, never
301    // emptied, with the count folded into the exponent whatever was left.
302    let max_length = i64::try_from(crate::working_length(digits)).unwrap_or(i64::MAX);
303    let (mut left_digits, mut left_exp) = (left.digits.as_slice(), left.exponent);
304    let (mut right_digits, mut right_exp) = (right.digits.as_slice(), right.exponent);
305    let first_min = left_exp.min(right_exp);
306    let left_shift = i64::from(left_exp) - i64::from(first_min);
307    let right_shift = i64::from(right_exp) - i64::from(first_min);
308    let left_over = signed_len(left_digits.len()) + left_shift - max_length;
309    let right_over = signed_len(right_digits.len()) + right_shift - max_length;
310    if left_over > 0 || right_over > 0 {
311        let shorten = |slice: &[u8], count: i64| -> usize {
312            slice
313                .len()
314                .saturating_sub(usize::try_from(count.max(0)).unwrap_or(usize::MAX))
315                .max(1)
316        };
317        // Neither operand is longer than the working length, so each count
318        // here is at most an exponent gap, which fits an `i32`.
319        let gap = |count: i64| i32::try_from(count).expect("at most an exponent gap");
320        let mut adjust = left_over.max(right_over);
321        if left_shift != 0 {
322            let taken = adjust.min(left_shift);
323            right_digits = &right_digits[..shorten(right_digits, taken)];
324            right_exp = right_exp.saturating_add(gap(taken));
325            adjust -= taken;
326        } else if right_shift != 0 {
327            let taken = adjust.min(right_shift);
328            left_digits = &left_digits[..shorten(left_digits, taken)];
329            left_exp = left_exp.saturating_add(gap(taken));
330            adjust -= taken;
331        }
332        if adjust != 0 {
333            left_digits = &left_digits[..shorten(left_digits, adjust)];
334            left_exp = left_exp.saturating_add(gap(adjust));
335            right_digits = &right_digits[..shorten(right_digits, adjust)];
336            right_exp = right_exp.saturating_add(gap(adjust));
337        }
338    }
339    let min_exp = left_exp.min(right_exp);
340    let left_width =
341        left_digits.len() + usize::try_from(i64::from(left_exp) - i64::from(min_exp)).ok()?;
342    let right_width =
343        right_digits.len() + usize::try_from(i64::from(right_exp) - i64::from(min_exp)).ok()?;
344    let widest = left_width.max(right_width);
345    if widest > WIDE_DIGITS || signed_len(widest) > max_length {
346        return None;
347    }
348    // Folded in a `u64` while nineteen digits fit one, which is every
349    // operand of a DIGITS 18 or narrower addition.
350    let magnitude = |digits: &[u8], width: usize| {
351        let split = digits.len().min(19);
352        let head = digits[..split]
353            .iter()
354            .fold(0u64, |acc, digit| acc * 10 + u64::from(*digit));
355        let mut value = u128::from(head);
356        for digit in &digits[split..] {
357            value = value * 10 + u128::from(*digit);
358        }
359        value * 10u128.pow(u32::try_from(width - digits.len()).expect("at most WIDE_DIGITS"))
360    };
361    let a = magnitude(left_digits, left_width);
362    let b = magnitude(right_digits, right_width);
363    let (raw, negative, raw_length) = if left_negative == right_negative {
364        let sum = a + b;
365        let carried = sum >= 10u128.pow(u32::try_from(widest).expect("at most WIDE_DIGITS"));
366        (sum, left_negative, widest + usize::from(carried))
367    } else {
368        match a.cmp(&b) {
369            std::cmp::Ordering::Equal => return Some(Number::zero()),
370            std::cmp::Ordering::Greater => (a - b, left_negative, widest),
371            std::cmp::Ordering::Less => (b - a, right_negative, widest),
372        }
373    };
374    // The raw digits, leading zeros included, written out through one
375    // division into two `u64` halves: a `u128` divided digit by digit is a
376    // library call per digit.
377    let mut buffer = [0u8; WIDE_DIGITS + 1];
378    let raw_digits = &mut buffer[..raw_length];
379    let split = 10u128.pow(19);
380    let half = |value: u128| u64::try_from(value).expect("below 2 * 10^18 or 10^19");
381    let (mut high, mut low) = if raw < split {
382        (0, half(raw))
383    } else {
384        (half(raw / split), half(raw % split))
385    };
386    for (place, slot) in raw_digits.iter_mut().rev().enumerate() {
387        if place < 19 {
388            *slot = last_digit(low);
389            low /= 10;
390        } else {
391            *slot = last_digit(high);
392            high /= 10;
393        }
394    }
395    // `round_change` on those digits: keep the first DIGITS of them, leading
396    // zeros counted, half up on the first dropped digit, and an all-nines
397    // carry keeps the length and moves the exponent.
398    let keep = usize::try_from(digits).unwrap_or(usize::MAX);
399    let mut exponent = min_exp;
400    let kept = if keep != 0 && raw_length > keep {
401        exponent += i32::try_from(raw_length - keep).expect("at most WIDE_DIGITS + 1");
402        let round_up = raw_digits[keep] >= 5;
403        let kept = &mut raw_digits[..keep];
404        if round_up {
405            match kept.iter().rposition(|digit| *digit != 9) {
406                Some(at) => {
407                    kept[at] += 1;
408                    kept[at + 1..].fill(0);
409                }
410                None => {
411                    kept.fill(0);
412                    kept[0] = 1;
413                    exponent += 1;
414                }
415            }
416        }
417        kept
418    } else {
419        raw_digits
420    };
421    let lead = kept.iter().take_while(|digit| **digit == 0).count();
422    if lead == kept.len() {
423        return Some(Number::zero());
424    }
425    Some(Number {
426        negative,
427        digits: Digits::from_slice(&kept[lead..]),
428        exponent,
429    })
430}

432/// Compares two digit vectors by value, ignoring leading zeros.
433fn compare_magnitudes(a: &[u8], b: &[u8]) -> std::cmp::Ordering {
434    let a_start = a.iter().take_while(|d| **d == 0).count();
435    let b_start = b.iter().take_while(|d| **d == 0).count();
436    let (a, b) = (&a[a_start..], &b[b_start..]);
437    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
438}

440/// Sum of two aligned digit vectors.
441#[inline(always)]
442fn add_magnitudes(a: &[u8], b: &[u8]) -> Digits {
443    let n = a.len().max(b.len());
444    let mut out = Digits::zeros(n);
445    let mut carry = 0u8;
446    for i in 0..n {
447        let x = a.len().checked_sub(i + 1).map_or(0, |k| a[k]);
448        let y = b.len().checked_sub(i + 1).map_or(0, |k| b[k]);
449        let sum = x + y + carry;
450        out[n - 1 - i] = sum % 10;
451        carry = sum / 10;
452    }
453    if carry > 0 {
454        out.insert_front(carry);
455    }
456    out
457}

459/// `a - b`, where `a >= b` by magnitude.
460#[inline(always)]
461fn sub_magnitudes(a: &[u8], b: &[u8]) -> Digits {
462    let n = a.len();
463    let mut out = Digits::zeros(n);
464    let mut borrow = 0i8;
465    for i in 0..n {
466        let x = a[n - 1 - i].cast_signed();
467        let y = b.len().checked_sub(i + 1).map_or(0, |k| b[k].cast_signed());
468        let mut d = x - y - borrow;
469        if d < 0 {
470            d += 10;
471            borrow = 1;
472        } else {
473            borrow = 0;
474        }
475        out[n - 1 - i] = d.cast_unsigned();
476    }
477    out
478}

480/// Removes `count` digits from the low end, by walking the end pointer
481/// backwards. Never empties the vector.
482fn drop_low_digits(digits: &mut Digits, count: i64) {
483    // Saturated: a count past `usize` (only where `usize` is 32 bits) is
484    // longer than any vector, and leaves the one digit kept as it would.
485    let count = usize::try_from(count.max(0)).unwrap_or(usize::MAX);
486    let keep = digits.len().saturating_sub(count).max(1);
487    digits.truncate(keep);
488}

499        right: &Number,

509        SHORTCUT_OFF.set(false);

511    }

562                        assert_eq!(

564                            long,
565                            "{} {} {} at DIGITS {digits}",
566                            left.format(40),
567                            if subtract { "-" } else { "+" },
568                            right.format(40)
569                        );
570                        let working = crate::working_length(digits);
571                        let l = left.truncated_to(working);
572                        let r = right.truncated_to(working);
573                        if !l.is_zero()
574                            && !r.is_zero()
575                            && exact_small_sum(&l, &r, l.negative, r.negative != subtract, digits)
576                                .is_some()
577                        {
578                            taken += 1;
579                        }
580                    }
581                }
582            }
583        }
584        // A shortcut that declined everything would satisfy the loop above
585        // without saying anything.
586        assert!(taken > 0, "the shortcut never fired");
587    }

589    /// Pseudo-random operands of up to forty digits at exponents near each
590    /// other, so that most pairs take [`super::aligned_wide_sum`] and its
591    /// rounding: both shortcuts on against both off.
592    #[test]
593    fn the_wide_sum_answers_what_the_digit_addition_answers() {
594        let mut state = 0x2545_f491_4f6c_dd1du64;
595        let mut next = move |bound: u64| {
596            state ^= state << 13;
597            state ^= state >> 7;
598            state ^= state << 17;
599            state % bound
600        };
601        let number = |next: &mut dyn FnMut(u64) -> u64| {
602            let length = 1 + usize::try_from(next(40)).expect("below 40");
603            let mut text = String::new();
604            if next(2) == 0 {
605                text.push('-');
606            }
607            for i in 0..length {
608                // Runs of nines and zeros, which is where carries and
609                // cancellations live.
610                let digit = match next(4) {
611                    0 => 9,
612                    1 => 0,
613                    _ => u8::try_from(next(10)).expect("a digit"),
614                };
615                text.push(char::from(b'0' + if i == 0 { digit.max(1) } else { digit }));
616            }
617            // Mostly near each other, sometimes far apart, which is what
618            // makes the alignment adjustment shorten an operand.
619            let spread = if next(4) == 0 { 90 } else { 30 };
620            text.push_str(&format!(
621                "E{}",
622                next(spread).cast_signed() - (spread / 2).cast_signed()
623            ));
624            Number::parse(&text).expect("a generated spelling parses")
625        };
626        let mut taken = 0;
627        for _ in 0..100_000 {
628            let left = number(&mut next);
629            let right = number(&mut next);
630            let digits = 1 + next(40);
631            let subtract = next(2) == 0;
632            let shortcut = if subtract {
633                left.sub(&right, digits)
634            } else {
635                left.add(&right, digits)
636            };
637            assert_eq!(
638                shortcut,
639                the_long_way(&left, &right, digits, subtract),
640                "{} {} {} at DIGITS {digits}",
641                left.format(40),
642                if subtract { "-" } else { "+" },
643                right.format(40)
644            );
645            let working = crate::working_length(digits);
646            let (l, r) = (left.truncated_to(working), right.truncated_to(working));
647            if super::aligned_wide_sum(&l, &r, l.negative, r.negative != subtract, digits).is_some()
648            {
649                taken += 1;
650            }
651        }
652        assert!(taken > 20_000, "the wide sum fired {taken} times");
653    }

655    /// `Scaled::add` answers what the digit addition answers wherever it
656    /// answers at all: operands of at most eighteen digits at DIGITS around
657    /// theirs, where its alignment adjustment and rounding both run.
658    #[test]
659    fn the_scaled_sum_answers_what_the_digit_addition_answers() {
660        let mut state = 0x9e37_79b9_7f4a_7c15u64;
661        let mut next = move |bound: u64| {
662            state ^= state << 13;
663            state ^= state >> 7;
664            state ^= state << 17;
665            state % bound
666        };
667        let number = |next: &mut dyn FnMut(u64) -> u64| {
668            let length = 1 + usize::try_from(next(18)).expect("below 18");
669            let mut text = String::new();
670            if next(2) == 0 {
671                text.push('-');
672            }
673            for i in 0..length {
674                let digit = match next(4) {
675                    0 => 9,
676                    1 => 0,
677                    _ => u8::try_from(next(10)).expect("a digit"),
678                };
679                text.push(char::from(b'0' + if i == 0 { digit.max(1) } else { digit }));
680            }
681            let spread = if next(4) == 0 { 60 } else { 20 };
682            text.push_str(&format!(
683                "E{}",
684                next(spread).cast_signed() - (spread / 2).cast_signed()
685            ));
686            if next(16) == 0 {
687                text = String::from("0");
688            }
689            Number::parse(&text).expect("a generated spelling parses")
690        };
691        let (mut taken, mut rounded) = (0, 0);
692        for _ in 0..200_000 {
693            let left = number(&mut next);
694            let right = number(&mut next);
695            let digits = 1 + next(20);
696            let subtract = next(2) == 0;
697            let (Some(l), Some(r)) = (left.scaled(), right.scaled()) else {
698                continue;
699            };
700            let Some(sum) = l.add(if subtract { r.negated() } else { r }, digits) else {
701                continue;
702            };
703            let expected = the_long_way(&left, &right, digits, subtract)
704                .expect("a sum the scaled path answers is in range");
705            assert_eq!(
706                sum.to_number(),
707                expected,
708                "{} {} {} at DIGITS {digits}",
709                left.format(40),
710                if subtract { "-" } else { "+" },
711                right.format(40)
712            );
713            taken += 1;
714            let width = |n: &Number| crate::signed_len(n.digits.len()) + i64::from(n.exponent);
715            if (width(&left) - i64::from(left.exponent.min(right.exponent))).unsigned_abs() > digits
716            {
717                rounded += 1;
718            }
719        }
720        assert!(taken > 50_000, "the scaled sum answered {taken} times");
721        assert!(rounded > 10_000, "the scaled sum rounded {rounded} times");
722    }

745    /// The shortcut declines every operand it must, so that the assertion
746    /// above is a statement about a real boundary rather than about an empty
747    /// set on one side of it.
748    #[test]
749    fn the_small_sum_shortcut_declines_what_it_cannot_answer_exactly() {
750        let one = Number::parse("1").expect("one");
751        let nine = Number::parse("9").expect("nine");
752        let wide = Number::parse("123456789012345678").expect("eighteen digits");
753        let far = Number::parse("1E+30").expect("a distant exponent");
754
755        // A sum that may carry past the precision has to be rounded, which
756        // the shortcut does not do. It is decided on the wider operand plus
757        // one rather than on the sum, so `9 + 9` is declined at DIGITS 1
758        // although `18` would round to `2E+1` either way.
759        assert!(exact_small_sum(&nine, &nine, false, false, 1).is_none());
760        assert!(exact_small_sum(&nine, &nine, false, false, 2).is_some());
761
762        // Operands too far apart to share an `i64` mantissa.
763        assert!(exact_small_sum(&far, &one, false, false, 40).is_none());
764
765        // A mantissa past what an `i64` holds.
766        assert!(exact_small_sum(&wide, &wide, false, false, 40).is_none());
767
768        // A cancelling pair answers the canonical zero rather than one
769        // carrying the operands' exponent.
770        let tenth = Number::parse("1.1").expect("a tenth");
771        assert_eq!(
772            exact_small_sum(&tenth, &tenth, false, true, 9).expect("the pair cancels"),
773            Number::zero()
774        );
775    }
