18    /// Interprets this number as a whole number expressible within `digits`
19    /// significant digits, which is what `**` requires of its exponent.
20    fn as_whole(&self, digits: u64) -> Option<i64> {
21        let rounded = self.round_to(digits);
22        let self_ = &rounded;
23        if self_.is_zero() {
24            return Some(0);
25        }
26        // Compared in i64, with `digits` saturated into it: a narrowing
27        // (`as i32` originally, `as i64` for a bare u64 now) wraps negative
28        // and inverts this comparison rather than panicking -- a silently
29        // wrong answer, in release and debug alike. Saturation is exact:
30        // the left side is a digit count plus an exponent, nowhere near
31        // i64::MAX.
32        if crate::signed_len(self_.digits.len()) + i64::from(self_.exponent)
33            > i64::try_from(digits).unwrap_or(i64::MAX)
34        {
35            return None;
36        }
37        if self_.exponent < 0 {
38            // Any digit past the decimal point makes it non-whole.
39            let frac = usize::cast_from(self_.exponent.unsigned_abs());
40            if frac >= self_.digits.len()
41                || self_.digits[self_.digits.len() - frac..]
42                    .iter()
43                    .any(|d| *d != 0)
44            {
45                return None;
46            }
47        }
48        if self_.exponent >= 0 {
49            self_.plain_integer(u64::cast_from(crate::ARGUMENT_DIGITS))
50        } else {
51            self_.whole_value(crate::ARGUMENT_DIGITS)
52        }
53    }

55    pub fn pow(&self, exponent: &Number, digits: u64) -> Result<Number, ArithError> {
56        // `additional()` renders 26.008's substitution from `exponent` at
57        // its own full precision, which is the closest this crate can get
58        // to the interpreter's "as originally written" text -- see
59        // `ArithError::message`'s doc comment.
60        let power = exponent
61            .as_whole(digits)
62            .ok_or_else(|| ArithError::PowerExponentNotWhole {
63                exponent: exponent.clone(),
64            })?;
65        let negative_power = power < 0;
66        let power = power.unsigned_abs();
67
68        let left = self.truncated_to(crate::working_length(digits));
69
70        if left.is_zero() {
71            // Zero to a negative power is an underflow, not infinity. Error
72            // 42.903, no substitution -- confirmed with `0 ** -1`.
73            if negative_power {
74                return Err(ArithError::ZeroToNegativePower);
75            }
76            // Rexx defines 0**0 as 1, though mathematically it is undefined.
77            return Ok(if power == 0 {
78                Number::one()
79            } else {
80                Number::zero()
81            });
82        }
83
84        // The magnitude of the result is knowable up front, so a hopeless
85        // computation is refused before it is attempted. Error 42.001,
86        // substituting the base, the literal "**", and the exponent --
87        // confirmed with `100 ** 999999999` and, to catch the base/exponent
88        // getting rounded to this crate's default precision instead of
89        // their own, `123456789012345678 ** 999999999` at DIGITS 15 (which
90        // still reports the base at its full 18-digit precision).
91        let magnitude = u64::from(left.adjusted_exponent().unsigned_abs()).saturating_mul(power);
92        if magnitude > u64::from(MAX_EXPONENT.unsigned_abs()) {
93            return Err(ArithError::PowerOverflow {
94                base: self.clone(),
95                exponent: exponent.clone(),
96            });
97        }
98
99        if power == 0 {
100            return Ok(Number::one());
101        }
102
103        // Working precision is raised by the number of decimal digits in the
104        // exponent, plus one, so the repeated squaring does not accumulate
105        // visible error. Saturating: `digits` is a bare u64, and precision
106        // must stay huge rather than wrap.
107        let extra = u64::cast_from(power.to_string().len());
108        let work = digits.saturating_add(extra + 1);
109
110        // Square and multiply, high bit first, exactly as the interpreter
111        // sequences it: the accumulator starts as the base itself, which
112        // consumes the power's leading 1-bit, and each remaining bit squares
113        // the accumulator and multiplies the base back in when set. The
114        // order matters: it changes where the intermediate roundings fall,
115        // and the reciprocal below exposes the accumulator's last working
116        // digits, not just the rounded result.
117        let mut acc = Number::clone(&left);
118        let top = 63 - power.leading_zeros();
119        for i in (0..top).rev() {
120            acc = acc.mul_intermediate(&acc, work)?;
121            if (power >> i) & 1 == 1 {
122                acc = acc.mul_intermediate(&left, work)?;
123            }
124        }
125
126        if negative_power {
127            // The reciprocal has its own division, not the general one: it
128            // neither rounds nor range-checks its quotient. Both are
129            // observable -- the only rounding the reciprocal ever sees is the
130            // final one below, where going through `div` would round twice
131            // and flip a knife-edge last digit; and a reciprocal near the
132            // exponent limit is out of range at working precision until that
133            // same rounding brings it back.
134            acc = divide_power(&acc, work);
135        }
136
137        let mut result = acc.into_round(digits);
138        result.check_range()?;
139        // Trailing zeros come off, by the same rule division uses.
140        while result.digits.len() > 1 && *result.digits.last().unwrap() == 0 {
141            result.digits.pop();
142            result.exponent += 1;
143        }
144        Ok(Number::assemble(
145            result.negative,
146            result.digits,
147            result.exponent,
148        ))
149    }
150}

152/// A digit of the reciprocal: the remainder enters each step below ten times
153/// the divisor, so its guesses add up to at most nine.
154fn quotient_digit(digit: i32) -> u8 {
155    u8::try_from(digit).expect("a quotient digit")
156}

158/// The reciprocal `1 / accum` for a negative power.
159fn divide_power(accum: &Number, digits: u64) -> Number {
160    debug_assert!(!accum.is_zero(), "pow never inverts a zero accumulator");
161    let divisor = &accum.digits;
162    let n = divisor.len();
163
164    // The dividend is a 1 padded with zeros to the divisor's length.
165    let mut left: Vec<u8> = vec![0; n];
166    left[0] = 1;
167
168    // The expected exponent of the result's last digit; every extension of the
169    // dividend below moves it down one. Carried wide, as a 64-bit whole number.
170    let mut calc_exp = -i64::from(accum.exponent) - crate::signed_len(n) + 1;
171
172    // Digit guesses divide by the divisor's first two digits plus one, so a
173    // guess is either correct or errs low, never high.
174    let mut div_char = i32::from(divisor[0]) * 10;
175    if n > 1 {
176        div_char += i32::from(divisor[1]);
177    }
178    div_char += 1;
179
180    // The reciprocal's digits become the returned number's, so they are built
181    // in the same representation; `left` above is the working dividend and
182    // never leaves this function.
183    let mut result = Digits::new();
184    let mut this_digit: i32 = 0;
185
186    // The outer loop yields one quotient digit per pass; the inner loop
187    // builds that digit by accumulating guesses until the remainder drops
188    // below the divisor.
189    'outer: loop {
190        loop {
191            let multiplier;
192            if left.len() == n {
193                // Equal lengths: a direct comparison tells us where we are.
194                match left[..].cmp(&divisor[..]) {
195                    // The remainder is smaller: this digit is complete.
196                    std::cmp::Ordering::Less => break,
197                    // Exactly equal: the current digit is one too small.
198                    // Adjust it and the entire division is done.
199                    std::cmp::Ordering::Equal => {
200                        result.push(quotient_digit(this_digit + 1));
201                        break 'outer;
202                    }
203                    std::cmp::Ordering::Greater => multiplier = i32::from(left[0]),
204                }
205            } else if left.len() > n {
206                // The remainder is longer, so it has at least two digits.
207                multiplier = i32::from(left[0]) * 10 + i32::from(left[1]);
208            } else {
209                break;
210            }
211
212            // A zero guess gets wrapped to 1.
213            let m = (multiplier * 10 / div_char).max(1);
214            this_digit += m;
215            subtract_multiple(&mut left, divisor, m);
216            strip_leading(&mut left);
217        }
218
219        // A non-zero digit always joins the result; zeros only follow a
220        // previous non-zero.
221        if !result.is_empty() || this_digit != 0 {
222            result.push(quotient_digit(this_digit));
223            this_digit = 0;
224            // Done once the remainder is zero or the result has grown to
225            // digits + 1, leaving one digit for the caller's rounding.
226            // (Saturated rather than narrowed: wrapping a bare huge
227            // `digits` around to a tiny cap would silently truncate the
228            // quotient.)
229            if left[0] == 0 || result.len() > usize::try_from(digits).unwrap_or(usize::MAX) {
230                break;
231            }
232        }
233        // Reduced to exactly zero before the first significant digit.
234        if left.len() == 1 && left[0] == 0 {
235            break;
236        }
237        calc_exp -= 1;
238        left.push(0);
239    }
240
241    // An exponent beyond i32 cannot be in range anyway; saturate and let the
242    // caller's range check reject it.
243    let exponent = i32::try_from(calc_exp.clamp(i64::from(i32::MIN), i64::from(i32::MAX)))
244        .expect("clamped to the i32 range");
245    Number {
246        negative: accum.negative,
247        digits: result,
248        exponent,
249    }
