1//! Whole-number kernels over base-10^9 limbs, for operands long enough that
2//! working one decimal digit at a time dominates.
3//!
4//! A limb vector is least significant first, each limb below [`BASE`]. Only
5//! exact integer results leave this module: the decimal digit vectors go in
6//! and come out unchanged in meaning, so rounding, exponents and the stopping
7//! rules stay with the digit-level callers.
8
9/// Nine decimal digits per limb: the product of two limbs plus two more
10/// limbs stays inside a `u64`.
11const BASE: u64 = interrexx_cast::u64_from_u32(LIMB_BASE);
12/// [`BASE`] at a limb's own width.
13const LIMB_BASE: u32 = 1_000_000_000;
14/// [`BASE`] signed, for the borrow arithmetic of the division.
15const SIGNED_BASE: i64 = interrexx_cast::i64_from_u32(LIMB_BASE);
16const LIMB_DIGITS: usize = 9;
17
18/// `value`, which is below [`BASE`], as a limb.
19fn to_limb(value: u64) -> u32 {
20    u32::try_from(value).expect("a limb is below BASE")
21}
22
23/// `value`, which is in `0..BASE`, as a limb.
24fn signed_limb(value: i64) -> u32 {
25    u32::try_from(value).expect("a limb is in 0..BASE")
26}
27
28/// The last decimal digit of `value`.
29fn last_digit(value: u32) -> u8 {
30    u8::try_from(value % 10).expect("a remainder by 10 is a digit")
31}
32
33/// Below this many limbs in the shorter operand, Karatsuba's extra additions
34/// cost more than the multiplications it saves.
35const KARATSUBA_LIMBS: usize = 48;
36
37/// The limbs of the whole number spelt by `digits` (most significant first)
38/// followed by `zeros` zero digits.
39pub(crate) fn from_digits(digits: &[u8], zeros: usize) -> Vec<u32> {
40    let total = digits.len() + zeros;
41    let mut limbs = Vec::with_capacity(total.div_ceil(LIMB_DIGITS));
42    // Whole zero limbs first, then the remainder of the padding shifts the
43    // digits within their limbs.
44    limbs.resize(zeros / LIMB_DIGITS, 0);
45    let partial = zeros % LIMB_DIGITS;
46    let mut limb: u32 = 0;
47    let mut scale: u32 = 10u32.pow(u32::try_from(partial).expect("below LIMB_DIGITS"));
48    let mut filled = partial;
49    for &digit in digits.iter().rev() {
50        limb += u32::from(digit) * scale;
51        filled += 1;
52        if filled == LIMB_DIGITS {
53            limbs.push(limb);
54            limb = 0;
55            scale = 1;
56            filled = 0;
57        } else {
58            scale *= 10;
59        }
60    }
61    if filled > 0 {
62        limbs.push(limb);
63    }
64    trim(&mut limbs);
65    limbs
66}

68/// The decimal digits of `limbs`, most significant first, with no leading
69/// zeros (a zero is the single digit `0`).
70pub(crate) fn to_digits(limbs: &[u32]) -> Vec<u8> {
71    let mut top = limbs.len();
72    while top > 1 && limbs[top - 1] == 0 {
73        top -= 1;
74    }
75    if top == 0 {
76        return vec![0];
77    }
78    let mut out = Vec::with_capacity(top * LIMB_DIGITS);
79    let mut buf = [0u8; LIMB_DIGITS];
80    let mut lead = limbs[top - 1];
81    let mut len = 0;
82    loop {
83        buf[len] = last_digit(lead);
84        lead /= 10;
85        len += 1;
86        if lead == 0 {
87            break;
88        }
89    }
90    out.extend(buf[..len].iter().rev());
91    for &limb in limbs[..top - 1].iter().rev() {
92        let mut v = limb;
93        for slot in buf.iter_mut().rev() {
94            *slot = last_digit(v);
95            v /= 10;
96        }
97        out.extend_from_slice(&buf);
98    }
99    out
100}

102fn trim(limbs: &mut Vec<u32>) {
103    while limbs.last() == Some(&0) {
104        limbs.pop();
105    }
106}

108/// `a * b`, exact, with `a.len() + b.len()` limbs (possibly a leading zero).
109pub(crate) fn mul(a: &[u32], b: &[u32]) -> Vec<u32> {
110    let mut out = vec![0u32; a.len() + b.len()];
111    mul_into(&mut out, a, b);
112    out
113}
114
115/// `out = a * b`, where `out` is zeroed and `a.len() + b.len()` limbs long.
116fn mul_into(out: &mut [u32], a: &[u32], b: &[u32]) {
117    let (short, long) = if a.len() <= b.len() { (a, b) } else { (b, a) };
118    if short.len() < KARATSUBA_LIMBS {
119        schoolbook(out, short, long);
120    } else if long.len() >= 2 * short.len() {
121        // Lopsided: Karatsuba on short-sized slices of the longer operand.
122        let mut scratch = vec![0u32; 2 * short.len()];
123        for (index, chunk) in long.chunks(short.len()).enumerate() {
124            let piece = &mut scratch[..short.len() + chunk.len()];
125            piece.fill(0);
126            mul_into(piece, short, chunk);
127            add_into(&mut out[index * short.len()..], piece);
128        }
129    } else {
130        karatsuba(out, a, b);
131    }
132}

134fn schoolbook(out: &mut [u32], a: &[u32], b: &[u32]) {
135    for (i, &x) in a.iter().enumerate() {
136        if x == 0 {
137            continue;
138        }
139        let x = u64::from(x);
140        let mut carry = 0u64;
141        for (j, &y) in b.iter().enumerate() {
142            let t = u64::from(out[i + j]) + x * u64::from(y) + carry;
143            out[i + j] = to_limb(t % BASE);
144            carry = t / BASE;
145        }
146        let mut k = i + b.len();
147        while carry != 0 {
148            let t = u64::from(out[k]) + carry;
149            out[k] = to_limb(t % BASE);
150            carry = t / BASE;
151            k += 1;
152        }
153    }
154}

156/// Karatsuba on two operands of comparable length, split at half the
157/// shorter one.
158fn karatsuba(out: &mut [u32], a: &[u32], b: &[u32]) {
159    let half = a.len().min(b.len()) / 2;
160    let (a0, a1) = a.split_at(half);
161    let (b0, b1) = b.split_at(half);
162
163    // z0 = a0*b0 and z2 = a1*b1 land in their own places of `out`.
164    let (low, high) = out.split_at_mut(2 * half);
165    mul_into(low, a0, b0);
166    mul_into(&mut high[..a1.len() + b1.len()], a1, b1);
167
168    // z1 = (a0+a1)(b0+b1) - z0 - z2, added in at `half`.
169    let sa = sum(a0, a1);
170    let sb = sum(b0, b1);
171    let mut z1 = vec![0u32; sa.len() + sb.len()];
172    mul_into(&mut z1, &sa, &sb);
173    sub_from(&mut z1, &out[..2 * half]);
174    sub_from(&mut z1, &out[2 * half..2 * half + a1.len() + b1.len()]);
175    add_into(&mut out[half..], &z1);
176}

178fn sum(a: &[u32], b: &[u32]) -> Vec<u32> {
179    let (short, long) = if a.len() <= b.len() { (a, b) } else { (b, a) };
180    let mut out = Vec::with_capacity(long.len() + 1);
181    let mut carry = 0u32;
182    for (i, &l) in long.iter().enumerate() {
183        let mut t = l + short.get(i).copied().unwrap_or(0) + carry;
184        carry = u32::from(t >= LIMB_BASE);
185        if carry != 0 {
186            t -= LIMB_BASE;
187        }
188        out.push(t);
189    }
190    if carry != 0 {
191        out.push(carry);
192    }
193    out
194}

196/// `acc += addend`, the carry running on into `acc`'s higher limbs. The
197/// caller guarantees the sum fits.
198fn add_into(acc: &mut [u32], addend: &[u32]) {
199    let mut carry = 0u32;
200    let mut i = 0;
201    while i < addend.len() || carry != 0 {
202        let mut t = acc[i] + addend.get(i).copied().unwrap_or(0) + carry;
203        carry = u32::from(t >= LIMB_BASE);
204        if carry != 0 {
205            t -= LIMB_BASE;
206        }
207        acc[i] = t;
208        i += 1;
209    }
210}

212/// `acc -= sub`, which the caller guarantees does not go negative; any limbs
213/// of `sub` past `acc`'s length are zero.
214fn sub_from(acc: &mut [u32], sub: &[u32]) {
215    let mut borrow = 0u32;
216    let mut i = 0;
217    while i < sub.len() || borrow != 0 {
218        let s = sub.get(i).copied().unwrap_or(0) + borrow;
219        if i >= acc.len() {
220            debug_assert_eq!(s, 0, "the difference is not negative");
221            i += 1;
222            continue;
223        }
224        if acc[i] >= s {
225            acc[i] -= s;
226            borrow = 0;
227        } else {
228            acc[i] = acc[i] + LIMB_BASE - s;
229            borrow = 1;
230        }
231        i += 1;
232    }
233}

235/// `(n / d, n % d)` for a non-zero `d` without leading zero limbs: the
236/// quotient, possibly with leading zero limbs, and the remainder, without
237/// any (so a zero remainder is empty).
238pub(crate) fn divide(n: &[u32], d: &[u32]) -> (Vec<u32>, Vec<u32>) {
239    debug_assert!(d.last().is_some_and(|top| *top != 0), "a trimmed divisor");
240    if n.len() < d.len() {
241        let mut rem = n.to_vec();
242        trim(&mut rem);
243        return (vec![0], rem);
244    }
245    if d.len() == 1 {
246        let divisor = u64::from(d[0]);
247        let mut q = vec![0u32; n.len()];
248        let mut rem = 0u64;
249        for i in (0..n.len()).rev() {
250            let cur = rem * BASE + u64::from(n[i]);
251            // `rem < divisor`, so `cur < divisor * BASE`.
252            q[i] = to_limb(cur / divisor);
253            rem = cur % divisor;
254        }
255        let rem = if rem == 0 { vec![] } else { vec![to_limb(rem)] };
256        return (q, rem);
257    }
258
259    // Knuth's algorithm D: normalise so the divisor's top limb is at least
260    // half the base, which keeps each quotient-limb estimate within two of
261    // the truth.
262    let k = d.len();
263    let factor = BASE / (u64::from(d[k - 1]) + 1);
264    let dd = scale(d, factor);
265    debug_assert_eq!(dd.len(), k, "normalising never grows the divisor");
266    let mut nn = scale(n, factor);
267    nn.resize(n.len() + 1, 0);
268    let m = n.len() - k;
269    let mut q = vec![0u32; m + 1];
270    let top = u64::from(dd[k - 1]);
271    let next = u64::from(dd[k - 2]);
272    for j in (0..=m).rev() {
273        let numerator = u64::from(nn[j + k]) * BASE + u64::from(nn[j + k - 1]);
274        let mut qhat = numerator / top;
275        let mut rhat = numerator % top;
276        while qhat >= BASE || qhat * next > rhat * BASE + u64::from(nn[j + k - 2]) {
277            qhat -= 1;
278            rhat += top;
279            if rhat >= BASE {
280                break;
281            }
282        }
283        // nn[j..=j+k] -= qhat * dd
284        let mut carry = 0u64;
285        let mut borrow = 0i64;
286        for i in 0..k {
287            let p = qhat * u64::from(dd[i]) + carry;
288            carry = p / BASE;
289            let t = i64::from(nn[j + i]) - i64::from(to_limb(p % BASE)) - borrow;
290            if t < 0 {
291                nn[j + i] = signed_limb(t + SIGNED_BASE);
292                borrow = 1;
293            } else {
294                nn[j + i] = signed_limb(t);
295                borrow = 0;
296            }
297        }
298        // `qhat` is below `2 * BASE` and a limb below `BASE`, so the carry is
299        // at most `2 * BASE`.
300        let carry = i64::try_from(carry).expect("a carry of at most 2 * BASE");
301        let t = i64::from(nn[j + k]) - carry - borrow;
302        if t < 0 {
303            // The estimate was one too high: add the divisor back.
304            nn[j + k] = signed_limb(t + SIGNED_BASE);
305            qhat -= 1;
306            let mut carry = 0u32;
307            for i in 0..k {
308                let mut s = nn[j + i] + dd[i] + carry;
309                carry = u32::from(s >= LIMB_BASE);
310                if carry != 0 {
311                    s -= LIMB_BASE;
312                }
313                nn[j + i] = s;
314            }
315            // The carry out cancels the borrow left in the top limb.
316            nn[j + k] = to_limb((u64::from(nn[j + k]) + u64::from(carry)) % BASE);
317        } else {
318            nn[j + k] = signed_limb(t);
319        }
320        // The estimate starts below `BASE^2 / (BASE / 2)` and only falls.
321        q[j] = u32::try_from(qhat).expect("an estimate below 2 * BASE");
322    }
323    // The remainder is what is left of the scaled numerator, scaled back.
324    let mut rem = vec![0u32; k];
325    let mut carry = 0u64;
326    for i in (0..k).rev() {
327        let cur = carry * BASE + u64::from(nn[i]);
328        // `carry < factor`, so `cur < factor * BASE`.
329        rem[i] = to_limb(cur / factor);
330        carry = cur % factor;
331    }
332    debug_assert_eq!(carry, 0, "the scaled remainder divides by the scale");
333    trim(&mut rem);
334    (q, rem)
335}

337/// `limbs * factor` for a factor below the base.
338fn scale(limbs: &[u32], factor: u64) -> Vec<u32> {
339    let mut out = Vec::with_capacity(limbs.len() + 1);
340    let mut carry = 0u64;
341    for &limb in limbs {
342        let t = u64::from(limb) * factor + carry;
343        out.push(to_limb(t % BASE));
344        carry = t / BASE;
345    }
346    if carry != 0 {
347        // `limb * factor + carry < BASE * factor`, so the carry is below the
348        // factor.
349        out.push(to_limb(carry));
350    }
351    out
352}

360    /// xorshift64*, so a failure reproduces from the seed.
361    struct Rng(u64);
362    impl Rng {
363        fn next(&mut self) -> u64 {
364            let mut x = self.0;
365            x ^= x >> 12;
366            x ^= x << 25;
367            x ^= x >> 27;
368            self.0 = x;
369            x.wrapping_mul(0x2545_F491_4F6C_DD1D)
370        }
371        fn below(&mut self, n: u64) -> u64 {
372            self.next() % n
373        }
374        /// A digit string with a non-zero lead, biased toward runs of 9s and
375        /// 0s, which is where carries and borrows go wrong.
376        fn digits(&mut self, len: usize) -> Vec<u8> {
377            let style = self.below(4);
378            (0..len)
379                .map(|i| {
380                    let d = match style {
381                        0 => 9,
382                        1 if i > 0 => 0,
383                        2 if self.below(3) == 0 => 9,
384                        _ => u8::try_from(self.below(10)).expect("a digit"),
385                    };
386                    if i == 0 && d == 0 { 1 } else { d }
387                })
388                .collect()
389        }
390    }

392    /// Digit-by-digit reference product.
393    fn reference_mul(a: &[u8], b: &[u8]) -> Vec<u8> {
394        let mut out = vec![0u32; a.len() + b.len()];
395        for (i, x) in a.iter().rev().enumerate() {
396            for (j, y) in b.iter().rev().enumerate() {
397                out[i + j] += u32::from(*x) * u32::from(*y);
398            }
399        }
400        let mut carry = 0;
401        for slot in out.iter_mut() {
402            let v = *slot + carry;
403            *slot = v % 10;
404            carry = v / 10;
405        }
406        let mut digits: Vec<u8> = out
407            .iter()
408            .rev()
409            .map(|d| u8::try_from(*d).expect("a digit"))
410            .collect();
411        let lead = digits.iter().take_while(|d| **d == 0).count();
412        digits.drain(..lead.min(digits.len() - 1));
413        digits
414    }

417    fn digits_round_trip_through_limbs() {
418        let mut rng = Rng(7);
419        for len in 1..60 {
420            for zeros in [0, 1, 8, 9, 10, 17, 18, 19] {
421                let digits = rng.digits(len);

426            }

432    #[test]
433    fn the_limb_product_is_the_digit_product() {
434        let mut rng = Rng(11);
435        // Lengths either side of the Karatsuba threshold, and lopsided pairs.
436        let karatsuba_digits = KARATSUBA_LIMBS * LIMB_DIGITS;
437        let lengths = [
438            1,
439            9,
440            10,
441            50,
442            karatsuba_digits - 1,
443            karatsuba_digits,
444            karatsuba_digits + 9,
445            2 * karatsuba_digits + 5,
446            5 * karatsuba_digits + 3,
447        ];
448        for &la in &lengths {
449            for &lb in &lengths {
450                for _ in 0..3 {
451                    let a = rng.digits(la);
452                    let b = rng.digits(lb);
453                    let product = mul(&from_digits(&a, 0), &from_digits(&b, 0));
454                    assert_eq!(to_digits(&product), reference_mul(&a, &b), "{la} x {lb}");

462        let mut rng = Rng(13);
463        for _ in 0..3000 {
464            let ld = 1 + usize::try_from(rng.below(80)).expect("below 80");
465            let lq = 1 + usize::try_from(rng.below(80)).expect("below 80");
466            let d = from_digits(&rng.digits(ld), 0);
467            let qq = from_digits(&rng.digits(lq), 0);
468            // Zero, or a remainder with fewer digits than the divisor.

470                vec![]
471            } else {
472                let len = 1 + usize::try_from(rng.below(u64::cast_from(ld) - 1)).expect("below ld");
473                from_digits(&rng.digits(len), 0)

475            let mut n = mul(&qq, &d);
476            let mut padded = r.clone();
477            padded.resize(n.len(), 0);
478            add_into(&mut n, &padded);
479            trim(&mut n);
480            let (q, rem) = divide(&n, &d);
481            assert_eq!(to_digits(&q), to_digits(&qq));
482            assert_eq!(to_digits(&rem), to_digits(&r));
