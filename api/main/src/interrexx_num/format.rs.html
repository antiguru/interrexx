8/// What `FORMAT` can fail with. Both are error 93 (`Incorrect call to
9/// method`); the interpreter further distinguishes 93.941/93.942, but this
10/// crate follows `ArithError`'s lead and exposes only the number a trapped
11/// Rexx program actually sees in `RC`, using the variant itself to carry the
12/// finer distinction for Rust callers. Every raise site lives in this file,
13/// so each variant carries the substitution *values* directly rather than
14/// pre-rendered text -- `message()` renders from the generated table on
15/// demand, and `additional()` exposes those same values in the
16/// interpreter's own order (what `condition('o')~additional` would return).
17#[derive(Clone, PartialEq, Eq, Debug)]
18pub enum FormatError {
19    /// `before` is too narrow for the integer part actually produced (which,
20    /// in exponential form, means the mantissa's integer digits). Error
21    /// 93.942; `additional()` is `[reported.format(digits), before]`.
22    /// ```text
23    /// format(1.5,0)                            "1.5"          no `after`, nothing rounded
24    /// format(1.5,0,0)                          "2"            `after` rounded it, and carried
25    /// format(99.996,2,2)                       "100.0"        the rounding's
26    /// digit count, not four format(-0.5,1,0)                         "-1"
27    /// the sign survives a full round-away format(123456.789,2)
28    /// "123456.789"   plain: not reframed ENGINEERING format(123456.789,2,,,0)
29    /// "123.456789"   reframed by the engineering exponent ENGINEERING
30    /// format(123456.789,2,3,,0)    "123.457"      reframed and then rounded
31    /// format(1e10,2,,0)                        "1E+10"        rendered through
32    /// the string value ENGINEERING format(1e10,0,,0)            "10E+9"
33    /// ...which honours FORM too
34    /// ```
35    BeforeOversize {
36        reported: Number,
37        digits: u64,
38        form: Form,
39        before: u32,
40    },
41    /// `expp` is too narrow to hold the exponent's digits. Error 93.941;
42    /// `additional()` is `[render(mantissa), width]`, where `render` is a
43    /// plain (no `before`/`after`) rendering of `mantissa` -- see the two
44    /// call sites (`format_with`'s inline check, and
45    /// `post_carry_exponent_error`) for what `mantissa` is reframed to and
46    /// why there are two of them.
47    ExponentOversize { mantissa: Number, width: u32 },
48}
49
50impl FormatError {
51    pub fn code(self) -> u16 {
52        93
53    }
54
55    /// The sub-message identifying this failure's exact table row -- see
56    /// `ArithError::sub_code`'s doc comment for why `code()` alone isn't
57    /// enough (here it would be, since both variants share major 93, but
58    /// the pattern matches the crate's other two error types).
59    fn sub(&self) -> u16 {
60        match self {
61            FormatError::BeforeOversize { .. } => 942,
62            FormatError::ExponentOversize { .. } => 941,
63        }
64    }
65
66    /// The full `(major, sub)` pair, for a caller that has to build the
67    /// interpreter's own condition object rather than a message.
68    pub fn sub_code(&self) -> (u16, u16) {
69        (93, self.sub())
70    }
71
72    /// The substitution values in the interpreter's own order -- what
73    /// `condition('o')~additional` would return for this failure.
74    pub fn additional(&self) -> Vec<String> {
75        match self {
76            FormatError::BeforeOversize {
77                reported,
78                digits,
79                form,
80                before,
81            } => {
82                vec![reported.format_form(*digits, *form), before.to_string()]
83            }
84            FormatError::ExponentOversize { mantissa, width } => {
85                let rendered =
86                    render_integer_padded(mantissa, None, None, None, 0, Form::Scientific)
87                        .expect("`before` is None, so the width check cannot fail");
88                vec![rendered, width.to_string()]
89            }
90        }
91    }
92
93    /// The interpreter's exact message text, rendered from the generated
94    /// table on demand.
95    pub fn message(&self) -> String {
96        let subs = self.additional();
97        let refs: Vec<&str> = subs.iter().map(String::as_str).collect();
98        crate::error_text(93, self.sub(), &refs)
99    }
100}

103    /// Renders the number the way automatic-conversion display and `FORMAT`'s
104    /// defaults do, honouring `NUMERIC FORM`. `Number::format` only ever
105    /// produces SCIENTIFIC; this is the form-aware entry point the brief
106    /// asks for, implemented as the all-defaults case of [`Number::format_with`]
107    /// so the two can never drift apart.
108    pub fn format_form(&self, digits: u64, form: Form) -> String {
109        self.format_with(digits, form, None, None, None, None)
110            .expect("no `before`/`expp` supplied, so no width check can fail")
111    }

113    /// Writes into `out` what [`Number::format_form`] answers at `digits`,
114    /// for a value with a fractional part that renders exactly and plainly
115    /// in at most `out.len()` bytes; `None`, writing nothing, otherwise.
116    ///
117    /// Exact means no digit is rounded away (the value holds at most
118    /// `digits` digits) and plain means no exponent, which is the same in
119    /// either form. Under those conditions the rendering spells every
120    /// stored digit and the negative exponent as the point's position, so
121    /// parsing it back yields this number's digits and exponent unchanged.
122    pub fn write_short_fraction(&self, digits: u64, out: &mut [u8]) -> Option<usize> {
123        let count = self.digits.len();
124        if self.exponent >= 0 || u64::cast_from(count) > digits {
125            return None;
126        }
127        let places = usize::cast_from(self.exponent.unsigned_abs());
128        // `format_with`'s exponential trigger for a value below one: more
129        // places after the point than twice DIGITS.
130        let adjusted = i64::from(self.exponent) + signed_len(count) - 1;
131        if adjusted < 0 && u64::cast_from(places) > digits.saturating_mul(2) {
132            return None;
133        }
134        let whole = count.saturating_sub(places);
135        let leading_zeros = places.saturating_sub(count);
136        let length = usize::from(self.negative) + whole.max(1) + 1 + places;
137        if length > out.len() {
138            return None;
139        }
140        let mut at = 0;
141        if self.negative {
142            out[at] = b'-';
143            at += 1;
144        }
145        if whole == 0 {
146            out[at] = b'0';
147            at += 1;
148        }
149        for &digit in &self.digits[..whole] {
150            out[at] = b'0' + digit;
151            at += 1;
152        }
153        out[at] = b'.';
154        at += 1;
155        for slot in &mut out[at..at + leading_zeros] {
156            *slot = b'0';
157        }
158        at += leading_zeros;
159        for &digit in &self.digits[whole..] {
160            out[at] = b'0' + digit;
161            at += 1;
162        }
163        debug_assert_eq!(at, length);
164        Some(length)
165    }

167    /// Implements the `FORMAT(number, before, after, expp, expt)` builtin.
168    pub fn format_with(
169        &self,
170        digits: u64,
171        form: Form,
172        before: Option<u32>,
173        after: Option<u32>,
174        expp: Option<u32>,
175        expt: Option<u32>,
176    ) -> Result<String, FormatError> {
177        // Every FORMAT call rounds to the *current* DIGITS first, regardless
178        // of what `expt` says -- `expt` only moves the exponential-form
179        // trigger, it does not add or remove precision. Confirmed with a
180        // 15-significant-digit literal at DIGITS 9: it always renders as if
181        // it had been rounded to 9 digits first, however `expt` is set.
182        let n1 = self.round_to(digits);
183        // `BeforeOversize`'s &1 comes from `(n1, digits)` -- see its doc
184        // comment for what exactly that renders and why. Stored as the
185        // `Number` and the `digits` it needs, not pre-rendered, since
186        // `additional()` does the rendering on demand.
187
188        // `expp == 0` is not "no padding": it suppresses exponential form
189        // altogether, so a number that would otherwise trigger it renders in
190        // full plain digits instead (verified with 1E100 at expp 0, which
191        // prints all 101 digits). Everything else about it -- notably the
192        // `before` check -- then operates on that plain rendering.
193        let expt = trigger(digits, expp, expt);
194
195        // `expp` is checked *twice*. The first check runs against the exponent
196        // derived from the number alone, before the section that applies
197        // `after` and can carry -- it has no notion yet of the value carry will
198        // eventually produce. So this first check (and its substitution) needs
199        // the uncarried trigger/grouping -- `initial_exponent`, which is also
200        // `resolve_exponential_shape`'s first guess -- not the final,
201        // carry-resolved one that drives the successful-render path below. This
202        // is shown by forcing a carry that bumps the exponent from 20 to 21
203        // digits (`9.996E+20` rounded to 0 decimals) while `expp` is too narrow
204        // for either: the reported mantissa is `"9.996"` (reframed at the
205        // pre-carry 20), not `"0.9996"` (reframed at the post-carry 21).
206        let initial_eng_exp = initial_exponent(Shape::of(&n1), form, expt);
207        if let Some(width) = expp {
208            if let Some(exp0) = initial_eng_exp {
209                let needed = u32::try_from(exp0.unsigned_abs().to_string().len())
210                    .expect("a u32 has at most ten digits");
211                if needed > width {
212                    return Err(FormatError::ExponentOversize {
213                        mantissa: reframe(&n1, exp0),
214                        width,
215                    });
216                }
217            }
218            // The whole trigger/width check is redone a *second* time, right
219            // after the decimal-rounding carry that the first check couldn't
220            // have known about. Two shapes need it: a carry that grows an
221            // *already* -triggered exponent's digit count (`9.996E+99` rounded
222            // to 0 decimals at DIGITS 9, `expp` 2 -- pre-carry exponent 99
223            // fits, post-carry 100 does not), and a carry that triggers
224            // exponential form for the first time on a value that started plain
225            // (`9999999999.6` at DIGITS 15, `expt` 10, `after` 0, `expp` 1 --
226            // adjusted exponent 9 does not clear the trigger, but rounding away
227            // the ".6" carries it to 10, which does). The first was a
228            // regression caught by review, not by any of the 21,296 cases the
229            // four curated FORMAT sets already ran.
230            if let Some(err) =
231                post_carry_exponent_error(&n1, initial_eng_exp, form, expt, after, width)
232            {
233                return Err(err);
234            }
235        }
236
237        let (eng_exp, rounded) = resolve_exponential_state(&n1, form, expt, after);
238        // What `BeforeOversize` substitutes, computed here rather than at the
239        // raise site because it needs both exponent choices -- see
240        // `reported_value` -- and computed only when `before` is supplied,
241        // since nothing else can reach that raise.
242        let reported = before.map(|_| reported_value(&n1, initial_eng_exp, eng_exp, after));
243
244        // The exponent check comes before the `before` check -- an exponent
245        // that doesn't fit is reported even when `before` would have been
246        // wide enough for the mantissa. It has already run above, so the
247        // `?` here is the later of the two.
248        let layout = pad_to_before(
249            plain_layout(Shape::of(&rounded), after),
250            before,
251            reported.as_ref(),
252            digits,
253            form,
254        )?;
255        let field = ExponentField::decide(eng_exp, expp);
256
257        // **Measured out first and written once**, which is what lets this
258        // one `String` be allocated at its final size and never grow.
259        let mut out = String::with_capacity(layout.len() + field.len());
260        emit_plain(&mut out, &rounded, layout);
261        field.push_to(&mut out);
262        // What makes the layout a promise about the bytes rather than a
263        // capacity hint for them -- and what says the allocation above stayed
264        // one: the final length equals the capacity asked for, so nothing
265        // pushed into it ever had to grow it.
266        debug_assert_eq!(
267            out.len(),
268            layout.len() + field.len(),
269            "the rendering disagrees with the layout it was emitted from"
270        );
271        Ok(out)
272    }

274    /// The exponent a [`format_with`](Self::format_with) call with these same arguments would
275    /// display, or `None` when the result is plain and no exponent field is
276    /// written at all.
277    pub fn format_exponent(
278        &self,
279        digits: u64,
280        form: Form,
281        after: Option<u32>,
282        expp: Option<u32>,
283        expt: Option<u32>,

286        resolve_exponential_state(&n1, form, trigger(digits, expp, expt), after).0

289    /// Implements `TRUNC(number, places)`. Truncates, does not round, and
290    /// -- unlike `FORMAT` -- never produces exponential form: `TRUNC(1E10)`
291    /// is the eleven-digit integer, not `1E+10`.
292    pub fn trunc(&self, digits: u64, places: u32) -> String {
293        let n = self.round_to(digits);
294        let truncated = truncate_to_places(&n, places);
295        // `before` is always `None` here, so the oversize check can never
296        // run and the placeholder `value`/`digits` are never read.
297        render_integer_padded(&truncated, None, Some(places), None, 0, Form::Scientific)
298            .expect("`before` is None, so the width check cannot fail")
299    }
300}

302/// The exponential trigger `format_with` compares against, with `expp`'s
303/// suppression folded in. `None` means "never exponential".
304fn trigger(digits: u64, expp: Option<u32>, expt: Option<u32>) -> Option<i64> {
305    if expp == Some(0) {
306        return None;
307    }
308    Some(
309        expt.map(i64::from)
310            .unwrap_or_else(|| i64::try_from(digits).unwrap_or(i64::MAX)),
311    )
312}

314/// Chooses ENGINEERING's exponent for a value whose most significant digit
315/// sits at `adjusted`: the largest multiple of 3 not exceeding it. Found by
316/// sweeping literal `12eK` across `-30..=30` and comparing engineering to
317/// scientific at each `K` -- `div_euclid` is exactly this floor division,
318/// including on the negative side (`-25 -> -27`, not `-24`).
319fn group(form: Form, adjusted: i32) -> i32 {
320    match form {
321        Form::Scientific => adjusted,
322        Form::Engineering => adjusted.div_euclid(3) * 3,
323    }
324}

326/// Rescales `n1` so its value reads as `mantissa * 10^exp`. Used both to turn
327/// a number into the digits that sit before `E`, and -- inside
328/// `resolve_exponential_shape` -- to redo that rescaling from scratch once
329/// rounding has changed which `exp` is the right one, rather than patching a
330/// stale mantissa.
331fn reframe(n1: &Number, exp: i32) -> Number {
332    Number {
333        negative: n1.negative,
334        digits: n1.digits.clone(),
335        exponent: Shape::of(n1).reframed(exp).exponent,
336    }
337}

339/// A number reduced to everything the exponential decision and the layout
340/// read: how many digits it has, where the point sits relative to them, and
341/// whether it has a sign.
342#[derive(Clone, Copy, PartialEq, Eq, Debug)]
343struct Shape {
344    len: usize,
345    exponent: i32,
346    negative: bool,
347}
348
349impl Shape {
350    fn of(n: &Number) -> Self {
351        Shape {
352            len: n.digits.len(),
353            exponent: n.exponent,
354            negative: n.negative,
355        }
356    }
357
358    /// The exponent of the most significant digit.
359    fn adjusted(self) -> i32 {
360        self.exponent + signed_count(self.len) - 1
361    }
362
363    /// The shape [`reframe`] produces: rescaling a value to read as
364    /// `mantissa * 10^exp` moves the point and leaves the digits and the sign
365    /// where they were.
366    fn reframed(self, exp: i32) -> Self {
367        Shape {
368            exponent: self.exponent - exp,
369            ..self
370        }
371    }
372}

374/// Whether a value of this shape renders exponentially before `after` has cut
375/// anything, and at what exponent -- `None` for plain form.
376fn initial_exponent(shape: Shape, form: Form, expt: Option<i64>) -> Option<i32> {
377    let expt = expt?;
378    let a = shape.adjusted();
379    // `saturating_mul`: `expt` may itself be the saturated-i64 form of a huge
380    // bare `digits` (see `trigger`), and doubling that must stay huge rather
381    // than wrap.
382    let triggered =
383        i64::from(a) >= expt || (a < 0 && i64::from(shape.exponent).abs() > expt.saturating_mul(2));
384    triggered.then(|| group(form, a))
385}

387/// Decides whether `n1` renders in exponential form and, if so, at what
388/// exponent, then applies `after` to the resulting mantissa (or to `n1`
389/// itself in plain form).
390fn resolve_exponential_shape(
391    n1: &Number,
392    form: Form,
393    expt: Option<i64>,
394    after: Option<u32>,
395) -> (Option<i32>, Shape) {
396    let base = Shape::of(n1);
397    // The shape a pass works on: `n1` reframed at the current guess, and cut
398    // to `after` places if there is a cut to make. Reframing at `0` -- what an
399    // untriggered guess gives -- is the identity, which is why the plain arm
400    // needs no case of its own.
401    let step = |eng_exp: Option<i32>| -> Shape {
402        let at = eng_exp.unwrap_or(0);
403        match after {
404            None => base.reframed(at),
405            Some(places) => Shape::of(&round_to_places(&reframe(n1, at), places)),
406        }
407    };
408
409    // The very first guess is the only place the low-end (fractional-value)
410    // trigger matters -- see `initial_exponent`. That is fine: rounding only
411    // ever grows a number, so a value already outside exponential range can
412    // only move further from the fractional trigger's territory, never into
413    // it.
414    let mut eng_exp = initial_exponent(base, form, expt);
415
416    for _ in 0..8 {
417        let rounded = step(eng_exp);
418        let true_adjusted = eng_exp.unwrap_or(0) + rounded.adjusted();
419        // Once exponential, always exponential (growth cannot shrink a
420        // magnitude back below the trigger); a number that started plain can
421        // only newly trigger via the upper (adjusted >= expt) arm.
422        let new_eng_exp = match expt {
423            Some(expt) if eng_exp.is_some() || i64::from(true_adjusted) >= expt => {
424                Some(group(form, true_adjusted))
425            }
426            _ => None,
427        };
428        if new_eng_exp == eng_exp {
429            return (eng_exp, rounded);
430        }
431        eng_exp = new_eng_exp;
432    }
433    (eng_exp, step(eng_exp))
434}

436/// The value [`resolve_exponential_shape`] decided on, materialised.
437fn resolve_exponential_state<'a>(
438    n1: &'a Number,
439    form: Form,
440    expt: Option<i64>,
441    after: Option<u32>,
442) -> (Option<i32>, Cow<'a, Number>) {
443    let (eng_exp, shape) = resolve_exponential_shape(n1, form, expt, after);
444    let framed = match eng_exp {
445        Some(exp) => Cow::Owned(reframe(n1, exp)),
446        None => Cow::Borrowed(n1),
447    };
448    let rounded = match after {
449        Some(places) => Cow::Owned(round_to_places(&framed, places)),
450        None => framed,
451    };
452    // The one place both halves are in hand at once: the shape the decision
453    // settled on is the shape of the value built from it. That decision is
454    // made in shape arithmetic, keeping no value it can be checked against --
455    // so this is what says the arithmetic tracks what `reframe` and
456    // `round_to_places` actually do.
457    debug_assert_eq!(
458        Shape::of(&rounded),
459        shape,
460        "the resolved value disagrees with the shape resolved for it"
461    );
462    (eng_exp, rounded)
463}

465/// Redoes the exponent-width trigger/check after the decimal-place cut `after`
466/// makes, as is done right after the rounding call -- see the call site's doc
467/// comment for *why* a second check exists at all. Returns `None` when no cut
468/// is even possible (`after` omitted), when the number has no decimals to cut,
469/// or when nothing is actually dropped -- the interpreter's own decimals
470/// section only reaches its redo under the same conditions.
471fn post_carry_exponent_error(
472    n1: &Number,
473    eng_exp0: Option<i32>,
474    form: Form,
475    expt: Option<i64>,
476    after: Option<u32>,
477    width: u32,
478) -> Option<FormatError> {
479    let after = after?;
480    // The state the rounding actually rounds: reframed to mantissa scale by the
481    // first check's exponent if it triggered, `n1` itself (a no-op reframe)
482    // otherwise.
483    let pre_round = reframe(n1, eng_exp0.unwrap_or(0));
484    if pre_round.exponent >= 0 {
485        return None; // no decimal places to cut
486    }
487    let adjusted_decimals = -i64::from(pre_round.exponent);
488    if adjusted_decimals <= i64::from(after) {
489        return None; // `after` already covers every decimal place present
490    }
491    let excess = adjusted_decimals - i64::from(after);
492    let len = signed_len(pre_round.digits.len());
493    if excess >= len {
494        // The interpreter's own "rounds away to a single digit or zero"
495        // branch, which does not redo the trigger/width check at all.
496        return None;
497    }
498    // `Number::round_to`'s carry -- bump the exponent, keep the digit count --
499    // is the rounding here; `round_to_places` (used for the successful render)
500    // grows the digit vector instead. See this function's doc comment for why
501    // that difference matters here specifically.
502    let rounded = pre_round.into_round((len - excess).unsigned_abs());
503    // Back to true scale: `rounded` is still relative to `eng_exp0`.
504    let true_scale = Number {
505        negative: rounded.negative,
506        digits: rounded.digits,
507        exponent: rounded.exponent + eng_exp0.unwrap_or(0),
508    };
509    let adjusted2 = Shape::of(&true_scale).adjusted();
510    let triggered = eng_exp0.is_some()
511        || matches!(
512            expt,
513            // `saturating_mul` for the same reason as the other two
514            // triggers: `expt` may be a saturated huge `digits` default.
515            Some(expt) if i64::from(adjusted2) >= expt
516                || (adjusted2 < 0 && i64::from(true_scale.exponent).abs() > expt.saturating_mul(2))
517        );
518    if !triggered {
519        return None;
520    }
521    let exp2 = group(form, adjusted2);
522    let needed =
523        u32::try_from(exp2.unsigned_abs().to_string().len()).expect("a u32 has at most ten digits");
524    if needed <= width {
525        return None;
526    }
527    Some(FormatError::ExponentOversize {
528        mantissa: reframe(&true_scale, exp2),
529        width,
530    })
531}

533/// The number the formatting is holding by the time it reports
534/// `BeforeOversize`, which is neither the operand nor the rendered result.
535fn reported_value(
536    n1: &Number,
537    eng_exp0: Option<i32>,
538    eng_exp: Option<i32>,
539    after: Option<u32>,
540) -> Number {
541    let framed = reframe(n1, eng_exp0.unwrap_or(0));
542    let cut = match after {
543        None => framed,
544        Some(places) => math_round_places(framed, places),
545    };
546    let true_scale = Number {
547        negative: cut.negative,
548        digits: cut.digits,
549        exponent: cut.exponent + eng_exp0.unwrap_or(0),
550    };
551    reframe(&true_scale, eng_exp.unwrap_or(0))
552}

554/// The decimals cut the formatting makes, digit for digit.
555fn math_round_places(n: Number, places: u32) -> Number {
556    if n.exponent >= 0 {
557        return n;
558    }
559    let adjusted_decimals = -i64::from(n.exponent);
560    if adjusted_decimals <= i64::from(places) {
561        return n;
562    }
563    let excess = adjusted_decimals - i64::from(places);
564    let len = signed_len(n.digits.len());
565    if excess < len {
566        return n.into_round((len - excess).unsigned_abs());
567    }
568    if excess == len && n.digits[0] >= 5 {
569        // `excess < adjusted_decimals` always holds here, and
570        // `adjusted_decimals` is within `MAX_EXPONENT`, so `places` fits i32.
571        return Number {
572            negative: n.negative,
573            digits: Digits::single(1),
574            exponent: -i32::try_from(places).expect("below the decimals"),
575        };
576    }
577    Number::zero()
578}

580/// Rounds (half up) `n` to exactly `places` digits after the decimal point --
581/// the cut FORMAT's `after` and TRUNC both make, just at a fixed decimal
582/// position rather than a fixed significant-digit count. `Number::round_to`
583/// cannot be reused here: its `digits == 0` is a no-op sentinel (see its own
584/// doc comment for why), whereas `places == 0` is an ordinary, meaningful
585/// cut for this crate's callers (`FORMAT(x, , 0)`), and must still round a
586/// `0.6` up to `1`.
587fn round_to_places(n: &Number, places: u32) -> Number {
588    // Widened to `i64` up front: `-(places as i32)` alone overflows once
589    // `places >= 2^31` (a `u32` value the interpreter accepts without
590    // complaint), so even deciding which branch to take must not go
591    // through `i32`.
592    let target_exponent_wide = -i64::from(places);
593    if i64::from(n.exponent) >= target_exponent_wide {
594        return n.clone();
595    }
596
597    // Only reachable once `target_exponent_wide > n.exponent`. `n.exponent`
598    // is always within `+/-MAX_EXPONENT` (comfortably inside `i32`), and
599    // `target_exponent_wide <= 0` always (`places` is never negative), so
600    // it is squeezed into `(n.exponent, 0]` here -- which always fits `i32`.
601    // A `places` large enough to overflow can therefore never reach this
602    // branch; it always takes the early return above instead.
603    let target_exponent =
604        i32::try_from(target_exponent_wide).expect("between the exponent and zero");
605    let drop = usize::cast_from((target_exponent - n.exponent).unsigned_abs());
606    let len = n.digits.len();
607    if drop > len {
608        // The cut falls to the left of every stored digit -- the digit that
609        // would decide round-up-or-down is an implicit, unstored zero, so
610        // there is nothing to round: `0.04` to 0 places rounds down to zero.
611        // But it is a *fixed-decimal* zero, not the canonical one --
612        // `Number::zero()`/`assemble` would collapse it to exponent 0 and
613        // lose the requested `places`, when the caller needs it preserved
614        // (`FORMAT`/`TRUNC` on `0.000012345` at 1 place is `0.0`, not `0`,
615        // confirmed against the interpreter). The sign is still dropped: a
616        // value that rounds away to nothing has no sign left to show,
617        // exactly as a whole-number underflow already does.
618        return Number {
619            negative: false,
620            digits: Digits::single(0),
621            exponent: target_exponent,
622        };
623    }
624    let keep = len - drop;
625    let mut kept = Digits::from_slice(&n.digits[..keep]);
626    if n.digits[keep] >= 5 {
627        // A full carry chain (all-9s down to `keep`) grows the digit count
628        // by one instead of shifting the exponent -- unlike `round_to`,
629        // which holds the digit count fixed and lets the exponent absorb the
630        // growth. Here the exponent (`places`, hence `target_exponent`) is
631        // what the caller fixed, so the growth has to show up as an extra
632        // digit: `99.96` to 1 place is `100.0`, not `10.0` at one exponent
633        // higher.
634        let mut i = keep;
635        loop {
636            if i == 0 {
637                kept.insert_front(1);
638                break;
639            }
640            i -= 1;
641            if kept[i] == 9 {
642                kept[i] = 0;
643            } else {
644                kept[i] += 1;
645                break;
646            }
647        }
648    }
649    if kept.is_empty() {
650        // `keep == 0` and no carry: rounds down to nothing, same underflow
651        // as the `drop > len` branch above, and the same reason `assemble`
652        // cannot be used -- it would collapse this to the canonical zero
653        // and lose `target_exponent` (`0.001` to 2 places is `0.00`, not
654        // `0`, confirmed against the interpreter). A carry reaching this
655        // point instead (`if i == 0` above) leaves `kept` non-empty, so it
656        // takes the normal `assemble` path below.
657        return Number {
658            negative: false,
659            digits: Digits::single(0),
660            exponent: target_exponent,
661        };
662    }
663    Number::assemble(n.negative, kept, target_exponent)
664}
665
666/// TRUNC's cut: same fixed decimal position as `round_to_places`, but drops
667/// the extra digits outright instead of deciding whether to carry. Padding
668/// (when `n` already has at most `places` decimal digits) is deferred to
669/// `render_integer_padded` the same way and for the same reason -- see
670/// `round_to_places`'s doc comment.
671fn truncate_to_places(n: &Number, places: u32) -> Number {
672    let target_exponent_wide = -i64::from(places);
673    if i64::from(n.exponent) >= target_exponent_wide {
674        return n.clone();
675    }
676    let target_exponent =
677        i32::try_from(target_exponent_wide).expect("between the exponent and zero");
678    let drop = usize::cast_from((target_exponent - n.exponent).unsigned_abs());
679    let len = n.digits.len();
680    if drop >= len {
681        // Everything gets dropped -- same underflow as `round_to_places`,
682        // and the same fix: keep `target_exponent` so the requested
683        // decimal-place count still shows (`TRUNC(0.000012345, 1)` is
684        // `0.0`, not `0`), constructed directly rather than through
685        // `assemble`, which would collapse it back to the canonical zero.
686        return Number {
687            negative: false,
688            digits: Digits::single(0),
689            exponent: target_exponent,
690        };
691    }
692    let keep = len - drop;
693    // `assemble` here only ever sees a genuinely nonzero leading digit
694    // (`keep > 0` and `n` is nonzero, which never stores a leading zero),
695    // so it cannot re-trigger the collapse above. At `places == 0` the
696    // branch above and this one agree anyway -- `target_exponent` is 0
697    // there, which is exactly `Number::zero()` -- so `TRUNC(-0.5, 0)` still
698    // drops its sign; it is only `places > 0` where preserving the exponent
699    // here matters.
700    Number::assemble(
701        n.negative,
702        Digits::from_slice(&n.digits[..keep]),
703        target_exponent,
704    )
705}

707/// A plain rendering of `n` on its own: [`plain_layout`] and [`pad_to_before`]
708/// decide it, [`emit_plain`] writes it. For the callers that want no exponent
709/// field at all -- `TRUNC`, and the substitution `ExponentOversize` carries.
710fn render_integer_padded(
711    n: &Number,
712    before: Option<u32>,
713    after: Option<u32>,
714    reported: Option<&Number>,
715    oversize_digits: u64,
716    oversize_form: Form,
717) -> Result<String, FormatError> {
718    let layout = pad_to_before(
719        plain_layout(Shape::of(n), after),
720        before,
721        reported,
722        oversize_digits,
723        oversize_form,
724    )?;
725    let mut out = String::with_capacity(layout.len());
726    emit_plain(&mut out, n, layout);
727    debug_assert_eq!(
728        out.len(),
729        layout.len(),
730        "the rendering disagrees with the layout it was emitted from"
731    );
732    Ok(out)
733}

735/// The exact byte counts a plain rendering is about to have, decided before
736/// any of those bytes exist.
737#[derive(Clone, Copy, PartialEq, Eq, Debug)]
738struct PlainLayout {
739    /// The spaces `before` asks for, ahead of the sign.
740    pad: usize,
741    /// The sign, written only for a negative value.
742    sign: usize,
743    /// Digits before the decimal point, including the lone `0` a value below
744    /// one shows there.
745    int_len: usize,
746    /// The decimal part as (the value's own digits, the trailing zeros
747    /// `after` adds beyond them). `None` is "no decimal part", which is not
748    /// the same as `Some((0, 0))`: the first prints no `.` and the second
749    /// would.
750    dec: Option<(usize, usize)>,
751    /// Where the decimal point falls inside the digit run. Meaningful only
752    /// for a negative exponent -- a non-negative one puts the point past
753    /// every digit, which is the arm that has no natural decimal part at all.
754    point: i32,
755}
756
757impl PlainLayout {
758    /// The `+ 1` is the point itself, and it is inside the `map_or` rather
759    /// than added unconditionally: a value with no decimal part prints no
760    /// point, and a count one byte over would be right by luck rather than by
761    /// construction.
762    fn len(self) -> usize {
763        self.pad
764            + self.sign
765            + self.int_len
766            + self.dec.map_or(0, |(natural, extra)| 1 + natural + extra)
767    }
768}
769
770/// The number of decimal digits `value` is written with, without writing it.
771fn decimal_width(value: u32) -> usize {
772    usize::cast_from(value.checked_ilog10().unwrap_or(0)) + 1
773}
774
775/// Everything a rendering carries after its mantissa, decided once so that
776/// the count reserving room for it and the code writing it read the same
777/// decision.
778#[derive(Clone, Copy, PartialEq, Eq, Debug)]
779enum ExponentField {
780    /// Nothing follows -- plain form, or a displayed exponent of exactly zero
781    /// with no explicit `expp` whose field would need reserving.
782    Absent,
783    /// A displayed exponent of exactly zero *with* an explicit `expp`: the
784    /// field it would have taken is reserved as this many blanks rather than
785    /// vanishing outright.
786    Reserved(usize),
787    /// `E`, a sign, `zeros` leading zeros, and then `exp`'s own digits.
788    Present { exp: i32, zeros: usize },
789}
790
791impl ExponentField {
792    fn decide(eng_exp: Option<i32>, expp: Option<u32>) -> Self {
793        let Some(exp) = eng_exp else {
794            return ExponentField::Absent;
795        };
796        if exp == 0 {
797            // A displayed exponent of exactly zero is never written as `E+0`
798            // -- confirmed with `format(3.14159,,,,0)` at DIGITS 5 (adjusted
799            // exponent 0 exactly), which prints `3.1416`, not `3.1416E+0`,
800            // and with ENGINEERING grouping an adjusted exponent of 1 or 2
801            // down to a displayed 0 the same way.
802            return match expp {
803                Some(width) => ExponentField::Reserved(usize::cast_from(width) + 2),
804                None => ExponentField::Absent,
805            };
806        }
807        // **Not `format!("{:0width$}", ..)`.** Rust's format width is a
808        // `u16`, so that spelling panics with "Formatting argument out of
809        // range" at `expp == 65536`, where the interpreter answers a
810        // 65,539-byte result. Measured both sides:
811        // `length(format(1e10,,,65535))` is 65538 and
812        // `length(format(1e10,,,65536))` is 65539.
813        let zeros = match expp {
814            Some(width) => {
815                usize::cast_from(width).saturating_sub(decimal_width(exp.unsigned_abs()))
816            }
817            None => 0,
818        };
819        ExponentField::Present { exp, zeros }
820    }
821
822    fn len(self) -> usize {
823        match self {
824            ExponentField::Absent => 0,
825            ExponentField::Reserved(blanks) => blanks,
826            // The `E` and the sign, then the padding and the digits.
827            ExponentField::Present { exp, zeros } => 2 + zeros + decimal_width(exp.unsigned_abs()),
828        }
829    }
830
831    fn push_to(self, out: &mut String) {
832        match self {
833            ExponentField::Absent => {}
834            ExponentField::Reserved(blanks) => push_repeated(out, ' ', blanks),
835            ExponentField::Present { exp, zeros } => {
836                out.push('E');
837                out.push(if exp < 0 { '-' } else { '+' });
838                push_repeated(out, '0', zeros);
839                // Written straight into `out` rather than through
840                // `to_string`, which would allocate a `String` to be copied
841                // into this one; `write!` to a `String` cannot fail.
842                let _ = write!(out, "{}", exp.unsigned_abs());
843            }
844        }
845    }
846}

848/// Decides [`PlainLayout`] for a value of this shape, before `before` has had
849/// its say. **Infallible, and that is deliberate**: `before` is the only
850/// argument that can raise, so keeping it out of here leaves a rendering that
851/// supplies none -- which is what a number's ordinary display is -- with no
852/// `Result` carrying a `Number` to construct and drop.
853fn plain_layout(shape: Shape, after: Option<u32>) -> PlainLayout {
854    let point = signed_count(shape.len) + shape.exponent;
855    let int_len = if shape.exponent >= 0 {
856        shape.len + usize::cast_from(shape.exponent.unsigned_abs())
857    } else if point > 0 {
858        usize::cast_from(point.unsigned_abs())
859    } else {
860        1 // the "0" before the point
861    };
862    // The digits this value naturally shows after the point, before `after`
863    // asks for more. `None` is "no decimal part", which is not the same as
864    // `Some(0)`: the first prints no `.` and the second would.
865    let natural_dec = if shape.exponent >= 0 {
866        None
867    } else if point > 0 {
868        Some(shape.len - usize::cast_from(point.unsigned_abs()))
869    } else {
870        Some(usize::cast_from(point.unsigned_abs()) + shape.len)
871    };
872    // The decimal part as (natural digits, trailing zeros `after` adds).
873    let dec = match after {
874        None => natural_dec.map(|natural| (natural, 0usize)),
875        Some(0) => None,
876        Some(places) => {
877            let natural = natural_dec.unwrap_or(0);
878            let extra = usize::cast_from(places).saturating_sub(natural);
879            Some((natural, extra))
880        }
881    };
882    PlainLayout {
883        pad: 0,
884        sign: usize::from(shape.negative),
885        int_len,
886        dec,
887        point,
888    }
889}

891/// Applies `before` to a layout: the leading spaces that widen the integer
892/// part to it, or the raise for an integer part already wider.
893fn pad_to_before(
894    layout: PlainLayout,
895    before: Option<u32>,
896    // `Some` exactly when `before` is, which is the only way the oversize
897    // raise below can be reached.
898    reported: Option<&Number>,
899    oversize_digits: u64,
900    // Only ever read to build `BeforeOversize`, which renders `reported`
901    // through the string value and so honours `NUMERIC FORM` as well as
902    // `DIGITS`. Every call that passes `before: None` can never reach that raise
903    // and passes `Form::Scientific` as an unread placeholder.
904    oversize_form: Form,
905) -> Result<PlainLayout, FormatError> {
906    let Some(before) = before else {
907        return Ok(layout);
908    };
909    let needed = signed_len(layout.int_len);
910    // Widened to `i64`: `before` is a legitimate `u32` up to `u32::MAX` (the
911    // interpreter accepts `FORMAT(1, 3000000000)` and returns the
912    // three-billion-character result), and casting it to `i32` first can turn
913    // it negative, producing a spurious oversize error instead of the huge
914    // space-padding the interpreter actually shows.
915    let available = i64::from(before) - signed_len(layout.sign);
916    if available < needed {
917        return Err(FormatError::BeforeOversize {
918            reported: reported
919                .expect("`before` is Some here, and the caller pairs the two")
920                .clone(),
921            digits: oversize_digits,
922            form: oversize_form,
923            before,
924        });
925    }
926    Ok(PlainLayout {
927        // At most `before`, a `u32`.
928        pad: usize::try_from(available - needed).expect("at most a u32"),
929        ..layout
930    })
931}

933/// Writes the bytes [`plain_layout`] counted, and decides nothing of its own:
934/// every branch here is keyed on the exponent and the `point` the layout
935/// already carries.
936fn emit_plain(out: &mut String, n: &Number, layout: PlainLayout) {
937    let (dec, point) = (layout.dec, layout.point);
938    push_repeated(out, ' ', layout.pad);
939    if n.negative {
940        out.push('-');
941    }
942    if n.exponent >= 0 {
943        push_digits(out, n.digits.as_slice());
944        push_repeated(out, '0', usize::cast_from(n.exponent.unsigned_abs()));
945    } else if point > 0 {
946        push_digits(
947            out,
948            &n.digits.as_slice()[..usize::cast_from(point.unsigned_abs())],
949        );
950    } else {
951        out.push('0');
952    }
953    if let Some((_, extra)) = dec {
954        out.push('.');
955        // The natural digits, which exist only for a negative exponent. A
956        // non-negative one reaches here only because `after` asked for places,
957        // and then the whole decimal part is the zeros below.
958        if n.exponent < 0 {
959            if point > 0 {
960                push_digits(
961                    out,
962                    &n.digits.as_slice()[usize::cast_from(point.unsigned_abs())..],
963                );
964            } else {
965                push_repeated(out, '0', usize::cast_from(point.unsigned_abs()));
966                push_digits(out, n.digits.as_slice());
967            }
968        }
969        push_repeated(out, '0', extra);
970    }
971}

973/// Appends `digits` -- one decimal digit per byte, as [`Number`] stores them --
974/// to `out` as characters.
975fn push_digits(out: &mut String, digits: &[u8]) {
976    out.extend(digits.iter().map(|d| char::from(b'0' + d)));
977}
978
979/// Appends `count` copies of `ch` to `out`.
980fn push_repeated(out: &mut String, ch: char, count: usize) {
981    for _ in 0..count {
982        out.push(ch);
983    }
