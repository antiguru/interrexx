9    Scientific,
10    Engineering,
11}

13#[derive(Clone, PartialEq, Eq, Debug)]
14pub enum SettingsError {
15    /// `DIGITS` was not a positive whole number *expressible within the
16    /// DIGITS currently in force* -- not-a-number, zero, negative,
17    /// fractional, and too-many-digits all land here alike. Error 26.005;
18    /// `additional()` is `[found]`. Confirmed with `NUMERIC DIGITS abc`, and
19    /// with `NUMERIC DIGITS 1000` at DIGITS 3.
20    DigitsNotWhole { found: String },
21    /// `FUZZ` was negative, not a whole number, or not expressible within
22    /// the DIGITS currently in force -- the same code as `DigitsNotWhole`
23    /// (26) but a different sub-message (26.006 vs 26.005), which is exactly
24    /// why this is its own variant rather than one shared `NotWholeNumber`
25    /// with no way to tell the two apart. `additional()` is `[found]`.
26    /// Confirmed with `NUMERIC FUZZ -1`, and with `NUMERIC FUZZ 12345` at
27    /// DIGITS 3 (26.006, *not* 33.001: the conversion fails before the
28    /// fuzz-below-digits comparison is ever reached).
29    FuzzNotWhole { found: String },
30    /// `FUZZ` would not be strictly less than `DIGITS`. Raised whichever of
31    /// the two is the one being set. Error 33.001; `additional()` is
32    /// `[digits, fuzz]` -- the pending pair the setting would end up with
33    /// (the just-rejected candidate for whichever is being set, the
34    /// unchanged stored value for the other), not necessarily either's
35    /// current field on `self`.
36    FuzzNotBelowDigits { digits: u64, fuzz: u64 },
37    /// A `FORM` that is neither `SCIENTIFIC` nor `ENGINEERING`. Error 25.011;
38    /// `additional()` is `[found]`.
39    InvalidForm { found: String },
40}
41
42impl SettingsError {
43    /// The interpreter's error number. These are part of the contract, not an
44    /// implementation detail -- programs trap on them.
45    pub fn code(&self) -> u16 {
46        match self {
47            SettingsError::InvalidForm { .. } => 25,
48            SettingsError::DigitsNotWhole { .. } | SettingsError::FuzzNotWhole { .. } => 26,
49            SettingsError::FuzzNotBelowDigits { .. } => 33,
50        }
51    }
52
53    /// The `(major, sub)` pair identifying this failure's exact table row --
54    /// see `ArithError::sub_code`'s doc comment for why `code()` alone
55    /// isn't enough.
56    fn sub_code(&self) -> (u16, u16) {
57        match self {
58            SettingsError::InvalidForm { .. } => (25, 11),
59            SettingsError::DigitsNotWhole { .. } => (26, 5),
60            SettingsError::FuzzNotWhole { .. } => (26, 6),
61            SettingsError::FuzzNotBelowDigits { .. } => (33, 1),
62        }
63    }
64
65    /// The substitution values in the interpreter's own order -- what
66    /// `condition('o')~additional` would return for this failure.
67    pub fn additional(&self) -> Vec<String> {
68        match self {
69            SettingsError::DigitsNotWhole { found }
70            | SettingsError::FuzzNotWhole { found }
71            | SettingsError::InvalidForm { found } => vec![found.clone()],
72            SettingsError::FuzzNotBelowDigits { digits, fuzz } => {
73                vec![digits.to_string(), fuzz.to_string()]
74            }
75        }
76    }
77
78    /// The interpreter's message text, rendered from the generated table on
79    /// demand.
80    pub fn message(&self) -> String {
81        let (major, sub) = self.sub_code();
82        let subs = self.additional();
83        let refs: Vec<&str> = subs.iter().map(String::as_str).collect();
84        crate::error_text(major, sub, &refs)
85    }
86}

88/// The largest value any whole-number conversion can produce on 64-bit, and
89/// therefore the absolute ceiling on `DIGITS` itself (confirmed: `NUMERIC DIGITS
90/// 999999999999999999` succeeds from DIGITS 18, and 10^18 fails even from DIGITS
91/// 30). Unrelated to `crate::MAX_EXPONENT`, which bounds a number's
92/// *exponent* -- conflating the two was exactly the defect this rule
93/// replaces.
94const MAX_WHOLENUMBER: u64 = 999_999_999_999_999_999;
95
96#[derive(Clone, PartialEq, Eq, Debug)]
97pub struct Settings {
98    digits: u64,
99    fuzz: u64,
100    form: Form,
101}

107            fuzz: 0,
108            form: Form::Scientific,
109        }
110    }
111}

113/// The largest unsigned whole number expressible in `digits` significant
114/// positions, capped at `MAX_WHOLENUMBER`'s eighteen.
115fn max_value_for_digits(digits: u64) -> u64 {
116    if digits >= 18 {
117        MAX_WHOLENUMBER
118    } else {
119        // `digits < 18`, so the power fits.
120        10u64.pow(u32::try_from(digits).expect("below 18")) - 1
121    }
122}
123
124/// Converts `text` the way the interpreter converts a NUMERIC DIGITS/FUZZ
125/// operand: an unsigned-number request at the current precision
126/// -- i.e. an unsigned whole number
127/// **within the DIGITS setting currently in force** -- there is no fixed
128/// cap. The value is rounded at that DIGITS boundary first (half-up on the
129/// first dropped digit), every decimal digit that remains must reduce to
130/// zero -- literal zeros, or all nines consumed by the carry -- and the
131/// result must be at most `max_value_for_digits(digits)`.
132fn unsigned_whole_number(text: &str, digits: u64) -> Option<u64> {
133    let n = Number::parse(text)?;
134    if n.is_zero() {
135        return Some(0);
136    }
137    if n.negative {
138        return None;
139    }
140
141    let mut exponent = i64::from(n.exponent);
142    let mut length = n.digits.len();
143    let mut carry = false;
144    // More stored digits than the DIGITS in force: the value is rounded at
145    // that boundary, and the first dropped digit decides the carry.
146    if u64::cast_from(length) > digits {
147        let keep = usize::try_from(digits).expect("below the length");
148        exponent += crate::signed_len(length - keep);
149        carry = n.digits[keep] >= 5;
150        length = keep;
151    }
152    let kept = &n.digits[..length];
153    let max_value = max_value_for_digits(digits);
154
155    if exponent >= 0 {
156        return create_unsigned_value(kept, carry, exponent, max_value);
157    }
158
159    // Some kept digits sit below the decimal point. They must all reduce to
160    // zero for the value to be whole.
161    let decimals = exponent.unsigned_abs();
162    if carry && decimals > u64::cast_from(length) {
163        // The carry lands below the lowest stored digit; the padding zeros
164        // in between can never ripple it up to the integer part.
165        return None;
166    }
167    let required = if carry { 9 } else { 0 };
168    let checked =
169        usize::try_from(decimals.min(u64::cast_from(length))).expect("at most the length");
170    if kept[length - checked..].iter().any(|d| *d != required) {
171        return None;
172    }
173    if decimals >= u64::cast_from(length) {
174        // Nothing left of the point: the value truncates to 0, or to 1 when
175        // the all-nines carry ripples all the way up.
176        return Some(u64::from(carry));
177    }
178    let decimals = usize::try_from(decimals).expect("below the length");
179    create_unsigned_value(&kept[..length - decimals], carry, 0, max_value)
180}
181
182/// The unsigned-value conversion: accumulate the integer digits, add the
183/// rounding carry, scale by the exponent's trailing zeros, and reject anything
184/// past `max_value`. The checked u64 arithmetic stands in for wrap tests, and
185/// bounds the scaling loop the same way, since any nonzero value overflows u64
186/// within twenty scalings however large `exponent` is.
187fn create_unsigned_value(digits: &[u8], carry: bool, exponent: i64, max_value: u64) -> Option<u64> {
188    let mut value: u64 = 0;
189    for &d in digits {
190        value = value.checked_mul(10)?.checked_add(u64::from(d))?;
191    }
192    if carry {
193        value = value.checked_add(1)?;
194    }
195    for _ in 0..exponent {
196        value = value.checked_mul(10)?;
197    }
198    (value <= max_value).then_some(value)
199}

201impl Settings {
202    /// The three settings as given, unchecked: the package digits and fuzz
203    /// setters store without the check `NUMERIC` makes. The caller owns `digits
204    /// >= 1`; `fuzz >= digits` is allowed through.
205    pub fn from_parts(digits: u64, fuzz: u64, form: Form) -> Settings {
206        Settings { digits, fuzz, form }
207    }
208
209    pub fn digits(&self) -> u64 {
210        self.digits
211    }
212
213    pub fn fuzz(&self) -> u64 {
214        self.fuzz
215    }
216
217    pub fn form(&self) -> Form {
218        self.form
219    }
220
221    pub fn set_digits_str(&mut self, text: &str) -> Result<(), SettingsError> {
222        // `found` is the raw text the caller passed, unmodified -- confirmed
223        // with a lowercase variable (`x = "abc"`) that the interpreter
224        // echoes back without uppercasing it.
225        let not_whole = || SettingsError::DigitsNotWhole {
226            found: text.to_string(),
227        };
228        // The candidate is judged against the DIGITS *currently in force*:
229        // `numeric digits 1000000000` is 26.005 at DIGITS 9 and legal at
230        // DIGITS 10. There is no fixed cap short of `MAX_WHOLENUMBER`.
231        let value = unsigned_whole_number(text, self.digits).ok_or_else(not_whole)?;
232        self.set_digits_checked(value, not_whole)
233    }
234
235    /// `::OPTIONS DIGITS n`, whose operand the directive parser has already
236    /// converted with [`ARGUMENT_DIGITS`](crate::ARGUMENT_DIGITS) rather than
237    /// with the setting in force.
238    pub fn set_digits(&mut self, value: u64) -> Result<(), SettingsError> {
239        self.set_digits_checked(value, || SettingsError::DigitsNotWhole {
240            found: value.to_string(),
241        })
242    }
243
244    /// `NUMERIC DIGITS` with no operand: the package default, and it makes
245    /// the same FUZZ check the operand form makes.
246    pub fn reset_digits(&mut self, default: u64) -> Result<(), SettingsError> {
247        if default <= self.fuzz {
248            return Err(SettingsError::FuzzNotBelowDigits {
249                digits: default,
250                fuzz: self.fuzz,
251            });
252        }
253        self.digits = default;
254        Ok(())
255    }
256
257    /// The half the two share. `not_whole` is a closure so that the text form
258    /// reports the bytes the program wrote -- `numeric digits 0.0` names
259    /// `"0.0"` and not the `0` it converted to.
260    fn set_digits_checked(
261        &mut self,
262        value: u64,
263        not_whole: impl Fn() -> SettingsError,
264    ) -> Result<(), SettingsError> {
265        if value < 1 {
266            return Err(not_whole());
267        }
268        if value <= self.fuzz {
269            // The pending DIGITS/FUZZ pair, not necessarily either's stored
270            // value: the candidate here (not yet committed to `self.digits`)
271            // and the unchanged `self.fuzz` -- confirmed with `NUMERIC FUZZ
272            // 5` then `NUMERIC DIGITS 3`, which reports "(\"3\") ...
273            // (\"5\")".
274            return Err(SettingsError::FuzzNotBelowDigits {
275                digits: value,
276                fuzz: self.fuzz,
277            });
278        }
279        self.digits = value;
280        Ok(())
281    }
282
283    pub fn set_fuzz_str(&mut self, text: &str) -> Result<(), SettingsError> {
284        let not_whole = || SettingsError::FuzzNotWhole {
285            found: text.to_string(),
286        };
287        // Same conversion as DIGITS, against the same in-force DIGITS
288        // setting -- so `numeric fuzz 12345` at DIGITS 3 is 26.006, not
289        // 33.001: the conversion fails before the comparison below is ever
290        // reached. (Negatives fail the conversion too; there is no separate
291        // sign check.)
292        let value = unsigned_whole_number(text, self.digits).ok_or_else(not_whole)?;
293        self.set_fuzz_checked(value)
294    }
295
296    /// `::OPTIONS FUZZ n`, the sibling of [`Settings::set_digits`] and
297    /// converted the same way.
298    pub fn set_fuzz(&mut self, value: u64) -> Result<(), SettingsError> {
299        self.set_fuzz_checked(value)
300    }
301
302    /// `NUMERIC FUZZ` with no operand: the package default, checked against
303    /// the DIGITS in force.
304    pub fn reset_fuzz(&mut self, default: u64) -> Result<(), SettingsError> {
305        if default >= self.digits {
306            return Err(SettingsError::FuzzNotBelowDigits {
307                digits: self.digits,
308                fuzz: default,
309            });
310        }
311        self.fuzz = default;
312        Ok(())
313    }
314
315    /// The half the two share. No `not_whole` closure here: the check below
316    /// reports numbers rather than the text it was given.
317    fn set_fuzz_checked(&mut self, value: u64) -> Result<(), SettingsError> {
318        if value >= self.digits {
319            // Same substitution rule as `set_digits_str`, mirrored: the
320            // unchanged `self.digits` and the rejected candidate fuzz --
321            // confirmed with `NUMERIC DIGITS 5` then `NUMERIC FUZZ 10`,
322            // which reports "(\"5\") ... (\"10\")".
323            return Err(SettingsError::FuzzNotBelowDigits {
324                digits: self.digits,
325                fuzz: value,
326            });
327        }
328        self.fuzz = value;
329        Ok(())
330    }
331
332    pub fn set_form_str(&mut self, text: &str) -> Result<(), SettingsError> {
333        // Only the two exact uppercase spellings: the runtime VALUE path --
334        // which a `&str` setter models -- does no uppercasing, no trimming,
335        // and no abbreviation. `engineering`, `Engineering`, `ENG`, and
336        // `' ENGINEERING'` are all 25.011, confirmed against
337        // `build/bin/rexx`. The *keyword* form written in source only looks
338        // case-insensitive because the tokenizer uppercases the token
339        // before the instruction ever sees it.
340        self.form = match text {
341            "SCIENTIFIC" => Form::Scientific,
342            "ENGINEERING" => Form::Engineering,
343            // `found` substitutes the raw text too, same rule as DIGITS/FUZZ.
344            _ => {
345                return Err(SettingsError::InvalidForm {
346                    found: text.to_string(),
347                });
348            }
349        };
350        Ok(())
351    }
352
353    /// `::OPTIONS FORM ENGINEERING`/`SCIENTIFIC` and `NUMERIC FORM`'s reset
354    /// to the package default, both of which name the setting rather than
355    /// spell it.
356    pub fn set_form(&mut self, form: Form) {
357        self.form = form;
358    }
359}

361#[cfg(test)]
362mod tests {
363    use super::{Form, Settings, SettingsError};
364
365    /// The reset refuses exactly what the operand form refuses, and names the
366    /// same two values -- the default it wanted to store and the FUZZ in force.
367    #[test]
368    fn the_digits_reset_refuses_what_the_operand_form_refuses() {
369        let mut settings = Settings::default();
370        settings
371            .set_digits_str("30")
372            .expect("30 is legal at DIGITS 9");
373        settings.set_fuzz_str("20").expect("20 is below DIGITS 30");
374
375        let mut spelled_out = settings.clone();
376        assert_eq!(
377            spelled_out.set_digits_str("9"),
378            Err(SettingsError::FuzzNotBelowDigits {
379                digits: 9,
380                fuzz: 20
381            }),
382            "the operand form still refuses the value the reset stores"
383        );
384        assert_eq!(spelled_out.digits(), 30, "a refused set changes nothing");
385
386        assert_eq!(
387            settings.reset_digits(crate::DEFAULT_DIGITS),
388            Err(SettingsError::FuzzNotBelowDigits {
389                digits: 9,
390                fuzz: 20
391            }),
392            "the reset makes the same check, naming the default it could not store"
393        );
394        assert_eq!(
395            (settings.digits(), settings.fuzz()),
396            (30, 20),
397            "a refused reset changes nothing"
398        );
399    }
400
401    /// Each no-operand form writes its own setting and leaves the other two.
402    /// Measured across a sweep of starting DIGITS, FUZZ and FORM: `numeric
403    /// digits` writes only DIGITS, `numeric fuzz` only FUZZ, and `numeric
404    /// form` only FORM.
405    #[test]
406    fn each_reset_leaves_the_settings_it_does_not_own() {
407        let start = || {
408            let mut settings = Settings::default();
409            settings
410                .set_digits_str("30")
411                .expect("30 is legal at DIGITS 9");
412            settings.set_fuzz_str("20").expect("20 is below DIGITS 30");
413            settings
414                .set_form_str("ENGINEERING")
415                .expect("a spelling the setter accepts");
416            settings
417        };
418
419        // DIGITS' reset cannot run from this state -- the default sits under
420        // the FUZZ in force -- so the pair it would have to leave alone is
421        // measured from a state where it succeeds.
422        let mut digits_reset = start();
423        digits_reset
424            .set_fuzz_str("5")
425            .expect("5 is below DIGITS 30");
426        digits_reset
427            .reset_digits(crate::DEFAULT_DIGITS)
428            .expect("9 exceeds a FUZZ of 5");
429        assert_eq!(
430            (
431                digits_reset.digits(),
432                digits_reset.fuzz(),
433                digits_reset.form()
434            ),
435            (9, 5, Form::Engineering)
436        );
437
438        let mut fuzz_reset = start();
439        fuzz_reset.reset_fuzz(0).expect("0 is below any DIGITS");
440        assert_eq!(
441            (fuzz_reset.digits(), fuzz_reset.fuzz(), fuzz_reset.form()),
442            (30, 0, Form::Engineering)
443        );
444
445        let mut form_reset = start();
446        form_reset
447            .set_form_str("SCIENTIFIC")
448            .expect("a spelling the setter accepts");
449        assert_eq!(
450            (form_reset.digits(), form_reset.fuzz(), form_reset.form()),
451            (30, 20, Form::Scientific)
452        );
453    }
