1//! Stopping a daemon on `SIGTERM`, `SIGINT` or `SIGHUP`.
2// The `unsafe` calls are the C library's signal-mask and wait functions, which no safe crate of
3// the workspace offers.
4#![allow(unsafe_code)]
5
6use std::io;
7
8/// The signals that stop a daemon.
9const STOPPING: [libc::c_int; 3] = [libc::SIGTERM, libc::SIGINT, libc::SIGHUP];
10
11/// Makes the first of the stopping signals call `stop`, on a thread of its own, whatever the
12/// signals' disposition and mask were when this process started: a client that starts the daemon
13/// may hold them blocked or ignored, and both survive an `exec`.
14///
15/// The signals are blocked in the calling thread, and so in every thread it starts afterwards,
16/// which is why only one thread waits for them. Call it while the process has one thread.
17pub(crate) fn stop_on_signal(stop: impl FnOnce() + Send + 'static) -> io::Result<()> {
18    // SAFETY: the set is initialised by `sigemptyset` before it is added to, through a pointer
19    // to a live local; `signal` with `SIG_DFL` installs no code of ours; the mask pointers are
20    // to live values or null.
21    let set = unsafe {
22        let mut set: libc::sigset_t = std::mem::zeroed();
23        libc::sigemptyset(&raw mut set);
24        for signal in STOPPING {
25            libc::sigaddset(&raw mut set, signal);
26            libc::signal(signal, libc::SIG_DFL);
27        }
28        let failed = libc::pthread_sigmask(libc::SIG_BLOCK, &raw const set, std::ptr::null_mut());
29        if failed != 0 {
30            return Err(io::Error::from_raw_os_error(failed));
31        }
32        set
33    };
34    std::thread::Builder::new()
35        .name("rxapi-signal".to_owned())
36        .spawn(move || {
37            let mut received: libc::c_int = 0;
38            // SAFETY: `set` is an initialised set and `received` a live local.
39            if unsafe { libc::sigwait(&raw const set, &raw mut received) } == 0 {
40                stop();
41            }
42        })?;
43    Ok(())
44}
