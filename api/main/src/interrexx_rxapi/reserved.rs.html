1//! Descriptors the interpreter owns for its own use, kept out of a program's reach.
2//!
3//! A program's `SockClose(n)` or `SysClose(n)` names a descriptor by number, and the numbers
4//! the process hands out first are the lowest free ones. So that a program closing small
5//! numbers cannot close the interpreter's own descriptors (the timer's wake socket the signal
6//! handler writes to, its deadline timer, the rxapi connection, a running command's pidfd),
7//! each is moved at creation to the lowest free number at or above [`floor`] and recorded
8//! here; `rxsock` and `rxunixsys` answer `EBADF` for a recorded number, as for a closed one.
9//!
10//! The floor is 1000, lowered to 24 below the `RLIMIT_NOFILE` soft limit where that is
11//! smaller, so the interpreter's descriptors sit above the numbers a program normally reaches
12//! and below the limit. Where no number at or above it is free, the descriptor keeps its number
13//! and is still recorded.
14
15use std::ops::Deref;
16use std::os::fd::{AsFd, AsRawFd, BorrowedFd, OwnedFd, RawFd};
17use std::sync::{Mutex, PoisonError};
18
19/// The highest floor; see the module documentation.
20pub const FLOOR: RawFd = 1000;
21
22/// A record of reserved numbers, in no order.
23struct Registry(Mutex<Vec<RawFd>>);
24
25impl Registry {
26    const fn new() -> Registry {
27        Registry(Mutex::new(Vec::new()))
28    }
29
30    fn numbers(&self) -> std::sync::MutexGuard<'_, Vec<RawFd>> {
31        self.0.lock().unwrap_or_else(PoisonError::into_inner)
32    }
33
34    fn contains(&self, fd: RawFd) -> bool {
35        self.numbers().contains(&fd)
36    }
37}
38
39/// The interpreter's numbers.
40static RESERVED: Registry = Registry::new();
41
42/// The number the interpreter's descriptors are moved to or above.
43pub fn floor() -> RawFd {
44    let soft = rustix::process::getrlimit(rustix::process::Resource::Nofile)
45        .current
46        .and_then(|soft| RawFd::try_from(soft).ok())
47        .unwrap_or(RawFd::MAX);
48    FLOOR.min(soft.saturating_sub(24)).max(3)
49}
50
51/// Whether `fd` is one of the interpreter's own descriptors.
52pub fn is_reserved(fd: RawFd) -> bool {
53    RESERVED.contains(fd)
54}
55
56/// An interpreter-owned descriptor, recorded while it is open.
57pub struct Reserved<T: AsRawFd> {
58    inner: Option<T>,
59    fd: RawFd,
60    /// Where its number is recorded.
61    registry: &'static Registry,
62}
63
64/// Moves `object`'s descriptor to [`floor`] or above, close-on-exec, and records it.
65pub fn reserve<T>(object: T) -> Reserved<T>
66where
67    T: AsRawFd + From<OwnedFd> + Into<OwnedFd>,
68{
69    reserve_in(&RESERVED, object)
70}
71
72/// [`reserve`], recording the number in `registry`.
73fn reserve_in<T>(registry: &'static Registry, object: T) -> Reserved<T>
74where
75    T: AsRawFd + From<OwnedFd> + Into<OwnedFd>,
76{
77    let owned: OwnedFd = object.into();
78    let owned = match rustix::io::fcntl_dupfd_cloexec(&owned, floor()) {
79        Ok(lifted) => lifted,
80        Err(_) => owned,
81    };
82    let fd = owned.as_raw_fd();
83    registry.numbers().push(fd);
84    Reserved {
85        inner: Some(T::from(owned)),
86        fd,
87        registry,
88    }
89}
90
91impl<T: AsRawFd> Deref for Reserved<T> {
92    type Target = T;
93
94    fn deref(&self) -> &T {
95        self.inner.as_ref().expect("held until drop")
96    }
97}
98
99impl<T: AsRawFd + AsFd> AsFd for Reserved<T> {
100    fn as_fd(&self) -> BorrowedFd<'_> {
101        (**self).as_fd()
102    }
103}
104
105impl<T: AsRawFd> AsRawFd for Reserved<T> {
106    fn as_raw_fd(&self) -> RawFd {
107        self.fd
108    }
109}
110
111impl<T: AsRawFd> Drop for Reserved<T> {
112    /// Closes the descriptor, then forgets its number: the other order would let a program
113    /// close it in between.
114    fn drop(&mut self) {
115        drop(self.inner.take());
116        let mut numbers = self.registry.numbers();
117        if let Some(at) = numbers.iter().position(|held| *held == self.fd) {
118            numbers.swap_remove(at);
119        }
120    }
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126
127    /// Recorded in a registry of its own: the process's is shared with every
128    /// test that connects, and one running beside this can be handed the number
129    /// this closes and record it before the last assertion reads it.
130    #[test]
131    fn a_reserved_descriptor_sits_at_the_floor_and_is_forgotten_when_closed() {
132        static OWN: Registry = Registry::new();
133        let (one, _other) = std::os::unix::net::UnixStream::pair().expect("a pair");
134        let low = one.as_raw_fd();
135        let held = reserve_in(&OWN, one);
136        let fd = held.as_raw_fd();
137        assert!(fd >= floor(), "{fd} below {}", floor());
138        assert!(OWN.contains(fd));
139        assert!(!OWN.contains(low));
140        drop(held);
141        assert!(!OWN.contains(fd));
142    }
143
144    /// [`reserve`] records in the registry [`is_reserved`] reads, for as long as
145    /// the descriptor is held.
146    #[test]
147    fn reserve_records_where_is_reserved_reads() {
148        let (one, _other) = std::os::unix::net::UnixStream::pair().expect("a pair");
149        let held = reserve(one);
150        assert!(is_reserved(held.as_raw_fd()));
151        assert!(std::ptr::eq(held.registry, &RESERVED));
152    }
153}
