1//! Closing the descriptors a started daemon inherited from its starter.
2// The one `unsafe` function, at daemon start-up.
3#![allow(unsafe_code)]
4
5use std::os::fd::{AsRawFd, BorrowedFd, RawFd};
6
7use rustix::fs::{Dir, Mode, OFlags};
8
9/// Closes every descriptor of this process above standard error except `keep`, so a daemon a
10/// client started holds none of that client's pipes or sockets open. Call it while the process
11/// has one thread and no descriptor besides `keep` and the standard three that anything owns.
12pub(crate) fn close_inherited(keep: &[BorrowedFd<'_>]) {
13    // Linux lists a process's descriptors under `/proc`, Apple's systems
14    // under `/dev/fd`. (FreeBSD's `/dev/fd` lists only 0 to 2 unless fdescfs
15    // is mounted, so a BSD port needs a listing of its own.)
16    let listed = if cfg!(target_os = "linux") {
17        "/proc/self/fd"
18    } else {
19        "/dev/fd"
20    };
21    let Ok(listing) = rustix::fs::open(
22        listed,
23        OFlags::RDONLY | OFlags::DIRECTORY | OFlags::CLOEXEC,
24        Mode::empty(),
25    ) else {
26        return;
27    };
28    // The listing names its own descriptor, which it closes itself.
29    let mut skip: Vec<RawFd> = keep.iter().map(AsRawFd::as_raw_fd).collect();
30    skip.push(listing.as_raw_fd());
31    let Ok(dir) = Dir::new(listing) else {
32        return;
33    };
34    // Listed in full, and the listing closed, before any is closed.
35    let open: Vec<RawFd> = dir
36        .filter_map(|entry| entry.ok()?.file_name().to_str().ok()?.parse().ok())
37        .collect();
38    for fd in open.into_iter().filter(|fd| *fd > 2 && !skip.contains(fd)) {
39        // SAFETY: `fd` was open at the listing, and the caller guarantees that no object of
40        // this process owns it, it being neither a standard stream nor in `keep` nor the
41        // listing's own, and that no other thread can close it or open another descriptor
42        // under its number before this close. So it is still open and closing it invalidates
43        // nothing any code will use.
44        unsafe { rustix::io::close(fd) };
45    }
46}
