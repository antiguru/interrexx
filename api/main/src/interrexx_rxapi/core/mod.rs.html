13/// Who a request came from and where its reply goes.
14///
15/// `connection` is the identity: one for a client's whole connection, and never reused. `call`
16/// tells apart the concurrent calls of one connection, so each parked pull has its own reply.
17#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
18pub struct ClientId {
19    pub connection: u64,
20    pub call: u64,
21}
22
23impl ClientId {
24    /// The first call of `connection`.
25    pub fn connection(connection: u64) -> ClientId {
26        ClientId {
27            connection,
28            call: 0,
29        }
30    }
31
32    /// Another call on the same connection.
33    pub fn call(self, call: u64) -> ClientId {
34        ClientId { call, ..self }
35    }
36}

38/// What `Core::handle` did with a request.
39#[derive(Clone, Debug, PartialEq, Eq)]
40pub enum Outcome {
41    Reply(Reply),
42    /// The reply will come out of `Core::take_ready` after a later call.
43    Parked,
44}
45
46/// All state the service holds.
47#[derive(Default)]
48pub struct Core {
49    queues: Queues,
50    registries: Registries,
51    macros: MacroSpace,
52}
53
54impl Core {
55    pub fn new() -> Core {
56        Core::default()
57    }
58
59    /// Runs one request to its reply, or parks it.
60    pub fn handle(&mut self, client: ClientId, request: Request) -> Outcome {
61        let reply = match request {
62            Request::CreateSessionQueue { session } => self.queues.attach(client, session),
63            Request::DeleteSessionQueue { .. } => Reply::Status(crate::message::code::OK),
64            Request::CreateQueue {
65                session,
66                requested,
67                size,
68            } => self.queues.create(session, requested, size),
69            Request::OpenQueue { name } => self.queues.open(name),
70            Request::QueueExists { name } => self.queues.exists(name),
71            Request::DeleteQueue { name } => self.queues.delete(name),
72            Request::ClearQueue { session, name } => self.queues.clear(client, session, name),
73            Request::QueryQueue { session, name } => self.queues.query(client, session, name),
74            Request::AddQueue {
75                session,
76                name,
77                data,
78                flag,
79                time,
80            } => self.queues.add(client, session, name, data, flag, time),
81            Request::PullQueue {
82                session,
83                name,
84                wait,
85            } => return self.queues.pull(client, session, name, wait),
86            Request::CancelPull { call } => self.queues.cancel(client.call(call)),
87            Request::Register {
88                registry,
89                name,
90                library,
91                procedure,
92                user_area,
93                owner_only,
94            } => self
95                .registries
96                .table(registry)
97                .register(client, name, library, procedure, user_area, owner_only),
98            Request::QueryRegistration {
99                registry,
100                name,
101                library,
102            } => self.registries.table(registry).query(name, library),
103            Request::Deregister {
104                registry,
105                name,
106                library,
107            } => self
108                .registries
109                .table(registry)
110                .deregister(client, name, library),
111            Request::AddMacro {
112                name,
113                image,
114                position,
115            } => self.macros.add(name, image, position),
116            Request::DropMacro { name } => self.macros.drop_macro(name),
117            Request::ClearMacroSpace => self.macros.clear(),
118            Request::QueryMacro { name } => self.macros.query(name),
119            Request::ReorderMacro { name, position } => self.macros.reorder(name, position),
120            Request::ResolveMacro { name } => self.macros.resolve(name),
121            Request::SaveMacros { names } => self.macros.save(names),
122            Request::LoadMacros { names, macros } => self.macros.load(names, macros),
123        };
124        Outcome::Reply(reply)
125    }
126
127    /// Replies to parked requests that have become ready since the last call.
128    pub fn take_ready(&mut self) -> Vec<(ClientId, Reply)> {
129        self.queues.take_ready()
130    }
131
132    /// A pull of `name` took `item`, and its client was gone. The item goes to the next waiter, or
133    /// back to the head of the queue.
134    pub fn undelivered(&mut self, session: u64, name: Option<Vec<u8>>, item: Item) {
135        self.queues.restore(session, name, item);
136    }
137
138    /// `client`'s connection closed: its parked pulls end and its session queues lose a user. A
139    /// session queue goes once it has no user, whatever it holds.
140    pub fn client_closed(&mut self, client: ClientId) {
141        self.queues.connection_closed(client.connection);
142    }
143
144    /// No named queue, registration or macro.
145    pub fn is_empty(&self) -> bool {
146        self.queues.hold_nothing() && self.registries.is_empty() && self.macros.is_empty()
147    }
148}

150/// The key a caseless name compares under.
151fn fold(name: &[u8]) -> Vec<u8> {
152    name.to_ascii_uppercase()
153}

155/// `name` cut to the length the service stores.
156fn truncated(mut name: Vec<u8>) -> Vec<u8> {
157    name.truncate(crate::message::MAX_NAME);
