1//! The per-user table of compiled routines.
2
3use std::collections::{BTreeMap, HashSet};
4
5use super::{fold, truncated};
6use crate::message::code::*;
7use crate::message::{Macro, Reply, macro_position};
8
9/// The image bytes a save may reach by listing a name repeatedly. A short request can name one
10/// large image many times, and the service would hold a copy for each.
11const MAX_SAVE_REPEAT: usize = 256 << 20;
12
13/// Macros by folded name. `add` and `reorder` take only a position before or
14/// after (Deviation 185).
15#[derive(Default)]
16pub(super) struct MacroSpace {
17    macros: BTreeMap<Vec<u8>, Macro>,
18}
19
20fn key(name: &[u8]) -> Vec<u8> {
21    fold(&name[..name.len().min(crate::message::MAX_NAME)])
22}
23
24impl MacroSpace {
25    /// Adds or replaces a macro, or answers `RXMACRO_INVALID_POSITION` and
26    /// changes nothing for a position that is neither before nor after.
27    pub(super) fn add(&mut self, name: Vec<u8>, image: Vec<u8>, position: u64) -> Reply {
28        if !macro_position::is_valid(position) {
29            return Reply::Status(MACRO_INVALID_POSITION);
30        }
31        let name = truncated(name);
32        self.macros
33            .entry(fold(&name))
34            .and_modify(|existing| {
35                existing.image = image.clone();
36                existing.position = position;
37            })
38            .or_insert(Macro {
39                name,
40                image,
41                position,
42            });
43        Reply::Status(OK)
44    }
45
46    pub(super) fn drop_macro(&mut self, name: Vec<u8>) -> Reply {
47        Reply::Status(match self.macros.remove(&key(&name)) {
48            Some(_) => OK,
49            None => MACRO_NOT_FOUND,
50        })
51    }
52
53    pub(super) fn clear(&mut self) -> Reply {
54        self.macros.clear();
55        Reply::Status(OK)
56    }
57
58    pub(super) fn query(&self, name: Vec<u8>) -> Reply {
59        match self.macros.get(&key(&name)) {
60            Some(found) => Reply::MacroPosition {
61                status: OK,
62                position: found.position,
63            },
64            None => Reply::MacroPosition {
65                status: MACRO_NOT_FOUND,
66                position: 0,
67            },
68        }
69    }
70
71    /// Moves a macro, or answers `RXMACRO_INVALID_POSITION` and changes
72    /// nothing for a position that is neither before nor after, whether or not
73    /// the macro is there.
74    pub(super) fn reorder(&mut self, name: Vec<u8>, position: u64) -> Reply {
75        if !macro_position::is_valid(position) {
76            return Reply::Status(MACRO_INVALID_POSITION);
77        }
78        Reply::Status(match self.macros.get_mut(&key(&name)) {
79            Some(found) => {
80                found.position = position;
81                OK
82            }
83            None => MACRO_NOT_FOUND,
84        })
85    }
86
87    pub(super) fn resolve(&self, name: Vec<u8>) -> Reply {
88        match self.macros.get(&key(&name)) {
89            Some(found) => Reply::MacroImage {
90                status: OK,
91                image: Some(found.image.clone()),
92            },
93            None => Reply::MacroImage {
94                status: MACRO_NOT_FOUND,
95                image: None,
96            },
97        }
98    }
99
100    pub(super) fn save(&self, names: Option<Vec<Vec<u8>>>) -> Reply {
101        let refused = |status| Reply::Macros {
102            status,
103            macros: Vec::new(),
104        };
105        let macros = match names {
106            None if self.macros.is_empty() => return refused(MACRO_FILE_ERROR),
107            None => self.macros.values().cloned().collect(),
108            Some(names) => {
109                let found: Option<Vec<&Macro>> = names
110                    .iter()
111                    .map(|name| self.macros.get(&key(name)))
112                    .collect();
113                let Some(found) = found else {
114                    return refused(MACRO_NOT_FOUND);
115                };
116                let wanted = found.iter().map(|entry| entry.image.len()).sum::<usize>();
117                let held = self.macros.values().map(|entry| entry.image.len()).sum();
118                if wanted > usize::max(held, MAX_SAVE_REPEAT) {
119                    return refused(MACRO_NO_STORAGE);
120                }
121                found.into_iter().cloned().collect()
122            }
123        };
124        Reply::Macros { status: OK, macros }
125    }
126
127    pub(super) fn load(&mut self, names: Option<Vec<Vec<u8>>>, macros: Vec<Macro>) -> Reply {
128        let wanted: Option<HashSet<Vec<u8>>> =
129            names.map(|names| names.iter().map(|name| key(name)).collect());
130        for loaded in macros {
131            let selected = wanted
132                .as_ref()
133                .is_none_or(|wanted| wanted.contains(&key(&loaded.name)));
134            // A macro whose position is refused ends the load as a cut file
135            // does: the macros before it stay loaded.
136            if selected && self.add(loaded.name, loaded.image, loaded.position) != Reply::Status(OK)
137            {
138                return Reply::Status(MACRO_FILE_ERROR);
139            }
140        }
141        Reply::Status(OK)
142    }
143
144    pub(super) fn is_empty(&self) -> bool {
145        self.macros.is_empty()
146    }
147}
