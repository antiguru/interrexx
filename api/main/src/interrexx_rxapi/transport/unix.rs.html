42/// The service directory for this user: `REXX_API_DIR`, else `$XDG_RUNTIME_DIR/interrexx`, else a
43/// directory of this user's under the temp directory. A candidate whose socket path the platform
44/// cannot bind is passed over, and a passed-over `REXX_API_DIR` is reported once on standard
45/// error.
46pub fn service_dir() -> PathBuf {
47    let (dir, skipped) = choose_dir(
48        std::env::var_os("REXX_API_DIR"),
49        std::env::var_os("XDG_RUNTIME_DIR"),
50        &std::env::temp_dir(),
51        rustix::process::getuid().as_raw(),
52    );
53    if let Some(skipped) = skipped {
54        static WARNED: std::sync::Once = std::sync::Once::new();
55        WARNED.call_once(|| {
56            use std::io::Write;
57            let _ = writeln!(
58                io::stderr(),
59                "rxapi: REXX_API_DIR {} is not used: its socket path {} is too long",
60                skipped.display(),
61                socket_path(&resolved(&skipped), PROTOCOL_VERSION).display()
62            );
63        });
64    }
65    dir
66}

68/// Whether a socket in `dir` has a path the service can bind and a client can connect to. The
69/// check is the one `UnixListener::bind` and `UnixStream::connect` make, which leaves room for
70/// the terminating NUL in `sun_path`: a path that fills `sun_path` exactly is refused there, so
71/// it is refused here.
72///
73/// The path measured is the one [`prepare_dir`] answers, with symbolic links resolved: on macOS
74/// `/var`, under which the temp directory lies, is a link to `/private/var`, eight bytes longer.
75fn socket_fits(dir: &Path) -> bool {
76    let socket = socket_path(&resolved(dir), PROTOCOL_VERSION);
77    std::os::unix::net::SocketAddr::from_pathname(&socket).is_ok()
78}

80/// `dir` absolute, as `fs::canonicalize` will answer it once the rest exists: each component
81/// that exists with its links resolved, and a `..` after one that does not yet taken as its
82/// parent, as creating the directories makes it.
83fn resolved(dir: &Path) -> PathBuf {
84    let absolute = std::path::absolute(dir).unwrap_or_else(|_| dir.to_path_buf());
85    let mut path = PathBuf::new();
86    for component in absolute.components() {
87        match component {
88            std::path::Component::CurDir => {}
89            // Every component so far is resolved or does not exist, so none is a link, and `..`
90            // is its parent.
91            std::path::Component::ParentDir => {
92                path.pop();
93            }
94            other => {
95                path.push(other);
96                if let Ok(canonical) = fs::canonicalize(&path) {
97                    path = canonical;
98                }
99            }
100        }
101    }
102    path
103}

105/// The directory, and `REXX_API_DIR`'s value where it was set and passed over.
106fn choose_dir(
107    api: Option<OsString>,
108    xdg: Option<OsString>,
109    temp: &Path,
110    uid: u32,
111) -> (PathBuf, Option<PathBuf>) {
112    let set = |value: Option<OsString>| value.filter(|value| !value.is_empty()).map(PathBuf::from);
113    let api = set(api);
114    let fallback = temp.join(format!("interrexx-{uid}"));
115    let chosen = api
116        .iter()
117        .cloned()
118        .chain(set(xdg).map(|xdg| xdg.join("interrexx")))
119        .find(|dir| socket_fits(dir))
120        .unwrap_or(fallback);
121    let skipped = api.filter(|api| *api != chosen);
122    (chosen, skipped)
123}

125/// Creates `dir` if needed and insists it is a `0700` directory of this user, answering its
126/// absolute path.
127pub fn prepare_dir(dir: &Path) -> io::Result<PathBuf> {
128    fs::DirBuilder::new()
129        .recursive(true)
130        .mode(0o700)
131        .create(dir)?;
132    let metadata = fs::symlink_metadata(dir)?;
133    let ours = metadata.uid() == rustix::process::getuid().as_raw();
134    if !metadata.is_dir() || !ours || metadata.mode() & 0o077 != 0 {
135        return Err(io::Error::new(
136            io::ErrorKind::PermissionDenied,
137            format!("{} is not private to this user", dir.display()),
138        ));
139    }
140    fs::canonicalize(dir)
141}

186/// The first `rxapi` that can run, beside this executable, else on `PATH`. The answer is absolute
187/// because the daemon starts in `/`.
188fn find_daemon() -> Result<PathBuf, TransportError> {
189    let beside = std::env::current_exe()
190        .ok()
191        .and_then(|exe| Some(exe.parent()?.join("rxapi")));
192    let on_path = std::env::var_os("PATH").unwrap_or_default();
193    let here = std::env::current_dir().map_err(|error| unavailable(&error))?;
194    beside
195        .into_iter()
196        .chain(std::env::split_paths(&on_path).map(|dir| here.join(dir).join("rxapi")))
197        .find(|candidate| candidate.is_file() && access(candidate, Access::EXEC_OK).is_ok())
198        .ok_or_else(|| TransportError::Unavailable("no rxapi executable found".to_owned()))
199}

201/// Set in the environment of an `rxapi` a client starts: that process starts the daemon as a
202/// child of its own and exits at once, so the daemon is no child of the client, whose program
203/// may wait for any child of its process (`SysWait`).
204pub const DETACH_VARIABLE: &str = "REXX_API_DETACH";
205
206fn start_daemon(daemon: &Path, dir: &Path) -> Result<(), TransportError> {
207    let mut command = Command::new(daemon);
208    command
209        .env("REXX_API_DIR", dir)
210        .env(DETACH_VARIABLE, "1")
211        .current_dir("/")
212        .stdin(Stdio::null())
213        .stdout(Stdio::null())
214        .stderr(Stdio::null());
215    // Started under the lock and waited for outside it, so that a client
216    // thread does not hold up descriptor creation while the service starts.
217    let status = crate::inheritance::spawning(|| command.spawn())
218        .and_then(|mut child| child.wait())
219        .map_err(|error| unavailable(&error))?;
220    if status.success() {
221        Ok(())
222    } else {
223        Err(TransportError::Unavailable(format!(
224            "rxapi could not start the service: {status}"
225        )))
226    }
227}

229/// In an `rxapi` a client started, starts the daemon from this executable with the arguments
230/// given and without [`DETACH_VARIABLE`], answering whether it did; the caller exits at once.
231pub fn detach() -> Option<io::Result<()>> {
232    std::env::var_os(DETACH_VARIABLE)?;
233    let spawned = std::env::current_exe().and_then(|exe| {
234        Command::new(exe)
235            .args(std::env::args_os().skip(1))
236            .env_remove(DETACH_VARIABLE)
237            .current_dir("/")
238            .stdin(Stdio::null())
239            .stdout(Stdio::null())
240            .stderr(Stdio::null())
241            .spawn()
242    });
243    Some(spawned.map(drop))
244}

250fn private(dir: &Path) -> Result<PathBuf, TransportError> {
251    prepare_dir(dir).map_err(|error| {
252        // A directory that is there and refused for its owner or mode, as against one that
253        // could not be made.
254        if error.kind() == io::ErrorKind::PermissionDenied && fs::symlink_metadata(dir).is_ok() {
255            TransportError::NotPrivate {
256                dir: dir.display().to_string(),
257            }
258        } else {
259            unavailable(&error)
260        }
261    })
262}

274/// Writes all of `bytes` without raising `SIGPIPE` in the embedding process.
275fn send_all(stream: &UnixStream, mut bytes: &[u8]) -> io::Result<()> {
276    while !bytes.is_empty() {
277        match send(stream, bytes, SEND_FLAGS) {
278            Ok(sent) => bytes = &bytes[sent..],
279            Err(rustix::io::Errno::INTR) => {}
280            Err(error) => return Err(error.into()),
281        }
282    }
283    Ok(())
284}

286/// Calls in flight, by call number.
287#[derive(Default)]
288struct Calls {
289    next: u64,
290    waiting: HashMap<u64, mpsc::Sender<Reply>>,
291    closed: bool,
292}
293
294type SharedCalls = Arc<Mutex<Calls>>;
295
296fn lock_calls(calls: &Mutex<Calls>) -> MutexGuard<'_, Calls> {
297    calls.lock().expect("the critical sections cannot panic")
298}
299
300/// Delivers replies until the connection ends, then fails every call in flight.
301fn read_replies(stream: Reserved<UnixStream>, calls: SharedCalls) {
302    while let Ok(Some((call, body))) = wire::read_frame(&mut &*stream) {
303        let Ok(reply) = wire::decode_reply(&body) else {
304            break;
305        };
306        if let Some(waiter) = lock_calls(&calls).waiting.remove(&call) {
307            let _ = waiter.send(reply);
308        }
309    }
310    let mut calls = lock_calls(&calls);
311    calls.closed = true;
312    calls.waiting.clear();
313}

315/// Why a connect attempt failed.
316enum ConnectFailure {
317    /// Nothing answered, or what did went away.
318    Absent(io::Error),
319    Refused {
320        theirs: u32,
321    },
322    /// Another user's process answered.
323    Foreign {
324        uid: u32,
325    },
326}
327
328impl ConnectFailure {
329    fn into_error(self, version: u32) -> TransportError {
330        match self {
331            ConnectFailure::Absent(error) => unavailable(&error),
332            ConnectFailure::Refused { theirs } => TransportError::Refused {
333                ours: version,
334                theirs,
335            },
336            ConnectFailure::Foreign { uid } => TransportError::Foreign { uid },
337        }
338    }
339}

351impl Unix {
352    /// Connects to the service in `dir`, starting it if nothing answers.
353    pub fn connect_or_start(dir: &Path) -> Result<Unix, TransportError> {
354        let dir = private(dir)?;
355        match Unix::attempt(&dir, PROTOCOL_VERSION, own_uid()) {
356            Err(ConnectFailure::Absent(_)) => {
357                Unix::start_and_connect(&dir, &find_daemon()?, own_uid())
358            }
359            result => result.map_err(|failure| failure.into_error(PROTOCOL_VERSION)),
360        }
361    }
362
363    /// As `connect_or_start`, with the executable to start.
364    pub fn connect_or_start_with(dir: &Path, daemon: &Path) -> Result<Unix, TransportError> {
365        Unix::start_and_connect(&private(dir)?, daemon, own_uid())
366    }
367
368    fn start_and_connect(dir: &Path, daemon: &Path, uid: u32) -> Result<Unix, TransportError> {
369        let deadline = now() + START_TIMEOUT;
370        let mut next_start = now();
371        // A start that fails is retried until the deadline, as one whose service never answers
372        // is (Resolution 9); the error then names the failed start where there was one.
373        let mut failed_start = None;
374        loop {
375            let error = match Unix::attempt(dir, PROTOCOL_VERSION, uid) {
376                Ok(connected) => return Ok(connected),
377                Err(ConnectFailure::Absent(error)) => error,
378                Err(failure) => return Err(failure.into_error(PROTOCOL_VERSION)),
379            };
380            let time = now();
381            if time >= deadline {
382                return Err(failed_start.unwrap_or_else(|| unavailable(&error)));
383            }
384            if time >= next_start {
385                failed_start = start_daemon(daemon, dir).err();
386                next_start = time + RESPAWN_INTERVAL;
387            }
388            pause();
389        }
390    }
391
392    /// Connects to a running service speaking `version`, starting nothing.
393    pub fn connect_version(dir: &Path, version: u32) -> Result<Unix, TransportError> {
394        Unix::attempt(&private(dir)?, version, own_uid())
395            .map_err(|failure| failure.into_error(version))
396    }

398    /// Connects to the service in the private `dir`, which must run as `uid`.
399    fn attempt(dir: &Path, version: u32, uid: u32) -> Result<Unix, ConnectFailure> {
400        // Made close-on-exec apart from the call where the host has no
401        // `SOCK_CLOEXEC`, so under the inheritance lock.
402        let stream = reserve(
403            crate::inheritance::creating(|| UnixStream::connect(socket_path(dir, version)))
404                .map_err(ConnectFailure::Absent)?,
405        );
406        no_sigpipe(&stream).map_err(ConnectFailure::Absent)?;
407        let peer = peer_uid(&stream).map_err(ConnectFailure::Absent)?;
408        if peer != uid {
409            return Err(ConnectFailure::Foreign { uid: peer });
410        }
411        handshake(&stream, version)?;
412        let reader_stream = reserve(stream.try_clone().map_err(ConnectFailure::Absent)?);
413        let calls = SharedCalls::default();
414        let reader = {
415            let calls = Arc::clone(&calls);
416            std::thread::Builder::new()
417                .name("rexx-api".to_owned())
418                .spawn(move || read_replies(reader_stream, calls))
419                .map_err(ConnectFailure::Absent)?
420        };
421        Ok(Unix {
422            stream: Mutex::new(stream),
423            calls,
424            reader: Some(reader),
425        })
426    }

428    /// Stops the service in `dir`, answering whether one was running. Returns once it closed
429    /// this connection, by which time it has removed its socket.
430    pub fn stop(dir: &Path) -> Result<bool, TransportError> {
431        Unix::stop_version(dir, PROTOCOL_VERSION)
432    }

434    /// As `stop`, for the service speaking `version`.
435    pub fn stop_version(dir: &Path, version: u32) -> Result<bool, TransportError> {
436        Unix::stop_as(&private(dir)?, version, own_uid())
437    }

439    fn stop_as(dir: &Path, version: u32, uid: u32) -> Result<bool, TransportError> {
440        let mut client = match Unix::attempt(dir, version, uid) {
441            Ok(client) => client,
442            Err(ConnectFailure::Absent(_)) => return Ok(false),
443            Err(failure) => return Err(failure.into_error(version)),
444        };
445        client
446            .send(0, &wire::encode_stop())
447            .map_err(|error| unavailable(&error))?;
448        if let Some(reader) = client.reader.take() {
449            let _ = reader.join();
450        }
451        Ok(true)
452    }
453
454    fn send(&self, call: u64, body: &[u8]) -> io::Result<()> {
455        let stream = self.stream.lock().expect("sends cannot panic");
456        send_all(&stream, &wire::frame(call, body))
457    }
458}

460fn handshake(stream: &UnixStream, version: u32) -> Result<(), ConnectFailure> {
461    let absent = ConnectFailure::Absent;
462    // The timeout is set before the `hello` goes out. A service of another version answers and
463    // closes at once, and macOS then fails `setsockopt` on the closed socket with `EINVAL`, which
464    // would turn a refusal into an absent service.
465    stream
466        .set_read_timeout(Some(HANDSHAKE_TIMEOUT))
467        .map_err(absent)?;
468    send_all(stream, &wire::hello(version)).map_err(absent)?;
469    let mut welcome = [0; wire::WELCOME_LENGTH];
470    (&*stream).read_exact(&mut welcome).map_err(absent)?;
471    match wire::read_welcome(&welcome) {
472        // Only an accepting service keeps the connection open, so only here can the timeout still
473        // be cleared.
474        (true, _) => stream.set_read_timeout(None).map_err(absent),
475        (false, theirs) => Err(ConnectFailure::Refused { theirs }),
476    }
477}

479impl Transport for Unix {
480    fn call_cancellable(
481        &self,
482        request: Request,
483        token: &CancelToken,
484    ) -> Result<Reply, TransportError> {
485        let lost = || TransportError::Unavailable("the connection is closed".to_owned());
486        let mut started = token.started();
487        if started.cancelled {
488            return Ok(Reply::Cancelled);
489        }
490        let (sender, receiver) = mpsc::channel();
491        let call = {
492            let mut calls = lock_calls(&self.calls);
493            if calls.closed {
494                return Err(lost());
495            }
496            calls.next += 1;
497            let call = calls.next;
498            calls.waiting.insert(call, sender);
499            call
500        };
501        started.call = Some(call);
502        if self.send(call, &wire::encode_request(&request)).is_err() {
503            lock_calls(&self.calls).waiting.remove(&call);
504            return Err(lost());
505        }
506        drop(started);
507        receiver.recv().map_err(|_| lost())
508    }
509}

511impl Drop for Unix {
512    fn drop(&mut self) {
513        if let Ok(stream) = self.stream.get_mut() {
514            let _ = stream.shutdown(std::net::Shutdown::Both);
515        }
516        if let Some(reader) = self.reader.take() {
517            let _ = reader.join();
518        }
519    }

594    /// [`prepare_dir`] puts the socket: a short link to a long directory does not fit, and a
595    /// directory below it that does not exist yet is measured there too.
596    #[test]
597    fn a_directory_behind_a_link_is_measured_where_it_leads() {
598        let root = std::env::temp_dir().join(format!("rxapi-link-{}", std::process::id()));
599        let long = root.join("l".repeat(100));
600        fs::create_dir_all(&long).expect("a long directory");
601        let link = root.join("s");
602        let _ = fs::remove_file(&link);
603        std::os::unix::fs::symlink(&long, &link).expect("a link");
604        let through = link.join("api");
605        assert!(
606            std::os::unix::net::SocketAddr::from_pathname(socket_path(&through, PROTOCOL_VERSION))
607                .is_ok(),
608            "the path through the link is short enough"
609        );
610        assert!(!socket_fits(&through), "the path it leads to is not");
611        assert_eq!(
612            resolved(&through),
613            fs::canonicalize(&long).expect("the target").join("api")
614        );
615        let _ = fs::remove_dir_all(&root);
616    }

637    /// A listener in a private directory, which `uid` here is never the owner of.
638    fn foreign_listener(name: &str) -> (PathBuf, std::os::unix::net::UnixListener, u32) {
639        let root = std::env::temp_dir().join(format!("rxapi-{name}-{}", std::process::id()));
640        let dir = prepare_dir(&root.join("api")).expect("a private directory");
641        let socket = socket_path(&dir, PROTOCOL_VERSION);
642        let listener = std::os::unix::net::UnixListener::bind(socket).expect("a listener");
643        listener.set_nonblocking(true).expect("non-blocking");
644        (root, listener, own_uid() + 1)
645    }

647    fn assert_told_nothing(listener: &std::os::unix::net::UnixListener) {
648        let (mut accepted, _) = listener.accept().expect("the attempt connected");
649        accepted.set_nonblocking(true).expect("non-blocking");
650        let mut byte = [0];
651        match accepted.read(&mut byte) {
652            Ok(0) => {}
653            Ok(_) => panic!("the peer was sent bytes"),
654            Err(error) => assert_eq!(error.kind(), io::ErrorKind::WouldBlock),
655        }
656    }

658    #[test]
659    fn a_service_of_another_user_is_an_error_not_an_absence() {
660        let (root, listener, uid) = foreign_listener("foreign");
661        let dir = root.join("api");
662        let foreign = TransportError::Foreign { uid: own_uid() };
663
664        assert_eq!(
665            Unix::stop_as(&dir, PROTOCOL_VERSION, uid).err(),
666            Some(foreign.clone())
667        );
668        assert_told_nothing(&listener);
669        let starting = Unix::start_and_connect(&dir, Path::new("/nonexistent"), uid);
670        assert_eq!(starting.err(), Some(foreign.clone()));
671        assert_told_nothing(&listener);
672        assert_eq!(foreign.status(), crate::message::code::NO_SERVICE);
673
674        fs::remove_dir_all(&root).expect("cleanup");
675    }
