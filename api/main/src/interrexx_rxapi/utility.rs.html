1//! The messages `rxqueue` and `rxsubcom` print: catalogue entries of major 999, by subcode.
2
3use interrexx_inventory::errors::{lookup, substitute};
4
5/// The catalogue major of the utilities' messages.
6const MAJOR: u16 = 999;
7
8/// `rxsubcom`: the parameters are incorrect, no action named.
9pub const SUBCOM_GENERAL: u16 = 0;
10/// `rxsubcom REGISTER`: the parameters are incorrect.
11pub const SUBCOM_REGISTER: u16 = 1;
12/// `rxsubcom DROP`: the parameters are incorrect.
13pub const SUBCOM_DROP: u16 = 2;
14/// `rxsubcom LOAD`: the parameters are incorrect.
15pub const SUBCOM_LOAD: u16 = 3;
16/// `rxsubcom QUERY`: the parameters are incorrect.
17pub const SUBCOM_QUERY: u16 = 11;
18/// `rxqueue`: the queuing system is not available.
19pub const QUEUE_NOT_INITIALISED: u16 = 4;
20/// `rxqueue`: the data size is refused (999.5).
21pub const QUEUE_SIZE: u16 = 5;
22/// `rxqueue`: queue storage ran out (999.6).
23pub const QUEUE_NO_MEMORY: u16 = 6;
24/// `rxqueue`: a queue name is not valid. Insert 1 is the name.
25pub const QUEUE_NAME: u16 = 7;
26/// `rxqueue`: the access mode is refused (999.8).
27pub const QUEUE_ACCESS: u16 = 8;
28/// `rxqueue`: no queue of that name (999.10). Insert 1 is the name.
29pub const QUEUE_EXIST: u16 = 10;
30/// `rxqueue`: the command's syntax is incorrect.
31pub const QUEUE_SYNTAX: u16 = 16;
32
33/// The text of message `999.sub` with `inserts` in place of `&1`, `&2`, ... Bytes that are not
34/// UTF-8 are replaced, as the terminal shows them.
35///
36/// Panics for a subcode the catalogue lacks: the constants above are all catalogued.
37pub fn message(sub: u16, inserts: &[&[u8]]) -> String {
38    let entry = lookup(MAJOR, sub).expect("a catalogued utility message");
39    String::from_utf8_lossy(&substitute(entry.text, inserts)).into_owned()
40}
41
42#[cfg(test)]
43mod tests {
44    use super::*;
45
46    const ALL: [u16; 12] = [
47        SUBCOM_GENERAL,
48        SUBCOM_REGISTER,
49        SUBCOM_DROP,
50        SUBCOM_LOAD,
51        SUBCOM_QUERY,
52        QUEUE_NOT_INITIALISED,
53        QUEUE_SIZE,
54        QUEUE_NO_MEMORY,
55        QUEUE_NAME,
56        QUEUE_ACCESS,
57        QUEUE_EXIST,
58        QUEUE_SYNTAX,
59    ];
60
61    #[test]
62    fn every_message_is_catalogued_and_distinct() {
63        let mut seen = Vec::new();
64        for sub in ALL {
65            assert!(lookup(MAJOR, sub).is_some(), "999.{sub}");
66            assert!(!seen.contains(&sub), "999.{sub} listed twice");
67            seen.push(sub);
68        }
69    }
70
71    #[test]
72    fn a_message_is_the_catalogue_text_with_its_inserts() {
73        for sub in ALL {
74            let entry = lookup(MAJOR, sub).expect("catalogued");
75            let with_none = message(sub, &[]);
76            assert_eq!(
77                with_none.as_bytes(),
78                substitute::<&[u8]>(entry.text, &[]).as_slice(),
79                "999.{sub}"
80            );
81        }
82        // A named queue reaches the text at insert 1, and nothing is left of the marker.
83        for sub in [QUEUE_NAME, QUEUE_EXIST] {
84            let text = message(sub, &[b"qq-probe"]);
85            assert!(text.contains("qq-probe"), "999.{sub}");
86            assert!(!text.contains("&1"), "999.{sub}");
87        }
88        // A message that takes no insert ignores the one handed to it.
89        assert_eq!(message(QUEUE_SIZE, &[b"x"]), message(QUEUE_SIZE, &[]));
90    }
91}
