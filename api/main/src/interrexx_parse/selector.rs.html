1//! Message selectors, interned while a source is parsed.
2
3use rustc_hash::FxHashSet;
4use std::sync::Arc;
5
6/// One message name, as the parse that read it holds it.
7#[derive(Clone, PartialEq, Eq, Hash, Debug)]
8pub struct Selector(Arc<[u8]>);
9
10impl Selector {
11    /// The name's bytes.
12    pub fn bytes(&self) -> &[u8] {
13        &self.0
14    }
15
16    /// Whether these two are the same interned name -- one allocation, which
17    /// is what interning buys over comparing bytes.
18    pub fn same(left: &Selector, right: &Selector) -> bool {
19        Arc::ptr_eq(&left.0, &right.0)
20    }
21}
22
23impl std::ops::Deref for Selector {
24    type Target = [u8];
25
26    fn deref(&self) -> &[u8] {
27        &self.0
28    }
29}
30
31/// One parse's pool of message names.
32#[derive(Default)]
33pub(crate) struct SelectorTable(FxHashSet<Arc<[u8]>>);
34
35impl SelectorTable {
36    pub(crate) fn new() -> SelectorTable {
37        SelectorTable::default()
38    }
39
40    /// The pool's selector for `name`, adding it if the pool has no such
41    /// spelling.
42    pub(crate) fn intern(&mut self, name: &[u8]) -> Selector {
43        if let Some(found) = self.0.get(name) {
44            return Selector(Arc::clone(found));
45        }
46        let fresh: Arc<[u8]> = Arc::from(name);
47        self.0.insert(Arc::clone(&fresh));
48        Selector(fresh)
49    }
50}

52#[cfg(test)]
53mod tests {
54    use super::*;
55
56    /// **The interning constraint's own test**: one spelling is one
57    /// allocation within a parse, so a resolution keyed on a selector's
58    /// identity answers the same for every site that names the method.
59    #[test]
60    fn one_spelling_interns_to_one_allocation_and_two_spellings_do_not() {
61        let mut pool = SelectorTable::new();
62        let first = pool.intern(b"LENGTH");
63        let second = pool.intern(b"LENGTH");
64        let other = pool.intern(b"REVERSE");
65
66        assert!(
67            Selector::same(&first, &second),
68            "the same spelling interned twice is two allocations"
69        );
70        assert_eq!(first, second);
71        assert!(!Selector::same(&first, &other));
72        assert_ne!(first, other);
73        assert_eq!(first.bytes(), b"LENGTH");
74    }
75
76    /// Two pools are two parses, and a selector's identity is a property of
77    /// the pool that minted it. Equality still holds across the two, which
78    /// is what every byte-keyed reader depends on.
79    #[test]
80    fn two_pools_agree_by_value_and_not_by_identity() {
81        let mut one = SelectorTable::new();
82        let mut two = SelectorTable::new();
83        let left = one.intern(b"LENGTH");
84        let right = two.intern(b"LENGTH");
85
86        assert_eq!(left, right);
87        assert!(!Selector::same(&left, &right));
88    }
89}
