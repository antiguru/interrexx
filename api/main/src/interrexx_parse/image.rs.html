1//! A parsed program as bytes, and back: what lets the interpreter's own
2//! library, parsed when the interpreter is built, start without being parsed
3//! again.
4//!
5//! An image holds everything a [`Program`] holds except its text, which the
6//! reader is handed and checks against the image: the image is refused
7//! unless it was written by this build's format from that same text and every
8//! expression arena it holds is consistent (see [`Exprs::check`]). A refused
9//! image is `None`, never a partial program: `None` is the caller's cue to
10//! parse the text instead.
11//!
12//! The image of the interpreter's own library is read without that check, by
13//! [`program_from_static_image`] and the lazy body reader: the build writes it
14//! with [`checked_program_image`], which runs the check in every profile, and
15//! the check would read every body at startup, which that image exists to
16//! avoid.
17

31/// What an image starts with.
32const MAGIC: &[u8; 8] = b"RXPARSED";
33
34/// The layout this module writes and reads. A change to what any type below
35/// writes changes it.
36const FORMAT: u32 = 4;
37
38/// `program`, which was parsed from its own source text, as an image.
39pub fn program_image(program: &Program) -> Vec<u8> {
40    let mut out = Writer(Vec::new());
41    out.0.extend_from_slice(MAGIC);
42    out.0.extend_from_slice(&FORMAT.to_le_bytes());
43    let (text, ..) = program.source.parts();
44    out.0.extend_from_slice(&fingerprint(text).to_le_bytes());
45    out.usize(text.len());
46    program.source.put(&mut out);
47    program.symbols.put(&mut out);
48    program.main.put(&mut out);
49    program.directives.put(&mut out);
50    out.0
51}

53/// [`program_image`] for an image that will be read without the arena check,
54/// such as one built into the interpreter: the check runs here instead.
55///
56/// # Errors
57///
58/// The first arena invariant `program` breaks, as [`Program::check`] reports
59/// it, and then no image is written.
60pub fn checked_program_image(program: &Program) -> Result<Vec<u8>, Inconsistent> {
61    program.check()?;
62    Ok(program_image(program))
63}

65/// The program `image` holds, given the text it was parsed from, or `None`
66/// where `image` is not this format's image of `text`.
67pub fn program_from_image(image: &[u8], text: &[u8]) -> Option<Program> {
68    read_program(image, None, text)
69}

71/// [`program_from_image`] for an image built into the interpreter together with
72/// `text`, by the one build step that wrote both: each code body is left encoded
73/// until it is first read (see [`Body`]), and only a debug build hashes `text` to
74/// check the image is its image, which is a quarter of what an empty program's
75/// startup spends reading the image. The format and the text's length are checked
76/// either way.
77pub fn program_from_static_image(image: &'static [u8], text: &[u8]) -> Option<Program> {
78    read_program(image, Some(image), text)
79}

81fn read_program(image: &[u8], whole: Option<&'static [u8]>, text: &[u8]) -> Option<Program> {
82    let mut input = Reader {
83        bytes: image,
84        whole,
85        selectors: SelectorTable::new(),
86        text: None,
87    };
88    let built_in = whole.is_some() && !cfg!(debug_assertions);
89    if input.take(MAGIC.len())? != MAGIC
90        || u32::from_le_bytes(input.take(4)?.try_into().ok()?) != FORMAT
91    {
92        return None;
93    }
94    let fingerprinted = u64::from_le_bytes(input.take(8)?.try_into().ok()?);
95    if (!built_in && fingerprinted != fingerprint(text)) || input.usize()? != text.len() {
96        return None;
97    }
98    input.text = Some(text.to_vec());
99    let program = Program {
100        source: Image::get(&mut input)?,
101        symbols: Image::get(&mut input)?,
102        main: Image::get(&mut input)?,
103        directives: Image::get(&mut input)?,
104    };
105    if !input.bytes.is_empty() || (whole.is_none() && program.check().is_err()) {
106        return None;
107    }
108    Some(program)
109}

111/// The code body a [`Body`] left encoded holds.
112pub(crate) fn decode_body(encoded: &[u8]) -> Option<CodeBody> {
113    let mut input = Reader {
114        bytes: encoded,
115        whole: None,
116        selectors: SelectorTable::new(),
117        text: None,
118    };
119    let body = CodeBody::get(&mut input)?;
120    input.bytes.is_empty().then_some(body)
121}

123/// A 64-bit FNV-1a hash of `text`, which ties an image to the text it was
124/// parsed from.
125fn fingerprint(text: &[u8]) -> u64 {
126    text.iter().fold(0xcbf2_9ce4_8422_2325, |hash, &byte| {
127        (hash ^ u64::from(byte)).wrapping_mul(0x0100_0000_01b3)
128    })
129}

131struct Writer(Vec<u8>);
132
133impl Writer {
134    fn u8(&mut self, value: u8) {
135        self.0.push(value);
136    }
137
138    /// LEB128.
139    fn usize(&mut self, mut value: usize) {
140        loop {
141            let byte = u8::cast_truncate(value & 0x7f);
142            value >>= 7;
143            if value == 0 {
144                self.0.push(byte);
145                return;
146            }
147            self.0.push(byte | 0x80);
148        }
149    }
150
151    fn bytes(&mut self, bytes: &[u8]) {
152        self.usize(bytes.len());
153        self.0.extend_from_slice(bytes);
154    }
155}

157struct Reader<'a> {
158    bytes: &'a [u8],
159    /// The whole image, where it lives as long as the program: `bytes` is
160    /// then its tail, and a body is left encoded as a slice of it.
161    whole: Option<&'static [u8]>,
162    /// Interns every selector read, as the parser does.
163    selectors: SelectorTable,
164    /// The source text, until the program's source takes it.
165    text: Option<Vec<u8>>,
166}
167
168impl<'a> Reader<'a> {
169    fn take(&mut self, count: usize) -> Option<&'a [u8]> {
170        if count > self.bytes.len() {
171            return None;
172        }
173        let (taken, rest) = self.bytes.split_at(count);
174        self.bytes = rest;
175        Some(taken)
176    }
177
178    fn u8(&mut self) -> Option<u8> {
179        Some(self.take(1)?[0])
180    }
181
182    fn usize(&mut self) -> Option<usize> {
183        let mut value = 0usize;
184        let mut shift = 0;
185        loop {
186            let byte = self.u8()?;
187            if shift >= usize::BITS {
188                return None;
189            }
190            value |= usize::from(byte & 0x7f) << shift;
191            if byte & 0x80 == 0 {
192                return Some(value);
193            }
194            shift += 7;
195        }
196    }
197
198    fn bytes(&mut self) -> Option<&'a [u8]> {
199        let count = self.usize()?;
200        self.take(count)
201    }
202}

204/// A value an image holds.
205trait Image: Sized {
206    fn put(&self, out: &mut Writer);
207    fn get(input: &mut Reader<'_>) -> Option<Self>;
208}

308impl<T: Image> Image for Option<T> {
309    fn put(&self, out: &mut Writer) {
310        match self {
311            None => out.u8(0),
312            Some(value) => {
313                out.u8(1);
314                value.put(out);
315            }
316        }
317    }

320        match input.u8()? {
321            0 => Some(None),
322            1 => Some(Some(T::get(input)?)),
323            _ => None,
324        }
325    }

338impl<T: Image> Image for Vec<T> {
339    fn put(&self, out: &mut Writer) {
340        out.usize(self.len());
341        for item in self {
342            item.put(out);
343        }
344    }
345
346    fn get(input: &mut Reader<'_>) -> Option<Self> {
347        let count = input.usize()?;
348        // Bounded by what is left, so a damaged count cannot ask for more
349        // than the image could hold: every item takes a byte at least.
350        if count > input.bytes.len() {
351            return None;
352        }
353        let mut items = Vec::with_capacity(count);
354        for _ in 0..count {
355            items.push(T::get(input)?);
356        }
357        Some(items)
358    }
359}

361impl Image for BTreeMap<Box<[u8]>, usize> {
362    fn put(&self, out: &mut Writer) {
363        out.usize(self.len());
364        for (key, value) in self {
365            key.put(out);
366            value.put(out);
367        }
368    }
369
370    fn get(input: &mut Reader<'_>) -> Option<Self> {
371        let count = input.usize()?;
372        let mut map = BTreeMap::new();
373        for _ in 0..count {
374            let key = Image::get(input)?;
375            map.insert(key, Image::get(input)?);
376        }
377        (map.len() == count).then_some(map)
378    }
379}

402/// A body's image, behind its length, so a reader can leave it encoded.
403impl Image for Body {
404    fn put(&self, out: &mut Writer) {
405        let mut body = Writer(Vec::new());
406        CodeBody::put(self, &mut body);
407        out.bytes(&body.0);
408    }
409
410    fn get(input: &mut Reader<'_>) -> Option<Self> {
411        let encoded = input.bytes()?;
412        match input.whole {
413            Some(whole) => {
414                let end = whole.len() - input.bytes.len();
415                Some(Body::encoded(&whole[end - encoded.len()..end]))
416            }
417            None => Some(Body::new(decode_body(encoded)?)),
418        }
419    }
420}

422/// The names in id order; read back, each is interned again, which gives it
423/// the id it had.
424impl Image for SymbolTable {
425    fn put(&self, out: &mut Writer) {
426        out.usize(self.len());
427        for index in 0..self.len() {
428            let id = SymbolId::from_index(index).expect("an id the table issued");
429            out.bytes(self.name(id).as_bytes());
430        }
431    }
432
433    fn get(input: &mut Reader<'_>) -> Option<Self> {
434        let count = input.usize()?;
435        let mut table = SymbolTable::default();
436        for index in 0..count {
437            let name = std::str::from_utf8(input.bytes()?).ok()?;
438            if table.intern(name).index() != index {
439                return None;
440            }
441        }
442        Some(table)
443    }
444}

446/// Everything but the text, which the reader holds.
447impl Image for ProgramSource {
448    fn put(&self, out: &mut Writer) {
449        let (_, lines, kind, interpret_line, traceable, image_name) = self.parts();
450        out.usize(lines.len());
451        for &(start, end) in lines {
452            out.usize(start);
453            out.usize(end);
454        }
455        kind.put(out);
456        out.usize(interpret_line);
457        traceable.put(out);
458        image_name.map(Box::<[u8]>::from).put(out);
459    }
460
461    fn get(input: &mut Reader<'_>) -> Option<Self> {
462        let count = input.usize()?;
463        if count > input.bytes.len() {
464            return None;
465        }
466        let mut lines = Vec::with_capacity(count);
467        for _ in 0..count {
468            lines.push((input.usize()?, input.usize()?));
469        }
470        let kind = Image::get(input)?;
471        let interpret_line = input.usize()?;
472        let traceable = Image::get(input)?;
473        let image_name = Image::get(input)?;
474        let text = input.text.take()?;
475        Some(ProgramSource::from_parts(
476            text,
477            lines,
478            kind,
479            interpret_line,
480            traceable,
481            image_name,
482        ))
483    }
484}

486/// A struct, field by field. The pattern names every field, so a field added
487/// to the struct and not here does not compile.
488macro_rules! image_struct {
489    ($ty:ident { $($field:ident),* $(,)? }) => {
490        impl Image for $ty {
491            fn put(&self, out: &mut Writer) {
492                let $ty { $($field),* } = self;
493                $($field.put(out);)*
494            }
495
496            fn get(input: &mut Reader<'_>) -> Option<Self> {
497                Some($ty { $($field: Image::get(input)?),* })
498            }
499        }
500    };
501}

503/// An enum, as a tag byte and then the variant's fields. The `match` names
504/// every variant, so a variant added to the enum and not here does not
505/// compile.
506macro_rules! image_enum {
507    ($ty:ident {
508        $($tag:literal => $variant:ident
509            $(( $($item:ident),* ))?
510            $({ $($field:ident),* })?
511        ),* $(,)?
512    }) => {
513        impl Image for $ty {
514            fn put(&self, out: &mut Writer) {
515                match self {
516                    $($ty::$variant $(( $($item),* ))? $({ $($field),* })? => {
517                        out.u8($tag);
518                        $($($item.put(out);)*)?
519                        $($($field.put(out);)*)?
520                    })*
521                }
522            }
523
524            fn get(input: &mut Reader<'_>) -> Option<Self> {
525                Some(match input.u8()? {
526                    $($tag => $ty::$variant
527                        $(( $({
528                            let $item = Image::get(input)?;
529                            $item
530                        }),* ))?
531                        $({ $($field: Image::get(input)?),* })?,)*
532                    _ => return None,
533                })
534            }
535        }
536    };
537}
