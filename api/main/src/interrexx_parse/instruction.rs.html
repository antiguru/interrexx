159/// The missing-`THEN` error, which names the instruction that wanted it:
160/// 18.1 for an `IF` and 18.2 for a `WHEN`.
161pub(crate) fn missing_then(which: PendingThen) -> Msg<1> {
162    match which {
163        PendingThen::If => E18_1,
164        PendingThen::When => E18_2,
165    }
166}

168/// Parses the clause the cursor is sitting on, advancing it.
169pub(crate) fn parse_instruction(
170    ctx: &ParseCtx,
171    cursor: &mut ClauseCursor,
172    block: &mut Block,
173) -> Result<Instruction, ParseError> {
174    let clause = cursor
175        .peek()
176        .expect("parse_instruction on an exhausted cursor")
177        .clone();
178    let mut parser = Inst::new(ctx, clause);
179
180    // Block translation looks for the THEN itself, before instruction parsing
181    // sees the clause at all, so this test comes ahead of the label and
182    // assignment tests rather than inside the keyword dispatch: measured, `if 1
183    // = 1` followed by `then = 7` is an error rather than an assignment.
184    if let Some((which, byte)) = cursor.take_expected_then() {
185        // A label clause is excluded, which is the one place this differs from
186        // a plain reading of the clause. There the label is still one clause
187        // with whatever follows the colon, so a label spelled THEN becomes the
188        // THEN and the leftover `:` then fails with 35.1. The colon is already
189        // split off here, leaving nothing that can fail, so the missing-THEN
190        // error fires instead. Both reject `if 1 = 1` followed by `then: nop`,
191        // with 35.1 for both the IF and the WHEN spelling, and only the number
192        // differs.
193        if parser.clause.label.is_some() || parser.first_keyword() != Some(KW_THEN) {
194            // Reported against THIS clause, the offending one, naming the IF's
195            // line: measured, `nop` / `if 1 = 1` / blank / blank / `nop`
196            // reports 18.1 at line 5, naming line 2 as the IF's.
197            return Err(parser.then_expected(which, byte));
198        }
199        let end_at = parser.keyword_end();
200        parser.next_real();
201        return Ok(parser.finish_split(
202            cursor,
203            InstructionKind::Then,
204            end_at,
205            Remainder::Instruction,
206        ));
207    }
208
209    parser.dispatch(cursor, block)
210}

290    /// Index of the `n`th token that is not a blank, counting the next one as
291    /// zero, without consuming.
292    fn nth_real_index(&self, n: usize) -> Option<usize> {
293        let mut i = self.cursor.peek()?;
294        let mut seen = 0;
295        loop {
296            while i < self.clause.tokens.end && self.ctx.tokens[i].kind.tag() == Tag::Blank {
297                i += 1;
298            }
299            if i >= self.clause.tokens.end {
300                return None;
301            }
302            if seen == n {
303                return Some(i);
304            }
305            seen += 1;
306            i += 1;
307        }
308    }

435    /// The order instruction parsing tries: label, assignment, message term,
436    /// keyword, command.
437    fn dispatch(
438        mut self,
439        cursor: &mut ClauseCursor,
440        block: &mut Block,
441    ) -> Result<Instruction, ParseError> {
442        if let Some(label) = self.clause.label.clone() {
443            let kind = self.label(label)?;
444            return Ok(self.finish(cursor, kind));
445        }
446        if let Some(kind) = self.assignment()? {
447            return Ok(self.finish(cursor, kind));
448        }
449        if let Some(kind) = self.message()? {
450            return Ok(self.finish(cursor, kind));
451        }
452        match self.first_keyword() {
453            Some(index) => self.keyword(cursor, index, block),
454            // Not a keyword, so the whole clause is a command.
455            None => self.command(cursor),
456        }
457    }

459    fn label(&self, label: Range<usize>) -> Result<InstructionKind, ParseError> {
460        if self.ctx.source.kind() == SourceKind::Interpret {
461            return Err(self.found(E47_1, Some(&self.ctx.tokens[label.start])));
462        }
463        // A label may be a literal too, `"here": nop`. Both spellings reach
464        // the label table through the token's value, which is why the name is
465        // bytes rather than a `SymbolId`: a literal was never seen as a symbol,
466        // so it is not in the read-only symbol table.
467        let name = match &self.ctx.tokens[label.start].kind {
468            TokenKind::Symbol { id, .. } => Box::from(self.ctx.symbols.name(*id).as_bytes()),
469            TokenKind::Literal { value } => value.clone(),
470            other => panic!("a label token is a symbol or a literal, not {other:?}"),
471        };
472        Ok(InstructionKind::Label { name })
473    }

475    /// The `symbol = expr` and `symbol (op)= expr` forms.
476    fn assignment(&mut self) -> Result<Option<InstructionKind>, ParseError> {
477        let Some(first) = self.peek_token(0) else {
478            return Ok(None);
479        };
480        let TokenKind::Symbol { id, class } = first.kind else {
481            return Ok(None);
482        };
483        let target_span = first.span.clone();
484        let Some(second) = self.peek_token(1) else {
485            return Ok(None);
486        };
487        let op = match &second.kind {
488            // `symbol == expr` is not an assignment and not an expression
489            // either: it is rejected outright.
490            TokenKind::Operator(Operator::StrictEqual) => {
491                return Err(self.found(E35_1, Some(second)));
492            }
493            TokenKind::Operator(Operator::Equal) => None,
494            TokenKind::Assignment(op) => Some(*op),
495            _ => return Ok(None),
496        };
497        need_variable(self.ctx, id, class, self.clause_byte)?;
498        self.seek(self.cursor.position() + 2);
499        // `a += b` is expanded into `a = a + b` at parse time, building the
500        // binary node itself, so there is one instruction form and not two.
501        // The target is a leaf of its own slot; the read of `a` is a second
502        // leaf, pushed before the value because it is evaluated before it.
503        let target = self.ctx.push(symbol_kind(id, class), target_span.clone());
504        let read = op.map(|op| (op, self.ctx.push(symbol_kind(id, class), target_span)));
505        let value = self.expr(Terminators::EOC, E35_918)?;
506        let value = match read {
507            None => value,
508            Some((op, read)) => push_binary(self.ctx, op, read, value),
509        };
510        Ok(Some(InstructionKind::Assignment { target, value }))
511    }

513    /// The four message-term forms.
514    fn message(&mut self) -> Result<Option<InstructionKind>, ParseError> {
515        let mut trial = self.trial();
516        // Taken before the trial, which adopts no node pushed before it.
517        let mark = self.ctx.exprs.borrow().mark();
518        let Some(term) = parse_message_term(self.ctx, &mut trial)? else {
519            return Ok(None);
520        };
521        // A term came back, so the trial's position is the real one.
522        self.cursor = trial;
523        let second = self.peek_token(0);
524        let op = match second.map(|token| &token.kind) {
525            // The whole clause was the message send.
526            None => {
527                return Ok(Some(InstructionKind::Message { term, value: None }));
528            }
529            Some(TokenKind::Operator(Operator::StrictEqual)) => {
530                return Err(self.found(E35_1, second));
531            }
532            Some(TokenKind::Operator(Operator::Equal)) => None,
533            Some(TokenKind::Assignment(op)) => Some(*op),
534            // Anything else means this was not a message instruction after all,
535            // so the clause goes back to keyword processing with its position
536            // reset, by falling through to the first token, and the term's
537            // nodes are dropped with it.
538            _ => {
539                self.cursor = TokenCursor::new(self.clause.tokens.clone());
540                self.ctx.exprs.borrow_mut().rewind(mark);
541                return Ok(None);
542            }
543        };
544        self.cursor.advance();
545        // A copy of the message term is the left operand and the original is
546        // turned into the assignment. The copy is pushed before the value,
547        // because it is evaluated before it.
548        let read = op.map(|op| (op, self.ctx.exprs.borrow_mut().copy(term)));
549        // A missing value names the `=` or assignment operator.
550        let Some(value) = self.opt_expr(Terminators::EOC)? else {
551            return Err(self.found(E35_1, second));
552        };
553        let value = match read {
554            None => value,
555            Some((op, read)) => push_binary(self.ctx, op, read, value),
556        };
557        Ok(Some(InstructionKind::Message {
558            term,
559            value: Some(value),
560        }))
561    }

570    /// The keyword instructions.
571    fn keyword(
572        mut self,
573        cursor: &mut ClauseCursor,
574        index: usize,
575        block: &mut Block,
576    ) -> Result<Instruction, ParseError> {
577        let keyword_end = self.keyword_end();
578        match index {
579            KW_NOP => {
580                self.next_real();
581                // Nothing may follow.
582                self.required_end(E21_901)?;
583                Ok(self.finish(cursor, InstructionKind::Nop))
584            }
585            KW_IF => {
586                self.next_real();
587                self.if_instruction(cursor, PendingThen::If, block)
588            }
589            KW_WHEN => {
590                self.next_real();
591                self.if_instruction(cursor, PendingThen::When, block)
592            }
593            // `ELSE` and `OTHERWISE` parse nothing at all. Both take their
594            // location from the keyword token and let block translation trim
595            // whatever shares the line.
596            KW_ELSE => {
597                self.next_real();
598                Ok(self.finish_split(
599                    cursor,
600                    InstructionKind::Else { then_exit: None },
601                    keyword_end,
602                    Remainder::Instruction,
603                ))
604            }
605            KW_OTHERWISE => {
606                self.next_real();
607                Ok(self.finish_split(
608                    cursor,
609                    InstructionKind::Otherwise,
610                    keyword_end,
611                    Remainder::Instruction,
612                ))
613            }
614            KW_CATCH => {
615                self.next_real();
616                let catch = self.catch()?;
617                Ok(self.finish(cursor, InstructionKind::Catch(Box::new(catch))))
618            }
619            KW_END => {
620                self.next_real();
621                let name = self.end_name()?;
622                Ok(self.finish(cursor, InstructionKind::End { name, closes: None }))
623            }
624            // `createLoop(false)` and `createLoop(true)`. Bare `DO` is a
625            // block and bare `LOOP` is `LOOP FOREVER`, which is the only
626            // difference between the two keywords.
627            KW_DO => {
628                self.next_real();
629                let body = self.create_loop(false)?;
630                Ok(self.finish(cursor, InstructionKind::Do(Box::new(body))))
631            }
632            KW_LOOP => {
633                self.next_real();
634                let body = self.create_loop(true)?;
635                Ok(self.finish(cursor, InstructionKind::Loop(Box::new(body))))
636            }
637            // `SAY`, `PUSH` and `QUEUE` are the same two lines: an optional
638            // expression to the end of the clause. PUSH and QUEUE share one
639            // instruction class and differ only in the instruction type, which
640            // is why they are separate variants here.
641            KW_SAY => {
642                self.next_real();
643                let expression = self.opt_expr(Terminators::EOC)?;
644                Ok(self.finish(cursor, InstructionKind::Say { expression }))
645            }
646            KW_PUSH => {
647                self.next_real();
648                let expression = self.opt_expr(Terminators::EOC)?;
649                Ok(self.finish(cursor, InstructionKind::Push { expression }))
650            }
651            KW_QUEUE => {
652                self.next_real();
653                let expression = self.opt_expr(Terminators::EOC)?;
654                Ok(self.finish(cursor, InstructionKind::Queue { expression }))
655            }
656            KW_DROP => {
657                self.next_real();
658                let variables = self.variable_list(E20_901)?;
659                Ok(self.finish(cursor, InstructionKind::Drop { variables }))
660            }
661            KW_EXPOSE => {
662                self.next_real();
663                // This is rejected inside INTERPRET, which happens at run time,
664                // because `rexxc` never parses the string: `interpret "expose
665                // a"` is rc 157, Error 99.908.
666                if self.ctx.source.kind() == SourceKind::Interpret {
667                    return Err(self.error(E99_908, []));
668                }
669                // Nothing may precede an EXPOSE, not even a label, because a
670                // label would give SIGNAL or CALL a target that skips it. Not
671                // method-specific, which was measured rather than assumed: `nop`
672                // / `expose a` in the MAIN program is
673                // 99.907 too.
674                if !block.at_body_start() {
675                    return Err(self.error(E99_907, []));
676                }
677                let variables = self.variable_list(E20_902)?;
678                for name in &variables {
679                    // Only the direct spelling is exposed for the GUARD check.
680                    // Measured: `expose (a)` / `guard on when a` is 99.913.
681                    if let VariableRef::Direct(id) = name {
682                        block.expose(Box::from(self.ctx.symbols.name(*id).as_bytes()));
683                    }
684                }
685                Ok(self.finish(cursor, InstructionKind::Expose { variables }))
686            }
687            // `PARSE`, `ARG` and `PULL`. The short forms have no options and no
688            // source keyword, and both imply UPPER.
689            KW_PARSE => {
690                self.next_real();
691                let body = self.parse_instruction_body(None)?;
692                Ok(self.finish(cursor, InstructionKind::Parse(Box::new(body))))
693            }
694            KW_ARG => {
695                self.next_real();
696                let body = self.parse_instruction_body(Some(ParseSource::Arg))?;
697                Ok(self.finish(cursor, InstructionKind::Arg(Box::new(body))))
698            }
699            KW_PULL => {
700                self.next_real();
701                let body = self.parse_instruction_body(Some(ParseSource::Pull))?;
702                Ok(self.finish(cursor, InstructionKind::Pull(Box::new(body))))
703            }
704            // The four instructions that are an optional expression and
705            // nothing else, plus the two that require one.
706            KW_RETURN => {
707                self.next_real();
708                let expression = self.opt_expr(Terminators::EOC)?;
709                Ok(self.finish(cursor, InstructionKind::Return { expression }))
710            }
711            KW_EXIT => {
712                self.next_real();
713                let expression = self.opt_expr(Terminators::EOC)?;
714                Ok(self.finish(cursor, InstructionKind::Exit { expression }))
715            }
716            KW_REPLY => {
717                self.next_real();
718                // Measured at run time: `interpret "reply 1"` is Error 99.924.
719                if self.ctx.source.kind() == SourceKind::Interpret {
720                    return Err(self.error(E99_924, []));
721                }
722                let expression = self.opt_expr(Terminators::EOC)?;
723                Ok(self.finish(cursor, InstructionKind::Reply { expression }))
724            }
725            KW_INTERPRET => {
726                self.next_real();
727                // Measured as 35.912.
728                let expression = self.expr(Terminators::EOC, E35_912)?;
729                Ok(self.finish(cursor, InstructionKind::Interpret { expression }))
730            }
731            KW_OPTIONS => {
732                self.next_real();
733                // Measured as 35.913.
734                let expression = self.expr(Terminators::EOC, E35_913)?;
735                Ok(self.finish(cursor, InstructionKind::Options { expression }))
736            }
737            KW_PROCEDURE => {
738                self.next_real();
739                // Only `EXPOSE` may follow, and the same error covers a
740                // non-symbol because both are 25.17. Measured: `procedure foo`
741                // is rc 231, Error 25.17.
742                let variables = match self.peek_real() {
743                    None => Vec::new(),
744                    Some(token) => {
745                        if self.sub_keyword(token) != Some(SUB_EXPOSE) {
746                            return Err(self.found(E25_17, Some(token)));
747                        }
748                        self.next_real();
749                        self.variable_list(E20_902)?
750                    }
751                };
752                Ok(self.finish(cursor, InstructionKind::Procedure { variables }))
753            }
754            KW_GUARD => {
755                self.next_real();
756                let guard = self.guard(block)?;
757                Ok(self.finish(cursor, InstructionKind::Guard(Box::new(guard))))
758            }
759            KW_FORWARD => {
760                self.next_real();
761                let forward = self.forward()?;
762                Ok(self.finish(cursor, InstructionKind::Forward(Box::new(forward))))
763            }
764            KW_RAISE => {
765                self.next_real();
766                let raise = self.raise()?;
767                Ok(self.finish(cursor, InstructionKind::Raise(Box::new(raise))))
768            }
769            KW_USE => {
770                self.next_real();
771                let use_ = self.use_instruction(block)?;
772                Ok(self.finish(cursor, InstructionKind::Use(Box::new(use_))))
773            }
774            KW_ADDRESS => {
775                self.next_real();
776                let address = self.address()?;
777                Ok(self.finish(cursor, InstructionKind::Address(Box::new(address))))
778            }
779            KW_NUMERIC => {
780                self.next_real();
781                let (setting, expression) = self.numeric()?;
782                Ok(self.finish(
783                    cursor,
784                    InstructionKind::Numeric {
785                        setting,
786                        expression,
787                    },
788                ))
789            }
790            KW_TRACE => {
791                self.next_real();
792                let trace = self.trace()?;
793                Ok(self.finish(cursor, InstructionKind::Trace(trace)))
794            }
795            KW_CALL => {
796                self.next_real();
797                let call = self.call()?;
798                Ok(self.finish(cursor, InstructionKind::Call(Box::new(call))))
799            }
800            KW_SIGNAL => {
801                self.next_real();
802                let signal = self.signal()?;
803                Ok(self.finish(cursor, InstructionKind::Signal(Box::new(signal))))
804            }
805            KW_SELECT => {
806                self.next_real();
807                let kind = self.select()?;
808                Ok(self.finish(cursor, kind))
809            }
810            // Both are built, and both parse bare: `leave` alone is rc 0 under
811            // `rexxc` and Error 28.1 only at run time, because the interpreter
812            // raises it. A parser that rejected it would diverge on every
813            // program holding an unreachable LEAVE.
814            KW_LEAVE => {
815                self.next_real();
816                let name = self.block_name(E20_907, E21_907)?;
817                Ok(self.finish(cursor, InstructionKind::Leave { name }))
818            }
819            KW_ITERATE => {
820                self.next_real();
821                let name = self.block_name(E20_908, E21_908)?;
822                Ok(self.finish(cursor, InstructionKind::Iterate { name }))
823            }
824            // A THEN that reaches the keyword dispatch is always misplaced,
825            // because a real one is consumed above.
826            KW_THEN => Err(self.error(E8_1, [])),
827            // Unreachable: `index` came from the keyword table and every entry has an arm
828            // above, which `tests::keyword_indices_still_name_their_own_spellings` pins.
829            _ => Err(self.unreachable_switch("keyword", self.peek_real())),
830        }
831    }

845    fn numeric(&mut self) -> Result<(NumericSetting, Option<ExprId>), ParseError> {
846        // Measured as 20.905 for a bare `numeric` and for `numeric "x"`.
847        let Some(token) = self.next_real() else {
848            return Err(self.found(E20_905, None));
849        };
850        if token.kind.tag() != Tag::Symbol {
851            return Err(self.found(E20_905, Some(token)));
852        }
853        match self.sub_keyword(token) {
854            Some(SUB_DIGITS) => Ok((NumericSetting::Digits, self.opt_expr(Terminators::EOC)?)),
855            Some(SUB_FUZZ) => Ok((NumericSetting::Fuzz, self.opt_expr(Terminators::EOC)?)),
856            Some(SUB_FORM) => {
857                let Some(index) = self.peek_real_index() else {
858                    // `NUMERIC FORM` alone resets to the package default.
859                    return Ok((NumericSetting::FormDefault, None));
860                };
861                let token = &self.ctx.tokens[index];
862                if token.kind.tag() != Tag::Symbol {
863                    // An implicit `NUMERIC FORM VALUE`, with the token left in
864                    // place. Measured: `numeric form (e)` is rc 0.
865                    return Ok((NumericSetting::FormValue, self.opt_expr(Terminators::EOC)?));
866                }
867                self.seek(index + 1);
868                match self.sub_keyword(token) {
869                    // Measured as 21.911 for `numeric form scientific x`.
870                    Some(SUB_SCIENTIFIC) => {
871                        self.required_end(E21_911)?;
872                        Ok((NumericSetting::FormScientific, None))
873                    }
874                    Some(SUB_ENGINEERING) => {
875                        self.required_end(E21_911)?;
876                        Ok((NumericSetting::FormEngineering, None))
877                    }
878                    // 35.917.
879                    Some(SUB_VALUE) => Ok((
880                        NumericSetting::FormValue,
881                        Some(self.expr(Terminators::EOC, E35_917)?),
882                    )),
883                    // Measured as 25.11 for `numeric form foo`.
884                    _ => Err(self.found(E25_11, Some(token))),
885                }
886            }
887            // Measured as 25.15 for `numeric foo`.
888            _ => Err(self.found(E25_15, Some(token))),
889        }
890    }

892    /// `TRACE`, in its four forms.
893    fn trace(&mut self) -> Result<Trace, ParseError> {
894        let Some(index) = self.peek_real_index() else {
895            return Ok(Trace::Default);
896        };
897        let token = &self.ctx.tokens[index];
898        match &token.kind {
899            TokenKind::Symbol { .. } | TokenKind::Literal { .. } => {
900                // `TRACE VALUE expr` is the one symbol that is not a setting.
901                if self.sub_keyword(token) == Some(SUB_VALUE) {
902                    self.seek(index + 1);
903                    // Measured as 35.916.
904                    return Ok(Trace::Value(self.expr(Terminators::EOC, E35_916)?));
905                }
906                self.seek(index + 1);
907                let value = self.value_of(token);
908                // Measured as 21.906 for `trace 5 x` and for `trace r x`.
909                self.required_end(E21_906)?;
910                match whole_number(&value, TRACE_DIGITS) {
911                    Some(skip) => Ok(Trace::Skip(skip)),
912                    None => {
913                        check_trace_setting(&value).map_err(|()| self.bad_trace_setting(&value))?;
914                        Ok(Trace::Setting(value))
915                    }
916                }
917            }
918            // `TRACE -n` and `TRACE +n`, the skip forms with a sign.
919            TokenKind::Operator(op @ (Operator::Subtract | Operator::Plus)) => {
920                let negate = *op == Operator::Subtract;
921                self.seek(index + 1);
922                // Measured: `trace -a` is rc 230, Error 26.7, so the number
923                // test happens after the end-of-clause test.
924                let Some(number) = self.next_real() else {
925                    return Err(self.found(E35_1, None));
926                };
927                if !matches!(number.kind.tag(), Tag::Symbol | Tag::Literal) {
928                    return Err(self.found(E35_1, Some(number)));
929                }
930                let value = self.value_of(number);
931                self.required_end(E21_906)?;
932                // 26.7.
933                let Some(skip) = whole_number(&value, TRACE_DIGITS) else {
934                    return Err(self.error(E26_7, [value.into_vec()]));
935                };
936                Ok(Trace::Skip(if negate { -skip } else { skip }))
937            }
938            // An implicit `TRACE VALUE`, with the token left in place.
939            // Measured: `trace (e)` is rc 0.
940            _ => match self.opt_expr(Terminators::EOC)? {
941                Some(expression) => Ok(Trace::Value(expression)),
942                None => Ok(Trace::Default),
943            },
944        }
945    }

947    fn guard(&mut self, block: &Block) -> Result<Guard, ParseError> {
948        // Measured at run time: `interpret "guard on"` is Error 99.912.
949        if self.ctx.source.kind() == SourceKind::Interpret {
950            return Err(self.error(E99_912, []));
951        }
952        // Measured as 25.913 for a bare `guard` and for `guard foo`.
953        let Some(token) = self.next_real() else {
954            return Err(self.found(E25_913, None));
955        };
956        let on = match self.sub_keyword(token) {
957            Some(SUB_ON) => true,
958            Some(SUB_OFF) => false,
959            _ => return Err(self.found(E25_913, Some(token))),
960        };
961        let mut watched = Vec::new();
962        let condition = match self.next_real() {
963            None => None,
964            Some(token) => {
965                // Measured as 25.912 for `guard on foo` and for `guard on 1`.
966                if self.sub_keyword(token) != Some(SUB_WHEN) {
967                    return Err(self.found(E25_912, Some(token)));
968                }
969                // 35.921, and like every
970                // other logical list the empty case is 35.929 first.
971                let condition = self.logical(Terminators::EOC, E35_929)?;
972                // A GUARD expression that names no exposed variable can never be
973                // woken again, so it is rejected. The names could be collected
974                // while the expression is evaluated, through guard-variable
975                // capture on the simple-variable and stem paths, and then
976                // counted. Walking the finished expression is equivalent because
977                // capture is unconditional over the whole expression and those
978                // two paths are reached for exactly the variable references in
979                // it.
980                self.guard_exposes(condition, block, &mut watched);
981                if watched.is_empty() {
982                    return Err(self.error(E99_913, []));
983                }
984                Some(condition)
985            }
986        };
987        Ok(Guard {
988            on,
989            condition,
990            watched,
991        })
992    }

994    /// Six options in any order, each at most once.
995    fn forward(&mut self) -> Result<Forward, ParseError> {
996        // Measured at run time: `interpret "forward to 1"` is Error 99.923.
997        if self.ctx.source.kind() == SourceKind::Interpret {
998            return Err(self.error(E99_923, []));
999        }
1000        let mut forward = Forward::default();
1001        while let Some(token) = self.next_real() {
1002            // Measured as 25.916.
1003            if token.kind.tag() != Tag::Symbol {
1004                return Err(self.found(E25_916, Some(token)));
1005            }
1006            match self.sub_keyword(token) {
1007                // 25.917 and
1008                // 35.925. Measured:
1009                // `forward to 1 to 2` is 25.917 and `forward to` is 35.925.
1010                Some(SUB_TO) => self.forward_option(&mut forward.to, E25_917, E35_925)?,
1011                Some(SUB_CLASS) => self.forward_option(&mut forward.class, E25_921, E35_928)?,
1012                Some(SUB_MESSAGE) => {
1013                    self.forward_option(&mut forward.message, E25_922, E35_927)?;
1014                }
1015                // ARGUMENTS and ARRAY exclude each other and share one error,
1016                // 25.918.
1017                Some(SUB_ARGUMENTS) => {
1018                    if forward.arguments.is_some() || forward.array.is_some() {
1019                        return Err(self.error(E25_918, []));
1020                    }
1021                    let Some(value) = parse_constant_expression(self.ctx, &mut self.cursor)? else {
1022                        return Err(self.error(E35_926, []));
1023                    };
1024                    forward.arguments = Some(value);
1025                }
1026                Some(SUB_ARRAY) => {
1027                    if forward.arguments.is_some() || forward.array.is_some() {
1028                        return Err(self.error(E25_918, []));
1029                    }
1030                    // Shared with RAISE ARRAY. Measured: `forward array 1` is
1031                    // 35.924.
1032                    let open = self
1033                        .next_real()
1034                        .filter(|token| token.kind.tag() == Tag::LeftParen);
1035                    if open.is_none() {
1036                        return Err(self.error(E35_924, []));
1037                    }
1038                    forward.array = Some(self.arg_list(Some(Tag::RightParen))?);
1039                }
1040                // Measured as 25.919.
1041                Some(SUB_CONTINUE) => {
1042                    if forward.continue_ {
1043                        return Err(self.error(E25_919, []));
1044                    }
1045                    forward.continue_ = true;
1046                }
1047                _ => return Err(self.found(E25_916, Some(token))),
1048            }
1049        }
1050        Ok(forward)
1051    }

1053    /// One `FORWARD` option that takes a constant expression.
1054    fn forward_option(
1055        &mut self,
1056        slot: &mut Option<ExprId>,
1057        duplicate: Msg<0>,
1058        missing: Msg<0>,
1059    ) -> Result<(), ParseError> {
1060        if slot.is_some() {
1061            return Err(self.error(duplicate, []));
1062        }
1063        match parse_constant_expression(self.ctx, &mut self.cursor)? {
1064            Some(value) => {
1065                *slot = Some(value);
1066                Ok(())
1067            }
1068            None => Err(self.error(missing, [])),
1069        }
1070    }

1072    /// A condition name, then five options.
1073    fn raise(&mut self) -> Result<Raise, ParseError> {
1074        // Measured as 20.914 for a bare `raise`.
1075        let Some(token) = self.next_real() else {
1076            return Err(self.error(E20_914, []));
1077        };
1078        let TokenKind::Symbol { id, .. } = token.kind else {
1079            return Err(self.error(E20_914, []));
1080        };
1081        let which = self.ctx.keywords.conditions.index_of(id);
1082        let mut raise = Raise {
1083            condition: self.value_of(token),
1084            propagate: false,
1085            rc: None,
1086            description: None,
1087            additional: None,
1088            array: None,
1089            result: None,
1090        };
1091        match which {
1092            // These three take a value after the condition name. SYNTAX also
1093            // needs run-time work, which the flag records there and the node
1094            // shape records here.
1095            Some(COND_FAILURE | COND_ERROR | COND_SYNTAX) => {
1096                let Some(value) = parse_constant_expression(self.ctx, &mut self.cursor)? else {
1097                    // Measured: `raise syntax` with no value is 35.1, naming
1098                    // the end of the clause.
1099                    return Err(self.found(E35_1, None));
1100                };
1101                raise.rc = Some(value);
1102            }
1103            Some(COND_USER) => {
1104                // Measured as 20.915.
1105                let Some(name) = self.next_real() else {
1106                    return Err(self.error(E20_915, []));
1107                };
1108                let TokenKind::Symbol { .. } = name.kind else {
1109                    return Err(self.error(E20_915, []));
1110                };
1111                let mut composed = b"USER ".to_vec();
1112                composed.extend_from_slice(&self.value_of(name));
1113                raise.condition = composed.into_boxed_slice();
1114            }
1115            Some(COND_PROPAGATE) => raise.propagate = true,
1116            Some(
1117                COND_HALT | COND_NOMETHOD | COND_NOSTRING | COND_NOTREADY | COND_NOVALUE
1118                | COND_LOSTDIGITS,
1119            ) => {}
1120            // ANY reaches here, because nothing can be raised for every
1121            // condition at once: measured, `raise any` is rc 231, Error 25.906,
1122            // as is `raise foo`.
1123            _ => return Err(self.found(E25_906, Some(token))),
1124        }
1125
1126        while let Some(token) = self.next_real() {
1127            // Measured as 25.907.
1128            if token.kind.tag() != Tag::Symbol {
1129                return Err(self.found(E25_907, Some(token)));
1130            }
1131            match self.sub_keyword(token) {
1132                // 25.908 and
1133                // 35.922.
1134                Some(SUB_DESCRIPTION) => {
1135                    self.forward_option(&mut raise.description, E25_908, E35_922)?;
1136                }
1137                // ADDITIONAL and ARRAY exclude each other and share 25.909.
1138                Some(SUB_ADDITIONAL) => {
1139                    if raise.additional.is_some() || raise.array.is_some() {
1140                        return Err(self.error(E25_909, []));
1141                    }
1142                    let Some(value) = parse_constant_expression(self.ctx, &mut self.cursor)? else {
1143                        return Err(self.error(E35_923, []));
1144                    };
1145                    raise.additional = Some(value);
1146                }
1147                Some(SUB_ARRAY) => {
1148                    if raise.additional.is_some() || raise.array.is_some() {
1149                        return Err(self.error(E25_909, []));
1150                    }
1151                    let open = self
1152                        .next_real()

1178    /// `USE` and `USE LOCAL`.
1179    fn use_instruction(&mut self, block: &mut Block) -> Result<Use, ParseError> {
1180        let first = self.peek_real();
1181        if first.and_then(|token| self.sub_keyword(token)) == Some(SUB_LOCAL) {
1182            self.next_real();
1183            return self.use_local(block);
1184        }
1185        let strict = first.and_then(|token| self.sub_keyword(token)) == Some(SUB_STRICT);
1186        if strict {
1187            self.next_real();
1188        }
1189        // 25.905 and `_use_strict` is
1190        // 25.929. Measured: `use foo` is 25.905 and `use strict foo` is 25.929.
1191        let arg = self.next_real();
1192        if arg.and_then(|token| self.sub_keyword(token)) != Some(SUB_ARG) {
1193            return Err(self.found(if strict { E25_929 } else { E25_905 }, arg));
1194        }
1195        let mut targets: Vec<Option<UseTarget>> = Vec::new();
1196        let mut allow_optionals = false;
1197        while let Some(index) = self.peek_real_index() {
1198            let token = &self.ctx.tokens[index];
1199            // A bare comma is an omitted position, which keeps the argument
1200            // numbering right.
1201            if token.kind.tag() == Tag::Comma {
1202                self.seek(index + 1);
1203                targets.push(None);
1204                continue;
1205            }
1206            // `...` ends the list and stops argument-count checking.
1207            // Measured as 99.930 for `use arg ..., a`.
1208            if token.kind.tag() == Tag::Symbol && &*self.value_of(token) == b"..." {
1209                self.seek(index + 1);
1210                allow_optionals = true;
1211                if !self.at_end() {
1212                    return Err(self.error(E99_930, []));
1213                }
1214                break;
1215            }
1216            // `>a` and `<a` alias the caller's variable instead of copying it.
1217            if matches!(
1218                token.kind,
1219                TokenKind::Operator(Operator::GreaterThan | Operator::LessThan)
1220            ) {
1221                self.seek(index + 1);
1222                // Measured as
1223                // 20.931 for `use arg >a.b`, because a compound cannot be
1224                // aliased.
1225                let Some(name) = self.next_real() else {
1226                    return Err(self.found(E20_931, None));
1227                };
1228                let TokenKind::Symbol { id, class } = name.kind else {
1229                    return Err(self.found(E20_931, Some(name)));
1230                };
1231                if !matches!(class, SymbolClass::Variable | SymbolClass::Stem) {
1232                    return Err(self.found(E20_931, Some(name)));
1233                }
1234                targets.push(Some(UseTarget {
1235                    target: self.ctx.push(symbol_kind(id, class), name.span.clone()),
1236                    default: None,
1237                    alias: true,
1238                }));
1239                match self.next_real() {
1240                    None => break,
1241                    Some(next) if next.kind.tag() == Tag::Comma => continue,
1242                    // Measured as 99.950 for `use arg >a = 1`.
1243                    Some(next) if matches!(next.kind, TokenKind::Operator(Operator::Equal)) => {
1244                        return Err(self.error(E99_950, []));
1245                    }
1246                    // Measured as 46.902.
1247                    Some(next) => return Err(self.found(E46_902, Some(next))),
1248                }
1249            }
1250            // A variable or a message term. Measured: `use arg q~x` is rc 0.
1251            let mut trial = self.trial_from(index);
1252            let target = match parse_variable_or_message_term(self.ctx, &mut trial)? {
1253                Some(target) => {
1254                    self.cursor = trial;
1255                    target
1256                }
1257                // 89.1.
1258                None => return Err(self.found(E89_1, Some(token))),
1259            };
1260            let mut default = None;
1261            match self.next_real() {
1262                None => {
1263                    targets.push(Some(UseTarget {
1264                        target,
1265                        default,
1266                        alias: false,
1267                    }));
1268                    break;
1269                }
1270                Some(next) if next.kind.tag() == Tag::Comma => {}
1271                Some(next) if matches!(next.kind, TokenKind::Operator(Operator::Equal)) => {
1272                    // Measured as
1273                    // 35.930 for `use arg a = `.
1274                    let Some(value) = parse_constant_expression(self.ctx, &mut self.cursor)? else {
1275                        return Err(self.error(E35_930, []));
1276                    };
1277                    default = Some(value);
1278                    match self.next_real() {
1279                        None => {
1280                            targets.push(Some(UseTarget {
1281                                target,
1282                                default,
1283                                alias: false,
1284                            }));
1285                            break;
1286                        }
1287                        Some(next) if next.kind.tag() == Tag::Comma => {}
1288                        Some(_) => return Err(self.error(E35_930, [])),
1289                    }
1290                }
1291                // `use arg a b` is not a two-variable list: after a target
1292                // that may take a default, `=` is expected too, which is
1293                // 46.903 (`r[instr.use.syntax]`).
1294                Some(next) => return Err(self.found(E46_903, Some(next))),
1295            }
1296            targets.push(Some(UseTarget {
1297                target,
1298                default,
1299                alias: false,
1300            }));
1301        }
1302        Ok(Use::Arg {
1303            strict,
1304            allow_optionals,
1305            targets,
1306        })
1307    }

1309    fn use_local(&mut self, block: &mut Block) -> Result<Use, ParseError> {
1310        // Measured at run time: `interpret "use local a"` is Error 99.915.
1311        if self.ctx.source.kind() == SourceKind::Interpret {
1312            return Err(self.error(E99_915, []));
1313        }
1314        // The same placement rule as EXPOSE and for the same reason.
1315        if !block.at_body_start() {
1316            return Err(self.error(E99_910, []));
1317        }
1318        // The five special names are made local before the list is read, so the
1319        // five special names are local even in a `USE LOCAL` with no names of
1320        // its own.
1321        block.auto_expose();
1322        let mut variables = Vec::new();
1323        while let Some(token) = self.next_real() {
1324            // 20.927.
1325            let TokenKind::Symbol { id, class } = token.kind else {
1326                return Err(self.error(E20_927, []));
1327            };
1328            self.need_variable_class(token, class)?;
1329            // Measured as 99.948 for `use local a.b`: only a simple variable or
1330            // a stem can be local.
1331            if class == SymbolClass::Compound {
1332                return Err(self.found(E99_948, Some(token)));
1333            }
1334            block.local_variable(Box::from(self.ctx.symbols.name(id).as_bytes()));
1335            variables.push(VariableRef::Direct(id));
1336        }
1337        Ok(Use::Local { variables })

1340    /// `CALL` and the three constructors above it, which are four distinct
1341    /// instruction objects.
1342    fn call(&mut self) -> Result<Call, ParseError> {
1343        // Measured as 19.2 for a bare `call`.
1344        let Some(token) = self.next_real() else {
1345            return Err(self.error(E19_2, []));
1346        };
1347        match &token.kind {
1348            TokenKind::Symbol { id, .. } => {
1349                // `CALL ns:name`. The colon is looked for with the next-token
1350                // read, so no blank may sit before it, and none can: a blank is
1351                // only a token when a symbol, a literal, `(` or `[` follows it.
1352                if self.peek_token(0).map(|token| token.kind.tag()) == Some(Tag::Colon) {
1353                    self.cursor.advance();
1354                    // Measured as
1355                    // 20.922 for `call ns:`.
1356                    let Some(name) = self.next_real() else {
1357                        return Err(self.found(E20_922, None));
1358                    };
1359                    let TokenKind::Symbol { id: name, .. } = name.kind else {
1360                        return Err(self.found(E20_922, Some(name)));
1361                    };
1362                    let args = self.arg_list(None)?;
1363                    return Ok(Call::Qualified {
1364                        namespace: *id,
1365                        name,
1366                        args,
1367                    });
1368                }
1369                match self.ctx.keywords.sub_keywords.index_of(*id) {
1370                    Some(index @ (SUB_ON | SUB_OFF)) => {
1371                        Ok(Call::Trap(self.condition_trap(index == SUB_ON, true)?))
1372                    }
1373                    _ => {
1374                        let name = self.value_of(token);
1375                        let args = self.arg_list(None)?;
1376                        self.ctx.note_symbol_call(*id);
1377                        Ok(Call::Named {
1378                            name,
1379                            literal: false,
1380                            args,
1381                        })
1382                    }
1383                }
1384            }
1385            // `CALL "name"` never resolves to an internal label, which is what
1386            // the no-internal flag records.
1387            TokenKind::Literal { value } => {
1388                let name = value.clone();
1389                let args = self.arg_list(None)?;
1390                self.ctx.note_literal_call(&name);
1391                Ok(Call::Named {
1392                    name,
1393                    literal: true,
1394                    args,
1395                })
1396            }
1397            // `CALL (expr) args`, whose target is only known at run time.
1398            TokenKind::LeftParen => {
1399                let Some(target) = parse_paren_expression(self.ctx, &mut self.cursor)? else {
1400                    return Err(self.error(E35_932, []));
1401                };
1402                let args = self.arg_list(None)?;
1403                Ok(Call::Dynamic { target, args })
1404            }
1405            _ => Err(self.error(E19_2, [])),
1406        }
1407    }

1452    fn catch(&mut self) -> Result<Catch, ParseError> {
1453        let variable = match (self.peek_real(), self.nth_real(1)) {
1454            (Some(token), Some(equal))
1455                if matches!(equal.kind, TokenKind::Operator(Operator::Equal)) =>
1456            {
1457                let TokenKind::Symbol { id, class } = token.kind else {
1458                    return Err(self.error(E20_936, []));
1459                };
1460                need_variable(self.ctx, id, class, self.clause_byte)?;
1461                self.next_real();
1462                self.next_real();
1463                Some(self.ctx.push(symbol_kind(id, class), token.span.clone()))
1464            }
1465            _ => None,
1466        };
1467
1468        let mut conditions = Vec::new();
1469        loop {
1470            let Some(token) = self.next_real() else {
1471                return Err(self.error(E20_936, []));
1472            };
1473            let TokenKind::Symbol { id, .. } = token.kind else {
1474                return Err(self.error(E20_936, []));
1475            };
1476            let Some(condition) = self.ctx.keywords.conditions.index_of(id) else {
1477                return Err(self.found(E25_939, Some(token)));
1478            };
1479            if condition == COND_PROPAGATE {
1480                return Err(self.found(E25_939, Some(token)));
1481            }
1482            let mut name = self.value_of(token);
1483            if condition == COND_USER {
1484                let Some(user) = self.next_real() else {
1485                    return Err(self.error(E20_915, []));
1486                };
1487                if user.kind.tag() != Tag::Symbol {
1488                    return Err(self.error(E20_915, []));
1489                }
1490                let mut qualified = b"USER ".to_vec();
1491                qualified.extend_from_slice(&self.value_of(user));
1492                name = qualified.into_boxed_slice();
1493            }
1494            let code = if matches!(condition, COND_SYNTAX | COND_ERROR | COND_FAILURE)
1495                && self
1496                    .peek_real()
1497                    .is_some_and(|token| token.kind.tag() != Tag::Comma)
1498            {
1499                Some(self.catch_code(condition == COND_SYNTAX)?)
1500            } else {
1501                None
1502            };
1503            conditions.push(CatchCondition { name, code });
1504            match self.next_real() {
1505                None => break,
1506                Some(separator) if separator.kind.tag() == Tag::Comma => {}
1507                token => return Err(self.found(E21_904, token)),
1508            }
1509        }
1510        Ok(Catch {
1511            variable,
1512            conditions,
1513            block: None,
1514        })
1515    }

1517    fn catch_code(&mut self, syntax: bool) -> Result<CatchCode, ParseError> {
1518        let token = self.next_real().expect("the caller saw a code");
1519        let value = match token.kind {
1520            TokenKind::Symbol {
1521                id,
1522                class: SymbolClass::Variable,
1523            } => {
1524                return Ok(CatchCode::Variable(id));
1525            }
1526            TokenKind::Symbol {
1527                class: SymbolClass::Constant,
1528                ..
1529            }
1530            | TokenKind::Literal { .. } => self.value_of(token).into_vec(),
1531            TokenKind::Operator(op @ (Operator::Plus | Operator::Subtract)) if !syntax => {
1532                let next = self.next_real();
1533                let Some(next) = next.filter(|t| {
1534                    matches!(
1535                        t.kind,
1536                        TokenKind::Symbol {
1537                            class: SymbolClass::Constant,
1538                            ..
1539                        }
1540                    )
1541                }) else {
1542                    return Err(self.found(E35_1, next));
1543                };
1544                let mut value = self.value_of(next).into_vec();
1545                if interrexx_num::Number::parse_bytes(&value).is_none() {
1546                    return Err(self.found(E35_1, Some(next)));
1547                }
1548                if op == Operator::Subtract {
1549                    value.insert(0, b'-');
1550                }
1551                return Ok(CatchCode::Literal(value.into_boxed_slice()));
1552            }
1553            _ => return Err(self.found(E35_1, Some(token))),
1554        };
1555        if syntax && !crate::convert::valid_catch_syntax_number(&value) {
1556            return Err(self.error(E35_937, [value]));
1557        }
1558        if !syntax
1559            && token.kind.tag() == Tag::Symbol
1560            && interrexx_num::Number::parse_bytes(&value).is_none()
1561        {
1562            return Err(self.found(E35_1, Some(token)));
1563        }
1564        Ok(CatchCode::Literal(value.into_boxed_slice()))
1565    }

1567    /// `CALL ON` and `SIGNAL ON`, which differ only in which conditions they
1568    /// accept and in four error numbers.
1569    fn condition_trap(&mut self, on: bool, is_call: bool) -> Result<ConditionTrap, ParseError> {
1570        // Measured: `call on` is 20.911 and `call off` is 20.912.
1571        let missing = if on { E20_911 } else { E20_912 };
1572        let Some(token) = self.next_real() else {
1573            return Err(self.error(missing, []));
1574        };
1575        let TokenKind::Symbol { id, .. } = token.kind else {
1576            return Err(self.error(missing, []));
1577        };
1578        let condition = self.ctx.keywords.conditions.index_of(id);
1579        // 25.1 and `_calloff` 25.2;
1580        // `_signalon` is 25.3 and `_signaloff` 25.4.
1581        let bad = match (is_call, on) {
1582            (true, true) => E25_1,
1583            (true, false) => E25_2,
1584            (false, true) => E25_3,
1585            (false, false) => E25_4,
1586        };
1587        let rejected = match condition {
1588            None => true,
1589            Some(COND_PROPAGATE) => true,
1590            // ANY is accepted by both, which is easy to assume otherwise:
1591            // measured, `call on any` and `signal on any` are both rc 0.
1592            Some(COND_ANY) => false,
1593            // Measured: `call on syntax` and `call on novalue` are 25.1, where
1594            // `signal on syntax` is rc 0.
1595            Some(COND_SYNTAX | COND_NOVALUE | COND_LOSTDIGITS | COND_NOMETHOD | COND_NOSTRING) => {
1596                is_call
1597            }
1598            _ => false,
1599        };
1600        if rejected {
1601            return Err(self.found(bad, Some(token)));
1602        }
1603        let mut label;
1604        let condition_name;
1605        if condition == Some(COND_USER) {
1606            // Measured as 20.915 for `call on user`.
1607            let Some(name) = self.next_real() else {
1608                return Err(self.error(E20_915, []));
1609            };
1610            let TokenKind::Symbol { .. } = name.kind else {
1611                return Err(self.error(E20_915, []));
1612            };
1613            let name = self.value_of(name);
1614            // The condition's own name is `USER name`, built the way
1615            // `concatToCstring("USER ")` builds it.
1616            let mut composed = b"USER ".to_vec();
1617            composed.extend_from_slice(&name);
1618            condition_name = composed.into_boxed_slice();
1619            label = Some(name);
1620        } else {
1621            let name = self.value_of(token);
1622            condition_name = name.clone();
1623            label = Some(name);
1624        }
1625
1626        if on {
1627            if let Some(token) = self.next_real() {
1628                // 25.914 and
1629                // `_signalonname` is 25.915.
1630                if self.sub_keyword(token) != Some(SUB_NAME) {
1631                    let msg = if is_call { E25_914 } else { E25_915 };
1632                    return Err(self.found(msg, Some(token)));
1633                }
1634                // Measured as 19.3 for `call on error name`.
1635                let Some(target) = self.next_real() else {
1636                    return Err(self.error(E19_3, []));
1637                };
1638                if !matches!(target.kind.tag(), Tag::Symbol | Tag::Literal) {
1639                    return Err(self.error(E19_3, []));
1640                }
1641                label = Some(self.value_of(target));
1642                // Measured as 21.903.
1643                self.required_end(E21_903)?;
1644            }
1645        } else {
1646            // The OFF form has no label at all, which is how the two are told
1647            // apart at run time.
1648            label = None;
1649            // Measured as 21.904 for `call off error x`.
1650            self.required_end(E21_904)?;
1651        }
1652
1653        Ok(ConditionTrap {
1654            on,
1655            condition: condition_name,
1656            label,
1657        })
1658    }

1725        let mut label = None;
1726        let mut case = None;
1727        if let Some(token) = self.peek_real() {
1728            // Anything that is not a symbol cannot be either keyword.
1729            if token.kind.tag() != Tag::Symbol {
1730                return Err(self.found(E25_923, Some(token)));
1731            }
1732            let mut token = token;
1733            if self.sub_keyword(token) == Some(SUB_LABEL) {
1734                self.next_real();
1735                let Some(name) = self.next_real() else {
1736                    return Err(self.error(E20_918, []));
1737                };
1738                let TokenKind::Symbol { id, .. } = name.kind else {
1739                    return Err(self.error(E20_918, []));
1740                };
1741                label = Some(id);
1742                match self.peek_real() {
1743                    Some(next) => token = next,
1744                    None => return Ok(select_node(label, case)),
1745                }
1746            }
1747            if token.kind.tag() == Tag::Symbol {
1748                if self.sub_keyword(token) != Some(SUB_CASE) {
1749                    return Err(self.found(E25_923, Some(token)));
1750                }
1751                self.next_real();
1752                case = Some(self.expr(Terminators::EOC, E35_933)?);
1753            }
1754            // Nothing else may follow either option.
1755            self.required_end(E25_923)?;
1756        }
1757        Ok(select_node(label, case))
1758    }

1760    /// Raises the error a non-variable symbol `token` of `class` gets in a
1761    /// `DROP`, `EXPOSE`, `PROCEDURE EXPOSE` or `USE LOCAL` list, naming it.
1762    fn need_variable_class(&self, token: &Token, class: SymbolClass) -> Result<(), ParseError> {
1763        match class {
1764            SymbolClass::Variable | SymbolClass::Stem | SymbolClass::Compound => Ok(()),
1765            SymbolClass::Constant => Err(self.found(E31_2, Some(token))),
1766            SymbolClass::Dummy | SymbolClass::DotSymbol => Err(self.found(E31_3, Some(token))),
1767        }
1768    }

1773            TokenKind::Symbol { id, .. } => Box::from(self.ctx.symbols.name(*id).as_bytes()),

1786    /// `IF` and the `SELECT` half of `WHEN`, which share one instruction class.
1787    fn if_instruction(
1788        mut self,
1789        cursor: &mut ClauseCursor,
1790        which: PendingThen,
1791        block: &Block,
1792    ) -> Result<Instruction, ParseError> {
1793        // 35.929 is the logical-list error, which the condition parser
1794        // raises before the required-expression check's own 35.902 or 35.903 can
1795        // be reached.
1796        let kind = match which {
1797            PendingThen::If => InstructionKind::If {
1798                condition: self.logical(Terminators::IF, E35_929)?,
1799                false_target: None,
1800            },
1801            // The control stack is asked which SELECT it is in, and the answer
1802            // decides three things: whether the clause is legal, which grammar
1803            // it follows, and which instruction class it becomes.
1804            PendingThen::When => match block.enclosing_select() {
1805                // Measured for a `WHEN` with nothing open at all and for one
1806                // inside a `DO` that is itself inside a `SELECT`, which is 9.1
1807                // too because the `DO` stops the search, and for one after an
1808                // `OTHERWISE`.
1809                EnclosingSelect::None => return Err(self.error(E9_1, [])),
1810                EnclosingSelect::Plain => InstructionKind::When {
1811                    condition: self.logical(Terminators::IF, E35_929)?,
1812                    false_target: None,
1813                    exit: None,
1814                },
1815                EnclosingSelect::Case => InstructionKind::WhenCase {
1816                    values: parse_case_when_list(self.ctx, &mut self.cursor, Terminators::IF)?,
1817                    false_target: None,
1818                    exit: None,
1819                },
1820            },
1821        };
1822        let end_at = self.condition_end();
1823        let byte = self.clause_byte;
1824        let instruction = self.finish_split(cursor, kind, end_at, Remainder::Then);
1825        cursor.expect_then(which, byte);
1826        Ok(instruction)
1827    }

1829    /// Adds to `watched` each variable exposed at this point that `condition`
1830    /// names, once: the variables `GUARD ... WHEN` watches, of which it
1831    /// requires at least one. A loop over the condition's nodes in index
1832    /// order, which meets the leaves left to right.
1833    fn guard_exposes(&self, condition: ExprId, block: &Block, watched: &mut Vec<Box<[u8]>>) {
1834        let mut watch = |name: &str| {
1835            if block.is_exposed(name.as_bytes())
1836                && !watched.iter().any(|seen| **seen == *name.as_bytes())
1837            {
1838                watched.push(name.as_bytes().into());
1839            }
1840        };
1841        let exprs = self.ctx.exprs.borrow();
1842        for node in exprs.range(condition) {
1843            match *node.kind() {
1844                NodeKind::Variable(id) | NodeKind::Stem(id) => watch(self.ctx.symbols.name(id)),
1845                // `>name` watches the variable it names, the same as reading
1846                // it (`r[instr.guard.when.exposed]`).
1847                NodeKind::VariableReference { symbol, .. } => watch(self.ctx.symbols.name(symbol)),
1848                // A compound contributes its stem and its variable tail
1849                // pieces, wherever else in the body it is referenced
1850                // (`r[instr.guard.when.exposed]`).
1851                NodeKind::Compound(id) => {
1852                    let (stem, tails) = compound_parts(self.ctx.symbols.name(id));
1853                    watch(stem);
1854                    for tail in tails {
1855                        if let Tail::Variable(piece) = tail {
1856                            watch(piece);
1857                        }
1858                    }
1859                }
1860                NodeKind::Literal(_)
1861                | NodeKind::Constant(_)
1862                | NodeKind::DotVariable(_)
1863                | NodeKind::ClassResolver { .. }
1864                | NodeKind::Omitted
1865                | NodeKind::Prefix { .. }
1866                | NodeKind::Binary { .. }
1867                | NodeKind::Call { .. }
1868                | NodeKind::QualifiedCall { .. }
1869                | NodeKind::Message { .. }
1870                | NodeKind::List(_)
1871                | NodeKind::Logical(_) => {}
1872            }
1873        }
1874    }
1875}
