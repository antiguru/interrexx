1//! `ADDRESS` and the redirections of its `WITH` clause.

3use interrexx_inventory::errors::{
4    E20_932, E20_933, E25_930, E25_931, E25_932, E25_933, E25_934, E35_914, E35_935,

16impl<'a> Inst<'a> {
17    pub(super) fn address(&mut self) -> Result<Address, ParseError> {
18        let mut address = Address::default();
19        let Some(index) = self.peek_real_index() else {
20            return Ok(address);
21        };
22        let token = &self.ctx.tokens[index];
23        if !matches!(token.kind.tag(), Tag::Symbol | Tag::Literal) {
24            // An implicit `ADDRESS VALUE`, with the token left in place.
25            // Measured: `address (e)` is rc 0.
26            address.dynamic = self.opt_expr(Terminators::PARSE_WITH)?;
27        } else if self.sub_keyword(token) == Some(SUB_VALUE) {
28            self.seek(index + 1);
29            // Measured as 35.914 for `address value`.
30            address.dynamic = Some(self.expr(Terminators::PARSE_WITH, E35_914)?);
31        } else {
32            self.seek(index + 1);
33            address.environment = Some(self.value_of(token));
34            if !self.at_end() {
35                // The command expression stops at WITH, so what follows is
36                // either that keyword or the end of the clause. This can come
37                // back empty, which is `ADDRESS env WITH ...`: a configuration
38                // with no command.
39                address.command = self.opt_expr(Terminators::PARSE_WITH)?;
40            }
41        }
42        let with = self
43            .peek_real()
44            .filter(|token| self.sub_keyword(token) == Some(SUB_WITH));
45        if with.is_some() {
46            self.next_real();
47            address.io = Some(Box::new(self.address_with()?));
48        }
49        Ok(address)
50    }

52    /// The `INPUT`, `OUTPUT` and `ERROR` redirections, each at most once and in
53    /// any order.
54    fn address_with(&mut self) -> Result<AddressIo, ParseError> {
55        let mut io = AddressIo::default();
56        // Measured as 20.933 for `address system with`.
57        if self.at_end() {
58            return Err(self.error(E20_933, []));
59        }
60        let mut seen = [false; 3];
61        while let Some(token) = self.next_real() {
62            if token.kind.tag() != Tag::Symbol {
63                return Err(self.error(E20_933, []));
64            }
65            // The three duplicate errors are 25.930, 25.931 and 25.932, one per
66            // stream. Measured: `with input normal input normal` is 25.930.
67            let stream = match self.sub_keyword(token) {
68                Some(SUB_INPUT) => 0,
69                Some(SUB_OUTPUT) => 1,
70                Some(SUB_ERROR) => 2,
71                // Measured as
72                // 25.934 for `address system with foo` and for `with 1`.
73                _ => return Err(self.found(E25_934, Some(token))),
74            };
75            if seen[stream] {
76                return Err(self.error([E25_930, E25_931, E25_932][stream], []));
77            }
78            seen[stream] = true;
79            // `NORMAL` resets the stream and takes no target.
80            let normal = self
81                .peek_real()
82                .filter(|token| self.sub_keyword(token) == Some(SUB_NORMAL));
83            if normal.is_some() {
84                self.next_real();
85                match stream {
86                    0 => io.input = Redirection::Normal,
87                    1 => io.output = Redirection::Normal,
88                    _ => io.error = Redirection::Normal,
89                }
90                continue;
91            }
92            // Only an output stream takes APPEND or REPLACE, and both are
93            // optional.
94            if stream != 0 {
95                let option = self.output_option();
96                if stream == 1 {
97                    io.output_option = option;
98                } else {
99                    io.error_option = option;
100                }
101            }
102            let target = self.redirect_target()?;
103            match stream {
104                0 => io.input = target,
105                1 => io.output = target,
106                _ => io.error = target,
107            }
108        }
109        Ok(io)
110    }

112    /// `APPEND` or `REPLACE`, or neither, in which case nothing is consumed.
113    fn output_option(&mut self) -> OutputOption {
114        let Some(token) = self.peek_real() else {
115            return OutputOption::Default;
116        };
117        let option = match self.sub_keyword(token) {
118            Some(SUB_REPLACE) => OutputOption::Replace,
119            Some(SUB_APPEND) => OutputOption::Append,
120            // Probably one of the target keywords, so leave it in place.
121            _ => return OutputOption::Default,
122        };
123        self.next_real();
124        option
125    }
126
127    /// Where one stream goes.
128    fn redirect_target(&mut self) -> Result<Redirection, ParseError> {
129        // Measured as 25.933 for `address system with input` and for `with input
130        // foo`.
131        let Some(token) = self.next_real() else {
132            return Err(self.found(E25_933, None));
133        };
134        if token.kind.tag() != Tag::Symbol {
135            return Err(self.found(E25_933, Some(token)));
136        }
137        match self.sub_keyword(token) {
138            Some(SUB_STEM) => {
139                // Measured as
140                // 20.932 for `with input stem a`, because `a` is a variable and
141                // not a stem.
142                let Some(name) = self.next_real() else {
143                    return Err(self.error(E20_932, []));
144                };
145                let TokenKind::Symbol {
146                    id,
147                    class: SymbolClass::Stem,
148                } = name.kind
149                else {
150                    return Err(self.error(E20_932, []));
151                };
152                Ok(Redirection::Stem(id))
153            }
154            // Both take the constant-expression form, so a bare variable is
155            // 35.1: measured, `with error using x` is rc 221 while
156            // `with error using (x)` would be accepted.
157            Some(index @ (SUB_STREAM | SUB_USING)) => {
158                match parse_constant_expression(self.ctx, &mut self.cursor)? {
159                    // Measured as
160                    // 35.935 for `with input stream`, naming the keyword and
161                    // the instruction.
162                    None => {
163                        let keyword: &[u8] = if index == SUB_STREAM {
164                            b"STREAM"
165                        } else {
166                            b"USING"
167                        };
168                        Err(self.error(E35_935, [keyword.to_vec(), b"ADDRESS".to_vec()]))
169                    }
170                    Some(value) => Ok(if index == SUB_STREAM {
171                        Redirection::Stream(value)
172                    } else {
173                        Redirection::Using(value)
174                    }),
175                }
176            }
177            _ => Err(self.found(E25_933, Some(token))),
178        }
179    }
