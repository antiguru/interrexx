68/// Stack size used to test parsing at [`MAX_EXPR_DEPTH`].
69pub const PARSER_STACK_BYTES: usize = 512 * 1024 * 1024;
70
71/// Runs `work` on a fresh thread with [`PARSER_STACK_BYTES`] of stack and
72/// waits for it to finish. The closure can borrow its inputs from the caller.
73///
74/// # Errors
75///
76/// Returns an I/O error if the thread cannot be spawned. A parse error
77/// returned by the closure is part of the closure's result.
78///
79/// # Panics
80///
81/// Propagates a panic from the worker to the caller.
82///
83/// # Examples
84///
85/// ```
86/// let line = b"say 'hello'".to_vec();
87/// let program = interrexx_parse::with_parser_stack(|| {
88///     interrexx_parse::parse_lines(&[&line])
89/// })??;
90/// assert_eq!(program.main.instructions.len(), 1);
91/// # Ok::<(), Box<dyn std::error::Error>>(())
92/// ```
93pub fn with_parser_stack<T: Send>(work: impl FnOnce() -> T + Send) -> std::io::Result<T> {
94    std::thread::scope(|scope| {
95        let worker = std::thread::Builder::new()
96            .name("rexx-parse".into())
97            .stack_size(PARSER_STACK_BYTES)
98            .spawn_scoped(scope, work)?;
99        match worker.join() {
100            Ok(result) => Ok(result),
101            Err(panic) => std::panic::resume_unwind(panic),
102        }
103    })
104}

106/// A whole program: everything `translate` produces from one source buffer.
107#[derive(Debug)]
108pub struct Program {
109    pub source: ProgramSource,
110    /// The main code body. A directive's own body is a `CodeBody` too, held
111    /// inside `directives` rather than here.
112    pub main: CodeBody,
113    /// The `::` directives, in source order, each carrying its own assembled
114    /// body.
115    pub directives: Vec<Directive>,
116    /// Retained because a `SymbolId` is meaningless without it: the executor
117    /// resolves names back to text to report them.
118    pub symbols: SymbolTable,
119}
120
121impl Program {
122    /// Checks the arena of every body and every `::CONSTANT` expression
123    /// against the roots its instructions name and against the source text
124    /// and symbol table its nodes index.
125    ///
126    /// # Errors
127    ///
128    /// The first invariant found broken, as [`Exprs::check`] reports it.
129    pub fn check(&self) -> Result<(), Inconsistent> {
130        let bounds = Bounds {
131            text: self.source.parts().0.len(),
132            symbols: self.symbols.len(),
133        };
134        self.main.check(bounds)?;
135        for directive in &self.directives {
136            let body = match &directive.kind {
137                DirectiveKind::Method(method) => method.body.as_ref(),
138                DirectiveKind::Attribute(attribute) => attribute.body.as_ref(),
139                DirectiveKind::Routine(routine) => routine.body.as_ref(),
140                DirectiveKind::Constant(constant) => {
141                    constant.value.check(bounds)?;
142                    None
143                }
144                DirectiveKind::Annotate(_)
145                | DirectiveKind::Class(_)
146                | DirectiveKind::Options(_)
147                | DirectiveKind::Requires(_)
148                | DirectiveKind::Resource(_) => None,
149            };
150            if let Some(body) = body {
151                body.check(bounds)?;
152            }
153        }
154        Ok(())
155    }
156}

183/// Parses a program `text` already known to translate, such as the
184/// interpreter's own library, without checking its directives against the
185/// ones declared before them.
186pub fn parse_known_program(text: Vec<u8>) -> Result<Program, ParseError> {
187    let source = ProgramSource::new(text, SourceKind::Program);
188    match parse(&source, false, None) {
189        Ok(parsed) => Ok(Program {
190            source,
191            main: parsed.main,
192            directives: parsed.directives,
193            symbols: parsed.symbols,
194        }),
195        Err((error, _)) => Err(error),
196    }
197}

216/// The minimum language level of a compiled program using 6.05 or 6.06 features.
217#[derive(Copy, Clone, Debug, Default, PartialEq, Eq)]
218#[repr(u32)]
219pub enum RequiredLanguageLevel {
220    #[default]
221    V6_05 = 60500,
222    V6_06 = 60600,
223}

225/// Parses a whole program and records which language level its translation requires.
226pub fn program_from_with_language_level(
227    source: ProgramSource,
228) -> Result<(Program, RequiredLanguageLevel), Rejected> {
229    let required_level = token::RequiredLanguage::default();
230    let program = program_from_tracking(source, Some(&required_level))?;
231    Ok((program, required_level.level.get()))
232}

234fn program_from_tracking(
235    source: ProgramSource,
236    required_level: Option<&token::RequiredLanguage>,
237) -> Result<Program, Rejected> {
238    match parse(&source, true, required_level) {
239        Ok(parsed) => Ok(Program {
240            source,
241            main: parsed.main,
242            directives: parsed.directives,
243            symbols: parsed.symbols,
244        }),
245        Err((error, binds)) => Err(Rejected {
246            error,
247            source,
248            binds,
249        }),
250    }
251}

253/// Parses `source` as an `INTERPRET` fragment, answering it back with the
254/// error when it does not parse.
255pub fn fragment_from(source: ProgramSource) -> Result<Fragment, Rejected> {
256    // `INTERPRET` text carries no directive, so it declares nothing to check.
257    let parsed = match parse(&source, false, None) {
258        Ok(parsed) => parsed,
259        Err((error, binds)) => {
260            return Err(Rejected {
261                error,
262                source,
263                binds,
264            });
265        }
266    };
267    debug_assert!(
268        parsed.directives.is_empty(),
269        "INTERPRET text cannot carry a directive: `parse` raises 99.914 first"
270    );
271    // Measured: a label inside `INTERPRET` text is 47.1, already
272    // enforced where `parse_instruction` builds a `Label` node, so this map is
273    // always empty rather than load-bearing.
274    debug_assert!(
275        parsed.main.labels.is_empty(),
276        "INTERPRET text cannot carry a label: `parse_instruction` raises 47.1 first"
277    );
278    Ok(Fragment {
279        source,
280        body: parsed.main,
281        symbols: parsed.symbols,
282    })
283}

285/// What one parse produces, before it is split into `Program` or `Fragment`.
286struct Parsed {
287    /// The main code body. A directive's body is inside the directive.
288    main: CodeBody,
289    directives: Vec<Directive>,
290    symbols: SymbolTable,
291}
292
293/// The composition shared by both entry points. A failure carries
294/// [`Rejected::binds`].
295fn parse(
296    source: &ProgramSource,
297    checked: bool,
298    required_level: Option<&token::RequiredLanguage>,
299) -> Result<Parsed, (ParseError, Vec<Directive>)> {
300    let (scanned, scan_error) = scan_to_error(source);
301    // Declared ahead of `ctx`, which borrows it, and dropped with the parse.
302    // A `Selector` in the arena holds the name's bytes itself, so what the
303    // pool's own entry buys is finding a spelling again while the parse is
304    // still reading names -- which is exactly as long as it lives.
305    let selectors = std::cell::RefCell::new(SelectorTable::new());
306    let exprs = std::cell::RefCell::new(Exprs::default());
307    let ctx = ParseCtx {
308        source,
309        tokens: &scanned.tokens,
310        symbols: &scanned.symbols,
311        selectors: &selectors,
312        exprs: &exprs,
313        keywords: scanned.keywords,
314        resources: &scanned.resources,
315        required_level,
316    };
317
318    let clauses = split_clauses(ctx.tokens).map_err(|error| (error, Vec::new()))?;
319    let mut cursor = ClauseCursor::new(clauses);
320    let mut directives = Vec::new();
321    let mut binds = Vec::new();
322    let parsed = translate(&ctx, &mut cursor, checked, &mut directives, &mut binds);
323    let bound = |directives: &[Directive]| -> Vec<Directive> {
324        binds
325            .iter()
326            .map(|&index| directives[index].clone())
327            .collect()
328    };
329    if let Some(error) = scan_error
330        && (parsed.is_ok() || cursor.read_past_end())
331    {
332        return Err((error, bound(&directives)));
333    }
334    match parsed {
335        Ok(main) => Ok(Parsed {
336            main,
337            directives,
338            symbols: scanned.symbols,
339        }),
340        Err(error) => Err((cursor.echo(error), bound(&directives))),
341    }
342}

344/// The grammar over the clauses `cursor` holds: the main body, then the
345/// directives into `directives`, with the index of each one whose `EXTERNAL` is
346/// resolved while translating in `binds`.
347fn translate<'a>(
348    ctx: &'a ParseCtx<'a>,
349    cursor: &mut ClauseCursor,
350    checked: bool,
351    directives: &mut Vec<Directive>,
352    binds: &mut Vec<usize>,
353) -> Result<CodeBody, ParseError> {
354    let main = translate_block(ctx, cursor)?;
355
356    // `translate` raises 99.914 exactly here, once, before the next directive is
357    // ever called: `INTERPRET` text may not carry a directive at all, so this is
358    // not a per-directive check. Measured via a `signal on syntax` trap around
359    // `interpret "::routine r"`: `condition('o')~code` is `99.914`.
360    if let Some(clause) = cursor.peek()
361        && ctx.source.kind() == SourceKind::Interpret
362    {
363        return Err(ParseError::raise(E99_914, clause.span.start, []));
364    }
365    if cursor.peek().is_none() {
366        return Ok(main);
367    }
368
369    let mut declared = if checked {
370        Declared::default()
371    } else {
372        Declared::unchecked()
373    };
374    while cursor.peek().is_some() {
375        let mut directive = parse_directive(ctx, cursor, &mut declared)?;
376        if let Some(required_level) = ctx.required_level {
377            // A directive's constant expression has no body-local labels.
378            required_level.finish_body(false);
379            if let DirectiveKind::Options(options) = &directive.kind
380                && options
381                    .iter()
382                    .any(|option| matches!(option, PackageOption::NumericInherit(_)))
383            {
384                required_level.level.set(RequiredLanguageLevel::V6_06);
385            }
386        }
387        if declared.bound {
388            binds.push(directives.len());
389        }
390        // Only `::METHOD`/`::ATTRIBUTE`/`::ROUTINE` can carry a body, and each
391        // already rejects one with its own specific error when its OWN shape
392        // does not allow it (`::CONSTANT`'s body is 99.938, for one). Every
393        // other directive kind never sets a body at all, and a clause trailing
394        // one of those needs no special case here: the NEXT iteration's
395        // `parse_directive` call sees a clause that does not start with `::` and
396        // raises 99.916 on its own -- measured for all five kinds that can never
397        // have a body (`::CLASS`, `::OPTIONS`, `::REQUIRES`, `::ANNOTATE`,
398        // `::RESOURCE`).
399        if let Some(slot) = directive_body(&mut directive.kind) {
400            *slot = translate_block(ctx, cursor)?;
401        }
402        directives.push(directive);
403    }
404    Ok(main)
405}

407/// The body slot of a directive that carries one, for the assembler to fill.
408pub(crate) fn directive_body(kind: &mut DirectiveKind) -> Option<&mut CodeBody> {
409    match kind {
410        DirectiveKind::Method(method) => method.body.as_deref_mut(),
411        DirectiveKind::Attribute(attribute) => attribute.body.as_deref_mut(),
412        DirectiveKind::Routine(routine) => routine.body.as_deref_mut(),
413        DirectiveKind::Annotate(_)
414        | DirectiveKind::Class(_)
415        | DirectiveKind::Constant(_)
416        | DirectiveKind::Options(_)
417        | DirectiveKind::Requires(_)
418        | DirectiveKind::Resource(_) => None,
419    }
420}
