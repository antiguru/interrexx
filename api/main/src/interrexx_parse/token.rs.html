17/// A parse-time error, identified the way the interpreter identifies it: a
18/// major number and a sub-number, as in `13.1` or `99.943`. One word, so the
19/// parser's `Result`s stay narrow: the fields live behind the box and read
20/// through `Deref` and [`ParseError::code`] and [`ParseError::sub`]. Nothing
21/// outside this module writes them, so the pair stays the one
22/// [`ParseError::raise`] was given, with the inserts its message needs.
23#[derive(Clone, PartialEq, Eq, Debug)]
24pub struct ParseError(Box<ParseErrorFields>);
25
26/// What a [`ParseError`] carries.
27#[derive(Clone, PartialEq, Eq, Debug)]
28pub struct ParseErrorFields {
29    /// The major error number, e.g. 13 for `Error 13: Invalid character in
30    /// program`.
31    pub(crate) code: u16,
32    /// The sub-number, e.g. 1 for `Error 13.1`. Zero means the major error
33    /// with no sub-code.
34    pub(crate) sub: u16,
35    /// Where the source the traceback line echoes ends; see [`ParseError::clause`].
36    /// Zero until the scanner or `parse` resolves it. An echo ending past
37    /// `u32::MAX` is cut there, and one starting past it is empty.
38    end: u32,
39    /// The byte offset the error is *reported against*, which is the start of
40    /// the clause being translated and not the offending character. The name
41    /// reads like the latter; it is not.
42    pub byte: usize,
43    /// The secondary message's inserts, in insert order.
44    inserts: Vec<Vec<u8>>,
45}
46
47impl std::ops::Deref for ParseError {
48    type Target = ParseErrorFields;
49
50    fn deref(&self) -> &ParseErrorFields {
51        &self.0
52    }
53}
54
55impl ParseError {
56    /// The error `msg` reports against the clause starting at `byte`, with
57    /// `inserts` for its `&1`, `&2`, ... in order.
58    #[cold]
59    #[inline(never)]
60    pub fn raise<const N: usize>(msg: Msg<N>, byte: usize, inserts: [Vec<u8>; N]) -> Self {
61        let (code, sub) = msg.code();
62        ParseError(Box::new(ParseErrorFields {
63            code,
64            sub,
65            end: 0,
66            byte,
67            inserts: Vec::from(inserts),
68        }))
69    }
70
71    /// The major error number, e.g. 13 for `Error 13: Invalid character in
72    /// program`.
73    pub fn code(&self) -> u16 {
74        self.0.code
75    }
76
77    /// The sub-number, e.g. 1 for `Error 13.1`. Zero means the major error
78    /// with no sub-code.
79    pub fn sub(&self) -> u16 {
80        self.0.sub
81    }
82
83    /// The values the secondary message substitutes, in insert order.
84    pub fn inserts(&self) -> &[Vec<u8>] {
85        &self.inserts
86    }
87
88    /// The source the error's traceback line echoes: the clause being
89    /// translated, from its first token to the last one scanned, as the stack
90    /// frame records it.
91    pub fn clause(&self) -> Range<usize> {
92        self.byte..usize::cast_from(self.end).max(self.byte)
93    }
94
95    /// This error reported against `clause` and echoing it, unless it already
96    /// echoes one.
97    pub(crate) fn echoing(mut self, clause: Range<usize>) -> Self {
98        if self.end == 0 {
99            self.0.byte = clause.start;
100            self.0.end = u32::try_from(clause.end).unwrap_or(u32::MAX);
101        }
102        self
103    }
104}

184/// What kind of thing a symbol names, from the symbol scan's classification.
185#[derive(Copy, Clone, PartialEq, Eq, Debug)]
186pub enum SymbolClass {
187    /// A lone `.`, the placeholder in a `PARSE` template.
188    Dummy,
189    /// Starts with a digit, or starts with `.` and scanned as a number: a
190    /// literal value, never a variable.
191    Constant,
192    /// Starts with `.` and is not a number, e.g. `.true` or `.array`.
193    DotSymbol,
194    /// A simple variable name, no periods.
195    Variable,
196    /// One period, at the end, e.g. `stem.`.
197    Stem,
198    /// A period that is not the only one or not at the end, e.g. `stem.i.j`.
199    Compound,
200}

202/// The 19 token classes, in a fixed order.
203#[derive(Clone, PartialEq, Eq, Debug)]
204pub enum TokenKind {
205    /// Never produced.
206    Null,
207    /// A blank that is an operator, under the two-sided rule in `scan`.
208    Blank,
209    /// A symbol. Carries the interned upcased spelling, not the source text:
210    /// the identity is the `SymbolId` and the occurrence is `Token::span`,
211    /// and neither substitutes for the other.
212    Symbol {
213        id: SymbolId,
214        class: SymbolClass,
215    },
216    /// A quoted literal, carrying its *decoded* value: doubled quotes
217    /// collapsed, and a `'…'x` or `'…'b` suffix already packed to bytes. This
218    /// is the one token kind that cannot be a slice of its own span, because
219    /// `'it''s'` has the value `it's`.
220    Literal {
221        value: Box<[u8]>,
222    },
223    Operator(Operator),
224    /// A clause terminator: `;`, an uncontinued line end, or end of file.
225    Eoc,
226    Comma,
227    /// Never produced.
228    Prefix,
229    LeftParen,
230    RightParen,
231    /// Never produced.
232    Point,
233    Colon,
234    Tilde,
235    DTilde,
236    LeftBracket,
237    RightBracket,
238    DColon,
239    /// Never produced: a `,` or `-` continuation is resolved inside the token
240    /// locator and becomes a blank or nothing.
241    Continue,
242    /// An operator immediately followed by `=`, e.g. `+=`, which the
243    /// interpreter treats as an assignment shortcut rather than as an
244    /// operator.
245    Assignment(Operator),
246}

248/// `TokenKind` without its payloads, for asserting token *shape*.
249#[derive(Copy, Clone, PartialEq, Eq, Debug)]
250pub enum Tag {
251    Null,
252    Blank,
253    Symbol,
254    Literal,
255    Operator,
256    Eoc,
257    Comma,
258    Prefix,
259    LeftParen,
260    RightParen,
261    Point,
262    Colon,
263    Tilde,
264    DTilde,
265    LeftBracket,
266    RightBracket,
267    DColon,
268    Continue,
269    Assignment,
270}
271
272impl TokenKind {
273    /// This kind with its payloads dropped.
274    pub fn tag(&self) -> Tag {
275        match self {
276            TokenKind::Null => Tag::Null,
277            TokenKind::Blank => Tag::Blank,
278            TokenKind::Symbol { .. } => Tag::Symbol,
279            TokenKind::Literal { .. } => Tag::Literal,
280            TokenKind::Operator(_) => Tag::Operator,
281            TokenKind::Eoc => Tag::Eoc,
282            TokenKind::Comma => Tag::Comma,
283            TokenKind::Prefix => Tag::Prefix,
284            TokenKind::LeftParen => Tag::LeftParen,
285            TokenKind::RightParen => Tag::RightParen,
286            TokenKind::Point => Tag::Point,
287            TokenKind::Colon => Tag::Colon,
288            TokenKind::Tilde => Tag::Tilde,
289            TokenKind::DTilde => Tag::DTilde,
290            TokenKind::LeftBracket => Tag::LeftBracket,
291            TokenKind::RightBracket => Tag::RightBracket,
292            TokenKind::DColon => Tag::DColon,
293            TokenKind::Continue => Tag::Continue,
294            TokenKind::Assignment(_) => Tag::Assignment,
295        }
296    }
297
298    /// Whether a blank *following* this token can be significant:
299    /// One half of the two-sided rule; the other half looks at what comes next.
300    pub fn makes_blank_significant(&self) -> bool {
301        matches!(
302            self,
303            TokenKind::Symbol { .. }
304                | TokenKind::Literal { .. }
305                | TokenKind::RightParen
306                | TokenKind::RightBracket
307        )
308    }
309}
