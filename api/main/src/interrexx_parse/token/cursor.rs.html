14/// Everything a `parse_*` function needs that is not the clause it is
15/// parsing.
16pub(crate) struct ParseCtx<'a> {
17    /// Read by the instruction parser for `SourceKind`, which decides whether
18    /// a label is error 47.1. The expression grammar needs only the tokens.
19    pub(crate) source: &'a ProgramSource,
20    pub(crate) tokens: &'a [Token],
21    /// Read-only by the time parsing starts: `scan` has already interned every
22    /// symbol in the program. It compares a clause's
23    /// first symbol against the pre-interned keyword ids and recovers a
24    /// label's spelling when `Program::labels` is built.
25    pub(crate) symbols: &'a SymbolTable,
26    /// The message names this parse has interned, which is what
27    /// `NodeKind::Message` holds one of.
28    pub(crate) selectors: &'a RefCell<SelectorTable>,
29    /// The arena of the code body being parsed, which the body takes when it
30    /// is finished. Every borrow of it lasts one statement, so no parse path
31    /// holds one across a call that takes another.
32    pub(crate) exprs: &'a RefCell<Exprs>,
33    /// Every reserved *spelling* this parser recognises, pre-interned by `scan`
34    /// before it reads any source, so their ids are fixed and every keyword
35    /// test is an integer comparison. Keywords are NOT reserved words, so this
36    /// is only ever consulted positionally.
37    pub(crate) keywords: &'a Keywords,
38    /// The `::RESOURCE` bodies `scan` copied out, keyed by the index of the
39    /// `::` token that opened each directive.
40    pub(crate) resources: &'a [ResourceBody],
41    /// Collected only when translating an image.
42    pub(crate) required_level: Option<&'a RequiredLanguage>,
43}
44
45impl ParseCtx<'_> {
46    /// Pushes a node over the children pushed just before it (see
47    /// [`Exprs::push`]).
48    pub(crate) fn push(&self, kind: NodeKind, extent: Range<usize>) -> ExprId {
49        self.exprs.borrow_mut().push(kind, extent)
50    }
51
52    /// Takes the arena built so far, leaving an empty one for the next body.
53    pub(crate) fn take_exprs(&self) -> Exprs {
54        std::mem::take(&mut *self.exprs.borrow_mut())
55    }
56
57    pub(crate) fn note_symbol_call(&self, id: SymbolId) {
58        if let Some(required_level) = self.required_level
59            && self.symbols.name(id) == "GC"
60        {
61            required_level.pending_gc.set(true);
62        }
63    }
64
65    pub(crate) fn note_literal_call(&self, name: &[u8]) {
66        if let Some(required_level) = self.required_level
67            && name == b"GC"
68        {
69            // A literal call skips internal labels.
70            required_level.level.set(RequiredLanguageLevel::V6_06);
71        }
72    }
73}
74
75#[derive(Default)]
76pub(crate) struct RequiredLanguage {
77    pub(crate) level: Cell<RequiredLanguageLevel>,
78    pending_gc: Cell<bool>,
79}
80
81impl RequiredLanguage {
82    /// The label may appear after the call, so resolve it after parsing the body.
83    pub(crate) fn finish_body(&self, has_gc_label: bool) {
84        if self.pending_gc.replace(false) && !has_gc_label {
85            self.level.set(RequiredLanguageLevel::V6_06);
86        }
87    }
88}

90/// A position inside one clause's token range, not inside the whole vector,
91/// so an expression parser cannot walk off the end of its clause.
92pub(crate) struct TokenCursor {
93    /// Index range into `ParseCtx::tokens` that this cursor may visit.
94    range: Range<usize>,
95    /// Next index to yield; always inside `range` or equal to `range.end`.
96    pos: usize,
97}
98
99impl TokenCursor {
100    /// The instruction parser builds one of these per clause, from
101    /// `Clause::tokens`; the expression grammar is handed one already built.
102    pub(crate) fn new(range: Range<usize>) -> Self {
103        Self {
104            pos: range.start,
105            range,
106        }
107    }
108
109    /// Index of the next token, or None at the end of the range.
110    pub(crate) fn peek(&self) -> Option<usize> {
111        (self.pos < self.range.end).then_some(self.pos)
112    }
113
114    /// Yield the next token index and step past it. Deliberately not called
115    /// `next`: `clippy::should_implement_trait` fires on an inherent `next`
116    /// with this signature, and the workspace runs clippy with `-D warnings`.
117    pub(crate) fn advance(&mut self) -> Option<usize> {
118        let i = self.peek()?;
119        self.pos += 1;
120        Some(i)
121    }
122
123    /// A next-real read without consuming: index of the next token that is not a
124    /// blank.
125    pub(crate) fn peek_real(&self, tokens: &[Token]) -> Option<usize> {
126        let mut i = self.peek()?;
127        while i < self.range.end && tokens[i].kind.tag() == Tag::Blank {
128            i += 1;
129        }
130        (i < self.range.end).then_some(i)
131    }
132
133    /// The next token that is not a blank, consumed.
134    pub(crate) fn advance_real(&mut self, tokens: &[Token]) -> Option<usize> {
135        let i = self.peek_real(tokens)?;
136        self.pos = i + 1;
137        Some(i)
138    }
139
140    pub(crate) fn position(&self) -> usize {
141        self.pos
142    }
143
144    /// Index of the first token this cursor may visit, whatever it has already
145    /// yielded.
146    pub(crate) fn start(&self) -> usize {
147        self.range.start
148    }
149
150    /// One past the last token index this cursor may visit.
151    pub(crate) fn end(&self) -> usize {
152        self.range.end
153    }
154}
