1//! Symbol interning: `SymbolId`, and the `SymbolTable` that hands them out.
2
3use interrexx_cast::{CastFrom, CastTruncate};
4use rustc_hash::FxBuildHasher;
5use std::hash::BuildHasher;
6
7/// A symbol's identity: the upcased spelling, interned. Two symbols with the
8/// same `SymbolId` name the same variable, method or label.
9#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
10pub struct SymbolId(u32);
11
12impl SymbolId {
13    /// This id as an index into the `SymbolTable` that interned it: **dense,
14    /// zero-based, and assigned in interning order**, so `n` distinct symbols
15    /// occupy exactly `0..n` and `SymbolTable::len` is the length a caller
16    /// needs to size a `Vec`.
17    pub fn index(self) -> usize {
18        usize::cast_from(self.0)
19    }
20
21    /// The id whose [`Self::index`] is `index`, where one can be.
22    pub(crate) fn from_index(index: usize) -> Option<Self> {
23        u32::try_from(index).ok().map(SymbolId)
24    }
25}
26
27/// Interns upcased symbol spellings. Owned by `ProgramSource`'s parse, handed
28/// to `Program` so the executor can resolve a `SymbolId` back to text for error
29/// messages and `SIGNAL`'s label lookup.
30///
31/// A table may sit on a frozen **base** table (`over`): the base's symbols keep
32/// their ids `0..base.len()`, and this table's own symbols follow from
33/// `base.len()`. The ids are then exactly those a table that first interned the
34/// base's symbols, in the base's order, would hand out, so ids stay dense and
35/// `len` still sizes a `Vec`. The scanner puts every chunk over one
36/// process-wide table of the keyword spellings, so compiling a short chunk (an
37/// `INTERPRET` fragment, a `Method~new` source) costs in proportion to the
38/// chunk rather than re-interning the ~150 keywords each time.
39///
40/// The index is a linear-probing table of its own rather than a `HashMap`, for
41/// two reasons a map cannot serve: a table over a base starts from a copy of
42/// the base's index (one small allocation and a copy, so a lookup is one hash
43/// and one probe sequence whichever table the symbol is in), and a symbol's
44/// spelling is stored once, in `names`, rather than once as a key and again
45/// for `name`.
46#[derive(Default, Debug)]
47pub struct SymbolTable {
48    base: Option<&'static SymbolTable>,
49    /// The index: a power-of-two number of slots, or none before the first
50    /// symbol. Each slot is empty or holds an id and its spelling's hash.
51    slots: Vec<Slot>,
52    /// This table's own spellings, upcased, by id less the base's length.
53    names: Vec<Box<str>>,
54}
55
56/// One index slot: `id_plus_one` 0 when empty, else the id plus one and the
57/// low half of its spelling's hash, which settles almost every mismatch
58/// without reading the spelling.
59#[derive(Copy, Clone, Default, Debug)]
60struct Slot {
61    hash: u32,
62    id_plus_one: u32,
63}
64
65/// The hash an upcased spelling's bytes are indexed by.
66fn hash_of(key: &[u8]) -> u32 {
67    // The low half: the index never has 2^32 slots.
68    u32::cast_truncate(FxBuildHasher.hash_one(key))
69}
70
71impl SymbolTable {
72    /// An empty table over the frozen `base`, which must itself have no base.
73    pub fn over(base: &'static SymbolTable) -> Self {
74        debug_assert!(base.base.is_none(), "a base table has no base of its own");
75        SymbolTable {
76            base: Some(base),
77            slots: base.slots.clone(),
78            names: Vec::new(),
79        }
80    }
81
82    /// How many ids the base holds: this table's own ids start here, and
83    /// every id below it names a keyword spelling every table shares.
84    pub fn base_len(&self) -> usize {
85        self.base.map_or(0, |base| base.names.len())
86    }
87
88    /// Intern `text`, upcasing it. Returns the same id for every spelling that
89    /// differs only in case.
90    pub fn intern(&mut self, text: &str) -> SymbolId {
91        // The upcased key is built without allocating whenever it fits a
92        // stack buffer, which every symbol a real program writes does: the
93        // lookup path is the common case by an order of magnitude, and source
94        // is mostly lower case. Only a genuine miss allocates, for the stored
95        // spelling.
96        // Bytes rather than `str` until a miss: the upcased copy is ASCII-
97        // upcased valid UTF-8, but proving that to `str` costs a validation per
98        // lookup.
99        let mut buffer = [0u8; 64];
100        let upcased: String;
101        let key: &[u8] = if !text.bytes().any(|b| b.is_ascii_lowercase()) {
102            text.as_bytes()
103        } else if let Some(bytes) = buffer.get_mut(..text.len()) {
104            bytes.copy_from_slice(text.as_bytes());
105            bytes.make_ascii_uppercase();
106            bytes
107        } else {
108            upcased = text.to_ascii_uppercase();
109            upcased.as_bytes()
110        };
111        let hash = hash_of(key);
112        let slot = match self.find(key, hash) {
113            Ok(id) => return id,
114            Err(slot) => slot,
115        };
116        let id = u32::try_from(self.len()).expect("symbols fit u32");
117        let spelling = std::str::from_utf8(key).expect("ASCII upcasing keeps UTF-8 valid");
118        self.names.push(spelling.into());
119        self.slots[slot] = Slot {
120            hash,
121            id_plus_one: id + 1,
122        };
123        // At most three quarters full, so a probe sequence stays short and
124        // always ends at an empty slot.
125        if self.len() * 4 > self.slots.len() * 3 {
126            self.grow();
127        }
128        SymbolId(id)
129    }
130
131    /// The id of `key`, or the empty slot where it belongs.
132    fn find(&mut self, key: &[u8], hash: u32) -> Result<SymbolId, usize> {
133        if self.slots.is_empty() {
134            self.slots = vec![Slot::default(); 16];
135        }
136        let mask = self.slots.len() - 1;
137        let mut at = usize::cast_from(hash) & mask;
138        loop {
139            let slot = self.slots[at];
140            if slot.id_plus_one == 0 {
141                return Err(at);
142            }
143            let id = SymbolId(slot.id_plus_one - 1);
144            if slot.hash == hash && self.name(id).as_bytes() == key {
145                return Ok(id);
146            }
147            at = (at + 1) & mask;
148        }
149    }
150
151    /// Doubles the index, re-placing every slot by its stored hash.
152    fn grow(&mut self) {
153        let doubled = vec![Slot::default(); self.slots.len() * 2];
154        let old = std::mem::replace(&mut self.slots, doubled);
155        let mask = self.slots.len() - 1;
156        for slot in old.into_iter().filter(|slot| slot.id_plus_one != 0) {
157            let mut at = usize::cast_from(slot.hash) & mask;
158            while self.slots[at].id_plus_one != 0 {
159                at = (at + 1) & mask;
160            }
161            self.slots[at] = slot;
162        }
163    }
164
165    /// The upcased spelling. Panics on an id from a different table, which is
166    /// a parser bug rather than a source error.
167    pub fn name(&self, id: SymbolId) -> &str {
168        let index = usize::cast_from(id.0);
169        match self.base {
170            Some(base) if index < base.names.len() => &base.names[index],
171            _ => &self.names[index - self.base_len()],
172        }
173    }
174
175    /// How many distinct symbols are interned, the base's included.
176    pub fn len(&self) -> usize {
177        self.base_len() + self.names.len()
178    }
179
180    /// Whether nothing is interned yet, in this table or its base.
181    pub fn is_empty(&self) -> bool {
182        self.len() == 0
183    }
184}
