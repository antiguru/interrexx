376    /// Runs `serve` against the host's [`Surface`], or records `slot` as
377    /// refused and answers `refused` where the host has none.
378    fn with_surface<R>(
379        &self,
380        slot: &'static str,
381        refused: R,
382        serve: impl FnOnce(&mut Conversion<'_>) -> R,
383    ) -> R {
384        let mut cx = self.conversion();
385        if cx.host.surface().is_none() {
386            refuse(slot);
387            return refused;
388        }
389        serve(&mut cx)
390    }

499    /// `ValuesToObject`: an array of each descriptor's object.
500    pub fn values_to_object(&self, described: &[(u16, Value, Option<Vec<u8>>)]) -> RexxObjectPtr {
501        // Each is registered as it is made, so that making the next cannot
502        // collect it.
503        let handles: Vec<RexxObjectPtr> = described
504            .iter()
505            .map(|(declared, value, text)| self.value_to_object(*declared, *value, text.as_deref()))
506            .collect();
507        let items: Vec<Option<ObjRef>> =
508            handles.iter().map(|handle| self.resolve(*handle)).collect();
509        self.with_surface(
510            "RexxThreadInterface.ValuesToObject",
511            std::ptr::null_mut(),
512            |cx| {
513                let surface = cx.host.surface().expect("checked by with_surface");
514                let array = surface.new_array(&items);
515                cx.host.register_local(array)
516            },
517        )
518    }

556    /// `RaiseCondition`, with `name` upper-cased as `new_upper_string` does.
557    pub fn raise_condition(
558        &self,
559        name: &[u8],
560        description: RexxObjectPtr,
561        additional: RexxObjectPtr,
562        result: RexxObjectPtr,
563    ) {
564        let (description, additional, result) = (
565            self.resolve(description),
566            self.resolve(additional),
567            self.resolve(result),
568        );

627    /// `DecodeConditionInfo` over the directory `handle` names, or `None`
628    /// for a handle this activation does not hold.
629    pub fn decode_condition(&self, handle: RexxObjectPtr) -> Option<DecodedCondition> {
630        let directory = self.resolve(handle)?;
631        self.with_surface("RexxThreadInterface.DecodeConditionInfo", None, |cx| {
632            let surface = cx.host.surface().expect("checked by with_surface");
633            let mut entry = |name: &[u8]| surface.directory_entry(directory, name);
634            let name = entry(b"CONDITION");
635            let code = entry(b"CODE");
636            let rc = entry(b"RC");
637            let position = entry(b"POSITION");
638            let (message, errortext, program) =
639                (entry(b"MESSAGE"), entry(b"ERRORTEXT"), entry(b"PROGRAM"));
640            let (description, additional) = (entry(b"DESCRIPTION"), entry(b"ADDITIONAL"));
641            // `RAISE` allows any object as the description and, but for
642            // `SYNTAX`, as the additional information: each is given only
643            // where it is of the type the structure declares.
644            let description = description.filter(|&object| surface.is_string(object));
645            let additional = additional.filter(|&object| surface.is_of_class(object, "Array"));
646            let mut handle = |found: Option<ObjRef>| {
647                found.map_or(std::ptr::null_mut(), |object| {
648                    cx.host.register_local(object)
649                })
650            };
651            let decoded = DecodedCondition {
652                code: 0,
653                rc: 0,
654                position: 0,
655                name: handle(name),
656                message: handle(message),
657                errortext: handle(errortext),
658                program: handle(program),
659                description: handle(description),
660                additional: handle(additional),
661            };
662            let text = |found: Option<ObjRef>, cx: &mut Conversion<'_>| {
663                Some(cx.host.string_value_text(found?))
664            };
665            let syntax = text(name, cx).is_some_and(|name| name == b"SYNTAX");
666            // Only a `SYNTAX` condition has an error code, and its `RC` is the
667            // error number; `ERROR` and `FAILURE` have a command's return
668            // code, any value, given where it is a whole number.
669            let (code, rc) = match syntax {
670                true => (
671                    text(code, cx).map_or(0, |code| message_number(&code)),
672                    text(rc, cx).map_or(0, |rc| message_number(&rc) / 1000),
673                ),
674                false => {
675                    let rc = rc
676                        .and_then(|rc| {
677                            cx.host
678                                .signed_integer(
679                                    rc,
680                                    -crate::values::MAX_WHOLENUMBER,
681                                    crate::values::MAX_WHOLENUMBER,
682                                )
683                                .ok()
684                                .flatten()
685                        })
686                        .and_then(|rc| isize::try_from(rc).ok())
687                        .unwrap_or(0);
688                    (0, rc)
689                }
690            };
691            Some(DecodedCondition {
692                code,
693                rc,
694                position: text(position, cx)
695                    .and_then(|position| std::str::from_utf8(&position).ok()?.parse().ok())
696                    .unwrap_or(0),
697                ..decoded
698            })
699        })
700    }

1034    /// [`Activation::send_to`], or [`Activation::send_own`] where `own` names
1035    /// the class.
1036    fn send_as(
1037        &self,
1038        slot: &'static str,
1039        receiver: Option<ObjRef>,
1040        own: Option<&'static str>,
1041        name: &[u8],
1042        arguments: &[Argument<'_>],
1043    ) -> Option<ObjRef> {
1044        let receiver = receiver?;
1045        let mut objects = Vec::with_capacity(arguments.len());
1046        for argument in arguments {
1047            objects.push(match argument {
1048                Argument::Handle(handle) => self.resolve(*handle),
1049                Argument::Text(bytes) => {
1050                    let mut cx = self.conversion();
1051                    let string = cx.host.new_string(bytes);
1052                    cx.host.register_local(string);
1053                    Some(string)
1054                }
1055                Argument::Number(number) => {
1056                    let mut cx = self.conversion();
1057                    let value = i64::try_from(*number).unwrap_or(i64::MAX);
1058                    let object = cx.host.whole_number(value);
1059                    cx.host.register_local(object);
1060                    Some(object)
1061                }
1062            });
1063        }
1064        self.with_surface(slot, None, |cx| {
1065            let surface = cx.host.surface().expect("checked by with_surface");
1066            // A receiver of another class than `own` names is sent to as any
1067            // message is, which is what the member did before it ran the
1068            // class's own method.
1069            let answered = match own.and_then(|id| surface.send_own(receiver, id, name, &objects)) {
1070                Some(answered) => answered,
1071                None => surface.send(receiver, name, None, &objects),
1072            };
1073            let answer = answered.ok().flatten()?;
1074            cx.host.register_local(answer);
1075            Some(answer)
1076        })
1077    }

2225/// `major.minor` as `major * 1000 + minor`, zero where it is not one.

2229    match (major.parse::<isize>(), minor.parse::<isize>()) {
2230        (Ok(major), Ok(minor)) if (1..100).contains(&major) && (0..1000).contains(&minor) => {
2231            major * 1000 + minor
2232        }
2233        _ => 0,
2234    }

2269/// each line feed, a carriage return before one dropped, and no empty line

2273        .split(|&byte| byte == b'\n')

2278    }
