1//! The constants the headers in `include/` define, by their C names.
2//!
3//! Each value is the one the interpreter uses: where Rust code already names
4//! the value, the table takes it from there, so the header and the code that
5//! reads what an extension passes cannot disagree. The header generator
6//! (`header.rs`) writes these tables, and the ABI dump (`tests/abi_dump.rs`)
7//! reads every value back through a C and a C++ compiler.

30/// How the header writes a constant's value.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub enum Form {
33    /// A decimal literal.
34    Decimal,
35    /// A hexadecimal literal, of at least `digits` digits.
36    Hex { digits: usize },
37    /// `REXX_OPTIONAL_ARGUMENT` ORed with `REXX_VALUE_<name>`.
38    Optional(&'static str),
39}
40
41/// One `#define` of a number.
42#[derive(Clone, Debug, PartialEq, Eq)]
43pub struct CConstant {
44    /// Its C name.
45    pub name: Cow<'static, str>,
46    /// Its value.
47    pub value: i64,
48    /// How the header writes the value.
49    pub form: Form,
50}
51
52impl CConstant {
53    /// The text the header defines the name as.
54    #[must_use]
55    pub fn text(&self) -> String {
56        match self.form {
57            Form::Decimal if self.value < 0 => format!("({})", self.value),
58            Form::Decimal => self.value.to_string(),
59            Form::Hex { digits } => format!("0x{:0digits$x}", self.value),
60            Form::Optional(name) => format!("(REXX_OPTIONAL_ARGUMENT | REXX_VALUE_{name})"),
61        }
62    }
63}
64
65const fn decimal(name: &'static str, value: i64) -> CConstant {
66    CConstant {
67        name: Cow::Borrowed(name),
68        value,
69        form: Form::Decimal,
70    }
71}
72
73const fn hex(name: &'static str, value: i64) -> CConstant {
74    CConstant {
75        name: Cow::Borrowed(name),
76        value,
77        form: Form::Hex { digits: 2 },
78    }
79}

81const fn version(name: &'static str, value: i64) -> CConstant {
82    CConstant {
83        name: Cow::Borrowed(name),
84        value,
85        form: Form::Hex { digits: 8 },
86    }
87}

375/// The value codes, in `oorexxapi.h`: the optional bit, `REXX_VALUE_<name>`
376/// for every row of the conversion table (`values.rs`) and every alias, and
377/// the optional forms.
378///
379/// # Panics
380/// If an alias or an optional form names a code the conversion table lacks.
381#[must_use]
382pub fn value_codes() -> Vec<CConstant> {
383    let code_of = |name: &str| -> i64 {
384        rows()
385            .find(|(_, row)| *row == name)
386            .map(|(code, _)| i64::from(code))
387            .unwrap_or_else(|| panic!("the conversion table has no code {name}"))
388    };
389    let named = |name: String, value: i64| CConstant {
390        name: Cow::Owned(name),
391        value,
392        form: Form::Decimal,
393    };
394    let mut out = vec![CConstant {
395        name: Cow::Borrowed("REXX_OPTIONAL_ARGUMENT"),
396        value: i64::from(OPTIONAL_ARGUMENT),
397        form: Form::Hex { digits: 4 },
398    }];
399    for (code, name) in rows() {
400        out.push(named(format!("REXX_VALUE_{name}"), i64::from(code)));
401    }
402    for (alias, name) in VALUE_ALIASES {
403        out.push(named(format!("REXX_VALUE_{alias}"), code_of(name)));
404    }
405    for name in OPTIONAL_VALUES {
406        out.push(CConstant {
407            name: Cow::Owned(format!("REXX_VALUE_OPTIONAL_{name}")),
408            value: i64::from(OPTIONAL_ARGUMENT) | code_of(name),
409            form: Form::Optional(name),
410        });
411    }
412    out
413}

484/// Defines [`ARGUMENT_TYPES`] and [`argument_type_names`] from each value
485/// code's name and the Rust type of the argument an entry receives for it.
486macro_rules! argument_types {
487    ($($name:ident: $ty:ty,)*) => {
488        /// Each value code's name and the tokens of the Rust type of the
489        /// argument an entry receives for it, `ARGUMENT_TYPE_<name>` in
490        /// `oorexxapi.h`.
491        pub const ARGUMENT_TYPES: &[(&str, &str)] = &[$((stringify!($name), stringify!($ty))),*];
492
493        /// Each value code's name and the Rust type of its argument, in the
494        /// order of [`ARGUMENT_TYPES`].
495        #[must_use]
496        pub fn argument_type_names() -> Vec<(&'static str, &'static str)> {
497            vec![$((stringify!($name), std::any::type_name::<$ty>())),*]
498        }
499    };
500}

556/// The value of the error code `Rexx_<symbol>` for message `major.sub`, which
557/// `oorexxerrors.h` defines for every message whose major code is below 100.
558#[must_use]
559pub fn error_code(major: u16, sub: u16) -> i64 {
560    i64::from(major) * 1000 + i64::from(sub)
561}

563/// Whether `oorexxerrors.h` defines a code for a message of major code
