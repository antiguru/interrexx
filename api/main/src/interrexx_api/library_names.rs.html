1//! The names the C API's two shared libraries go by on each host: the name a
2//! program or library that links them records, which the loader looks for,
3//! and how the one that forwards finds the one that owns the state. Build
4//! scripts include this file by path, with the target's `CARGO_CFG_TARGET_OS`;
5//! `librexxapi`'s build script hands the names to its source as constants, so
6//! that library links no code of this crate. Code that runs on the host reads
7//! [`host_api_libraries`].
8
9/// The C API's libraries on one host.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub struct ApiLibraries {
12    /// The interpreter's library, which owns all of the API's state.
13    pub librexx: &'static str,
14    /// The classic API's library, which forwards each name to [`ApiLibraries::librexx`].
15    pub librexxapi: &'static str,
16    /// The name [`ApiLibraries::librexx`] records for itself, which a program linked against it
17    /// records as its dependency: the SONAME on ELF hosts, the install name on Apple's.
18    pub librexx_id: &'static str,
19    /// The name [`ApiLibraries::librexxapi`] records for itself.
20    pub librexxapi_id: &'static str,
21    /// The path the forwarding library records for [`ApiLibraries::librexx`] and opens it by:
22    /// beside itself, where the two are installed together.
23    pub librexx_beside: &'static str,
24}
25
26/// The libraries on the host `target_os` names, as Rust spells it (`linux`, `macos`).
27/// Other Unix targets share the ELF names.
28#[must_use]
29pub fn api_libraries(target_os: &str) -> ApiLibraries {
30    match target_os {
31        "macos" | "ios" => ApiLibraries {
32            librexx: "librexx.4.dylib",
33            librexxapi: "librexxapi.4.dylib",
34            librexx_id: "@rpath/librexx.4.dylib",
35            librexxapi_id: "@rpath/librexxapi.4.dylib",
36            librexx_beside: "@loader_path/librexx.4.dylib",
37        },
38        "windows" => ApiLibraries {
39            librexx: "rexx.dll",
40            librexxapi: "rexxapi.dll",
41            librexx_id: "rexx.dll",
42            librexxapi_id: "rexxapi.dll",
43            librexx_beside: "rexx.dll",
44        },
45        _ => ApiLibraries {
46            librexx: "librexx.so.4",
47            librexxapi: "librexxapi.so.4",
48            librexx_id: "librexx.so.4",
49            librexxapi_id: "librexxapi.so.4",
50            librexx_beside: "librexx.so.4",
51        },
52    }
53}
54
55/// The libraries on the host this was compiled for.
56#[must_use]
57pub fn host_api_libraries() -> ApiLibraries {
58    api_libraries(std::env::consts::OS)
59}
