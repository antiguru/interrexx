50pub enum Body {
51    /// A value whose identity is its bytes (D15). `num` is a tri-state cache
52    /// of the one parse those bytes ever get: `None` is "not yet asked", and
53    /// must keep meaning exactly that -- treating it as "definitely not a
54    /// number" after some other value has been filled in answers wrongly for
55    /// a value nobody has asked about yet. The two `Result` arms tell "is a
56    /// number" from "is not" apart so a non-numeric string is not re-parsed
57    /// on every comparison.
58    Text {
59        bytes: Bytes,
60        num: Option<Result<Box<Number>, NotNumeric>>,
61    },
62    /// A value whose identity is its number (D15). `created_digits` and
63    /// `created_form` are the `NUMERIC DIGITS`/`NUMERIC FORM` in force when
64    /// this value was produced, and `text` -- formatted under exactly those,
65    /// never the settings in force when it is later read -- is fixed for the
66    /// object's whole lifetime once filled in.
67    Num {
68        value: Number,
69        created_digits: u32,
70        created_form: Form,
71        text: Option<Vec<u8>>,
72    },
73    /// A stem: its own name, an optional default, and its tails keyed by the
74    /// tail's value verbatim. A tail entry present and mapped to
75    /// `None` is a tombstone -- an explicitly dropped tail, which must not
76    /// fall back to the default -- and is distinct from the key being
77    /// absent, which does.
78    Stem {
79        name: Box<[u8]>,
80        default: Option<ObjRef>,
81        /// **Each tail's insertion ordinal is load-bearing.** `allIndexes`,
82        /// `supplier` and the other listings answer the tails in assignment
83        /// order (`r[class.stem.deviation.1]`), and a
84        /// `HashMap` cannot say which came first, so [`crate::Tails`] keeps
85        /// the ordinal with the value.
86        tails: crate::Tails,
87        /// The tails an `EXPOSE` or `PROCEDURE EXPOSE` made another stem's,
88        /// each naming that stem: an exposed compound element's real element.
89        /// Such a tail is also in `tails`, as a tombstone, so that it keeps its
90        /// place and holds no value of its own.
91        exposed: Option<Box<crate::NameMap<Vec<u8>, ObjRef>>>,
92    },
93    /// An array's slots, in index order, `None` for a slot that holds no
94    /// object at all.
95    Array {
96        slots: ArraySlots,
97        dimensions: Option<Box<[usize]>>,
98    },
99    /// A user-defined object: the class it belongs to, the behaviour it
100    /// dispatches against, the name something gave it, and its instance
101    /// variables, one pool per scope.
102    Instance {
103        class: ObjRef,
104        behaviour: BehaviourHandle,
105        name: Option<Box<[u8]>>,
106        pools: ScopePools,
107        own: Option<Box<ObjectMethods>>,
108        native: Option<Box<NativeState>>,
109    },
110    /// An object the interpreter builds for itself rather than one a program
111    /// constructs: `.environment`, `.local`, a package's `.methods` table and
112    /// an activation's `.context`.
113    Native(Box<NativeObject>),
114    /// A reference that does not keep its target alive. Traces to nothing --
115    /// that is the whole point -- and the collector rewrites the target to
116    /// `ObjRef::NIL` once it dies.
117    WeakRef(ObjRef),
118    /// A class object.
119    Class { owned: Vec<ObjRef> },
120    /// The object a `>name` term answers: a variable named rather than read.
121    VarRef(Box<VarRef>),
122    /// What keeps the storage of a variable a `>name` reference was taken
123    /// to: the index of that storage in [`crate::RootSet`]'s cells, which the
124    /// collector frees with this object, and the name of the variable that
125    /// owned it.
126    Cell(u32, Box<[u8]>),
127}

272/// Bytes whose first one is 16-byte aligned, with a length of their own.
273#[derive(Clone, Debug)]
274pub struct AlignedBytes {
275    words: Vec<Word>,
276    len: usize,
277}
278
279impl AlignedBytes {
280    /// `len` zero bytes, in at least one word, so that even an empty run has
281    /// an address no other shares.
282    pub fn zeroed(len: usize) -> AlignedBytes {
283        AlignedBytes {
284            words: vec![Word::default(); len.div_ceil(16).max(1)],
285            len,
286        }
287    }
288
289    pub fn len(&self) -> usize {
290        self.len
291    }
292
293    pub fn is_empty(&self) -> bool {
294        self.len == 0
295    }
296
297    /// A copy of the bytes.
298    pub fn to_vec(&self) -> Vec<u8> {
299        let mut bytes: Vec<u8> = self
300            .words
301            .iter()
302            .flat_map(|word| word.0.to_ne_bytes())
303            .collect();
304        bytes.truncate(self.len);
305        bytes
306    }
307
308    /// Writes `bytes` from `at`, as far as the length reaches.
309    pub fn write(&mut self, at: usize, bytes: &[u8]) {
310        for (offset, byte) in bytes.iter().enumerate() {
311            let index = at + offset;
312            if index >= self.len {
313                break;
314            }
315            let word = &mut self.words[index / 16];
316            let mut raw = word.0.to_ne_bytes();
317            raw[index % 16] = *byte;
318            word.0 = u128::from_ne_bytes(raw);
319        }
320    }
321
322    /// The address of the first byte.
323    pub fn as_mut_ptr(&mut self) -> *mut u8 {
324        self.words.as_mut_ptr().cast()
325    }
326}

666/// A `MutableBuffer`'s bytes, whose address native calls in flight may hold
667/// While any does ([`BufferBytes::lend`]), growth
668/// moves the bytes to new storage and keeps the old until the last of those
669/// calls ends ([`BufferBytes::release`]); a shrink keeps the storage.
670///
671/// Only the growth methods here may reallocate: the slice it dereferences to
672/// cannot, and [`BufferBytes::edit`] refuses to.
673#[derive(Debug)]
674pub struct BufferBytes {
675    bytes: Vec<u8>,
676    lent: usize,
677    retired: Vec<Vec<u8>>,
678}
679
680impl BufferBytes {
681    pub fn capacity(&self) -> usize {
682        self.bytes.capacity()
683    }
684
685    /// Room for `additional` bytes past the length.
686    pub fn try_reserve_exact(&mut self, additional: usize) -> Result<(), TryReserveError> {
687        if self.bytes.capacity() - self.bytes.len() >= additional {
688            return Ok(());
689        }
690        if self.lent == 0 {
691            return self.bytes.try_reserve_exact(additional);
692        }
693        let mut moved = Vec::new();
694        moved.try_reserve_exact(self.bytes.len().saturating_add(additional))?;
695        moved.extend_from_slice(&self.bytes);
696        self.retired.push(std::mem::replace(&mut self.bytes, moved));
697        Ok(())
698    }
699
700    /// [`BufferBytes::try_reserve_exact`], aborting where `Vec`'s own growth
701    /// would.
702    fn reserve_exact(&mut self, additional: usize) {
703        if self.try_reserve_exact(additional).is_err() {
704            std::alloc::handle_alloc_error(std::alloc::Layout::new::<u8>());
705        }
706    }
707
708    pub fn extend_from_slice(&mut self, bytes: &[u8]) {
709        self.reserve_exact(bytes.len());
710        self.bytes.extend_from_slice(bytes);
711    }
712
713    pub fn resize(&mut self, length: usize, value: u8) {
714        self.reserve_exact(length.saturating_sub(self.bytes.len()));
715        self.bytes.resize(length, value);
716    }
717
718    pub fn truncate(&mut self, length: usize) {
719        self.bytes.truncate(length);
720    }
721
722    pub fn clear(&mut self) {
723        self.bytes.clear();
724    }
725
726    /// `Vec::shrink_to`, which keeps lent storage where it is.
727    pub fn shrink_to(&mut self, capacity: usize) {
728        if self.lent == 0 {
729            self.bytes.shrink_to(capacity);
730        }
731    }
732
733    /// Runs `edit` over the bytes as a `Vec` that it may shorten or rewrite
734    /// but not grow past its capacity.
735    ///
736    /// # Panics
737    /// Where `edit` reallocated lent storage.
738    pub fn edit<R>(&mut self, edit: impl FnOnce(&mut Vec<u8>) -> R) -> R {
739        let address = self.bytes.as_ptr();
740        let answer = edit(&mut self.bytes);
741        assert!(
742            self.lent == 0 || std::ptr::eq(address, self.bytes.as_ptr()),
743            "a buffer edit reallocated lent storage"
744        );
745        answer
746    }
747
748    /// One more native call holds the bytes' address.
749    pub fn lend(&mut self) {
750        self.lent += 1;
751    }
752
753    /// One call that held the bytes' address has ended; the last frees the
754    /// storage growth moved them out of.
755    pub fn release(&mut self) {
756        self.lent = self.lent.saturating_sub(1);
757        if self.lent == 0 {
758            self.retired.clear();
759        }
760    }
761
762    /// How many storages growth has moved the bytes out of while lent.
763    pub fn retired(&self) -> usize {
764        self.retired.len()
765    }
766}

800    /// Room for `added` more bytes, taking the larger of what is needed and
801    /// twice the capacity. Answers how far the capacity grew.
802    pub fn ensure_capacity(&mut self, added: usize) -> Result<usize, TryReserveError> {
803        let needed = self.bytes.len().saturating_add(added);
804        if needed > self.capacity {
805            let capacity = needed.max(self.capacity.saturating_mul(2));
806            self.bytes.try_reserve_exact(capacity - self.bytes.len())?;
807            let grown = capacity - self.capacity;
808            self.capacity = capacity;
809            return Ok(grown);
810        }
811        Ok(0)
812    }
813
814    /// Room for `length` bytes in all. Only a length past the capacity grows it,
815    /// by [`Self::ensure_capacity`] for the bytes beyond the contents, so the
816    /// doubling rule still applies. Answers how far the capacity grew.
817    pub fn set_capacity(&mut self, length: usize) -> Result<usize, TryReserveError> {
818        if length > self.capacity {
819            return self.ensure_capacity(length.saturating_sub(self.bytes.len()));
820        }
821        Ok(0)
822    }
823
824    /// The bytes' address, writable up to the capacity answered with it.
825    ///
826    /// A copy's `Vec` holds only its length, so the allocation is grown to
827    /// the capacity here; where that fails the capacity drops to what the
828    /// allocation holds.
829    pub fn writable(&mut self) -> (*mut u8, usize) {
830        if self.bytes.capacity() < self.capacity
831            && self
832                .bytes
833                .try_reserve_exact(self.capacity.saturating_sub(self.bytes.len()))
834                .is_err()
835        {
836            self.capacity = self.bytes.capacity();
837        }
838        (self.bytes.as_mut_ptr(), self.capacity)
839    }
840
841    /// Zero empties the contents and takes the capacity back to `default_size`;
842    /// any other size becomes the capacity and truncates contents longer than
843    /// it. Answers how far the capacity grew.
844    pub fn set_buffer_size(&mut self, size: usize) -> Result<usize, TryReserveError> {
845        let before = self.capacity;
846        if size == 0 {
847            self.bytes.clear();
848            if self.capacity > self.default_size {
849                self.capacity = self.default_size;
850                self.bytes.shrink_to(self.default_size);
851            }
852        } else if size != self.capacity {
853            self.bytes.truncate(size);
854            if size > self.bytes.capacity() {
855                self.bytes.try_reserve_exact(size - self.bytes.len())?;
856            } else {
857                self.bytes.shrink_to(size);
858            }
859            self.capacity = size;
860        }
861        Ok(self.capacity.saturating_sub(before))
862    }
863}

893/// One name a [`ScopePools`] binds: the scope it is bound in, its name, and
894/// its value.
895#[derive(Clone, Debug)]
896pub struct PoolEntry {
897    pub scope: ObjRef,
898    pub name: Rc<[u8]>,
899    pub value: ObjRef,

902/// The variable pools of one object, held as one list of entries so that an
903/// object with a single scope costs one allocation.
904///
905/// **Each scope's entries are contiguous**, in binding order, and scopes
906/// follow one another in the order they first bound a name, **including a
907/// scope that has dropped every name and binds again**, which goes back where
908/// it was. A new name for a scope that already has entries is inserted after
909/// that scope's last entry, which keeps [`ScopePools::entries`] a slice and
910/// [`ScopePools::find`]'s "created last" order.
911///
912/// What remembers a scope's place once it has no entries is a tail of **order
913/// records**, written the first time a scope drops its last name and absent
914/// until then: one entry per scope that has bound a name, in first-binding
915/// order, with `ObjRef::MARKER` as their scope, no name, and the scope they
916/// stand for as their value. They are never read as variables, and sit after
917/// every real entry. The marker is not nil because a real scope can be: a
918/// method run by `~run` exposes in the nil scope.
919/// They are kept in the list, not beside it, because a field here would widen
920/// every object.
921///
922/// A name with no value is **absent** rather than present-and-empty: `DROP`
923/// removes the entry, and a reader cannot tell the two apart -- measured, a
924/// class method exposing `v`, assigning it, and dropping it leaves a later
925/// `expose v` reading the derived name `V`, which is what a never-bound name
926/// reads as. A name is shared rather than owned so that a caller binding the
927/// same names on many objects (a hash store's five) can hand over a copy of
928/// one it keeps instead of allocating each afresh.
929#[derive(Clone, Debug, Default)]
930pub struct ScopePools {
931    entries: Vec<PoolEntry>,
932}

941    /// The value `name` holds in `scope`'s pool, or `None` when the pool does
942    /// not hold it -- which is the uninitialised state and not an error.
943    pub fn get(&self, scope: ObjRef, name: &[u8]) -> Option<ObjRef> {
944        self.entries
945            .iter()
946            .find(|entry| entry.scope == scope && *entry.name == *name)
947            .map(|entry| entry.value)
948    }

950    /// [`ScopePools::get`], tried first where `hint` says the name last was.
951    #[inline]
952    pub fn get_hinted(&self, scope: ObjRef, name: &[u8], hint: &PoolHint) -> Option<ObjRef> {
953        if let Some(entry) = self.entries.get(hint.get())
954            && entry.scope == scope
955            && *entry.name == *name
956        {
957            return Some(entry.value);
958        }
959        let at = self
960            .entries
961            .iter()
962            .position(|entry| entry.scope == scope && *entry.name == *name)?;
963        hint.set(at);
964        Some(self.entries[at].value)
965    }

967    /// [`ScopePools::set`], tried first where `hint` says the name last was.
968    #[inline]
969    pub fn set_hinted(&mut self, scope: ObjRef, name: &[u8], value: ObjRef, hint: &PoolHint) {
970        if let Some(entry) = self.entries.get_mut(hint.get())
971            && entry.scope == scope
972            && *entry.name == *name
973        {
974            entry.value = value;
975            return;
976        }
977        let at = self.bind(scope, name, value);
978        hint.set(at);
979    }

987    /// Binds `name` to `value` in `scope`'s pool, and answers where its entry
988    /// is.
989    fn bind(&mut self, scope: ObjRef, name: &[u8], value: ObjRef) -> usize {
990        match self
991            .entries
992            .iter()
993            .position(|entry| entry.scope == scope && *entry.name == *name)
994        {
995            Some(at) => {
996                self.entries[at].value = value;
997                at
998            }
999            None => self.insert(scope, name.into(), value),
1000        }
1001    }

1003    /// Adds an entry the pool does not hold, after the last entry of `scope`'s
1004    /// run, and answers where it went. A scope with no entries goes ahead of
1005    /// the first scope that follows it in first-binding order, or at the end.
1006    fn insert(&mut self, scope: ObjRef, name: Rc<[u8]>, value: ObjRef) -> usize {
1007        let at = match self.entries.iter().rposition(|entry| entry.scope == scope) {
1008            Some(last) => last + 1,
1009            None if self
1010                .entries
1011                .last()
1012                .is_none_or(|last| !is_order_record(last)) =>
1013            {
1014                self.entries.len()
1015            }
1016            None => self.place_new_run(scope),
1017        };
1018        self.entries.insert(at, PoolEntry { scope, name, value });
1019        at
1020    }
1021
1022    /// Where the first entry of `scope`, which has none, goes when order
1023    /// records exist: ahead of the run of the first scope after it in
1024    /// first-binding order that still has entries, or after the last real
1025    /// entry. Records `scope` as bound if it is new.
1026    #[cold]
1027    #[inline(never)]
1028    fn place_new_run(&mut self, scope: ObjRef) -> usize {
1029        let real = self
1030            .entries
1031            .iter()
1032            .rposition(|entry| !is_order_record(entry))
1033            .map_or(0, |last| last + 1);
1034        let Some(rank) = self.entries[real..]
1035            .iter()
1036            .position(|record| record.value == scope)
1037        else {
1038            // Order records exist here, so the last entry is one, and its
1039            // empty name is shared rather than allocated afresh.
1040            let empty = self.entries.last().map(|last| Rc::clone(&last.name));
1041            self.entries.push(order_record(
1042                scope,
1043                empty.unwrap_or_else(|| Rc::from(&b""[..])),
1044            ));
1045            return real;
1046        };
1047        self.entries[real + rank + 1..]
1048            .iter()
1049            .find_map(|later| {
1050                self.entries[..real]
1051                    .iter()
1052                    .position(|entry| entry.scope == later.value)
1053            })
1054            .unwrap_or(real)
1055    }

1057    /// [`Self::set`] for each of `entries` in turn, with each name a copy of
1058    /// the caller's own, and, on an empty list, room for all of them made at
1059    /// once.
1060    pub fn set_all(&mut self, scope: ObjRef, entries: &[(&Rc<[u8]>, ObjRef)]) {
1061        if self.entries.is_empty() {
1062            self.entries.reserve_exact(entries.len());
1063        }
1064        for &(name, value) in entries {
1065            match self
1066                .entries
1067                .iter_mut()
1068                .find(|entry| entry.scope == scope && *entry.name == **name)
1069            {
1070                Some(entry) => entry.value = value,
1071                None => {
1072                    self.insert(scope, Rc::clone(name), value);
1073                }
1074            }
1075        }
1076    }

1078    /// Binds `value` to the entry of `scope`'s pool whose name is `name`
1079    /// itself, the same allocation rather than equal bytes, and answers
1080    /// whether there was one. A caller that bound a name from a copy it keeps
1081    /// finds its entry this way without comparing any name's bytes.
1082    #[inline]
1083    pub fn set_shared(&mut self, scope: ObjRef, name: &Rc<[u8]>, value: ObjRef) -> bool {
1084        match self
1085            .entries
1086            .iter_mut()
1087            .find(|entry| entry.scope == scope && Rc::ptr_eq(&entry.name, name))
1088        {
1089            Some(entry) => {
1090                entry.value = value;
1091                true
1092            }
1093            None => false,
1094        }
1095    }

1097    /// Returns `name` to the uninitialised state in `scope`'s pool, which is
1098    /// what `DROP` on an exposed variable does. A name the pool never held is
1099    /// a no-op, exactly as `DROP` on a never-assigned local is.
1100    pub fn clear(&mut self, scope: ObjRef, name: &[u8]) {
1101        let Some(at) = self
1102            .entries
1103            .iter()
1104            .position(|entry| entry.scope == scope && *entry.name == *name)
1105        else {
1106            return;
1107        };
1108        let shares_run = (at > 0 && self.entries[at - 1].scope == scope)
1109            || self
1110                .entries
1111                .get(at + 1)
1112                .is_some_and(|next| next.scope == scope);
1113        if !shares_run
1114            && self
1115                .entries
1116                .last()
1117                .is_none_or(|last| !is_order_record(last))
1118        {
1119            self.write_order_records();
1120        }
1121        self.entries.remove(at);
1122    }

1124    /// Appends the order records, from the list as it stands, which still
1125    /// holds the scope about to leave it.
1126    #[cold]
1127    #[inline(never)]
1128    fn write_order_records(&mut self) {
1129        let mut seen: Vec<ObjRef> = Vec::new();
1130        for entry in &self.entries {
1131            if seen.last() != Some(&entry.scope) {
1132                seen.push(entry.scope);
1133            }
1134        }
1135        let empty: Rc<[u8]> = Rc::from(&b""[..]);
1136        self.entries.extend(
1137            seen.into_iter()
1138                .map(|scope| order_record(scope, Rc::clone(&empty))),
1139        );
1140    }

1152    /// Every name `scope`'s pool binds, with its value, in binding order.
1153    pub fn entries(&self, scope: ObjRef) -> &[PoolEntry] {
1154        let Some(start) = self.entries.iter().position(|entry| entry.scope == scope) else {
1155            return &[];
1156        };
1157        let run = self.entries[start..]
1158            .iter()
1159            .take_while(|entry| entry.scope == scope)
1160            .count();
1161        &self.entries[start..start + run]
1162    }

1164    /// The first `N` entries of `scope`'s run, when they all belong to
1165    /// `scope`, found without measuring the run.
1166    #[inline]
1167    pub fn leading<const N: usize>(&self, scope: ObjRef) -> Option<&[PoolEntry; N]> {
1168        let start = self.entries.iter().position(|entry| entry.scope == scope)?;
1169        let first = self.entries[start..].first_chunk::<N>()?;
1170        first
1171            .iter()
1172            .all(|entry| entry.scope == scope)
1173            .then_some(first)
1174    }

1176    /// Appends every object these pools reach: each scope's own identity and
1177    /// every value bound in it.
1178    fn trace(&self, out: &mut Vec<ObjRef>) {
1179        let mut previous = None;
1180        for entry in &self.entries {
1181            if is_order_record(entry) {
1182                // The scope it stands for is reached through this record when it
1183                // has no entries. One that still has entries is pushed again
1184                // here, which the collector's marks absorb.
1185                out.push(entry.value);
1186                continue;
1187            }
1188            if previous != Some(entry.scope) {
1189                out.push(entry.scope);
1190                previous = Some(entry.scope);
1191            }
1192            out.push(entry.value);
1193        }
1194    }

1332/// What a `Method`, `Routine` or `Package` object is bound to, sharing one
1333/// slot so that every other native object stays its size.

1336    Nothing,
1337    /// A method object's scope.
1338    Scope(ObjRef),
1339    /// The package a `Method` or `Routine` object's code belongs to.

1349    /// stays its size.

1597    /// Appends every object this one can reach.
1598    pub fn trace(&self, out: &mut Vec<ObjRef>) {
1599        match self {
1600            // Neither reaches an `ObjRef`: a `Number` and a byte string are
1601            // both plain data, never a handle into the heap.
1602            // `Heap::collect`'s mark loop relies on this: it reads a `Text`
1603            // survivor's bytes and never calls `trace` on it.
1604            Body::Text { .. } => {}
1605            Body::Num { .. } => {}
1606            Body::Stem {
1607                default,
1608                tails,
1609                exposed,
1610                ..
1611            } => {
1612                out.extend(default.iter().copied());
1613                // A tombstone (`None`) reaches nothing, same as a weak
1614                // reference clearing to `.nil` -- it is present but dead.
1615                out.extend(tails.values().flatten());
1616                out.extend(exposed.iter().flat_map(|exposed| exposed.values().copied()));
1617            }
1618            // An empty slot reaches nothing, the same as a stem's tombstone
1619            // above.
1620            Body::Array { slots, .. } => out.extend(slots.iter().filter_map(|item| *item)),
1621            // Every scope's pool, walked by the storage's own type -- see
1622            // [`ScopePools::trace`] for why the walk lives there.
1623            // The class handle is traced for the reason the `Native` arm
1624            // below traces its own: it names no arena slot today
1625            // ([`crate::CLASS_SLOT_BASE`]), and the arm stays correct on the
1626            // day a class object is allocated like anything else.
1627            Body::Instance {
1628                class,
1629                pools,
1630                own,
1631                native,
1632                ..
1633            } => {
1634                out.push(*class);
1635                pools.trace(out);
1636                if let Some(native) = native {
1637                    native.trace(out);
1638                }
1639                // A one-off method's scope, in the position the class handle
1640                // above is in and traced for the same reason.
1641                if let Some(own) = own {
1642                    own.trace(out);
1643                }
1644            }
1645            Body::Native(native) => {
1646                // The class handle travels with the values. It names no arena
1647                // slot today ([`crate::CLASS_SLOT_BASE`]), so the collector
1648                // drops it on the floor; tracing it anyway is what keeps this
1649                // arm correct on the day a class object is allocated like
1650                // anything else.
1651                out.push(native.class);
1652                out.extend(native.entries.values().copied());
1653                // A method object's scope, in the position the class handle
1654                // above is in and traced for the same reason.
1655                out.extend(native.scope());
1656                out.extend(native.executable_package());
1657                // Reachable from the object alone once a program holds the
1658                // table and drops every other handle on it.
1659                out.extend(native.annotations());
1660                if let Some(state) = native.package() {
1661                    state.trace(out);
1662                }
1663            }
1664            // Deliberately reaches nothing: a weak reference must not keep
1665            // its target alive.
1666            Body::WeakRef(_) => {}
1667            Body::Class { owned } => out.extend(owned.iter().copied()),
1668            // The owning object of an instance variable is traced because the
1669            // reference may be the only handle a program still has on it.
1670            Body::VarRef(reference) => match reference.home {
1671                VarRefHome::Cell(slot) => out.extend(slot.cell()),
1672                VarRefHome::Instance { owner, scope } => {
1673                    out.push(owner);
1674                    out.push(scope);
1675                }
1676            },
1677            // The value is in the root set's cell, which `Heap::collect`
1678            // marks when it marks this object.
1679            Body::Cell(..) => {}
1680        }
1681    }
1682}
