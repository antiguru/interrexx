1//! A reference to a Rexx object.
2
3use std::marker::PhantomData;
4
5use interrexx_cast::{CastFrom, u8_truncate_u64, u32_truncate_u64, u64_from_u8, u64_from_u32};
6
7const TAG_BITS: u32 = 2;
8const TAG_MASK: u64 = 0b11;
9const TAG_HEAP: u64 = 0b00;
10const TAG_INT: u64 = 0b01;
11const TAG_NIL: u64 = 0b10;
12/// Short byte strings, stored in the handle itself.
13const TAG_TEXT: u64 = 0b11;
14
15/// How many bytes fit in a handle.
16pub const INLINE_TEXT: usize = 7;
17
18const TEXT_LEN_SHIFT: u32 = TAG_BITS;
19const TEXT_LEN_MASK: u64 = 0b111;
20/// Bytes start at a byte boundary so that decoding is one shift and one
21/// `to_le_bytes`, rather than a shift per byte. The three bits between the
22/// length and the first byte are unused and always zero.
23const TEXT_DATA_SHIFT: u32 = 8;
24
25const SLOT_SHIFT: u32 = TAG_BITS;
26const SLOT_BITS: u32 = 32;
27const SLOT_MASK: u64 = (1 << SLOT_BITS) - 1;
28const GEN_SHIFT: u32 = SLOT_SHIFT + SLOT_BITS;
29const GEN_BITS: u32 = 30;
30
31/// The highest generation a slot can reach. A slot that would exceed this is
32/// retired rather than reused, so a stale handle can never alias a live one.
33pub const GENERATION_MAX: u32 = (1 << GEN_BITS) - 1;
34
35/// Inclusive bounds of the inline integer range.
36pub const SMALL_INT_MAX: i64 = (1 << 61) - 1;
37pub const SMALL_INT_MIN: i64 = -(1 << 61);
38
39/// Not [`Send`] or [`Sync`]: a handle is meaningless off the island that
40/// allocated its slot, and only `interrexx`'s `Islanded` carries one to
41/// another OS thread (design section 2.5).
42///
43/// ```compile_fail
44/// fn require_send<T: Send>(_: T) {}
45/// require_send(interrexx_core::ObjRef::NIL);
46/// ```
47///
48/// ```compile_fail
49/// fn require_sync<T: Sync>(_: T) {}
50/// require_sync(interrexx_core::ObjRef::NIL);
51/// ```
52///
53/// A type that actually is `Send` and `Sync` passes the same two checks, so
54/// the failures above are `ObjRef`'s doing and not some unrelated mismatch:
55///
56/// ```
57/// fn require_send<T: Send>(_: T) {}
58/// fn require_sync<T: Sync>(_: T) {}
59/// require_send(0u64);
60/// require_sync(0u64);
61/// ```
62#[derive(Copy, Clone, PartialEq, Eq, Hash, Debug)]
63/// `repr(transparent)`, so a register holding one is the `u64` itself, which
64/// is what code generated for a frame reads and writes.
65#[repr(transparent)]
66pub struct ObjRef(u64, PhantomData<*const ()>);
67
68/// A byte string held in the handle, with no heap object behind it.
69#[derive(Copy, Clone, PartialEq, Eq, Debug)]
70pub struct InlineText {
71    len: u8,
72    buf: [u8; INLINE_TEXT],
73}
74
75impl std::ops::Deref for InlineText {
76    type Target = [u8];
77
78    fn deref(&self) -> &[u8] {
79        &self.buf[..usize::from(self.len)]
80    }
81}

83#[derive(Copy, Clone, PartialEq, Eq, Debug)]
84pub enum Decoded {
85    Heap {
86        slot: u32,
87        generation: u32,
88    },
89    SmallInt(i64),
90    /// Bytes carried in the handle. See [`InlineText`].
91    Text(InlineText),
92    Nil,
93}
94
95impl ObjRef {
96    pub const NIL: ObjRef = ObjRef(TAG_NIL, PhantomData);
97
98    /// A handle equal to no handle a program can hold, for a list of handles
99    /// that needs a marker: nil's tag with a payload. The field is private to
100    /// this module, and each constructor here sets its own tag, nil's alone
101    /// with no payload; the code generated for a frame builds handles only as
102    /// small integers, nil and the two logicals, and otherwise copies handles
103    /// it was given. Never handed to a program, so it is never decoded.
104    pub(crate) const MARKER: ObjRef = ObjRef(TAG_NIL | (1 << TAG_BITS), PhantomData);
105
106    /// The handle's own bits, for a caller that has to *index* by identity
107    /// rather than compare two handles.
108    pub const fn bits(self) -> u64 {
109        self.0
110    }
111
112    /// The heap reference `bits` are, if they carry the heap tag: what an
113    /// extension's handle names once the table that minted it is gone. The
114    /// slot may be free or reused; the heap's own lookup decides.
115    pub fn heap_bits(bits: u64) -> Option<Self> {
116        (bits & TAG_MASK == TAG_HEAP).then_some(ObjRef(bits, PhantomData))
117    }
118
119    /// The value `bits` carries, if they are exactly the bits of an integer,
120    /// a short string or `.nil` held in the handle itself; `None` for a heap
121    /// reference and for bits no constructor produces. Such a value names no
122    /// heap object, so it is the same value wherever the bits come from.
123    pub fn carried(bits: u64) -> Option<Self> {
124        let candidate = ObjRef(bits, PhantomData);
125        let canonical = match candidate.decode() {
126            Decoded::Heap { .. } => return None,
127            Decoded::SmallInt(value) => ObjRef::small_int(value)?,
128            Decoded::Text(text) => ObjRef::inline_text(&text)?,
129            Decoded::Nil => ObjRef::NIL,
130        };
131        (canonical == candidate).then_some(candidate)
132    }
133
134    pub const fn heap(slot: u32, generation: u32) -> Self {
135        debug_assert!(generation <= GENERATION_MAX);
136        ObjRef(
137            (u64_from_u32(generation) << GEN_SHIFT) | (u64_from_u32(slot) << SLOT_SHIFT) | TAG_HEAP,
138            PhantomData,
139        )
140    }
141
142    pub const fn small_int(value: i64) -> Option<Self> {
143        if value > SMALL_INT_MAX || value < SMALL_INT_MIN {
144            return None;
145        }
146        Some(ObjRef(
147            ((value.cast_unsigned() << TAG_BITS) & !TAG_MASK) | TAG_INT,
148            PhantomData,
149        ))
150    }
151
152    /// A handle carrying `bytes` itself, or `None` if there are too many.
153    #[inline]
154    pub fn inline_text(bytes: &[u8]) -> Option<Self> {
155        if bytes.len() > INLINE_TEXT {
156            return None;
157        }
158        let data = pack_short(bytes);
159        Some(ObjRef(
160            (data << TEXT_DATA_SHIFT) | (u64::cast_from(bytes.len()) << TEXT_LEN_SHIFT) | TAG_TEXT,
161            PhantomData,
162        ))
163    }
164
165    /// The inline-text handle for a single byte, as a constant.
166    pub const fn inline_byte(byte: u8) -> Self {
167        ObjRef(
168            (u64_from_u8(byte) << TEXT_DATA_SHIFT) | (1 << TEXT_LEN_SHIFT) | TAG_TEXT,
169            PhantomData,
170        )
171    }
172
173    /// **`always` rather than a hint**, and the difference was measured
174    /// rather than assumed. Under this workspace's `lto = "fat"` and
175    /// `codegen-units = 1`, a plain `#[inline]` left the out-of-line symbol
176    /// in the binary and moved `instructions:u` on every benchmark program
177    /// by 0.000%; the callers this matters in are large enough that LLVM's
178    /// cost model declines them. With `always`, `bench-programs/emptyloop.
179    /// rex` is -3.23% and `varlookup.rex` -1.43%, and `.text` does not grow.
180    #[inline(always)]
181    pub const fn decode(self) -> Decoded {
182        match self.0 & TAG_MASK {
183            TAG_HEAP => Decoded::Heap {
184                slot: u32_truncate_u64((self.0 >> SLOT_SHIFT) & SLOT_MASK),
185                generation: u32_truncate_u64(self.0 >> GEN_SHIFT),
186            },
187            TAG_INT => Decoded::SmallInt(self.0.cast_signed() >> TAG_BITS),
188            TAG_TEXT => {
189                let len = u8_truncate_u64((self.0 >> TEXT_LEN_SHIFT) & TEXT_LEN_MASK);
190                let full = (self.0 >> TEXT_DATA_SHIFT).to_le_bytes();
191                Decoded::Text(InlineText {
192                    len,
193                    buf: [
194                        full[0], full[1], full[2], full[3], full[4], full[5], full[6],
195                    ],
196                })
197            }
198            _ => Decoded::Nil,
199        }
200    }
201}
202
203/// `bytes`, at most eight of them, as a little-endian `u64` with the bytes
204/// past the end zero.
205///
206/// Two loads that may overlap, of the widest power of two the length holds,
207/// rather than a copy per byte or a `memcpy` call: a short slice is what
208/// every handle-sized string is built from.
209#[inline(always)]
210fn pack_short(bytes: &[u8]) -> u64 {
211    let len = bytes.len();
212    debug_assert!(len <= 8);
213    if len >= 4 {
214        let low = u64::from(u32::from_le_bytes([bytes[0], bytes[1], bytes[2], bytes[3]]));
215        let at = len - 4;
216        let high = u64::from(u32::from_le_bytes([
217            bytes[at],
218            bytes[at + 1],
219            bytes[at + 2],
220            bytes[at + 3],
221        ]));
222        low | (high << (8 * at))
223    } else if len >= 2 {
224        let low = u64::from(u16::from_le_bytes([bytes[0], bytes[1]]));
225        let at = len - 2;
226        let high = u64::from(u16::from_le_bytes([bytes[at], bytes[at + 1]]));
227        low | (high << (8 * at))
228    } else if len == 1 {
229        u64::from(bytes[0])
230    } else {
231        0
232    }
233}

235#[cfg(test)]
236mod inline_byte_tests {
237    use super::*;
238
239    /// The `const` route and the slice route must agree, or every caller that
240    /// compares against a constant logical silently stops matching.
241    #[test]
242    fn a_single_byte_inlines_the_same_way_either_route() {
243        for byte in 0u8..=255 {
244            assert_eq!(
245                Some(ObjRef::inline_byte(byte)),
246                ObjRef::inline_text(&[byte]),
247                "byte {byte} disagrees between the const and the slice route"
248            );
249        }
250    }
251
252    /// Every length a handle holds packs to the bytes in order with zeros
253    /// past the end, which is what decoding reads back.
254    #[test]
255    fn short_slices_pack_byte_for_byte() {
256        let source = [0x11u8, 0x22, 0x33, 0x44, 0x55, 0x66, 0x77, 0x88];
257        for len in 0..=8 {
258            let mut expected = [0u8; 8];
259            expected[..len].copy_from_slice(&source[..len]);
260            assert_eq!(
261                pack_short(&source[..len]),
262                u64::from_le_bytes(expected),
263                "length {len}"
264            );
265        }
266        for len in 0..=INLINE_TEXT {
267            let handle = ObjRef::inline_text(&source[..len]).expect("a handle-sized slice");
268            let Decoded::Text(text) = handle.decode() else {
269                panic!("an inline text handle decoded as something else");
270            };
271            assert_eq!(&*text, &source[..len]);
272        }
273    }
274}
