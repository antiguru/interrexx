1//! A [`Body::Text`]'s bytes, held inline while they are few.
2
3use std::fmt;
4use std::mem::MaybeUninit;
5use std::ops::Deref;
6
7/// How many bytes fit without a heap allocation.
8pub const INLINE_BYTES: usize = 54;
9
10/// **This is the payload width the capacity above is chosen against**, and it
11/// is asserted rather than described because a layout claim defended by a
12/// sentence has rotted in this repository before. `Body`'s and `Slot`'s own
13/// bounds are the two that matter to the arena; this one says which capacity
14/// produced them.
15const _: () = assert!(size_of::<Bytes>() <= 56);
16
17/// A byte string, stored inline when it is at most [`INLINE_BYTES`] long.
18/// Immutable but for [`Bytes::extend_from_slice`], whose doc says when that
19/// is allowed.
20#[derive(Clone)]
21pub struct Bytes(Repr);
22
23/// What a debug build writes past `len`, so that a read which should never
24/// happen produces something a test can recognise. `0xAB` is not a byte any
25/// test source here produces.
26#[cfg(debug_assertions)]
27const POISON: u8 = 0xAB;
28
29#[derive(Clone)]
30enum Repr {
31    Inline {
32        len: u8,
33        buf: [MaybeUninit<u8>; INLINE_BYTES],
34    },
35    Heap(Vec<u8>),
36}
37
38impl Bytes {
39    /// A copy of `source`, inline when it fits.
40    pub fn from_slice(source: &[u8]) -> Bytes {
41        if source.len() <= INLINE_BYTES {
42            let mut buf = [MaybeUninit::<u8>::uninit(); INLINE_BYTES];
43            for (slot, byte) in buf.iter_mut().zip(source) {
44                slot.write(*byte);
45            }
46            // See the type's doc: this exists so the invariant has a test.
47            #[cfg(debug_assertions)]
48            for slot in &mut buf[source.len()..] {
49                slot.write(POISON);
50            }
51            Bytes(Repr::Inline {
52                len: u8::try_from(source.len()).expect("at most INLINE_BYTES"),
53                buf,
54            })
55        } else {
56            Bytes(Repr::Heap(source.to_vec()))
57        }
58    }
59
60    /// `left`, then `separator` if there is one, then `right`, inline when the
61    /// whole fits: a concatenation's result written straight into its body
62    /// rather than joined somewhere else and copied in.
63    pub fn joined(left: &[u8], separator: Option<u8>, right: &[u8]) -> Bytes {
64        let total = left.len() + usize::from(separator.is_some()) + right.len();
65        if total > INLINE_BYTES {
66            let mut joined = Vec::with_capacity(total);
67            joined.extend_from_slice(left);
68            joined.extend(separator);
69            joined.extend_from_slice(right);
70            return Bytes(Repr::Heap(joined));
71        }
72        let mut buf = [MaybeUninit::<u8>::uninit(); INLINE_BYTES];
73        for (slot, byte) in buf.iter_mut().zip(left) {
74            slot.write(*byte);
75        }
76        let mut at = left.len();
77        if let Some(byte) = separator {
78            buf[at].write(byte);
79            at += 1;
80        }
81        for (slot, byte) in buf[at..].iter_mut().zip(right) {
82            slot.write(*byte);
83        }
84        #[cfg(debug_assertions)]
85        for slot in &mut buf[total..] {
86            slot.write(POISON);
87        }
88        Bytes(Repr::Inline {
89            len: u8::try_from(total).expect("at most INLINE_BYTES"),
90            buf,
91        })
92    }
93
94    /// [`from_slice`](Self::from_slice), for a caller that already owns the bytes.
95    pub fn from_vec(source: Vec<u8>) -> Bytes {
96        if source.len() <= INLINE_BYTES {
97            Bytes::from_slice(&source)
98        } else {
99            Bytes(Repr::Heap(source))
100        }
101    }
102
103    /// **The only reader of `Repr::Inline`'s buffer**, which is the other half
104    /// of what makes the invariant local: every comparison, hash, `Debug` and
105    /// deref a caller performs arrives here first.
106    pub fn as_slice(&self) -> &[u8] {
107        match &self.0 {
108            Repr::Inline { len, buf } => {
109                let len = usize::from(*len);
110                // Not a redundant bound: it is what says the slice below stays
111                // inside the array, stated where the `unsafe` can see it
112                // rather than left to `len`'s type.
113                debug_assert!(len <= INLINE_BYTES, "an inline length past the buffer");
114                // SAFETY: `from_slice` and `joined` are the only constructors
115                // of `Repr::Inline`, and each writes exactly `len` bytes before
116                // setting `len`, so the first `len` elements of `buf` are
117                // initialised. `len` is a `u8` and `INLINE_BYTES` is 54, and
118                // the assertion above holds it inside the array on every debug
119                // run of the suite. The only other writer is
120                // `extend_from_slice`, which writes the bytes past `len` before
121                // it lengthens `len` over them, and only while the sum stays
122                // within `INLINE_BYTES`: `Repr` is private to this file and
123                // neither field is written anywhere else.
124                unsafe { std::slice::from_raw_parts(buf.as_ptr().cast::<u8>(), len) }
125            }
126            Repr::Heap(v) => v,
127        }
128    }
129
130    /// The length of the separate allocation these bytes are held in, or 0
131    /// when they are held inline.
132    #[inline]
133    pub fn heap_len(&self) -> usize {
134        match &self.0 {
135            Repr::Inline { .. } => 0,
136            Repr::Heap(v) => v.len(),
137        }
138    }
139
140    /// Appends `more`, moving the bytes out to a separate allocation once they
141    /// no longer fit inline. **For a body nothing else can read yet** -- the
142    /// one caller is a concatenation growing the string it has just built,
143    /// which no variable, table or other register holds -- since everywhere
144    /// else a string is a value and these bytes are its identity.
145    pub fn extend_from_slice(&mut self, more: &[u8]) {
146        match &mut self.0 {
147            Repr::Inline { len, buf } if usize::from(*len) + more.len() <= INLINE_BYTES => {
148                let at = usize::from(*len);
149                for (slot, byte) in buf[at..].iter_mut().zip(more) {
150                    slot.write(*byte);
151                }
152                // The new length is set only after the bytes it covers are
153                // written, which is `as_slice`'s invariant.
154                *len = u8::try_from(at + more.len()).expect("at most INLINE_BYTES");
155            }
156            Repr::Inline { .. } => {
157                let mut grown = Vec::with_capacity(2 * (self.as_slice().len() + more.len()));
158                grown.extend_from_slice(self.as_slice());
159                grown.extend_from_slice(more);
160                self.0 = Repr::Heap(grown);
161            }
162            Repr::Heap(vec) => vec.extend_from_slice(more),
163        }
164    }
165
166    /// Whether these bytes are held inline rather than in a separate
167    /// allocation.
168    pub fn is_inline(&self) -> bool {
169        matches!(self.0, Repr::Inline { .. })
170    }
171}
172
173impl Deref for Bytes {
174    type Target = [u8];
175
176    fn deref(&self) -> &[u8] {
177        self.as_slice()
178    }
179}

181/// Shows the live bytes and never the padding behind them: the bytes past
182/// `len` are not part of the value, and printing them would make two equal
183/// values look different.
184impl fmt::Debug for Bytes {
185    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
186        fmt::Debug::fmt(self.as_slice(), f)
187    }
188}

194    /// Appending answers the bytes a copy of the whole would, on both sides
195    /// of the inline capacity and across it.
196    #[test]
197    fn extending_answers_the_joined_bytes() {
198        for first in [0, 1, 20, INLINE_BYTES - 1, INLINE_BYTES, INLINE_BYTES + 3] {
199            for more in [0, 1, 7, INLINE_BYTES, 100] {
200                let a: Vec<u8> = (0..first)
201                    .map(|i| u8::try_from(i % 251).expect("below 251"))
202                    .collect();
203                let b: Vec<u8> = (0..more)
204                    .map(|i| u8::try_from(i % 13).expect("below 13") + 100)
205                    .collect();
206                let mut grown = Bytes::from_slice(&a);
207                grown.extend_from_slice(&b);
208                let joined = [a.as_slice(), b.as_slice()].concat();
209                assert_eq!(grown.as_slice(), &joined[..], "{first} + {more}");
210                assert_eq!(
211                    grown.is_inline(),
212                    joined.len() <= INLINE_BYTES,
213                    "{first} + {more}"
214                );
215                assert_eq!(
216                    grown.heap_len(),
217                    if grown.is_inline() { 0 } else { joined.len() }
218                );
219            }
220        }
221    }

223    /// A join answers the bytes of the concatenation and takes the arm its
224    /// whole length asks for, with and without the separator, on both sides
225    /// of the capacity and exactly at it.
226    #[test]
227    fn joining_answers_the_concatenation_on_the_arm_its_length_asks_for() {
228        let lengths = [0, 1, 7, 26, INLINE_BYTES - 1, INLINE_BYTES, 60];
229        for left in lengths {
230            for right in lengths {
231                for separator in [None, Some(b' ')] {
232                    let a: Vec<u8> = (0..left)
233                        .map(|i| u8::try_from(i % 251).expect("below 251"))
234                        .collect();
235                    let b: Vec<u8> = (0..right)
236                        .map(|i| u8::try_from(i % 13).expect("below 13") + 100)
237                        .collect();
238                    let expected = [&a[..], separator.as_slice(), &b[..]].concat();
239                    let joined = Bytes::joined(&a, separator, &b);
240                    let shape = format!("{left} {separator:?} {right}");
241                    assert_eq!(joined.as_slice(), &expected[..], "{shape}");
242                    assert_eq!(
243                        joined.is_inline(),
244                        expected.len() <= INLINE_BYTES,
245                        "{shape}"
246                    );
247                    #[cfg(debug_assertions)]
248                    if joined.is_inline() {
249                        assert!(
250                            !joined.as_slice().contains(&POISON),
251                            "a read went past len at {shape}"
252                        );
253                    }
254                }
255            }
256        }
257    }

259    /// The capacity is a boundary and both sides of it have to answer the
260    /// same bytes, including the two lengths that bracket it exactly.
261    #[test]
262    fn both_arms_answer_the_bytes_they_were_given() {
263        for len in 0..=(INLINE_BYTES + 8) {
264            let source: Vec<u8> = (0..len)
265                .map(|i| u8::try_from(i % 251).expect("below 251"))
266                .collect();
267
268            let from_slice = Bytes::from_slice(&source);
269            assert_eq!(from_slice.as_slice(), &source[..], "from_slice at {len}");
270            assert_eq!(&*from_slice, &source[..], "deref at {len}");
271            assert_eq!(from_slice.len(), len, "len at {len}");
272
273            let from_vec = Bytes::from_vec(source.clone());
274            assert_eq!(from_vec.as_slice(), &source[..], "from_vec at {len}");
275
276            let cloned = from_slice.clone();
277            assert_eq!(cloned.as_slice(), &source[..], "clone at {len}");
278
279            assert_eq!(
280                from_slice.is_inline(),
281                from_vec.is_inline(),
282                "the two constructors disagree about the arm at {len}"
283            );
284        }
285    }

287    /// The capacity claim itself: `INLINE_BYTES` is inline and one more is
288    /// not, through both constructors.
289    #[test]
290    fn the_capacity_is_where_the_constant_says_it_is() {
291        let fits = vec![b'x'; INLINE_BYTES];

293

331    /// **What [`POISON`] is for, and it is not extra mutation coverage.**
332    /// Three mutations were applied and all three redden this file:
333    /// `as_slice` answering `INLINE_BYTES` bytes, `as_slice` answering
334    /// `len + 1`, and `from_slice` writing one byte too few. Every one of them
335    /// is caught by the value assertions that were already here, so this test
336    /// finds nothing they do not.
337    #[cfg(debug_assertions)]
338    #[test]
339    fn the_bytes_past_len_are_never_part_of_the_value() {
340        for len in 0..=INLINE_BYTES {
341            let source: Vec<u8> = (0..len)
342                .map(|i| u8::try_from(i % 97).expect("below 97"))
343                .collect();
344            assert!(
345                !source.contains(&POISON),
346                "the source itself carries the poison byte at {len}"
347            );
348            let bytes = Bytes::from_slice(&source);
349            assert!(bytes.is_inline(), "not the arm under test at {len}");
350            assert_eq!(bytes.as_slice(), &source[..], "bytes at {len}");
351            assert!(
352                !bytes.as_slice().contains(&POISON),
353                "a read went past len at {len}: {:?}",
354                bytes.as_slice()
355            );
356        }
357    }
