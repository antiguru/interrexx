17fn code(status: u64) -> RexxReturnCode {

21/// The bytes of the C string at `pointer`, `None` for null.
22///
23/// # Safety
24/// `pointer` is null or a NUL-terminated string that outlives `'a`.
25unsafe fn text<'a>(pointer: *const c_char) -> Option<&'a [u8]> {
26    // SAFETY: the caller's guarantee.
27    (!pointer.is_null()).then(|| unsafe { CStr::from_ptr(pointer) }.to_bytes())
28}

30/// The bytes of a name the API requires, the empty string for null.
31///
32/// # Safety
33/// As [`text`].
34unsafe fn name<'a>(pointer: *const c_char) -> &'a [u8] {
35    // SAFETY: the caller's guarantee.
36    unsafe { text(pointer) }.unwrap_or_default()
37}

39/// The 16 bytes of a user area, `None` for null.
40///
41/// # Safety
42/// `pointer` is null or addresses 16 readable bytes.
43unsafe fn user_area(pointer: *const c_char) -> Option<[u8; 16]> {
44    // SAFETY: the caller's guarantee.
45    (!pointer.is_null()).then(|| unsafe { pointer.cast::<[u8; 16]>().read_unaligned() })
46}

48/// The names a macrospace call lists, `None` for a null list.
49///
50/// # Safety
51/// `names` is null or addresses `count` pointers, each as [`text`] asks.
52unsafe fn names(count: usize, names: *const *const c_char) -> Option<Vec<Vec<u8>>> {
53    if names.is_null() {
54        return None;
55    }
56    // SAFETY: the caller's guarantee.
57    let list = unsafe { std::slice::from_raw_parts(names, count) };
58    // SAFETY: as above, for each element.
59    Some(
60        list.iter()
61            .map(|&each| unsafe { name(each) }.to_vec())
62            .collect(),
63    )
64}

66/// Hands `bytes` back through `data`: into the caller's buffer where it is strictly shorter than
67/// the buffer, else into a new block `RexxFreeMemory` releases, with a NUL after the bytes.
68/// `false` where no block could be allocated.
69///
70/// # Safety
71/// `data` addresses an initialised `RXSTRING` whose non-null `strptr` has `strlength` writable
72/// bytes.
73unsafe fn hand_back(data: *mut RXSTRING, bytes: &[u8]) -> bool {
74    // SAFETY: the caller's guarantee.
75    let data = unsafe { &mut *data };
76    if !data.strptr.is_null() && bytes.len() < data.strlength {
77        // SAFETY: the buffer holds `strlength` bytes, more than `bytes.len()`.
78        unsafe { std::ptr::copy_nonoverlapping(bytes.as_ptr(), data.strptr.cast(), bytes.len()) };
79        data.strlength = bytes.len();
80        return true;
81    }
82    // SAFETY: `malloc` has no precondition.
83    let block = unsafe { libc::malloc(bytes.len() + 1) }.cast::<u8>();
84    if block.is_null() {
85        return false;
86    }
87    // SAFETY: the block holds `bytes.len() + 1` bytes.
88    unsafe {
89        std::ptr::copy_nonoverlapping(bytes.as_ptr(), block, bytes.len());
90        block.add(bytes.len()).write(0);
91    }
92    data.strptr = block.cast();
93    data.strlength = bytes.len();
94    true
95}

112/// `RexxPullFromQueue` and `RexxPullQueue`: the status, with the item's time where an item was
113/// handed back.
114///
115/// # Safety
116/// As [`hand_back`], and `name` as [`text`].
117unsafe fn pull(
118    name: *const c_char,
119    data: *mut RXSTRING,
120    wait: usize,
121) -> (RexxReturnCode, Option<Timestamp>) {
122    // SAFETY: the caller's guarantee.
123    let (status, item) = capi::pull_queue(unsafe { text(name) }, u64::cast_from(wait));
124    let Some(item) = item else {
125        return (code(status), None);
126    };
127    // SAFETY: the caller's guarantee.
128    if !data.is_null() && !unsafe { hand_back(data, &item.data) } {
129        return (code(capi::MEMORY_FAILURE), None);
130    }
131    (code(status), Some(item.time))
132}

134/// An entry-point registration.
135///
136/// # Safety
137/// `name` and `user` as [`text`] and [`user_area`] ask; a non-null `entry` is a handler of
138/// `registry`'s signature, mapped for the process's life.
139unsafe fn register_entry(
140    registry: Registry,
141    name: *const c_char,
142    entry: REXXPFN,
143    user: *const c_char,
144) -> RexxReturnCode {
145    // SAFETY: the caller's guarantee that `entry` is such a handler.
146    let Some(entry) = (unsafe { ClassicEntry::new(entry) }) else {
147        return code(capi::BAD_ENTRY);
148    };
149    // SAFETY: the caller's guarantee.
150    let status =
151        unsafe { capi::register_entry(registry, self::name(name), entry, user_area(user)) };
152    code(status)
153}

155/// A query: `*flags` is 0, then 1 when found, and `user` gets the stored area when found.
156///
157/// # Safety
158/// As [`RexxQuerySubcom`].
159unsafe fn query(
160    registry: Registry,
161    name: *const c_char,
162    library: *const c_char,
163    flags: *mut u16,
164    user: *mut c_char,
165) -> RexxReturnCode {
166    if !flags.is_null() {
167        // SAFETY: the caller's guarantee.
168        unsafe { flags.write(0) };
169    }
170    // SAFETY: the caller's guarantee.
171    let (status, area) =
172        unsafe { capi::query_registration(registry, self::name(name), text(library)) };
173    if status == 0 {
174        if !flags.is_null() {
175            // SAFETY: as above.
176            unsafe { flags.write(1) };
177        }
178        if let Some(area) = area
179            && !user.is_null()
180        {
181            // SAFETY: the caller's guarantee of 16 writable bytes.
182            unsafe { user.cast::<[u8; 16]>().write_unaligned(area) };
183        }
184    }
185    code(status)
186}

469/// `RexxQueryMacro`: `*position` is 0 where the macro is absent.
470///
471/// # Safety
472/// As [`RexxRegisterSubcomDll`].
473#[unsafe(no_mangle)]
474pub unsafe extern "C" fn RexxQueryMacro(name: *const c_char, position: *mut u16) -> RexxReturnCode {
475    // SAFETY: the caller's guarantee.
476    let (status, found) = capi::query_macro(unsafe { self::name(name) });
477    if !position.is_null() {
478        // The service stores only position 1 or 2 (Deviation 185) and answers
479        // 0 for an absent macro, so the header's `unsigned short` holds the
480        // answer whole. A service started by an earlier build may hold another
481        // position, which keeps its low bits, as ooRexx answers one.
482        // SAFETY: the caller's guarantee.
483        unsafe { position.write(u16::cast_truncate(found)) };
484    }
485    code(status)
486}

507/// `RexxCreateQueue`.
508///
509/// # Safety
510/// As [`RexxRegisterSubcomDll`]; `name` has `size` writable bytes.
511#[unsafe(no_mangle)]
512pub unsafe extern "C" fn RexxCreateQueue(
513    name: *mut c_char,
514    size: usize,
515    requested: *const c_char,
516    duplicate: *mut usize,
517) -> RexxReturnCode {
518    // SAFETY: the caller's guarantee.
519    let (status, created, flag) = capi::create_queue(unsafe { text(requested) }, size);
520    if let Some(created) = created
521        && !name.is_null()
522        && size > 0
523    {
524        let length = created.len().min(size - 1);
525        // SAFETY: `name` holds `size` bytes, and `length + 1` is at most that.
526        unsafe {
527            std::ptr::copy_nonoverlapping(created.as_ptr(), name.cast::<u8>(), length);
528            name.add(length).write(0);
529        }
530    }
531    if let Some(flag) = flag
532        && !duplicate.is_null()
533    {
534        // SAFETY: the caller's guarantee.
535        unsafe { duplicate.write(usize::from(flag)) };
536    }
537    code(status)
538}

645/// `RexxPullQueue`: `RexxPullFromQueue` with the older time stamp, `valid` set.
646///
647/// # Safety
648/// As [`RexxPullFromQueue`].
649#[unsafe(no_mangle)]
650pub unsafe extern "C" fn RexxPullQueue(
651    name: *const c_char,
652    data: *mut RXSTRING,
653    time: *mut RexxLegacyQueueTime,
654    wait: usize,
655) -> RexxReturnCode {
656    // SAFETY: the caller's guarantee.
657    let (status, stamp) = unsafe { pull(name, data, wait) };
658    if let Some(stamp) = stamp
659        && !time.is_null()
660    {
661        let stamp = queue_time(stamp);
662        #[cfg(unix)]
663        let older = RexxLegacyQueueTime {
664            hours: stamp.hours,
665            minutes: stamp.minutes,
666            seconds: stamp.seconds,
667            hundredths: 0,
668            day: stamp.day,
669            month: stamp.month,
670            year: stamp.year,
671            weekday: stamp.weekday,
672            microseconds: 0,
673            yearday: usize::cast_from(stamp.yearday),
674            valid: 1,
675        };
676        #[cfg(windows)]
677        let older = RexxLegacyQueueTime {
678            year: stamp.year,
679            month: stamp.month,
680            weekday: stamp.weekday,
681            day: stamp.day,
682            hours: stamp.hours,
683            minutes: stamp.minutes,
684            seconds: stamp.seconds,
685            milliseconds: u16::try_from(stamp.microseconds / 1000)
686                .expect("`queue_time` sets no microseconds"),
687        };
688        // SAFETY: the caller's guarantee.
689        unsafe { time.write(older) };
690    }
691    status
692}
