1//! The builtin and error message tables, generated from the checked-in
2//! interface tables in `data/` and the message texts in `messages/`.
3//!
4//! # Message texts
5//!
6//! `messages/<major>.tsv` holds interrexx's own text for each error message
7//! of one major, one row per message: `major<TAB>sub<TAB>text`. A text writes
8//! insert `k` as `&k` (the utility messages of major 999 too, though the
9//! source marks theirs `%k`), and only for a `k` in the row's `rendered` column of
10//! `data/errors.tsv`. A line that is empty or starts with `#` is a comment.
11//! A text has no tab and no line break. The build checks every file: its
12//! name is the major of each of its rows, each code is in `errors.tsv`, no
13//! code has two texts, every code has one, and no text uses an insert
14//! outside the rendered set. [`errors::MESSAGES`] carries these texts.
15

19    /// The text of `message` with its inserts filled in, in position order.
20    /// A test that expects a message renders it through this function rather
21    /// than quoting the text, so that a change of wording alone fails
22    /// nothing.
23    pub fn render<const N: usize>(message: Msg<N>, inserts: [&str; N]) -> String {
24        let (major, sub) = message.code();
25        let text = lookup(major, sub)
26            .expect("a message constant names a row")
27            .text;
28        String::from_utf8(substitute(text, &inserts)).expect("inserts and texts are UTF-8")
29    }
30
31    /// `expected` with each `⟨M.S¦v1¦v2⟩` replaced by the text of message
32    /// `M.S` with the inserts `v1`, `v2`, ... filled in, and each `⟨M⟩` by the
33    /// text of the major `M` on its own. A test writes the message lines it
34    /// expects with these markers instead of quoting a wording, so that the
35    /// wording can change without the test: the code and the inserts stay in
36    /// the expectation.
37    pub fn expand(expected: &str) -> String {
38        let mut out = String::with_capacity(expected.len());
39        let mut rest = expected;
40        while let Some(open) = rest.find('⟨') {
41            out.push_str(&rest[..open]);
42            let after = &rest[open + '⟨'.len_utf8()..];
43            let close = after.find('⟩').expect("a ⟨ marker is closed by ⟩");
44            let mut parts = after[..close].split('¦');
45            let code = parts.next().expect("split yields one part");
46            let inserts: Vec<&str> = parts.collect();
47            let (major, sub) = match code.split_once('.') {
48                Some((major, sub)) => (major, sub),
49                None => (code, "0"),
50            };
51            let message = lookup(
52                major.parse().expect("a marker names a major number"),
53                sub.parse().expect("a marker names a sub number"),
54            )
55            .unwrap_or_else(|| panic!("{code} is not a message"));
56            let text = String::from_utf8(substitute(message.text, &inserts))
57                .expect("inserts and texts are UTF-8");
58            out.push_str(&text);
59            rest = &after[close + '⟩'.len_utf8()..];
60        }
61        out.push_str(rest);
62        out
63    }
64
65    /// `text` with `&1`, `&2`, ... replaced by the values of the same
66    /// position. A position with no value stays as written.
67    pub fn substitute<V: AsRef<[u8]>>(text: &str, values: &[V]) -> Vec<u8> {
68        let mut out = Vec::with_capacity(text.len());
69        let mut bytes = text.as_bytes().iter().copied().peekable();
70        while let Some(byte) = bytes.next() {
71            if byte != b'&' {
72                out.push(byte);
73                continue;
74            }
75            match bytes.peek().copied() {
76                Some(digit @ b'1'..=b'9') => {
77                    bytes.next();
78                    match values.get(usize::from(digit - b'1')) {
79                        Some(value) => out.extend_from_slice(value.as_ref()),
80                        None => out.extend_from_slice(&[b'&', digit]),
81                    }
82                }
83                _ => out.push(b'&'),
84            }
85        }
86        out
87    }
88}

90pub mod builtins {
91    include!(concat!(env!("OUT_DIR"), "/builtins.rs"));

93    /// The builtins excluded, whole or in part, each with its owner and the
94    /// reason: the whole exclusions followed by the partial rows.
95    pub const EXCLUDED: &[(&str, &str, &str)] = &[];
96
97    /// The owner of `name`'s excluded form and its reason, or `None` for
98    /// a name [`EXCLUDED`] does not list.
99    pub fn owner_of(name: &str) -> Option<(&'static str, &'static str)> {
100        EXCLUDED
101            .iter()
102            .find(|(excluded, _, _)| *excluded == name)
103            .map(|(_, owner, reason)| (*owner, *reason))
104    }
105
106    /// Every excluded name, in [`EXCLUDED`] order.
107    pub fn excluded_names() -> Vec<&'static str> {
108        EXCLUDED.iter().map(|(name, _, _)| *name).collect()
109    }
110
111    /// The subset of [`EXCLUDED`] that is excluded only in part, and so is
112    /// still in scope in its other form. Named rather than written as a
113    /// literal `3` at the two places that subtract it, because "in
114    /// `EXCLUDED`" and "excluded outright" are different sets and the
115    /// difference is exactly these names.
116    pub const PARTIALLY_EXCLUDED: &[&str] = &[];
117
118    /// The builtins excluded outright: [`EXCLUDED`] less
119    /// [`PARTIALLY_EXCLUDED`], in `EXCLUDED` order.
120    pub fn wholly_excluded() -> Vec<&'static str> {
121        EXCLUDED
122            .iter()
123            .map(|(name, _, _)| *name)
124            .filter(|name| !PARTIALLY_EXCLUDED.contains(name))
125            .collect()
126    }
127
128    /// The names of [`NAMES`] that are builtins only at a language level above
129    /// the one this crate implements (`docs/design/decisions.md`, the
130    /// language level), each with that level. At this crate's level each is
131    /// a routine of the `REXX` package, so a call to one is resolved past the
132    /// builtins and the running package's routines, and the name is neither
133    /// in scope nor excluded.
134    pub const LATER_LEVEL: &[(&str, &str)] = &[
135        ("BEEP", "6.07"),
136        ("DIRECTORY", "6.07"),
137        ("FILESPEC", "6.07"),
138    ];
139
140    /// Whether `name` is listed in [`LATER_LEVEL`].
141    pub fn is_later_level(name: &str) -> bool {
142        LATER_LEVEL.iter().any(|(later, _)| *later == name)
143    }
144
145    /// The builtins this crate must be able to run: every name in [`NAMES`] that
146    /// is not excluded outright and not of a [later level](LATER_LEVEL), in
147    /// `NAMES` order. A partially excluded name is in here, because its
148    /// in-scope form still has to work.
149    pub fn in_scope() -> Vec<&'static str> {
150        let whole = wholly_excluded();
151        NAMES
152            .iter()
153            .copied()
154            .filter(|name| !whole.contains(name) && !is_later_level(name))
155            .collect()
156    }
