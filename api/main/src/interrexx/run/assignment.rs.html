4    /// Everything one assignment does once its value has been evaluated:
5    /// `>>>`, then the write and the lines the write itself produces.
6    #[inline]
7    pub(crate) fn assign_evaluated(
8        &mut self,
9        code: &Code<'_>,
10        target: ExprId,
11        value: ObjRef,
12        at: Option<usize>,
13        shared_key: bool,
14    ) -> Result<(), Failure> {
15        self.roots.activity_mut().push_temp(value);
16        // `>>>` fires before the assignment itself
17        // (evaluate, trace, *then* assign), which matters only in that the traced value can never be
18        // affected by the write it precedes.
19        // Reads `current_value_indent` rather than recomputing
20        // `static_indent(index)` independently -- the clause unit already
21        // computed exactly this value for this same instruction right before the clause
22        // ran, and a second computation of the identical quantity is how the
23        // two drift, which is exactly what happened here before this fix: this
24        // site's own copy never learned about the offset when the field was
25        // added.
26        let indent = self.activity.clause_state.current_value_indent;
27        let target = code.body.expr(target);
28        // The entry gate excludes a result line when evaluation enabled tracing.
29        if self.activity.clause_state.instructions_traced_at_entry
30            && let Some(rendered) = self.result_text(value)?
31        {
32            self.trace_result_before_effect(indent, &rendered)?;
33        }
34        if shared_key && let ExprView::Compound(id) = target.kind() {
35            return self.assign_compound(code, id, value, indent, true);
36        }
37        self.assign_expr_target(code, target, value, indent, at)
38    }

40    /// Writes `value` through the compound target `id`, and traces the
41    /// write. `shared_key` says that the value expression read this same
42    /// compound with [`crate::eval::SymbolRead::CompoundShared`], whose key
43    /// is used here where that read still holds it.
44    fn assign_compound(
45        &mut self,
46        code: &Code<'_>,
47        id: SymbolId,
48        value: ObjRef,
49        indent: usize,
50        shared_key: bool,
51    ) -> Result<(), Failure> {
52        // **Borrowed, not copied.** `Code::stem` and `SymbolTable::name`
53        // both answer with the lifetime of the `Code`, which is the program
54        // rather than this `Interp`, so neither needs an owned copy to
55        // survive the `&mut self` calls below. The `Variable` arm of
56        // `assign_expr_target` has always passed its name borrowed; this arm
57        // copied both, and measured with `heaptrack` on
58        // `samples/rexxcps.rex` that cost two allocations per compound
59        // write.
60        let tag = code.symbols.name(id).as_bytes();
61        let (stem_name, stem_at) = code.stem(id);
62        let mut key = self.take_key_buffer();
63        // The read's key is the key this write would build: the read held
64        // it only where no activation ran after it began, and the value
65        // expression between it and here calls nothing.
66        let mut held = None;
67        if shared_key && self.shared_key_for.take() == Some(id) {
68            std::mem::swap(&mut key, &mut self.shared_key);
69            held = self.shared_tail.take();
70        } else if let Err(failure) = self.tail_key_into(code, id, &mut key) {
71            self.give_key_buffer(key);
72            return Err(failure);
73        }
74        // The tail the read found, written by its ordinal where it is still
75        // that tail, and by its key otherwise.
76        if !held.is_some_and(|held| self.stem_set_held(stem_name, stem_at, held, &key, value)) {
77            self.stem_set_at(stem_name, stem_at, &key, value);
78        }
79        // **The resolved name is built only when a line will print it.**
80        // `trace_compound_name` returns at once unless intermediates are on,
81        // so joining the stem to the tail key ahead of that check allocated a
82        // name to discard. This is the rule the value rendering below
83        // follows -- `trace.rs`'s own doc comment gives the reasoning for the
84        // value half, and the name half is the same argument.
85        if self.tracing_intermediates() {
86            let mut resolved = stem_name.to_vec();
87            resolved.extend_from_slice(&key);
88            self.trace_compound_name(indent, tag, &resolved);
89        }
90        if let Some(rendered) = self.intermediate_text(value)? {
91            self.trace_assignment(indent, tag, &rendered, value);
92        }
93        self.give_key_buffer(key);
94        Ok(())
95    }

97    /// Writes `value` into the slot a plain-variable target is already bound
98    /// to, and traces the write. `rendered` is the assigned value's own text
99    /// where a `>=>` line will print it, and `None` where none will.
100    ///
101    /// Panics in a debug build when `id` does not name a simple variable: the
102    /// slot written here is the whole name's, which a stem or a compound tail
103    /// does not have.
104    #[inline]
105    pub(crate) fn assign_bound_variable(
106        &mut self,
107        code: &Code<'_>,
108        id: SymbolId,
109        slot: usize,
110        value: ObjRef,
111        rendered: Option<&[u8]>,
112        indent: usize,
113    ) {
114        debug_assert!(
115            matches!(
116                shape_of(code.symbols.name(id).as_bytes()),
117                NameShape::Simple
118            ),
119            "a bound slot was written for {}, which is not a simple variable",
120            code.symbols.name(id),
121        );
122        let frame = self.activation().frame;
123        self.set_variable(frame, slot, value);
124        // The name is reached only where a line will print it, which is what
125        // the general path below pays on every call.
126        if let Some(rendered) = rendered {
127            self.trace_assignment(indent, code.symbols.name(id).as_bytes(), rendered, value);
128        }
129    }

131    /// Writes `value` through one assignment *target expression*, and traces
132    /// the write.
133    pub(crate) fn assign_expr_target(
134        &mut self,
135        code: &Code<'_>,
136        target: ExprRef<'_>,
137        value: ObjRef,
138        indent: usize,
139        at: Option<usize>,
140    ) -> Result<(), Failure> {
141        match target.kind() {
142            ExprView::Variable(id) => {
143                self.assign_name_target(code, NameShape::Simple, id, value, indent, at)
144            }
145            ExprView::Stem(id) => {
146                self.assign_name_target(code, NameShape::Stem, id, value, indent, at)
147            }
148            ExprView::Compound(id) => {
149                self.assign_name_target(code, NameShape::Compound, id, value, indent, at)
150            }
151            // the term's `NAME=` sent with the value ahead of its own
152            // arguments, each argument traced as a result.
153            ExprView::Message {
154                target: receiver,
155                name,
156                super_class,
157                args,
158                ..
159            } => {
160                let receiver = self.eval(code, receiver)?;
161                self.roots.activity_mut().push_temp(receiver);
162                let scope = match super_class {
163                    Some(super_class) => {
164                        let scope = self.eval(code, super_class)?;
165                        self.roots.activity_mut().push_temp(scope);
166                        Some(scope)
167                    }
168                    None => None,
169                };
170                let mut values = vec![Some(value)];
171                for arg in args.iter() {
172                    let evaluated = match arg {
173                        Some(expr) => {
174                            let evaluated = self.eval(code, expr)?;
175                            self.roots.activity_mut().push_temp(evaluated);
176                            Some(evaluated)
177                        }
178                        None => None,
179                    };
180                    if let Some(shown) = evaluated
181                        .map(|value| self.result_text(value))
182                        .transpose()?
183                        .flatten()
184                    {
185                        self.trace_result(indent, &shown);
186                    } else if evaluated.is_none() {
187                        self.trace_result(indent, b"");
188                    }
189                    values.push(evaluated);
190                }
191                let mut setter = name.to_vec();
192                setter.push(b'=');
193                self.validate_scope_override(receiver, scope)?;
194                let caller = self.caller();
195                let answer = pinned!(
196                    self,
197                    crate::pinning::PinKind::TreeSend,
198                    self.send_message(receiver, &setter, scope, &values, caller)
199                )?;
200                if let Some(answer) = answer
201                    && let Some(shown) = self.intermediate_text(answer)?
202                {
203                    self.trace_assignment(indent, &setter, &shown, answer);
204                }
205                Ok(())
206            }
207            other => Err(Loud::expression(other).into()),
208        }
209    }

211    /// Writes `value` through the variable `id` names, of the given shape,
212    /// and traces the write.
213    pub(crate) fn assign_name_target(
214        &mut self,
215        code: &Code<'_>,
216        shape: NameShape,
217        id: SymbolId,
218        value: ObjRef,
219        indent: usize,
220        at: Option<usize>,
221    ) -> Result<(), Failure> {
222        match shape {
223            NameShape::Simple => {
224                let name = code.symbols.name(id).as_bytes();
225                let slot = match at {
226                    Some(slot) => slot,
227                    None => self.slot_of(name),
228                };
229                let frame = self.activation().frame;
230                self.set_variable(frame, slot, value);
231                if let Some(rendered) = self.intermediate_text(value)? {
232                    self.trace_assignment(indent, name, &rendered, value);
233                }
234            }
235            // `stem. = expr`: replace-and-rebind, through the
236            // library `stem_assign` already builds -- this arm is the
237            // dispatch, not new stem logic.
238            NameShape::Stem => {
239                // The tripwire `crate::ir::drive`'s `Op::Store` arm carries
240                // for the compiled side, here where **both** engines pass:
241                // a slot handed to this arm was resolved against the symbol's
242                // own id and is silently shadowed below, so a caller that
243                // started supplying one would write through the entry's slot
244                // and never learn that its own was ignored.
245                debug_assert!(
246                    at.is_none(),
247                    "a stem write was handed a slot, and the slot it writes comes from the entry"
248                );
249                let name = code.symbols.name(id).as_bytes();
250                let at = code.compound(id).and_then(|entry| entry.stem_at);
251                self.stem_assign_at(name, at, value);
252                if let Some(rendered) = self.intermediate_text(value)? {
253                    self.trace_assignment(indent, name, &rendered, value);
254                }
255            }
256            // `a.b = expr`: resolve the tail key the same way reading
257            // `a.b` would (`eval_node`'s own `Compound` arm), then
258            // mutate that one tail in place through `stem_set`.
259            NameShape::Compound => {
260                // `read_symbol`'s own compound tripwire, on the writing
261                // side. A compound-shaped name **can** reach
262                // `Plan::by_symbol`: `Plan::bind` puts every name it binds
263                // whole there, and `note_loop` and `note_parse` both call it
264                // with spellings that may be compound-shaped. So a caller
265                // reaching for a slot by this symbol's id can find one, and it
266                // is not the slot this arm writes.
267                debug_assert!(
268                    at.is_none(),
269                    "a compound write was handed a slot, and the slot it writes is the stem's"
270                );
271                self.assign_compound(code, id, value, indent, false)?;
272            }
273        }
274        Ok(())
275    }
