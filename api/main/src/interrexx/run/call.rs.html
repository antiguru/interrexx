23/// Swaps `args` with the buffer `slot` holds, which nothing else shares
24/// between calls.
25#[inline(always)]
26fn lend_arguments(slot: &mut Rc<Vec<Option<ObjRef>>>, args: &mut Vec<Option<ObjRef>>) {
27    match Rc::get_mut(slot) {
28        Some(buffer) => std::mem::swap(buffer, args),
29        None => lend_shared_arguments(slot, args),
30    }
31}
32
33/// [`lend_arguments`] where something else holds `slot` too.
34#[cold]
35#[inline(never)]
36fn lend_shared_arguments(slot: &mut Rc<Vec<Option<ObjRef>>>, args: &mut Vec<Option<ObjRef>>) {
37    let mut owned = Vec::clone(slot);
38    std::mem::swap(&mut owned, args);
39    *slot = Rc::new(owned);
40}

225    /// The half of a call's resolution decided before the arguments run: a
226    /// label or a builtin (the parser's builtin index), or the refusal of a
227    /// builtin this crate excludes. `None` for a name that is none of those.
228    pub(crate) fn resolve_fixed_call(
229        &self,
230        name: &[u8],
231        search_labels: bool,
232    ) -> Result<Option<Resolved>, Failure> {
233        // **Resolved against the running *activation's* body, not against the
234        // body a caller is walking, and the two differ inside an `INTERPRET`
235        // fragment.** A fragment's `labels` is always empty -- a label in
236        // interpreted text is error 47.1 -- so searching the walked body would
237        // make every `CALL` inside a fragment unresolvable. For example,
238        // `interpret "call sub"` runs the enclosing program's `sub:`. Found by
239        // running the composition rather than by reading the code: the first
240        // version of this searched the walked body and passed every test that
241        // had no `INTERPRET` in it.
242        let program = Rc::clone(&self.activation().program);
243        let selector = self.activation().body;
244        let Some(activation_body) = body_of(&program, selector) else {
245            return Err(Loud::missing_body().into());
246        };
247        let label = if search_labels {
248            activation_body.labels.get(name).copied()
249        } else {
250            None
251        };
252        let resolved = match label {
253            Some(target) => Resolved::Label(target),
254            // **`resolve` rather than `is_builtin`, and it answers the same
255            // question.** Every row's name is in scope
256            // (`every_implemented_row_names_an_in_scope_builtin`) and a name in
257            // scope with no row comes back `Gap`, so `resolve(name).is_some()`
258            // and `is_builtin(name)` agree on every name. It costs the same
259            // one lookup and keeps which builtin it found.
260            None if let Some(target) = builtin::resolve(name) => Resolved::Builtin(target),
261            // A builtin excluded outright is still a builtin, so it
262            // sits here rather than behind the routine lookup -- see
263            // `builtin::is_excluded_builtin`'s own doc for why neither the
264            // routine step nor 43.1 is an acceptable answer for one.
265            None if builtin::is_excluded_builtin(name) => {
266                return Err(Loud::unresolved_call(name).into());
267            }
268            None => return Ok(None),
269        };
270        Ok(Some(resolved))
271    }

273    /// The half of a call's resolution searched once the arguments have run,
274    /// for a name [`Interp::resolve_fixed_call`] answered `None` for.
275    pub(crate) fn resolve_routine_call(&mut self, name: &[u8]) -> Result<Resolved, Failure> {
276        let resolved = match self.package_routine_lookup(name)? {
277            Some(resolved) => resolved,
278            // A macro searched before the routines below.
279            None if crate::builtin::rexxutil::macros::macro_position(name)
280                == Some(crate::builtin::rexxutil::macros::BEFORE) =>
281            {
282                Resolved::Macro
283            }
284            // **A routine of the interpreter's own internal packages**, which
285            // is consulted here -- after the running package's routines and
286            // before the external file search. Answering 43.1 for one of these
287            // says "no such routine" for a name that does exist, which a
288            // program cannot tell from its own typo.
289            None if let Some(row) = crate::internal_routines::lookup(name) => match row.body {
290                Some(_) => Resolved::Internal(row),
291                None => return Err(Loud::internal_routine(name).into()),
292            },
293            // **A routine a loaded library exports**, whichever load registered
294            // it. These are kept in the table the internal packages' routines
295            // are in, where a library routine replaces an internal one of the
296            // same name; here the internal one is found first.
297            None if let Some(slot) = self.package_routine(name) => Resolved::LibraryRoutine(slot),
298            // **The external file search, which is performed before answering 43.1.**
299            // Only whether it resolves is decided here; the path is searched for again
300            // where the file is entered, because a `Resolved` is `Copy` and this
301            // resolver is `&self`. **A registered classic function**, which is tried
302            // after the macros searched first and before the external file search.
303            None if self.classic_function(name).is_some() => Resolved::Registered,
304            None if self.external_program(name).is_some() => Resolved::External,
305            None if crate::builtin::rexxutil::macros::macro_position(name)
306                == Some(crate::builtin::rexxutil::macros::AFTER) =>
307            {
308                Resolved::Macro
309            }
310            // 43.1, once the search above has found nothing, and
311            // raised by the consumer rather than here. Measured in a
312            // clean directory with nothing of that name beside the
313            // program: `call zorkolo` gives 43.1 rc 213 naming
314            // `ZORKOLO`.
315            None => Resolved::Unresolved,
316        };
317        Ok(resolved)
318    }

320    /// The routine `name` reaches from the running package, through
321    /// [`Interp::find_routine`].
322    ///
323    /// **The parent step is what code built by a send resolves through**:
324    /// `Routine~newFile` and `Method~newFile` record the context's package or,
325    /// with none, the caller's (`new_file_executable`), and `Package~new`
326    /// records a context's (`package_from_source`). A file reached by a
327    /// *call* records none, so it still sees its own package alone --
328    /// measured, the same body finds the caller's `::ROUTINE` through
329    /// `Routine~newFile` and raises 43.1 through `call`.
330    fn package_routine_lookup(&mut self, name: &[u8]) -> Result<Option<Resolved>, Failure> {
331        let Some(program) = self.running_activation().map(|running| running.program_id) else {
332            return Ok(None);
333        };
334        let upper = name.to_ascii_uppercase();
335        let package = self.program_package(program);
336        if !self.routines_written {
337            return Ok(self
338                .find_routine(package, &upper)
339                .map(crate::MergedRoutine::resolved));
340        }
341        match self.find_written_routine(package, &upper)? {
342            Some(crate::FoundRoutine::Merged(merged)) => Ok(Some(merged.resolved())),
343            Some(crate::FoundRoutine::Object(object)) => {
344                self.routine_object_resolution(object).map(Some)
345            }
346            None => Ok(None),
347        }
348    }

365    /// One builtin call: its arguments evaluated in the caller, then the row
366    /// run over them.
367    pub(crate) fn invoke_builtin_call(
368        &mut self,
369        code: &Code<'_>,
370        target: crate::builtin::BuiltinTarget,
371        name: &[u8],
372        args: Args<'_>,
373    ) -> Result<ObjRef, Failure> {
374        // **Pushed onto the shared stack one at a time**, rather than into a
375        // buffer lent for the whole loop: each argument's evaluation calls
376        // back into `&mut self`, so nothing may hold the `Vec` across it, and
377        // a run that is only ever appended to needs no borrow between pushes.
378        // The compiled path's `Op::PushArg` writes the same stack.
379        let mark = self.activity.value_buffer.len();
380        for arg in args.iter() {
381            let value = match arg {
382                None => {
383                    self.trace_argument(self.activity.clause_state.current_value_indent, b"");
384                    None
385                }
386                Some(expr) if self.leaf_argument(expr) => {
387                    Some(self.eval_leaf_argument(code, expr)?)
388                }
389                Some(expr) => Some(pinned!(
390                    self,
391                    crate::pinning::PinKind::TreeEval,
392                    self.eval_traced_argument(code, expr)
393                )?),
394            };
395            self.activity.value_buffer.push(value);
396        }
397        self.run_over_pushed_args(mark, |interp, values| {
398            builtin::run(interp, name, target, values)
399        })
400    }

416    /// [`Interp::invoke_call`] up to the point its callee's body would run.
417    pub(crate) fn begin_invoke_call(
418        &mut self,
419        code: &Code<'_>,
420        resolution: CallResolution<'_>,
421        name: &[u8],
422        args: Args<'_>,
423        call_type: CallType,
424    ) -> Result<Begun, Failure> {
425        // **Evaluated in the caller, before anything is pushed**, which is
426        // where the argument expressions' own variables live. Observable
427        // through failure, and the failure is real and measured: `call sub
428        // 1/0` is Error 42.3
429        // reported against the `CALL` clause, at rc 214, and a version that
430        // skipped evaluation would run the callee instead.
431        if let CallResolution::Settled(Resolved::Builtin(target)) = resolution {
432            let answer = self.invoke_builtin_call(code, target, name, args);
433            return Ok(Begun::Done(Ended::Returned(builtin::answered_nothing(
434                self, answer,
435            )?)));
436        }
437        // A fresh `Vec` and not a lent one: this path always hands the
438        // arguments to the callee, which keeps them, so there is nothing to
439        // give back and a pool would allocate on every call anyway.
440        let mut arguments: Vec<Option<ObjRef>> = Vec::with_capacity(args.len());
441        for arg in args.iter() {
442            match arg {
443                None => {
444                    // An omitted position traces an **empty** value line, not
445                    // no line, and measured above.
446                    self.trace_argument(self.activity.clause_state.current_value_indent, b"");
447                    arguments.push(None);
448                }
449                Some(expr) if self.leaf_argument(expr) => {
450                    arguments.push(Some(self.eval_leaf_argument(code, expr)?));
451                }
452                Some(expr) => arguments.push(Some(pinned!(
453                    self,
454                    crate::pinning::PinKind::TreeEval,
455                    self.eval_traced_argument(code, expr)
456                )?)),
457            }
458        }
459        let (resolved, keep) = self.settle_after_arguments(resolution, name)?;
460        let begun = self.begin_evaluated_call(resolved, name, arguments, call_type);
461        if let Some((slot, keep)) = keep {
462            self.site_call(slot, keep, SiteCall::of_begun(&begun));
463        }
464        begun
465    }

467    /// [`Interp::begin_invoke_call`] past its argument loop, over `arguments`
468    /// already evaluated, for any callee but a builtin.
469    fn begin_evaluated_call(
470        &mut self,
471        resolved: Resolved,
472        name: &[u8],
473        mut arguments: Vec<Option<ObjRef>>,
474        call_type: CallType,
475    ) -> Result<Begun, Failure> {
476        // An internal-package routine runs no activation, and neither does a
477        // library routine, so they take the builtin discipline over the values.
478        if let Resolved::Internal(_) | Resolved::LibraryRoutine(_) = resolved {
479            if let Some(handled) = self.call_checkpoint(name, &arguments)? {
480                return Ok(Begun::Done(Ended::Returned(handled)));
481            }
482            return match resolved {
483                Resolved::Internal(row) => {
484                    Ok(internal_begun(self.run_internal(row, &mut arguments, 0)?))
485                }
486                Resolved::LibraryRoutine(slot) => self
487                    .run_package_routine(slot, name, &arguments)
488                    .map(library_begun),
489                _ => unreachable!("only the two arms above reach here"),
490            };
491        }
492        let begun = self.begin_call(resolved, name, &mut arguments, 0, call_type)?;
493        // An enclosing builtin's run stands on the stack while this call's
494        // callee clears it at its first clause.
495        if let Begun::Entered = begun {
496            let values = std::mem::take(&mut self.activity.value_buffer);
497            self.lend_stack(true, values);
498        }
499        Ok(begun)
500    }

502    /// What a call runs, once its arguments have run, and what its site is
503    /// to keep once it returns.
504    fn settle_after_arguments<'c>(
505        &mut self,
506        resolution: CallResolution<'c>,
507        name: &[u8],
508    ) -> Result<(Resolved, Option<(crate::ir::CallSiteSlot<'c>, SiteKeep)>), Failure> {
509        let site = match resolution {
510            CallResolution::Settled(resolved) => return Ok((resolved, None)),
511            CallResolution::AfterArguments { site } => site,
512            CallResolution::Qualified { namespace } => {
513                return self
514                    .resolve_qualified_call(namespace, name)
515                    .map(|resolved| (resolved, None));
516            }
517        };
518        let generation = self.routine_generation;
519        if let Some(site) = site
520            && let Some(resolved) = site.resolved(generation)
521        {
522            return Ok((resolved, None));
523        }
524        let resolved = self.resolve_routine_call(name)?;
525        Ok((
526            resolved,
527            site.map(|site| (site, self.site_keep(site.site(), resolved, generation))),
528        ))
529    }

531    /// A [`SiteKeep`] for call site `site`, its copy pinned.
532    pub(crate) fn site_keep(&mut self, site: u16, resolved: Resolved, generation: u32) -> SiteKeep {
533        let copy = resolved.copy();
534        if copy != ObjRef::NIL {
535            self.pin_copy(copy);
536        }
537        let holder = self.activation().program_id;
538        let hold = match resolved {
539            Resolved::Routine(installed) if installed.program != holder => {
540                Some(crate::run::SiteHold {
541                    holder,
542                    program: installed.program,
543                })
544            }
545            _ => None,
546        };
547        SiteKeep {
548            site,
549            resolved,
550            generation,
551            hold,
552        }
553    }

555    /// Ends or hands on `keep`, for its call through `slot` that `ended`: a
556    /// call that entered a body files it on the activity under the index its
557    /// op's parked call takes ([`Interp::take_entered_keep`]).
558    pub(crate) fn site_call(
559        &mut self,
560        slot: crate::ir::CallSiteSlot<'_>,
561        keep: SiteKeep,
562        ended: SiteCall,
563    ) {
564        match ended {
565            SiteCall::Entered => {
566                let at = self.activity.parked_calls.len();
567                debug_assert!(
568                    self.activity
569                        .parked_keeps
570                        .last()
571                        .is_none_or(|(filed, _)| *filed < at),
572                    "an entered call's keep was left unclaimed"
573                );
574                self.activity.parked_keeps.push((at, keep));
575            }
576            SiteCall::Returned | SiteCall::Raised => {
577                self.end_site_keep(slot.chunk(), keep, ended == SiteCall::Returned);
578            }
579        }
580    }

582    /// Ends `keep`: its site keeps the resolution where the call `returned`.
583    pub(crate) fn end_site_keep(
584        &mut self,
585        chunk: &crate::ir::Chunk,
586        keep: SiteKeep,
587        returned: bool,
588    ) {
589        if returned {
590            let repin =
591                chunk
592                    .call_site_slot(keep.site)
593                    .remember(keep.resolved, keep.generation, keep.hold);
594            self.repin(repin);
595        }
596        let copy = keep.resolved.copy();
597        if copy != ObjRef::NIL {
598            self.unpin_copy(copy);
599        }
600    }

602    /// The keep filed for the parked call the next push would make, which
603    /// a call op entered with no parked call: one its own callee has not had
604    /// the chance to file under the same index.

716    /// The security manager's `CALL` checkpoint,
717    /// asked for every name
718    /// that gets past the running package's own `::ROUTINE`s -- including
719    /// one nothing answers, which is why it runs before 43.1.
720    ///
721    /// `Some(result)` where the manager handled the call, nothing then
722    /// running; `None` where it declined or none is installed.
723    fn call_checkpoint(
724        &mut self,
725        name: &[u8],
726        values: &[Option<ObjRef>],
727    ) -> Result<Option<Option<ObjRef>>, Failure> {
728        if self.effective_security_manager().is_none() {
729            return Ok(None);
730        }
731        let called = self.text(name);
732        self.roots.activity_mut().push_temp(called);
733        let arguments = self.security_arguments_array(values);
734        let entries = [
735            (crate::security::key::NAME, called),
736            (crate::security::key::ARGUMENTS, arguments),
737        ];
738        let Some(info) = self.security_check(crate::security::message::CALL, &entries)? else {
739            return Ok(None);
740        };
741        let result = self.security_entry(info, crate::security::key::RESULT)?;
742        Ok(Some(result))
743    }

765        &mut self,

774    /// [`Interp::run_internal_as`] for the `Routine` object `executable`
775    /// standing for the row, `.nil` for the row's own, which its frame
776    /// answers.
777    pub(crate) fn run_internal_given(
778        &mut self,
779        row: &'static crate::internal_routines::InternalRoutine,
780        name: InternalName<'_>,
781        args: &mut Vec<Option<ObjRef>>,
782        from: usize,
783        executable: ObjRef,
784    ) -> Result<Started<ObjRef>, Failure> {
785        let Some(body) = row.body else {
786            return Err(Loud::internal_routine(row.name.as_bytes()).into());
787        };
788        let caller = self.running_activation().map(|activation| activation.id);
789        let depth = self.activity.internal_depth;
790        let level = match self.activity.internal_levels.get_mut(depth) {
791            Some(level) => level,
792            None => add_internal_level(&mut self.activity.internal_levels),
793        };
794        level.caller = caller;
795        level.row = row;
796        level.from = from;
797        level.executable = executable;
798        level.name = match name {
799            InternalName::Upper => LevelName::Upper,
800            InternalName::Row => LevelName::Row,
801            InternalName::Called(name) => {
802                level.called.clear();
803                level.called.extend_from_slice(name);
804                LevelName::Called
805            }
806        };
807        lend_arguments(&mut level.arguments, args);
808        let shared = Rc::clone(&level.arguments);
809        self.activity.internal_depth = depth + 1;
810        let outcome = body(self, row.name.as_bytes(), &shared[from..]);
811        self.activity.internal_depth = depth;
812        drop(shared);
813        if let Some(level) = self.activity.internal_levels.get_mut(depth) {
814            lend_arguments(&mut level.arguments, args);
815        }
816        if let Err(failure) = &outcome {
817            // One of these is reported under its own routine line, whether the
818            // argument marshalling or the body raised: `filespec('D')` and
819            // `SysSleep('abc')` both carry it.
820            self.blame_internal_routine(row, name, &args[from..], failure, executable);
821        }
822        let value = outcome?;
823        Ok(match self.activity.native_park {
824            Some(_) => Started::Entered,
825            None => Started::Ran(value),
826        })
827    }

829    /// [`Interp::begin_call_over_values`] for a call whose site `slot` keeps
830    /// `keep` once the call returns.
831    #[cold]
832    #[inline(never)]
833    fn begin_kept_call_over_values(
834        &mut self,
835        resolved: Resolved,
836        name: &[u8],
837        values: &mut Vec<Option<ObjRef>>,
838        mark: usize,
839        slot: crate::ir::CallSiteSlot<'_>,
840        keep: SiteKeep,
841    ) -> Result<Started<ObjRef>, Failure> {
842        if let Resolved::Internal(_) | Resolved::LibraryRoutine(_) = resolved {
843            let started = self.begin_call_over_values(resolved, name, values, mark);
844            self.site_call(slot, keep, SiteCall::of(&started));
845            return started;
846        }
847        // Kept before the callee's value is checked: a routine that returned
848        // nothing to a function call is kept.
849        let begun = self.begin_call(resolved, name, values, mark, CallType::Function);
850        self.site_call(slot, keep, SiteCall::of_begun(&begun));
851        function_started(begun?, name)
852    }

854    /// [`Interp::begin_call_over_pushed_args`] with the run in hand.
855    #[inline(always)]
856    fn begin_call_over_values(
857        &mut self,
858        resolved: Resolved,
859        name: &[u8],
860        values: &mut Vec<Option<ObjRef>>,
861        mark: usize,
862    ) -> Result<Started<ObjRef>, Failure> {
863        if let Resolved::Internal(row) = resolved {
864            if let Some(handled) = self.call_checkpoint(name, &values[mark..])? {
865                return handled
866                    .map(Started::Ran)
867                    .ok_or_else(|| Raised::no_data_returned(name).into());
868            }
869            return self.run_internal(row, values, mark);
870        }
871        if let Resolved::LibraryRoutine(slot) = resolved {
872            let answered = match self.call_checkpoint(name, &values[mark..])? {
873                Some(handled) => Started::Ran(handled),
874                None => self.run_package_routine(slot, name, &values[mark..])?,
875            };
876            return match answered {
877                Started::Ran(value) => value
878                    .map(Started::Ran)
879                    .ok_or_else(|| Raised::no_data_returned(name).into()),
880                Started::Entered => Ok(Started::Entered),
881            };
882        }
883        let begun = self.begin_call(resolved, name, values, mark, CallType::Function)?;
884        function_started(begun, name)
885    }

981    /// [`Interp::invoke_call_over`] up to the point its callee's body would
982    /// run: the whole call for a callee that runs no activation, or the
983    /// activation pushed and the caller's level state saved. The arguments
984    /// are `arguments` from `from`.
985    pub(crate) fn begin_call(
986        &mut self,
987        resolved: Resolved,
988        name: &[u8],
989        arguments: &mut Vec<Option<ObjRef>>,
990        from: usize,
991        call_type: CallType,
992    ) -> Result<Begun, Failure> {
993        // **The builtin outcome ends here**, before `SIGL`, before the depth
994        // guard and before any activation is pushed -- each of those three is
995        // the label path's, and the builtin path has none of them (`builtin`'s
996        // own module doc carries the probe for each). Every argument is still
997        // rooted by the loop above, so the allocation a builtin's result costs
998        // happens with the inputs reachable, and the value handed back is
999        // rooted by whichever caller receives it exactly as a callee's `RETURN`
1000        // value already is. **Every arm below the running package's own
1001        // `::ROUTINE`s passes the manager first**, an unresolvable name
1002        // included.
1003        if matches!(
1004            resolved,
1005            Resolved::External | Resolved::Macro | Resolved::Registered | Resolved::Unresolved
1006        ) {
1007            if let Some(handled) = self.call_checkpoint(name, &arguments[from..])? {
1008                return Ok(Begun::Done(Ended::Returned(handled)));
1009            }
1010            // The function exits next.
1011            if self.exits.is_some()
1012                && let Some(handled) = self.function_exits(name, call_type, &arguments[from..])?
1013            {
1014                return Ok(Begun::Done(Ended::Returned(handled)));
1015            }
1016        }
1017        if let Resolved::Unresolved = resolved {
1018            // The scripting exit, after every search (step 4).
1019            if self.exits.is_some()
1020                && let Some(handled) = self.scripting_exit(name, call_type, &arguments[from..])?
1021            {
1022                return Ok(Begun::Done(Ended::Returned(handled)));
1023            }
1024            return Err(Raised::routine_not_found(name).into());
1025        }
1026        // **And the external file ends here for the same reason**, which is
1027        // the whole of why it is not an `Entered`: a file the search found is
1028        // a program, entered the way the command line's is.
1029        if let Resolved::External = resolved {
1030            return self
1031                .enter_external_program(name, arguments[from..].to_vec(), call_type)
1032                .map(|value| Begun::Done(Ended::Returned(value)));
1033        }
1034        if let Resolved::Macro = resolved {
1035            return self
1036                .enter_macro(name, arguments[from..].to_vec(), call_type)
1037                .map(|value| Begun::Done(Ended::Returned(value)));
1038        }
1039        if let Resolved::Registered = resolved {
1040            return self
1041                .call_classic_function(name, &arguments[from..])
1042                .map(|value| Begun::Done(Ended::Returned(value)));
1043        }
1044
1045        // A library routine merged into the package, and a `::ROUTINE` bound
1046        // to native code, run no activation either, and are found before the
1047        // manager is asked.
1048        if let Resolved::MergedLibraryRoutine(code, copy) = resolved {
1049            let executable = self.found_routine_object(code, copy);
1050            return self
1051                .run_library_routine(code, name, &arguments[from..], executable)
1052                .map(library_begun);
1053        }
1054        if let Resolved::MergedInternal(row, copy) = resolved {
1055            return self
1056                .run_internal_given(row, InternalName::Called(name), arguments, from, copy)
1057                .map(internal_begun);
1058        }
1059        if let Resolved::Routine(installed) = resolved {
1060            if let Some(code) = self.library_routine_code(installed) {
1061                return self
1062                    .run_library_routine(
1063                        code,
1064                        name,
1065                        &arguments[from..],
1066                        crate::RoutineObject::OfCode,
1067                    )
1068                    .map(library_begun);
1069            }
1070            if let Some(row) = self.rexx_routine_row(installed) {
1071                return self
1072                    .run_internal_as(row, InternalName::Called(name), arguments, from)
1073                    .map(internal_begun);
1074            }
1075            if let Some(classic) = self.classic_routine_entry(installed) {
1076                return self
1077                    .call_classic_entry(name, &arguments[from..], |name, arguments, queue| {
1078                        classic.call_function(name, arguments, queue)
1079                    })
1080                    .map(|value| Begun::Done(Ended::Returned(value)));
1081            }
1082        }
1083
1084        let entered = match resolved {
1085            // Answered above, before the loop that just ran.
1086            Resolved::Builtin(_) => unreachable!("the builtin path returns before this"),
1087            Resolved::Internal(_) => unreachable!("the internal path returns before this"),
1088            Resolved::LibraryRoutine(_)
1089            | Resolved::MergedLibraryRoutine(..)
1090            | Resolved::MergedInternal(..) => {
1091                unreachable!("a library or merged internal routine is run before this")
1092            }
1093            Resolved::Label(target) => Entered::Label(target),
1094            Resolved::Routine(installed) => Entered::Routine(installed),
1095            Resolved::External => unreachable!("the external path returns just above"),
1096            Resolved::Macro => unreachable!("the macro path returns just above"),
1097            Resolved::Registered => unreachable!("the registered path returns just above"),
1098            Resolved::Unresolved => unreachable!("an unresolved name raises just above"),
1099        };
1100
1101        // `SIGL`, set here rather than before the argument loop above: an
1102        // internal call receives its arguments already evaluated by its caller,
1103        // so they are evaluated under whatever `SIGL` was already in force, and
1104        // only then does the transfer overwrite it. So `signal there` / `there:
1105        // call sub sigl` into `sub: use arg a` reports the argument as `1` (the
1106        // `SIGNAL`'s own line, still in force during evaluation) and `sub`'s
1107        // own `SIGL` as the `CALL`'s line -- a version setting `SIGL` before
1108        // evaluating arguments would report the argument as the `CALL`'s own
1109        // line instead.
1110        if matches!(entered, Entered::Label(_)) {
1111            self.set_sigl(self.activity.clause_state.line());
1112        }
1113
1114        // One Rust frame per activation, plus this counter, so an unbounded
1115        // recursion becomes a reportable condition instead of a native abort.
1116        // `Raised::insufficient_stack` already existed (`error.rs`); the same
1117        // 11.1 is answered at rc 245 for the same program.
1118        let probe = 0u8;
1119        if self.nesting_depth() >= MAX_ACTIVATION_DEPTH || self.stack_exhausted(&raw const probe) {
1120            return Err(Raised::insufficient_stack().into());
1121        }
1122
1123        let callee_id = self.next_activation_id();
1124        match entered {
1125            Entered::Label(target) => {
1126                // **D9r's default: a shared pool.** The callee reuses the
1127                // caller's `SlotFrame`, so it reads and writes the caller's
1128                // variables and its writes survive the return -- measured,
1129                // and `pop_slots` is deliberately not called on the way out
1130                // because the frame is not this activation's to free. Task
1131                // 5's `PROCEDURE` is what will ever push a frame of its own.
1132                let caller = self.activation();
1133                // The caller's own program and body selector, which a label
1134                // callee inherits: the same pair `resolve_call` searched.
1135                let program = Rc::clone(&caller.program);
1136                let program_id = caller.program_id;
1137                let selector = caller.body;
1138                let plan = Rc::clone(&caller.plan);
1139                let frame = caller.frame;
1140                // The `call_type` parameter is deliberately not read here: an
1141                // internal label answers `PARSE SOURCE`'s second word from the
1142                // activation it was called out of, whichever form called it.
1143                // Measured, a `::routine` invoked as a function whose body
1144                // `CALL`s a label reads `FUNCTION` inside that label, not
1145                // `SUBROUTINE`.
1146                let caller_call_type = caller.call_type;
1147                let caller_rs = caller.rs;
1148                let settings = caller.settings.clone();
1149                let trace_mode = caller.trace_mode;
1150                // Inherited with `settings` and for its reason: a `SIGNAL ON`
1151                // in the caller turned the package's escalation off for the
1152                // rest of that activation, and an internal call is still
1153                // inside it.
1154                let condition_syntax = caller.condition_syntax;
1155                let extra = caller.extra.clone();
1156                // Cloned with `extra`, and for the same reason: a label
1157                // reached without `PROCEDURE` shares the caller's pool, and an
1158                // exposed name is part of that pool. Measured -- a class
1159                // method exposing `v`, calling a label that assigns `v`, and a
1160                // second class method reading `v` back -- the assignment
1161                // reaches the object variable. Without this the label writes
1162                // the frame slot the exposure left empty and the write is
1163                // lost.
1164                let exposed = caller.exposed.clone();
1165                // Cloned in and never written back, exactly like `settings`
1166                // and `trace_mode` beside it -- `Activation::traps`' own doc
1167                // comment has the three probes that measure the inheritance
1168                // and its one-way direction.
1169                let traps = caller.traps.clone();
1170                // Both halves of the pair, not just the current one:
1171                // measured, a callee's own bare `ADDRESS` swaps to the
1172                // *caller's* alternate. `Activation::address`' own doc
1173                // comment has the transcript.
1174                let address = caller.address.clone();
1175                let caller_io_configs = caller.io_configs.clone();
1176                // Same one-way rule again: an internal call sees the caller's
1177                // `CONDITION()` answers and a reset inside the callee dies
1178                // with it. `TrappedCondition`'s own doc comment has the
1179                // four-line transcript. `RAISE PROPAGATE`'s condition and the
1180                // elapsed clock are copied the same way.
1181                let cold = caller
1182                    .cold
1183                    .as_deref()
1184                    .and_then(crate::activation::ActivationCold::inherited);
1185                let (cached_clock, elapsed_anchor, elapsed_reset) = (
1186                    caller.cached_clock,
1187                    caller.elapsed_anchor,
1188                    caller.flags.elapsed_reset(),
1189                );
1190                let debug = caller.debug;
1191                let mut callee = Activation::nested(
1192                    callee_id,
1193                    program,
1194                    program_id,
1195                    selector,
1196                    plan,
1197                    frame,
1198                    target,
1199                    Inherited {
1200                        call_type: caller_call_type,
1201                        rs: caller_rs,
1202                        settings,
1203                        condition_syntax,
1204                        trace_mode,
1205                        address,
1206                        // The refcount, not the table: a write in the callee
1207                        // clones first, which is how the inheritance stays
1208                        // one-way.
1209                        io_configs: caller_io_configs,
1210                        traps,
1211                        cold,
1212                        cached_clock,
1213                        elapsed_anchor,
1214                        elapsed_reset,
1215                        debug,
1216                    },
1217                );
1218                callee.extra = extra;
1219                callee.exposed = exposed;
1220                self.push_activation(callee);
1221            }
1222            // **A pool of its own, and not one of the five inheritances**
1223            // -- `Activation::routine` is where that is stated and
1224            // `Activation::nested`'s own doc carries the six probes. The
1225            // plan is the routine body's own, cached under its own
1226            // `BodyKey`, and it is what sizes the frame: a routine's names
1227            // are not the caller's, so a frame sized from the caller's plan
1228            // would be the wrong length.
1229            Entered::Routine(installed) => {
1230                // Reached through `programs` rather than through the running
1231                // activation's own `Rc`, so the plan's cache key and the
1232                // activation's program are the same program by construction
1233                // (`InstalledRoutine`'s own doc).
1234                let Some(routine_program) = self.programs.get(installed.program).map(Rc::clone)
1235                else {
1236                    return Err(Loud::missing_body().into());
1237                };
1238                let Some(body) = body_of(&routine_program, Some(installed.directive)) else {
1239                    return Err(Loud::missing_body().into());
1240                };
1241                let plan = self.plan_for(
1242                    BodyKey {
1243                        program: installed.program,
1244                        directive: Some(installed.directive),
1245                    },
1246                    body,
1247                    &routine_program.symbols,
1248                    &routine_program.source,
1249                );
1250                // The routine's own package, which is not always the caller's
1251                // -- `installed.program` is where the `::ROUTINE` was
1252                // declared. The caller's settings go with it for
1253                // `::OPTIONS NUMERIC INHERIT`, the one option that reads
1254                // them.
1255                // A routine started on an activity of its own has no caller.
1256                let start = self.package_start(
1257                    installed.program,
1258                    self.running_activation().map(|caller| &caller.settings),
1259                );
1260                let frame = self.roots.push_slots(plan.len());
1261                let mut callee = Activation::routine(
1262                    callee_id,
1263                    routine_program,
1264                    installed.program,
1265                    installed.directive,
1266                    plan,
1267                    frame,
1268                    call_type,
1269                );
1270                if let Some(start) = start {
1271                    start.apply(&mut callee);
1272                }
1273                self.push_activation(callee);
1274                if self.trace_mode().labels {
1275                    // The `>I>` line's `STACKFRAME` names the routine by the
1276                    // name it was called by (`r[class.stackframe.name]`), and
1277                    // its calling convention is installed only after this.
1278                    let called_as = self.activity.invocation_name(name);
1279                    let called_with = self.shared_arguments(&arguments[from..]);
1280                    let callee = self.activation_mut();
1281                    callee.call_name = Some(called_as);
1282                    callee.call_arguments = Some(called_with);
1283                }
1284                self.trace_package_invocation_entry();
1285            }
1286        }
1287
1288        // Level state for the callee, five pieces, saved here and restored
1289        // on both paths by `finish_call`. `Interpret`'s own arm is the model for four
1290        // of them, and one differs from it deliberately -- the fifth,
1291        // `clause_state`, is not level state for the callee at all, and is
1292        // saved and restored for a different reason stated where it is:
1293        let saved_clause_state = self.save_clause_state();
1294        let callee_indent = match entered {
1295            Entered::Label(_) => saved_clause_state.value_indent() + 2,
1296            Entered::Routine(_) => 0,
1297        };
1298        let saved_base = std::mem::replace(&mut self.activity.activation_indent, callee_indent);
1299        let saved_line = std::mem::take(&mut self.activity.clause_line_override);
1300        let inherited = entered_receiver(entered, self.activity.call_context.receiver);
1301        let arguments = self.shared_arguments(&arguments[from..]);
1302        let name = self.activity.invocation_name(name);
1303        let saved_context = std::mem::replace(
1304            &mut self.activity.call_context,
1305            CallContext {
1306                name,
1307                arguments,
1308                // Read out of the caller's own convention before this
1309                // replaces it, which is the only place it can be read from:
1310                // `entered_receiver` carries the rule and the measurement
1311                // for each of its arms.
1312                receiver: inherited,
1313            },
1314        );
1315
1316        self.activity.call_tails.push(CallTail {
1317            kind: match entered {
1318                Entered::Label(_) => TailKind::Label,
1319                Entered::Routine(_) => TailKind::Routine,
1320            },
1321            saved_clause_state,
1322            saved_base,
1323            saved_line,
1324            saved_context,
1325            stack: LentStack::Kept,
1326        });
1327        // `RXTRACE=ON` starts every `::ROUTINE` that is called under `TRACE ?R`, as it
1328        // starts the program. Here and not beside the activation's push, because the
1329        // `>I>` line it may write takes the margin the level state above has just set.
1330        if matches!(entered, Entered::Routine(_)) {
1331            self.start_external_trace();
1332        }
1333        // A routine is a program-level call, which runs the initialization exit before its
1334        // first clause; its failure ends the call there.
1335        if matches!(entered, Entered::Routine(_))
1336            && self.calls_program_exits()
1337            && let Err(failure) = self.initialization_exit()
1338        {
1339            return match self.finish_call(Err(failure)) {
1340                Err(failure) => Err(failure),
1341                Ok(_) => Err(Loud::scheduler_inconsistency(
1342                    "a failed initialization exit ended its call normally",
1343                )
1344                .into()),
1345            };
1346        }
1347        Ok(Begun::Entered)
1348    }

1350    /// [`Interp::invoke_call_over`] from the point its callee's body has run
1351    /// and ended `ended`.
1352    #[inline(always)]
1353    pub(crate) fn finish_call(&mut self, ended: Result<Ended, Failure>) -> Result<Ended, Failure> {
1354        let Some(CallTail {
1355            kind,
1356            saved_clause_state,
1357            saved_base,
1358            saved_line,
1359            saved_context,
1360            stack,
1361        }) = self.activity.call_tails.pop()
1362        else {
1363            return Err(Loud::op_not_driven("a call with no saved level state").into());
1364        };
1365        // A routine that returned or exited runs the termination exit while its variables are
1366        // live; its failure is the call's.
1367        let ended = if kind == TailKind::Routine
1368            && self.calls_program_exits()
1369            && matches!(ended, Ok(_) | Err(Failure::Exited(_)))
1370        {
1371            match self.termination_exit() {
1372                Ok(()) => ended,
1373                Err(failure) => Err(failure),
1374            }
1375        } else {
1376            ended
1377        };
1378        if let Err(failure) = &ended {
1379            self.capture_activation_frame(failure);
1380        }
1381
1382        // Termination releases the guard lock first. With one
1383        // activity there is nothing to release; a `<I<` drops the flag first.
1384        if kind == TailKind::Method
1385            && (self.activities.guards.is_live() || !self.activity.guard_waits.is_empty())
1386        {
1387            self.guard_off_at_end();
1388        }
1389        // Before the pop, because both halves of `<I<`'s gate are the callee's
1390        // own -- its `trace_entry` state and its `TRACE` setting. It is part of
1391        // termination, which is likewise inside the activation.
1392        self.trace_invocation_exit(|| match &ended {
1393            Ok(ended) => ended.value_ref(),
1394            Err(Failure::Exited(value)) => *value,
1395            Err(_) => None,
1396        });
1397
1398        // Popped on both paths, and unconditionally: `run_activation`'s own
1399        // loop asserts the activation stack is where it found it after every
1400        // step, so a `CALL` that returned with the callee still on it would
1401        // trip that assertion in the caller rather than quietly running the
1402        // wrong frame's `pc`.
1403        let mut callee = self.pop_activation().expect("the activation just pushed");
1404        let mut ended = ended;
1405        match kind {
1406            TailKind::Method => {
1407                if let Some(replied) = self.release_method_activation(callee, saved_context)
1408                    && ended.is_ok()
1409                {
1410                    ended = Ok(Ended::Returned(replied));
1411                }
1412            }
1413            TailKind::Label | TailKind::Routine => {
1414                // **The two halves of "was the pool shared" are one bool, and
1415                // both are needed.** A `PROCEDURE` callee pushed a frame of its
1416                // own, so that frame is popped here -- on the error path as
1417                // well, which is why this is not inside the `Ok` arm below. It
1418                // also keeps its own run-time name bindings, so they are *not*
1419                // moved back: doing that would overwrite the caller's `extra`
1420                // with the callee's isolated one. A shared-pool callee is the
1421                // opposite on both counts, and its `extra` write-back is what
1422                // makes a name bound inside it survive the return (measured,
1423                // `interpret "zork = 42"` in a callee).
1424                if callee.owns_frame {
1425                    self.roots.activity_mut().pop_slots(callee.frame);
1426                } else {
1427                    // Taken rather than moved out, so the box stays whole and
1428                    // can be parked: moving a field out of a `Box` moves the
1429                    // whole of it out and frees the box, which is the
1430                    // allocation the pool exists to keep. What is left behind
1431                    // is the empty map a fresh activation starts with.
1432                    self.activation_mut().extra = std::mem::take(&mut callee.extra);
1433                }
1434                self.recycle_activation(callee);
1435                self.activity.call_context = saved_context;
1436            }
1437        }
1438        self.activity.activation_indent = saved_base;
1439        self.activity.clause_line_override = saved_line;
1440        self.restore_clause_state(saved_clause_state);
1441        match stack {
1442            LentStack::Kept => {}
1443            LentStack::Emptied => self.activity.value_buffer.clear(),
1444            LentStack::Lent(values) => self.activity.value_buffer = values,
1445        }
1446        // A returning activation runs the `UNINIT`s a collection readied (the
1447        // uninit queue is checked here). A method's run in
1448        // `Interp::finish_send`, once the primitive methods waiting on it have
1449        // finished, since an `UNINIT`'s own send would otherwise find their
1450        // tails at its depth.
1451        if !self.uninit_ready.is_empty()
1452            && kind != TailKind::Method
1453            && let Ok(Ended::Returned(value) | Ended::Exited(value)) = &ended
1454        {
1455            self.run_uninits_at_return(*value)?;
1456        }
1457
1458        // **`EXIT` inside a `::ROUTINE` ends the routine, not the program**,
1459        // where `EXIT` inside a `CALL`ed label ends the program. `RETURNED` is
1460        // set outright for a program-level call activation and only otherwise
1461        // does the walk go up, and a routine invocation is one of those. For
1462        // example, rc 0 every time, and the shapes matter because the exit
1463        // arrives here by three different routes:
1464        // ```text
1465        // call rtn / say result       ::routine rtn ; exit 5      ->  5, and main runs on
1466        // n1 = rtn() / say n1         ::routine rtn ; exit 7      ->  7
1467        // call rtn / say result       ::routine rtn ; exit        ->  RESULT, unset
1468        // call rtn / say 'after'      a label INSIDE the routine exits 9 -> "after" runs
1469        // call rtn / say 'after'      interpret "exit 4" in the routine  -> "after" runs
1470        // ```
1471        // A method body's `EXIT` answers its sender the same way.
1472        let ended = match ended {
1473            Ok(Ended::Exited(value)) | Err(Failure::Exited(value)) if kind != TailKind::Label => {
1474                return Ok(Ended::Returned(value));
1475            }
1476            other => other,
1477        };
1478
1479        match ended {
1480            Ok(ended) => Ok(ended),
1481            Err(failure) => {
1482                // Seal before the failure leaves the callee, never after --
1483                // `seal_site_level`'s own rule, and the same one `run_fragment`
1484                // follows. Without it the callee's clause would win
1485                // `record_failure_at`'s first-wins race outright and the call
1486                // would never be echoed. One echo is printed per level,
1487                // innermost first: a `say 1/0` in a routine called from a
1488                // routine called from a `DO` gives three lines, at indents 6, 4
1489                // and 2.
1490                self.seal_site_level();
1491                Err(failure)
1492            }
1493        }
1494    }

1522    /// Builds the object a `>name` or `<name` term answers: the variable
1523    /// `inner` names, rather than the value it holds.
1524    pub(crate) fn variable_reference(
1525        &mut self,
1526        code: &Code<'_>,
1527        id: SymbolId,
1528        stem: bool,
1529    ) -> Result<ObjRef, Failure> {
1530        let slot = match code.slot_for(id) {
1531            Some(slot) => slot,
1532            None => self.slot_of(code.symbols.name(id).as_bytes()),
1533        };
1534        let frame = self.activation().frame;
1535        let spelling = code.symbols.name(id).as_bytes();
1536        // A reference uses the stem variable it names, which creates it.
1537        if stem && self.variable(frame, slot).is_none() {
1538            self.create_stem_variable(frame, slot, spelling);
1539        }
1540        // Resolved here, where the name's own home is: an alias this
1541        // activation holds is still addressable, which is what makes `>p`
1542        // work when its `p` came from *its* caller, and an `EXPOSE` binding
1543        // is on this activation, which is what makes it work on an object
1544        // variable. Chasing the slot for an exposed name would name the
1545        // empty slot the exposure left behind. Either way the reference names
1546        // the variable the name resolves to, under that variable's own name:
1547        // an alias of the caller's `v` is a reference to `V`.
1548        let (home, name) = match self.exposure(frame, slot) {
1549            Some(var) => (
1550                VarRefHome::Instance {
1551                    owner: var.owner,
1552                    scope: var.scope,
1553                },
1554                Box::from(&*var.name),
1555            ),
1556            None => {
1557                let cell = self.promote_slot(frame, slot, spelling);
1558                let name = self.cell_owner_name(frame, slot).unwrap_or(spelling).into();
1559                (VarRefHome::Cell(cell), name)
1560            }
1561        };
1562        // A reference's string value is its referenced value's conversion,
1563        // which a rendering of the reference does not give.
1564        self.reqstr_armed = true;
1565        Ok(self.alloc_with(
1566            BehaviourId::OBJECT,
1567            Body::VarRef(Box::new(interrexx_core::VarRef { name, home })),
1568        ))
1569    }

1626    /// [`Interp::resolved_after_arguments`]' failing half: run the arguments
1627    /// for their trace lines and their own conditions, then report the
1628    /// resolution failure if none of them raised first.
1629    #[cold]
1630    #[inline(never)]
1631    fn arguments_before_failure(
1632        &mut self,
1633        code: &Code<'_>,
1634        args: Args<'_>,
1635        failure: Failure,
1636    ) -> Failure {
1637        // The same three shapes `invoke_call`'s own loop has, for their trace
1638        // lines and their failures; the values themselves are dropped, since
1639        // there is no callee to hand them to.
1640        for arg in args.iter() {
1641            let raised = match arg {
1642                None => {
1643                    self.trace_argument(self.activity.clause_state.current_value_indent, b"");
1644                    continue;
1645                }
1646                Some(expr) if self.leaf_argument(expr) => self.eval_leaf_argument(code, expr).err(),
1647                Some(expr) => pinned!(
1648                    self,
1649                    crate::pinning::PinKind::TreeEval,
1650                    self.eval_traced_argument(code, expr)
1651                )
1652                .err(),
1653            };
1654            if let Some(raised) = raised {
1655                return raised;
1656            }
1657        }
1658        failure
1659    }

1710    /// What a `CALL` does with the outcome its callee handed back: the
1711    /// caller's own `>>>` and `RESULT`.
1712    pub(crate) fn settle_call_result(
1713        &mut self,
1714        ended: Ended,
1715        base_indent: usize,
1716    ) -> Result<Flow, Failure> {
1717        let value = match ended {
1718            // `EXIT` inside the callee ends the program rather than the
1719            // call, and so does running off the end of the body -- measured
1720            // both ways. Forwarded unchanged; `RESULT` is never touched on
1721            // this path.
1722            Ended::Exited(value) => return Ok(Flow::Exit(value)),
1723            Ended::Returned(value) => value,
1724        };
1725
1726        // **`RESULT` is settled on return and not at the call.** Measured:
1727        // a caller setting `result = 'before'` and calling a no-`PROCEDURE`
1728        // routine has the callee print `inside result= before`, so nothing
1729        // is cleared on the way in. After `return 42` the caller reads `42`;
1730        // after a bare `return` it reads the derived name `RESULT`, which is
1731        // what an unset variable renders as.
1732        let slot = self.reserved_result_slot();
1733        let frame = self.activation().frame;
1734        match value {
1735            Some(value) => {
1736                // Re-rooted in the caller: `Op::Clause`'s region popped the
1737                // callee's temps frame around every clause it ran, this one
1738                // included, so the `push_temp` the `RETURN` arm did is gone
1739                // by now. Same window `Flow::Exit`'s own arm documents,
1740                // closed here rather than left open, because unlike an exit
1741                // value this one goes on to be stored and read.
1742                self.roots.activity_mut().push_temp(value);
1743                // The caller's own `>>>`, at the `CALL` clause's indent --
1744                // `base_indent`, saved before the callee overwrote
1745                // `current_value_indent` with its own clauses'.
1746                if !self.activity.clause_state.instance_trace_started
1747                    && let Some(rendered) = self.result_text(value)?
1748                {
1749                    self.trace_result(base_indent, &rendered);
1750                }
1751                self.set_variable(frame, slot, value);
1752            }
1753            None => self.clear_variable(frame, slot),
1754        }
1755        Ok(Flow::Next)
1756    }
1757
1758    /// Runs a started `CALL`'s callee body on this Rust stack, and settles its
1759    /// `RESULT` at `base_indent`.
1760    pub(crate) fn complete_subroutine(
1761        &mut self,
1762        started: Started<Ended>,
1763        base_indent: usize,
1764    ) -> Result<Flow, Failure> {
1765        let ended = match started {
1766            Started::Ran(ended) => ended,
1767            Started::Entered if self.activity.native_park.is_some() => {
1768                return Err(self.refuse_unrooted_park());
1769            }
1770            Started::Entered => {
1771                let ended = self.run_activation();
1772                self.finish_call(ended)?
1773            }
1774        };
1775        self.settle_call_result(ended, base_indent)
1776    }
1777
1778    /// A `CALL` over arguments the compiled stream has already evaluated onto
1779    /// the argument stack above `mark`, up to the point its callee's body
1780    /// would run: the argument run is lent out, turned into `Argument`s, and
1781    /// removed again whichever way the call went on.
1782    #[inline(always)]
1783    pub(crate) fn begin_subroutine_over_pushed_args(
1784        &mut self,
1785        resolved: Resolved,
1786        name: &[u8],
1787        mark: usize,
1788    ) -> Result<Started<Ended>, Failure> {
1789        let mut values = std::mem::take(&mut self.activity.value_buffer);
1790        let outcome = match resolved {
1791            // No activation, no `Argument`s built -- the shortcut
1792            // `begin_call_over_values` takes, and the reason a `CALL` to a builtin
1793            // reaches `Ended::Returned(Some(_))` with a value to settle.
1794            Resolved::Builtin(target) => {
1795                let answer = builtin::run(self, name, target, &values[mark..]);
1796                builtin::answered_nothing(self, answer)
1797                    .map(|value| Started::Ran(Ended::Returned(value)))
1798            }
1799            // **Named rather than left to the arm below**, which has a
1800            // catch-all: an internal routine sent into `invoke_call_over`
1801            // would reach a `match` that has no arm for it.
1802            Resolved::Internal(row) => match self.call_checkpoint(name, &values[mark..]) {
1803                Ok(Some(handled)) => Ok(Started::Ran(Ended::Returned(handled))),
1804                Ok(None) => self
1805                    .run_internal(row, &mut values, mark)
1806                    .map(|started| subroutine_started(internal_begun(started))),
1807                Err(failure) => Err(failure),
1808            },
1809            Resolved::LibraryRoutine(slot) => match self.call_checkpoint(name, &values[mark..]) {
1810                Ok(Some(handled)) => Ok(Started::Ran(Ended::Returned(handled))),
1811                Ok(None) => self
1812                    .run_package_routine(slot, name, &values[mark..])
1813                    .map(|started| subroutine_started(library_begun(started))),
1814                Err(failure) => Err(failure),
1815            },
1816            _ => self
1817                .begin_call(resolved, name, &mut values, mark, CallType::Subroutine)
1818                .map(subroutine_started),
1819        };
1820        values.truncate(mark);
1821        self.lend_stack(self.entered_body(&outcome), values);
1822        outcome
1823    }
1824}

1826/// A begun function call as a call op starts it.
1827#[inline(always)]
1828pub(crate) fn function_started(begun: Begun, name: &[u8]) -> Result<Started<ObjRef>, Failure> {
1829    match begun {
1830        Begun::Done(ended) => function_value(ended, name).map(Started::Ran),
1831        Begun::Entered => Ok(Started::Entered),
1832    }
1833}

1836fn internal_begun(started: Started<ObjRef>) -> Begun {
1837    match started {
1838        Started::Ran(value) => Begun::Done(Ended::Returned(Some(value))),
1839        Started::Entered => Begun::Entered,
1840    }

1847        Started::Ran(value) => Begun::Done(Ended::Returned(value)),
1848        Started::Entered => Begun::Entered,

1853pub(crate) fn subroutine_started(begun: Begun) -> Started<Ended> {
1854    match begun {
1855        Begun::Done(ended) => Started::Ran(ended),

1857    }

1860/// A function call's value, from the way its callee ended.
1861#[inline(always)]
1862fn function_value(ended: Ended, name: &[u8]) -> Result<ObjRef, Failure> {
1863    match ended {
1864        // `EXIT` inside the routine, or the routine falling off its own
1865        // end, ends the whole program exactly as it does when the same
1866        // routine is reached through `CALL` (`Interp::invoke_call`'s
1867        // own doc, `run/call.rs`). Propagated as `Failure::Exited` because
1868        // `eval`'s own return type is a plain `ObjRef` with no `Flow` to
1869        // carry the event through instead -- see that variant's own doc
1870        // (`error.rs`) for why every intervening `?` needs no special
1871        // handling to still unwind every nested `CALL` correctly.
1872        Ended::Exited(value) => Err(Failure::Exited(value)),
1873        Ended::Returned(Some(value)) => Ok(value),
1874        // For example, a routine reached through the expression form and
1875        // returning nothing (a bare `RETURN`) is Error 44.1 rc 212, naming the function -- the expression form's own answer to
1876        // "nothing to use here", which `CALL` never has to give since its own
1877        // value only ever reaches `RESULT`, unset or not.
1878        Ended::Returned(None) => Err(Raised::no_data_returned(name).into()),
1879    }
