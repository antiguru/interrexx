14    /// How many further pauses to skip, from a numeric `TRACE`
15    /// typed at one. A negative count also suppresses the echo -- measured,
16    /// `trace 2` traces the two clauses it skips the pause for and
17    /// `trace -2` traces neither.
18    fn set_debug_skip(&mut self, count: i64) -> Result<(), Failure> {
19        if !self.debug_pause() {
20            return Err(raised_numeric_trace_interactive_only().into());
21        }
22        self.activation_mut().debug.skip = count.abs();
23        if count < 0 {
24            // **Through `set_trace_mode`, which keeps the cache
25            // `Interp::trace_mode` reads.** Writing the activation's field
26            // directly left the cache holding the un-suppressed setting, and
27            // only the accessor's own `debug_assert` would have said so.
28            let current = self.trace_mode();
29            self.activation_mut().debug.saved = Some(current);
30            self.set_trace_mode(crate::trace::TraceMode {
31                debug: true,
32                letter: current.letter,
33                ..crate::trace::TraceMode::OFF
34            });
35        }
36        // The setting changed from inside the pause, so the pause ends
37        // without reading another line.
38        self.end_debug_stop();
39        Ok(())
40    }

42    /// Ends the stop a line typed at a pause runs in, once the line is done,
43    /// as `r[env.debugging.input.trace]` has any `TRACE` the line runs do,
44    /// whatever setting it leaves. Does nothing outside a pause.
45    pub(crate) fn end_debug_stop(&mut self) {
46        if self.debug_pause() {
47            self.activation_mut().debug.bypass = true;
48        }
49    }

51    /// One parsed `TRACE` setting, merged with the setting in force.
52    fn apply_trace_request(&mut self, request: &crate::trace::TraceRequest) {
53        let merged = crate::trace::applied(self.trace_mode(), request);
54        self.set_trace_mode(merged);
55        self.end_debug_stop();
56    }

58    /// Whether a `TRACE` instruction leaves the setting alone, and calls
59    /// neither the setting change nor the `>I>` it makes.
60    ///
61    /// **A `TRACE` instruction is ignored while interactive debug is on**,
62    /// and traced like any other clause: measured, a program that reaches
63    /// `trace ?` and then `trace off` under `?R` still answers `?R` from
64    /// `TRACE()` and keeps tracing. The builtin has no such guard, which is
65    /// why it lives here rather than in `set_trace_mode`.
66    ///
67    /// **"In debug" means debug is on and no pause is running**, so the same
68    /// instruction the program cannot use is what a line typed at the
69    /// pause uses to end debug -- measured, `trace off` at a prompt ends
70    /// the session and the pause with it.
71    fn ignores_trace_instruction(&self) -> bool {
72        self.trace_mode().debug && !self.debug_pause()
73    }

75    /// `NUMERIC DIGITS`/`FUZZ`/`FORM`, in every spelling the parser produces
76    /// (`NumericSetting`, `interrexx-parse`'s own `instruction.rs::numeric`).
77    pub(super) fn exec_trace(&mut self, code: &Code<'_>, setting: &Trace) -> Result<(), Failure> {
78        match setting {
79            Trace::Default => {
80                // The normal setting, which is silent here and is *not*
81                // `TRACE OFF` -- measured, `trace r` then bare `trace` then
82                // `trace()` gives `N`, where `trace off` gives `O`.
83                if self.ignores_trace_instruction() {
84                    return Ok(());
85                }
86                self.set_trace_mode(crate::trace::TraceMode::NORMAL);
87                self.end_debug_stop();
88            }
89            Trace::Setting(bytes) => {
90                if self.ignores_trace_instruction() {
91                    return Ok(());
92                }
93                let request = crate::trace::parse_trace_request(bytes)
94                    .expect("interrexx-parse's check_trace_setting already validated this byte");
95                self.apply_trace_request(&request);
96            }
97            // Valid only from
98            // a pause, and 24.901 anywhere else -- measured, `trace 0` in a
99            // program raises it exactly like `trace 5`.
100            Trace::Skip(count) => {
101                self.set_debug_skip(*count)?;
102            }
103            // `TRACE VALUE expr`: computed at run time, then classified
104            // exactly like a literal `TRACE` setting would have been --
105            // measured, `trace value 5` raises 24.901 like `trace 5`, and
106            // `trace value 'R'` behaves exactly like `trace r` (this
107            // task's report has both transcripts). A whole number is a
108            // skip count checked *before* trying it as a letter, matching
109            // `interrexx-parse`'s own `trace` parser's order (`instruction.rs`'s
110            // `whole_number` attempt precedes its `check_trace_setting`
111            // fallback).
112            Trace::Value(expression) => {
113                let value = self.eval(code, code.body.expr(*expression))?;
114                return self.exec_trace_value(code, value);
115            }
116        }
117        self.announce_entry_after_trace(code);
118        Ok(())
119    }

121    /// `TRACE VALUE`'s work after its expression, from its value: what the
122    /// walked instruction does once it has evaluated it, and what
123    /// `Op::ExecValue` does for a compiled one.
124    pub(crate) fn exec_trace_value(
125        &mut self,
126        code: &Code<'_>,
127        value: ObjRef,
128    ) -> Result<(), Failure> {
129        self.roots.activity_mut().push_temp(value);
130        // `>K>   "VALUE" => text`, naming the object before the
131        // required-string conversion below, like `SIGNAL VALUE`.
132        let shown = self.to_text(value).to_vec();
133        self.trace_keyword(
134            self.activity.clause_state.current_value_indent,
135            "VALUE",
136            &shown,
137        );
138        // The string request, and 24.1 names the conversion: `trace value .K`
139        // with a class-side `makeString` returning `'ZZ'` reports `found
140        // "Z"`.
141        let value = self.required_string_value(value)?;
142        let text = self.to_text(value).to_vec();
143        if is_whole_number(&text) {
144            let count = String::from_utf8_lossy(&text)
145                .trim()
146                .parse::<i64>()
147                .unwrap_or(0);
148            self.set_debug_skip(count)?;
149            return Ok(());
150        }
151        // Ignored before it is read as a setting, so an invalid one
152        // raises nothing there.
153        if self.ignores_trace_instruction() {
154            return Ok(());
155        }
156        let request =
157            crate::trace::parse_trace_request(&text).map_err(raised_invalid_trace_letter)?;
158        self.apply_trace_request(&request);
159        self.announce_entry_after_trace(code);
160        Ok(())
161    }

163    /// The entry is announced right after the new flags are installed, which
164    /// is the route every `>I>` takes. `Trace::Skip` never gets here -- it
165    /// returns before -- and sets no flags and calls nothing.
166    fn announce_entry_after_trace(&mut self, code: &Code<'_>) {
167        self.allow_entry_after_exposes(code);
168        self.trace_invocation_entry();
169    }

171    /// The entry announcement stays allowed past an `EXPOSE`, so a `TRACE` that
172    /// only `EXPOSE`s precede in the body may still announce `>I>`. Nothing but
173    /// those `EXPOSE`s can have run before it: any other way to reach it passes
174    /// a label, a `DO` or another instruction first.
175    fn allow_entry_after_exposes(&mut self, code: &Code<'_>) {
176        if self.activation().trace_entry != TraceEntry::Spent
177            || self.activity.clause_line_override.is_some()
178        {
179            return;
180        }
181        let index = self.activity.clause_state.clause_index();
182        let exposes_only = code.body.instructions.get(..index).is_some_and(|before| {
183            !before.is_empty()
184                && before.iter().all(|clause| {
185                    matches!(clause.kind, interrexx_parse::InstructionKind::Expose { .. })
186                })
187        });
188        if exposes_only {
189            self.activation_mut().trace_entry = TraceEntry::Allowed;
190        }
191    }

193    /// `>I>` while the activation can still announce its entry and its
194    /// setting traces labels, including a queued external announcement.
195    pub(crate) fn trace_invocation_entry(&mut self) {
196        let entry = self.activation().trace_entry;
197        if !entry.may_announce() && entry != TraceEntry::External {
198            return;
199        }
200        let Some(subject) = self.invocation_subject() else {
201            return;
202        };
203        if !self.trace_mode().labels {
204            return;
205        }
206        self.activation_mut().trace_entry = TraceEntry::Done;
207        self.activation_mut().debug.source_traced = true;
208        if self.entry_traced_before_retry() {
209            return;
210        }
211        let package = self
212            .program_display_name(self.activation().program_id)
213            .to_vec();
214        self.trace_invocation(">I>", &subject, &package);
215    }

217    /// `>I>` for a body whose package's `::OPTIONS TRACE` put a

219    pub(crate) fn trace_package_invocation_entry(&mut self) {
220        if !self.trace_mode().labels || self.activation().trace_entry != TraceEntry::Pending {
221            return;
222        }
223        let Some(subject) = self.invocation_subject() else {
224            return;
225        };
226        self.activation_mut().trace_entry = TraceEntry::Done;
227        self.activation_mut().debug.source_traced = true;
228        if self.entry_traced_before_retry() {
229            return;
230        }
231        let package = self
232            .program_display_name(self.activation().program_id)
233            .to_vec();
234        self.trace_invocation(">I>", &subject, &package);

237    /// `<I<` on every way a traced routine activation can end, after any
238    /// external entry still due. The value from `ending` stays rooted while
239    /// the lines are delivered.
240    pub(crate) fn trace_invocation_exit(&mut self, ending: impl FnOnce() -> Option<ObjRef>) {
241        let entry = self.activation().trace_entry;
242        if !matches!(entry, TraceEntry::Done | TraceEntry::External) {
243            return;
244        }
245        if self.activation().entry == crate::activation::Entry::Method {
246            self.guard_off_at_end();
247        }
248        let Some(subject) = self.invocation_subject() else {
249            return;
250        };
251        if !self.trace_mode().labels {
252            return;
253        }
254        let package = self
255            .program_display_name(self.activation().program_id)
256            .to_vec();
257        let parked = self
258            .roots
259            .activity_mut()
260            .park(ending().into_iter().collect());
261        if entry == TraceEntry::External {
262            self.trace_invocation_entry();
263        }
264        self.trace_invocation("<I<", &subject, &package);
265        self.roots.activity_mut().release(parked);
266    }

268    /// What the running activation announces itself as, or `None` when it
269    /// announces nothing at all -- which is the method-or-routine half of
270    /// the gate, expressed as the lookup that would supply the substitutions.
271    pub(crate) fn invocation_subject(&self) -> Option<Announced> {
272        if let Some(identity) = &self.activation().method_identity {
273            return Some(Announced::Method {
274                name: identity.name.to_vec(),
275                scope: self.class_id_text(identity.scope).as_bytes().to_vec(),
276            });
277        }
278        // A program called as a routine is announced under its package name, and
279        // so is a prologue, which runs as an external call.
280        if self.activation().entry == crate::activation::Entry::TopLevel
281            && matches!(
282                self.activation().call_type,
283                crate::activation::CallType::Subroutine
284                    | crate::activation::CallType::Function
285                    | crate::activation::CallType::Requires
286            )
287        {
288            let path = self.program_display_name(self.activation().program_id);
289            return Some(Announced::Routine {
290                name: path.to_vec(),
291            });
292        }
293        let index = self.activation().body?;
294        match &self.activation().program.directives.get(index)?.kind {
295            DirectiveKind::Routine(routine) => Some(Announced::Routine {
296                name: routine.name.to_vec(),
297            }),
298            _ => None,
299        }
300    }

302    /// `ADDRESS`'s three environment-naming forms. The caller has already
303    /// turned the command and `WITH` forms away.
304    /// ```text
305    /// address envC                    ->  ENVC
306    /// address 'LiTeRaL'               ->  LiTeRaL
307    /// nm = 'mIxEd'; address value nm  ->  mIxEd
308    /// nm = 'mIxEd'; address (nm)      ->  mIxEd
309    /// ```
310    pub(super) fn exec_address(
311        &mut self,
312        code: &Code<'_>,
313        address: &interrexx_parse::Address,
314    ) -> Result<(), Failure> {
315        // `environment` before `dynamic`, because the environment is tested
316        // ahead of the `ADDRESS VALUE` arm. The parser never fills both, so the
317        // order decides nothing today.
318        match (&address.environment, &address.dynamic) {
319            (None, None) => {
320                self.activation_mut().address.toggle();
321                return Ok(());
322            }
323            (Some(environment), _) => {
324                if environment.len() > MAX_ADDRESS_NAME_LENGTH {
325                    return Err(Raised::environment_name_too_long(
326                        MAX_ADDRESS_NAME_LENGTH,
327                        environment,
328                    )
329                    .into());
330                }
331                self.activation_mut().address.set_bytes(environment);
332                self.store_io_config(code, environment, address);
333            }
334            (None, Some(expression)) => {
335                let value = self.eval(code, code.body.expr(*expression))?;
336                return self.exec_address_value(code, address, value);
337            }
338        }
339        Ok(())
340    }

342    /// `ADDRESS VALUE expr`'s work after its expression, from its value: what
343    /// the walked instruction does once it has evaluated it, and what
344    /// `Op::ExecValue` does for a compiled one.
345    pub(crate) fn exec_address_value(
346        &mut self,
347        code: &Code<'_>,
348        address: &interrexx_parse::Address,
349        value: ObjRef,
350    ) -> Result<(), Failure> {
351        self.roots.activity_mut().push_temp(value);
352        // The string request comes before the `>>>`,
353        // so the trace
354        // names the conversion: measured, `trace r` over
355        // `address (.K)` with a class-side `makeString` returning
356        // `'CMD'` prints `>>>   "CMD"` and `ADDRESS()` answers `CMD`.
357        let value = self.required_string_value(value)?;
358        // Built in the lent buffer: the rendering exists only to be
359        // traced and then copied into the `Rc`, so an owned `Vec` of
360        // its own would be an allocation and a free per `ADDRESS
361        // VALUE` clause.
362        let mut text = self.take_result_buffer();
363        text.extend_from_slice(&self.to_text(value));
364        self.trace_result(self.activity.clause_state.current_value_indent, &text);
365        if text.len() > MAX_ADDRESS_NAME_LENGTH {
366            let raised = Raised::environment_name_too_long(MAX_ADDRESS_NAME_LENGTH, &text);
367            self.give_result_buffer(text);
368            return Err(raised.into());
369        }
370        self.activation_mut().address.set_bytes(&text);
371        self.store_io_config(code, &text, address);
372        self.give_result_buffer(text);
373        Ok(())
374    }

389    /// `NUMERIC` once its expression, if it has one, has answered `value`:
390    /// everything the instruction does past the evaluation, which comes
391    /// first in every form. [`crate::ir::Op::ExecValue`] runs this behind the
392    /// expression's native ops.
393    pub(crate) fn exec_numeric_value(
394        &mut self,
395        setting: &NumericSetting,
396        value: Option<ObjRef>,
397    ) -> Result<(), Failure> {
398        match setting {
399            NumericSetting::Digits => {
400                match self.numeric_operand(value, "DIGITS")? {
401                    Some((operand, text)) => {
402                        let parsed = String::from_utf8_lossy(&text);
403                        let outcome = self.activation_mut().settings.set_digits_str(&parsed);
404                        self.give_result_buffer(text);
405                        if let Err(error) = outcome {
406                            let named = self.to_text(operand).to_vec();
407                            return Err(raised_naming_the_operand(error, &named).into());
408                        }
409                    }
410                    // The reset stores the default and makes the operand
411                    // form's FUZZ check against it, which is the arm the
412                    // interpreter runs: it can fail, and the value it names
413                    // is the default rather than the setting in force.
414                    None => {
415                        let default = self.package_default_numeric().digits();
416                        self.activation_mut()
417                            .settings
418                            .reset_digits(default)
419                            .map_err(raised_from_settings)?;
420                    }
421                }
422            }
423            NumericSetting::Fuzz => match self.numeric_operand(value, "FUZZ")? {
424                Some((operand, text)) => {
425                    let parsed = String::from_utf8_lossy(&text);
426                    let outcome = self.activation_mut().settings.set_fuzz_str(&parsed);
427                    self.give_result_buffer(text);
428                    if let Err(error) = outcome {
429                        let named = self.to_text(operand).to_vec();
430                        return Err(raised_naming_the_operand(error, &named).into());
431                    }
432                }
433                None => {
434                    let default = self.package_default_numeric().fuzz();
435                    self.activation_mut()
436                        .settings
437                        .reset_fuzz(default)
438                        .map_err(raised_from_settings)?;
439                }
440            },
441            // **`FormDefault` and `FormScientific` part company once
442            // `::OPTIONS FORM ENGINEERING` moves the package default**:
443            // measured, a bare `numeric form` in such a file answers
444            // `ENGINEERING` where `numeric form scientific` answers
445            // `SCIENTIFIC`.
446            NumericSetting::FormDefault => {
447                let default = self.package_default_numeric().form();
448                self.activation_mut().settings.set_form(default);
449            }
450            NumericSetting::FormScientific => {
451                self.activation_mut().settings.set_form(Form::Scientific);
452            }
453            NumericSetting::FormEngineering => {
454                self.activation_mut().settings.set_form(Form::Engineering);
455            }
456            NumericSetting::FormValue => {
457                // The parser only ever produces this with an expression: an
458                // explicit `VALUE` with none is 35.917 at parse time
459                // (`instruction.rs::numeric`), and the implicit spelling
460                // (`NUMERIC FORM (expr)`) only takes this branch once a token
461                // is already known to be there. Loud rather than a panic, on
462                // this crate's own rule against aborting on a shape the
463                // grammar rules out but the type does not.
464                let Some(value) = value else {
465                    return Err(Loud::numeric_form_without_expression().into());
466                };
467                // `set_form_str`'s own doc comment: the runtime `VALUE` path
468                // does no uppercasing, no trimming and no abbreviation, unlike
469                // the keyword spellings above -- measured, `numeric form
470                // value 'engineering'` is 25.11, not accepted
471                // case-insensitively.
472                self.roots.activity_mut().push_temp(value);
473                let text = self.to_text(value).to_vec();
474                // `>K>   "FORM" => "engineering"` (F2, branch review): fires
475                // before `set_form_str`'s own validation, same as `DIGITS`/
476                // `FUZZ` below and `setup_controlled`'s own `TO`/`BY`/`FOR`
477                // -- measured, `numeric form value 'engineering'` under
478                // `trace r` traces `>K>` and *then* raises 25.11, not the
479                // reverse. Untranslated, matching the error report's own
480                // `found "engineering"` substitution, which is also
481                // unmodified case -- `set_form_str`'s own no-uppercasing
482                // rule for this one path, unlike the two keyword spellings
483                // above.
484                self.trace_numeric_operand("FORM", &text)?;
485                // The required-string protocol between the `>K>` and the
486                // validation, which is where the string request sits.
487                let converted = self.required_string_value(value)?;
488                let parsed = String::from_utf8_lossy(&self.to_text(converted)).into_owned();
489                if let Err(error) = self.activation_mut().settings.set_form_str(&parsed) {
490                    return Err(raised_naming_the_operand(error, &text).into());
491                }
492            }
493        }
494        Ok(())
495    }

497    /// The operand `value` of `NUMERIC DIGITS`/`FUZZ` with its `>K>` line
498    /// and its string request, or `None` when the instruction has none.
499    fn numeric_operand(
500        &mut self,
501        value: Option<ObjRef>,
502        keyword: &str,
503    ) -> Result<Option<(ObjRef, Vec<u8>)>, Failure> {
504        let Some(value) = value else {
505            return Ok(None);
506        };
507        self.roots.activity_mut().push_temp(value);
508        let mut text = self.take_result_buffer();
509        text.extend_from_slice(&self.to_text(value));
510        if let Err(failure) = self.trace_numeric_operand(keyword, &text) {
511            self.give_result_buffer(text);
512            return Err(failure);
513        }
514        // **The required-string protocol runs after the `>K>` and its answer
515        // replaces the bytes**, which is the string request. The trace above
516        // names the object and the parse below reads the conversion: `numeric
517        // digits .K` with a class-side `makeString` returning `12` traces `>K>
518        // "DIGITS" => "The K class"` and then answers `12` to `DIGITS()`.
519        let converted = self.required_string_value(value)?;
520        if converted != value {
521            text.clear();
522            text.extend_from_slice(&self.to_text(converted));
523        }
524        Ok(Some((value, text)))
525    }

527    /// Traces the operand before its string conversion or any setting change.
528    fn trace_numeric_operand(&mut self, keyword: &str, text: &[u8]) -> Result<(), Failure> {
529        self.trace_keyword(
530            self.activity.clause_state.current_value_indent,
531            keyword,
532            text,
533        );
534        if !self.activity.pending_traps.is_empty()
535            && let Some(failure) = self.take_trace_failure()
536        {
537            return Err(failure);
538        }
539        Ok(())
540    }
