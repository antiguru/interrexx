10    /// A `SELECT CASE`'s own `CASE` expression: the whole of what the
11    /// `SELECT` header clause does, and the value every `WHEN CASE` of that
12    /// `SELECT` is compared against.
13    pub(crate) fn select_case(
14        &mut self,
15        code: &Code<'_>,
16        case_expr: ExprRef<'_>,
17    ) -> Result<ObjRef, Failure> {
18        // The clause unit set this to the `SELECT`'s own printed indent on the
19        // way in.
20        let value = self.eval(code, case_expr)?;
21        self.select_case_value(value)
22    }

24    /// What a `SELECT CASE` owes once its expression has answered `value`:
25    /// the value rooted for the construct, its `>K>` line, and the block
26    /// opened. [`crate::ir::Op::SelectCase`] runs this behind the
27    /// expression's native ops.
28    pub(crate) fn select_case_value(&mut self, value: ObjRef) -> Result<ObjRef, Failure> {
29        // The clause unit set this to the `SELECT`'s own printed indent on the
30        // way in.
31        let indent = self.activity.clause_state.current_value_indent;
32        self.roots.activity_mut().push_temp(value);
33        let text = self.to_text(value).to_vec();
34        // The `>K>` line is at the `SELECT`'s own level -- measured, `>K>   "CASE" => "2"` sits
35        // at the same indent as `select case ...` itself, not the `WHEN`-scan
36        // level a `WhenCase`'s own comparison lines are indented to.
37        self.trace_keyword(indent, "CASE", &text);
38        // **The block opens after this**, so the header clause's own boundary
39        // runs one level in -- the block is opened next once the scrutinee has
40        // been evaluated and its `>K>` traced, and a `CALL ON` handler
41        // delivered at that boundary is based on whatever the counter reads
42        // then. The handler `h:` of a `select case raiser()` at top level
43        // echoes at `12 *-* h:`, where the `SELECT` clause itself echoes
44        // unindented.
45        self.settle_block_indent(true, indent);
46        Ok(value)
47    }

49    /// One listed `WHEN`/`WHEN CASE`'s own condition, and whether it holds.
50    pub(crate) fn scan_when(
51        &mut self,
52        code: &Code<'_>,
53        when_instruction: &Instruction,
54        case: Option<ObjRef>,
55    ) -> Result<bool, Failure> {
56        // The clause unit set this to this `WHEN`'s own printed indent on the
57        // way in, which is what its condition's `>>>` lines trace at.
58        let indent = self.activity.clause_state.current_value_indent;
59        match &when_instruction.kind {
60            InstructionKind::When { condition, .. } => self.eval_condition(
61                code,
62                code.body.expr(*condition),
63                ConditionTrace::Result(indent),
64                raised_when_not_logical,
65            ),
66            InstructionKind::WhenCase { values, .. } => match case {
67                Some(case) => self.test_case_when(code, code.body.items(values), case, indent),
68                // A listed `WhenCase` with no `case` expression: a plain
69                // `SELECT` with no `CASE` at all, which the parser should
70                // never produce for a `WhenCase` node (only `SELECT CASE`
71                // ever builds one, `ast.rs`'s own doc comment) -- F-EX3, an
72                // unproven parser invariant, and formerly an `.expect()`
73                // here that crashed on it. Evaluates `values` for side effects and
74                // never matches, the identical fallback.
75                None => {
76                    for value in values {
77                        let v = self.eval(code, code.body.expr(*value))?;
78                        self.roots.activity_mut().push_temp(v);
79                    }
80                    Ok(false)
81                }
82            },
83            other => panic!("a SELECT's whens holds only When/WhenCase, not {other:?}"),
84        }
85    }

87    /// Leaving a `SELECT`'s `OTHERWISE` branch: the escape elevation is
88    /// restored now that the whole dispatch -- marker and body alike -- is
89    /// finished reading it, and then `leave_select` decides where control
90    /// goes.
91    pub(crate) fn leave_otherwise(
92        &mut self,
93        code: &Code<'_>,
94        index: usize,
95        label: Option<SymbolId>,
96        otherwise_end: usize,
97        end: Option<usize>,
98        flow: Flow,
99    ) -> Result<Flow, Failure> {
100        debug_assert_eq!(
101            otherwise_end,
102            otherwise_range(code.body.instructions.len(), end),
103            "an OTHERWISE branch was run over a range that is not its own"
104        );
105        let resume = otherwise_resume(code.body.instructions.len(), end);
106        self.leave_select(code, index, label, resume, flow)
107    }
108
109    /// Turns the `Flow` a `SELECT`'s own matched `WHEN` or `OTHERWISE` body
110    /// produced into this `SELECT`'s own answer.
111    pub(crate) fn leave_select(
112        &mut self,
113        code: &Code<'_>,
114        index: usize,
115        label: Option<SymbolId>,
116        resume: SelectResume,
117        flow: Flow,
118    ) -> Result<Flow, Failure> {
119        match flow {
120            Flow::Next => Ok(Flow::Goto(resume.done)),
121            Flow::Leave(Some(name), _) if label == Some(name) => Ok(Flow::Goto(resume.left)),
122            Flow::Iterate(Some(name), origin) if label == Some(name) => {
123                self.record_leave_failure(&origin);
124                Err(raised_iterate_wrong_kind(code.symbols.name(name).as_bytes()).into())
125            }
126            // Not consumed: this SELECT is being "popped" by the search,
127            // so its own indent becomes the new residual before the flow
128            // continues outward.
129            Flow::Leave(name, origin) => Ok(Flow::Leave(
130                name,
131                self.pop_search_frame(code, index, origin),
132            )),
133            Flow::Iterate(name, origin) => Ok(Flow::Iterate(
134                name,
135                self.pop_search_frame(code, index, origin),
136            )),
137            other => Ok(other),
138        }
139    }
140
141    /// Resets `origin.indent` to `index`'s own `static_indent`, for a
142    /// `SELECT`/`DO`/`LOOP` that owns a search frame and is being forwarded
143    /// past (not matched) -- the update `LeaveOrigin`'s own doc comment
144    /// describes as "restoring the indent to the value saved when that
145    /// frame was pushed." Shared by `leave_select` (always calls it, since
146    /// a `SELECT` always owns a frame) and `do_body_outcome` (calls it only
147    /// when the `Do`/`Loop` in question owns one, i.e. skips an unlabelled
148    /// `Simple` block).
149    pub(super) fn pop_search_frame(
150        &self,
151        code: &Code<'_>,
152        index: usize,
153        mut origin: Box<LeaveOrigin>,
154    ) -> Box<LeaveOrigin> {
155        origin.indent =
156            static_indent(&code.body.instructions, index) + self.activity.activation_indent;
157        // `site` and `clause_line` are left alone: this resets the *indent*
158        // the search reports at, and the clause line stays the
159        // `LEAVE`/`ITERATE`'s own however many frames it is forwarded past --
160        // measured, `iterate lab` inside an inner loop attributes the outer
161        // loop's re-test to the `ITERATE`'s line, not to anything about the
162        // frames in between.
163        origin
164    }
165}

167/// What [`Interp::run_bounded`]'s absorption rule says about one clause's
168/// `Flow`, in a range bounded by `[start, end]`.
169pub(crate) enum Absorbed {
170    /// Nothing to redirect: continue after the clause that produced it.
171    Advance,
172    /// An in-range `Flow::Goto`: continue at this instruction.
173    Resume(usize),
174    /// Not this range's: hand it back to the caller unchanged.
175    Escaped(Flow),
176}
177
178/// [`Absorbed`] for `flow` in the range `[start, end]`.
179pub(crate) fn absorb(flow: Flow, start: usize, end: usize) -> Absorbed {
180    match flow {
181        Flow::Next => Absorbed::Advance,
182        Flow::Goto(target) if target >= start && target <= end => Absorbed::Resume(target),
183        other => Absorbed::Escaped(other),
184    }
185}

187/// Where an `IF` sends control on each of its two paths.
188pub(crate) struct IfTargets {
189    /// Where control goes when the condition is false: the `ELSE` when there
190    /// is one, otherwise the instruction after the `THEN` branch.
191    pub(crate) false_target: usize,
192    /// Where control resumes once the *true* branch has finished, which is
193    /// past the `ELSE` branch when there is one and identical to
194    /// `false_target` when there is not.
195    pub(crate) resume: usize,
196}
197
198/// [`IfTargets`] for an `If` whose parsed `false_target` is `raw`, against the
199/// body it belongs to.
200pub(crate) fn if_targets(instructions: &[Instruction], raw: Option<usize>) -> IfTargets {
201    // `None` is the end of this body (`InstructionKind::If`'s own doc), which
202    // is one past the last instruction and exactly what an empty range there
203    // needs.
204    let false_target = raw.unwrap_or(instructions.len());
205    IfTargets {
206        false_target,
207        resume: skip_else(instructions, false_target),
208    }
209}

211/// Where a listed `WHEN` sends control once its own condition holds.
212pub(crate) struct WhenTargets {
213    /// One past the last instruction of this `WHEN`'s own branch: the next
214    /// listed `WHEN`, the `OTHERWISE`, or the enclosing `SELECT`'s `END`.
215    pub(crate) body_end: usize,
216    /// Where control resumes once that branch has finished, which is past the
217    /// whole `SELECT`, because one true `WHEN` ends it.
218    pub(crate) resume: usize,
219}
220
221/// [`SelectResume`] for a matched listed `WHEN`: one answer twice, because one
222/// true `WHEN` ends the whole `SELECT` whether its branch finished or was left
223/// by name.
224pub(crate) fn when_resume(targets: &WhenTargets) -> SelectResume {
225    SelectResume {
226        done: targets.resume,
227        left: targets.resume,
228    }
229}
230
231/// [`SelectResume`] for the `OTHERWISE` branch: falling off its end runs the
232/// `END`, and a `LEAVE` naming this `SELECT` resumes past it.
233pub(crate) fn otherwise_resume(len: usize, end: Option<usize>) -> SelectResume {
234    SelectResume {
235        done: otherwise_range(len, end),
236        left: select_exit(len, end),
237    }
238}
239
240/// [`WhenTargets`] for a listed `When`/`WhenCase` node, against the body it
241/// belongs to.
242pub(crate) fn when_targets(kind: &InstructionKind, len: usize) -> WhenTargets {
243    match kind {
244        InstructionKind::When {
245            false_target, exit, ..
246        }
247        | InstructionKind::WhenCase {
248            false_target, exit, ..
249        } => WhenTargets {
250            body_end: false_target.unwrap_or(len),
251            resume: exit.unwrap_or(len),
252        },
253        other => panic!("a SELECT's whens holds only When/WhenCase, not {other:?}"),
254    }
255}

257/// What a `SELECT` node tells whoever is running one of its branches.
258pub(crate) struct SelectParts {
259    /// `SELECT LABEL name`'s own label, which a `LEAVE`/`ITERATE` may name.
260    pub(crate) label: Option<SymbolId>,
261    /// This `SELECT`'s own `OTHERWISE` marker, if it has one.
262    pub(crate) otherwise: Option<usize>,
263    /// The `END` that closes it.
264    pub(crate) end: Option<usize>,
265}
266
267/// [`SelectParts`] for a `Select` node, and `None` for anything else.
268pub(crate) fn select_parts(kind: &InstructionKind) -> Option<SelectParts> {
269    match kind {
270        InstructionKind::Select {
271            label,
272            otherwise,
273            end,
274            ..
275        } => Some(SelectParts {
276            label: *label,
277            otherwise: *otherwise,
278            end: *end,
279        }),
280        _ => None,
281    }
282}

284/// One past the last instruction of a `SELECT`'s `OTHERWISE` branch, which is
285/// also where control resumes once that branch has finished: its own `END`,
286/// where the `EndStyle::Otherwise` arm does nothing.
287pub(crate) fn otherwise_range(len: usize, end: Option<usize>) -> usize {
288    end.unwrap_or(len)
289}
290
291/// Where a `LEAVE` naming a `SELECT` resumes: past the `END` that closes it.
292pub(crate) fn select_exit(len: usize, end: Option<usize>) -> usize {
293    match end {
294        Some(end) => (end + 1).min(len),
295        None => len,
296    }
297}

299/// Where a `SELECT` sends control when one of its branches is over, which is
300/// **two answers and not one**.
301#[derive(Clone, Copy)]
302pub(crate) struct SelectResume {
303    /// A branch that ran off its own end. `OTHERWISE`'s falls through onto the
304    /// `END`, which executes and does nothing; a matched `WHEN`'s resumes past
305    /// the `END`, because one true `WHEN` ends the whole `SELECT`.
306    pub(crate) done: usize,
307    /// A `LEAVE` naming this `SELECT`, which resumes past the `END` from
308    /// **either** branch. Measured under `trace r`: `select label s` /
309    /// `when 1 = 0 then nop` / `otherwise leave s` / `end` echoes no `end`
310    /// clause at all, where the same `OTHERWISE` falling through echoes one.
311    pub(crate) left: usize,
312}

314/// What a `SELECT` does with a `Flow` that escaped the branch it was running.
315pub(crate) enum SelectEscape {
316    /// The flow lands exactly on this `SELECT`'s own `OTHERWISE` marker, so
317    /// that branch runs **with this `SELECT`'s search frame still standing**.
318    Otherwise(usize),
319    /// Anything else, `leave_select`'s to resolve.
320    Forward(Flow),
321}
322
323/// [`SelectEscape`] for `flow` against a `SELECT` whose `OTHERWISE` is at
324/// `otherwise`.
325pub(crate) fn select_escape(otherwise: Option<usize>, flow: Flow) -> SelectEscape {
326    match flow {
327        Flow::Goto(target) if otherwise == Some(target) => SelectEscape::Otherwise(target),
328        other => SelectEscape::Forward(other),
329    }
330}

332/// Where the *true* branch resumes, given where the false one goes.
333pub(super) fn skip_else(instructions: &[Instruction], target: usize) -> usize {
334    match instructions.get(target).map(|i| &i.kind) {
335        Some(InstructionKind::Else { then_exit }) => then_exit.unwrap_or(instructions.len()),
336        _ => target,
337    }
