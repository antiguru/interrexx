14    /// `SIGNAL ON`/`OFF` and `CALL ON`/`OFF`, which are one instruction with
15    /// one flag between them.
16    pub(crate) fn exec_condition_trap(
17        &mut self,
18        trap: &ConditionTrap,
19        call: bool,
20    ) -> Result<Flow, Failure> {
21        let delayed = self
22            .activation()
23            .traps
24            .get(&trap.condition)
25            .is_some_and(|entry| entry.delayed);
26        match &trap.label {
27            Some(label) => {
28                // An arming route of the required-string protocol: a
29                // NOSTRING trap is what turns an object with no string value
30                // from a rendering into a raise. `ANY` counts, measured --
31                // `signal on any` over `say .environment` runs the handler
32                // with `CONDITION('C')` `NOSTRING`. See
33                // `Interp::reqstr_armed` for why this only ever sets.
34                if matches!(&trap.condition[..], b"NOSTRING" | b"ANY") {
35                    self.reqstr_armed = true;
36                }
37                // The same for LOSTDIGITS, which only a `SIGNAL ON` trap
38                // takes: measured, `CALL ON ANY` over `1 / 9999999999` does
39                // not run its handler.
40                if !call && matches!(&trap.condition[..], b"LOSTDIGITS" | b"ANY") {
41                    self.lostdigits_armed = true;
42                }
43                let entry = Trap {
44                    call,
45                    label: std::rc::Rc::from(label.as_ref()),
46                    delayed: false,
47                };
48                self.activation_mut()
49                    .traps
50                    .insert(trap.condition.clone(), entry);
51                // Arming a trap also turns
52                // off whatever `::OPTIONS ... SYNTAX` asked for the same
53                // condition, so the trap takes it rather than a SYNTAX error
54                // preempting the trap. `novalue_trapped` is false because the
55                // guard is disarming's alone.
56                self.activation_mut()
57                    .condition_syntax
58                    .disable_for(&trap.condition, !call, false);
59            }
60            None => {
61                self.activation_mut().traps.remove(&trap.condition);
62                // Disarming has a second half too, with the one guard
63                // that arming's does not: a `NOVALUE` or
64                // `ANY` trap still in the table keeps `NOVALUE`'s escalation
65                // where it was.
66                let traps = &self.activation().traps;
67                let novalue_trapped = traps.get(b"NOVALUE".as_slice()).is_some()
68                    || traps.get(b"ANY".as_slice()).is_some();
69                self.activation_mut().condition_syntax.disable_for(
70                    &trap.condition,
71                    !call,
72                    novalue_trapped,
73                );
74            }
75        }
76        if delayed {
77            return self.release_delayed(&trap.condition, trap.label.is_some(), call);
78        }
79        Ok(Flow::Next)
80    }

82    /// What rexxref (`condtra.xml`, CALL ON) does with the conditions held
83    /// for the delayed trap `key` once a trap instruction for it has been
84    /// processed: after `CALL ON` each handler is called, after `SIGNAL ON`
85    /// the first is signalled with the object it was raised with, and after
86    /// `CALL OFF` or `SIGNAL OFF` each takes its default action, which only
87    /// `HALT`'s reports.
88    #[cold]
89    #[inline(never)]
90    fn release_delayed(&mut self, key: &[u8], on: bool, call: bool) -> Result<Flow, Failure> {
91        let Some(delayed) = self.delayed_trap_mut(key) else {
92            return Ok(Flow::Next);
93        };
94        let held = std::mem::take(&mut delayed.held);
95        match (on, call) {
96            (true, true) => {
97                let here = self.activation().id;
98                for mut pending in held {
99                    pending.activation = here;
100                    pending.fragment_depth = self.activity.fragment_depth;
101                    pending.queued_during_delivery = false;
102                    self.activity.pending_traps.push_back(pending);
103                }
104                Ok(Flow::Next)
105            }
106            (true, false) => {
107                let Some(pending) = held.into_iter().next() else {
108                    return Ok(Flow::Next);
109                };
110                let mut raised = Raised::condition(condition_name(&pending.condition));
111                raised.rc = pending.rc;
112                raised.description = pending.description;
113                match pending.object {
114                    Some(object) => {
115                        self.activity.reraised_object = Some(object);
116                        raised.delivery.held = true;
117                    }
118                    None => {
119                        raised.position =
120                            u32::try_from(self.activity.clause_state.line()).unwrap_or(0);
121                    }
122                }
123                Err(raised.into())
124            }
125            (false, _) if held.iter().any(|pending| *pending.condition == *b"HALT") => {
126                Err(Raised::halt().into())
127            }
128            (false, _) => Ok(Flow::Next),
129        }
130    }

132    /// Moves into `held` each condition queued for the running activation's
133    /// trap `key` since `queued_before`, while that trap was delayed: a `RAISE
134    /// ... RETURN` in the handler offers its condition to this activation. A
135    /// condition [`ignored_while_delayed`] is dropped instead.
136    #[cold]
137    #[inline(never)]
138    fn take_queued_for_delayed(
139        &mut self,
140        queued_before: usize,
141        key: &[u8],
142        held: &mut Vec<PendingTrap>,
143    ) {
144        let here = self.activation().id;
145        let depth = self.activity.fragment_depth;
146        let mut row = queued_before;
147        while row < self.activity.pending_traps.len() {
148            let pending = &self.activity.pending_traps[row];
149            let matches = !pending.request
150                && pending.activation == here
151                && pending.fragment_depth == depth
152                && (*pending.condition == *key
153                    || (key == b"ANY"
154                        && self.activation().traps.get(&pending.condition).is_none()));
155            if matches && let Some(pending) = self.activity.pending_traps.remove(row) {
156                if !ignored_while_delayed(&pending.condition) {
157                    held.push(pending);
158                }
159            } else {
160                row += 1;
161            }
162        }
163    }

165    /// Holds `pending`, whose trap is delayed, for the innermost delayed trap
166    /// it matches; a condition [`ignored_while_delayed`] is dropped instead.
167    #[cold]
168    #[inline(never)]
169    fn hold_delayed(&mut self, pending: PendingTrap) {
170        if ignored_while_delayed(&pending.condition) {
171            return;
172        }
173        let key: &[u8] = match self
174            .trap_frame()
175            .is_some_and(|frame| frame.traps.get(&pending.condition).is_some())
176        {
177            true => &pending.condition,
178            false => b"ANY",
179        };
180        let key: Box<[u8]> = key.into();
181        if let Some(delayed) = self.delayed_trap_mut(&key) {
182            delayed.held.push(pending);
183        }
184    }

186    /// The innermost activation's trap `key` delayed while its handler runs.
187    fn delayed_trap_mut(&mut self, key: &[u8]) -> Option<&mut crate::DelayedTrap> {
188        let activity = &mut self.activity;
189        activity
190            .running
191            .as_deref_mut()
192            .into_iter()
193            .chain(activity.suspended.iter_mut().rev().map(Box::as_mut))
194            .filter_map(|activation| activation.cold.as_deref_mut()?.delayed.as_mut())
195            .find(|delayed| *delayed.key == *key)
196    }

198    /// The trap the running activation has enabled for `condition`, if any.
199    pub(crate) fn trap_for(&self, condition: &[u8]) -> Option<Trap> {
200        let frame = self.trap_frame()?;
201        if ignored_in_debug_pause(frame, condition) {
202            return None;
203        }
204        let traps = &frame.traps;
205        // A delayed trap is answered too: its condition queues as for any
206        // `CALL ON` trap, and its delivery holds it (`Interp::hold_delayed`).
207        // A `CALL ON ANY` trap does not take what only `SIGNAL` can trap,
208        // whoever raises it (`call_on_any_takes`).
209        traps
210            .get(condition)
211            .or_else(|| {
212                traps
213                    .get(b"ANY".as_slice())
214                    .filter(|any| !any.call || call_on_any_takes(condition))
215            })
216            .cloned()
217    }

219    /// Raises `NOTREADY` for the stream named `name` -- the name **as the
220    /// program wrote it**, which is what `CONDITION('D')` answers.
221    ///
222    /// Three outcomes, all measured. A `SIGNAL ON` trap takes it
223    /// as a failure, so the raising clause is abandoned. A `CALL ON` trap
224    /// queues it against the running activation, and `leave_clause` runs the
225    /// handler once the clause finishes -- measured, the clause prints its
226    /// own empty value first and the next clause runs after the handler.
227    /// Untrapped it is silent unless `::OPTIONS NOTREADY SYNTAX` is in force,
228    /// which makes it 98.974 at rc 158.
229    ///
230    /// **The caller answers normally after `Ok`**: a raise never changes what
231    /// the failing call returns -- the residual, `1`, or the null string.
232    /// The condition object carries `stream` as `ADDITIONAL` and `result`, the
233    /// value the caller answers, as `RESULT`.
234    pub(crate) fn raise_notready(
235        &mut self,
236        name: &[u8],
237        stream: ObjRef,
238        result: ObjRef,
239    ) -> Result<(), Failure> {
240        // A native `LINEIN` the input reader sent runs in the reader's own
241        // activation.
242        let running = self.running_activation().map(|activation| activation.id);
243        if self.input_dispatcher_takes(running) {
244            return Ok(());
245        }
246        // One `Raised` for both branches, so the queued form can build its
247        // condition object here rather than at delivery.
248        let raised = Raised {
249            description: Some(name.to_vec()),
250            ..Raised::condition(Cow::Borrowed("NOTREADY"))
251        };
252        let trapping = self
253            .trap_frame()
254            .map_or(self.activation().id, |activation| activation.id);
255        match self.trap_for(b"NOTREADY") {
256            // Raised again while its handler runs, the trap being delayed. One
257            // raised while an earlier one is only queued is held as well, and
258            // calls the handler again (`r[cond.traps.delay]`, upstream bug
259            // #2084).
260            Some(trap) if trap.call && trap.delayed => Ok(()),
261            Some(trap) if trap.call => {
262                self.activity.pending_additional = Some(stream);
263                self.activity.pending_result = Some(result);
264                let object = self.build_condition_object(&raised, Some(true))?;
265                self.activity.native_frames_owed.push(object);
266                self.activity.pending_traps.push_back(PendingTrap {
267                    condition: b"NOTREADY".as_slice().into(),
268                    rc: None,
269                    description: Some(name.to_vec()),
270                    object: Some(object),
271                    // The activation whose table matched, not its caller: a
272                    // native method raises inside the clause that sent to it,
273                    // and that clause is still running. Under a non-continuing
274                    // `FORWARD` that is the frame beneath the forwarding
275                    // phantoms. The routine-return path queues against the
276                    // caller because its own activation is being popped;
277                    // nothing is popped here.
278                    activation: trapping,
279                    queued_during_delivery: false,
280                    request: false,
281                    fragment_depth: self.activity.fragment_depth,
282                    trace_failure: None,
283                });
284                Ok(())
285            }
286            Some(_) => {
287                self.activity.pending_additional = Some(stream);
288                self.activity.pending_result = Some(result);
289                Err(Raised {
290                    description: Some(name.to_vec()),
291                    ..Raised::condition(Cow::Borrowed("NOTREADY"))
292                }
293                .into())
294            }
295            None => {
296                if self.condition_raises_syntax(b"NOTREADY") {
297                    return Err(Raised::notready_syntax(name).into());
298                }
299                Ok(())
300            }
301        }
302    }

315    /// [`Interp::novalue_check`]'s uninitialised half: what the read's
316    /// `.local~novalue` hook came to where the read sent it, and otherwise
317    /// whether the setting in force turns this read into a raised `NOVALUE`,
318    /// into `::OPTIONS NOVALUE SYNTAX`'s 98.986, or into neither -- the
319    /// derived name `read_at` already produced.
320    #[cold]
321    #[inline(never)]
322    fn novalue_raised(&self, read: ObjRef) -> Result<(), Failure> {
323        if let Some(hooked) = self.novalue_hooked.take() {
324            return hooked;
325        }
326        if self.trap_for(b"NOVALUE").is_none_or(|trap| trap.call) {
327            if self.condition_raises_syntax(b"NOVALUE") {
328                return Err(Raised::unassigned_variable(&self.derived_name_text(read)).into());
329            }
330            return Ok(());
331        }
332        // The variable's own derived name, which `CONDITION('D')` and the
333        // condition object's `DESCRIPTION` both report: `ZZUNDEF` for `say
334        // zzundef` under `SIGNAL ON NOVALUE`. `derived_name_text` is the same
335        // value the `::OPTIONS NOVALUE SYNTAX` branch above already builds;
336        // leaving it off here is what made both answer empty.
337        let mut raised = Raised::condition(Cow::Borrowed("NOVALUE"));
338        raised.description = Some(self.derived_name_text(read));
339        Err(raised.into())
340    }

342    /// The value of an uninitialised read whose derived name is `derived`, in
343    /// this order: where `::OPTIONS NOVALUE SYNTAX` is about to raise 98.986,
344    /// `derived`; otherwise the `.local~novalue` hook's answer, or `derived`
345    /// where it answers none. The read is still [`Novalue::Unset`]: where the
346    /// hook answered or failed, that outcome waits in `Interp::novalue_hooked`
347    /// for the `novalue_check` after the read.
348    #[cold]
349    #[inline(never)]
350    pub(crate) fn novalue_answer(&mut self, derived: ObjRef) -> ObjRef {
351        let syntax = self.trap_for(b"NOVALUE").is_none_or(|trap| trap.call)
352            && self.condition_raises_syntax(b"NOVALUE");
353        if syntax {
354            return derived;
355        }
356        match self.novalue_hook(derived) {
357            Ok(Some(value)) => {
358                self.novalue_hooked.set(Some(Ok(())));
359                value
360            }
361            // The `NOVALUE` exit's pass, once the hook has answered none: a value it gives is
362            // the variable's from then on.
363            Ok(None) if self.exits.is_some() => match self.novalue_exit(derived) {
364                Ok(Some(value)) => {
365                    let value = value.unwrap_or(ObjRef::NIL);
366                    let name = self.derived_name_text(derived);
367                    self.assign_by_name(&name, value);
368                    self.novalue_hooked.set(Some(Ok(())));
369                    value
370                }
371                Ok(None) => derived,
372                Err(failure) => {
373                    self.novalue_hooked.set(Some(Err(failure)));
374                    derived
375                }
376            },
377            Ok(None) => derived,
378            Err(failure) => {
379                self.novalue_hooked.set(Some(Err(failure)));
380                derived
381            }
382        }
383    }

385    /// The object `.local` holds under `NOVALUE`, sent `NOVALUE` with the
386    /// uninitialised variable's derived name: an answer other than `.nil` is
387    /// the read's value, and the variable stays unset.
388    fn novalue_hook(&mut self, read: ObjRef) -> Result<Option<ObjRef>, Failure> {
389        let Some(handler) = self.local_route(b"NOVALUE")? else {
390            return Ok(None);
391        };
392        self.roots.activity_mut().push_temp(read);
393        let caller = self.caller();
394        let answer = pinned!(
395            self,
396            crate::pinning::PinKind::NovalueHook,
397            self.send_message(handler, b"NOVALUE", None, &[Some(read)], caller)
398        )?;
399        Ok(answer.filter(|value| *value != ObjRef::NIL))
400    }

402    /// The bytes of the derived name an uninitialised read answered.
403    fn derived_name_text(&self, read: ObjRef) -> Vec<u8> {
404        match read.decode() {
405            interrexx_core::Decoded::Text(inline) => inline.to_vec(),
406            interrexx_core::Decoded::Heap { .. } => {
407                match self.heap.get(read).map(|object| &object.body) {
408                    Some(interrexx_core::Body::Text { bytes, .. }) => bytes.to_vec(),
409                    other => {
410                        debug_assert!(
411                            false,
412                            "an uninitialised read answered something other than a string: {other:?}"
413                        );
414                        Vec::new()
415                    }
416                }
417            }
418            other => {
419                debug_assert!(
420                    false,
421                    "an uninitialised read answered something other than a string: {other:?}"
422                );
423                Vec::new()
424            }
425        }
426    }

460    /// at DIGITS minus FUZZ (rexxref, "Numeric Comparisons"), so under FUZZ
461    /// each operand is checked again at that precision, whatever the signs.

472            return Ok(());
473        }

475        self.lostdigits_raise(right, right_value)?;

477            return Ok(());
478        }
479        self.lostdigits_fuzz_raise(
480            left,
481            |interp| interp.unsent_string_value_text(left_value),

483            |interp| interp.unsent_string_value_text(right_value),
484            fuzz,

554    /// [`Interp::lostdigits_raise_with`] against `digits` rather than the
555    /// setting in force.
556    fn lostdigits_raise_at(
557        &mut self,
558        operand: &Number,
559        digits: u64,
560        description: impl FnOnce(&mut Self) -> Vec<u8>,
561    ) -> Result<(), Failure> {
562        let digits = usize::try_from(digits).unwrap_or(usize::MAX);
563        if operand.digit_count() <= digits {
564            return Ok(());
565        }
566        if self.trap_for(b"LOSTDIGITS").is_some_and(|trap| !trap.call) {
567            let text = description(self);
568            return Err(Raised {
569                description: Some(text),
570                ..Raised::condition(Cow::Borrowed("LOSTDIGITS"))
571            }
572            .into());
573        }
574        if self.condition_raises_syntax(b"LOSTDIGITS") {
575            let text = description(self);
576            return Err(Raised::lostdigits(&text).into());
577        }
578        Ok(())
579    }

581    /// Offers a failure escaping the running activation to that activation's
582    /// trap table, and either transfers control or hands the failure back to
583    /// keep unwinding.
584    pub(crate) fn offer_to_trap(&mut self, failure: Failure) -> Result<Flow, Failure> {
585        let Failure::Raised(raised) = &failure else {
586            return Err(failure);
587        };
588        if raised.condition == "SYNTAX"
589            && let Some(replaced) = self.notify_slot_failure(raised)
590        {
591            return self.offer_to_trap(replaced);
592        }
593        match raised.delivery.search {
594            Search::Here => {}
595            // One level up, and this is that level's turn to decline. The
596            // rewrite is what makes the *next* loop out offer it: without
597            // it, `Caller` would skip every activation rather than one.
598            Search::Caller => {
599                let Failure::Raised(mut raised) = failure else {
600                    unreachable!("matched Failure::Raised immediately above")
601                };
602                raised.delivery.search = Search::Here;
603                return Err(Failure::Raised(raised));
604            }
605            // an internal call hands the raise to its parent, and the first
606            // activation that is not one raises it, from there outward.
607            Search::Top if self.activation().entry == Entry::InternalCall => return Err(failure),
608            Search::Top => {
609                let Failure::Raised(mut raised) = failure else {
610                    unreachable!("matched Failure::Raised immediately above")
611                };
612                raised.delivery.search = Search::Here;
613                return self.offer_to_trap(Failure::Raised(raised));
614            }
615            Search::AboveTop if self.activation().entry == Entry::InternalCall => {
616                return Err(failure);
617            }
618            Search::AboveTop => {
619                let Failure::Raised(mut raised) = failure else {
620                    unreachable!("matched Failure::Raised immediately above")
621                };
622                raised.delivery.search = Search::Here;
623                return Err(Failure::Raised(raised));
624            }
625            Search::Nobody => return Err(failure),
626        }
627        // A phantom does not trap: whether the activation is forwarded is
628        // checked before any trap table is looked at, and this crate reaches
629        // the frame it drills to by declining here and letting the failure
630        // leave this activation. [`Interp::trap_for`] does the drilling for the
631        // callers that ask whether a condition would be trapped at all, which
632        // is the same question the trap test answers.
633        if self
634            .running_activation()
635            .is_some_and(|a| a.flags.forwarded())
636        {
637            return Err(failure);
638        }
639        let Some(trap) = self.trap_for(raised.condition.as_bytes()) else {
640            return Err(failure);
641        };
642        if trap.call {
643            return Err(failure);
644        }
645        // Resolved **before** anything is cleared or removed, because a label
646        // that does not exist is not a trap that fired: `signal on syntax name
647        // nosuchlabel` with `say 1/0` on line 3 reports `Error 16.1 Label
648        // "NOSUCHLABEL" not found` against line 3 -- the raising clause's own
649        // site, which is still the one `Op::Clause`'s region recorded a moment
650        // ago and which the clearing below would have thrown away. `SIGNAL ON`
651        // turns the trap off before it raises 16.1, so the 16.1 is offered to
652        // this same table: `signal on novalue` with no `NOVALUE:` label is
653        // trapped by `signal on syntax` at the raising clause (defect 1877).
654        let target = match self.resolve_signal_target(&trap.label) {
655            Ok(target) => target,
656            Err(missing) => {
657                // The 16.1 is raised here, at the raising clause of the Rexx
658                // level holding the trap: nothing of the condition that found
659                // the trap, nor of the levels it left, belongs to it.
660                let site = self.activity.failure_site.take();
661                self.clear_failure_levels();
662                self.activity.failure_site = site;
663                self.activity.pending_additional = None;
664                self.activity.pending_result = None;
665                self.activity.pending_description = None;
666                self.activity.pending_rc = None;
667                let traps = &mut self.activation_mut().traps;
668                match traps.get(raised.condition.as_bytes()) {
669                    Some(_) => traps.remove(raised.condition.as_bytes()),
670                    None => traps.remove(b"ANY".as_slice()),
671                }
672                return self.offer_to_trap(missing);
673            }
674        };
675        let Failure::Raised(mut raised) = failure else {
676            unreachable!("matched Failure::Raised immediately above")
677        };
678        // Removed when it fires (`Activation::traps`' own doc comment has
679        // the two probes). Only the entry that matched goes: the condition's
680        // own entry when there is one (`trap_for` prefers it), else `ANY`.
681        // Each trap keeps its own state, so a specific trap that fires leaves
682        // `ANY` on to answer a later condition.
683        let traps = &mut self.activation_mut().traps;
684        match traps.get(raised.condition.as_bytes()) {
685            Some(_) => traps.remove(raised.condition.as_bytes()),
686            None => traps.remove(b"ANY".as_slice()),
687        };
688        // **Inherited item I11, and the reason it is this task's.** Both
689        // halves of the echo stack are dropped: a trapped condition prints
690        // no report at all, so the sites it accumulated must not survive to
691        // be printed against a *later*, untrapped one. Measured -- `say 1/0`
692        // trapped on line 3 and `say 2/0` untrapped on line 8 inside the
693        // handler reports line 8, alone, and a version that kept the first
694        // site reports line 3.
695        let site = self.activity.failure_site.take();
696        let sites = std::mem::take(&mut self.activity.failure_sites);
697        let frames = std::mem::take(&mut self.activity.failure_frames);
698        for &frame in &frames {
699            self.roots.activity_mut().push_temp(frame);
700        }
701        self.activity.failure_frame = None;
702        let origin = self.activity.failure_origin.take();
703        self.hold_origin_package(origin);
704        let propagated = std::mem::take(&mut self.activity.failure_propagated);
705        let reraised = std::mem::take(&mut self.activity.failure_reraised);
706        self.activity.propagate_absorb = 0;
707        // A propagated condition trapped here is reported by nothing, so a
708        // later untrapped one must not show its `TRACEBACK`.
709        self.activity.traceback_object = None;
710        self.set_sigl(self.activity.clause_state.line());
711        if let Some(rc) = &raised.rc {
712            // A `RAISE` that named an object for `RC` holds it for the
713            // condition object, and the variable has that object.
714            let value = match self.activity.pending_rc {
715                Some(object) => object,
716                None => self.text(rc),
717            };
718            self.assign_by_name(b"RC", value);
719        }
720        // What `CONDITION()` reports for the rest of this activation, kept
721        // apart from `active_condition` below, which carries the echo sites
722        // `RAISE PROPAGATE` puts back.
723        // Built here, on the raising clause, because everything in it is a
724        // raise-time fact: `POSITION` is the innermost level's line and
725        // `STACKFRAMES` the levels left above the stack as it stands now.
726        let unwound = crate::condition::Unwound {
727            sites: &sites,
728            frames,
729            origin,
730            propagated,
731            reraised,
732        };
733        let object = match self.activity.reraised_object.take() {
734            Some(object) if raised.delivery.held => self.trap_held_condition_object(object)?,
735            // Any other condition is re-raised with the
736            // object it was built with, where a SYNTAX condition's position
737            // moves to this level.
738            Some(object) if raised.condition != "SYNTAX" => {
739                self.trap_propagated_condition_object(object)?
740            }
741            Some(object) => self.trap_reraised_condition_object(object, unwound.origin)?,
742            None => self.build_trapped_condition_object(&raised, &unwound)?,
743        };
744        raised.delivery.held = false;
745        self.attach_condition(object);
746        self.activation_mut()
747            .replace_condition(Some(TrappedCondition {
748                name: raised.condition.as_bytes().into(),
749                call: false,
750                description: raised.description.clone(),
751                object: Some(object),
752            }));
753        // What a later `RAISE PROPAGATE` re-raises. See `exec_raise_
754        // propagate` for what is and is not measured about it.
755        let replied = self.activation().reply != ReplyState::None;
756        self.activation_mut().cold_mut().active_condition = Some(Rc::new(ActiveCondition {
757            raised: *raised,
758            site,
759            sites,
760            replied,
761        }));
762        // **The failed clause's own boundary, and the one place it can
763        // happen**. `Op::Clause`'s region ends a *completing* clause; a clause
764        // that raised has not completed, and at that moment nothing yet knows
765        // whether it will be trapped here or unwind the activation. This is the
766        // point where that is decided in favour of "trapped here, execution
767        // continues", so it is the point where a `CALL ON` handler queued by
768        // the same clause is owed its run. A halt request waits for the label's
769        // clause: an untrapped one is reported there, and `SIGL` names it for a
770        // trapped one.
771        // The trapped condition takes the place of a heap-limit raise the
772        // clause owes (r[env.invocation.heap-limit.raise]), which would
773        // otherwise fall on the handler's first clause, past this trap. The
774        // collection that found the heap past its limit may have run while
775        // the condition object was being built.
776        if self.heap_over_limit && self.running_user_program() {
777            self.heap_limit_raised();
778        }
779        if let Some(exit) = self.deliver_pending(false)? {
780            return Ok(Flow::Exit(exit.value()));
781        }
782        Ok(Flow::Signal(target))
783    }

786    /// conditions were queued, and stops early only if one of them ends the

792    /// [`Interp::deliver_pending_traps`], leaving a halt request queued where
793    /// `requests` is false.
794    fn deliver_pending(&mut self, requests: bool) -> Result<Option<HandlerExit>, Failure> {
795        // Snapshotted rather than re-read, so that anything a handler queues
796        // lands beyond the prefix this boundary is answering for.
797        let mut owed = self.activity.pending_traps.len();
798        while owed > 0 {
799            // Only the activation whose trap table matched delivers, and only
800            // once it is running again -- `PendingTrap::activation`'s own doc
801            // comment has the three transcripts this identity check answers,
802            // including the two a stack depth got wrong.
803            let here = self.activation().id;
804            // Both keys, and they answer different questions: the identity says
805            // which activation's trap table matched, the depth says which
806            // `INTERPRET` fragment's queue the condition is sitting in. A
807            // fragment is an activation of its own in the reference model and
808            // is not one here, so the second is what the first cannot see.
809            let depth = self.activity.fragment_depth;
810            let Some(at) = self
811                .activity
812                .pending_traps
813                .iter()
814                .take(owed)
815                .position(|pending| {
816                    pending.activation == here
817                        && pending.fragment_depth == depth
818                        && (requests || !pending.request)
819                })
820            else {
821                return Ok(None);
822            };
823            let pending = self
824                .activity
825                .pending_traps
826                .remove(at)
827                .expect("position answered an index inside the queue");
828            owed -= 1;
829            if let Some(failure) = pending.trace_failure {
830                return Err(failure);
831            }
832            if pending.request {
833                self.raise_requested_halt(
834                    pending.description,
835                    pending.object == Some(ObjRef::NIL),
836                )?;
837                continue;
838            }
839            if let Some(exit) = self.deliver_one_pending_trap(pending)? {
840                return Ok(Some(exit));
841            }
842        }
843        Ok(None)
844    }

846    /// One queued condition's handler, run at the boundary that owes it, and
847    /// run again for each condition held while it ran.
848    fn deliver_one_pending_trap(
849        &mut self,
850        pending: PendingTrap,
851    ) -> Result<Option<HandlerExit>, Failure> {
852        let (exit, held) = self.run_pending_handler(pending)?;
853        if exit.is_some() || held.is_empty() {
854            return Ok(exit);
855        }
856        self.redeliver_held(held)
857    }

859    /// rexxref (`condtra.xml`, CALL ON): on the handler's `RETURN` the trap
860    /// is no longer delayed and the handler is called again at once for
861    /// each condition `held` for it, in the order raised.
862    #[cold]
863    #[inline(never)]
864    fn redeliver_held(&mut self, held: Vec<PendingTrap>) -> Result<Option<HandlerExit>, Failure> {
865        let frame = self.roots.activity_mut().push_frame();
866        let mut queue = std::collections::VecDeque::from(held);
867        let mut delivered = Ok(None);
868        while let Some(pending) = queue.pop_front() {
869            // The conditions waiting are reachable from this queue alone.
870            self.roots.activity_mut().pop_frame(frame);
871            for object in std::iter::once(&pending)
872                .chain(&queue)
873                .filter_map(|pending| pending.object)
874            {
875                self.roots.activity_mut().push_temp(object);
876            }
877            match self.run_pending_handler(pending) {
878                Ok((None, held)) => {
879                    for pending in held.into_iter().rev() {
880                        queue.push_front(pending);
881                    }
882                }
883                ended => {
884                    delivered = ended.map(|(exit, _)| exit);
885                    break;
886                }
887            }
888        }
889        self.roots.activity_mut().pop_frame(frame);
890        delivered
891    }

893    /// [`Interp::deliver_one_pending_trap`]'s one handler run, answering with
894    /// the conditions held for its trap while it ran.
895    fn run_pending_handler(
896        &mut self,
897        pending: PendingTrap,
898    ) -> Result<(Option<HandlerExit>, Vec<PendingTrap>), Failure> {
899        let Some(trap) = self.trap_for(&pending.condition) else {
900            return Ok((None, Vec::new()));
901        };
902        if trap.delayed {
903            self.hold_delayed(pending);
904            return Ok((None, Vec::new()));
905        }
906        if !trap.call {
907            // A `SIGNAL ON` trap never gets here: `exec_raise` throws for
908            // that half instead, so the transfer happens where the raise is
909            // rather than one clause later. Declining rather than asserting
910            // keeps a future raiser that forgets the distinction from
911            // silently running a `SIGNAL` handler as a call.
912            return Ok((None, Vec::new()));
913        }
914        // No `SIGL` here: a handler that is a label sets it when the call
915        // enters it, and one that is a `::ROUTINE` or a builtin leaves it.
916        if let Some(rc) = &pending.rc {
917            // The condition object has the object the `RAISE` named, which the
918            // variable has too.
919            let entry = pending
920                .object
921                .and_then(|object| self.condition_entry_held(object, b"RC"));
922            let value = match entry {
923                Some(entry) => entry,
924                None => self.text(rc),
925            };
926            self.assign_by_name(b"RC", value);
927        }
928        // A `CALL ON` handler is running a condition too, and `RAISE
929        // PROPAGATE` inside one asks for it -- measured, `raise propagate`
930        // in a `CALL ON USER FOO` handler ends the program silently at rc 0,
931        // where the same clause with no handler running at all is 98.918.
932        // Recording nothing here would give the second answer for the first
933        // program. No sites travel with it: nothing failed, so nothing was
934        // cleared.
935        let mut raised = Raised::condition(condition_name(&pending.condition));
936        raised.rc = pending.rc.clone();
937        // **Saved and restored, not cleared**: clearing is right only when
938        // nothing was active before -- and one clause can queue a `CALL ON`
939        // condition *and* raise a `SIGNAL ON`-trapped one, so a `SIGNAL`
940        // handler can be running when a `CALL` handler is delivered inside it.
941        // So `zq = sub() + 1/0` under both traps, with the `SIGNAL` handler
942        // ending in `raise propagate`, is the original 42.3 at rc 214; clearing
943        // to `None` would give 98.918 at rc 158, and never clearing at all
944        // would give silence at rc 0. Restoring is right in all three shapes,
945        // the "nothing was active, restore `None`" one included. Set here for
946        // the handler's activation to inherit, and put back once it returns.
947        let enclosing = self
948            .activation_mut()
949            .cold_mut()
950            .active_condition
951            .replace(Rc::new(ActiveCondition {
952                raised,
953                site: None,
954                sites: Vec::new(),
955                replied: false,
956            }));
957        let key: Box<[u8]> = pending.condition.clone();
958        // The entry the trap was found under, `ANY` where the condition has
959        // none: the trap instruction's own condition is what is delayed.
960        let delayed: Box<[u8]> = match self
961            .trap_frame()
962            .is_some_and(|frame| frame.traps.get(&key).is_some())
963        {
964            true => key.clone(),
965            false => b"ANY".as_slice().into(),
966        };
967        // **Delayed, not removed** (`Trap::delayed`). The two are the same
968        // to every lookup that decides whether to trap, and different to
969        // `CONDITION('S')` alone, which reports `DELAY` here and `OFF` for a
970        // trap that is absent. Nothing else moves: an earlier version of
971        // this comment claimed the flag also protects a handler's own `CALL
972        // OFF`, and that is false -- the handler's table is a copy, so its
973        // `CALL OFF` never reached this one to be undone.
974        let recorded = match self.activation_mut().traps.get_mut(&delayed) {
975            Some(trap) => {
976                trap.delayed = true;
977                true
978            }
979            None => false,
980        };
981        if recorded {
982            self.activation_mut().cold_mut().delayed = Some(crate::DelayedTrap {
983                key: delayed.clone(),
984                held: Vec::new(),
985            });
986        }
987        // What `CONDITION()` answers inside the handler. Set on *this*
988        // activation and restored afterwards, because the handler inherits
989        // its copy at call time and the caller must be left as it was --
990        // measured, `condition()` back in the caller after a `CALL ON`
991        // handler returns is the null string.
992        let enclosing_condition = self
993            .activation_mut()
994            .replace_condition(Some(TrappedCondition {
995                name: key.clone(),
996                call: true,
997                description: pending.description.clone(),
998                // Carried from the queue rather than built here. By delivery the
999                // raising clause has finished and its routine may have returned,
1000                // so `POSITION` and `STACKFRAMES` no longer exist to be read --
1001                // measured, `POSITION` is 7 for a command inside a routine, not
1002                // the caller's own `call` line.
1003                object: pending.object,
1004            }));
1005        let queued_before = self.activity.pending_traps.len();
1006        // `CallType::Subroutine` because a `CALL ON` handler is a `CALL`, and
1007        // that reaches `PARSE SOURCE` when the trap's name resolves to a
1008        // `::ROUTINE` rather than to a label. A trapped `USER` condition whose
1009        // handler is a `::routine` running `parse source` answers `SUBROUTINE`,
1010        // where the same handler written as a label answers whatever the
1011        // trapping activation answers. A label handler has the trapping
1012        // activation's receiver, as an internal `CALL` has
1013        // (`entered_receiver`). The condition object is the handler's one
1014        // argument, whether it is a label, a builtin or an external routine.
1015        let arguments = pending.object.map(Some).into_iter().collect();
1016        let ended = pinned!(
1017            self,
1018            crate::pinning::PinKind::TrapHandler,
1019            self.resolve_and_run_call_over(&trap.label, true, arguments, CallType::Subroutine,)
1020        );
1021        // A trap queued by the handler that just ran is not one the
1022        // interrupted clause owes, and `in_clause`'s tripwire has to be able
1023        // to tell the two apart -- see the field's own doc comment.
1024        for pending in self.activity.pending_traps.iter_mut().skip(queued_before) {
1025            pending.queued_during_delivery = true;
1026        }
1027        self.activation_mut().replace_condition(enclosing_condition);
1028        // The delay is lifted again. The `if let` tests the handler for null before
1029        // enabling it; nothing a Rexx program can do removes the entry between here and
1030        // the delay above, so the arm is structural rather than a case anything reaches.
1031        if let Some(trap) = self.activation_mut().traps.get_mut(&delayed) {
1032            trap.delayed = false;
1033        }
1034        let held = match recorded {
1035            true => self
1036                .activation_mut()
1037                .cold
1038                .as_deref_mut()
1039                .and_then(|cold| cold.delayed.take())
1040                .map(|delayed| delayed.held)
1041                .unwrap_or_default(),
1042            false => Vec::new(),
1043        };
1044        match ended {
1045            // The handler returned; execution resumes at the clause after
1046            // the one that finished.
1047            Ok(Ended::Returned(_)) => {
1048                self.activation_mut().cold_mut().active_condition = enclosing;
1049                let mut held = held;
1050                if recorded {
1051                    self.take_queued_for_delayed(queued_before, &delayed, &mut held);
1052                }
1053                Ok((None, held))
1054            }
1055            // The handler failed rather than returned.
1056            Err(failure) => {
1057                self.activation_mut().cold_mut().active_condition = enclosing;
1058                Err(failure)
1059            }
1060            // `EXIT` inside the handler ends the program, exactly as it does
1061            // inside any other called routine. Nothing will read
1062            // `active_condition` again, so it is left as it is.
1063            Ok(exited @ Ended::Exited(_)) => Ok((HandlerExit::from_ended(exited), Vec::new())),
1064        }
1065    }

1099    /// Resolves a `SIGNAL`/`SIGNAL VALUE` target against the running
1100    /// *activation's* own body -- not `code.body`, which differs inside an
1101    /// `INTERPRET` fragment (whose own `labels` is always empty, a label in
1102    /// interpreted text being 47.1). Mirrors `Interp::resolve_call`'s
1103    /// identical fix for `CALL` (`run.rs:2153-2154` in
1104    /// the tree this task started from) -- found there by running the
1105    /// composition rather than reading the code, and true of `SIGNAL` for
1106    /// the same reason: measured, `interpret "signal there"` reaches an
1107    /// enclosing `there:`, and `call sub` into `sub:` containing `signal
1108    /// caller_label` reaches a label back in the caller's own text, because
1109    /// an internal `CALL` target shares its caller's exact body -- not
1110    /// because `SIGNAL` reaches across an activation boundary on its own.
1111    pub(crate) fn signal_to_label(&mut self, name: &[u8]) -> Result<Flow, Failure> {
1112        let target = self.resolve_signal_target(name)?;
1113        // Set only once the target actually resolves -- an unresolved `SIGNAL` (16.1) ends the
1114        // program regardless.
1115        self.set_sigl(self.activity.clause_state.line());
1116        Ok(Flow::Signal(target))
1117    }

1119    /// `SIGNAL VALUE expr`, past the expression: its `>K>` echo, then the
1120    /// same search and transfer a written label takes.
1121    pub(crate) fn signal_to_value(&mut self, value: ObjRef) -> Result<Flow, Failure> {
1122        let text = self.to_text(value).to_vec();
1123        // `>K>` names the object and the search reads the conversion, which is
1124        // the order of the steps: the `>K>` trace first, then the string
1125        // request. For example, `signal value .K` with a class-side
1126        // `makeString` returning `'NOSUCH'` reports 16.1 naming `NOSUCH`,
1127        // so 16.1 names the conversion where NUMERIC's own errors name
1128        // the object.
1129        self.trace_keyword(
1130            self.activity.clause_state.current_value_indent,
1131            "VALUE",
1132            &text,
1133        );
1134        let converted = self.required_string_value(value)?;
1135        if converted == value {
1136            return self.signal_to_label(&text);
1137        }
1138        let text = self.to_text(converted).to_vec();
1139        self.signal_to_label(&text)
1140    }

1142    fn resolve_signal_target(&self, name: &[u8]) -> Result<usize, Failure> {
1143        let program = Rc::clone(&self.activation().program);
1144        let selector = self.activation().body;
1145        let Some(activation_body) = body_of(&program, selector) else {
1146            return Err(Loud::missing_body().into());
1147        };
1148        match activation_body.labels.get(name) {
1149            Some(target) => Ok(*target),
1150            None => Err(Raised::label_not_found(name).into()),
1151        }
1152    }
1153}

1155/// Where [`Interp::raise_sender`] offers a raised condition.
1156struct Sender {
1157    /// The level the raise leaves from: the running one, or the first that
1158    /// is not an internal call.
1159    top: usize,
1160    /// The level whose traps are asked.
1161    depth: usize,
1162    /// Whether that level is the running one, which an `INTERPRET` fragment
1163    /// with a `RETURN` tail offers it to.
1164    in_fragment: bool,
1165    returns: bool,
1166    /// Whether `top` has replied, which an `INTERPRET` fragment's sender
1167    /// inherits: a value it returns is then 98.936 unless a
1168    /// `SIGNAL ON` trap takes the condition first.
1169    reply_issued: bool,
1170    /// The identity of the level at `depth`, unless that level is beyond a
1171    /// `top` that has replied or beyond a `CallRoutine` call's base.
1172    activation: Option<crate::activation::ActivationId>,
1173    /// Whether the level at `depth` is the activation a `CallRoutine` or
1174    /// `CallProgram` call was made from: the code it runs has no Rexx
1175    /// caller, and the native code that made the call keeps the condition.
1176    api: bool,
1177}
1178
1179impl Sender {
1180    /// How a `SIGNAL ON` trap at the sender is reached.
1181    fn search(&self) -> Search {
1182        match (self.in_fragment, self.top) {
1183            (true, _) => Search::Here,
1184            (false, 0) => Search::Caller,
1185            (false, _) => Search::AboveTop,
1186        }
1187    }
1188}

1190/// Whether `condition`, raised while its `CALL ON` trap is delayed, is
1191/// ignored rather than held (rexxref `condtra.xml`, CALL ON and NOTREADY;
1192/// `iostrms.xml`, the DELAY state).
1193fn ignored_while_delayed(condition: &[u8]) -> bool {
1194    matches!(condition, b"ERROR" | b"FAILURE" | b"NOTREADY")
