64/// Converts a `interrexx-num` settings failure into a `Raised`.
65/// ```text
66/// raise syntax 40.4       -> 40.4       the catalogue entry
67/// raise syntax 40         -> 40.0       ditto, major line only (sub 0)
68/// raise syntax 40.001     -> 40.1       ".001" is the integer 1
69/// raise syntax '4E1'      -> 40.0       each half is a Rexx number, not an int literal
70/// raise syntax '40.1E2'   -> 98.941     found "40.100"
71/// raise syntax 40.10      -> 98.941     found "40.10"
72/// raise syntax 1          -> 98.941     found "1.0"
73/// raise syntax 3.5        -> 98.941     found "3.5"
74/// raise syntax 0 / 100 / 999 / 'abc' / 40.1000 / '40.'  -> 33.904
75/// ```
76pub(super) fn raise_syntax_condition(text: &[u8], additional: Vec<Vec<u8>>) -> Raised {
77    match syntax_code(text) {
78        Some(code) => raised_for_code(code, additional),
79        None => Raised::syntax(33, 904, Vec::new()),
80    }
81}

83/// Whether the text of a RAISE SYNTAX number is no error number, or one that
84/// names no error with a message: the clause's own faults, 33.904 and 98.941,
85/// raised where the clause is (r[instr.raise.number]).
86pub(super) fn names_no_error(text: &[u8]) -> bool {
87    let Some(code) = syntax_code(text) else {
88        return true;
89    };
90    let (major, sub) = (code / 1000, code % 1000);
91    let known = u16::try_from(major)
92        .ok()
93        .zip(u16::try_from(sub).ok())
94        .is_some_and(|(major, sub)| interrexx_inventory::errors::lookup(major, sub).is_some());
95    !known
96}
97
98/// The `SYNTAX` condition for error `code`, `major * 1000 + sub`, or 98.941
99/// where the catalogue has no text for it, naming the code as `major.sub`
100/// (r[instr.raise.number]).
101pub(crate) fn raised_for_code(code: u64, additional: Vec<Vec<u8>>) -> Raised {
102    let (major, sub) = (code / 1000, code % 1000);
103    let known = u16::try_from(major)
104        .ok()
105        .zip(u16::try_from(sub).ok())
106        .filter(|&(major, sub)| interrexx_inventory::errors::lookup(major, sub).is_some());
107    if let Some((major, sub)) = known {
108        return Raised::syntax(major, sub, additional);
109    }
110    Raised::syntax(98, 941, vec![format!("{major}.{sub}").into_bytes()])
111}

113/// [`raised_from_settings`] with the operand's own rendering in place of the
114/// string the required-string protocol converted it to.
115pub(super) fn raised_naming_the_operand(error: SettingsError, operand: &[u8]) -> Raised {
116    let names_the_operand = matches!(
117        error,
118        SettingsError::InvalidForm { .. }
119            | SettingsError::DigitsNotWhole { .. }
120            | SettingsError::FuzzNotWhole { .. }
121    );
122    let mut raised = raised_from_settings(error);
123    if names_the_operand {
124        raised.additional = vec![operand.to_vec()];
125    }
126    raised
127}
128
129pub(crate) fn raised_from_settings(error: SettingsError) -> Raised {
130    let additional = crate::error::into_substitutions(error.additional());
131    let (number, sub): (u16, u16) = match &error {
132        SettingsError::InvalidForm { .. } => (25, 11),
133        SettingsError::DigitsNotWhole { .. } => (26, 5),
134        SettingsError::FuzzNotWhole { .. } => (26, 6),
135        SettingsError::FuzzNotBelowDigits { .. } => (33, 1),
136    };
137    Raised::syntax(number, sub, additional)
138}
