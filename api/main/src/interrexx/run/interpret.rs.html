11    /// Parses `text` as an `INTERPRET` fragment and runs it **inside the
12    /// current activation**.
13    /// ```text
14    ///      2 *-*   leave outer
15    ///      2 *-*   interpret "leave outer"
16    /// ```
17    pub(super) fn run_fragment(&mut self, text: Vec<u8>) -> Result<Flow, Failure> {
18        let source =
19            interrexx_parse::ProgramSource::interpret(text, self.activity.clause_state.line());
20        let fragment: Rc<Fragment> = match interrexx_parse::fragment_from(source) {
21            Ok(fragment) => Rc::new(fragment),
22            Err(rejected) => return Err(self.raise_parse_failure(&rejected, None)),
23        };
24
25        // An owned `Fragment` would do here, since nothing but this loop reads
26        // it. It is an `Rc` because an `INTERPRET` inside a fragment makes this
27        // function reentrant and each level anchors its own.
28        if let Some(level) = self.activity.fragments.last_mut() {
29            level.fragment = Some(Rc::clone(&fragment));
30        }
31        let fragment_plan = self.fragment_plan(&fragment);
32        let code = Code {
33            body: &fragment.body,
34            symbols: &fragment.symbols,
35            // **A fragment carries its own plan, with every slot translated
36            // into the enclosing frame** (`Interp::fragment_plan`). It used to
37            // carry none, so its clause indents and compound splits were
38            // computed the way they were before either table existed; it needs
39            // one now because a fragment compiles, and a chunk's `Op::Load`
40            // and `Op::Store` carry plan slots.
41            plan: Some(&fragment_plan),
42        };
43
44        // `exit` inside `INTERPRET` ends the program, not the fragment, so
45        // it has to propagate rather than stop here -- `run_bounded`'s own
46        // catch-all does exactly that for anything it does not own, `Exit`
47        // included, with nothing fragment-specific to add.
48        let chunk = match self.compile_chunk(&fragment.body, &fragment_plan, self.chunk_trace()) {
49            Ok(chunk) => chunk,
50            Err(_) => {
51                self.seal_site_level();
52                return Err(Loud::chunk_refused().into());
53            }
54        };
55        // An `INTERPRET`ed string's loops stay driven: each execution compiles
56        // a chunk of its own, so a compiled loop would be compiled again every
57        // time, and no fragment is expected to be fast.
58        #[cfg(feature = "jit")]
59        chunk.jit.refuse_all();
60        // The literals this chunk interns are interned after this count.
61        let interned_from = self.heap.immortal_count();
62        // Truncated on the way out, so a temp this chunk leaks does not outlive it.
63        let temps = self.roots.activity_mut().push_frame();
64        let arena = self.roots.activity().frames();
65        let registers = arena.reserve(chunk.registers);
66        let ran = self.run_bounded(
67            &code,
68            0,
69            code.body.instructions.len(),
70            Some(&fragment.source),
71            BodyEngine::Chunk {
72                chunk: &chunk,
73                registers,
74            },
75        );
76        // Released on both paths, exactly as `Interp::drive` does: a frame left
77        // behind would keep its registers rooted for the rest of the run.
78        arena.release(registers);
79        self.roots.activity_mut().pop_frame(temps);
80        self.heap.release_immortal(interned_from, |handle| {
81            chunk.interned_values().any(|value| value == handle)
82        });
83        if !self.pinned_copies.is_empty() || self.site_hold_count != 0 {
84            self.unpin_chunk_copies(&chunk);
85        }
86        let flow = match ran {
87            Ok(flow) => flow,
88            Err(failure) => {
89                self.capture_fragment_frame(&failure);
90                self.seal_site_level();
91                return Err(failure);
92            }
93        };
94
95        // The exhausted search, at the fragment's own boundary rather than the
96        // program's: the `LEAVE`/`ITERATE` search does not cross an `INTERPRET`
97        // (this function's own doc comment has the six transcripts).
98        // Byte-identical in shape to `run_activation`'s own four arms,
99        // deliberately -- same four constructors, same `record_leave_failure`
100        // call -- because it is the same event happening at a different
101        // boundary.
102        match flow {
103            Flow::Leave(name, origin) => {
104                self.record_leave_failure(&origin);
105                let failure = match name {
106                    None => raised_leave_no_loop(),
107                    Some(id) => raised_leave_no_match(fragment.symbols.name(id).as_bytes()),
108                }
109                .into();
110                self.capture_fragment_frame(&failure);
111                self.seal_site_level();
112                Err(failure)
113            }
114            Flow::Iterate(name, origin) => {
115                self.record_leave_failure(&origin);
116                let failure = match name {
117                    None => raised_iterate_no_loop(),
118                    Some(id) => raised_iterate_no_match(fragment.symbols.name(id).as_bytes()),
119                }
120                .into();
121                self.capture_fragment_frame(&failure);
122                self.seal_site_level();
123                Err(failure)
124            }
125            other => {
126                if !self.uninit_ready.is_empty() {
127                    self.drain_leaving_fragment(&other)?;
128                }
129                Ok(other)
130            }
131        }
132    }

134    /// An `INTERPRET` activation leaves through the `RETURNED` block that runs
135    /// the `UNINIT`s a collection readied, a
136    /// `SIGNAL` out of it too.
137    #[cold]
138    #[inline(never)]
139    fn drain_leaving_fragment(&mut self, flow: &Flow) -> Result<(), Failure> {
140        let value = match flow {
141            Flow::Next | Flow::Signal(_) => None,
142            Flow::Return(value) | Flow::Exit(value) => *value,
143            _ => return Ok(()),
144        };
145        self.run_uninits_at_return(value)
146    }

148    /// Drops one `DROP` target: a plain variable, a whole stem, one tail, or
149    /// the `(v)` indirect form.
150    /// ```text
151    /// a=1; b=2; v='a b'      ; drop (v); say a; say b   ->  A / B  (both dropped)
152    /// x=1;      v=' x '      ; drop (v); say x           ->  X      (trimmed)
153    /// v='9'                  ; drop (v)                  ->  Error 31.2
154    /// v='.x'                 ; drop (v)                  ->  Error 31.3
155    /// w=1;      v='(w)'      ; drop (v)                  ->  Error 20.928
156    /// a=1;      v='a'        ; drop (v); say a           ->  A      (agrees with the old reading)
157    /// ```
158    pub(super) fn drop_variable(
159        &mut self,
160        code: &Code<'_>,
161        variable: &VariableRef,
162    ) -> Result<(), Failure> {
163        match variable {
164            VariableRef::Direct(id) => {
165                let name = code.symbols.name(*id);
166                if shape_of(name.as_bytes()) == NameShape::Compound {
167                    let (stem_name, stem_at) = code.stem(*id);
168                    let key = self.tail_key(code, *id)?;
169                    self.stem_drop_tail_at(stem_name, stem_at, &key);
170                } else {
171                    self.drop_by_name(name.as_bytes());
172                }
173            }
174            VariableRef::Indirect(id) => {
175                let value = self.read_list_selector(code, *id)?;
176                let value = self.required_string_value(value)?;
177                let text = self.to_text(value).into_owned();
178                let mut names = Vec::new();
179                for word in split_indirect_words(&text) {
180                    names.push(validate_indirect_word(word)?);
181                }
182                for name in &names {
183                    self.drop_list_name(name)?;
184                }
185            }
186        }
187        Ok(())
188    }

190    /// Drops one validated name of a `DROP (list)`. A compound name has the
191    /// variables of its tail replaced by their values as DROP reaches it, so
192    /// a name earlier in the list that changed one counts.
193    fn drop_list_name(&mut self, name: &[u8]) -> Result<(), Failure> {
194        if shape_of(name) == NameShape::Compound {
195            let (stem_name, key) = self.split_compound(name)?;
196            self.stem_drop_tail(&stem_name, &key);
197        } else {
198            self.drop_by_name(name);
199        }
200        Ok(())
201    }

203    /// Drops the variable, whole stem, or one verbatim-keyed tail `name`'s
204    /// own spelling names, dispatched by `shape_of`.
205    pub(super) fn drop_by_name(&mut self, name: &[u8]) {
206        match shape_of(name) {
207            NameShape::Simple => {
208                let slot = self.slot_of(name);
209                let frame = self.activation().frame;
210                self.clear_variable(frame, slot);
211            }
212            NameShape::Stem => self.stem_drop(name),
213            NameShape::Compound => {
214                let dot = name
215                    .iter()
216                    .position(|&b| b == b'.')
217                    .expect("NameShape::Compound guarantees at least one period");
218                let (stem_name, key) = name.split_at(dot + 1);
219                self.stem_drop_tail(stem_name, key);
220            }
221        }
222    }

224    /// The pause owed once `clause` has done its own work under a setting
225    /// that is interactive, `flow` being what it answered if it answered one.
226    /// Answers whether the clause is to run again, which is what `=` asks for.
227    /// A failure the pause raises is recorded against `clause`.
228    ///
229    /// The decision is per instruction class, against flags the trace setting
230    /// derives from the letter: an instruction pauses under `A`, `R` and `I`
231    /// and a label wherever labels trace. A command pauses from
232    /// [`Interp::debug_pause_after_command`] instead. A negative skip count
233    /// suppresses the echo and not the pauses it counts, so the flags come from
234    /// the setting it saved. An `IF`, a `SELECT` or a block `DO` that answers a
235    /// `Flow` ran its whole construct; a repeating `DO` that answers one ended
236    /// before any pass, and pauses after its header. A flow that leaves the
237    /// instruction for good owes no pause, and a `LEAVE` or an `ITERATE` pauses
238    /// once the loop it reaches has taken it, so one that reaches none raises
239    /// first.
240    #[cold]
241    #[inline(never)]
242    pub(crate) fn debug_pause_after_clause(
243        &mut self,
244        code: &Code<'_>,
245        index: usize,
246        clause: &Instruction,
247        source: Option<&ProgramSource>,
248        flow: Option<&Flow>,
249    ) -> Result<Pause, Failure> {
250        let kind = &clause.kind;
251        if let Some(flow) = flow
252            && (matches!(
253                flow,
254                Flow::Exit(_)
255                    | Flow::Return(_)
256                    | Flow::Signal(_)
257                    | Flow::Leave(..)
258                    | Flow::Iterate(..)
259            ) || matches!(
260                kind,
261                InstructionKind::If { .. } | InstructionKind::Select { .. }
262            ) || matches!(
263                kind,
264                InstructionKind::Do(body) | InstructionKind::Loop(body)
265                    if matches!(body.kind, interrexx_parse::LoopKind::Simple)
266            ))
267        {
268            return Ok(Pause::Go);
269        }
270        let mode = self.pausing_mode();
271        let pausing = match kind {
272            InstructionKind::Label { .. } => mode.labels,
273            InstructionKind::Command { .. } => false,
274            InstructionKind::Address(address) if address.command.is_some() => false,
275            _ => mode.all && pauses_after(kind),
276        };
277        if !pausing {
278            return Ok(Pause::Go);
279        }
280        self.debug_pause_now()
281            .inspect_err(|_| self.record_failure_site(code, index, source, clause))
282    }

284    /// The pause an instruction takes from
285    /// inside its own work: `INTERPRET`'s before its fragment, and `LEAVE`'s
286    /// and `ITERATE`'s once their loop has taken them. Answers whether the
287    /// instruction is to run again.
288    pub(crate) fn debug_pause_instruction(&mut self) -> Result<Pause, Failure> {
289        let mode = self.pausing_mode();
290        if !(mode.debug && mode.all) {
291            return Ok(Pause::Go);
292        }
293        self.debug_pause_now()
294    }

296    /// The pause taken once a command it traced
297    /// has settled `RC` and raised its condition. Answers whether the command
298    /// is to run again.
299    pub(crate) fn debug_pause_after_command(
300        &mut self,
301        status: crate::command::ReturnStatus,
302    ) -> Result<Pause, Failure> {
303        let mode = self.pausing_mode();
304        let traced = mode.all
305            || mode.commands
306            || match status {
307                crate::command::ReturnStatus::Error => mode.errors,
308                crate::command::ReturnStatus::Failure => mode.failures,
309                crate::command::ReturnStatus::Normal => false,
310            };
311        if !(mode.debug && traced) {
312            return Ok(Pause::Go);
313        }
314        self.debug_pause_now()
315    }

325    /// Answers whether the
326    /// clause is to run again.
327    ///
328    /// The prompt is printed once per activation, and a line that is neither
329    /// empty nor `=` runs as an `INTERPRET` fragment; the pause ends when the
330    /// fragment ended debug or changed a setting from inside it.
331    pub(crate) fn debug_pause_now(&mut self) -> Result<Pause, Failure> {
332        if self.activity.reading_debug_input || self.silenced() {
333            return Ok(Pause::Go);
334        }
335        if self.activation().debug.bypass {
336            self.activation_mut().debug.bypass = false;
337            return Ok(Pause::Go);
338        }
339        let skip = self.activation().debug.skip;
340        if skip > 0 {
341            let left = skip - 1;
342            self.activation_mut().debug.skip = left;
343            if left == 0
344                && let Some(saved) = self.activation_mut().debug.saved.take()
345            {
346                self.set_trace_mode(saved);
347            }
348            return Ok(Pause::Go);
349        }
350        if self.untraceable_code() {
351            return Ok(Pause::Go);
352        }
353        if !self.activation().debug.prompt_issued {
354            self.activation_mut().debug.prompt_issued = true;
355            let line = crate::error::Raised::debug_prompt_line();
356            let start = self.trace.len();
357            self.trace.extend_from_slice(&line);
358            self.trace.push(b'\n');
359            self.route_trace_line(start);
360        }
361        loop {
362            self.activity.reading_debug_input = true;
363            let line = self.debug_input_line();
364            self.activity.reading_debug_input = false;
365            let line = line?;
366            if line.is_empty() {
367                return Ok(Pause::Go);
368            }
369            if line == b"=" {
370                return Ok(Pause::Again);
371            }
372            if let Some(flow) = self.run_debug_fragment(line)? {
373                return Ok(Pause::Transfer(flow));
374            }
375            if self.activation().debug.bypass {
376                self.activation_mut().debug.bypass = false;
377                return Ok(Pause::Go);
378            }
379            if !self.trace_mode().debug {
380                return Ok(Pause::Go);
381            }
382        }
383    }

385    /// The debug input line: `LINEIN` sent to
386    /// `.local~DEBUGINPUT`, an absent entry or a `.nil` answer read as the
387    /// null string. A condition the send raises is the pause's failure.
388    fn debug_input_line(&mut self) -> Result<Vec<u8>, Failure> {
389        // The I/O exit's first pass.
390        if self.exits.is_some()
391            && let Some(line) = self.debug_read_exit()?
392        {
393            return Ok(line);
394        }
395        let Some(route) = self.local_route(b"DEBUGINPUT")? else {
396            return Ok(Vec::new());
397        };
398        let caller = self.caller();
399        let answer = pinned!(
400            self,
401            crate::pinning::PinKind::TraceWrapper,
402            self.send_message(route, crate::dispatch::LINEIN, None, &[], caller)
403        )?;
404        self.input_answer_text(answer)
405    }

407    /// One line typed at a pause, run as an `INTERPRET` fragment that traces
408    /// nothing. A SYNTAX condition inside it is reported under the two
409    /// `+++` error lines and swallowed, and the pause goes on --
410    /// measured, `zz = 1/0` at a pause reports 42.3 and the next line still
411    /// runs.
412    fn run_debug_fragment(&mut self, text: Vec<u8>) -> Result<Option<Flow>, Failure> {
413        let saved_pause = self.replace_debug_pause(true);
414        // The paused clause's line and indent, which a condition it queued
415        // reads at its boundary after the pause.
416        let saved_clause = self.save_clause_state();
417        self.activity.fragment_depth += 1;
418        let (line, indent) = (
419            self.activity.clause_state.line(),
420            self.activity.clause_state.current_value_indent,
421        );
422        self.enter_fragment_level(line, indent);
423        let nested = self.activity.clause_line_override.is_some();
424        let saved_entry = self.enter_fragment(nested);
425        // The clauses of the line answer to the paused clause's line, as the
426        // clauses of an `INTERPRET` do: a `CALL` of a label in the line sets
427        // `SIGL` to it.
428        let saved_override = self.activity.clause_line_override.replace(line);
429        let outcome = pinned!(
430            self,
431            crate::pinning::PinKind::Interpret,
432            self.run_fragment(text)
433        );
434        self.activity.clause_line_override = saved_override;
435        let depth = self.activity.fragment_depth;
436        self.activity
437            .pending_traps
438            .retain(|pending| pending.fragment_depth != depth);
439        self.activity.fragment_depth -= 1;
440        self.leave_fragment_level();
441        self.leave_fragment(saved_entry);
442        self.replace_debug_pause(saved_pause);
443        self.restore_clause_state(saved_clause);
444        match outcome {
445            // An `EXIT`, `RETURN` or `SIGNAL` the line ran leaves the paused
446            // clause as it would from the program.
447            Ok(flow @ (Flow::Exit(_) | Flow::Return(_) | Flow::Signal(_))) => Ok(Some(flow)),
448            Ok(_) => Ok(None),
449            Err(Failure::Raised(raised)) => {
450                self.report_debug_error(&raised);
451                self.clear_failure_levels();
452                Ok(None)
453            }
454            Err(other) => Err(other),
455        }
456    }

458    /// The two lines a failing pause line prints: the major and the sub, each
459    /// behind the debug prefix (101.4).
460    fn report_debug_error(&mut self, raised: &crate::error::Raised) {
461        for line in raised.debug_error_lines() {
462            let start = self.trace.len();
463            self.trace.extend_from_slice(&line);
464            self.trace.push(b'\n');
465            self.route_trace_line(start);
466        }
467    }

469    /// Records a fragment entered from the running activation's clause at
470    /// `line` and `indent`.
471    pub(super) fn enter_fragment_level(&mut self, line: usize, indent: usize) {
472        let owner = self.activation().id;
473        self.activity.fragments.push(crate::FragmentLevel {
474            owner,
475            fragment: None,
476            line,
477            indent,
478            outer_clause: self.activity.fragment_clause,
479            invocation: None,
480        });
481    }

483    /// Forgets the innermost fragment, whose enclosing one's clause is
484    /// current again.
485    pub(super) fn leave_fragment_level(&mut self) {
486        if let Some(level) = self.activity.fragments.pop() {
487            self.activity.fragment_clause = level.outer_clause;
488        }
489    }

491    /// Gives a fragment its own `>I>` count, and answers with the enclosing
492    /// state for [`Interp::leave_fragment`] to put back.
493    pub(super) fn enter_fragment(&mut self, nested: bool) -> TraceEntry {
494        let enclosing = self.activation().trace_entry;
495        let entry = if enclosing.may_announce() && !nested {
496            TraceEntry::Pending
497        } else {
498            TraceEntry::Spent
499        };
500        self.activation_mut().trace_entry = entry;
501        enclosing
502    }

504    /// Puts the enclosing state back after a fragment, **except** that a
505    /// fragment which announced leaves the activation `Done`.
506    pub(super) fn leave_fragment(&mut self, enclosing: TraceEntry) {
507        if self.activation().trace_entry != TraceEntry::Done {
508            self.activation_mut().trace_entry = enclosing;
509        }
510    }

513/// How a debug pause ended.
514pub(crate) enum Pause {
515    /// The program goes on with its next clause.
516    Go,
517    /// `=`: the paused clause runs again.
518    Again,
519    /// A line typed at the pause ran `EXIT`, `RETURN` or `SIGNAL`: the paused
520    /// clause is left with this flow.
521    Transfer(Flow),
522}
523
524/// Whether a debug pause follows an instruction of `kind` once its own work is
525/// done, as instructions end in a pause, or a block header's in a conditional
526/// one. `INTERPRET` pauses before its fragment instead. `CALL ON` and `CALL
527/// OFF` do not pause. `REPLY` pauses once the activation is marked as having
528/// replied and before the caller goes on, so its banner comes there and the
529/// continuation, which shares the activation's debug state, does not print it
530/// again.
531pub(crate) fn pauses_after(kind: &InstructionKind) -> bool {
532    !matches!(
533        kind,
534        InstructionKind::Then
535            | InstructionKind::Else { .. }
536            | InstructionKind::Otherwise
537            | InstructionKind::End { .. }
538            | InstructionKind::Return { .. }
539            | InstructionKind::Exit { .. }
540            | InstructionKind::Signal(_)
541            | InstructionKind::Raise(_)
542            | InstructionKind::Guard(_)
543            | InstructionKind::Interpret { .. }
544    ) && !matches!(kind, InstructionKind::Call(call) if matches!(**call, interrexx_parse::Call::Trap(_)))
545}
