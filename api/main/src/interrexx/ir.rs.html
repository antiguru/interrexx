43/// **The stream's width, asserted so that a change to it is a compile error.**
44/// Every op in every chunk is as wide as the widest variant, so the width is a
45/// fact about the whole array rather than about the variant that sets it, and a
46/// variant that outgrows this number grows every op there is. That is what this
47/// line catches, at the moment it happens, rather than leaving it to a
48/// measurement somebody has to take again. The widest payload has tail padding
49/// for the discriminant to sit in, which is why sixteen is an equality and not
50/// a bound.
51const _: () = assert!(size_of::<Op>() == 16);
52

63/// One step in a compiled stream.
64pub(crate) enum Op {
65    /// Opens the promoted clause of the instruction at `index`. `end` is the
66    /// op index one past this clause's last op -- the mark the register
67    /// allocator releases to when the clause finishes (the plan's Decisions
68    /// section: "a promoted clause takes a mark when its `Clause` op is
69    /// emitted and releases to it at `end`").
70    Clause { index: u32, end: u32 },
71    /// [`Op::Clause`] for a region holding an op a callee can resume after:
72    /// [`Op::CallExpr`], [`Op::CallArgs`] or [`Op::Send`].
73    CallingClause { index: u32, end: u32 },
74    /// Echoes the `>K>` line of one `DO`/`LOOP` header value, from register
75    /// `src`, under the tag [`HeaderRole`] gives it.
76    TraceKeyword { role: HeaderRole, src: u16 },
77    /// Validates the `DO`/`LOOP` header value in register `src` for the role
78    /// it plays and files it for [`Op::LoopRun`].
79    LoopHeaderValue { role: HeaderRole, src: u16 },
80    /// Runs the `DO`/`LOOP` at `index` from the header values the ops before it
81    /// filed, with **its body's clauses stepped from this chunk**.
82    LoopRun { index: u32 },
83    /// The bottom of one pass of the flattened
84    /// `DO`/`LOOP` at `index`: what the body just did, then the next pass's
85    /// header test, then either back to the body or on past the `END`.
86    LoopNext { index: u32 },
87    /// Opens the header test of the flat `DO`/`LOOP` at `index` whose
88    /// `WHILE` or `UNTIL` condition compiled into the stream, as a region of
89    /// that instruction ending at `end`: the clause the test runs in, at the
90    /// header's own line, entered here rather than by the pass boundary that
91    /// jumped to it. The condition's ops follow, and [`Op::LoopTested`] ends
92    /// the region.
93    LoopTest { index: u32, end: u32 },
94    /// The end of the header test [`Op::LoopTest`] opened: the condition
95    /// value in register `src` decides the pass, and the header clause's
96    /// boundary runs. Continues at the body or settles the loop's answer.
97    LoopTested { index: u32, src: u16 },
98    /// Files the `SELECT CASE` value of the instruction at `index`, which the
99    /// ops before it left in register `src`: rooted, echoed as `>K>` and the
100    /// block opened, as the walked expression's caller does.
101    SelectCase { index: u32, src: u16 },
102    /// Runs the instruction at `index` from the value its expression left in
103    /// register `src`: everything the instruction does past evaluating it.
104    ExecValue { index: u32, src: u16 },
105    /// Runs the instruction at `index` from the value its expression left in
106    /// register `src`, as [`Op::Exec`] runs a whole instruction: pinned, and
107    /// answering an `ExecOutcome` whose `Flow` ends the region. For the
108    /// instructions whose work past the evaluation can transfer control or
109    /// run Rexx code of its own, which [`Op::ExecValue`]'s cannot. A `RAISE`
110    /// raises from the registers its [`Op::RaisePart`]s filled, the first of
111    /// which is `src`.
112    ExecWith { index: u32, src: u16 },
113    /// The check the `REPLY` at `index` makes before it evaluates its
114    /// expression, ahead of that expression's ops: its refusal outside a
115    /// method, so that the expression does not run where the instruction may
116    /// not.
117    ReplyCheck { index: u32 },
118    /// A `FORWARD` option's check, conversion or keyword trace, before the
119    /// next option is evaluated. The opening method check reads no register.
120    ForwardPart {
121        index: u32,
122        part: ForwardPart,
123        src: u16,
124    },
125    /// One part of the `RAISE` at `index`: what the instruction does with the
126    /// value its expression left in register `src` before it evaluates the
127    /// next, its results kept in the registers [`crate::run::RaiseLayout`]
128    /// gives the part. [`Op::ExecWith`] then raises from them.
129    RaisePart {
130        index: u32,
131        part: RaisePart,
132        src: u16,
133    },
134    /// Echoes the `*-*` line of the clause of the instruction at `index`.
135    TraceClause { index: u32 },
136    /// Evaluates expression `slot` of the instruction at `index` into
137    /// register `dst`, still dispatched through `eval.rs` rather than
138    /// reimplemented here (the plan's Decisions section: trace ops land
139    /// before the first *native* expression op, so this stays
140    /// trace-identical to the `eval.rs` call it wraps).
141    EvalExpr { index: u32, slot: u32, dst: u16 },
142    /// Runs the call at `path` inside expression `slot` of the instruction at
143    /// `index`, into register `dst`, through the resolution site `site` keeps.
144    CallExpr {
145        index: u32,
146        slot: u16,
147        path: NodePath,
148        site: u16,
149        dst: u16,
150    },
151    /// Appends the value in `src` to the driver's argument stack, or an
152    /// omitted position when `src` is [`Op::ARG_OMITTED`].
153    PushArg { src: u16 },
154    /// The `>A>` line one argument owes, for the value now in `src`, or the
155    /// **empty** value line an omitted position owes when `src` is
156    /// [`Op::ARG_OMITTED`] -- which is not the absence of a line.
157    TraceArgument { src: u16 },
158    /// A call over the `argc` arguments now on top of the driver's argument
159    /// stack.
160    CallArgs {
161        slot: u16,
162        path: NodePath,
163        site: u16,
164        argc: u16,
165        dst: u16,
166    },
167    /// The value line the node at `path` inside expression `slot` owes, for
168    /// the value now in `src`: `>F>` behind [`Op::CallExpr`] or
169    /// [`Op::CallArgs`], `>O>` behind [`Op::Reference`], `>N>` behind
170    /// [`Op::ClassAt`].
171    TraceFunction {
172        index: u32,
173        slot: u16,
174        path: NodePath,
175        src: u16,
176    },
177    /// Tests the listed `WHEN`/`WHEN CASE` at `index` and stores whether it
178    /// holds in register `dst`, as the Rexx logical value [`Op::JumpUnless`]
179    /// reads back.
180    WhenTest {
181        index: u32,
182        case: Option<u16>,
183        dst: u16,
184    },
185    /// Closes the branch control has just run off the end of: the clause
186    /// boundary a promoted construct owes where wrapper
187    /// around the whole arm runs one.
188    EndBranch,
189    /// Closes the [`Op::EnterWhen`]/[`Op::EnterOtherwise`] frame whose branch
190    /// ends here, if one is open.
191    EndWhen,
192    /// Opens a frame over the branch of the listed `WHEN` at `when`, which
193    /// belongs to the `SELECT` at `select`.
194    EnterWhen { select: u32, when: u32 },
195    /// Opens a frame over the `OTHERWISE` branch of the `SELECT` at `select`,
196    /// which is [`Op::EnterWhen`]'s counterpart for the branch no `WHEN`
197    /// owns.
198    EnterOtherwise { select: u32 },
199    /// Loads constant `konst` of [`Chunk::consts`] into register `dst`.
200    Const { dst: u16, konst: u32 },
201    /// Loads the value of the constant symbol `symbol` into register `dst`:
202    /// its own upcased spelling, which is observable rather than incidental --
203    /// `say 1e5` prints `1E5`.
204    LoadConstant { symbol: SymbolId, dst: u16 },
205    /// Echoes the `>L>` line of the literal or constant symbol in register
206    /// `src`.
207    TraceLiteral { src: u16 },
208    /// Loads the value of the bare symbol `symbol` into register `dst`, by the
209    /// read [`SymbolRead`] names. `at` is the symbol's frame slot, except for
210    /// [`SymbolRead::Environment`], which has none: there it is the chunk's
211    /// memo site for the read ([`Chunk::environment_site`]).
212    Load {
213        symbol: SymbolId,
214        read: SymbolRead,
215        at: PlanSlot,
216        dst: u16,
217    },
218    /// Echoes the `>V>` line of the read in register `src`, and the `>C>` line
219    /// a compound read owes in front of it.
220    TraceRead {
221        symbol: SymbolId,
222        read: SymbolRead,
223        src: u16,
224    },
225    /// Computes `lhs op rhs` into register `dst`, for the operators
226    /// `eval::is_arithmetic` names.
227    Arith {
228        op: Operator,
229        /// This site's own slot in [`Chunk::hints`] -- **not** a position in
230        /// the op stream. The table is dense over the ops that can specialise,
231        /// keyed off the op the way [`Chunk::consts`] is, so a stream carries
232        /// no entry for the ops that never quicken and the driver's own loop
233        /// needs no counter walking beside it to find one.
234        hint: u32,
235        lhs: u16,
236        rhs: u16,
237        dst: u16,
238    },
239    /// Computes `lhs op rhs` into register `dst`, for the concatenation,
240    /// comparison and logical operators -- every operator `eval::
241    /// is_native_binary` names and `eval::is_arithmetic` does not.
242    Binary {
243        op: Operator,
244        lhs: u16,
245        rhs: u16,
246        dst: u16,
247        /// A comparison operator one of whose operands is a literal that is
248        /// not a number ([`crate::eval::never_a_number`]), decided where the
249        /// op is compiled: two plain strings then compare by the string rule
250        /// at once, the answer the numeric attempt would come to.
251        text: bool,
252        /// A concatenation whose left operand is the concatenation compiled
253        /// just before it in the same chain, `(a || b) || c`: what is in `lhs`
254        /// is the string that op built, if it built one, and this op may
255        /// append to it ([`Interp::concat_onto`]) rather than copy it. The
256        /// driver checks that it is that string before relying on it.
257        ///
258        /// [`Interp::concat_onto`]: crate::Interp::concat_onto
259        onto: bool,
260    },
261    /// Echoes the `>O>` line of the operator result in register `src`.
262    TraceOperator { op: Operator, src: u16 },
263    /// Computes `op src` into register `dst`, for the prefix operators `+`,
264    /// `-` and `\`.
265    Prefix { op: PrefixOp, src: u16, dst: u16 },
266    /// Echoes the `>P>` line of the prefix-operator result in register `src`.
267    TracePrefix { op: PrefixOp, src: u16 },
268    /// Writes register `src` through the target of the `Assignment` at
269    /// `index`, and traces the write. `shared` says that the target is a
270    /// compound its value reads with [`SymbolRead::CompoundShared`], whose
271    /// key the write uses.
272    Store {
273        index: u32,
274        at: PlanSlot,
275        src: u16,
276        shared: bool,
277    },
278    /// Prints the `SAY` at `index` from register `src`, or a blank line when
279    /// it has no expression at all.
280    Say { index: u32, src: Option<u16> },
281    /// The `SIGNAL` at `index`, in whichever of its three forms that
282    /// instruction carries.
283    Signal { index: u32, src: Option<u16> },
284    /// One step of a `USE ARG` clause that has a default expression.
285    UseArg(UseArgOp),
286    /// The `PARSE`, `ARG` or `PULL` at `index`: takes its first parse
287    /// string, from register `src` for `PARSE VALUE`, and walks the
288    /// template's entries up to `to`, the first whose operand the stream
289    /// evaluates. Where entries are left it files the walk in the registers
290    /// from `base` on ([`crate::parse_template::WALK_REGISTERS`] of them) for
291    /// the [`Op::ParseRun`]s after it; a template with none has `base`
292    /// [`Op::ARG_OMITTED`]. A clause the walk does not take -- a line will be
293    /// traced, a fold, a source read as a line, a parse string that is not a
294    /// string it can read in place -- is run whole by `Interp::exec_parse`,
295    /// and the runs after this op are not driven.
296    ParseBegin {
297        index: u32,
298        src: u16,
299        base: u16,
300        to: u16,
301        opening: ParseOpening,
302        strings: u8,
303    },
304    /// A literal template piece prepared by the plan, after the opening
305    /// checked that the clause doesn't owe parse trace lines.
306    ParseStore {
307        slot: u32,
308        value: ObjRef,
309        scratch: u16,
310    },
311    /// Walks template entries `from..to` of the `PARSE` at `index` on from
312    /// the walk filed from register `base`: entry `from` with its operand in
313    /// register `operand`, and the entries after it, whose operands are read
314    /// off the template.
315    ParseRun {
316        index: u32,
317        base: u16,
318        from: u16,
319        to: u16,
320        operand: u16,
321    },
322    /// Ends the activation at `index` with the value in register `src`, or
323    /// with no value at all for the bare form, and answers the `Flow` its
324    /// keyword calls for.
325    Return {
326        index: u32,
327        src: Option<u16>,
328        keyword: ReturnKeyword,
329    },
330    /// Queues the line in register `src` at the end its keyword names, or the
331    /// null string for the bare form, for the `PUSH`/`QUEUE` at `index`.
332    Queue {
333        index: u32,
334        src: Option<u16>,
335        keyword: QueueKeyword,
336    },
337    /// Runs the named or qualified `CALL` at `index` and settles `RESULT` from
338    /// what came back. Only a call without a namespace can cache its target.
339    Call { index: u32, site: u16 },
340    /// [`Op::Call`] with its arguments already on the driver's argument
341    /// stack, which is what [`Op::PushArg`] put there.
342    CallNamed { index: u32, site: u16, argc: u16 },
343    /// Runs the message-send clause at `index`: the receiver, the scope
344    /// override, the arguments and their `>A>` lines, the send, the `>M>`

347    Message { index: u32 },

351    Expose { index: u32 },
352    /// Runs the instruction at `index` through the arm of
353    /// `Interp::exec_instruction` that its kind names -- the shared
354    /// implementation named, entered without `Op::Clause`'s region around it.
355    Exec { index: u32 },
356    /// Answers the `Flow` the `LEAVE`/`ITERATE` at `index` resolves to: the
357    /// instruction's own name, if it has one, and the [`crate::run::
358    /// LeaveOrigin`] captured here rather than reconstructed later.
359    Escape { index: u32 },
360    /// Continues at op `target`.
361    Jump { target: u32 },
362    /// Continues at op `target` unless register `reg` holds the logical value
363    /// `1`, and at the next op if it does.
364    JumpUnless { reg: u16, target: u32 },
365    /// Validates the condition value in `reg`, emits the `>>>` line it owes,
366    /// and continues at op `target` unless it holds -- [`Op::JumpUnless`]'s
367    /// own branch, taken from the logical value rather than from a register
368    /// this op would have to write it to first.
369    ConditionJump {
370        index: u32,
371        reg: u16,
372        keyword: ConditionKeyword,
373        target: u32,

494/// The route from an expression slot's root down to the node one op names.
495#[derive(Clone, Copy, PartialEq, Eq)]
496pub(crate) struct NodePath(u32);
497
498impl NodePath {
499    /// The slot's own expression: the sentinel alone, with no steps below it.
500    pub(crate) const ROOT: NodePath = NodePath(1);
501
502    /// One step further down, into the right child when `right` and the left
503    /// or only child otherwise, or `None` when this width has no room for it.
504    pub(crate) fn child(self, right: bool) -> Option<NodePath> {
505        (self.0 >> (u32::BITS - 1) == 0).then(|| NodePath(self.0 << 1 | u32::from(right)))
506    }
507
508    /// One step further down, into child `index` of a node with a list of
509    /// children: `index` clear steps and a set one. `None` when this width
510    /// has no room for them.
511    pub(crate) fn nth(self, index: usize) -> Option<NodePath> {
512        let width = u32::try_from(index).ok()?.checked_add(1)?;
513        (self.0.leading_zeros() >= width).then(|| NodePath(self.0 << width | 1))
514    }
515
516    /// The steps below the sentinel, **outermost first**: the order a descent
517    /// from the slot's root walks them in.
518    pub(crate) fn steps(self) -> impl Iterator<Item = bool> {
519        // The sentinel is the highest set bit, so its position is how many
520        // steps sit below it -- zero for `ROOT`, whose value is the sentinel
521        // alone.
522        let depth = u32::BITS - 1 - self.0.leading_zeros();
523        (0..depth).rev().map(move |bit| self.0 >> bit & 1 == 1)
524    }
525}

527/// The frame slot a symbol resolves to, worked out when this chunk was
528/// compiled, or the absence of one.
529#[derive(Clone, Copy, PartialEq, Eq)]
530pub(crate) struct PlanSlot(u32);
531
532impl PlanSlot {
533    /// No slot resolved: the read or write works its own out, which is what
534    /// every one of them did before there was a compiler to do it earlier.
535    pub(crate) const UNRESOLVED: PlanSlot = PlanSlot(u32::MAX);
536
537    /// The slot `at`, or [`PlanSlot::UNRESOLVED`] when it does not fit this
538    /// width.
539    pub(crate) fn of(at: usize) -> PlanSlot {
540        match u32::try_from(at) {
541            Ok(at) if at != PlanSlot::UNRESOLVED.0 => PlanSlot(at),
542            _ => PlanSlot::UNRESOLVED,
543        }
544    }
545
546    /// The slot, or `None` when this op carries none.
547    pub(crate) fn resolved(self) -> Option<usize> {
548        (self != PlanSlot::UNRESOLVED).then_some(usize::cast_from(self.0))
549    }
550}

552/// The one way [`compile`] can fail: a body that does not fit the index
553/// widths this stream commits to, not a language construct it refuses (the
554/// plan's Decisions section: "the compiler has one error, and it is a
555/// machine width rather than a language construct"). Every instruction
556/// still compiles; this reports that the *stream* built from them does not
557/// fit some index's width.
558#[derive(Debug)]
559pub(crate) struct ChunkTooLarge {
560    /// What overflowed: the `u32` op stream, or the `u16` register file.
561    #[allow(dead_code, reason = "no production caller renders a refusal's reason")]
562    pub(crate) what: &'static str,
563}
564
565/// Whether a compiled arithmetic site keeps a hint about which path to try
566/// first.
567const QUICKENING: bool = true;
568
569/// The state a site starts in and stays in while the small-integer path keeps
570/// answering: try that path first.
571const TRY_SMALL_INT: u32 = 0;
572
573/// The state a site moves to the first time the small-integer path answers
574/// `None`: go straight to the general one.
575const GENERAL: u32 = 1;
576
577/// One arithmetic site's hint.
578struct PatchSlot(Cell<u32>);
579
580impl PatchSlot {
581    fn new() -> PatchSlot {
582        PatchSlot(Cell::new(TRY_SMALL_INT))
583    }
584
585    fn get(&self) -> u32 {
586        self.0.get()
587    }
588
589    fn set(&self, state: u32) {
590        self.0.set(state);
591    }
592}
593
594/// One chunk's patch table: a hint per [`Op::Arith`], indexed by that op's own
595/// `hint` field.
596struct Hints {
597    /// One slot per arithmetic op, or **empty** when [`QUICKENING`] is off.
598    slots: Vec<PatchSlot>,
599    /// How many arithmetic ops have taken a slot.
600    next: u32,
601}
602
603impl Hints {
604    fn new() -> Hints {
605        Hints {
606            slots: Vec::new(),
607            next: 0,
608        }
609    }
610
611    /// Reserves the slot for one arithmetic op, answering the index it
612    /// carries.
613    fn reserve(&mut self) -> Result<u32, ChunkTooLarge> {
614        let at = self.next;
615        self.next = at.checked_add(1).ok_or(ChunkTooLarge {
616            what: "arithmetic sites past u32",
617        })?;
618        if QUICKENING {
619            self.slots.push(PatchSlot::new());
620        }
621        Ok(at)
622    }
623
624    /// Whether site `at` is still worth trying the small-integer path for.
625    fn tries_small_int(&self, at: u32) -> bool {
626        !QUICKENING
627            || self
628                .slots
629                .get(usize::cast_from(at))
630                .is_none_or(|slot| slot.get() == TRY_SMALL_INT)
631    }
632
633    /// Records that site `at` has had the small-integer path answer `None`.
634    fn saw_general(&self, at: u32) {
635        if QUICKENING && let Some(slot) = self.slots.get(usize::cast_from(at)) {
636            slot.set(GENERAL);
637        }
638    }
639}

641/// Whether a compiled call site keeps the resolution it made.
642const CALL_SITE_CACHE: bool = true;
643
644/// A call whose target is resolved anew on each entry, with no cache slot.
645const UNCACHED_CALL: u16 = u16::MAX;
646
647/// One call site's kept resolution, or none yet, with the
648/// `Interp::routine_generation` it was made under. A `Routine` copy the kept
649/// resolution names is pinned, and the program of a `::ROUTINE` it names in
650/// another program is held ([`Interp::repin`]), while the site keeps it.
651struct CallSite {
652    resolved: Cell<Option<Resolved>>,
653    generation: Cell<u32>,
654    held: Cell<Option<SiteHold>>,
655}
656
657impl CallSite {
658    fn new() -> CallSite {
659        CallSite {
660            resolved: Cell::new(None),
661            generation: Cell::new(0),
662            held: Cell::new(None),
663        }
664    }
665
666    /// The copy the kept resolution names, or `.nil`.
667    fn pinned(&self) -> ObjRef {
668        self.resolved.get().map_or(ObjRef::NIL, Resolved::copy)
669    }
670
671    /// The kept resolution, or `None` where it is one looked up on every call
672    /// and `generation` has moved on.
673    fn get(&self, generation: u32) -> Option<Resolved> {
674        let resolved = self.resolved.get()?;
675        if resolved.kept_until_routines_change() && self.generation.get() != generation {
676            return None;
677        }
678        Some(resolved)
679    }
680
681    /// Keeps `resolved` under `generation`, with `hold` on its program.
682    fn set(&self, resolved: Resolved, generation: u32, hold: Option<SiteHold>) -> Repin {
683        let mut repin = Repin::to(self.pinned(), resolved.copy());
684        if hold.is_some() || self.held.get().is_some() {
685            let release = self.held.replace(hold);
686            if release != hold {
687                repin.program = Some(ProgramRepin { release, hold });
688            }
689        }
690        self.resolved.set(Some(resolved));
691        self.generation.set(generation);
692        repin
693    }
694}

696/// One [`Op::Send`]'s message name, and the method the site last resolved
697/// it to.
698struct SendSite {
699    name: Box<[u8]>,
700    memo: Cell<Option<crate::dispatch::SendMemo>>,
701}
702
703/// One chunk's resolution table: a slot per call op -- [`Op::Call`] and
704/// [`Op::CallExpr`] alike -- indexed by that op's own `site` field, and the
705/// send site each [`Op::Send`]'s `site` names.
706struct Calls {
707    /// One slot per call op, or **empty** when [`CALL_SITE_CACHE`] is off.
708    slots: Vec<CallSite>,
709    sends: Vec<SendSite>,
710    /// One memo per `.NAME` read, indexed by its [`Op::Load`]'s `at`.
711    environments: Vec<Cell<Option<crate::environment::EnvMemo>>>,
712    /// How many call ops have taken a slot, kept separately from `slots.len()`
713    /// so that the indices the ops carry are the same whether or not the table
714    /// exists -- [`Hints::next`]'s own reason, so that a golden op stream reads
715    /// identically under either setting.
716    next: u16,
717}
718
719impl Calls {
720    fn new() -> Calls {
721        Calls {
722            slots: Vec::new(),
723            sends: Vec::new(),
724            environments: Vec::new(),
725            next: 0,
726        }
727    }
728
729    /// Files one send's message name, answering the site it has.
730    fn reserve_send(&mut self, name: Box<[u8]>) -> Result<u32, ChunkTooLarge> {
731        let at = u32::try_from(self.sends.len()).map_err(|_| ChunkTooLarge {
732            what: "send sites past u32",
733        })?;
734        self.sends.push(SendSite {
735            name,
736            memo: Cell::new(None),
737        });
738        Ok(at)
739    }
740
741    /// Files one `.NAME` read's memo, answering the site its load carries.
742    fn reserve_environment(&mut self) -> PlanSlot {
743        let at = PlanSlot::of(self.environments.len());
744        if at != PlanSlot::UNRESOLVED {
745            self.environments.push(Cell::new(None));
746        }
747        at
748    }
749
750    /// Reserves the slot for one call op, answering the index it carries.
751    fn reserve(&mut self) -> Result<u16, ChunkTooLarge> {
752        let at = self.next;
753        self.next = at.checked_add(1).ok_or(ChunkTooLarge {
754            what: "call sites past u16",
755        })?;
756        if CALL_SITE_CACHE {
757            self.slots.push(CallSite::new());
758        }
759        Ok(at)
760    }
761
762    /// What site `at` resolved to last time, or `None` for a site that has not
763    /// resolved yet or has no slot at all -- which is what makes
764    /// [`CALL_SITE_CACHE`] off behave as no table rather than as a table that
765    /// answers wrongly.
766    fn resolved(&self, at: u16, generation: u32) -> Option<Resolved> {
767        self.slots
768            .get(usize::cast_from(at))
769            .and_then(|slot| slot.get(generation))
770    }
771
772    /// Records what site `at` resolved to, under `generation`.
773    ///
774    /// **A miss is never kept.** `Resolved::Unresolved` is the answer for a
775    /// name nothing has yet, and a file the search would find next time must
776    /// not be masked by this table -- the same reason a raising resolution
777    /// was never cached.
778    fn remember(
779        &self,
780        at: u16,
781        resolved: Resolved,
782        generation: u32,
783        hold: Option<SiteHold>,
784    ) -> Repin {
785        if matches!(resolved, Resolved::Unresolved) {
786            return Repin::NONE;
787        }
788        match self.slots.get(usize::cast_from(at)) {
789            Some(slot) => slot.set(resolved, generation, hold),
790            None => Repin::NONE,
791        }
792    }
793}

795/// One body's compiled instruction stream, cached on `Interp` under its
796/// `Plan`'s `BodyKey` **and the [`ChunkTrace`] it was compiled under**
797/// (`Interp::chunk_for`, in `plan.rs`).
798/// A clause's source line and its nesting indent, both fixed by the source
799/// text and so decided once, where the chunk is compiled.
800#[derive(Copy, Clone, PartialEq, Eq, Debug)]
801pub(crate) struct ClausePosition {
802    /// `Plan::lines`' own entry: the source line the clause starts on.
803    pub(crate) line: u32,
804    /// `Plan::indents`' own entry: the nesting indent **before**
805    /// `Activity::activation_indent` is added,
806    /// which is what `Interp::printed_indent` adds to it.
807    pub(crate) indent: u32,
808}

810/// What [`Chunk::opening_at`] answers for one clause.
811#[derive(Copy, Clone, PartialEq, Eq, Debug)]
812pub(crate) struct ClauseOpening {
813    /// The clause's line and indent, meaningful only where `tabled`.
814    pub(crate) position: ClausePosition,
815    /// Whether `position` holds the clause's own answer. **False for every
816    /// clause of a body whose plan cannot supply every entry** -- a body
817    /// planned without source has no line table, and a clause line or indent
818    /// too wide for a `u32` has no entry either -- and then the driver reads
819    /// the tables as it always did, so this is a shortcut and never the only
820    /// route to an answer.
821    pub(crate) tabled: bool,
822    /// The register top when the clause opens: every register at or above it
823    /// is dead there, because the allocator is a stack and what an enclosing
824    /// construct keeps live across its clauses (a loop header's values, a
825    /// `SELECT CASE` value) sits below it. The driver clears them as the
826    /// clause opens, so that no value a finished clause computed stays rooted
827    /// by its register.
828    pub(crate) register_top: u16,
829}

849pub(crate) struct Chunk {
850    /// The trace setting this stream's ops were emitted for, which is half of
851    /// the key it is cached under and the thing the driver compares the
852    /// setting in force against.
853    trace: ChunkTrace,
854    /// Whether `trace_flow` settled each clause's setting at compile time and
855    /// left out the echoes it found would not print (`compile`), rather than
856    /// keeping every echo behind its run-time gate (`compile_unsettled`).
857    settled: bool,
858    /// One entry per instruction, in instruction order: every instruction
859    /// compiles, so an instruction no specialised op covers still gets one.
860    ops: valid::ValidOps,
861    /// Instruction index -> **the op control resumes at** when it arrives at
862    /// that instruction. One entry per instruction, in order, plus one final
863    /// entry at `ops.len()`: `run_bounded`'s absorption guard is inclusive,
864    /// so a construct's resume point can be one past its last instruction,
865    /// and a map that stopped at `len - 1` would panic there instead of
866    /// failing loudly at compile.
867    op_of: Vec<u32>,
868    /// The register allocator's high-water mark (the plan's Decisions
869    /// section: "the chunk records its high-water mark").
870    /// `Interp::drive` reserves this many registers before running a
871    /// chunk and truncates them away on the way out, and `Interp::run_fragment`
872    /// does the same for the chunk an `INTERPRET` compiles.
873    pub(crate) registers: u16,
874    /// Instruction index -> what the driver reads as that clause opens, one
875    /// entry per instruction in instruction order. One table rather than one
876    /// per field, so opening a clause pays for one bounds check and one base
877    /// pointer.
878    openings: Box<[ClauseOpening]>,
879    /// The literal values [`Op::Const`] loads, **one entry per distinct
880    /// literal** rather than one per occurrence.
881    consts: Vec<Box<[u8]>>,
882    /// The value each entry of [`Chunk::consts`] was interned to, or
883    /// [`ObjRef::NIL`] for one not built yet.
884    interned: Vec<Cell<ObjRef>>,
885    /// The same for [`Op::LoadConstant`], indexed by the constant symbol's own
886    /// `SymbolId` rather than by a slot `compile` assigned.
887    interned_symbols: Vec<Cell<ObjRef>>,
888    /// What `trace_flow::analyse` answered for each instruction, kept only
889    /// where `debug_assert` is compiled in: the driver checks a `Known`
890    /// answer against the setting actually in force at that clause, which is
891    /// what makes the analysis checked rather than argued.
892    #[cfg(debug_assertions)]
893    settings: Box<[trace_flow::Setting]>,
894    /// The quickening hints [`Op::Arith`] reads, one per such op.
895    hints: Hints,
896    /// The resolutions [`Op::Call`] reads and writes, one per such op.
897    calls: Calls,
898    /// **The `WHILE`/`UNTIL` conditions, outside the stream.** A loop's
899    /// condition is evaluated inside its header's own clause, which the flat
900    /// loop opens and closes around it, so it is not a region the driver
901    /// walks: `Interp::run_condition` runs these ops for it. Value ops only
902    /// -- loads, constants and operators -- and no trace op, so a condition
903    /// is run from here only while `TRACE` shows no intermediates.
904    conditions: valid::ValidOps,
905    /// Instruction index -> the range of [`Chunk::conditions`] its loop's
906    /// condition compiled to, for a `DO`/`LOOP` whose condition did.
907    condition_of: Box<[Option<ConditionOps>]>,
908    /// Instruction index -> the header test region of a `DO`/`LOOP` whose
909    /// condition calls or sends, which [`Chunk::conditions`] cannot hold
910    /// because a call op parks the region it is in.
911    loop_test_of: Box<[Option<LoopTestOps>]>,
912    /// Which of this chunk's loops run compiled (`crate::jit`). Boxed: held
913    /// in place it makes the chunk two words wider than the build without
914    /// the feature, and the driver's loop compiles worse around a chunk of
915    /// that shape (#849).
916    #[cfg(feature = "jit")]
917    pub(crate) jit: Box<drive::jit_host::Loops>,
918}
