143fn forward_slot<'a>(body: &'a CodeBody, forward: &'a Forward, slot: u32) -> Option<ExprRef<'a>> {
144    if let Some(item) = slot.checked_sub(FORWARD_ARRAY_SLOT) {

151    }
152    forward_option_parts(body, forward)[usize::cast_from(slot)].1

413/// Splits on the default separator:
414/// a piece per line end, one `\r` dropped from a piece that ends in one, and
415/// a trailing piece only where the text does not end at a separator.
416fn makearray_lines(text: &[u8]) -> Vec<&[u8]> {
417    let mut pieces = Vec::new();
418    let mut start = 0;
419    while let Some(offset) = text[start..].iter().position(|byte| *byte == b'\n') {
420        let separator = start + offset;
421        let mut end = separator;
422        if end > start && text[end - 1] == b'\r' {
423            end -= 1;
424        }
425        pieces.push(&text[start..end]);
426        start = separator + 1;
427    }
428    if start < text.len() {
429        pieces.push(&text[start..]);
430    }
431    pieces
432}

566    /// The calling convention [`Interp::prepare_level`] records on the
567    /// running activation.
568    #[inline(always)]
569    pub(crate) fn record_call_convention(&mut self) {
570        let arguments = Rc::clone(&self.activity.call_context.arguments);
571        // **The name is shared only where nothing else records it.** A
572        // method activation already carries the message name it was entered
573        // under on `Activation::method_identity`, and that is the hot path:
574        // measured, `instructions:u` on `bench-programs/dispatch.rex`, one
575        // no-argument send per iteration, +1.732% against BASE with an
576        // `Rc<[u8]>` built here for every send and +0.093% with this test in
577        // front of it.
578        let name = (self.activation().entry != crate::activation::Entry::Method)
579            .then(|| Rc::clone(&self.activity.call_context.name));
580        let activation = self.activation_mut();
581        if let Some(name) = name {
582            activation.call_name = Some(name);
583        }
584        activation.call_arguments = Some(arguments);
585    }

587    /// The running activation's program, plan and chunk.
588    #[inline(always)]
589    pub(crate) fn running_level(&mut self) -> Result<crate::ir::Level, Failure> {
590        let program = Rc::clone(&self.activation().program);
591        let plan = Rc::clone(&self.activation().plan);
592        let selector = self.activation().body;
593        // The key this body's plan was cached under, and the key its chunk is
594        // cached under. **They have to be the same key**, and nothing but this
595        // says so: `plan` is read off the activation while `key` is rebuilt
596        // from the activation's program id and body selector, so a push that
597        // paired a plan with the wrong id would put a body's chunk under
598        // another body's name -- a wrong answer rather than a miss, exactly
599        // what `BodyKey::directive`'s own doc says about the selector.
600        let key = self.activation().body_key();
601        debug_assert!(
602            self.plans
603                .get(&key)
604                .is_some_and(|cached| Rc::ptr_eq(cached, &plan)),
605            "the running activation's body key does not name the plan it is running with, so \
606             its chunk would be cached under another body's name"
607        );
608
609        // **Every activation's body runs from a compiled chunk.** There is
610        // no second engine and no selection left to make.
611        let trace = self.chunk_trace();
612        let chunk = match plan.last_chunk(key, trace) {
613            Some(chunk) => chunk,
614            None => {
615                // A selector that resolves to nothing is an internal inconsistency
616                // and not a program error: it can only be built by a resolution step
617                // that already looked the body up. Loud rather than a panic, matching
618                // this crate's standing rule -- an abort is precisely the outcome
619                // that rule exists to exclude.
620                let Some(body) = body_of(&program, selector) else {
621                    return Err(Loud::missing_body().into());
622                };
623                let Some(chunk) = self.chunk_for(key, trace, body, &plan) else {
624                    return Err(Loud::chunk_refused().into());
625                };
626                chunk
627            }
628        };
629        Ok(crate::ir::Level {
630            program,
631            plan,
632            chunk,
633            selector,
634        })
635    }

647    /// Applies one clause's `Flow` to this activation.
648    pub(crate) fn apply_flow(
649        &mut self,
650        code: &Code<'_>,
651        flow: Flow,
652    ) -> Result<Option<Ended>, Failure> {
653        match flow {
654            Flow::Next => self.activation_mut().pc += 1,
655            Flow::Goto(target) => self.activation_mut().pc = target,
656            // `SIGNAL`, once its target has escaped every nested construct
657            // and every `INTERPRET` fragment it fired from (`Flow::Signal`'s
658            // own doc comment has why it cannot ride `Goto` to get here).
659            // The only consumer, matching `Goto`'s own arm exactly: `target`
660            // already resolved against this activation's own body
661            // (`resolve_signal_target`), which is exactly the body `code` is
662            // bound to.
663            Flow::Signal(target) => {
664                self.activation_mut().pc = target;
665                self.activity.activation_indent = 0;
666            }
667            // **The one place an activation's value stops being a clause's
668            // temporary**, which is why the root that outlives the temps
669            // stack is taken here rather than at each of the half-dozen
670            // constructs that can produce one. `EXIT`, a top-level `RETURN`,
671            // a `RAISE` with an `EXIT` tail and a handler's own exit all
672            // arrive as one of these two variants; every one of them can end
673            // up as the value `execute` hands `exit_code_for`, and by then
674            // the frame that rooted it has been popped. See
675            // [`Interp::root_exit_value`] for the measurement.
676            Flow::Exit(value) => {
677                if let Some(value) = value {
678                    self.root_exit_value(value);
679                }
680                return Ok(Some(Ended::Exited(value)));
681            }
682            // The activation boundary `Flow::Return` was added to reach.
683            // Every construct between the `RETURN` and here forwarded it
684            // untouched; this is the one consumer.
685            Flow::Return(value) => {
686                if let Some(value) = value {
687                    self.root_exit_value(value);
688                }
689                return Ok(Some(Ended::Returned(value)));
690            }
691            // A `LEAVE`/`ITERATE` that reached the very top of the
692            // program -- nothing anywhere, at any nesting depth, ever
693            // matched it. This is the exhausted-search family, 28.1 (bare
694            // `LEAVE`)/28.2 (bare `ITERATE`)/28.3 (named `LEAVE`)/28.4
695            // (named `ITERATE`). `origin.indent` already holds this family's
696            // own answer by the time it gets here -- every `Select`/`Do`
697            // frame the search walked through on the way up has already
698            // reset it to its own `static_indent` as it forwarded past
699            // (`LeaveOrigin`'s own doc comment has the rule: it is **not**
700            // always zero, only when every popped
701            // frame along the way happened to sit at top level). 28.5 (a
702            // named `ITERATE` that *did* match something, just not a loop)
703            // is a different family, raised where the match was found, in
704            // `Select`/`Do`'s own arms, and never reaches here.
705            Flow::Leave(name, origin) => {
706                self.record_leave_failure(&origin);
707                let raised = match name {
708                    None => raised_leave_no_loop(),
709                    Some(n) => raised_leave_no_match(code.symbols.name(n).as_bytes()),
710                };
711                return Err(raised.into());
712            }
713            Flow::Iterate(name, origin) => {
714                self.record_leave_failure(&origin);
715                let raised = match name {
716                    None => raised_iterate_no_loop(),
717                    Some(n) => raised_iterate_no_match(code.symbols.name(n).as_bytes()),
718                };
719                return Err(raised.into());
720            }
721        }
722        Ok(None)
723    }

725    /// One instruction's own work, entered by [`crate::ir::Op::Exec`] from
726    /// inside the [`crate::ir::Op::Clause`] region that already opened the
727    /// clause, and what the driver does next.
728    pub(crate) fn exec_instruction(
729        &mut self,
730        code: &Code<'_>,
731        index: usize,
732        instruction: &Instruction,
733        source: Option<&ProgramSource>,
734        first_instruction: bool,
735    ) -> Result<ExecOutcome, Failure> {
736        #[cfg(test)]
737        if let Some(parked) = self.scripted_park() {
738            return Ok(parked);
739        }
740        match &instruction.kind {
741            InstructionKind::Guard(guard) => self.exec_guard(code, index, guard, None),
742            InstructionKind::Reply { expression } => {
743                self.exec_reply(code, expression.as_ref().map(|e| code.body.expr(*e)))
744            }
745            // `FORWARD` and its options. See `exec_forward`.
746            InstructionKind::Forward(forward) => self.exec_forward(code, forward),
747            InstructionKind::Command { .. } | InstructionKind::Address(_)
748                if self.blocked_command_woken(index) =>
749            {
750                self.end_blocked_command(index, instruction, source)
751            }
752            // A command clause: the string is evaluated, handed to the
753            // `ADDRESS` environment in force, and its return code settles
754            // `RC`, `.RS` and any condition. See `command.rs`.
755            //
756            // The expression is `Option` because `opt_expr` builds it, but a
757            // clause reaching the command fallback has a term in it -- an empty
758            // one is a null clause and is dropped before this. For example,
759            // `;`, a blank line and `;;` each run nothing at all, and the null
760            // it would need would fault. The assertion is the tripwire rather
761            // than a sentence claiming it cannot happen.
762            InstructionKind::Command { expression } => {
763                debug_assert!(
764                    expression.is_some(),
765                    "a command clause reached execution with no expression"
766                );
767                match expression {
768                    Some(expression) => self.exec_command(
769                        code,
770                        index,
771                        instruction,
772                        source,
773                        code.body.expr(*expression),
774                        None,
775                        None,
776                    ),
777                    None => Ok(ExecOutcome::Done(Flow::Next)),
778                }
779            }
780            // `ADDRESS`: the forms that only name an environment -- the
781            // constant `ADDRESS env`, the computed `ADDRESS VALUE expr` and
782            // the bare toggle -- go to `exec_address`. `ADDRESS env command`
783            // runs one command against that name and writes neither half of
784            // the activation's pair, measured. `WITH`'s redirection is what
785            // is still owed.
786            InstructionKind::Address(address) => {
787                if let Some(command) = &address.command {
788                    let environment = match (&address.environment, &address.dynamic) {
789                        (Some(name), _) => name.to_vec(),
790                        (None, Some(expression)) => {
791                            let value = self.eval(code, code.body.expr(*expression))?;
792                            self.roots.activity_mut().push_temp(value);
793                            let value = self.required_string_value(value)?;
794                            self.to_text(value).into_owned()
795                        }
796                        (None, None) => Vec::new(),
797                    };
798                    // **The configuration is this command's alone.** The
799                    // arm that carries a command never sets the address,
800                    // so it stores nothing under the name -- measured, a
801                    // later command in the same program is not redirected
802                    // and the stem it filled keeps its one line.
803                    return self.exec_command(
804                        code,
805                        index,
806                        instruction,
807                        source,
808                        code.body.expr(*command),
809                        Some(&environment),
810                        address.io.as_deref(),
811                    );
812                }
813                self.exec_address(code, address)?;
814                Ok(ExecOutcome::Done(Flow::Next))
815            }
816            _ => self
817                .exec_flow(code, index, instruction, source, first_instruction)
818                .map(ExecOutcome::Done),
819        }
820    }

822    /// [`Interp::exec_instruction`] for the instructions that always run to
823    /// their `Flow`.
824    fn exec_flow(
825        &mut self,
826        code: &Code<'_>,
827        index: usize,
828        instruction: &Instruction,
829        source: Option<&ProgramSource>,
830        first_instruction: bool,
831    ) -> Result<Flow, Failure> {
832        match &instruction.kind {
833            InstructionKind::Say { expression } => {
834                let value = match expression {
835                    Some(expression) => Some(self.eval(code, code.body.expr(*expression))?),
836                    None => None,
837                };
838                self.say_evaluated(value)?;
839                Ok(Flow::Next)
840            }
841
842            InstructionKind::Assignment { target, value } => {
843                let value = self.eval(code, code.body.expr(*value))?;
844                // `None`: this engine resolves nothing ahead of time, so the
845                // write resolves its own slot the way it always has.
846                self.assign_evaluated(code, *target, value, None, false)?;
847                Ok(Flow::Next)
848            }
849
850            // A simple variable back to unset (`Interp::clear_variable`, added
851            // expressly for this and never `ObjRef::NIL`, which is a value
852            // and not an absence -- `x = .nil; say x` and `y = .nil; drop y;
853            // say y` render differently, measured in `drop_variable`'s own
854            // doc comment), a whole stem, one tail, or the `(v)` indirect
855            // form. See `drop_variable`.
856            InstructionKind::Drop { variables } => {
857                for variable in variables {
858                    self.drop_variable(code, variable)?;
859                }
860                Ok(Flow::Next)
861            }
862
863            // `NUMERIC DIGITS`/`FUZZ`/`FORM`, every spelling `NumericSetting`
864            // has. See `exec_numeric`.
865            InstructionKind::Numeric {
866                setting,
867                expression,
868            } => {
869                self.exec_numeric(
870                    code,
871                    setting,
872                    expression.as_ref().map(|e| code.body.expr(*e)),
873                )?;
874                Ok(Flow::Next)
875            }
876
877            // `TRACE`: sets the running activation's own trace mode, or
878            // raises 24.901 for
879            // the interactive-only skip-count forms. See `exec_trace`.
880            InstructionKind::Trace(setting) => {
881                self.exec_trace(code, setting)?;
882                Ok(Flow::Next)
883            }
884
885            // `EXIT`, bare or with a result. The value crosses out of the instruction loop
886            // as `Flow::Exit`, unconverted -- `Interp::exit_code_for` (`lib.rs`)
887            // is what turns it into a process exit code, and it runs once in
888            // `execute` rather than here, because a `Flow::Exit` can also
889            // come from inside a fragment (`run_fragment`'s own propagating
890            // arm), and the conversion needs nothing this loop knows
891            // that `execute` does not already have.
892            InstructionKind::Exit { expression } => {
893                let value = match expression {
894                    Some(expression) => Some(self.eval(code, code.body.expr(*expression))?),
895                    None => None,
896                };
897                self.returned_value(value, ReturnKeyword::Exit)
898            }
899
900            // A label is a traced no-op: executing a label instruction only
901            // traces it (a `Label` clause is echoed here via `Op::Clause`'s
902            // region, same as any other instruction) and does nothing else
903            // besides. `SIGNAL`/`CALL` reach a label by jumping to the
904            // instruction after it; nothing ever executes the label node for
905            // its own effect.
906            InstructionKind::Label { .. } => Ok(Flow::Next),
907
908            InstructionKind::Nop => Ok(Flow::Next),
909
910            // `INTERPRET expr`: evaluate to a string, parse it as a fragment,
911            // run it against **this** activation, through `run_fragment`. The
912            // arm is thin because `run_fragment` is where the work is.
913            InstructionKind::Interpret { expression } => {
914                let value = self.eval(code, code.body.expr(*expression))?;
915                self.exec_interpret_value(index, instruction, source, value)
916            }
917
918            // `IF`/`THEN`/`ELSE`. This arm resolves the whole construct
919            // itself rather than leaving the outer loop to fall through the
920            // flat list -- see `run_bounded`'s doc comment for why that is
921            // not optional. `false_target`'s own doc comment: "the ELSE if
922            // there is one, otherwise the instruction after the THEN
923            // branch" -- confirmed by tracing `block.rs` by hand, it is the
924            // `Else` instruction's own index when there is one, landing
925            // *on* it rather than past it.
926
927            // A pure marker: only ever reached inside `If`'s own bounded
928            // sub-loop (the true branch, right after the `IF`) or via
929            // ordinary fallthrough on the false path. Never independently
930            // dispatched for a decision of its own.
931            InstructionKind::Then => Ok(Flow::Next),
932
933            // Also a pure marker (`ast.rs`'s own doc comment: "executing an
934            // ELSE only traces"). Reached only by ordinary fallthrough on
935            // the false path -- the true path's `Goto` in the `If` arm above
936            // skips straight past it to `then_exit`, so this is never asked
937            // to decide anything.
938            InstructionKind::Else { .. } => Ok(Flow::Next),
939
940            // `SELECT`/`SELECT CASE`. Evaluates `case` at most once (if this
941            // is a `SELECT CASE`), then tests each of its own *listed*
942            // `whens` in source order by reading the `When`/`WhenCase` node
943            // directly as data (`condition`/`values`, `false_target`,
944            // `exit`) rather than dispatching through `Op::Clause`'s region
945            // -- a *listed* `When`/`WhenCase` node (one collected into this
946            // `whens` list, `ast.rs`'s own doc comment) must never be
947            // independently stepped for a decision of its own, only ever
948            // run past inside a bounded sub-loop.
949            InstructionKind::Otherwise => Ok(Flow::Next),
950            InstructionKind::Catch(_) => Err(Loud::instruction(&instruction.kind).into()),
951
952            // `DO`/`LOOP`, `IF` and `SELECT` are **not** reachable here, and
953            // neither is a `WHEN`/`WHEN CASE`: every one is a listed branch of
954            // its `SELECT`, which tests it without stepping it, since the
955            // parser refuses a `WHEN` anywhere else with 9.1.
956            InstructionKind::Do(_)
957            | InstructionKind::Loop(_)
958            | InstructionKind::If { .. }
959            | InstructionKind::Select { .. }
960            | InstructionKind::When { .. }
961            | InstructionKind::WhenCase { .. } => Err(Loud::instruction(&instruction.kind).into()),
962
963            // `LEAVE`/`ITERATE`, bare or by name. Resolves to
964            // data, not a failure (`Flow::Leave`'s own doc comment): whether
965            // this instruction's own name matches anything is answered by
966            // whichever `Do`/`Select` (or `run_activation`'s own top level,
967            // if none does) inspects the `Flow` this returns, never here.
968            InstructionKind::Leave { name } => Ok(Flow::Leave(
969                *name,
970                Box::new(self.leave_origin(code, index, source, instruction)),
971            )),
972            InstructionKind::Iterate { name } => Ok(Flow::Iterate(
973                *name,
974                Box::new(self.leave_origin(code, index, source, instruction)),
975            )),
976
977            // `END`. `Select`'s two non-7.3 closings (`OTHERWISE` present)
978            // are reached only by that `OTHERWISE`'s own ordinary body
979            // fallthrough and do nothing. `EndStyle::Select`'s own doc
980            // comment: "Reaching this END at run time is error 7.3, because
981            // every WHEN was false" -- `Select`'s own arm sends every other
982            // path around this instruction.
983            InstructionKind::End { closes, .. } => {
984                let closes = closes
985                    .as_ref()
986                    .expect("an End's closes is only None while its body is still being assembled");
987                match closes.style {
988                    EndStyle::Select => Err(raised_select_no_when().into()),
989                    EndStyle::Otherwise
990                    | EndStyle::LabeledOtherwise
991                    | EndStyle::Do
992                    | EndStyle::LabeledDo
993                    | EndStyle::Loop => Ok(Flow::Next),
994                }
995            }
996
997            // `CALL name`, `CALL "name"`, `CALL (expr)` and `CALL ON`/`CALL
998            // OFF`.
999            InstructionKind::Call(call) => match &**call {
1000                // `name` arrives already upcased for the symbol form and
1001                // verbatim for the quoted one (`interrexx-parse`'s own `Call`
1002                // doc). `literal` inverts into "may this search the label
1003                // table": measured, `call "SUB"` with `sub:` present is
1004                // Error 43.1 and not a call, so the quoted form bypasses the
1005                // search entirely rather than merely matching case-sensitively.
1006                interrexx_parse::Call::Named {
1007                    name,
1008                    literal,
1009                    args,
1010                } => self.exec_call(code, name, !*literal, code.body.args(args)),
1011                // `CALL (expr)`: the target is evaluated in the caller, its
1012                // value is traced, and the **verbatim** text is what the
1013                // label search sees. Both halves are measured and they pull
1014                // in opposite directions from the quoted form: `nm = 'SUB';
1015                // call (nm)` runs `sub:`, so this form *does* search labels,
1016                // while `nm = 'sub'; call (nm)` is Error 43.1 naming `sub`,
1017                // so the value is not upcased on the way in.
1018                interrexx_parse::Call::Dynamic { target, args } => {
1019                    let value = self.eval(code, code.body.expr(*target))?;
1020                    self.exec_dynamic_call(code, value, code.body.args(args))
1021                }
1022                // `CALL ON cond NAME label` / `CALL OFF cond`. Shares every
1023                // line of its implementation with `SIGNAL ON`/`OFF` except
1024                // the one `bool` that decides how the handler runs -- see
1025                // `exec_condition_trap`, and `Trap`'s own doc comment
1026                // (`activation.rs`) for the two measured behaviours that
1027                // `bool` selects between.
1028                interrexx_parse::Call::Trap(trap) => self.exec_condition_trap(trap, true),
1029                // `CALL ns:name`, whose target is a public routine of that
1030                // namespace and nothing else.
1031                // It settles
1032                // `RESULT` exactly as `Named` does, which is why it joins the
1033                // same second half; what it does not do is search labels or
1034                // builtins, so it has no `search_labels` to pass.
1035                interrexx_parse::Call::Qualified {
1036                    namespace,
1037                    name,
1038                    args,
1039                } => {
1040                    let namespace = code.symbols.name(*namespace).as_bytes();
1041                    let name = code.symbols.name(*name).as_bytes();
1042                    self.invoke_named_call(
1043                        code,
1044                        CallResolution::Qualified { namespace },
1045                        name,
1046                        code.body.args(args),
1047                    )
1048                }
1049            },
1050
1051            // `RETURN`, bare or with a value. Unwinds to the activation
1052            // boundary; `Flow::Return`'s own doc comment has why none of the
1053            // other variants expresses that, and why the main body's own
1054            // `RETURN` ends the program.
1055            InstructionKind::Return { expression } => {
1056                let value = match expression {
1057                    Some(expression) => Some(self.eval(code, code.body.expr(*expression))?),
1058                    None => None,
1059                };
1060                self.returned_value(value, ReturnKeyword::Return)
1061            }
1062
1063            // `SIGNAL label` and `SIGNAL VALUE`.
1064            InstructionKind::Signal(signal) => match &**signal {
1065                // `name` is already upcased for a bare symbol and verbatim
1066                // for a quoted one (`interrexx-parse`'s own `Signal` doc), and
1067                // **both forms search the label table** -- unlike `CALL
1068                // "name"`, which never does, because `SIGNAL` has no
1069                // builtin/external fallback for a literal spelling to
1070                // deliberately bypass into. Measured: `signal "sub"` with
1071                // `sub:` present still raises 16.1 (case-sensitive against
1072                // the label's own upcased spelling, so the lowercase quoted
1073                // form does not match), while `signal Sub` (bare, mixed
1074                // case) and `signal "SUB"` both run it.
1075                interrexx_parse::Signal::Label(name) => self.signal_to_label(name),
1076                // `SIGNAL VALUE expr`. Its own `>K>` -- `"VALUE" => text`, at
1077                // this clause's own indent with no `+2` the way `WHILE`/
1078                // `UNTIL` carry (measured one `DO` deep: `signal value
1079                // target` traces `>K>     "VALUE" => "THERE"` at the same
1080                // indent as its own clause echo, unlike those two, which are
1081                // evaluated as part of the *enclosing* `DO`/`LOOP`'s own
1082                // step). The rendered text is then searched exactly like
1083                // `Label`'s own bytes, with **no shape check on the value at
1084                // all** -- measured, a number, an empty string and an
1085                // ordinary non-label string all raise 16.1 naming that exact
1086                // text, none of them a different error.
1087                interrexx_parse::Signal::Value(expr) => {
1088                    let value = self.eval(code, code.body.expr(*expr))?;
1089                    // Rooted here rather than inside `signal_to_value`, which
1090                    // the compiled stream reaches with the value already in a
1091                    // register: the temp is what roots it across the render
1092                    // there, and a second one would be a frame this clause
1093                    // does not own.
1094                    self.roots.activity_mut().push_temp(value);
1095                    self.signal_to_value(value)
1096                }
1097                // `SIGNAL ON cond NAME label` / `SIGNAL OFF cond`. Unlike the
1098                // two arms above it transfers no control of its own: it edits
1099                // this activation's trap table and falls through to the next
1100                // clause, and the transfer happens later, if the condition is
1101                // ever raised.
1102                interrexx_parse::Signal::Trap(trap) => self.exec_condition_trap(trap, false),
1103            },
1104
1105            // `PROCEDURE`, bare or with an `EXPOSE` list (D9r). Isolates the
1106            // callee's variable pool and aliases the exposed names back into
1107            // the pool they came from. See `exec_procedure`.
1108            InstructionKind::Procedure { variables } => {
1109                self.exec_procedure(code, variables, first_instruction)?;
1110                Ok(Flow::Next)
1111            }
1112
1113            // `EXPOSE`: binds this method's names to the receiving object's
1114            // pool for the scope the method was declared in. See
1115            // `exec_expose`.
1116            InstructionKind::Expose { variables } => {
1117                self.exec_expose(code, variables)?;
1118                Ok(Flow::Next)
1119            }
1120
1121            // `USE ARG`/`USE STRICT ARG`/`USE LOCAL`. See `exec_use`.
1122            InstructionKind::Use(use_) => {
1123                self.exec_use(code, use_, first_instruction)?;
1124                Ok(Flow::Next)
1125            }
1126
1127            // `RAISE`, in all of its forms. See `exec_raise`, whose doc
1128            // comment carries the delivery table -- which is the whole of
1129            // this instruction and is not derivable from the grammar.
1130            InstructionKind::Raise(raise) => self.exec_raise(code, raise),
1131
1132            // `PUSH`/`QUEUE line`. One arm, not two copies that can
1133            // drift: the two spellings differ only in
1134            // which end of the queue the value lands on, decided below by
1135            // which variant matched. The rendering, the `>>>` line and the
1136            // write are `Interp::queue_evaluated`'s, shared with
1137            // `crate::ir::Op::Queue`.
1138            InstructionKind::Push { expression } | InstructionKind::Queue { expression } => {
1139                let value = match expression {
1140                    Some(expression) => Some(self.eval(code, code.body.expr(*expression))?),
1141                    None => None,
1142                };
1143                let keyword = if matches!(instruction.kind, InstructionKind::Push { .. }) {
1144                    QueueKeyword::Push
1145                } else {
1146                    QueueKeyword::Queue
1147                };
1148                self.queue_evaluated(value, keyword)?;
1149                Ok(Flow::Next)
1150            }
1151
1152            // `PARSE`, in every source spelling, plus the two short forms that
1153            // are the same instruction with `UPPER` already set: `ARG
1154            // template` is `PARSE UPPER ARG template` and `PULL template` is
1155            // `PARSE UPPER PULL template`. One arm, because `interrexx-parse`
1156            // builds the identical `Parse` body for all three (its
1157            // `parse_instruction_body` takes the implied source and sets
1158            // `upper` from it), so a second arm here would be a second copy of
1159            // the dispatch and nothing else.
1160            InstructionKind::Parse(parse)
1161            | InstructionKind::Arg(parse)
1162            | InstructionKind::Pull(parse) => {
1163                self.exec_parse(code, parse, None)?;
1164                Ok(Flow::Next)
1165            }
1166
1167            // A message send as a whole clause: `q~append(1)`, `q~~append(1)`
1168            // and the message-assignment form `q[1] = 2`. See
1169            // `exec_message`.
1170            InstructionKind::Message { term, value } => self.exec_message(code, *term, *value),
1171
1172            // `OPTIONS`: the expression is evaluated, converted and traced, and
1173            // nothing else.
1174            InstructionKind::Options { expression } => {
1175                let value = self.eval(code, code.body.expr(*expression))?;
1176                self.exec_options_value(value)?;
1177                Ok(Flow::Next)
1178            }
1179
1180            other => Err(Loud::instruction(other).into()),
1181        }
1182    }

1184    /// A message send that is a clause of its own: `term` and the
1185    /// assignment form's `value`, as nodes of `code`'s body.
1186    pub(crate) fn exec_message(
1187        &mut self,
1188        code: &Code<'_>,
1189        term: ExprId,
1190        value: Option<ExprId>,
1191    ) -> Result<Flow, Failure> {
1192        let term = code.body.expr(term);
1193        let value = value.map(|value| code.body.expr(value));
1194        let ExprView::Message {
1195            target,
1196            name,
1197            super_class,
1198            args,
1199            cascade,
1200        } = term.kind()
1201        else {
1202            // `interrexx-parse` builds this variant only from a message term
1203            // (`instruction.rs`'s `message`), so nothing else can arrive;
1204            // loud rather than a panic, on the standing rule that a parser
1205            // guarantee the type system does not carry must not abort.
1206            return Err(Loud::expression(term.kind()).into());
1207        };
1208        // **`message_term` directly rather than through `Interp::eval`**,
1209        // even for the form that is an ordinary expression: `eval`'s own
1210        // `NodeKind::Message` arm turns a valueless send into 91.999, which
1211        // is the expression position's error and not this one's -- measured,
1212        // a whole-clause `.K~m` on a method ending in a bare `return` is
1213        // rc 0.
1214        let mut assigned_name;
1215        let result = match value {
1216            None => {
1217                let probe = 0u8;
1218                self.enter_eval_node(&raw const probe)?;
1219                let sent = self.message_term(
1220                    code,
1221                    &crate::dispatch::MessageTerm {
1222                        target,
1223                        name,
1224                        super_class,
1225                        args,
1226                        cascade,
1227                        assigned: None,
1228                    },
1229                );
1230                self.activity.depth -= 1;
1231                sent?
1232            }
1233            Some(value) => {
1234                assigned_name = name.to_vec();
1235                assigned_name.push(b'=');
1236                self.message_term(
1237                    code,
1238                    &crate::dispatch::MessageTerm {
1239                        target,
1240                        name: &assigned_name,
1241                        super_class,
1242                        args,
1243                        // This form is built as `KEYWORD_MESSAGE` whatever the
1244                        // term's own tilde count, so a `~~` written here is not
1245                        // a cascade.
1246                        cascade: false,
1247                        assigned: Some(value),
1248                    },
1249                )?
1250            }
1251        };
1252        let slot = self.reserved_result_slot();
1253        let frame = self.activation().frame;
1254        // **A send that produced no value drops `RESULT`** rather than
1255        // leaving the previous one in place -- the same rule a bare `return`
1256        // from a `CALL` follows. Measured: `result = 'unset'` then `.K~m`
1257        // then `symbol('RESULT')` is `LIT`.
1258        match result {
1259            Some(result) => {
1260                self.roots.activity_mut().push_temp(result);
1261                self.set_variable(frame, slot, result);
1262            }
1263            None => self.clear_variable(frame, slot),
1264        }
1265        Ok(Flow::Next)
1266    }

1268    /// `PROCEDURE`, with or without an `EXPOSE` list (D9r). Out of line:
1269    /// measured, inlined into `exec_flow` it costs `bench-programs/fibcall.rex`
1270    /// 2.4% in instructions.
1271    #[inline(never)]
1272    fn exec_procedure(
1273        &mut self,
1274        code: &Code<'_>,
1275        variables: &[VariableRef],
1276        first_instruction: bool,
1277    ) -> Result<(), Failure> {
1278        // 17.1 covers every shape but one: the first instruction executed
1279        // after an internal `CALL` or function invocation. Both halves are
1280        // needed -- every other entry fails the second, and anything after
1281        // another instruction in the same activation fails the first.
1282        let entered_by_internal_call = match self.activation().entry {
1283            Entry::InternalCall => true,
1284            Entry::TopLevel | Entry::Routine | Entry::Method => false,
1285        };
1286        if !(first_instruction && entered_by_internal_call) {
1287            return Err(Raised::procedure_out_of_place().into());
1288        }
1289
1290        // The swap at the end of this function gives the callee a frame of
1291        // its own, and an activation that already owned one would be pushing
1292        // a second onto the same stack. That is the state
1293        // `ActivityRoots::pop_slots` catches a step or two later,
1294        // by which time the instruction that caused it has returned, so the
1295        // invariant is asserted where it is established rather than where
1296        // the damage surfaces. `Activation::nested` builds the entry kind
1297        // admitted above, and it starts the callee sharing.
1298        assert!(
1299            !self.activation().owns_frame,
1300            "a PROCEDURE admitted in an activation that already owns its frame"
1301        );
1302
1303        let names = self.expose_names(code, variables)?;
1304
1305        // Resolved against the pool still in force, which is the caller's:
1306        // this activation has not swapped in a frame of its own yet.
1307        let outer = self.activation().frame;
1308        let mut bindings: Vec<(Box<[u8]>, usize, VarHome)> = Vec::with_capacity(names.len());
1309        let mut tails: Vec<(Box<[u8]>, Box<[u8]>, Vec<u8>, ObjRef, bool)> = Vec::new();
1310        for (name, traced) in names {
1311            // Whole stems alias fine -- the stem object lives in one slot, so
1312            // aliasing that slot shares the object and every measured stem
1313            // transcript falls out of it. A single tail is exposed once the
1314            // callee has a frame of its own: it is resolved and made in the
1315            // caller, and the caller's stem is found here, in the caller's
1316            // pool.
1317            let shape = shape_of(&name);
1318            if shape == NameShape::Compound {
1319                let (stem_name, key) = self.split_compound(&name)?;
1320                let home = self.read_stem(&stem_name);
1321                tails.push((name, stem_name, key, home, traced));
1322                continue;
1323            }
1324            let slot = self.slot_of(&name);
1325            // Exposing a stem uses the caller's stem variable, which creates it.
1326            if shape == NameShape::Stem && self.variable(outer, slot).is_none() {
1327                self.create_stem_variable(outer, slot, &name);
1328            }
1329            // A name the enclosing method exposed has no frame storage to
1330            // alias: its home is the object's pool, and the callee gets the
1331            // same home rather than a slot. Measured -- a class method
1332            // exposing `v` and calling `inner: procedure expose v`, which
1333            // assigns `v` -- the object variable is what changes.
1334            let target = match self.exposure(outer, slot) {
1335                Some(var) => VarHome::Instance(Box::new(var.clone())),
1336                None => VarHome::Slot(self.roots.activity().slot_ref(outer, slot)),
1337            };
1338            bindings.push((name, slot, target));
1339        }
1340
1341        // Any name that needed a fresh slot just grew the caller's frame and
1342        // was recorded in this activation's `extra`. Move those bindings
1343        // to the caller before replacing the table with the isolated one.
1344        let resolved = std::mem::take(&mut self.activation_mut().extra);
1345        if let Some(caller) = self.caller_activation_mut() {
1346            caller.extra = resolved;
1347        }
1348
1349        // Sized from the caller's *current* frame length rather than from
1350        // `plan.len()`: an exposed name may sit at an index the caller grew
1351        // into, and that same index has to address something on this side of
1352        // the alias too.
1353        let len = self.roots.activity().frame_len(outer);
1354        let inner = self.roots.push_slots(len);
1355        let mut exposed: Vec<(usize, InstanceVar)> = Vec::new();
1356        for (_, slot, target) in &bindings {
1357            match target {
1358                VarHome::Slot(target) => {
1359                    self.roots.activity_mut().alias_slot(inner, *slot, *target)
1360                }
1361                VarHome::Instance(var) => exposed.push((*slot, (**var).clone())),
1362            }
1363        }
1364
1365        // The callee's own run-time bindings start empty -- that is the
1366        // isolation -- except for exposed names the plan never saw, which
1367        // must keep resolving to the index the alias was installed at.
1368        let plan = Rc::clone(&self.activation().plan);
1369        let mut extra = ExtraNames::default();
1370        for (name, slot, _) in bindings {
1371            if plan.slot_of(&name).is_none() {
1372                extra.insert(name, slot);
1373            }
1374        }
1375
1376        let activation = self.activation_mut();
1377        activation.frame = inner;
1378        activation.owns_frame = true;
1379        activation.extra = extra;
1380        // **Replaced, not extended.** This activation inherited the caller's
1381        // exposures when it was pushed, and a `PROCEDURE` isolates the pool:
1382        // a name the caller exposed and this list does not name is an
1383        // ordinary local here. Measured -- a class method exposing `v` and
1384        // calling `inner: procedure` with no list, which assigns `v` -- the
1385        // object variable is unchanged.
1386        activation.exposed = (!exposed.is_empty())
1387            .then(|| Box::new(crate::activation::Exposures::from_list(exposed)));
1388        if let Some(cold) = activation.cold.as_deref_mut() {
1389            cold.auto_expose = None;
1390        }
1391        for (name, stem_name, key, home, traced) in tails {
1392            self.expose_tail(home, &stem_name, &key);
1393            if !traced {
1394                self.trace_exposed_tail(&name, &stem_name, &key);
1395            }
1396        }
1397        Ok(())
1398    }

1400    /// `EXPOSE`: bind every name it lists to the receiving object's variable
1401    /// pool for the scope the running method was declared in.
1402    pub(crate) fn exec_expose(
1403        &mut self,
1404        code: &Code<'_>,
1405        variables: &[VariableRef],
1406    ) -> Result<(), Failure> {
1407        let Some(identity) = self.activation().method_identity.as_ref() else {
1408            return Err(Raised::expose_outside_method().into());
1409        };
1410        let scope = identity.scope;
1411        let receiver = identity.receiver;
1412        let owner = self.pool_owner(receiver)?;
1413        for variable in variables {
1414            match variable {
1415                VariableRef::Direct(id) => {
1416                    let name = match code.plan {
1417                        Some(plan) => plan.exposed_name(*id, code.symbols),
1418                        None => code.symbols.name(*id).as_bytes().into(),
1419                    };
1420                    match simple_slot(code, *id) {
1421                        Some(slot) => self.expose_slot(slot, InstanceVar { owner, scope, name }),
1422                        None => self.bind_exposed(owner, scope, name)?,
1423                    }
1424                }
1425                // **The selector's own name is bound before its value is
1426                // read**, and the order is observable rather than tidy.
1427                // Measured: with a class-scope `LISTER` holding `'BETA'` and a
1428                // class-scope `BETA` holding `'beta-value'`, `expose (lister)`
1429                // in a third method reads `[BETA][beta-value]` -- so `LISTER`
1430                // was read out of the object's pool and `BETA` was exposed
1431                // from it. Reading the selector first, out of the frame, gets
1432                // `[BETA][BETA]`: the frame's `LISTER` is unset, so its
1433                // derived name `LISTER` is what spells the list.
1434                VariableRef::Indirect(id) => {
1435                    let name = code.symbols.name(*id).as_bytes().into();
1436                    self.bind_exposed(owner, scope, name)?;
1437                    let value = self.read_list_selector(code, *id)?;
1438                    // The conversion goes through the string request, so a
1439                    // selector holding an object spells its list from the
1440                    // conversion. For example, `zz = 1; x = .K; drop (x)` with
1441                    // a class-side `makeString` returning `'zz'` leaves
1442                    // `SYMBOL('ZZ')` at `LIT`.
1443                    let value = self.required_string_value(value)?;
1444                    let text = self.to_text(value).into_owned();
1445                    for word in split_indirect_words(&text) {
1446                        let word = validate_indirect_word(word)?;
1447                        self.bind_exposed(owner, scope, word.into())?;
1448                    }
1449                }
1450            }
1451        }
1452        Ok(())
1453    }

1455    /// Binds one name to `scope`'s pool on `owner`, for the rest of this
1456    /// activation. A simple symbol the body's plan bound a slot to goes
1457    /// straight to [`Interp::expose_slot`] instead: see [`simple_slot`].
1458    fn bind_exposed(
1459        &mut self,
1460        owner: ObjRef,
1461        scope: ObjRef,
1462        name: Rc<[u8]>,
1463    ) -> Result<(), Failure> {
1464        // A whole stem is one value in one pool entry, so it binds like any
1465        // other name; a single tail is aliasing *inside* a stem object. For
1466        // example, `expose a.1` in one class method assigning `a.1` and `a.2`
1467        // and the same in another reading them back: `[tail-one][A.2]`, so tail
1468        // 1 is shared and tail 2 is the method's own local.
1469        match shape_of(&name) {
1470            NameShape::Simple => {
1471                let slot = self.slot_of(&name);
1472                self.expose_slot(slot, InstanceVar { owner, scope, name });
1473                Ok(())
1474            }
1475            NameShape::Compound => self.expose_object_tail(owner, scope, &name),
1476            NameShape::Stem => self.expose_object_stem(InstanceVar { owner, scope, name }),
1477        }
1478    }

1481    /// object's stem variable, which creates it.
1482    #[cold]
1483    #[inline(never)]
1484    fn expose_object_stem(&mut self, var: InstanceVar) -> Result<(), Failure> {

1488        let frame = self.activation().frame;
1489        if self.variable(frame, slot).is_none() {
1490            self.create_stem_variable(frame, slot, &name);
1491        }
1492        Ok(())
1493    }

1507    /// The tail `name` spells becomes the same
1508    /// tail of `scope`'s stem on `owner`, which is made if the object has
1509    /// none, and the resolved name is traced.
1510    fn expose_object_tail(
1511        &mut self,
1512        owner: ObjRef,
1513        scope: ObjRef,
1514        name: &[u8],
1515    ) -> Result<(), Failure> {
1516        let (stem_name, key) = self.split_compound(name)?;
1517        let home = match self
1518            .pools_of(owner)
1519            .and_then(|pools| pools.get(scope, &stem_name))
1520        {
1521            Some(stem) => stem,
1522            None => {
1523                let stem = self.empty_stem(&stem_name);
1524                self.set_pool_variable(owner, scope, &stem_name, stem);
1525                stem
1526            }
1527        };
1528        self.expose_tail(home, &stem_name, &key);
1529        self.trace_exposed_tail(name, &stem_name, &key);
1530        Ok(())
1531    }

1543    /// The object whose [`interrexx_core::ScopePools`] a send to `receiver` binds
1544    /// into.
1545    pub(crate) fn pool_owner(&mut self, receiver: ObjRef) -> Result<ObjRef, Failure> {
1546        if !self.heap.is_class(receiver) {
1547            if matches!(
1548                self.heap.get(receiver).map(|object| &object.body),
1549                Some(Body::Instance { .. })
1550            ) {
1551                return Ok(receiver);
1552            }
1553            return Err(Loud::expose_receiver().into());
1554        }
1555        if let Some(owner) = self.class_variables.get(&receiver) {
1556            return Ok(*owner);
1557        }
1558        // The pools of the class *object*, whose own class is its metaclass --
1559        // what `.K~class` answers. This object is storage and never a value a
1560        // program holds.
1561        let holds = self.classes().class_of(receiver);
1562        let behaviour = self.classes().instance_behaviour_handle(holds);
1563        let owner = self.alloc_with(
1564            BehaviourId::OBJECT,
1565            Body::Instance {
1566                class: holds,
1567                behaviour,
1568                name: None,
1569                pools: ScopePools::new(),
1570                own: None,
1571                native: None,
1572            },
1573        );
1574        // Held by the class before anything else can allocate, which is the
1575        // rule `.environment` and `.local` are created under.
1576        self.class_owns(receiver, owner);
1577        self.class_variables.insert(receiver, owner);
1578        Ok(owner)
1579    }

1581    /// Every name one `PROCEDURE EXPOSE` list names, in source order, each
1582    /// with whether its `>C>` line has already been traced.
1583    fn expose_names(
1584        &mut self,
1585        code: &Code<'_>,
1586        variables: &[VariableRef],
1587    ) -> Result<Vec<(Box<[u8]>, bool)>, Failure> {
1588        let mut names: Vec<(Box<[u8]>, bool)> = Vec::new();
1589        for variable in variables {
1590            match variable {
1591                VariableRef::Direct(id) => {
1592                    names.push((code.symbols.name(*id).as_bytes().into(), false));
1593                }
1594                VariableRef::Indirect(id) => {
1595                    // The selector itself, then the names its value spells.
1596                    let name: Box<[u8]> = code.symbols.name(*id).as_bytes().into();
1597                    let value = if shape_of(&name) == NameShape::Compound {
1598                        let (stem_name, key) = self.split_compound(&name)?;
1599                        self.trace_exposed_tail(&name, &stem_name, &key);
1600                        names.push((name, true));
1601                        self.procedure_selector_value(code, *id, &names)?
1602                    } else {
1603                        names.push((name, false));
1604                        self.read_list_selector(code, *id)?
1605                    };
1606                    let value = self.required_string_value(value)?;
1607                    let text = self.to_text(value).into_owned();
1608                    for word in split_indirect_words(&text) {
1609                        names.push((validate_indirect_word(word)?.into(), false));
1610                    }
1611                }
1612            }
1613        }
1614        Ok(names)
1615    }

1617    /// A compound `PROCEDURE EXPOSE` selector's value as the procedure's own
1618    /// frame reads it (the read comes after
1619    /// exposing): a tail variable and the compound itself have a value only
1620    /// through a name `exposed` already holds, and are unset otherwise, with
1621    /// the `NOVALUE` check and `TRACE I` lines of an ordinary read.
1622    fn procedure_selector_value(
1623        &mut self,
1624        code: &Code<'_>,
1625        id: SymbolId,
1626        exposed: &[(Box<[u8]>, bool)],
1627    ) -> Result<ObjRef, Failure> {
1628        let spelling = code.symbols.name(id).as_bytes().to_vec();
1629        let compound = CompoundName::split(&String::from_utf8_lossy(&spelling));
1630        let visible = |name: &[u8]| exposed.iter().any(|(held, _)| **held == *name);
1631        let mut key = Vec::new();
1632        for (index, piece) in compound.tails.iter().enumerate() {
1633            if index > 0 {
1634                key.push(b'.');
1635            }
1636            match piece {
1637                TailPiece::Constant(text) => key.extend_from_slice(text),
1638                TailPiece::Variable { name, .. } if visible(name) => {
1639                    let value = self.read_by_name(name);
1640                    let value = self.required_string_value(value)?;
1641                    self.write_text(value, &mut key);
1642                }
1643                TailPiece::Variable { name, .. } => key.extend_from_slice(name),
1644            }
1645        }
1646        let mut resolved = compound.stem.to_vec();
1647        resolved.extend_from_slice(&key);
1648        if self.tracing_intermediates() {
1649            let indent = self.activity.clause_state.current_value_indent;
1650            self.trace_compound_name(indent, &spelling, &resolved);
1651        }
1652        let mut reachable = visible(&compound.stem);
1653        for (held, _) in exposed {
1654            if reachable {
1655                break;
1656            }
1657            if shape_of(held) == NameShape::Compound {
1658                let (stem_name, held_key) = self.split_compound(held)?;
1659                reachable = stem_name == compound.stem && held_key == key;
1660            }
1661        }
1662        let (value, novalue) = if reachable {
1663            self.stem_get_at(&compound.stem, None, &key)
1664        } else {
1665            let derived = self.text(&resolved);
1666            (self.novalue_answer(derived), Novalue::Unset)
1667        };
1668        self.novalue_check(novalue, value)?;
1669        if self.tracing_intermediates() {
1670            let indent = self.activity.clause_state.current_value_indent;
1671            let text = self.traced_string_value_text(value)?;
1672            self.trace_variable(indent, &spelling, &text, value);
1673        }
1674        Ok(value)
1675    }

1677    /// Everything one `SAY` does once its expression has been evaluated:
1678    /// `>>>`, then the line itself.
1679    #[inline]
1680    pub(crate) fn say_evaluated(&mut self, value: Option<ObjRef>) -> Result<(), Failure> {
1681        let line = match value {
1682            Some(value) => {
1683                self.roots.activity_mut().push_temp(value);
1684                let value = self.required_string_value(value)?;
1685                // With no trace, exit or route to read the line, its bytes go
1686                // from the value to the buffer, and no third copy of a large
1687                // value is made on the way.
1688                if !self.traced_mode().results
1689                    && self.exits.is_none()
1690                    && self.output_route()?.is_none()
1691                {
1692                    let rendered = self.render(value);
1693                    let mut out = std::mem::take(&mut self.out);
1694                    let text = rendered.text(self);
1695                    // Room for the newline too: growing a buffer the line
1696                    // filled exactly can move it, and a host whose
1697                    // reallocation copies then holds the line twice.
1698                    out.reserve(text.len() + 1);
1699                    out.extend_from_slice(text);
1700                    self.out = out;
1701                    self.write_out_line(b"");
1702                    return self.say_written();
1703                }
1704                self.to_text(value).to_vec()
1705            }
1706            None => Vec::new(),
1707        };
1708        self.trace_result_before_effect(self.activity.clause_state.current_value_indent, &line)?;
1709        // The I/O exit's first pass.
1710        if self.exits.is_some() && self.say_exit(&line)? {
1711            return Ok(());
1712        }
1713        // `.OUTPUT` gets the line as a `SAY` message and its reply is dropped;
1714        // an entry
1715        // that is missing **or holds `.nil`** takes the buffer instead.
1716        //
1717        // The `.nil` test belongs here rather than in `resolve_stream`, and
1718        // the asymmetry is measured: `.local~output = .nil` then `say` then
1719        // `lineout(,)` prints the `SAY` line and *then* fails 97.1 on the
1720        // builtin, because the stream lookup has no such test and sends to the
1721        // `.nil` it found.
1722        // **The direct write when nothing has redirected `.OUTPUT`**, which is
1723        // the same bytes the monitor would forward and none of the cost:
1724        // measured, delivering every `SAY` is 16,360 instructions a line and
1725        // puts `sayloop` at 9.63x. A match guard cannot ask, because a guard
1726        // may not borrow `self` mutably.
1727        match self.output_route()? {
1728            Some(route) => {
1729                let argument = self.text_built(line);
1730                // **Rooted before the send.** The line can be long enough to
1731                // take the owned-`Bytes` path, and the send allocates: without
1732                // this the collect-on-every-allocation gate fails `a live
1733                // value` at `dispatch.rs:1465`. Measured, adding it takes the
1734                // library suite from 788 to 790.
1735                self.roots.activity_mut().push_temp(argument);
1736                let caller = self.caller();
1737                pinned!(
1738                    self,
1739                    crate::pinning::PinKind::OutputWrapper,
1740                    self.send_message(route, crate::dispatch::SAY, None, &[Some(argument)], caller)
1741                )?;
1742            }
1743            _ => {
1744                self.write_out_line(&line);
1745                return self.say_written();
1746            }
1747        }
1748        Ok(())
1749    }

1760    /// `GUARD ON`/`GUARD OFF`, with or without a `WHEN` expression.
1761    /// A reserve another activity holds the lock for, and a `WHEN` that is
1762    /// false, answer a park; the instruction's op runs it again once woken,
1763    /// and it goes on from the wait.
1764    #[inline(never)]
1765    fn exec_guard(
1766        &mut self,
1767        code: &Code<'_>,
1768        index: usize,
1769        guard: &Guard,
1770        compiled: Option<(&crate::ir::Chunk, RegFrame<'_>)>,
1771    ) -> Result<ExecOutcome, Failure> {
1772        use crate::guards::{GuardExec, GuardExecWait};
1773        let Some(key) = crate::guards::GuardKey::of(self.activation()) else {
1774            return Err(Raised::guard_outside_method().into());
1775        };
1776        #[derive(Clone, Copy)]
1777        enum Step {
1778            Reserve,
1779            Evaluate,
1780            Done,
1781        }
1782        let reserved = if guard.condition.is_some() {
1783            Step::Evaluate
1784        } else {
1785            Step::Done
1786        };
1787        let (mut exec, mut step) = match self.take_guard_exec(index) {
1788            Some(exec) => {
1789                let step = match exec.wait {
1790                    GuardExecWait::Reserve => {
1791                        debug_assert!(self.granted(key), "a GUARD woke without its lock");
1792                        self.activation_mut().flags.set_reserved(true);
1793                        reserved
1794                    }
1795                    GuardExecWait::When { reacquire: true } => Step::Reserve,
1796                    GuardExecWait::When { reacquire: false } => Step::Evaluate,
1797                };
1798                (exec, step)
1799            }
1800            None => {
1801                park_point!(
1802                    self,
1803                    match guard.condition {
1804                        Some(_) => Some(crate::pinning::ParkKind::GuardWhen),
1805                        None => guard.on.then_some(crate::pinning::ParkKind::GuardOn),
1806                    }
1807                );
1808                let watched = self.watch_guard_variables(&guard.watched);
1809                let exec = GuardExec {
1810                    activation: self.activation().id,
1811                    index,
1812                    wait: GuardExecWait::Reserve,
1813                    watched,
1814                };
1815                let step = if guard.on {
1816                    Step::Reserve
1817                } else {
1818                    self.guard_off(key);
1819                    reserved
1820                };
1821                (exec, step)
1822            }
1823        };
1824        loop {
1825            step = match step {
1826                Step::Reserve => match self.guard_on(key) {
1827                    Ok(true) => reserved,
1828                    Ok(false) => {
1829                        exec.wait = GuardExecWait::Reserve;
1830                        self.activity.guard_exec = Some(Box::new(exec));
1831                        return Ok(ExecOutcome::Park(crate::scheduler::ParkReason::Guard(key)));
1832                    }
1833                    Err(failure) => {
1834                        self.end_guard_exec(exec);
1835                        return Err(failure);
1836                    }
1837                },
1838                Step::Evaluate => {
1839                    let Some(condition) = &guard.condition else {
1840                        break;
1841                    };
1842                    self.activity.guard_posted = false;
1843                    let halted = std::mem::take(&mut self.activity.woken_by_halt);
1844                    let frame = self.roots.activity_mut().push_frame();
1845                    let trace = ConditionTrace::Keyword(
1846                        self.activity.clause_state.current_value_indent,
1847                        "WHEN",
1848                    );
1849                    // The chunk's side-table ops where it compiled them and
1850                    // `TRACE` shows no intermediates, as a loop condition's.
1851                    let holds = match compiled {
1852                        Some((chunk, registers)) => self.loop_condition(
1853                            code,
1854                            chunk,
1855                            registers,
1856                            index,
1857                            code.body.expr(*condition),
1858                            trace,
1859                            raised_guard_not_logical,
1860                        ),
1861                        None => self.eval_condition(
1862                            code,
1863                            code.body.expr(*condition),
1864                            trace,
1865                            raised_guard_not_logical,
1866                        ),
1867                    };
1868                    self.roots.activity_mut().pop_frame(frame);
1869                    match holds {
1870                        Ok(true) => Step::Done,
1871                        Ok(false) => {
1872                            let reacquire = self.activation().flags.reserved();
1873                            self.guard_off(key);
1874                            if !self.activity.guard_posted {
1875                                // A `CALL ON HALT` trap queues, and the wait
1876                                // parks again.
1877                                if halted && let Err(failure) = self.take_requested_halt() {
1878                                    self.end_guard_exec(exec);
1879                                    return Err(failure);
1880                                }
1881                                exec.wait = GuardExecWait::When { reacquire };
1882                                self.activity.guard_exec = Some(Box::new(exec));
1883                                return Ok(ExecOutcome::Park(
1884                                    crate::scheduler::ParkReason::GuardWhen,
1885                                ));
1886                            }
1887                            if reacquire {
1888                                Step::Reserve
1889                            } else {
1890                                Step::Evaluate
1891                            }
1892                        }
1893                        Err(failure) => {
1894                            self.end_guard_exec(exec);
1895                            return Err(failure);
1896                        }
1897                    }
1898                }
1899                Step::Done => break,
1900            };
1901        }
1902        self.end_guard_exec(exec);
1903        Ok(ExecOutcome::Done(Flow::Next))
1904    }

1941    /// `REPLY`), then the split. The walked instruction and `Op::ExecWith`
1942    /// both run it.
1943    fn exec_reply_value(&mut self, value: Option<ObjRef>) -> Result<ExecOutcome, Failure> {
1944        // The rooting and the `>>>` line are `RETURN`'s, for the same reason
1945        // and at the same indent: the expression evaluation is shared.
1946        if let Some(value) = value {
1947            self.roots.activity_mut().push_temp(value);
1948            if let Some(rendered) = self.result_text(value)? {
1949                self.trace_result(self.activity.clause_state.current_value_indent, &rendered);
1950            }
1951        }
1952        if self.activation().reply != ReplyState::None {
1953            return Err(Raised::reply_twice().into());
1954        }
1955        // Past its driver's level the only pin is this `Op::Exec`'s or
1956        // `Op::ExecWith`'s own.
1957        if self.activity.pin_depth != self.activity.driver_pins + 1 {
1958            immovable_reply!(self);
1959            return Err(Loud::immovable_reply().into());
1960        }
1961        let activation = self.activation_mut();
1962        activation.reply = ReplyState::Owed;
1963        activation.replied_a_value = value.is_some();
1964        Ok(ExecOutcome::Split(value))
1965    }

1967    /// `FORWARD`, with any of `TO`, `MESSAGE`, `CLASS`, `ARGUMENTS`, `ARRAY`
1968    /// and `CONTINUE`, up to its send. Kept out of line so that
1969    /// [`Interp::exec_instruction`], which every [`crate::ir::Op::Exec`]
1970    /// instruction enters, does not carry its frame.
1971    #[inline(never)]
1972    fn exec_forward(&mut self, code: &Code<'_>, forward: &Forward) -> Result<ExecOutcome, Failure> {
1973        let Some(identity) = self.activation().method_identity.as_ref() else {
1974            return Err(Raised::forward_outside_method().into());
1975        };
1976        let receiver = identity.receiver;
1977        let own_name = Rc::clone(&identity.name);
1978        let indent = self.activity.clause_state.current_value_indent;
1979
1980        // The option order is observable in the trace: `TO`, `MESSAGE`,
1981        // `CLASS`, then whichever of `ARGUMENTS` and `ARRAY` is present. Under
1982        // `trace i`, `forward to (t) message('OTHER') array(1,2)` emits `>K>
1983        // "TO"`, `>K> "MESSAGE"` and `>K> "ARRAY"` in that order, with the
1984        // `ARRAY` items' own `>A>` lines ahead of its keyword line.
1985        let target = match &forward.to {
1986            None => receiver,
1987            Some(expr) => self.forward_keyword(code, code.body.expr(*expr), "TO")?,
1988        };
1989        let message = match &forward.message {
1990            None => own_name,
1991            Some(expr) => {
1992                let value = self.forward_keyword(code, code.body.expr(*expr), "MESSAGE")?;
1993                let text = self.required_string_value(value)?;
1994                Rc::from(self.to_text(text).to_ascii_uppercase())
1995            }
1996        };
1997        let start_scope = match &forward.class {
1998            None => None,
1999            Some(expr) => {
2000                let value = self.eval(code, code.body.expr(*expr))?;
2001                self.roots.activity_mut().push_temp(value);
2002                self.forward_class(value)?;
2003                Some(value)
2004            }
2005        };
2006
2007        let (mut values, mark) = self.take_value_buffer();
2008        let evaluated = self.forward_arguments(code, forward, &mut values, mark);
2009        self.prepare_forward(
2010            ForwardSend {
2011                target,
2012                message,
2013                start_scope,
2014                mark,
2015                continue_: forward.continue_,
2016                indent,
2017            },
2018            values,
2019            evaluated,
2020        )
2021    }

2081    fn prepare_forward(
2082        &mut self,
2083        send: ForwardSend,
2084        values: Vec<Option<ObjRef>>,
2085        evaluated: Result<(), Failure>,
2086    ) -> Result<ExecOutcome, Failure> {
2087        // A non-continuing FORWARD cannot answer a sender already answered
2088        // by REPLY. Ask after the options, before committing to the send.
2089        let owed = evaluated.and_then(|()| self.forward_after_reply(send.continue_));
2090        // Conditions from a non-continuing send bypass this level's traps.
2091        if owed.is_ok() && !send.continue_ {
2092            self.activation_mut().flags.set_forwarded(true);
2093        }
2094        if let Err(failure) =
2095            owed.and_then(|()| self.validate_scope_override(send.target, send.start_scope))
2096        {
2097            self.give_value_buffer(values, send.mark);
2098            return Err(failure);
2099        }
2100        self.activity.value_buffer = values;
2101        Ok(ExecOutcome::Forward(Box::new(send)))
2102    }

2295        }

2299    /// `ARGUMENTS expr`'s value through the array request,
2300    /// appended to `values`.
2301    fn forward_arguments_converted(
2302        &mut self,
2303        value: ObjRef,
2304        values: &mut Vec<Option<ObjRef>>,
2305    ) -> Result<(), Failure> {
2306        match self.forward_arguments_conversion(value) {
2307            Conversion::Array => {
2308                let slots = self.array_slots_of(value).unwrap_or_default();

2310            }
2311            Conversion::Lines => {
2312                let text = self.to_text(value).into_owned();
2313                for line in makearray_lines(&text) {
2314                    self.push_converted_argument(line, values);
2315                }
2316            }
2317            Conversion::Tails => {
2318                for tail in self.stem_assigned_tails(value) {

2323            Conversion::Requested => match self.single_dimension_request(value)? {
2324                Some((_, store)) => {

2327                }
2328                None => return Err(Raised::forward_arguments().into()),
2329            },
2330        }

2353    /// Which arm of the array request this value takes.
2354    fn forward_arguments_conversion(&self, value: ObjRef) -> Conversion {
2355        match value.decode() {
2356            Decoded::Nil => return Conversion::Refused,
2357            // An integer and a number both
2358            // hand the work to their string value's.
2359            Decoded::SmallInt(_) | Decoded::Text(_) => return Conversion::Lines,
2360            Decoded::Heap { .. } => {}
2361        }
2362        match self.heap.get(value).map(|object| &object.body) {
2363            // A class object is a primitive with no `makeArray` of its own,
2364            // so the array request stops at `.nil`.
2365            Some(Body::Class { .. }) => Conversion::Refused,
2366            // A multi-dimensional array shares `.nil`'s raise rather
2367            // than converting.
2368            Some(Body::Array { .. }) => {
2369                if self.is_multi_dimensional_array(value) {
2370                    Conversion::Refused
2371                } else {
2372                    Conversion::Array
2373                }
2374            }
2375            Some(Body::Text { .. } | Body::Num { .. }) => Conversion::Lines,
2376            // A stem converts to its assigned
2377            // tails and never the default: measured, `a. = 'dflt'` with no
2378            // tail assigned forwards no arguments at all.
2379            Some(Body::Stem { .. }) => Conversion::Tails,
2380            // The array request sends `REQUEST('ARRAY')` for a non-primitive,
2381            // which looks `MAKEARRAY` up in the behaviour and sends it, and
2382            // otherwise answers `.nil`.
2383            Some(Body::Instance { .. }) => Conversion::Requested,
2384            // **Not the referent's conversion**, which is what
2385            // `~request('ARRAY')` answers and is a different route: the array
2386            // request looks `MAKEARRAY` up in the receiver's own behaviour
2387            // rather than sending it, and a reference's behaviour holds no such
2388            // name, so the lookup fails and `.nil` is the answer. For example,
2389            // `forward arguments (>v)` over a `v` holding `'val'` is `98.946`,
2390            // where the same instruction over `v` itself is rc 0 -- so this arm
2391            // may not chase the way the string conversion beside it does.
2392            Some(Body::VarRef(_)) => Conversion::Refused,
2393            Some(Body::Native(_) | Body::WeakRef(_) | Body::Cell(..)) | None => {
2394                Conversion::Requested
2395            }
2396        }
2397    }

2415    /// Everything a `RETURN` or an `EXIT` does once its expression has been
2416    /// evaluated: its `>>>`, and the `Flow` that leaves the activation.
2417    pub(crate) fn returned_value(
2418        &mut self,
2419        value: Option<ObjRef>,
2420        keyword: ReturnKeyword,
2421    ) -> Result<Flow, Failure> {
2422        if let Some(value) = value {
2423            self.roots.activity_mut().push_temp(value);
2424            if let Some(rendered) = self.result_text(value)? {
2425                self.trace_result(self.activity.clause_state.current_value_indent, &rendered);
2426            }
2427        }
2428        // **LEGALITY.** A `REPLY` has already answered the sender, so a value
2429        // here has nobody to go to whatever activity the rest of the body runs
2430        // on: 98.936 for `RETURN`, 98.937 for `EXIT`. Both are asked **after**
2431        // the trace line above, which is the order of the checks: `RETURN`
2432        // evaluates its expression first and only then checks. `EXIT` checks in
2433        // the same order. The bare form of either is legal after a reply:
2434        // `reply 'v'` then `say 'tail'` then `return` is rc 0 with both lines
2435        // printed and an empty stderr.

2437            ReturnKeyword::Return => {

2439                    return Err(Raised::return_after_reply().into());
2440                }
2441                Ok(Flow::Return(value))
2442            }
2443            ReturnKeyword::Exit => {
2444                self.exit_after_reply(value)?;
2445                Ok(Flow::Exit(value))
2446            }
2447        }
2448    }

2451    /// that is not an internal call (`EXIT` walks up to it). An `EXIT` in a
2452    /// `CALL ON` handler run at the `REPLY` clause's end ends the method
2453    /// before its split, and its result replaces the reply's, so the sender

2472    /// Everything a `PUSH` or a `QUEUE` does once its expression has been
2473    /// evaluated: the `>>>` line, and the line itself onto one end of the
2474    /// queue.
2475    pub(crate) fn queue_evaluated(
2476        &mut self,
2477        value: Option<ObjRef>,
2478        keyword: QueueKeyword,
2479    ) -> Result<(), Failure> {
2480        let line = match value {
2481            Some(value) => {
2482                self.roots.activity_mut().push_temp(value);
2483                // The same string request again, so the `>>>` below traces
2484                // the conversion: measured, `trace r` over `push .K` with a
2485                // class-side `makeString` returning `'pv'` prints
2486                // `>>>   "pv"` and `pull` reads `PV`.
2487                let value = self.required_string_value(value)?;
2488                self.to_text(value).to_vec()
2489            }
2490            None => Vec::new(),
2491        };
2492        self.trace_result_before_effect(self.activity.clause_state.current_value_indent, &line)?;
2493        // The queue exit's first pass (`Activity::queue`).
2494        if self.exits.is_some() && self.push_exit(&line, keyword == QueueKeyword::Push)? {
2495            return Ok(());
2496        }
2497        let Some(queue) = self.local_route(b"STDQUE")? else {
2498            match keyword {
2499                QueueKeyword::Push => self.queue.push(line),
2500                QueueKeyword::Queue => self.queue.queue(line),
2501            }
2502            return Ok(());
2503        };
2504        let message: &[u8] = match keyword {
2505            QueueKeyword::Push => b"PUSH",
2506            QueueKeyword::Queue => b"QUEUE",
2507        };
2508        let frame = self.roots.activity_mut().push_frame();
2509        self.roots.activity_mut().push_temp(queue);
2510        let line = self.text(&line);
2511        self.roots.activity_mut().push_temp(line);
2512        let caller = self.caller();
2513        let sent = pinned!(
2514            self,
2515            crate::pinning::PinKind::PullWrapper,
2516            self.send_message(queue, message, None, &[Some(line)], caller)
2517        );
2518        self.roots.activity_mut().pop_frame(frame);
2519        sent.map(drop)
2520    }

2571    /// Opens a stepped clause of `code`: everything
2572    /// [`crate::ir::Op::Clause`] owes before the clause's own work
2573    /// runs.
2574    #[inline(always)]
2575    pub(crate) fn enter_stepped_clause(
2576        &mut self,
2577        code: &Code<'_>,
2578        index: usize,
2579        instruction: &Instruction,
2580        source: Option<&ProgramSource>,
2581        position: (bool, crate::ir::ClausePosition),
2582        counted: crate::clause::DeadlineCounted,
2583    ) -> SteppedClause {
2584        // `DATE`/`TIME`'s per-clause clock cache (`activation.rs`'s own doc on
2585        // `Activation::clock_stale`) is invalidated **unconditionally, once per
2586        // call, on whichever activation is executing right now**, so the
2587        // timestamp is invalidated right after each instruction executes. This
2588        // is that same one place every flat instruction position -- including
2589        // one nested inside `IF`/ `SELECT`/`DO`/`INTERPRET`, per this
2590        // function's own doc, and including a callee's own body reached through
2591        // `resolve_and_run_ call` -- passes through, so a clause's first clock
2592        // read gets a fresh one, any further read of the same clause sees the
2593        // value that read cached, and a callee's own instructions invalidate
2594        // only the callee's own cache rather than reaching back into the
2595        // caller's. The cached *value* itself (`Activation::cached_clock`) is
2596        // left untouched here -- `TIME('R')`'s own lazy reset needs it still
2597        // readable one call later, and clearing it here is what an earlier
2598        // version of this did instead.
2599        let activation = self.activation_mut();
2600        activation.clock_stale = true;
2601        // `>I>`'s own "am I still on the first instruction" decay
2602        // (`TraceEntry`, `activation.rs`), spent **here** and not in
2603        // `run_activation`'s loop. The flag is cleared at the bottom of the
2604        // instruction loop, and an `IF`'s then-clause, a `DO` body clause and a
2605        // fragment's clauses are each a separate instruction in that loop --
2606        // where here they are nested inside their enclosing clause's step. This
2607        // function is the one place a clause is stepped at any depth, so it is
2608        // the only site that counts them all. Before the move, `if 1=1 then
2609        // trace l` as a routine's first clause announced the pair here where it
2610        // should announce nothing.
2611        // Written only where it changes: past an activation's second clause
2612        // the state is settled, and a store per clause is what it would cost.
2613        if activation.trace_entry.steps() {
2614            activation.trace_entry = activation.trace_entry.stepped();
2615        }
2616        // `TRACE`'s own `*-*` clause echo, and the single insertion point for
2617        // it -- exactly the analogue of `eval`'s own split from `eval_node`,
2618        // since this is the one place `run_bounded`'s loop (this function's
2619        // only non-test caller) visits every flat instruction position it
2620        // steps, markers (`Then`/`Else`/ `Otherwise`/`When`/`WhenCase`/`Label`)
2621        // included, so every one of those is traced from its own execution.
2622        let (tabled, position) = position;
2623        // Both halves are about a clause an interactive-debug `=` re-runs, so
2624        // a clause with neither pending reads the two fields and writes
2625        // nothing.
2626        let rerun = if self.activity.rerun_pending() {
2627            self.note_rerun(index)
2628        } else {
2629            None
2630        };
2631        let shortcut = tabled
2632            && source.is_some()
2633            && self.activity.clause_line_override.is_none()
2634            && rerun.is_none();
2635        let indent = if let Some(indent) = rerun {
2636            indent
2637        } else if shortcut {
2638            usize::cast_from(position.indent) + self.activity.activation_indent
2639        } else {
2640            self.printed_indent(code, index)
2641        };
2642        debug_assert!(
2643            rerun.is_some() || indent == self.printed_indent(code, index),
2644            "the chunk's indent table disagrees with the plan's for instruction {index}"
2645        );
2646        self.activity.clause_state.current_value_indent = indent;
2647        // Set here with the indent, and for the same reason that one is: this
2648        // is the one place every stepped instruction passes before its `step`
2649        // call runs, which is where it is read. See the field.
2650        self.activity.clause_state.instructions_traced_at_entry = self.trace_mode().all;
2651        // Set unconditionally, exactly like `current_value_indent` just
2652        // above and for the identical reason (that field's own doc comment):
2653        // `SIGL` (`lib.rs`'s doc on `current_clause_line`) has to stay
2654        // correct whether or not `TRACE` is on, and this is the one place
2655        // every stepped instruction, `SIGNAL`/`CALL` included, passes
2656        // through before its own `step` call runs.
2657        // `enter_clause` rather than a bare assignment: the clause line and
2658        // the clause boundary are one operation (`clause.rs`), and the
2659        // `ClauseEntry` this hands back is what `leave_stepped_clause` spends
2660        // on the matching half.
2661        let line = if shortcut {
2662            usize::cast_from(position.line)
2663        } else {
2664            self.clause_line_at(code, index, instruction, source)
2665                .unwrap_or_else(|| self.activity.clause_state.line())
2666        };
2667        debug_assert_eq!(
2668            line,
2669            self.clause_line_at(code, index, instruction, source)
2670                .unwrap_or_else(|| self.activity.clause_state.line()),
2671            "the chunk's line table disagrees with the plan's for instruction {index}"
2672        );
2673        // The frame readers' own view of this clause
2674        // ([`crate::activation::ClauseSnapshot`], which carries where this has
2675        // to be and why the line cannot come from `Activation::pc`). **Skipped
2676        // while a fragment is running**, so the enclosing `INTERPRET` clause
2677        // stays in force for this activation -- the answer given for the frame
2678        // beneath the `FRAME_INTERPRET` one.
2679        self.enter_clause_index(index);
2680        let entry = self.enter_clause(line, counted);
2681        // The debug tripwire I22 asks for, alongside `ActivityRoots::temps_len`, its
2682        // one prerequisite. `SteppedClause` carries the watermark and the
2683        // frame across to the matching half, which is where the check reads
2684        // them; that type's own doc comment has what it checks and why it is
2685        // there rather than in `pop_frame`.
2686        let temps_at_entry = self.roots.activity().temps_len();
2687        let frame = self.roots.activity_mut().push_frame();
2688        SteppedClause {
2689            entry,
2690            frame,
2691            temps_at_entry,
2692        }
2693    }

2695    /// `INTERPRET`'s work after its expression, from its value: the string
2696    /// request, the `>>>` line, the debug pause and the fragment's run. The
2697    /// walked instruction and `Op::ExecWith` both run it.
2698    pub(crate) fn exec_interpret_value(
2699        &mut self,
2700        index: usize,
2701        instruction: &Instruction,
2702        source: Option<&ProgramSource>,
2703        value: ObjRef,
2704    ) -> Result<Flow, Failure> {
2705        self.roots.activity_mut().push_temp(value);
2706        // The string request every instruction that evaluates a
2707        // single string expression shares -- so the `>>>` below traces the
2708        // conversion, measured: `trace r` over `interpret .K` with a
2709        // class-side `makeString` returning `'nop'` prints
2710        // `>>>   "nop"` and then the fragment's own `*-* nop`.
2711        let value = self.required_string_value(value)?;
2712        let text = self.to_text(value).to_vec();
2713        // `>>>` on the interpreted text itself, before the fragment
2714        // runs -- the same `trace_result` every other value-producing
2715        // arm calls (`Say`, `Assignment`), at the same
2716        // `current_value_indent`. Review finding I1(a): the arm shipped
2717        // without this and was the only value-producing arm in the
2718        // crate that traced nothing. With `trace r`, `zz = 'nop'` and
2719        // `interpret zz`, the trace shows
2720        // ```text
2721        //      3 *-* interpret zz
2722        //        >>>   "nop"
2723        //      3 *-* nop
2724        // ```
2725        self.trace_result(self.activity.clause_state.current_value_indent, &text);
2726        match self.debug_pause_instruction()? {
2727            interpret::Pause::Go => {}
2728            interpret::Pause::Again => return Ok(Flow::Goto(index)),
2729            interpret::Pause::Transfer(flow) => return Ok(flow),
2730        }
2731        // **Nested fragments are bounded by the stack, as calls are.** A
2732        // fragment runs on this activation, so no activation count grows, and
2733        // `Op::ExecWith` hands over a value read from a register, so no
2734        // expression level checked the stack on the way here either. Without
2735        // this, `zs = 'interpret zs'; interpret zs` overflows the interpreter
2736        // thread's stack and aborts the process instead of raising 11.1.
2737        // After the `>>>` line and the pause, as on the oracle: under `trace
2738        // r` the deepest `INTERPRET` still traces its value before the 11.1.
2739        // Before anything below is saved, so the raise leaves nothing to undo.
2740        let probe = 0u8;
2741        if self.stack_exhausted(&raw const probe) {
2742            return Err(Raised::insufficient_stack().into());
2743        }
2744        // **The fragment's level, with delta 0.** Measured: a
2745        // fragment's clauses print at the enclosing `INTERPRET`
2746        // clause's own absolute indent plus whatever nests them
2747        // *inside* the fragment -- `interpret "do jj = 1 to 1; say 2
2748        // & 1; end"` at top level echoes the inner clause at 2 and
2749        // the `INTERPRET` at 0, and the identical fragment two `DO`s
2750        // deep echoes them at 6 and 4. So the base is the enclosing
2751        // clause's printed indent exactly, with no bump of its own --
2752        // unlike a called routine's, which the same measurements put
2753        // two spaces further in (`call sub1` at printed indent 4 into
2754        // a flat routine echoes the callee's clause at 6).
2755        let base_indent = self.activity.clause_state.current_value_indent;
2756        let base_line = self.clause_site(source, instruction).map(|(line, _)| line);
2757        let saved_base = std::mem::replace(&mut self.activity.activation_indent, base_indent);
2758        let saved_line = std::mem::replace(&mut self.activity.clause_line_override, base_line);
2759        // **The fragment's own condition queue**, which the depth is the
2760        // key to rather than a second collection --
2761        // `Activity::fragment_depth` has why a fragment has one.
2762        // Incremented rather than replaced the way the values saved
2763        // above are, because a nested fragment inherits each of those
2764        // and needs a level of its own here.
2765        self.activity.fragment_depth += 1;
2766        let line = base_line.unwrap_or_else(|| self.activity.clause_state.line());
2767        self.enter_fragment_level(line, base_indent);
2768        // `saved_line` read before the replace above is also the
2769        // answer to "is a fragment already running", which is the one
2770        // extra thing `enter_fragment` needs and the only place it is
2771        // in hand. Nothing new is tracked for it.
2772        let saved_entry = self.enter_fragment(saved_line.is_some());
2773        let flow = pinned!(
2774            self,
2775            crate::pinning::PinKind::Interpret,
2776            self.run_fragment(text)
2777        );
2778        // **A condition still queued at this depth dies with the
2779        // fragment**, on the failing path as well as this one, because
2780        // the activation whose queue it was in is what ends. With
2781        // `interpret 'zq = raiser()'` and a handler that requeues, the
2782        // first handler runs and the second never does, and the
2783        // enclosing clause's own boundary does not pick it up. Dropping
2784        // the entries also keeps a program that runs `INTERPRET` in a
2785        // loop from accumulating undeliverable ones.
2786        let depth = self.activity.fragment_depth;
2787        self.activity
2788            .pending_traps
2789            .retain(|pending| pending.fragment_depth != depth);
2790        // **Nothing deeper than the fragment just left may survive
2791        // it**, which is the invariant that lets the delivery key be

2936        registers: RegFrame<'_>,
2937        index: usize,

2939        source: Option<&ProgramSource>,

2974            // blocked `Op::Exec` re-runs its instruction, and ends there.

3117    /// [`crate::ir::Op::Clause`] owes once the clause's own work has
3118    /// run.
3119    #[inline(always)]
3120    pub(crate) fn leave_stepped_clause<T: ClauseValue>(
3121        &mut self,
3122        entry: SteppedClause,
3123        code: &Code<'_>,
3124        index: usize,
3125        instruction: &Instruction,
3126        source: Option<&ProgramSource>,
3127        ran: Result<T, Failure>,
3128    ) -> Result<ClauseOutcome<T>, Failure> {

3926    /// The expression an op's address names: expression `slot` of
3927    /// `instruction`, then `path`'s steps down from that slot's root.
3928    pub(crate) fn chunk_node_at<'a>(
3929        body: &'a CodeBody,
3930        instruction: &'a Instruction,
3931        slot: u16,
3932        path: NodePath,
3933    ) -> Option<ExprRef<'a>> {
3934        let mut node = match (&instruction.kind, slot) {
3935            (InstructionKind::Assignment { value, .. }, 0) => body.expr(*value),
3936            // The one expression of an instruction that computes a value and
3937            // does something with it, slot `0`. A bare form holds no
3938            // expression at all, so it names no node and matches no arm here.
3939            (
3940                InstructionKind::Say {
3941                    expression: Some(expression),
3942                }
3943                | InstructionKind::Return {
3944                    expression: Some(expression),
3945                }
3946                | InstructionKind::Exit {
3947                    expression: Some(expression),
3948                }
3949                | InstructionKind::Push {
3950                    expression: Some(expression),
3951                }
3952                | InstructionKind::Queue {
3953                    expression: Some(expression),
3954                },
3955                0,
3956            ) => body.expr(*expression),
3957            (InstructionKind::Forward(forward), slot) => {
3958                forward_slot(body, forward, u32::from(slot))?
3959            }
3960            // The two instructions whose slot `0` exists only in one of their
3961            // forms. Both are reachable: the expression is compiled through
3962            // `push_value` like any other, so a call inside it emits an
3963            // `Op::CallExpr` that addresses the node from here.
3964            (InstructionKind::Signal(signal), 0) => match &**signal {
3965                interrexx_parse::Signal::Value(expression) => body.expr(*expression),
3966                interrexx_parse::Signal::Label(_) | interrexx_parse::Signal::Trap(_) => {
3967                    return None;
3968                }
3969            },
3970            (
3971                InstructionKind::Parse(parse)
3972                | InstructionKind::Arg(parse)
3973                | InstructionKind::Pull(parse),
3974                0,
3975            ) => match &parse.source {
3976                interrexx_parse::ParseSource::Value(Some(expression)) => body.expr(*expression),
3977                _ => return None,
3978            },
3979            (
3980                InstructionKind::Parse(parse)
3981                | InstructionKind::Arg(parse)
3982                | InstructionKind::Pull(parse),
3983                slot,
3984            ) => body.expr(parse.template.get(usize::from(slot) - 1)?.as_ref()?.value?),
3985            (InstructionKind::If { condition, .. }, 0) => body.expr(*condition),
3986            // An instruction run by `Op::ExecValue`, whose expression is
3987            // slot `0`.
3988            (kind, 0) if let Some(expression) = Self::exec_value_expression(body, kind) => {
3989                expression
3990            }
3991            // A `SELECT CASE`'s expression.
3992            (
3993                InstructionKind::Select {
3994                    case: Some(case), ..
3995                },
3996                0,
3997            ) => body.expr(*case),
3998            (InstructionKind::When { condition, .. }, 0) => body.expr(*condition),
3999            (
4000                InstructionKind::If { condition, .. } | InstructionKind::When { condition, .. },
4001                slot,
4002            ) => match body.expr(*condition).kind() {
4003                ExprView::Logical(items) => items.get(usize::from(slot - 1))??,
4004                _ => return None,
4005            },
4006            // A `DO`/`LOOP` header's `slot`th expression, resolved through the
4007            // same `loop_header_slot` the compiler emitted the slot's ops from
4008            // and the same one `eval_chunk_expr` evaluates a declining slot
4009            // with. Every slot of a header is addressable, not just one, which
4010            // is why this arm binds `slot` rather than matching a number.
4011            // Its `WHILE`/`UNTIL` condition, compiled into the loop's test
4012            // region, at the one slot no header expression takes.
4013            (InstructionKind::Do(loop_body) | InstructionKind::Loop(loop_body), slot)
4014                if u32::from(slot) == LOOP_CONDITION_SLOT =>
4015            {
4016                body.expr(loop_body.conditional.as_ref()?.condition)
4017            }
4018            (InstructionKind::Do(loop_body) | InstructionKind::Loop(loop_body), slot) => {
4019                loop_header_slot(body, loop_body, u32::from(slot))?
4020            }
4021            // A `CALL`'s argument `slot`.
4022            (InstructionKind::Call(call), slot) => match &**call {
4023                interrexx_parse::Call::Named { args, .. }
4024                | interrexx_parse::Call::Qualified { args, .. } => {
4025                    body.args(args).get(usize::from(slot))??
4026                }
4027                _ => return None,
4028            },
4029            // A message instruction's term, and the assignment form's value.
4030            (InstructionKind::Message { term, .. }, 0) => body.expr(*term),
4031            (
4032                InstructionKind::Message {
4033                    value: Some(value), ..
4034                },
4035                1,
4036            ) => body.expr(*value),
4037            // A `RAISE`'s expression `slot`, as `raise_slot` numbers them.
4038            (InstructionKind::Raise(raise), slot) => raise_slot(body, raise, slot)?,
4039            (InstructionKind::Use(use_), slot) => {
4040                Self::use_arg_default(body, use_, usize::from(slot))?
4041            }
4042            _ => return None,
4043        };
4044        let mut steps = path.steps();
4045        while let Some(step) = steps.next() {
4046            node = match (node.branches(), step) {
4047                (Branches::Binary(left, _), false) => left,
4048                (Branches::Binary(_, right), true) => right,
4049                (Branches::Prefix(operand), false) => operand,
4050                // A list step: its index in clear steps, then a set one.
4051                (Branches::Items(args), _) => {
4052                    let index = list_step(step, &mut steps)?;
4053                    args.get(index)??
4054                }
4055                (Branches::Message { target, args }, _) => {
4056                    match list_step(step, &mut steps)?.checked_sub(1) {
4057                        None => target,
4058                        Some(index) => args.get(index)??,
4059                    }
4060                }
4061                _ => return None,
4062            };
4063        }
4064        Some(node)
4065    }

4067    /// Expression `slot` of `instruction`, evaluated as the compiled stream's
4068    /// [`crate::ir::Op::EvalExpr`] asks.
4069    pub(crate) fn eval_chunk_expr(
4070        &mut self,
4071        code: &Code<'_>,
4072        instruction: &Instruction,
4073        slot: u32,
4074    ) -> Result<ObjRef, Failure> {
4075        match (&instruction.kind, slot) {
4076            (kind, 0) if let Some(expression) = Self::exec_value_expression(code.body, kind) => {
4077                self.eval(code, expression)
4078            }
4079            (InstructionKind::Use(use_), slot) => {
4080                let Some(expr) = Self::use_arg_default(code.body, use_, usize::cast_from(slot))
4081                else {
4082                    return Err(Loud::instruction(&instruction.kind).into());
4083                };
4084                self.eval(code, expr)
4085            }
4086            // A `RAISE`'s expression `slot`, evaluated and nothing else: each
4087            // part's work is its own `Op::RaisePart`.
4088            (InstructionKind::Raise(raise), slot) => {
4089                let Some(expr) = u16::try_from(slot)
4090                    .ok()
4091                    .and_then(|slot| raise_slot(code.body, raise, slot))
4092                else {
4093                    return Err(Loud::instruction(&instruction.kind).into());
4094                };
4095                self.eval(code, expr)
4096            }
4097            (InstructionKind::If { condition, .. }, 0) => {
4098                let holds = self.eval_if_condition(code, code.body.expr(*condition))?;
4099                // In range unconditionally: `SMALL_INT_MAX` is far above one.
4100                Ok(ObjRef::small_int(i64::from(holds)).unwrap_or(ObjRef::NIL))
4101            }
4102            (
4103                InstructionKind::Select {
4104                    case: Some(case_expr),
4105                    ..
4106                },
4107                0,
4108            ) => self.select_case(code, code.body.expr(*case_expr)),
4109            // A `DO`/`LOOP` header's `slot`th expression, in the order
4110            // `loop_header_plan` puts them in -- the order they were written,
4111            // which is the order they are evaluated. The expression is
4112            // evaluated and nothing else: its `>K>` echo and its validation are
4113            // ops of their own, because both have to happen before the next
4114            // expression is evaluated at all.
4115            (InstructionKind::Do(body) | InstructionKind::Loop(body), slot) => {
4116                let Some(expr) = loop_header_slot(code.body, body, slot) else {

4118                };

4120            }
4121            // An `Assignment`'s value, and the one expression of a `SAY`, a

4124            // this arm only for an expression `compile` did not emit a native

4127            // to arm here because this is the same
4128            // `eval` call it makes.
4129            (InstructionKind::Forward(forward), slot) => {
4130                let Some(expr) = forward_slot(code.body, forward, slot) else {
4131                    return Err(Loud::instruction(&instruction.kind).into());
4132                };
4133                self.eval(code, expr)
4134            }
4135            // `SIGNAL VALUE`'s own expression, for the shapes `compile` emits
4136            // no native op for. The `>K>` echo and the search are
4137            // `Op::Signal`'s, not this arm's, exactly as a loop header's
4138            // validation is its own op.
4139            (InstructionKind::Signal(signal), 0) => {
4140                let interrexx_parse::Signal::Value(expr) = &**signal else {
4141                    return Err(Loud::instruction(&instruction.kind).into());
4142                };
4143                self.eval(code, code.body.expr(*expr))
4144            }
4145            // `PARSE VALUE expr WITH`'s own expression, on the same terms. The
4146            // `>K>` echo belongs to `Op::ParseBegin`. This fallback is only for
4147            // the source expression, since an unsupported trigger always runs
4148            // in the template engine.
4149            (
4150                InstructionKind::Parse(parse)
4151                | InstructionKind::Arg(parse)
4152                | InstructionKind::Pull(parse),
4153                0,
4154            ) => {
4155                let interrexx_parse::ParseSource::Value(Some(expr)) = &parse.source else {
4156                    return Err(Loud::instruction(&instruction.kind).into());
4157                };
4158                self.eval(code, code.body.expr(*expr))
4159            }
4160            (InstructionKind::Assignment { value, .. }, 0) => {
4161                self.eval(code, code.body.expr(*value))
4162            }
4163            (
4164                InstructionKind::Say {
4165                    expression: Some(expression),
4166                }
4167                | InstructionKind::Return {
4168                    expression: Some(expression),
4169                }
4170                | InstructionKind::Exit {
4171                    expression: Some(expression),
4172                }
4173                | InstructionKind::Push {
4174                    expression: Some(expression),
4175                }
4176                | InstructionKind::Queue {
4177                    expression: Some(expression),
4178                },
4179                0,
4180            ) => self.eval(code, code.body.expr(*expression)),

4182        }

4248    /// Everything an evaluated condition value still owes: its `>>>` or `>K>`
4249    /// line, and the answer to whether it holds.
4250    pub(crate) fn condition_value(
4251        &mut self,
4252        value: ObjRef,
4253        trace: ConditionTrace<'_>,
4254        checked: bool,
4255        raise: fn(&[u8]) -> Raised,
4256    ) -> Result<bool, Failure> {
4257        // **The test's own temps frame, and it is a per-pass frame for the two
4258        // callers that are loop headers.** `WHILE` and `UNTIL` re-evaluate
4259        // their condition once per pass inside the enclosing `DO`
4260        // instruction's single frame, so the push below accumulates one
4261        // `ObjRef` per pass -- and one pinned heap object per pass whenever
4262        // the condition's result is one -- for the whole of the loop's run.
4263        // The value is never handed back (this answers a `bool`), so a frame
4264        // closed here releases it at the right time for every caller,
4265        // `IF`/`WHEN` included. The `?` path below leaves the pop to the outer
4266        // truncation, as `pop_frame`'s own doc describes.
4267        let frame = self.roots.activity_mut().push_frame();
4268        self.roots.activity_mut().push_temp(value);
4269        if self.trace_mode().results {
4270            let text = self.to_text(value).to_vec();
4271            match trace {
4272                // `IF`/plain `WHEN`'s own `>>>` (`select` / `when 1 = 1 then`
4273                // measured to show the identical
4274                // shape) -- measured, `WHILE`/`UNTIL` never get this (this
4275                // task's report: `trace r` over a `DO WHILE` shows only `>K>
4276                // "WHILE" => ...`, no bare `>>>` alongside it), which is why
4277                // this is a variant a caller picks rather than something
4278                // `eval_condition` decides on its own. `SELECT CASE`'s own
4279                // `WHEN`/`WhenCase` comparison never reaches this function at
4280                // all -- see `test_case_when`'s own trace calls instead.
4281                ConditionTrace::Result(indent) => self.trace_result(indent, &text),
4282                // `WHILE`/`UNTIL`'s own `>K>` -- re-fires every pass because the
4283                // condition is re-evaluated every pass, which the flat loop's
4284                // own call site already does.
4285                ConditionTrace::Keyword(indent, keyword) => {
4286                    self.trace_keyword(indent, keyword, &text);
4287                }
4288            }
4289        }
4290        // A comma list answers a logical constant of its own, its elements
4291        // having been judged one by one.
4292        debug_assert!(
4293            !checked || crate::eval::truth_without_conversion(value).is_some(),
4294            "a checked condition answered something other than a logical constant"
4295        );
4296        let decided = self.truth(value, raise);
4297        self.roots.activity_mut().pop_frame(frame);
4298        decided
4299    }

4423/// The index a list step names, `first` its first step and `rest` the ones
4424/// after it: how many clear steps come before a set one.
4425fn list_step(first: bool, rest: &mut impl Iterator<Item = bool>) -> Option<usize> {
4426    let mut index = 0;
4427    let mut step = first;
4428    while !step {
4429        index += 1;
4430        step = rest.next()?;
4431    }

4435pub(crate) fn shape_of(name: &[u8]) -> NameShape {
4436    // The first dot decides it: none is a simple name, one in the last

4438    // Found by a search that stops there rather than by counting every dot,

4440    match name.iter().position(|&b| b == b'.') {
4441        None => NameShape::Simple,
4442        Some(at) if at + 1 == name.len() => NameShape::Stem,
4443        Some(_) => NameShape::Compound,
4444    }

4455/// One legal symbol character, by the scanner's own character table
4456/// (`scanner.rs`'s `is_symbol_char`: `! . ? _ 0-9 A-Z a-z`, ASCII only --
4457/// `SymbolTable::intern`'s own doc says a non-ASCII byte cannot be part of a
4458/// symbol at all).
4459fn is_symbol_byte(b: u8) -> bool {
4460    matches!(b, b'!' | b'.' | b'?' | b'_' | b'0'..=b'9' | b'A'..=b'Z' | b'a'..=b'z')
4461}

4463/// The longest name a subsidiary-list word may have, counted over the whole
4464/// compound symbol with its periods.
4465const MAX_SYMBOL_LENGTH: usize = 250;
4466
4467/// Validates one word of an indirect subsidiary list and answers its upcased
4468/// name, or the condition raised for it.
4469fn validate_indirect_word(word: &[u8]) -> Result<Vec<u8>, Failure> {
4470    if !word.iter().copied().all(is_symbol_byte) {
4471        return Err(raised_symbol_expected(word).into());
4472    }
4473    match word[0] {
4474        b'0'..=b'9' => return Err(raised_digit_led(word).into()),
4475        b'.' => return Err(raised_dot_led(word).into()),
4476        _ => {}
4477    }
4478    if word.len() > MAX_SYMBOL_LENGTH {
4479        return Err(raised_symbol_expected(word).into());
4480    }
4481    Ok(word.to_ascii_uppercase())
4482}

4487impl Interp {
4488    /// The object a stream builtin sends its
4489    /// message to. An omitted or empty name and `STDIN`/`STDOUT`/`STDERR`
4490    /// (caselessly, with or without a trailing colon) are the standard
4491    /// streams; anything else is qualified and looked up in the owning
4492    /// activation's table by that qualified name, and a miss builds a
4493    /// `Stream` with the name **as written** -- `~string` answers what the
4494    /// program passed, while the key is the resolved path, so `a.txt` and
4495    /// `./a.txt` are one entry.
4496    ///
4497    /// `added` says whether a hit-or-miss may populate the table: every
4498    /// builtin passes it except `STATE`, `DESCRIPTION` and a `STREAM` command
4499    /// that is not OPEN, CLOSE or SEEK, which is why `stream(n,'S')` on an
4500    /// unknown name leaves nothing behind.
4501    pub(crate) fn resolve_stream(
4502        &mut self,
4503        name: &[u8],
4504        input: bool,
4505        added: bool,
4506    ) -> Result<ObjRef, Failure> {
4507        if let Some(standard) = standard_stream_name(name, input) {
4508            // **The entry of `.local`, not the symbol**: a class of that name
4509            // or a package's `local` entry changes what `.OUTPUT` finds and
4510            // not where the builtins write. A `.local` without the entry
4511            // falls back to the symbol.
4512            let entry = standard.strip_prefix(b".").unwrap_or(standard);
4513            return match self.local_route(entry)? {
4514                Some(found) => Ok(found),
4515                None => self.dot_variable(standard),
4516            };
4517        }
4518        let qualified = crate::builtin::platform::qualify_name(self, name);
4519        if let Some(found) = self
4520            .stream_table()
4521            .and_then(|table| table.get(qualified.as_slice()))
4522        {
4523            return Ok(found.object());
4524        }
4525        // **The manager sees the qualified name, and only a table miss.**
4526        // `.Stream~new` reaches none of this, so a program that builds its
4527        // own stream object is never checked.
4528        if let Some(replacement) = self.check_stream_access(&qualified)? {
4529            if let Some(table) = self.stream_table_mut() {
4530                table.get_or_insert_with(Box::default).insert(
4531                    qualified.into_boxed_slice(),
4532                    StreamEntry::Replacement(replacement),
4533                );
4534            }
4535            return Ok(replacement);
4536        }
4537        let Some(class) = self.rexx_package_class(b"STREAM") else {
4538            // Only before the stream class library has installed, which is the
4539            // library bootstrap's own step.
4540            return Err(Loud::environment_symbol(b".STREAM").into());
4541        };
4542        let argument = self.text(name);
4543        self.roots.activity_mut().push_temp(argument);
4544        let caller = self.caller();
4545        let built = pinned!(
4546            self,
4547            crate::pinning::PinKind::StreamWrapper,
4548            self.send_message(class, b"NEW", None, &[Some(argument)], caller)
4549        )?
4550        .ok_or_else(|| Failure::from(Raised::no_result(b"NEW")))?;
4551        if added && let Some(table) = self.stream_table_mut() {
4552            table
4553                .get_or_insert_with(Box::default)
4554                .insert(qualified.into_boxed_slice(), StreamEntry::Opened(built));
4555        }
4556        Ok(built)
4557    }
4558}

4561    /// The security manager's `STREAM` checkpoint:
4562    /// the object to use in place of a new `Stream`, or `None` where the manager declined or none
4563    /// is installed.
4564    ///
4565    /// **The manager's entry is `STREAM`, not `RESULT`.**
4566    fn check_stream_access(&mut self, qualified: &[u8]) -> Result<Option<ObjRef>, Failure> {
4567        if self.effective_security_manager().is_none() {
4568            return Ok(None);
4569        }
4570        let name = self.text(qualified);
4571        self.roots.activity_mut().push_temp(name);
4572        let entries = [(crate::security::key::NAME, name)];
4573        let Some(info) = self.security_check(crate::security::message::STREAM, &entries)? else {
4574            return Ok(None);
4575        };

4577    }

4579    /// Drops `name`'s entry from the owning activation's table, which is what
4580    /// `LINEOUT(name)` with no string and a `CLOSE` command do after their
4581    /// send: the stream stays closed and the next builtin to name it builds a
4582    /// fresh one.
4583    pub(crate) fn forget_stream(&mut self, name: &[u8]) {
4584        let qualified = crate::builtin::platform::qualify_name(self, name);
4585        if let Some(table) = self.stream_table_mut().and_then(Option::as_deref_mut) {
4586            table.remove(qualified.as_slice());
4587        }
4588    }
4589
4590    /// A manager's replacement is closed in the owning level. Removing the
4591    /// entries before the sends lets a trap after a failed close use the
4592    /// names again (spec: env.security.stream.close).
4593    #[cold]
4594    #[inline(never)]
4595    pub(crate) fn close_stream_replacements(
4596        &mut self,
4597        ended: &Result<Ended, Failure>,
4598    ) -> Option<Failure> {
4599        let table = self.activation_mut().streams.as_deref_mut()?;
4600        let mut replacements = Vec::new();
4601        table.retain(|_, entry| match *entry {
4602            StreamEntry::Opened(_) => true,
4603            StreamEntry::Replacement(object) => {
4604                replacements.push(object);
4605                false
4606            }
4607        });
4608        if replacements.is_empty() {
4609            return None;
4610        }
4611
4612        let frame = self.roots.activity_mut().push_frame();
4613        let returned = match ended {
4614            Ok(ended) => ended.value_ref(),
4615            Err(Failure::Exited(value)) => *value,
4616            _ => None,
4617        };
4618        for object in replacements.iter().copied().chain(returned) {
4619            self.roots.activity_mut().push_temp(object);
4620        }
4621        let program_failed = self.activation().entry == Entry::TopLevel
4622            && matches!(ended, Err(failure) if !matches!(failure, Failure::Exited(_)));
4623        let previous = match ended {
4624            Ok(_) | Err(Failure::Exited(_)) => None,
4625            Err(_) => Some(self.take_stream_close_condition()),
4626        };
4627        let mut first = None;
4628        for stream in replacements {
4629            let caller = self.caller();
4630            let closed = pinned!(
4631                self,
4632                crate::pinning::PinKind::StreamWrapper,
4633                self.send_message(stream, b"CLOSE", None, &[], caller)
4634            );
4635            if let Err(failure) = closed {
4636                let condition = self.take_stream_close_condition();
4637                if first.is_none() && !program_failed {
4638                    first = Some((failure, condition));
4639                }
4640            }
4641        }
4642        let (failure, condition) = match first {
4643            Some((failure, condition)) => (Some(failure), Some(condition)),
4644            None => (None, previous),
4645        };
4646        if let Some((levels, [additional, result, description, rc])) = condition {
4647            self.restore_failure_levels(levels);
4648            self.activity.pending_additional = additional;
4649            self.activity.pending_result = result;
4650            self.activity.pending_description = description;
4651            self.activity.pending_rc = rc;
4652        }
4653        self.roots.activity_mut().pop_frame(frame);
4654        failure
4655    }
4656
4657    /// Roots a condition while the remaining `CLOSE` sends run.
4658    fn take_stream_close_condition(&mut self) -> (FailureLevels, [Option<ObjRef>; 4]) {
4659        let levels = self.take_failure_levels();
4660        self.hold_origin_package(levels.origin);
4661        let pending = [
4662            self.activity.pending_additional.take(),
4663            self.activity.pending_result.take(),
4664            self.activity.pending_description.take(),
4665            self.activity.pending_rc.take(),
4666        ];
4667        for object in pending.into_iter().flatten() {
4668            self.roots.activity_mut().push_temp(object);
4669        }
4670        (levels, pending)
4671    }

4674/// The `.local` name a stream name spells, or `None` for an ordinary one.
4675/// Matched caselessly with an optional trailing colon, and an omitted or empty
4676/// name is the default input or output rather than a file.
4677///
4678/// **The monitors, not the streams behind them.** the stream lookup answers
4679/// `.INPUT`, `.OUTPUT` and `.ERROR`, so a redirected route is followed: with
4680/// `.output~destination(.stderr)` pushed, `SAY`, `LINEOUT(,)` and `CHAROUT(,)`
4681/// go to stderr while `.stdout~lineout` stays on stdout. Note `STDERR` reaches
4682/// `.ERROR` rather than an `.STDERR` monitor, which is absent.
4683fn standard_stream_name(name: &[u8], input: bool) -> Option<&'static [u8]> {
4684    let bare = name.strip_suffix(b":").unwrap_or(name);
4685    // An omitted or empty name is the default input or output, and which one
4686    // depends on the caller -- measured with stdin redirected from a file,
4687    // `linein()` reads `l1` and `chars()` answers 9, while `charout( ,'W')`
4688    // writes `W` to stdout.
4689    if bare.is_empty() {
4690        return Some(if input { b".INPUT" } else { b".OUTPUT" });
4691    }
4692    if bare.eq_ignore_ascii_case(b"STDIN") {
4693        return Some(b".INPUT");
4694    }
4695    if bare.eq_ignore_ascii_case(b"STDOUT") {
4696        return Some(b".OUTPUT");
4697    }
4698    if bare.eq_ignore_ascii_case(b"STDERR") {
4699        return Some(b".ERROR");
4700    }
4701    None
4702}
