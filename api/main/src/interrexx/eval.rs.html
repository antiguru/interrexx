16/// The evaluation-depth limit: `eval`'s own recursion, one level per
17/// left-deep term, refuses anything past this depth with 11.1 ("Insufficient
18/// control stack space") rather than letting the interpreter thread's guard
19/// page abort the process silently.
20const MAX_EVAL_DEPTH: usize = 100_000;
21
22/// Which of the three bare-symbol reads an expression is, carried where the
23/// `NodeKind` itself is not.
24pub(crate) const LOGICAL_TRUE: ObjRef = ObjRef::inline_byte(b'1');
25/// See [`LOGICAL_TRUE`].
26pub(crate) const LOGICAL_FALSE: ObjRef = ObjRef::inline_byte(b'0');
27
28/// The value a comparison or logical operator answers with.
29pub(crate) const fn logical(holds: bool) -> ObjRef {
30    if holds { LOGICAL_TRUE } else { LOGICAL_FALSE }
31}

39#[derive(Clone, Copy, PartialEq, Eq, Debug)]
40pub(crate) enum SymbolRead {
41    /// `NodeKind::Variable`: one slot, and the derived name when it is unset.
42    Simple,
43    /// `NodeKind::Stem`: one slot too, but a miss allocates a real
44    /// `Body::Stem` there rather than deriving a name (`Interp::read_stem`).
45    Stem,
46    /// `NodeKind::Compound`: a stem slot and a tail key resolved at the read
47    /// site, neither of which is the symbol's own slot -- which is why a
48    /// compound is the kind a compiled read has no slot for.
49    Compound,
50    /// A [`SymbolRead::Compound`] in the value of an assignment to the same
51    /// compound, whose write uses the key this read resolves rather than
52    /// resolving it again (`share_tail_key` in `ir/compile.rs` says
53    /// where). The read holds its key in `Interp::shared_key` for that
54    /// write.
55    CompoundShared,
56    /// `NodeKind::DotVariable`: an environment symbol, which traces `>E>`.
57    Environment,
58}

61    /// Evaluates one expression node, and keeps the depth bookkeeping the
62    /// evaluation-depth limit needs.
63    pub(crate) fn enter_eval_node<T>(&mut self, anchor: *const T) -> Result<(), Failure> {
64        let here = anchor.addr();
65
66        self.activity.depth += 1;
67        if self.activity.depth == 1 {
68            self.activity.stack_entry = here;
69        }
70        if self.activity.depth > self.activity.max_depth {
71            self.activity.max_depth = self.activity.depth;
72            self.activity.stack_first = self.activity.stack_entry;
73            self.activity.stack_deepest = here;
74        }
75
76        // The evaluation-depth limit, checked **after** the bookkeeping above
77        // so a run that trips it still gets an accurate `StackSpan` (the
78        // deepest level reached is this one, not the one before it). `>
79        // MAX_EVAL_DEPTH`, not `>=`: a 100,000-term left-deep expression
80        // reaches depth exactly 100,000 and survives to print an answer at that
81        // depth (Deviation 2), so the check must fire one level *above* the
82        // deepest one that must be handled, or the one depth we are supposed to
83        // match becomes the first one we refuse.
84        if self.activity.depth > MAX_EVAL_DEPTH || self.stack_exhausted(anchor) {
85            self.activity.depth -= 1;
86            return Err(Raised::insufficient_stack().into());
87        }
88        Ok(())
89    }

91    pub(crate) fn eval(&mut self, code: &Code<'_>, expr: ExprRef<'_>) -> Result<ObjRef, Failure> {
92        let probe = 0u8;
93        self.enter_eval_node(&raw const probe)?;
94        let value = self.eval_node(code, expr);
95        self.activity.depth -= 1;
96        // `TRACE I`'s own value events, inserted here and in
97        // [`Interp::eval_in_parse`], which repeats this depth bookkeeping, so a
98        // post-order event with the value already in hand needs no threading
99        // through `eval_node`'s own arms at all. Only `Ok` gets one: a raise
100        // has no value to show, and a trace never shows one for a failing
101        // sub-expression either (the *clause* echo already happened, and the
102        // raise's own report is what follows).
103        if let Ok(v) = &value
104            && self.tracing_intermediates()
105            && self.trace_intermediate_held(code, expr, *v)
106        {
107            return Err(self.take_held_trace_failure());
108        }
109        value
110    }

112    /// [`Interp::eval`] for PARSE's source and patterns, taking its trace
113    /// hook's `Result` in place: measured by callgrind, `eval`'s own shape
114    /// there costs `parse` +0.32%, and this shape in `eval` costs
115    /// `pingguard` +0.07%.
116    pub(crate) fn eval_in_parse(
117        &mut self,
118        code: &Code<'_>,
119        expr: ExprRef<'_>,
120    ) -> Result<ObjRef, Failure> {
121        let probe = 0u8;
122        self.enter_eval_node(&raw const probe)?;
123        let value = self.eval_node(code, expr);
124        self.activity.depth -= 1;
125        if let Ok(v) = &value {
126            self.trace_intermediate(code, expr, *v)?;
127        }
128        value
129    }

141    /// (`>A>` is not here: an argument's line belongs to the *call site*

146    /// (`TRACE I` only; `TRACE R` never reaches any of these, measured),
147    /// so the match below only ever runs its own `to_text` cost under `I`.

150        &mut self,
151        code: &Code<'_>,
152        expr: ExprRef<'_>,
153        value: ObjRef,
154    ) -> Result<(), Failure> {
155        if !self.tracing_intermediates() {
156            return Ok(());
157        }
158        self.trace_intermediate_event(code, expr, value)
159    }

161    /// [`Interp::trace_intermediate`]'s body, entered only under `TRACE I`.
162    /// `value` is rooted while its line is written, because a delivery to
163    /// `.TraceOutput` runs Rexx clauses that can collect.
164    #[inline(never)]
165    fn trace_intermediate_event(
166        &mut self,
167        code: &Code<'_>,
168        expr: ExprRef<'_>,
169        value: ObjRef,
170    ) -> Result<(), Failure> {
171        let frame = self.roots.activity_mut().push_frame();
172        self.roots.activity_mut().push_temp(value);
173        let traced = self.trace_intermediate_line(code, expr, value);
174        self.roots.activity_mut().pop_frame(frame);
175        traced
176    }

178    fn trace_intermediate_line(
179        &mut self,
180        code: &Code<'_>,
181        expr: ExprRef<'_>,
182        value: ObjRef,
183    ) -> Result<(), Failure> {
184        let indent = self.activity.clause_state.current_value_indent;
185        match expr.kind() {
186            // `Constant`'s own shape is reasoned from `Literal`'s measured
187            // one, not independently probed (this crate's own report says
188            // so) -- both are `>L>` with no tag.
189            // `echo_literal` rather than an open-coded render plus
190            // `trace_literal`, because the compiled stream's own
191            // `Op::TraceLiteral` emits the identical line from a register and
192            // the two must not be able to disagree about it.
193            ExprView::Literal(_) | ExprView::Constant(_) => self.echo_literal(value),
194            // The three bare-symbol reads, all through the one function
195            // `crate::ir::Op::TraceRead` enters from a register -- for the
196            // reason `echo_literal` above is one function: a compiled read
197            // emits nothing of its own, so the line has to come from an op,
198            // and the two emissions must not be able to disagree.
199            // `echo_symbol_read`'s own doc comment has what a compound owes
200            // that this does not emit.
201            ExprView::Variable(id) | ExprView::Stem(id) | ExprView::Compound(id) => {
202                self.echo_symbol_read(code, id, value)?;
203            }
204            // `.NIL`/`.TRUE`/`.FALSE` -- `>E>`. The other `DotVariable` names all fail loudly
205            // before reaching here (`eval_node`'s own arm), so this is
206            // exhaustive over what can arrive.
207            ExprView::DotVariable(id) => self.echo_environment_symbol(code, id, value)?,
208            // `>P>`, not `>O>` -- a prefix operator's line is its own, which
209            // is why this is not the arm below with a different operator type.
210            // `echo_prefix_op` rather than an open-coded render plus
211            // `trace_prefix_op`, because the compiled stream's own
212            // `crate::ir::Op::TracePrefix` emits the identical line from a
213            // register and the two must not be able to disagree -- the reason
214            // `echo_literal` above is one function.
215            ExprView::Prefix { op, .. } => self.echo_prefix_op(op, value),
216            // Every binary operator, arithmetic, comparison, logical and
217            // concatenation alike, traces as `>O>` -- one `match` arm is shared
218            // by every family. `echo_operator` rather than an open-coded render
219            // plus `trace_operator`, because the compiled stream's own
220            // `crate::ir::Op::TraceOperator` emits the identical line from a
221            // register and the two must not be able to disagree -- the reason
222            // `echo_literal` above is one function.
223            ExprView::Binary { op, .. } => self.echo_operator(op, value),
224            // `>name`/`<name` traces as an **operator**, not as the read of the
225            // variable it names: `>O> ">" => "PQ"`, the referenced variable's
226            // own *name* as the operator's value (the reference operator never
227            // reads the variable's value at all, so there is no `>V>` line to
228            // go with it). At both spellings and in both positions -- `say
229            // >pq`, `say <pq`, `call sub >pq` and `call sub <pq` -- all trace
230            // `>O> ">" => "PQ"`, the `<` form included: the literal `">"` is
231            // passed regardless of which byte was written.
232            ExprView::VariableReference { symbol, .. } => {
233                // The reference's own name, `value->getName()`: an alias
234                // made by `USE ARG >x` names the variable it refers to.
235                let name = match self.as_variable_reference(value) {
236                    Some(reference) => reference.name.to_vec(),
237                    None => code.symbols.name(symbol).as_bytes().to_vec(),
238                };
239                self.trace_operator(indent, b">", &name);
240            }
241            // `>F>`, the **function form only** -- `trace_function`'s own
242            // doc comment has the measured pair that separates it from
243            // `CALL`. The tag is the name as the call site spells it: a
244            // symbol target arrives already upcased and a literal one
245            // verbatim (`interrexx-parse`'s own `CallTarget`), which is the same
246            // pair of spellings `eval_call` resolves against.
247            ExprView::Call { target, .. } => {
248                let name = match target {
249                    Target::Symbol(id) => code.symbols.name(id).as_bytes().to_vec(),
250                    Target::Literal(bytes) => bytes.to_vec(),
251                };
252                let text = self.traced_string_value_text(value)?;
253                self.trace_function(indent, &name, &text);
254            }
255            // `>F>` too, tagged with the routine name **alone** -- measured,
256            // `trace i` over `say w:wroutine()` traces
257            // `>F>   WROUTINE => "wroutine ran"`, with no qualifier on the
258            // tag, which is why this shares the arm above's emitter rather
259            // than `>N>`'s.
260            ExprView::QualifiedCall { name, .. } => {
261                let name = code.symbols.name(name).as_bytes().to_vec();
262                let text = self.traced_string_value_text(value)?;
263                self.trace_function(indent, &name, &text);
264            }
265            // `>N>`, tagged `namespace:class` -- `traceClassResolution`,
266            // whose tag is the two symbols joined by a colon.
267            ExprView::ClassResolver { namespace, name } => {
268                let mut tag = code.symbols.name(namespace).as_bytes().to_vec();
269                tag.push(b':');
270                tag.extend_from_slice(code.symbols.name(name).as_bytes());
271                let text = self.traced_string_value_text(value)?;
272                self.trace_namespace(indent, &tag, &text);
273            }
274            // **`>M>` is not here**, and that is the one value prefix this
275            // hook does not own: the message-assignment form is an
276            // instruction that never evaluates its term as an expression, so
277            // a hook arm would print the line for one of the two forms only.
278            // `Interp::message_term` (`dispatch.rs`) emits it instead, at the
279            // same point in the sequence this hook would have.
280            ExprView::Message { .. } => {}
281            // A comma list (`NodeKind::Logical`) has no value line of its
282            // own -- each element is a full `eval` call in its own right
283            // (`eval_logical_list`), and traces itself through this same
284            // function when it runs. Every other unimplemented `NodeKind`
285            // never reaches here at all (`eval_node`'s own loud fallback).
286            _ => {}
287        }
288        Ok(())
289    }

291    /// One bare symbol's read: the whole of what `eval_node`'s own
292    /// `Variable`/`Stem`/`Compound` arms do, entered from there and from
293    /// `crate::ir::Op::Load`. `chunk` is the load's own, which holds the
294    /// memo an environment read's `at` names; the tree walk has none.
295    pub(crate) fn read_symbol(
296        &mut self,
297        code: &Code<'_>,
298        chunk: Option<&crate::ir::Chunk>,
299        read: SymbolRead,
300        id: SymbolId,
301        at: Option<usize>,
302    ) -> Result<ObjRef, Failure> {
303        match read {
304            SymbolRead::Simple => {
305                let (value, novalue) = self.read_at(code, id, at);
306                self.novalue_check(novalue, value)?;
307                Ok(value)
308            }
309            SymbolRead::Stem => Ok(self.read_stem_at(code.symbols.name(id).as_bytes(), at)),
310            // `id` names the *whole* compound (its interned spelling is the
311            // full dotted text); the split decomposes it into the stem's own
312            // name and the tail pieces `tail_key` (`stem.rs`) resolves into
313            // the one key `stem_get` looks up.
314            SymbolRead::Compound | SymbolRead::CompoundShared => {
315                debug_assert!(
316                    at.is_none(),
317                    "a compound read was handed a slot, and the slot it reads is the stem's"
318                );
319                let (stem_name, stem_at) = code.stem(id);
320                // Borrowed and handed back, rather than a fresh key per read
321                // -- see `Interp::key_buffer`. The `?` below is why the return
322                // is placed before it: an early exit there would lose the
323                // buffer, which is safe but forfeits the reuse.
324                // A shared read holds its key only where no activation was
325                // pushed from here to the end of the read: a tail's string
326                // request or a NOVALUE hook runs code, and that code could
327                // assign a tail variable. Set here, the push clears it.
328                let shared = read == SymbolRead::CompoundShared;
329                if shared {
330                    self.shared_key_for = Some(id);
331                }
332                let mut key = self.take_key_buffer();
333                if let Err(failure) = self.tail_key_into(code, id, &mut key) {
334                    self.give_key_buffer(key);
335                    self.shared_key_for = None;
336                    return Err(failure);
337                }
338                // `>C>`, here rather than in `echo_symbol_read_line`, so that
339                // the tail is resolved once per reference -- that function's
340                // own doc comment has the reason.
341                if self.tracing_intermediates() {
342                    let tag = code.symbols.name(id).as_bytes().to_vec();
343                    let mut printed = stem_name.to_vec();
344                    printed.extend_from_slice(&key);
345                    let indent = self.activity.clause_state.current_value_indent;
346                    self.trace_compound_name(indent, &tag, &printed);
347                }
348                let (value, novalue, held) = self.stem_get_holding(stem_name, stem_at, &key);
349                // Held only untraced, so that a traced write resolves and
350                // prints its own name the way it always has.
351                if shared {
352                    if self.shared_key_for == Some(id) && !self.tracing_intermediates() {
353                        self.shared_key.clear();
354                        self.shared_key.extend_from_slice(&key);
355                        self.shared_tail = held;
356                    } else {
357                        self.shared_key_for = None;
358                    }
359                }
360                self.give_key_buffer(key);
361                if let Err(failure) = self.novalue_check(novalue, value) {
362                    self.shared_key_for = None;
363                    return Err(failure);
364                }
365                Ok(value)
366            }
367            SymbolRead::Environment => match chunk {
368                Some(chunk) => self.load_environment(code, chunk, id, at),
369                None => self.environment_symbol(code, id),
370            },
371        }
372    }

374    /// An environment symbol's value.
375    ///
376    /// **The parser's own names never reach resolution and every other one
377    /// does.** The parser installs a `SpecialDotVariable` retriever for `.NIL`,
378    /// `.TRUE` and `.FALSE`, so those three are parse-time constants. With
379    /// `::class True` in the file, `say .TRUE` still prints `1` where `say
380    /// value('.TRUE')` prints `The TRUE class`, because only the second goes
381    /// through variable resolution.
382    fn environment_symbol(&mut self, code: &Code<'_>, id: SymbolId) -> Result<ObjRef, Failure> {
383        match code.symbols.name(id) {
384            ".NIL" => Ok(ObjRef::NIL),
385            // `.true`/`.false` need no representation of their own (D15):
386            // they are the shared one-byte strings "1" and "0", the
387            // `LOGICAL_TRUE`/`LOGICAL_FALSE` constants every built-in boolean
388            // answers as well.
389            ".TRUE" => Ok(LOGICAL_TRUE),
390            ".FALSE" => Ok(LOGICAL_FALSE),
391            other => self.dot_variable(other.as_bytes()),
392        }
393    }

395    /// [`Interp::environment_symbol`] for a compiled read, whose class
396    /// answer the memo at `chunk`'s site `at` keeps: see
397    /// [`Interp::dot_variable_memoized`]. Out of line, so the op loop pays
398    /// one call for it as it did for [`Interp::read_symbol`].
399    #[inline(never)]
400    pub(crate) fn load_environment(
401        &mut self,
402        code: &Code<'_>,
403        chunk: &crate::ir::Chunk,
404        id: SymbolId,
405        at: Option<usize>,
406    ) -> Result<ObjRef, Failure> {
407        let Some(memo) = at.and_then(|site| chunk.environment_site(site)) else {
408            return self.environment_symbol(code, id);
409        };
410        match code.symbols.name(id) {
411            ".NIL" => Ok(ObjRef::NIL),
412            ".TRUE" => Ok(LOGICAL_TRUE),
413            ".FALSE" => Ok(LOGICAL_FALSE),
414            other => self.dot_variable_memoized(other.as_bytes(), memo),
415        }
416    }

467    pub(crate) fn eval_node(
468        &mut self,
469        code: &Code<'_>,
470        expr: ExprRef<'_>,
471    ) -> Result<ObjRef, Failure> {
472        match expr.kind() {
473            ExprView::Literal(bytes) => Ok(self.literal(bytes)),
474            // A constant's value is its own upcased spelling, which is
475            // observable rather than incidental: `say 1e5` prints `1E5`.
476            ExprView::Constant(id) => Ok(self.literal(code.symbols.name(id).as_bytes())),
477            // The three bare-symbol reads, each through the one function
478            // `crate::ir::Op::Load` enters with the slot already resolved.
479            // They are three *different* operations sharing one entry point,
480            // not one operation under three spellings -- `read_symbol`'s own
481            // doc comment has what separates them -- so the dispatch is on a
482            // kind rather than absent.
483            ExprView::Variable(id) => self.read_symbol(code, None, SymbolRead::Simple, id, None),
484            ExprView::Stem(id) => self.read_symbol(code, None, SymbolRead::Stem, id, None),
485            ExprView::Compound(id) => self.read_symbol(code, None, SymbolRead::Compound, id, None),
486
487            ExprView::DotVariable(id) => self.environment_symbol(code, id),
488
489            ExprView::Prefix { op, operand } => self.eval_prefix(code, op, operand),
490
491            // Arithmetic (`+ - * / % // **`), through `interrexx-num` under the
492            // settings in force *now* -- the rendering of the result is fixed
493            // at creation (D15),
494            // but the DIGITS/FORM an operation computes *under* is always
495            // the activation's current ones, read fresh on every call. Ahead
496            // of the arm below, which is every *other* binary operator, so
497            // that arithmetic reaches `eval_arithmetic`: `**`'s exponent is
498            // not converted the way its base is, and that asymmetry has
499            // nowhere to live in a shared operand path.
500            ExprView::Binary { op, left, right } if is_arithmetic(op) => {
501                self.eval_arithmetic(code, op, left, right)
502            }
503
504            // Concatenation, comparison and `&`/`|`/`&&`, which share this operand
505            // prologue exactly: **both operands are always evaluated, left
506            // then right, and never short-circuited** -- measured, `say (0 &
507            // 'x')` raises 34.901 on `"x"` though the result is already
508            // decided. What each operator then does with two values is
509            // `apply_binary`, entered from here and from
510            // `crate::ir::Op::Binary`.
511            ExprView::Binary { op, left, right } if is_native_binary(op) && !is_arithmetic(op) => {
512                let frame = self.roots.activity_mut().push_frame();
513                let left_value = self.eval(code, left)?;
514                self.roots.activity_mut().push_temp(left_value);
515                let right_value = self.eval(code, right)?;
516                self.roots.activity_mut().push_temp(right_value);
517                // Bound rather than propagated with `?`, so the frame is
518                // popped on the failure path too -- an operator's own raise
519                // discards its operands here, where the `?` on an operand's
520                // evaluation above leaves them for `Op::Clause`'s region to
521                // truncate. The module doc has why both are safe.
522                let text = compares_as_text(op, left, right);
523                let result = self.apply_binary_op(op, text, left_value, right_value);
524                self.roots.activity_mut().pop_frame(frame);
525                result
526            }
527
528            // The comma-separated conditional list (`IF a, b THEN`, and
529            // `WHEN`/`GUARD`/`WHILE`/`UNTIL`'s own versions of the same
530            // syntax) -- see `eval_logical_list`'s own doc comment for the
531            // short-circuit and sub-number this arm alone can give it.
532            ExprView::Logical(items) => self.eval_logical_list(code, items),
533
534            // `>name`/`<name` answers a `VariableReference`, the variable
535            // itself rather than its value: `vr = 5; o = >vr; say o~class~id`
536            // is `VariableReference`, and `o~value = 7` writes `vr`.
537            ExprView::VariableReference { symbol, stem } => {
538                self.variable_reference(code, symbol, stem)
539            }
540
541            // `f(...)`/`"f"(...)` -- see `eval_call`'s own doc.
542            ExprView::Call { target, args } => self.eval_call(code, target, args),
543
544            // `target~name(...)`, `target~~name(...)` and `target[...]`
545            // -- see `Interp::message_term` (`dispatch.rs`) for
546            // the evaluation order and `Interp::resolve_in`/`Interp::invoke`
547            // for the send itself. The **expression** form only: the
548            // message-assignment form is an instruction and never an
549            // expression, so `assigned` is `None` here.
550            ExprView::Message {
551                target,
552                name,
553                super_class,
554                args,
555                cascade,
556            } => pinned!(
557                self,
558                crate::pinning::PinKind::TreeSend,
559                self.message_term(
560                    code,
561                    &crate::dispatch::MessageTerm {
562                        target,
563                        name,
564                        super_class,
565                        args,
566                        cascade,
567                        assigned: None,
568                    },
569                )
570            )?
571            .ok_or_else(|| Raised::no_result(name).into()),
572
573            _ => self.eval_cold(code, expr),
574        }
575    }

577    /// Every expression form the match above does not name: `(a, b, ...)`, and
578    /// the ones that fail loudly.
579    #[inline(never)]
580    fn eval_cold(&mut self, code: &Code<'_>, expr: ExprRef<'_>) -> Result<ObjRef, Failure> {
581        match expr.kind() {
582            // The list evaluation builds an array of the list's own length and fills the
583            // positions that were written.
584            ExprView::List(items) => self.eval_list(code, items),
585            // `ns:Name`, the qualified class lookup. The `>N>` line it owes
586            // is this hook's sibling below, in `trace_intermediate`.
587            ExprView::ClassResolver { namespace, name } => {
588                self.resolve_class_at(code, namespace, name)
589            }
590            // `ns:name(...)`, resolved against that namespace's public
591            // routines alone -- so none of `resolve_call`'s four steps runs,
592            // and a builtin of the same name is never reached. Measured,
593            // `rexx:length('abc')` is 43.902 rather than `3`.
594            ExprView::QualifiedCall {
595                namespace,
596                name,
597                args,
598            } => {
599                let namespace = code.symbols.name(namespace).as_bytes();
600                let name = code.symbols.name(name).as_bytes();
601                self.eval_call_resolved(code, CallResolution::Qualified { namespace }, name, args)
602            }
603            other => Err(Loud::expression(other).into()),
604        }
605    }

621    /// `NodeKind::List`: a fresh `.Array` whose slots are the list's
622    /// positions.
623    #[inline(never)]
624    fn eval_list(&mut self, code: &Code<'_>, items: Args<'_>) -> Result<ObjRef, Failure> {
625        let frame = self.roots.activity_mut().push_frame();
626        let indent = self.activity.clause_state.current_value_indent;
627        let mut slots = Vec::with_capacity(items.len());
628        for item in items.iter() {
629            let Some(expr) = item else {
630                slots.push(None);
631                continue;
632            };
633            let value = self.eval(code, expr)?;
634            self.roots.activity_mut().push_temp(value);
635            if let Some(rendered) = self.intermediate_text(value)? {
636                self.trace_argument(indent, &rendered);
637            }
638            slots.push(Some(value));
639        }
640        let array = self.alloc_charged(interrexx_core::BehaviourId::ARRAY, Body::array(slots));
641        self.roots.activity_mut().push_temp(array);
642        if let Some(rendered) = self.result_text(array)? {
643            self.trace_result(indent, &rendered);
644        }
645        self.roots.activity_mut().pop_frame(frame);
646        Ok(array)
647    }

680    /// [`Interp::eval_call_resolved`] up to the point its callee's body would
681    /// run.
682    pub(crate) fn begin_eval_call(
683        &mut self,
684        code: &Code<'_>,
685        resolution: CallResolution<'_>,
686        name: &[u8],
687        args: Args<'_>,
688    ) -> Result<Started<ObjRef>, Failure> {
689        // **A builtin goes straight to its own entry point**, which is the
690        // same call `invoke_call` would make and answers the value this
691        // function wants: none of `function_value`'s arms can apply to it, since
692        // it runs no activation and so can neither exit nor return nothing.
693        // What the detour cost is the `Ended` -- wider than a register pair,
694        // so built in memory here and read back out one line later, on the
695        // path every `length(...)`/`substr(...)` in a program takes.
696        if let CallResolution::Settled(Resolved::Builtin(target)) = resolution {
697            return self
698                .invoke_builtin_call(code, target, name, args)
699                .map(Started::Ran);
700        }
701        // `CallType::Function`: this is the function-invocation route, and a
702        // `::ROUTINE` reached this way answers `FUNCTION` as `PARSE SOURCE`'s
703        // second word where the same body reached by `CALL` answers
704        // `SUBROUTINE`. Measured in one program, the same routine both ways.
705        let begun = self.begin_invoke_call(code, resolution, name, args, CallType::Function)?;
706        function_started(begun, name)
707    }

728    /// `op value` for the prefix operators `+`, `-` and `\`.
729    pub(crate) fn apply_prefix(&mut self, op: PrefixOp, value: ObjRef) -> Result<ObjRef, Failure> {
730        // **An integer operand within DIGITS stays on the tagged integers**,
731        // as `0 + a` does through [`Interp::arith_small_int`]: `+a` is `0 + a`
732        // and `-a` is `0 - a`, the operand needs no rounding so there is no
733        // lost-digits check to make, and the answer is the one the general
734        // path below renders. Anything else -- a value spelled with blanks, a
735        // sign, an exponent or more digits than DIGITS -- falls through.
736        let dyadic = match op {
737            PrefixOp::Plus => Some(Operator::Plus),
738            PrefixOp::Minus => Some(Operator::Subtract),
739            PrefixOp::Not => None,
740        };
741        if let Some(dyadic) = dyadic
742            && let Some(operand) = small_int_operand(value)
743        {
744            let digits = self.activation().settings.digits();
745            if let Some(result) = small_int_arith(dyadic, 0, operand, digits) {
746                return Ok(result);
747            }
748        }
749        let result = self.apply_prefix_body(op, value);
750        // Blamed on any failure, the same reason `Interp::arith_general`
751        // gives: measured, `s. = 'abc'; say \s.` is 34.901 with the frame,
752        // even though `\` never converts its operand to a number at all --
753        // the frame belongs to the forwarded activation, not to which of
754        // its checks raised.
755        if result.is_err() {
756            self.blame_forwarded_operator(op.spelling().as_bytes(), value);
757        }
758        result
759    }

761    /// [`Interp::apply_prefix`]'s own computation, wrapped by it for the
762    /// reason [`Interp::arith_general_body`] is.
763    fn apply_prefix_body(&mut self, op: PrefixOp, value: ObjRef) -> Result<ObjRef, Failure> {
764        let result = match op {
765            PrefixOp::Plus | PrefixOp::Minus => {
766                let number = match self.arith_left_operand(op.spelling(), value)? {
767                    ArithOperand::Number(number) => number,
768                    ArithOperand::Send(target) => {
769                        return self.send_operator(op.spelling(), target, &[]);
770                    }
771                };
772                self.lostdigits_check(&number, value)?;
773                let digits = self.activation().settings.digits();
774                let form = self.activation().settings.form();
775                let result = if op == PrefixOp::Plus {
776                    Number::zero().add(&number, digits)
777                } else {
778                    Number::zero().sub(&number, digits)
779                }
780                .map_err(Raised::from)?;
781                self.number(result, saturate_digits(digits), form)
782            }
783            PrefixOp::Not => {
784                // `\.array` is 97.1, the same message send the dyadic operators
785                // make, and it is asked ahead of the truth test for the reason
786                // [`Interp::logical_values_body`] states.
787                if let Some(target) = self.operator_message_receiver(value) {
788                    return self.send_operator(op.spelling(), target, &[]);
789                }
790                if let Some(kind) = self.operator_operand_gap(value) {
791                    return Err(Loud::operator_operand(op.spelling(), kind).into());
792                }
793                let holds = self.truth(value, Raised::not_logical)?;
794                logical(!holds)
795            }
796        };
797        Ok(result)
798    }

800    /// The seven arithmetic operators, sharing one operand-evaluation and
801    /// error-conversion path.
802    fn eval_arithmetic(
803        &mut self,
804        code: &Code<'_>,
805        op: Operator,
806        left: ExprRef<'_>,
807        right: ExprRef<'_>,
808    ) -> Result<ObjRef, Failure> {
809        let frame = self.roots.activity_mut().push_frame();
810        let left_value = self.eval(code, left)?;
811        self.roots.activity_mut().push_temp(left_value);
812        let right_value = self.eval(code, right)?;
813        self.roots.activity_mut().push_temp(right_value);
814
815        // The two halves in the order they are tried, which is the whole of
816        // what this function decides once its operands are values. Both are
817        // entered from `crate::ir::Op::Arith` as well, which is why they are
818        // functions rather than the two blocks they used to be: the compiled
819        // op has its operands in registers rather than in nodes, and
820        // everything past that point is the same arithmetic.
821        let value = match self.arith_small_int(op, left_value, right_value) {
822            Some(value) => value,
823            None => self.arith_general(op, left_value, right_value)?,
824        };
825
826        self.roots.activity_mut().pop_frame(frame);
827        Ok(value)
828    }

830    /// `left op right` on the small-integer path, or `None` when the general
831    /// path must run instead. Inlined into the driver's arithmetic arm, which
832    /// is what keeps that arm's common case free of a call.
833    #[inline(always)]
834    pub(crate) fn arith_small_int(
835        &self,
836        op: Operator,
837        left_value: ObjRef,
838        right_value: ObjRef,
839    ) -> Option<ObjRef> {
840        let digits = self.activation().settings.digits();
841        // **The tagged pair is tested first and on its own**, exactly as it
842        // was before the spelled operand was admitted. Reading a spelled one
843        // costs a scan of its bytes, and a clause whose operands are both
844        // already tagged must not pay for a widening it cannot use.
845        match (left_value.decode(), right_value.decode()) {
846            (Decoded::SmallInt(left_int), Decoded::SmallInt(right_int)) => {
847                small_int_arith(op, left_int, right_int, digits)
848            }
849            _ => spelled_int_arith(op, left_value, right_value, digits),
850        }
851    }

853    /// `left op right` through `interrexx-num`, the path every operand shape
854    /// reaches and the one [`Interp::arith_small_int`] falls through to.
855    pub(crate) fn arith_general(
856        &mut self,
857        op: Operator,
858        left_value: ObjRef,
859        right_value: ObjRef,
860    ) -> Result<ObjRef, Failure> {
861        let result = self.arith_general_body(op, left_value, right_value);
862        // **Blamed on any failure past this point, not only the left operand's
863        // own conversion.** The forwarded native method runs to completion or
864        // fails; whichever of its own steps raises -- the base's conversion,
865        // the argument's, or the arithmetic itself -- is still inside that one
866        // activation, each of them past `arith_left_operand`'s own return. So
867        // `s. = 1; say s. / 0` is 42.3 with the frame, `s. = 1; say s. **
868        // 999999999999` is 26.8 with the frame (the base converts fine; the
869        // *exponent* fails), and `s. = 1; say s. + .array` is 41.1 with the
870        // frame even though it is the *right* operand's own conversion that
871        // fails. The receiver test is `blame_forwarded_operator`'s own, so this
872        // site gates on failure alone and hands it every failing path here,
873        // leaving a non-stem receiver for that predicate to discard.
874        if result.is_err() {
875            self.blame_forwarded_operator(op.spelling().as_bytes(), left_value);
876        }
877        result
878    }

880    /// [`Interp::arith_general`]'s own computation, wrapped by it so every
881    /// failing path -- not only [`Interp::arith_left_operand`]'s -- is
882    /// blamed once, in the one place that already knows the receiver.
883    fn arith_general_body(
884        &mut self,
885        op: Operator,
886        left_value: ObjRef,
887        right_value: ObjRef,
888    ) -> Result<ObjRef, Failure> {
889        let digits = self.activation().settings.digits();
890        let form = self.activation().settings.form();
891
892        // **An addition, subtraction, multiplication, `%` or `//` of two
893        // short numbers on their scaled mantissas**, where `Scaled` answers
894        // exactly what the `Number` operation would. Neither operand can be
895        // an operator receiver or fail to convert (a plain string or number
896        // is all `scaled_operand` reads). One that `+`, `-` or `*` accepts is
897        // no longer than DIGITS, so LOSTDIGITS has nothing to say; `%` and
898        // `//` accept one digit more and are not taken while LOSTDIGITS is
899        // trapped. The answer is made into a value exactly as the general
900        // path makes its own.
901        if !matches!(op, Operator::Divide | Operator::Power)
902            && let Some(answer) = self.scaled_arith(op, left_value, right_value, digits)
903        {
904            return Ok(match crate::value::scaled_handle(answer, digits) {
905                Some(handle) => handle,
906                None => self.number(answer.to_number(), saturate_digits(digits), form),
907            });
908        }
909
910        let left_number = match self.arith_left_operand(op.spelling(), left_value)? {
911            ArithOperand::Number(number) => number,
912            ArithOperand::Send(target) => {
913                return self.send_operator(op.spelling(), target, &[Some(right_value)]);
914            }
915        };
916
917        // The argument of the operator method the receiver answers, converted
918        // after the receiver itself is accepted -- `apply_binary`'s own doc has
919        // the details, and the arithmetic reports the **original argument**
920        // rather than the conversion when the converted string is not numeric:
921        // `'2' + .K` with a class-side `makeString` returning `'xx'` is 41.1
922        // with `The K class` as its item.
923        let converted = self.required_string_value(right_value)?;
924
925        let result = if op == Operator::Power {
926            let exponent = match self.to_number(converted) {
927                Ok(number) => number,
928                Err(NotNumeric) => {
929                    return Err(
930                        self.rendered_failure(right_value, Raised::power_exponent_not_whole)
931                    );
932                }
933            };
934            // The base alone: measured, `1.23456789 ** 1` at DIGITS 3 is
935            // 98.972 on the base and `2 ** 1234` is 26.8 on the *exponent*'s
936            // own whole-number conversion, which runs first and never
937            // reaches LOSTDIGITS.
938            self.lostdigits_check(&left_number, left_value)?;
939            match left_number.pow(&exponent, digits) {
940                // The exponent as the program wrote it, not the number
941                // parsed from it.
942                Err(ArithError::PowerExponentNotWhole { .. }) => {
943                    return Err(
944                        self.rendered_failure(right_value, Raised::power_exponent_not_whole)
945                    );
946                }
947                // Both operands as the program wrote them, not the numbers
948                // parsed from them.
949                Err(ArithError::PowerOverflow { .. }) => {
950                    let base = self.string_value_text(left_value)?;
951                    let exponent = self.string_value_text(right_value)?;
952                    return Err(Raised::power_overflow(&base, &exponent).into());
953                }
954                other => other,
955            }
956        } else {
957            let right_number = match self.to_number(converted) {
958                Ok(number) => number,
959                Err(NotNumeric) => {
960                    return Err(self.rendered_failure(right_value, Raised::nonnumeric));
961                }
962            };
963            // Left first: `987654321 + 123456789` at DIGITS 3 names 987654321.
964            // The right operand is named by its string value.
965            self.lostdigits_check2(&left_number, left_value, &right_number, converted)?;
966            match op {
967                Operator::Plus => left_number.add(&right_number, digits),
968                Operator::Subtract => left_number.sub(&right_number, digits),
969                Operator::Multiply => left_number.mul(&right_number, digits),
970                Operator::Divide => left_number.div(&right_number, digits, DivOp::Divide),
971                Operator::IntDiv => left_number.div(&right_number, digits, DivOp::IntegerDivide),
972                Operator::Remainder => left_number.div(&right_number, digits, DivOp::Remainder),
973                _ => unreachable!("eval_node only dispatches the seven arithmetic operators here"),
974            }
975        }
976        .map_err(Raised::from)?;
977
978        Ok(self.number(result, saturate_digits(digits), form))
979    }

981    /// [`Interp::arith_general_body`]'s `+`, `-`, `*`, `%` or `//` on two short
982    /// numbers, as [`Scaled`] answers it, or `None` where it is not that
983    /// exact short case.
984    #[inline(never)]
985    fn scaled_arith(
986        &mut self,
987        op: Operator,
988        left_value: ObjRef,
989        right_value: ObjRef,
990        digits: u64,

993        let right = self.scaled_operand(right_value)?;
994        match op {
995            Operator::Plus => left.add(right, digits),
996            Operator::Subtract => left.add(right.negated(), digits),
997            Operator::Multiply => left.mul(right, digits),
998            _ if self.lostdigits_armed => None,
999            _ => left.div_integer(right, digits, op == Operator::Remainder),
1000        }
1001    }

1042    /// Whether `value` is a stem or a variable reference whose forwarded
1043    /// operator would actually reach a native method to run: true for an
1044    /// unset stem (its own name is a `String`), an unset referenced variable
1045    /// (its name), and one whose value is itself `String`/`Number`-valued;
1046    /// false for `.nil` and for anything [`Interp::operator_operand_gap`]
1047    /// already named.
1048    fn is_forwarding_receiver(&self, value: ObjRef) -> bool {
1049        let Decoded::Heap { .. } = value.decode() else {
1050            return false;
1051        };

1111    ) -> Result<ObjRef, Failure> {

1118    /// The shared body of `||`/`Abuttal` (no separator) and `Blank` (one
1119    /// space).
1120    fn concat_values(
1121        &mut self,
1122        left_value: ObjRef,
1123        right_value: ObjRef,
1124        separator: Option<u8>,
1125    ) -> Result<ObjRef, Failure> {
1126        // **Nearly every operand is a string or a whole number**, whose bytes
1127        // `plain_bytes` reaches with one look, and the join is written
1128        // straight into the handle or into the new string's body. Any other
1129        // value is rendered first.
1130        let (mut left_slot, mut right_slot) = (PieceSlot::new(), PieceSlot::new());
1131        if let (Some(left), Some(right)) = (
1132            self.plain_bytes(left_value, &mut left_slot),
1133            self.plain_bytes(right_value, &mut right_slot),
1134        ) {
1135            let total = left.len() + usize::from(separator.is_some()) + right.len();
1136            let joined = if total <= INLINE_TEXT {
1137                let mut buffer = [0u8; INLINE_TEXT];
1138                buffer[..left.len()].copy_from_slice(left);
1139                let mut at = left.len();
1140                if let Some(byte) = separator {
1141                    buffer[at] = byte;
1142                    at += 1;
1143                }
1144                buffer[at..total].copy_from_slice(right);
1145                ObjRef::inline_text(&buffer[..total]).expect("a join no longer than the handle")
1146            } else {
1147                // Handed to `text_bytes` as `Bytes::joined` built it, which
1148                // is that function's own condition for a body built in its
1149                // slot rather than copied there.
1150                let bytes = Bytes::joined(left, separator, right);
1151                let joined = self.text_bytes(bytes);
1152                self.charge_text(total);
1153                joined
1154            };
1155            self.fresh_concat = joined;
1156            return Ok(joined);
1157        }
1158        // Any other operand is rendered, and both operands' bytes are read
1159        // through shared borrows, which can be live at once.
1160        let left_rendered = self.render(left_value);
1161        let right_rendered = self.render(right_value);
1162        let left_bytes = left_rendered.text(self);
1163        let right_bytes = right_rendered.text(self);
1164        // **A join that fits the object's own inline bytes is built on the
1165        // stack and never touches the lent buffer at all.** `text_built` ends
1166        // such a join by copying it into the value and handing the buffer
1167        // straight back, so everything the `Vec` did for it -- the take, the
1168        // reservation, the length bookkeeping, the return -- was setup for a
1169        // container the result does not end up in. The bound is the one
1170        // `text_built` branches on, so this arm and its own are the same
1171        // arm -- what differs is that the buffer is never taken.
1172        let total = left_bytes.len() + usize::from(separator.is_some()) + right_bytes.len();
1173        if total <= INLINE_BYTES {
1174            let mut buffer = [0u8; INLINE_BYTES];
1175            let (head, rest) = buffer.split_at_mut(left_bytes.len());
1176            head.copy_from_slice(left_bytes);
1177            let rest = match separator {
1178                Some(byte) => {
1179                    let (slot, rest) = rest.split_at_mut(1);
1180                    slot[0] = byte;
1181                    rest
1182                }
1183                None => rest,
1184            };
1185            rest[..right_bytes.len()].copy_from_slice(right_bytes);
1186            let joined = self.text(&buffer[..total]);
1187            self.fresh_concat = joined;
1188            return Ok(joined);
1189        }
1190        // **Built in the lent buffer and finished with `text_built`**, which
1191        // is what makes the one remaining allocation conditional rather than
1192        // certain. Only a result too long to live inline reaches here, and it
1193        // keeps the buffer, which is the same trade `text_built` always makes.
1194        let mut bytes = self.take_result_buffer();
1195        bytes.reserve(left_bytes.len() + usize::from(separator.is_some()) + right_bytes.len());
1196        bytes.extend_from_slice(left_bytes);
1197        if let Some(byte) = separator {
1198            bytes.push(byte);
1199        }
1200        bytes.extend_from_slice(right_bytes);
1201        let joined = self.text_built(bytes);
1202        self.fresh_concat = joined;
1203        Ok(joined)
1204    }

1206    /// `left op right` for a concatenation op whose left operand is the
1207    /// string the previous concatenation of the same chain built: `right` is
1208    /// appended to `left` itself, which is answered. The caller has checked
1209    /// that `left` is [`Interp::fresh_concat`], which is what makes it a
1210    /// string nothing else holds -- so the append is the concatenation's own
1211    /// answer, and the copy of `left` it saves is invisible. Anything other
1212    /// than a plain text body takes [`Interp::apply_binary`] instead.
1213    pub(crate) fn concat_onto(
1214        &mut self,
1215        op: Operator,
1216        left: ObjRef,
1217        right: ObjRef,
1218    ) -> Result<ObjRef, Failure> {
1219        debug_assert!(
1220            is_concatenation(op),
1221            "concat_onto is for the concatenation operators"
1222        );
1223        debug_assert_eq!(
1224            left, self.fresh_concat,
1225            "concat_onto on a string not just built"
1226        );
1227        if !matches!(
1228            self.heap
1229                .get(left)
1230                .map(|object| (&object.body, object.behaviour)),
1231            Some((Body::Text { .. }, interrexx_core::BehaviourId::STRING))
1232        ) {
1233            return self.apply_binary(op, left, right);
1234        }
1235        // `apply_binary`'s own sequence for a string receiver: the right
1236        // operand through the required-string protocol, then the join.
1237        let right = self.required_string_value(right)?;
1238        // An operand whose bytes live in its handle is appended from a local
1239        // copy; one in the heap goes through the lent buffer, because its
1240        // bytes cannot be borrowed while the string they join is written.
1241        let mut slot = PieceSlot::new();
1242        let mut held = [0u8; 1 + TEXT_SCRATCH];
1243        let mut piece = None;
1244        let joined: &[u8] = if let Some(right) = slot.handle_bytes(right) {
1245            let start = usize::from(op != Operator::Blank);
1246            held[0] = b' ';
1247            held[1..=right.len()].copy_from_slice(right);
1248            &held[start..=right.len()]
1249        } else {
1250            let mut buffer = self.take_result_buffer();
1251            if op == Operator::Blank {
1252                buffer.push(b' ');
1253            }
1254            let mut heap_slot = PieceSlot::new();
1255            match self.plain_bytes(right, &mut heap_slot) {
1256                Some(bytes) => buffer.extend_from_slice(bytes),
1257                None => {
1258                    let rendered = self.render(right);
1259                    buffer.extend_from_slice(rendered.text(self));
1260                }
1261            }
1262            piece.insert(buffer)
1263        };
1264        let Some(object) = self.heap.get_mut(left) else {
1265            unreachable!("a register roots the string a chain is building");
1266        };
1267        let Body::Text { bytes, num } = &mut object.body else {
1268            unreachable!("checked to be a text body above");
1269        };
1270        let before = bytes.heap_len();
1271        bytes.extend_from_slice(joined);
1272        *num = None;
1273        let after = bytes.heap_len();
1274        if let Some(buffer) = piece {
1275            self.give_result_buffer(buffer);
1276        }
1277        if after > before {
1278            self.charge_body_bytes(after - before);
1279        }
1280        // Set again: evaluating the right operand may have built strings of
1281        // its own.
1282        self.fresh_concat = left;
1283        Ok(left)
1284    }

1286    /// The comparison operators (numeric-or-string `= \= <> >< > < >= <= \> \<`, and strict
1287    /// `== \== >> << >>= <<= \>> \<<`), all through `interrexx-num`'s
1288    /// `compare_numbers` and `compare_strings` -- **no comparison rule is
1289    /// written here**, per the plan's own instruction and this crate's
1290    /// standing rule against a second copy of logic `interrexx-num` already owns
1291    /// (the same rule `compare.rs`'s own module doc states for its entry
1292    /// points). Those two are `compare_decoded`'s own arms, called directly
1293    /// because this function has already decided which arm applies: each
1294    /// arm's own doc comment says so, and says what a caller buys by
1295    /// choosing.
1296    fn compare_values(
1297        &mut self,
1298        op: Operator,
1299        left_value: ObjRef,
1300        right_value: ObjRef,
1301    ) -> Result<ObjRef, Failure> {
1302        let result = self.compare_values_body(op, left_value, right_value);
1303        // Blamed on any failure, the same reason `Interp::arith_general`
1304        // gives -- and comparison does have one, corrected from this task's
1305        // own earlier premise: `compare_numbers` converts both operands to
1306        // a `Number` and can overflow when both are numeric, exactly as an
1307        // arithmetic operator's operands can. Measured: `numeric digits 1;
1308        // s. = '9.9E999999999'; say s. > 1` is rc 214 with the frame; the
1309        // same overflow with `1 > s.` (the receiver `1`, not a stem) is the
1310        // identical rc 214 with no frame, because `is_forwarding_receiver`
1311        // answers `false` for the plain-number receiver regardless of which
1312        // operand overflowed. Strict `==`/`>>` and the non-numeric fallback
1313        // (`compare_strings`) never reach `compare_numbers` at all, so they
1314        // stay rc 0 and frameless, unaffected by this wrapping.
1315        if result.is_err() {
1316            self.blame_forwarded_operator(op.spelling().as_bytes(), left_value);
1317        }
1318        result
1319    }

1321    /// [`Interp::compare_values`]'s own computation, wrapped by it for the
1322    /// reason [`Interp::arith_general_body`] is.
1323    fn compare_values_body(
1324        &mut self,
1325        op: Operator,
1326        left_value: ObjRef,
1327        right_value: ObjRef,
1328    ) -> Result<ObjRef, Failure> {
1329        // Every `&mut` read this comparison needs happens first -- the two
1330        // parses, and the two settings -- so that the byte reads below can be
1331        // shared borrows taken together. `to_number` hands back an owned
1332        // `Number`, so nothing here outlives its own statement, and the
1333        // operand copies the old order forced are gone.
1334        let digits = self.activation().settings.digits();
1335        let fuzz = self.activation().settings.fuzz();
1336        // **Two tagged small integers answer this comparison themselves**,
1337        // where the general path renders both operands to text first --
1338        // measured with `heaptrack` on `samples/rexxcps.rex`, that rendering
1339        // was the largest single allocation width in the interpreter, at
1340        // 4,220,217 allocations of nineteen bytes on a 400,000-clause run
1341        // (`SMALL_INT_MAX` has nineteen digits, which is what pre-sizes them).
1342        if fuzz == 0
1343            && let Some(holds) = small_int_compare(op, left_value, right_value, digits)
1344        {
1345            return Ok(logical(holds));
1346        }
1347        let strict = is_strict_compare(op);
1348        // **Operands in their handles, compared without a `Number` built**:
1349        // two numbers that align within DIGITS compare exactly on their
1350        // scaled mantissas, and two texts at least one of which is not a
1351        // number take the string rule on the bytes the handles hold. Neither
1352        // has an operator gap to report, and an operand that aligns within
1353        // DIGITS loses no digits, so nothing below would raise.
1354        if !strict && fuzz == 0 {
1355            match (
1356                self.quick_operand(left_value),
1357                self.quick_operand(right_value),
1358            ) {
1359                (QuickOperand::Scaled(left, _), QuickOperand::Scaled(right, _)) => {
1360                    if let Some(holds) = left.compare(right, digits, compare_op(op)) {
1361                        return Ok(logical(holds));
1362                    }
1363                }
1364                (
1365                    QuickOperand::NotNumber(left) | QuickOperand::Scaled(_, Some(left)),
1366                    QuickOperand::NotNumber(right),
1367                )
1368                | (QuickOperand::NotNumber(left), QuickOperand::Scaled(_, Some(right))) => {
1369                    return Ok(logical(interrexx_num::compare_strings(
1370                        &left,
1371                        &right,
1372                        compare_op(op),
1373                    )));
1374                }
1375                _ => {}
1376            }
1377        }
1378        // **The `Result` is kept rather than turned into an `Option`.** A
1379        // `Number` is forty bytes and `Result::ok` moves one, so the two calls
1380        // here moved eighty bytes for nothing: every read below is `is_ok`,
1381        // `is_err` or a borrow out of the `Ok` arm, and none of them needs an
1382        // `Option`. Measured on `samples/rexxcps.rex`, -0.531% of its retired
1383        // instructions and -0.60% of its cycles over six alternating runs --
1384        // the first pair of runs read -5%, which was the cold start and not
1385        // this.
1386        let left_number = if strict {
1387            Err(NotNumeric)
1388        } else {
1389            self.to_number(left_value)
1390        };
1391        // **Behind the fast path above, because a comparison of renderings
1392        // never fails and so offers nothing to ride.** Comparing an object
1393        // against the very text it renders as answers `0` and not `1`; see
1394        // `Loud::operator_operand`.
1395        debug_assert!(
1396            left_number.is_err()
1397                || unshared!(self, self.operator_operand_gap(left_value)).is_none(),
1398            "a left operand that parsed as a number reported an operator gap"
1399        );
1400        if left_number.is_err()
1401            && let Some(kind) = self.operator_operand_gap(left_value)
1402        {
1403            return Err(Loud::operator_operand(op.spelling(), kind).into());
1404        }
1405        let right_number = if strict {
1406            Err(NotNumeric)
1407        } else {
1408            self.to_number(right_value)
1409        };
1410
1411        // **Both operands parsed means the bytes have no reader**, so they are
1412        // not produced. `compare_decoded`'s own `(Some, Some)` arm is
1413        // `compare_numbers` and nothing else, and it is handed the same two
1414        // values this would have passed it -- so this is where the rendering
1415        // happens rather than what the comparison answers. What it skips is
1416        // `render` plus `text` on both sides, whose result reaches only the
1417        // string fallback below.
1418        if let (Ok(left), Ok(right)) = (&left_number, &right_number) {
1419            // The **numeric** arm only: `1.23456789 = 'abc'` at DIGITS 3 is rc
1420            // 0, because a comparison that falls through to the string rule
1421            // converts no operand at all. A strict operator never reaches here
1422            // either.
1423            self.lostdigits_check_compare(left, left_value, right, right_value, fuzz)?;
1424            let holds = interrexx_num::compare_numbers(left, right, digits, fuzz, compare_op(op))
1425                .map_err(Raised::from)?;
1426            return Ok(logical(holds));
1427        }
1428        // Either operand failed to parse, or the operator is strict and neither
1429        // was parsed at all. Both routes compare the operands' own text, which
1430        // is what these two renderings are for.
1431        let left_rendered = self.render(left_value);
1432        let right_rendered = self.render(right_value);
1433        let left_bytes = left_rendered.text(self);
1434        let right_bytes = right_rendered.text(self);
1435        // The claim the line below rests on, checked rather than argued: an
1436        // operand `to_number` refused is one whose own rendering does not
1437        // parse either, so the arm `compare_decoded` would have reached after
1438        // reparsing is the arm taken here. A strict operator is exempt because
1439        // it consults no `Number` at all.
1440        debug_assert!(
1441            strict || left_number.is_ok() || Number::parse_bytes(left_bytes).is_none(),
1442            "a left operand to_number refused parses from its own rendering"
1443        );
1444        debug_assert!(
1445            strict || right_number.is_ok() || Number::parse_bytes(right_bytes).is_none(),
1446            "a right operand to_number refused parses from its own rendering"
1447        );
1448        let holds = interrexx_num::compare_strings(left_bytes, right_bytes, compare_op(op));
1449
1450        let result = logical(holds);
1451        Ok(result)
1452    }

1454    /// `&`/`|`/`&&`. **Both operands are always evaluated and checked,
1455    /// never short-circuited**, and the left is checked before the right
1456    /// when both are bad -- measured: `say (0 & 'x')` and `say (1 | 'x')`
1457    /// both raise 34.901 on `"x"` even though the result is already
1458    /// decided by the first operand alone, and `say ('y' & 'x')` (both
1459    /// bad) reports `"y"`, the left operand's own text, not `"x"`. The
1460    /// opposite rule from `NodeKind::Logical`'s comma list, below, which
1461    /// does short-circuit -- both are implemented exactly as measured rather
1462    /// than made to agree with each other. The evaluation half of that rule
1463    /// belongs to whoever calls this: the check half is here, and it checks
1464    /// the operand it was handed second even when the first already decided
1465    /// the answer.
1466    fn logical_values(
1467        &mut self,
1468        op: Operator,
1469        left_value: ObjRef,
1470        right_value: ObjRef,
1471    ) -> Result<ObjRef, Failure> {

1473        // Blamed on any failure, the same reason `Interp::arith_general`
1474        // gives: measured, `s. = 1; say s. & 'x'` is 34.901 with the frame
1475        // even though it is the *right* operand's own text that fails the
1476        // logical check.
1477        if result.is_err() {
1478            self.blame_forwarded_operator(op.spelling().as_bytes(), left_value);
1479        }
1480        result
1481    }

1483    /// [`Interp::logical_values`]'s own computation, wrapped by it for the
1484    /// reason [`Interp::arith_general_body`] is.
1485    fn logical_values_body(
1486        &mut self,
1487        op: Operator,
1488        left_value: ObjRef,
1489        right_value: ObjRef,
1490    ) -> Result<ObjRef, Failure> {
1491        // **Ahead of the truth test, not behind it.** `&`/`|`/`&&` are sent to
1492        // the left operand as a message -- `.array & 1` is 97.1 where this
1493        // crate's own 34.901 quotes the object's rendering -- so an operand
1494        // this names refuses however it renders, and a rendering that is itself
1495        // `0` or `1` is not an answer: `o & 1` for an instance named `'1'` is
1496        // `97.1`, naming the object `1` and the message `&`.
1497        if let Some(kind) = self.operator_operand_gap(left_value) {
1498            return Err(Loud::operator_operand(op.spelling(), kind).into());
1499        }
1500        let left_bool = self.truth(left_value, Raised::not_logical)?;
1501        let right_bool = self.truth(right_value, Raised::not_logical)?;
1502
1503        let holds = match op {
1504            Operator::And => left_bool && right_bool,
1505            Operator::Or => left_bool || right_bool,
1506            Operator::Xor => left_bool != right_bool,
1507            other => return Err(Loud::binary_operator(other).into()),
1508        };
1509
1510        let result = logical(holds);
1511        Ok(result)
1512    }

1514    /// The object an operator is **sent to as a message**, or `None` for an
1515    /// operand the operator converts itself.
1516    pub(crate) fn operator_message_receiver(&self, value: ObjRef) -> Option<ObjRef> {
1517        // **`.nil` is an operator receiver.** It answers `==`, `\\==`, `=`,
1518        // `<>` and `><` by identity and understands no other operator, so an
1519        // ordering or arithmetic operator against it is 97.1 -- measured,
1520        // `.nil < 'x'` and `.nil + 1` both raise it, with `The NIL object` as
1521        // the receiver item. Reaching that through the send is
1522        // what `Object`'s own operator methods do; converting `.nil` to its
1523        // string value instead makes it equal to the text `The NIL object`
1524        // and orderable against anything.
1525        if value == ObjRef::NIL {
1526            return Some(value);
1527        }
1528        // A small integer and an inline string leave on this line, which is
1529        // what keeps this off the cost of a comparison between two numbers.
1530        let Decoded::Heap { .. } = value.decode() else {
1531            return None;
1532        };
1533        match &self.heap.get(value)?.body {
1534            // **A class object is an operator receiver on the same terms `.nil`
1535            // is.** `Class` declares `=`, `==`, `\\=`, `\\==`, `<>` and `><`
1536            // and no other operator, so the six are `Object`'s identity test
1537            // and every other operator is a name the behaviour does not hold:
1538            // `.array = .array` is `1` and `.array = 'The Array class'` is `0`;
1539            // `.array > .array` and `.array + 1` both raise 97.1 with
1540            // `The Array class` as the receiver item.
1541            Body::Class { .. } => Some(value),
1542            Body::Instance { .. } => Some(value),
1543            // Neither `Array` nor a class this crate builds as a native body
1544            // declares an operator but `Object`'s and `Pointer`'s, so the
1545            // send answers by identity or 97.1.
1546            Body::Array { .. } | Body::Native(_) => Some(value),
1547            Body::Stem {
1548                default: Some(default),
1549                ..
1550            } => self.operator_message_receiver(*default),
1551            Body::VarRef(_) => self.reference_message_receiver(value),
1552            _ => None,
1553        }

1602    /// The noun for an operand no operator here can take, or `None` for one
1603    /// every operator can.
1604    pub(crate) fn operator_operand_gap(&self, value: ObjRef) -> Option<&'static str> {
1605        // A small integer, an inline string and `.nil` all leave on this
1606        // line: only a heap-tagged handle can be either shape.
1607        let Decoded::Heap { .. } = value.decode() else {
1608            return None;
1609        };
1610        match &self.heap.get(value)?.body {
1611            Body::Class { .. } => Some("a class object"),
1612            Body::Native(_) => Some("one of the interpreter's own objects"),
1613            // An array *does* have a string value -- `Array~makeString` joins
1614            // its items -- so concatenation, which never asks here, is
1615            // unaffected. An operator that does ask never gets that string
1616            // value: `.Array` defines no arithmetic or logical method, so `a =
1617            // (1,); say a + 1` is 97.1; and where a comparison finds one it is
1618            // `Object`'s own identity test, so `say (a = 1)` is `0` at rc 0
1619            // rather than a comparison of `1` against `1`.
1620            Body::Array { .. } => Some("an array"),
1621            // The operator is sent as a message here too, and it is the send
1622            // that fails: `o + 1` on an instance is `97.1`, naming the object `a K` and
1623            // the message `+`, and the same after `o~objectName =
1624            // '123'` -- the rendering being numeric does not make it a
1625            // conversion.
1626            Body::Instance { .. } => Some("an instance of a user class"),
1627            // **The redirect every conversion here takes, taken here too.** A
1628            // stem with a default answers *as* that default -- `to_text` and
1629            // `to_number` both chase it -- so a check that stopped at the stem
1630            // handle would let `a. = .array; say (a. == 'The Array class')`
1631            // answer 1 where 0 is right, which is this gap's original shape
1632            // reached through one assignment. A compound read (`a.zz`) yields
1633            // the class handle itself and needs no redirect, which is what says
1634            // the hole was the indirection rather than the operator arm.
1635            Body::Stem {
1636                default: Some(default),
1637                ..
1638            } => self.operator_operand_gap(*default),
1639            // The redirect one arm up, for the same reason: every operator
1640            // reaches the referenced value through `UNKNOWN`, so it is that
1641            // value the message is sent to. For example, `zz = .array~new(2);
1642            // say (>zz) + 1` is `97.1` naming the array and the message `+`
1643            // -- the array's own answer, substituting the array
1644            // rather than the reference.
1645            Body::VarRef(_) => self.reference_operand_gap(value),
1646            _ => None,
1647        }
1648    }

1684    /// `left op right` by the string rule, where both are strings or numbers
1685    /// in their own right -- a small integer, an inline string or a heap
1686    /// string or number -- and `None` for any other operand, which
1687    /// [`Interp::apply_binary`] has to judge. For a comparison operator with
1688    /// an operand that is not a number this is the answer `apply_binary`
1689    /// gives: neither operand is an operator receiver or converts through
1690    /// the required-string protocol, the numeric attempt fails on the
1691    /// non-number, and the fallback compares the two renderings. Nothing
1692    /// here can raise, which is why no blame is needed.
1693    fn compare_plain_strings(&mut self, op: Operator, left: ObjRef, right: ObjRef) -> Option<bool> {
1694        if !self.plain_string(left) || !self.plain_string(right) {
1695            return None;
1696        }
1697        // **Against the null string, equality needs no ordering**: by the
1698        // string rule the other side is equal when it is all blanks, and
1699        // strictly when it is empty, so only its own bytes are read.
1700        // The handle every null string is made as, so one comparison of the
1701        // bits finds it; a null string with any other handle takes the
1702        // general route below, which answers the same.
1703        let null_handle = ObjRef::inline_text(b"");
1704        let null = |value: ObjRef| Some(value) == null_handle;
1705        let other = if null(right) {
1706            Some(left)
1707        } else if null(left) {
1708            Some(right)
1709        } else {
1710            None
1711        };
1712        if let Some(other) = other {
1713            let other = self.render(other);
1714            if let Some(holds) = interrexx_num::compare_with_null(other.text(self), compare_op(op))
1715            {
1716                return Some(holds);
1717            }
1718        }
1719        let left = self.render(left);
1720        let right = self.render(right);
1721        Some(interrexx_num::compare_strings(
1722            left.text(self),
1723            right.text(self),
1724            compare_op(op),
1725        ))
1726    }

1729    /// own string value and no operator's receiver.

1733            Decoded::SmallInt(_) | Decoded::Text(_) => true,
1734            Decoded::Heap { .. } => matches!(
1735                self.heap.get(value).map(|object| &object.body),
1736                Some(Body::Text { .. } | Body::Num { .. })
1737            ),
1738            _ => false,
1739        }

1742    pub(crate) fn apply_binary(
1743        &mut self,
1744        op: Operator,
1745        left: ObjRef,
1746        right: ObjRef,
1747    ) -> Result<ObjRef, Failure> {
1748        // **`reqstr`'s dyadic-operator context is the operand on the *right*.** The section names
1749        // "Rexx dyadic operators when the receiving object (the object to the left of the operator)
1750        // is a string", and the receiver's own method is what converts its argument -- string
1751        // concatenation, comparison and logical operators all start by asking the right operand for
1752        // its string. The left operand is the receiver and is not converted at all: measured,
1753        // `.array + 1` is 97.1 where `1 + .array` is 41.1, and `eval.rs`'s `object_operand_tests`
1754        // is that half.
1755        if let Some(target) = self.operator_message_receiver(left) {
1756            if is_concatenation(op)
1757                && (self.as_variable_reference(left).is_some() || self.is_stem_object(left))
1758            {
1759                return self.concat_reference(op, left, right);
1760            }
1761            let answer = self.send_operator(op.spelling(), target, &[Some(right)]);
1762            // Whatever the method answered may be held anywhere, including a
1763            // string it built itself, so no chain may append to it.
1764            self.fresh_concat = ObjRef::NIL;
1765            return answer;
1766        }
1767        // **A string or number compared with `.nil` answers without reading it.** String
1768        // equality returns false for `.nil` before it looks at any bytes, and every other
1769        // comparison operator of `String`, `Integer` and `NumberString` answers `.false`
1770        // for it, except `\=`, `<>`, `><` and `\==`, which answer `.true`. So `'The NIL
1771        // object' == .nil` is `0`, `'a' < .nil` is `0` with no NOSTRING, and `'a' ||
1772        // .nil`, which is not a comparison, reads the string value.
1773        if is_comparison(op) {
1774            if right == ObjRef::NIL {
1775                return Ok(logical(is_inequality(op)));
1776            }
1777            let right = self.comparison_string_value(right)?;
1778            return self.compare_values(op, left, right);
1779        }
1780        let right = self.required_string_value(right)?;
1781        match op {
1782            Operator::Concatenate | Operator::Abuttal => self.concat_values(left, right, None),
1783            Operator::Blank => self.concat_values(left, right, Some(b' ')),
1784            Operator::And | Operator::Or | Operator::Xor => self.logical_values(op, left, right),
1785            op => Err(Loud::binary_operator(op).into()),
1786        }
1787    }

1789    /// `NodeKind::Logical`: the comma-separated conditional list `IF a, b,
1790    /// c THEN` (and `WHEN`/`GUARD`/`WHILE`/`UNTIL`'s own versions)
1791    /// desugars to, "a logical AND of its parts" (`ast.rs`'s own doc
1792    /// comment).
1793    fn eval_logical_list(&mut self, code: &Code<'_>, items: Args<'_>) -> Result<ObjRef, Failure> {
1794        let frame = self.roots.activity_mut().push_frame();
1795        let indent = self.activity.clause_state.current_value_indent;
1796        let mut holds = true;
1797        for item in items.iter().flatten() {
1798            let value = self.eval(code, item)?;
1799            if !self.logical_list_item(value, indent)? {
1800                holds = false;
1801                break;
1802            }
1803        }
1804
1805        let result = logical(holds);
1806        self.roots.activity_mut().pop_frame(frame);
1807        Ok(result)
1808    }

1956/// `Operator` -> `interrexx-num`'s `CompareOp`, for the `Operator` variants
1957/// `apply_binary` dispatches to `compare_values`. **Several operators share one
1958/// `CompareOp`**: `\=`/`<>`/`><` (`BackslashEqual`/`LessThanGreaterThan`/
1959/// `GreaterThanLessThan`) all mean `NotEqual` (`compare.rs`'s own doc comment:
1960/// the interpreter's operator table repeats one method pointer for them), and
1961/// `\>`/`\<` with their strict siblings `\>>`/`\<<` invert their positive
1962/// counterpart's sense rather than getting a `CompareOp` of their own: `\>`
1963/// ("not greater than") is `LessEqual`, `\<` is `GreaterEqual`, `\>>` is
1964/// `StrictLessEqual` and `\<<` is `StrictGreaterEqual`. `\==` is **not** among
1965/// them -- it has `StrictNotEqual` to itself -- and `\=` is one of the
1966/// operators sharing `NotEqual` above. Not derived from the names alone:
1967/// `\>>`/`\<<` are strict-family byte comparisons, where "greater"/"less" do
1968/// not carry the numeric intuition their spelling suggests.
1969fn compare_op(op: Operator) -> CompareOp {
1970    use Operator::*;
1971    match op {
1972        Equal => CompareOp::Equal,
1973        BackslashEqual | LessThanGreaterThan | GreaterThanLessThan => CompareOp::NotEqual,
1974        GreaterThan => CompareOp::Greater,
1975        LessThan => CompareOp::Less,
1976        GreaterThanEqual => CompareOp::GreaterEqual,
1977        LessThanEqual => CompareOp::LessEqual,
1978        BackslashGreaterThan => CompareOp::LessEqual,
1979        BackslashLessThan => CompareOp::GreaterEqual,
1980        StrictEqual => CompareOp::StrictEqual,
1981        StrictBackslashEqual => CompareOp::StrictNotEqual,
1982        StrictGreaterThan => CompareOp::StrictGreater,
1983        StrictLessThan => CompareOp::StrictLess,
1984        StrictGreaterThanEqual => CompareOp::StrictGreaterEqual,
1985        StrictLessThanEqual => CompareOp::StrictLessEqual,
1986        StrictBackslashGreaterThan => CompareOp::StrictLessEqual,
1987        StrictBackslashLessThan => CompareOp::StrictGreaterEqual,
1988        other => unreachable!(
1989            "apply_binary only dispatches the comparison operators here, got {other:?}"
1990        ),
1991    }
1992}

2099/// here, to decide whether calling `to_number` is worth it at all before
2100/// `compare_decoded` is reached (a strict comparison never looks at a
2101/// `Number`, so parsing one first would be pure waste).
2102/// Whether two tagged small integers settle `op` between them, and how.
2103fn small_int_compare(op: Operator, left: ObjRef, right: ObjRef, digits: u64) -> Option<bool> {
2104    let (Decoded::SmallInt(left), Decoded::SmallInt(right)) = (left.decode(), right.decode())
2105    else {
2106        return None;
2107    };
2108    // `i128` so the bound itself cannot overflow at the `DIGITS` a program may
2109    // set, and so `abs` has a value for `i64::MIN`.
2110    let bound = 10i128.checked_pow(u32::try_from(digits).ok()?)?;
2111    if i128::from(left).abs() >= bound || i128::from(right).abs() >= bound {
2112        return None;
2113    }
2114    Some(match compare_op(op) {
2115        CompareOp::Equal | CompareOp::StrictEqual => left == right,
2116        CompareOp::NotEqual | CompareOp::StrictNotEqual => left != right,
2117        CompareOp::Greater => left > right,
2118        CompareOp::Less => left < right,
2119        CompareOp::GreaterEqual => left >= right,
2120        CompareOp::LessEqual => left <= right,
2121        CompareOp::StrictGreater
2122        | CompareOp::StrictLess
2123        | CompareOp::StrictGreaterEqual
2124        | CompareOp::StrictLessEqual => return None,
2125    })
2126}
