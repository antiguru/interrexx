41    /// Why a directory is being read, which decides whether the security
42    /// manager sees the read at all. **Deliberately not `Copy`**, for the
43    /// reason the clearance beside it is not: one trip through the seam
44    /// carries one reason.
45    pub(super) enum Access {
46        /// The class lookup of the environment-symbol search, whose two
47        /// manager checks bracket the `.local` read.
48        Resolve,
49        /// Every other read. The activity's local-environment access and a
50        /// class directive's own search reach the directories directly, so a
51        /// manager never sees an output route or a directive's target.
52        Direct,
53    }
54
55    /// What the seam decided about one lookup.
56    pub(super) enum Admission {
57        /// The manager answered the name itself, and the directory is not
58        /// read at all.
59        Replaced(ObjRef),
60        /// Read the directory.
61        Permitted(Admitted),
62    }
63
64    /// Records the directory handles at bootstrap.
65    pub(super) fn hold(environment: ObjRef, local: ObjRef) -> Directories {
66        Directories { environment, local }
67    }

69    /// **The directory chokepoint (D12, site two).** Every read of `.local`
70    /// and of `.environment` passes here, and an [`Access::Resolve`] read is
71    /// the one the security manager's `LOCAL` and `ENVIRONMENT` checkpoints
72    /// see.
73    pub(super) fn admit(
74        interp: &mut Interp,
75        scope: EnvScope,
76        name: &[u8],
77        access: &Access,
78    ) -> Result<Admission, Failure> {
79        let manager = interp.package_security_manager();
80        if matches!(access, Access::Direct) || manager.is_none() {
81            return Ok(Admission::Permitted(Admitted(())));
82        }
83        let checkpoint = match scope {
84            EnvScope::Local => message::LOCAL,
85            EnvScope::Environment => message::ENVIRONMENT,
86        };
87        let index = interp.text(name);
88        interp.roots.activity_mut().push_temp(index);
89        let Some(info) = interp.security_check_by(manager, checkpoint, &[(key::NAME, index)])?
90        else {
91            return Ok(Admission::Permitted(Admitted(())));
92        };
93        // A manager that handles the name and sets no `RESULT` has hidden it:
94        // `checkLocalAccess` answers `OREF_NULL` for that, which its caller
95        // reads as a miss.
96        match interp.security_entry(info, key::RESULT)? {
97            Some(value) => Ok(Admission::Replaced(value)),
98            None => Ok(Admission::Permitted(Admitted(()))),
99        }
100    }

102    /// Which of the two `handle` is, or `None` for any other object.
103    pub(super) fn which(held: &Directories, handle: ObjRef) -> Option<EnvScope> {
104        if handle == held.environment {
105            Some(EnvScope::Environment)
106        } else if handle == held.local {
107            Some(EnvScope::Local)
108        } else {
109            None
110        }
111    }

332    fn environment_model(&mut self) -> &EnvironmentModel {
333        if self.environment.is_none() {
334            let model = self.build_environment();

336        }

338            .as_ref()
339            .expect("just built above if it was absent")

342    fn build_environment(&mut self) -> EnvironmentModel {
343        let context = self
344            .classes()
345            .lookup("RexxContext")
346            .expect("RexxContext is a native class");
347        let object_class = self
348            .classes()
349            .lookup("Object")
350            .expect("Object is a native class");
351
352        // Every object below is rooted the instant it exists and before the
353        // next allocation, which can collect.
354        let frame = self.roots.activity_mut().push_frame();
355        let mut owed: [ObjRef; 1] = [ObjRef::NIL];
356        for (at, held) in owed.iter_mut().enumerate() {
357            *held = self.alloc_with(
358                BehaviourId::OBJECT,
359                Body::Native(Box::new(NativeObject::new(
360                    object_class,
361                    b"an entry the library bootstrap fills",
362                ))),
363            );
364            self.roots.add_global(&format!(".owed{at}"), *held);
365        }
366        let environment =
367            self.interpreter_directory(ENVIRONMENT_ROOT, b"The Environment Directory");
368        let local = self.interpreter_directory(LOCAL_ROOT, b"The Local Directory");
369
370        // Native extensions can retain these handles across callbacks. The
371        // interpreter already roots both directories for its lifetime.
372        for object in [local, environment] {
373            self.borrow_native_handle(object);
374        }
375
376        // `completeSystemClass` puts every native class into the environment
377        // under its uppercased id, and that is the whole of what this crate
378        // has to put there beyond the entries `addToEnvironment` adds by
379        // hand.
380        let mut known: FxHashMap<Box<[u8]>, ObjRef> = self
381            .classes()
382            .registered()
383            .map(|(name, id)| (name.as_bytes().into(), id))
384            .collect();
385        // `addToEnvironment` registers these by hand.
386        let true_value = crate::eval::LOGICAL_TRUE;
387        let false_value = crate::eval::LOGICAL_FALSE;
388        // `REXXINFO` is a pre-built *instance* of a class no environment name
389        // reaches, which is why the entry renders as `a RexxInfo` and answers
390        // `~class~id` `RexxInfo`. So `.RexxInfo~isA(.Class)` is `0` and
391        // `.RexxInfo~class~superClass` is `The Object class`.
392        let rexx_info_class = self
393            .classes()
394            .system_lookup("RexxInfo")
395            .expect("RexxInfo is a native class in the kernel directory");
396        let rexx_info = self.alloc_with(
397            BehaviourId::OBJECT,
398            Body::Native(Box::new(NativeObject::new(rexx_info_class, b"a RexxInfo"))),
399        );
400        self.roots.activity_mut().push_temp(rexx_info);
401        // `.ENDOFLINE` is the platform's line terminator as a String --
402        // measured, `c2x(.endOfLine)` is `0A` here and its length is 1. The
403        // ooTest framework's own prologue reads it (`OOREXXUNIT.CLS:77`).
404        let end_of_line = self.text(crate::version::LINE_END);
405        self.roots.activity_mut().push_temp(end_of_line);
406        for (name, value) in [
407            (b"ENVIRONMENT".as_slice(), environment),
408            (b"NIL", ObjRef::NIL),
409            (b"TRUE", true_value),
410            (b"FALSE", false_value),
411            (b"REXXINFO", rexx_info),
412            (b"ENDOFLINE", end_of_line),
413        ] {
414            known.insert(name.into(), value);
415        }
416
417        // In the fixed order, so every entry sits at its position and a later
418        // write of a name replaces it in place. A name this crate has not built
419        // yet is owed until the library bootstrap puts it.
420        for name in ENVIRONMENT_SEQUENCE {
421            if *name == "LOCAL" {
422                continue;
423            }
424            let value = known.remove(name.as_bytes()).unwrap_or(owed[0]);
425            self.put_interpreter_entry(environment, name.as_bytes(), value);
426        }
427        let mut rest: Vec<(Box<[u8]>, ObjRef)> = known.into_iter().collect();
428        rest.sort_by(|a, b| a.0.cmp(&b.0));
429        for (name, value) in rest {
430            self.put_interpreter_entry(environment, &name, value);
431        }
432        // `LOCAL` is a method answering the running activity's local
433        // directory, and this crate runs one activity.
434        hash::directory_put_method_value(self, environment, b"LOCAL", local)
435            .expect("a string-keyed store accepts a string index");
436        // `Interp::mint_local_directory` builds these before any program runs,
437        // once the classes it needs exist.
438        for name in LOCAL_SEQUENCE {
439            self.put_interpreter_entry(local, name.as_bytes(), owed[0]);
440        }
441        self.roots.activity_mut().pop_frame(frame);
442
443        EnvironmentModel {
444            directories: env_seam::hold(environment, local),
445            owed,
446            views: [None, None],
447            context,
448        }
449    }

484    /// [`Interp::dot_variable`] for a compiled read, which keeps what the
485    /// class lookup answered -- a class, or that it found none -- in `memo`.
486    /// The memo is the running program's, and holds while
487    /// [`Interp::class_lookup_stamp`] and the class graph's epoch stand: the
488    /// class lookup reads the running package's tables and its parents', the
489    /// REXX package's, and the class registry's names, and a write to any of
490    /// them moves one or the other (the registry's names change only with a
491    /// class defined or collected, each of which moves the epoch).
492    ///
493    /// Past a class lookup that found none, the memo also keeps an item the
494    /// package local, `.local` or `.environment` held ([`EnvEntry`]), while no
495    /// hash store has been written. Those directories are this interpreter's
496    /// own, every change to their entries, methods or unknown method writes a
497    /// store or moves its generation, and the package local's miss is
498    /// answered under the same two counters
499    /// ([`Interp::package_local_symbol`]). What a security manager or a
500    /// stored method answered is never kept, and no item is kept while the
501    /// running package has a manager to ask: installing one writes the
502    /// package's data, which moves [`Interp::class_lookup_stamp`].
503    pub(crate) fn dot_variable_memoized(
504        &mut self,
505        dotted: &[u8],
506        memo: &Cell<Option<EnvMemo>>,
507    ) -> Result<ObjRef, Failure> {
508        let known = memo.get().filter(|held| {
509            held.stamp == self.class_lookup_stamp
510                && held.epoch == self.classes().epoch()
511                && held.program == self.running_program()
512        });
513        if let Some(held) = known {
514            if let Some(class) = held.class {
515                return Ok(class);
516            }
517            if let Some(entry) = held.entry
518                && entry.generation == self.store_generation
519                && entry.writes == self.store_writes
520            {
521                debug_assert!(
522                    unshared!(self, self.env_entry_recomputed(dotted, entry.value)),
523                    "the environment symbol memo answered {dotted:?}, so a write to a \
524                     directory it reads moved neither store_generation nor store_writes"
525                );
526                return Ok(entry.value);
527            }
528        }
529        self.dot_variable_in(dotted, Some(memo), known.map(|held| held.class))
530    }

532    /// Whether [`Interp::dot_variable_in`]'s directory steps still answer
533    /// `value` for `dotted` as a directory's own item.
534    fn env_entry_recomputed(&mut self, dotted: &[u8], value: ObjRef) -> bool {
535        let bare = dotted.strip_prefix(b".").unwrap_or(dotted);
536        if self.package_locals
537            && let Some(program) = self.running_program()
538            && !matches!(
539                self.package_local_probe(program, bare),
540                Ok(hash::DirectoryEntry::Absent)
541            )
542        {
543            return false;
544        }
545        matches!(
546            self.directory_read(
547                &[EnvScope::Local, EnvScope::Environment],
548                bare,
549                &env_seam::Access::Resolve,
550            ),
551            Ok((hash::DirectoryEntry::Found(found), true)) if found == value
552        )
553    }

555    /// The one body behind [`Interp::dot_variable`] and
556    /// [`Interp::dot_variable_memoized`]: `known` is the class lookup's
557    /// answer where a memo still holds it, and `memo` is filled with a fresh
558    /// one.
559    #[inline(never)]
560    fn dot_variable_in(
561        &mut self,
562        dotted: &[u8],
563        memo: Option<&Cell<Option<EnvMemo>>>,
564        known: Option<Option<ObjRef>>,
565    ) -> Result<ObjRef, Failure> {
566        let bare = dotted.strip_prefix(b".").unwrap_or(dotted);
567        let class = match known {
568            Some(class) => class,
569            None => {
570                let class = self
571                    .running_package_class(bare)
572                    .or_else(|| self.rexx_package_class(bare));
573                // Stamped after the lookup, which writes no package and no
574                // class: `rexx_package_class` only fills its own cache.
575                if let Some(memo) = memo {
576                    memo.set(Some(EnvMemo {
577                        program: self.running_program(),
578                        stamp: self.class_lookup_stamp,
579                        epoch: self.classes().epoch(),
580                        class,
581                        entry: None,
582                    }));
583                }
584                class
585            }
586        };
587        if let Some(class) = class {
588            return Ok(class);
589        }
590
591        let generation = self.store_generation;
592        let writes = self.store_writes;
593        if self.package_locals
594            && let Some(program) = self.running_program()
595        {
596            match self.package_local_symbol(program, bare)? {
597                hash::DirectoryEntry::Found(found) => return Ok(found),
598                hash::DirectoryEntry::Owed => {
599                    return Err(Loud::environment_symbol(dotted).into());
600                }
601                hash::DirectoryEntry::Absent => {}
602            }
603        }
604
605        match self.directory_read(
606            &[EnvScope::Local, EnvScope::Environment],
607            bare,
608            &env_seam::Access::Resolve,
609        )? {
610            (hash::DirectoryEntry::Found(found), stored) => {
611                // Kept only where nothing wrote a store during the search,
612                // so the counters it is kept under are the ones it read.
613                if stored
614                    && let Some(memo) = memo
615                    && let Some(mut held) = memo.get()
616                    && held.class.is_none()
617                    && generation == self.store_generation
618                    && writes == self.store_writes
619                    && self.package_security_manager().is_none()
620                {
621                    held.entry = Some(EnvEntry {
622                        generation,
623                        writes,
624                        value: found,
625                    });
626                    memo.set(Some(held));
627                }
628                return Ok(found);
629            }
630            (hash::DirectoryEntry::Owed, _) => {
631                return Err(Loud::environment_symbol(dotted).into());
632            }
633            (hash::DirectoryEntry::Absent, _) => {}
634        }
635
636        if let Some(found) = self.rexx_variable(bare)? {
637            return Ok(found);
638        }
639
640        // The dotted name is built the same way on the route `VALUE` takes.

706    /// The class an **unqualified** `::CLASS` directive's `SUBCLASS`,
707    /// `INHERIT` or `METACLASS` keyword names, when the file's own directives
708    /// do not declare it. A `ns:Name` target never comes here --
709    /// `Interp::resolve_class_target` answers it from the namespace table
710    /// instead, which is `ClassResolver::lookup`'s own split.
711    ///
712    /// Fails with what a directory lookup raised, which a stored method or an
713    /// `UNKNOWN` method there can.
714    pub(crate) fn directive_class(
715        &mut self,
716        installing: ProgramId,
717        upper: &[u8],
718    ) -> Result<Option<ObjRef>, Failure> {
719        // `findClass` on the installing package: its installed classes and then its
720        // parents', then its merged public classes and then its parents' public
721        // ones.
722        let package = self.program_package(installing);
723        if let Some(found) = self.chain_installed_class(package, upper) {
724            return Ok(Some(found));
725        }
726        let data = package.and_then(|package| self.package_data(package));
727        if let Some(found) = data.and_then(|data| data.merged_public_classes.get(upper)) {
728            return Ok(Some(*found));
729        }
730        let parent = data.and_then(|data| data.parent);
731        let local = data.and_then(|data| data.local);
732        if let Some(found) = self.chain_public_class(parent, upper) {
733            return Ok(Some(found));
734        }
735        // The rest of `findClass`: the REXX package's public classes, then
736        // the package's own local directory, `.local` and `.environment`, the
737        // first of them holding the name deciding. Uncached, because a
738        // directive's target is resolved once. `ClassDirective`'s own search
739        // reads the two directories without the security manager.
740        //
741        // The directory model is built here even when the REXX package
742        // answers, so that the bootstrap's first directive builds it. Built at
743        // a later read, its allocations shift the heap for the rest of the
744        // run: measured, `bench-programs/interpret.rex` +0.06% whole process,
745        // all of it in malloc.
746        let _ = self.environment_model();
747        if let Some(found) = self.rexx_package_class_uncached(upper) {
748            return Ok(Some(found));
749        }
750        let mut entry = match local {
751            Some(local) => self.package_local_entry(local, upper)?,
752            None => hash::DirectoryEntry::Absent,
753        };
754        if matches!(entry, hash::DirectoryEntry::Absent) {
755            entry = self.directory_lookup(
756                &[EnvScope::Local, EnvScope::Environment],
757                upper,
758                &env_seam::Access::Direct,
759            )?;
760        }
761        // A name that answers an object which is not a class is 99.949,
762        // `ClassResolver::lookup`. For example, `::class k subclass rexxinfo`,
763        // and `::class t subclass input`, `.INPUT` being `.local`'s stream.
764        match entry {
765            hash::DirectoryEntry::Absent => Ok(None),
766            hash::DirectoryEntry::Found(found) if self.heap.is_class(found) => Ok(Some(found)),
767            _ => Err(crate::error::Raised::not_a_valid_class(upper).into()),
768        }
769    }

774        scopes: &[EnvScope],
775        bare: &[u8],
776        access: &env_seam::Access,

782    /// [`Interp::directory_lookup`], and whether the answer is a directory's
783    /// own item rather than what a security manager or a stored method
784    /// answered: see [`hash::directory_read`].
785    fn directory_read(
786        &mut self,
787        scopes: &[EnvScope],
788        bare: &[u8],
789        access: &env_seam::Access,
790    ) -> Result<(hash::DirectoryEntry, bool), Failure> {
791        let key = hash::Key::new(self.hash_seed, bare);
792        for &scope in scopes {
793            match env_seam::admit(self, scope, bare, access)? {
794                env_seam::Admission::Replaced(value) => {
795                    return Ok((hash::DirectoryEntry::Found(value), false));
796                }
797                env_seam::Admission::Permitted(admitted) => {
798                    let at = match scope {
799                        EnvScope::Local => 0,
800                        EnvScope::Environment => 1,
801                    };
802                    let (handle, view) = {
803                        let model = self.environment_model();
804                        (
805                            env_seam::directory(&model.directories, admitted, scope),
806                            model.views[at],
807                        )
808                    };
809                    let (entry, stored, fresh) =
810                        hash::directory_read(self, handle, key, view.as_ref())?;
811                    if let Some(fresh) = fresh
812                        && let Some(model) = self.environment.as_mut()
813                    {
814                        model.views[at] = Some(fresh);
815                    }
816                    match entry {
817                        hash::DirectoryEntry::Absent => {}
818                        entry => return Ok((entry, stored)),
819                    }
820                }
821            }
822        }
823        Ok((hash::DirectoryEntry::Absent, false))
824    }

859    /// name's bytes: the package-local lookup of the class search, the step
860    /// between the REXX package's public classes and `.local`.
861    #[inline(never)]

863        &mut self,
864        local: ObjRef,
865        bare: &[u8],
866    ) -> Result<hash::DirectoryEntry, Failure> {
867        let view = match self.package_local_view {
868            Some((held, view)) if held == local => Some(view),
869            _ => None,
870        };

873            local,

875            view.as_ref(),

878            self.package_local_view = Some((local, fresh));
879        }
880        Ok(entry)

883    /// [`Interp::package_local_entry`] for `program`'s own local, answering a
884    /// repeat of the last miss without a probe while no hash store has been
885    /// written since.
886    fn package_local_symbol(
887        &mut self,
888        program: ProgramId,
889        bare: &[u8],
890    ) -> Result<hash::DirectoryEntry, Failure> {
891        let miss = &self.package_local_miss;
892        if miss.program == Some(program)
893            && miss.generation == self.store_generation
894            && miss.writes == self.store_writes
895            && miss.name == bare
896        {
897            debug_assert!(
898                matches!(
899                    unshared!(self, self.package_local_probe(program, bare)),
900                    Ok(hash::DirectoryEntry::Absent)
901                ),
902                "the package local miss memo answered {bare:?} absent, so a write to the \
903                 local's store moved neither store_generation nor store_writes"
904            );
905            return Ok(hash::DirectoryEntry::Absent);
906        }
907        let entry = self.package_local_probe(program, bare)?;
908        if matches!(entry, hash::DirectoryEntry::Absent) {
909            let miss = &mut self.package_local_miss;
910            miss.program = Some(program);
911            miss.generation = self.store_generation;
912            miss.writes = self.store_writes;
913            miss.name.clear();
914            miss.name.extend_from_slice(bare);
915        }
916        Ok(entry)
917    }

931    /// `TheRexxPackage->findPublicClass`, step 4 of the documented
932    /// environment-symbol search order (`rexxpg` `classes.xml:838`) and the
933    /// step the class search takes between a package's imports and its own
934    /// local.
935    #[inline]
936    pub(crate) fn rexx_package_class(&mut self, upper: &[u8]) -> Option<ObjRef> {
937        if let Some(&hit) = self.rexx_class_cache.get(upper) {
938            debug_assert_eq!(
939                Some(hit),
940                unshared!(self, self.rexx_package_class_uncached(upper)),
941                "the .NAME cache answered {upper:?} with a class the search no longer finds, so \
942                 some table this cache is derived from was mutated without \
943                 Interp::invalidate_rexx_class_cache"
944            );
945            return Some(hit);
946        }
947        let found = self.rexx_package_class_uncached(upper)?;
948        // Only a hit is cached. A miss must stay a miss: a later `::CLASS`
949        // install or a `~addPackage` can turn one into a hit, and a cached
950        // miss would outlive that where a cached hit is invalidated with the
951        // table it came from.
952        self.rexx_class_cache.insert(upper.into(), found);
953        Some(found)
954    }

992    /// The `.local` entries built before any program runs: the command-line
993    /// words, then the streams and monitors, and the session queue, each
994    /// stored as it is built. A new instance of the library's `LocalServer`
995    /// builds the same streams, monitors and queue.
996    ///
997    /// **Here rather than when the directory is built**, because the library
998    /// program evaluates `.environment` while it installs, before it has
999    /// declared `Stream`. Each write replaces an owed entry in place, so the
1000    /// order is fixed whatever order these are built in.
1001    ///
1002    /// **Through the package publics, not the class registry.** A `::CLASS`
1003    /// directive calls `define_unregistered_class`, which by its own contract
1004    /// records the class "without registering the name", so
1005    /// `classes().lookup("Stream")` never finds the one the library program
1006    /// declared.
1007    ///
1008    /// **No panic on a miss, at any step.** A step that cannot build its
1009    /// entry leaves that entry and every later one owed.
1010    pub(crate) fn mint_local_directory(&mut self) {
1011        let frame = self.roots.activity_mut().push_frame();
1012        self.mint_local_entries();
1013        self.roots.activity_mut().pop_frame(frame);
1014    }

1016    /// [`Interp::mint_local_directory`]'s steps, inside its root frame.
1017    fn mint_local_entries(&mut self) {
1018        // **`Body::array`, not `.Array~of`'s shape** -- measured, a
1019        // `.SYSCARGS` built from no words answers `~dimension` `0` as
1020        // `.array~new` does, where `.array~of()` answers `1`.
1021        let stored = match self.command_words_base {
1022            Some(base) => {
1023                let mut slots: Vec<Option<ObjRef>> = Vec::with_capacity(self.command_words.len());
1024                if !self.command_words.is_empty() {
1025                    slots.resize(base.saturating_sub(1), None);
1026                }
1027                for at in 0..self.command_words.len() {
1028                    let word = std::mem::take(&mut self.command_words[at]);
1029                    let text = self.text(&word);
1030                    self.roots.activity_mut().push_temp(text);
1031                    slots.push(Some(text));
1032                    self.command_words[at] = word;
1033                }
1034                let arguments = self.alloc_charged(BehaviourId::ARRAY, Body::array(slots));
1035                self.roots.activity_mut().push_temp(arguments);
1036                self.set_directory_entry(EnvScope::Local, b"SYSCARGS", arguments)
1037            }
1038            None => self.remove_directory_entry(EnvScope::Local, b"SYSCARGS"),
1039        };
1040        if stored.is_err() {
1041            return;
1042        }
1043        let Some(stream_class) = self.rexx_package_class(b"STREAM") else {
1044            return;
1045        };
1046        for (name, which) in [
1047            (b"STDIN".as_slice(), interrexx_core::StandardStream::In),
1048            (b"STDOUT".as_slice(), interrexx_core::StandardStream::Out),
1049            (b"STDERR".as_slice(), interrexx_core::StandardStream::Err),
1050        ] {
1051            let Ok(built) = crate::dispatch::stream::standard_stream(self, stream_class, which)
1052            else {
1053                return;
1054            };
1055            if name == b"STDERR".as_slice() {
1056                self.bootstrap_stderr = Some(built);
1057            }
1058            if name == b"STDOUT".as_slice() {
1059                self.bootstrap_stdout = Some(built);
1060            }
1061            if self
1062                .set_directory_entry(EnvScope::Local, name, built)
1063                .is_err()
1064            {
1065                return;
1066            }
1067        }
1068        let Some(monitor_class) = self.rexx_package_class(b"MONITOR") else {
1069            return;
1070        };
1071        for (name, over, rendered) in [
1072            (
1073                b"INPUT".as_slice(),
1074                b"STDIN".as_slice(),
1075                "The INPUT monitor",
1076            ),
1077            (
1078                b"DEBUGINPUT".as_slice(),
1079                b"INPUT".as_slice(),
1080                "The DEBUG INPUT monitor",
1081            ),
1082            (
1083                b"OUTPUT".as_slice(),
1084                b"STDOUT".as_slice(),
1085                "The OUTPUT monitor",
1086            ),
1087            (
1088                b"ERROR".as_slice(),
1089                b"STDERR".as_slice(),
1090                "The ERROR monitor",
1091            ),
1092            (
1093                b"TRACEOUTPUT".as_slice(),
1094                b"ERROR".as_slice(),
1095                "The TRACE OUTPUT monitor",
1096            ),
1097        ] {
1098            let Ok(Some(target)) = self.local_route(over) else {
1099                return;
1100            };
1101            let caller = self.caller();
1102            let Ok(Some(built)) =
1103                self.send_message(monitor_class, b"NEW", None, &[Some(target)], caller)
1104            else {
1105                return;
1106            };
1107            self.roots.activity_mut().push_temp(built);
1108            if let Some(object) = self.heap.get_mut(built)
1109                && let Body::Instance { name: held, .. } = &mut object.body
1110            {
1111                *held = Some(rendered.as_bytes().into());
1112            }
1113            if self
1114                .set_directory_entry(EnvScope::Local, name, built)
1115                .is_err()
1116            {
1117                return;
1118            }
1119        }
1120        let Some(queue_class) = self.rexx_package_class(b"REXXQUEUE") else {
1121            return;
1122        };
1123        let session = self.text(b"SESSION");
1124        self.roots.activity_mut().push_temp(session);
1125        let caller = self.caller();
1126        let Ok(Some(queue)) =
1127            self.send_message(queue_class, b"NEW", None, &[Some(session)], caller)
1128        else {
1129            return;
1130        };
1131        self.roots.activity_mut().push_temp(queue);
1132        let _ = self.set_directory_entry(EnvScope::Local, b"STDQUE", queue);
1133    }

1247    fn build_package_string_table(
1248        &mut self,
1249        package: ObjRef,
1250        kind: PackageTable,
1251        force: bool,
1252    ) -> Result<Option<ObjRef>, Failure> {
1253        let Some(data) = self.package_data(package) else {
1254            return Ok(None);
1255        };
1256        if let Some(found) = data.tables[kind.index()] {
1257            return Ok(Some(found));
1258        }
1259        // A copy of the REXX package has no directives, and gets a table only
1260        // for a write, which is where `~addRoutine` files its routines.
1261        let source = match data.program {
1262            Some(program) if !self.untranslated.contains(&program) => {
1263                let source = self.programs.get(program);
1264                debug_assert!(
1265                    source.is_some(),
1266                    "{program:?} was freed while its package was held"
1267                );
1268                let Some(source) = source else {
1269                    return Err(Loud::unknown_receiver("a package whose program was freed").into());
1270                };
1271                Some(std::rc::Rc::clone(source))
1272            }
1273            _ => None,
1274        };
1275        let entries = match (data.program, &source) {
1276            (Some(program), Some(source)) => package_table_entries(program, source, kind),
1277            _ => Vec::new(),
1278        };
1279        if entries.is_empty() && !force {
1280            return Ok(None);
1281        }
1282        // The table is rooted before anything it will hold is allocated, and
1283        // each value goes into it as soon as it exists: `alloc_with` collects
1284        // before it allocates.
1285        self.roots.activity_mut().push_temp(package);
1286        let table = crate::dispatch::hash::new_string_table(self)?;
1287        let data = self
1288            .package_data_mut(package)
1289            .expect("the package read above is rooted and keeps its data");
1290        data.tables[kind.index()] = Some(table);
1291        self.routines_tables |= kind == PackageTable::Routines;
1292        let frame = self.roots.activity_mut().push_frame();
1293        for (name, value) in entries {
1294            let value = match value {
1295                TableValue::Instance {
1296                    class: id,
1297                    site,
1298                    declared: (program, directive),
1299                    routine,
1300                } => {
1301                    let installed = crate::InstalledRoutine { program, directive };
1302                    // A library-backed `::ROUTINE` is the one object its
1303                    // library procedure has, shared by every directive naming
1304                    // it.
1305                    if let Some(code) = routine
1306                        .then(|| self.library_routine_code(installed))
1307                        .flatten()
1308                    {
1309                        let object = self.library_routine_object(code);
1310                        self.routine_objects.insert(installed, object);
1311                        object
1312                    } else if let Some(row) =
1313                        routine.then(|| self.rexx_routine_row(installed)).flatten()
1314                    {
1315                        let index = crate::internal_routines::index_of(row);
1316                        let object = self.internal_routine_object(index);
1317                        self.routine_objects.insert(installed, object);
1318                        object
1319                    } else {
1320                        let class = self
1321                            .classes()
1322                            .lookup(id)
1323                            .expect("every TableValue::Instance names a native class");
1324                        let object = self.native_instance(class);
1325                        self.attach_annotations(object, site);
1326                        // `entries` is empty unless `source` is this program's.
1327                        let method = source.as_ref().filter(|_| !routine).map(|source| {
1328                            let method =
1329                                self.unattached_method_id(program, source, directive, &name);
1330                            self.executable_objects.insert(method, object);
1331                            method
1332                        });
1333                        self.record_executable(
1334                            object,
1335                            crate::ExecutableRecord {
1336                                source: crate::ExecutableSource::Directive { program, directive },
1337                                installed: method,
1338                                routine: routine.then_some((program, directive)),
1339                                owns_installed: false,
1340                            },
1341                        )
1342                        .expect("the package whose table this is holds its program");
1343                        if routine {
1344                            self.routine_objects.insert(installed, object);
1345                        }
1346                        object
1347                    }
1348                }
1349                TableValue::Lines(lines) => self.line_array(&lines),
1350            };
1351            self.roots.activity_mut().push_temp(value);
1352            if let Err(failure) = crate::dispatch::hash::directory_put(self, table, &name, value) {
1353                self.roots.activity_mut().pop_frame(frame);
1354                return Err(failure);
1355            }
1356        }
1357        self.roots.activity_mut().pop_frame(frame);
1358        Ok(Some(table))
1359    }

1361    /// A fresh `StringTable` holding `entries`, put in name order: the copy
1362    /// the package's class and routine listings answer, which a program can
1363    /// ask everything a `.StringTable~new` answers.
1364    pub(crate) fn string_table_of(
1365        &mut self,
1366        mut entries: Vec<(Box<[u8]>, ObjRef)>,
1367    ) -> Result<ObjRef, Failure> {
1368        entries.sort_by(|a, b| a.0.cmp(&b.0));
1369        let frame = self.roots.activity_mut().push_frame();
1370        for (_, value) in &entries {
1371            self.roots.activity_mut().push_temp(*value);
1372        }
1373        let table = crate::dispatch::hash::new_string_table(self)?;
1374        self.roots.activity_mut().push_temp(table);
1375        for (name, value) in entries {
1376            let index = self.text(&name);
1377            self.roots.activity_mut().push_temp(index);
1378            crate::dispatch::hash::store_insert(self, table, index, value)?;
1379        }
1380        self.roots.activity_mut().pop_frame(frame);
1381        self.roots.activity_mut().push_temp(table);
1382        Ok(table)
1383    }

1617/// What `kind`'s table holds for `program`, keyed as the table keys it.
1618fn package_table_entries(
1619    id: ProgramId,
1620    program: &interrexx_parse::Program,
1621    kind: PackageTable,
1622) -> Vec<(Vec<u8>, TableValue)> {
1623    let unattached_method = |name: &[u8], index: usize| TableValue::Instance {
1624        class: "Method",
1625        site: Annotated::Unattached(id, name.into()),
1626        declared: (id, index),
1627        routine: false,
1628    };
1629    let mut entries = Vec::new();
1630    let mut seen_class = false;
1631    for (index, directive) in program.directives.iter().enumerate() {
1632        // A synthetic directive is in no package table either -- `install.rs`'s
1633        // `Interp::install_directives` carries the rule and what asserts it.
1634        if directive.clause_span.is_empty() {
1635            continue;
1636        }
1637        let unattached = !seen_class;
1638        match &directive.kind {
1639            interrexx_parse::DirectiveKind::Class(_) => seen_class = true,
1640            interrexx_parse::DirectiveKind::Method(method)
1641                if unattached && kind == PackageTable::UnattachedMethods =>
1642            {
1643                let upper = method.name.to_ascii_uppercase();
1644                // `ATTRIBUTE` files an accessor pair, the plain name and that
1645                // name with `=`. Measured, `::method z attribute` puts `Z`
1646                // and `Z=` in `.METHODS`; that spelling is filed as an
1647                // attribute getter and setter pair.
1648                if method.attribute {
1649                    let setter = crate::accessor_setter_name(&upper);
1650                    let value = unattached_method(&setter, index);
1651                    entries.push((setter, value));
1652                }
1653                let value = unattached_method(&upper, index);
1654                entries.push((upper, value));
1655            }
1656            interrexx_parse::DirectiveKind::Attribute(attribute)
1657                if unattached && kind == PackageTable::UnattachedMethods =>
1658            {
1659                let upper = attribute.name.to_ascii_uppercase();
1660                let setter = crate::accessor_setter_name(&upper);
1661                // The same split `Interp::install_attribute` makes over the
1662                // same field, because the getter and the setter here are the
1663                // ones it installs, under the same names. Measured,
1664                // `::attribute zz` puts `ZZ` and `ZZ=` in `.METHODS`,
1665                // `::attribute zz get` puts `ZZ` alone and `::attribute zz
1666                // set` puts `ZZ=` alone.
1667                match attribute.style {
1668                    interrexx_parse::AttributeStyle::Both => {
1669                        let getter_value = unattached_method(&upper, index);
1670                        let setter_value = unattached_method(&setter, index);
1671                        entries.push((upper, getter_value));
1672                        entries.push((setter, setter_value));
1673                    }
1674                    interrexx_parse::AttributeStyle::Get => {
1675                        let value = unattached_method(&upper, index);
1676                        entries.push((upper, value));
1677                    }
1678                    interrexx_parse::AttributeStyle::Set => {
1679                        let value = unattached_method(&setter, index);
1680                        entries.push((setter, value));
1681                    }
1682                }
1683            }
1684            interrexx_parse::DirectiveKind::Constant(constant)
1685                if unattached && kind == PackageTable::UnattachedMethods =>
1686            {
1687                // One getter, `createConstantGetterMethod`'s own
1688                // `addMethod(name, method, false)`. Measured, `::constant sep
1689                // '/'` leaves `.methods~items` `1`.
1690                let upper = constant.name.to_ascii_uppercase();
1691                let value = unattached_method(&upper, index);
1692                entries.push((upper, value));
1693            }
1694            interrexx_parse::DirectiveKind::Routine(routine) if kind == PackageTable::Routines => {
1695                // Both access scopes, not the public ones alone: `.ROUTINES`
1696                // is `package->routines` and `publicRoutines` is a second
1697                // table nothing here reads. Measured, a file with one `PUBLIC`
1698                // and one plain `::ROUTINE` leaves `.routines~items` `2`.
1699                entries.push((
1700                    routine.name.to_ascii_uppercase(),
1701                    TableValue::Instance {
1702                        class: "Routine",
1703                        site: Annotated::Routine(id, index),
1704                        declared: (id, index),
1705                        routine: true,
1706                    },
1707                ));
1708            }
1709            interrexx_parse::DirectiveKind::Resource(resource)
1710                if kind == PackageTable::Resources =>
1711            {
1712                let lines = resource
1713                    .lines
1714                    .iter()
1715                    .map(|span| {
1716                        program
1717                            .source
1718                            .span_bytes(span.clone())
1719                            .expect("a resource body line is a span of its own program's source")
1720                            .to_vec()
1721                    })
1722                    .collect();
1723                entries.push((resource.name.to_ascii_uppercase(), TableValue::Lines(lines)));
1724            }
1725            _ => {}
1726        }
1727    }
1728    entries
1729}
