1//! The `unsafe` that [`super`] needs and nothing else: the calls a host
2//! offers through no safe crate. On Apple's systems those are `setgroups`,
3//! which `nix` leaves out there, and `select` and `fcntl` on a descriptor
4//! number that may name no open descriptor, which no safe wrapper takes.
5//! Other hosts need none of it.
6
7#![allow(unsafe_code)]
8
9/// The calling thread's `errno`, as the last failed call left it.
10#[cfg(target_vendor = "apple")]
11fn errno() -> nix::errno::Errno {
12    nix::errno::Errno::last()
13}
14
15/// Drops every supplementary group of the process: `setgroups(0, NULL)`.
16#[cfg(target_vendor = "apple")]
17pub(super) fn setgroups_none() -> nix::Result<()> {
18    // SAFETY: with a count of 0, `setgroups` reads no element of the list,
19    // so the null pointer is never read. The call changes only the process's
20    // credentials, which no Rust object holds.
21    match unsafe { libc::setgroups(0, std::ptr::null()) } {
22        0 => Ok(()),
23        _ => Err(errno()),
24    }
25}

27/// Whether `fd` names an open descriptor of this process, as
28/// `fcntl(F_GETFD)` finds it.
29#[cfg(target_vendor = "apple")]
30pub(super) fn is_open(fd: i32) -> bool {
31    // SAFETY: `fcntl(F_GETFD)` takes no pointer and changes nothing; it reads
32    // the descriptor flags of any number and fails with `EBADF` for one that
33    // names no open descriptor.
34    unsafe { libc::fcntl(fd, libc::F_GETFD) != -1 }
35}

37/// Whether `select`, asked at once whether the descriptor numbered `fd` is
38/// readable, answers `EBADF`: `fd` is closed and inside the process's
39/// descriptor table. `select` answers `EINVAL` for an `fd` at or above
40/// `FD_SETSIZE`, and that is `false`.
41#[cfg(target_vendor = "apple")]
42pub(super) fn select_rejects(fd: i32) -> bool {
43    /// Bits in one word of an `fd_set`, as Apple's headers lay it out.
44    const WORD: usize = 32;
45    let Ok(index) = usize::try_from(fd) else {
46        return false;
47    };
48    // A set of at least an `fd_set`'s size, with `fd`'s bit as the headers'
49    // `FD_SET` places it.
50    let words = (index / WORD + 1).max(std::mem::size_of::<libc::fd_set>() / 4);
51    let mut set = vec![0u32; words];
52    set[index / WORD] |= 1 << (index % WORD);
53    let mut timeout = libc::timeval {
54        tv_sec: 0,
55        tv_usec: 0,
56    };
57    // SAFETY: `select` reads and writes the first `fd + 1` bits of the read
58    // set, which `set` holds (it has `index / 32 + 1` words or more, and is
59    // at least an `fd_set` long, aligned as its `i32` words are); the other
60    // sets are null, which `select` takes as empty; `timeout` is a live
61    // local, zero so the call does not wait. The descriptor's own state is
62    // only read.
63    let answer = unsafe {
64        libc::select(
65            fd.saturating_add(1),
66            set.as_mut_ptr().cast::<libc::fd_set>(),
67            std::ptr::null_mut(),
68            std::ptr::null_mut(),
69            &raw mut timeout,
70        )
71    };
72    answer == -1 && errno() == nix::errno::Errno::EBADF
73}
