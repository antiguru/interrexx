14/// `SysCls()`: `system("clear")`, answering its raw wait status.
15/// The child's output reaches this interpreter's own sinks.
16pub(crate) fn cls(
17    interp: &mut Interp,
18    _name: &'static [u8],
19    args: &[Option<ObjRef>],
20) -> Result<ObjRef, Failure> {
21    use std::os::unix::ffi::OsStrExt as _;
22    use std::os::unix::process::{CommandExt as _, ExitStatusExt as _};
23    check_arity(args, 0)?;
24    let mut command = std::process::Command::new("/bin/sh");
25    // `system`'s own argument vector, which names the shell in its messages.
26    command.arg0("sh").arg("-c").arg("clear").env_clear();
27    for (name, value) in &interp.env {
28        command.env(
29            std::ffi::OsStr::from_bytes(name),
30            std::ffi::OsStr::from_bytes(value),
31        );
32    }
33    command.current_dir(std::ffi::OsStr::from_bytes(interp.cwd_bytes()));
34    // `output` would give the child `/dev/null`; `system` inherits it.
35    command.stdin(std::process::Stdio::inherit());
36    // The output is collected as `output` collects it, but the child starts
37    // apart from the wait, so `host::spawning` holds no lock while it runs.
38    command.stdout(std::process::Stdio::piped());
39    command.stderr(std::process::Stdio::piped());
40    interp.sync_input_offset();
41    let output =
42        crate::host::spawning(|| command.spawn()).and_then(std::process::Child::wait_with_output);
43    interp.adopt_input_offset();
44    let status = match output {
45        Ok(output) => {
46            interp.write_out(&output.stdout);
47            interp.write_err(&output.stderr);
48            output.status.into_raw()
49        }
50        // `system` answers -1 where no child could be made.
51        Err(_) => -1,
52    };
53    Ok(answer(interp, i64::from(status)))
54}

56/// `SysGetKey([echo])`: one byte read from standard input with the terminal
57/// out of canonical mode, echoed unless `NOECHO`. The null string where
58/// standard input is not a terminal, as `getkey` answers there.
59pub(crate) fn get_key(
60    interp: &mut Interp,
61    _name: &'static [u8],
62    args: &[Option<ObjRef>],
63) -> Result<ObjRef, Failure> {
64    check_arity(args, 1)?;
65    let echo = match optional_text(interp, args, 1)? {
66        None => true,
67        Some(text) => {
68            let text = c_string(&text);
69            if text.eq_ignore_ascii_case(b"NOECHO") {
70                false
71            } else if text.eq_ignore_ascii_case(b"ECHO") {
72                true
73            } else {
74                return Err(Raised::invalid_routine_option(
75                    "SysGetKey",
76                    "echo",
77                    "'ECHO' or 'NOECHO'",
78                    text,
79                )
80                .into());
81            }
82        }
83    };
84    let stdin = std::io::stdin();
85    if !interp.input_is_process_stdin() || !rustix::termios::isatty(stdin.as_fd()) {
86        return Ok(interp.text_built(Vec::new()));
87    }
88    interp.hand_over_output();
89    let tty = std::fs::File::open("/dev/tty").ok();
90    let key = read_key(tty.as_ref().map(AsFd::as_fd), stdin.as_fd(), echo);
91    Ok(interp.text_built(key_text(key)))
92}

94/// What `getkey` stores of the `getchar` answer `key`: the byte as a C
95/// string, so a NUL is the null string, and `EOF` truncated to `'FF'x`.
96fn key_text(key: Option<u8>) -> Vec<u8> {
97    match key {
98        Some(0) => Vec::new(),
99        Some(byte) => vec![byte],
100        None => vec![0xFF],
101    }
102}

104/// Reads one byte from `input` with `tty`, where there is one, out of
105/// canonical mode and with echo off unless `echo`, then restores `tty`'s
106/// settings. `None` at end of input or on an error, interrupted reads
107/// included.
108fn read_key(tty: Option<BorrowedFd<'_>>, input: BorrowedFd<'_>, echo: bool) -> Option<u8> {
109    let original = tty.and_then(|tty| rustix::termios::tcgetattr(tty).ok());
110    if let (Some(tty), Some(original)) = (tty, &original) {
111        let mut raw = original.clone();
112        raw.local_modes.remove(LocalModes::ICANON);
113        if !echo {
114            raw.local_modes.remove(LocalModes::ECHO);
115        }
116        raw.special_codes[SpecialCodeIndex::VMIN] = 1;
117        raw.special_codes[SpecialCodeIndex::VTIME] = 0;
118        let _ = rustix::termios::tcsetattr(tty, OptionalActions::Now, &raw);
119    }
120    let mut byte = [0u8; 1];
121    let key = match rustix::io::read(input, &mut byte) {
122        Ok(1) => Some(byte[0]),
123        _ => None,
124    };
125    if let (Some(tty), Some(original)) = (tty, &original) {
126        let _ = rustix::termios::tcsetattr(tty, OptionalActions::Now, original);
127    }
128    key
129}

131/// `SysQueryProcess([selector])`: `PID` (the default), `PPID`, `PGID` (the
132/// **parent's** group, `getpgid(getppid())`), `PPRIO`, `PTIME`, `PMEM`,
133/// `PSWAPS` or `PRCVDSIG`, without regard to case; 40.1 for any other.
134pub(crate) fn query_process(
135    interp: &mut Interp,
136    name: &'static [u8],
137    args: &[Option<ObjRef>],
138) -> Result<ObjRef, Failure> {
139    check_arity(args, 1)?;
140    let selector = optional_text(interp, args, 1)?;
141    let selector = selector.as_deref().map(c_string);
142    let is = |wanted: &[u8]| selector.is_some_and(|text| text.eq_ignore_ascii_case(wanted));
143    let pid_number =
144        |pid: Option<rustix::process::Pid>| i64::from(rustix::process::Pid::as_raw(pid));
145    if selector.is_none() || is(b"PID") {
146        let pid = pid_number(Some(rustix::process::getpid()));
147        return Ok(answer(interp, pid));
148    }
149    if is(b"PPID") {
150        let pid = pid_number(rustix::process::getppid());
151        return Ok(answer(interp, pid));
152    }
153    if is(b"PGID") {
154        let group = rustix::process::getpgid(rustix::process::getppid())
155            .map_or(-1, |group| pid_number(Some(group)));
156        return Ok(answer(interp, group));
157    }
158    if is(b"PPRIO") {
159        let priority = rustix::process::getpriority_process(None).unwrap_or(-1);
160        return Ok(answer(interp, i64::from(priority)));
161    }
162    let usage = match Usage::of_this_process() {
163        Ok(usage) => usage,
164        Err(errno) => {
165            let text = format!("System error; errno = {}", errno.raw_os_error());
166            return Ok(interp.text_built(text.into_bytes()));
167        }
168    };
169    let text = if is(b"PTIME") {
170        usage.cpu_summary()
171    } else if is(b"PMEM") {
172        format!("Max_Memory_RSS: {}", usage.max_rss)
173    } else if is(b"PSWAPS") {
174        "Memory_swaps: 0".to_owned()
175    } else if is(b"PRCVDSIG") {
176        "Received_signals: 0".to_owned()
177    } else {
178        return Err(Raised::routine_failed(&name.to_ascii_uppercase()).into());
179    };
180    Ok(interp.text_built(text.into_bytes()))
181}

183/// The `getrusage(RUSAGE_SELF)` fields `SysQueryProcess` reads, from
184/// `/proc/self`. Linux keeps no `ru_nswap` or `ru_nsignals`, which are always
185/// 0 (`getrusage(2)`), so they have no field here.
186struct Usage {
187    /// User and system CPU time as seconds and microseconds, at the
188    /// resolution of a clock tick.
189    user: (u64, u64),
190    system: (u64, u64),
191    /// The resident set's high-water mark in kilobytes: `VmHWM`.
192    max_rss: u64,
193}
194
195impl Usage {
196    /// The usage `getrusage` reports, where no `/proc` keeps it. Apple's
197    /// `ru_maxrss` counts bytes, where Linux's counts kilobytes.
198    #[cfg(not(target_os = "linux"))]
199    fn of_this_process() -> Result<Usage, Errno> {
200        use nix::sys::resource::{UsageWho, getrusage};
201        let usage = getrusage(UsageWho::RUSAGE_SELF)
202            .map_err(|error| Errno::from_raw_os_error(crate::builtin::errno_number(error)))?;
203        let time = |value: nix::sys::time::TimeVal| {
204            (
205                u64::try_from(value.tv_sec()).unwrap_or(0),
206                u64::try_from(value.tv_usec()).unwrap_or(0),
207            )
208        };
209        let max_rss = u64::try_from(usage.max_rss()).unwrap_or(0);
210        let max_rss = if cfg!(target_vendor = "apple") {
211            max_rss / 1024
212        } else {
213            max_rss
214        };
215        Ok(Usage {
216            user: time(usage.user_time()),
217            system: time(usage.system_time()),
218            max_rss,
219        })
220    }
221
222    #[cfg(target_os = "linux")]
223    fn of_this_process() -> Result<Usage, Errno> {
224        let read = |path: &str| {
225            std::fs::read(path).map_err(|error| {
226                Errno::from_raw_os_error(error.raw_os_error().unwrap_or(Errno::IO.raw_os_error()))
227            })
228        };
229        let stat = read("/proc/self/stat")?;
230        // The fields after the parenthesised command name, which may itself
231        // hold blanks: `state` first, then `utime` and `stime` (`proc(5)`
232        // fields 14 and 15) at offsets 11 and 12.
233        let after_name = stat
234            .iter()
235            .rposition(|byte| *byte == b')')
236            .map_or(&stat[..], |at| &stat[at + 1..]);
237        let fields: Vec<&[u8]> = after_name
238            .split(|byte| *byte == b' ')
239            .filter(|f| !f.is_empty())
240            .collect();
241        let ticks = |at: usize| -> Result<u64, Errno> {
242            fields
243                .get(at)
244                .and_then(|field| std::str::from_utf8(field).ok()?.parse().ok())
245                .ok_or(Errno::IO)
246        };
247        let hertz = rustix::param::clock_ticks_per_second().max(1);
248        let time = |ticks: u64| (ticks / hertz, (ticks % hertz) * 1_000_000 / hertz);
249        let user = time(ticks(11)?);
250        let system = time(ticks(12)?);
251        let status = read("/proc/self/status")?;
252        let max_rss = status
253            .split(|byte| *byte == b'\n')
254            .find_map(|line| line.strip_prefix(b"VmHWM:"))
255            .and_then(|rest| {
256                std::str::from_utf8(rest)
257                    .ok()?
258                    .split_whitespace()
259                    .next()?
260                    .parse()
261                    .ok()
262            })
263            .unwrap_or(0);
264        Ok(Usage {
265            user,
266            system,
267            max_rss,
268        })
269    }
270
271    /// `PTIME`'s text: the total, kernel and user times, each as
272    /// `%2d:%.2d:%.2d:%.3d` hours, minutes, seconds and milliseconds, the
273    /// total's milliseconds the sum of the two truncated parts.
274    fn cpu_summary(&self) -> String {
275        // The work is in `unsigned int`.
276        let narrow = |value: u64| u32::cast_truncate(value);
277        let mut milliseconds = narrow(self.user.1 / 1000) + narrow(self.system.1 / 1000);
278        let mut seconds = 0;
279        if milliseconds >= 1000 {
280            seconds = milliseconds / 1000;
281            milliseconds %= 1000;
282        }
283        seconds += narrow(self.user.0);
284        seconds += narrow(self.system.0);
285        let kernel = clock(narrow(self.system.0), narrow(self.system.1 / 1000));
286        let user = clock(narrow(self.user.0), narrow(self.user.1 / 1000));
287        format!(
288            "CPU_Time Summary: {}  Kernel: {}  User: {}",
289            clock(seconds, milliseconds),
290            kernel,
291            user
292        )
293    }
294}

296/// `seconds` and `milliseconds` as `%2d:%.2d:%.2d:%.3d`.
297fn clock(seconds: u32, milliseconds: u32) -> String {
298    format!(
299        "{:2}:{:02}:{:02}:{:03}",
300        seconds / 3600,
301        seconds / 60 % 60,
302        seconds % 60,
303        milliseconds
304    )
305}

307/// `SysSetPriority([class], priority)`: `setpriority` of this thread's nice
308/// value to `-priority`, answering `0` or its errno. The class is checked as
309/// a whole number and otherwise ignored.
310pub(crate) fn set_priority(
311    interp: &mut Interp,
312    _name: &'static [u8],
313    args: &[Option<ObjRef>],
314) -> Result<ObjRef, Failure> {
315    check_arity(args, 2)?;
316    if let Some(class) = args.first().copied().flatten() {
317        int_argument(interp, class, 1)?;
318    }
319    let value = required(args, 2)?;
320    let priority = int_argument(interp, value, 2)?;
321    let nice = i32::try_from(priority)
322        .expect("int_argument answers an int")
323        .wrapping_neg();
324    let code = match rustix::process::setpriority_process(None, nice) {
325        Ok(()) => 0,
326        Err(errno) => errno.raw_os_error(),
327    };
328    Ok(answer(interp, i64::from(code)))
329}

341/// What `SysWait` answers where `wait` finds no child or fails: the reference
342/// run answers a status it never set (oracle defect,
343/// `corpus/oracle-crashes.txt` entry 45).
344const NO_CHILD: i64 = -1;
345
346/// `SysWait()`: `wait` for any child of this process, answering its raw status,
347/// or [`NO_CHILD`]. Any child: another activity's command, or one of another
348/// interpreter in the process, as `wait` reaps any.
349pub(crate) fn wait(
350    interp: &mut Interp,
351    _name: &'static [u8],
352    args: &[Option<ObjRef>],
353) -> Result<ObjRef, Failure> {
354    check_arity(args, 0)?;
355    let status = match rustix::process::wait(rustix::process::WaitOptions::empty()) {
356        Ok(Some((_, status))) => i64::from(status.as_raw()),
357        Ok(None) | Err(_) => NO_CHILD,
358    };
359    Ok(answer(interp, status))
360}

362/// `SysCreatePipe([blocking])`: a pipe as `"read write"`, its descriptors
363/// left open and inherited by every child, the read end non-blocking where
364/// the argument starts with `N` or `n`. A first character other than `B`,
365/// `b`, `N` or `n` is 40.920.
366pub(crate) fn create_pipe(
367    interp: &mut Interp,
368    _name: &'static [u8],
369    args: &[Option<ObjRef>],
370) -> Result<ObjRef, Failure> {
371    check_arity(args, 1)?;
372    let blocking = match optional_text(interp, args, 1)? {
373        None => true,
374        Some(text) => match c_string(&text).first() {
375            Some(b'n' | b'N') => false,
376            Some(b'b' | b'B') => true,
377            _ => {
378                return Err(Raised::invalid_routine_option(
379                    "SysCreatePipe",
380                    "blocking",
381                    "'B' or 'N'",
382                    c_string(&text),
383                )
384                .into());
385            }
386        },
387    };
388    let (reader, writer) = match std::io::pipe() {
389        Ok(ends) => ends,
390        Err(error) => return Ok(pipe_failed(interp, "Creating pipe", &error)),
391    };
392    let inherited = [reader.as_fd(), writer.as_fd()]
393        .into_iter()
394        .try_for_each(|end| rustix::io::fcntl_setfd(end, rustix::io::FdFlags::empty()));
395    if let Err(errno) = inherited {
396        return Ok(pipe_failed(interp, "Creating pipe", &errno.into()));
397    }
398    if !blocking {
399        let flags = rustix::fs::fcntl_getfl(&reader).and_then(|flags| {
400            rustix::fs::fcntl_setfl(&reader, flags | rustix::fs::OFlags::NONBLOCK)
401        });
402        if let Err(errno) = flags {
403            return Ok(pipe_failed(interp, "Setting NONBLOCK flag", &errno.into()));
404        }
405    }
406    let text = format!("{} {}", reader.into_raw_fd(), writer.into_raw_fd());
407    Ok(interp.text_built(text.into_bytes()))
408}

410/// `perror` of `what` on standard error, and the null string.
411fn pipe_failed(interp: &mut Interp, what: &str, error: &std::io::Error) -> ObjRef {
412    let message = std::io::Error::from_raw_os_error(error.raw_os_error().unwrap_or(0)).to_string();
413    let message = message
414        .rsplit_once(" (os error")
415        .map_or(message.as_str(), |(text, _)| text);
416    interp.write_err(format!("*** ERROR: {what}: {message}\n").as_bytes());
417    interp.text_built(Vec::new())
418}
