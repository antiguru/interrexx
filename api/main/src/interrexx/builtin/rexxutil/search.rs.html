13/// Where `SysFileSearch` and `SysFileTree` put what they find.
14#[derive(Clone, Copy)]
15pub(super) enum Found {
16    /// An `Array`, emptied first and appended to.
17    Array(ObjRef),
18    /// A stem, written from tail 1 with the count in tail 0 at the end. Tails
19    /// past the count are left as they were.
20    Stem(ObjRef),
21}
22
23impl Found {
24    /// The argument at `position` as a stem handler resolves it: an
25    /// instance of `Array` itself, a stem, or a string naming a stem variable.
26    /// 40.925 for anything else.
27    pub(super) fn resolve(
28        interp: &mut Interp,
29        value: ObjRef,
30        position: usize,
31    ) -> Result<Found, Failure> {
32        if interp.array_body(value).is_some() {
33            let caller = interp.caller();
34            interp.send_message(value, b"EMPTY", None, &[], caller)?;
35            return Ok(Found::Array(value));
36        }
37        if interp.is_stem_object(value) {
38            return Ok(Found::Stem(value));
39        }
40        // `ResolveStemVariable` clears whatever the string conversion raised.
41        if let Ok(Some(text)) = interp.string_conversion(value) {
42            let name = interp.to_text(text).into_owned();
43            if let Some(stem) = interp.context_stem_named(&name) {
44                return Ok(Found::Stem(stem));
45            }
46        }
47        Err(Raised::not_an_array_or_stem(position).into())
48    }
49
50    /// Adds `line` as the `count`th value.
51    pub(super) fn add(
52        self,
53        interp: &mut Interp,
54        count: usize,
55        line: Vec<u8>,
56    ) -> Result<(), Failure> {
57        let value = interp.text_built(line);
58        match self {
59            Found::Array(array) => {
60                interp.array_append(array, value)?;
61            }
62            Found::Stem(stem) => {
63                interp.stem_object_write(stem, count.to_string().as_bytes(), value)
64            }
65        }
66        Ok(())
67    }
68
69    /// Stores a stem's count into tail 0.
70    pub(super) fn complete(self, interp: &mut Interp, count: usize) {
71        if let Found::Stem(stem) = self {
72            let count = answer(interp, i64::try_from(count).unwrap_or(i64::MAX));
73            interp.stem_object_write(stem, b"0", count);
74        }
75    }
76}

78/// The lines `LineReader` reads: each ends at a LF, which is dropped with one
79/// CR before it, and a last line with no LF ends at the end of the file.
80fn lines(contents: &[u8]) -> impl Iterator<Item = &[u8]> {
81    let mut rest = contents;
82    std::iter::from_fn(move || {
83        if rest.is_empty() {
84            return None;
85        }
86        match rest.iter().position(|byte| *byte == b'\n') {
87            Some(end) => {
88                let line = &rest[..end];
89                rest = &rest[end + 1..];
90                Some(line.strip_suffix(b"\r").unwrap_or(line))
91            }
92            None => Some(std::mem::take(&mut rest)),
93        }
94    })
95}

97/// `mystrstr`: whether `needle` occurs in `line`, ASCII case folded when
98/// `caseless`. An empty needle occurs nowhere.
99fn contains(line: &[u8], needle: &[u8], caseless: bool) -> bool {
100    if needle.is_empty() || needle.len() > line.len() {
101        return false;
102    }
103    line.windows(needle.len()).any(|window| {
104        if caseless {
105            window.eq_ignore_ascii_case(needle)
106        } else {
107            window == needle
108        }
109    })
110}

113/// `snprintf` buffer less the NUL.
114pub(super) const OPTION_MESSAGE_LIMIT: usize = 255;

116/// `SysFileSearch(needle, file, stem [, options])`: the lines of `file`
117/// holding `needle`, into a stem or an array, each after its line number and
118/// a blank under option `N`, matched without case unless option `C`. Answers
119/// `0`, or `3` where the file is a directory or does not open, with nothing
120/// found.
121pub(crate) fn file_search(
122    interp: &mut Interp,
123    _name: &'static [u8],
124    args: &[Option<ObjRef>],
125) -> Result<ObjRef, Failure> {
126    check_arity(args, 4)?;
127    let needle = required(args, 1)?;
128    let Some(needle) = interp.string_conversion(needle)? else {
129        return Err(marshalled(Raised::argument_needs_a_string_value(1)));
130    };
131    let needle = interp.to_text(needle).into_owned();
132    let file = required_c_string(interp, args, 2)?;
133    let target = required(args, 3)?;
134    let options = optional_text(interp, args, 4)?;
135
136    let (mut numbered, mut caseless) = (false, true);
137    if let Some(options) = &options {
138        let options = c_string(options);
139        for letter in options {
140            match letter.to_ascii_uppercase() {
141                b'N' => numbered = true,
142                b'I' => caseless = true,
143                b'C' => caseless = false,
144                _ => {
145                    let mut text = b"SysFileSearch options argument must be a combination of \
146                                     C, I, or N; found \""
147                        .to_vec();
148                    text.extend_from_slice(options);
149                    text.push(b'"');
150                    text.truncate(OPTION_MESSAGE_LIMIT);
151                    return Err(Raised::routine_user_defined(&text).into());
152                }
153            }
154        }
155    }
156
157    let found = Found::resolve(interp, target, 3)?;
158    let path = qualify_name(interp, &file);
159    let path = os_path(&path);
160    let contents = if path.is_dir() {
161        None
162    } else {
163        std::fs::read(path).ok()
164    };
165    let Some(contents) = contents else {
166        found.complete(interp, 0);
167        return Ok(interp.text(b"3"));
168    };
169    let mut count = 0;
170    for (index, line) in lines(&contents).enumerate() {
171        if !contains(line, &needle, caseless) {
172            continue;
173        }
174        let mut value = Vec::with_capacity(line.len() + 8);
175        if numbered {
176            value.extend_from_slice(format!("{} ", index + 1).as_bytes());
177        }
178        value.extend_from_slice(line);
179        count += 1;
180        found.add(interp, count, value)?;
181    }
182    found.complete(interp, count);
183    Ok(interp.text(b"0"))
184}

186/// `SysSearchPath(variable, file [, option])`: the qualified name of the first
187/// regular file called `file` along the directories the environment variable
188/// lists, the current directory first unless the option starts `N`. The null
189/// string where it is not found.
190pub(crate) fn search_path(
191    interp: &mut Interp,
192    _name: &'static [u8],
193    args: &[Option<ObjRef>],
194) -> Result<ObjRef, Failure> {
195    check_arity(args, 3)?;
196    let variable = required_c_string(interp, args, 1)?;
197    let file = required_c_string(interp, args, 2)?;
198    let current_first = match optional_text(interp, args, 3)? {
199        None => true,
200        Some(text) => {
201            let option = c_string(&text);
202            match option.first().map(u8::to_ascii_uppercase) {
203                Some(b'C') => true,
204                Some(b'N') => false,
205                _ => {
206                    return Err(Raised::invalid_routine_option(
207                        "SysSearchPath",
208                        "option",
209                        "'C' or 'N'",
210                        option,
211                    )
212                    .into());
213                }
214            }
215        }
216    };
217    let value = interp.env_get(&variable).unwrap_or_default();
218    let found = search_path_for_file(interp, &file, value, current_first);
219    Ok(interp.text_built(found))
220}

222/// The search-path lookup. A name starting `/` or `~` is only qualified.
223/// Otherwise, `./name` and `../name` included, each non-empty entry of `path`
224/// is tried in turn, and the first that names anything ends the search: a
225/// regular file is the answer, anything else is not found.
226#[cfg(unix)]
227fn search_path_for_file(interp: &Interp, file: &[u8], path: &[u8], current_first: bool) -> Vec<u8> {
228    let has_directory = file.first() == Some(&b'~') || file.first() == Some(&b'/');
229    if has_directory {
230        let qualified = qualify_name(interp, file);
231        return if os_path(&qualified).is_file() {
232            qualified
233        } else {
234            Vec::new()
235        };
236    }
237    let current_and_path;
238    let path = if current_first {
239        let mut combined = interp.cwd_bytes().to_vec();
240        if !path.is_empty() {
241            combined.push(b':');
242            combined.extend_from_slice(path);
243        }
244        current_and_path = combined;
245        &current_and_path
246    } else {
247        path
248    };
249    for entry in path.split(|byte| *byte == b':') {
250        if entry.is_empty() {
251            continue;
252        }
253        let mut candidate = entry.to_vec();
254        if candidate.last() != Some(&b'/') {
255            candidate.push(b'/');
256        }
257        candidate.extend_from_slice(file);
258        let qualified = qualify_name(interp, &candidate);
259        if qualified.is_empty() {
260            continue;
261        }
262        if let Ok(metadata) = std::fs::metadata(os_path(&qualified)) {
263            return if metadata.is_file() {
264                qualified
265            } else {
266                Vec::new()
267            };
268        }
269    }
270    Vec::new()
271}

273#[cfg(windows)]
274fn search_path_for_file(interp: &Interp, file: &[u8], path: &[u8], current_first: bool) -> Vec<u8> {
275    let file_path = os_path(file);
276    if file.is_empty() || file.iter().any(|byte| matches!(byte, b'*' | b'?')) {
277        return Vec::new();
278    }
279    if file_path.has_root()
280        || matches!(
281            file_path.components().next(),
282            Some(std::path::Component::Prefix(_))
283        )
284    {
285        let qualified = qualify_name(interp, file);
286        return if os_path(&qualified).is_file() {
287            qualified
288        } else {
289            Vec::new()
290        };
291    }
292
293    let current = current_first.then(|| os_path(interp.cwd_bytes()).to_path_buf());
294    let directories = std::env::split_paths(os_path(path).as_os_str());
295    for directory in current.into_iter().chain(directories) {
296        if directory.as_os_str().is_empty() {
297            continue;
298        }
299        let candidate = directory.join(file_path);
300        let Some(candidate) = candidate.to_str() else {
301            continue;
302        };
303        let qualified = qualify_name(interp, candidate.as_bytes());
304        if os_path(&qualified).is_file() {
305            return qualified;
306        }
307    }
308    Vec::new()
309}

311/// `SysTempFileName(template [, filler])`. Each filler
312/// character in the template becomes a digit, the first the lowest, from a
313/// random start counting up until the qualified name names nothing. The null
314/// string where the template has no filler or more than nine, or every name
315/// is taken.
316pub(crate) fn temp_file_name(
317    interp: &mut Interp,
318    _name: &'static [u8],
319    args: &[Option<ObjRef>],
320) -> Result<ObjRef, Failure> {
321    check_arity(args, 2)?;
322    let template = required_c_string(interp, args, 1)?;
323    let filler = match optional_text(interp, args, 2)? {
324        None => b'?',
325        Some(text) => match c_string(&text) {
326            [filler] => *filler,
327            other => {
328                return Err(Raised::routine_argument_not_a_character(
329                    "SysTempFileName",
330                    "filler",
331                    other,
332                )
333                .into());
334            }
335        },
336    };
337    let name = unique_name(interp, &template, filler);
338    Ok(interp.text_built(name))
339}

341/// [`temp_file_name`]'s search.
342fn unique_name(interp: &Interp, template: &[u8], filler: u8) -> Vec<u8> {
343    let places = template.iter().filter(|byte| **byte == filler).count();
344    if places == 0 || places > 9 {
345        return Vec::new();
346    }
347    let limit = 10u64.pow(u32::try_from(places).expect("nine at most"));
348    let start = super::random_name_seed() % limit;
349    let mut number = start;
350    let mut candidate = template.to_vec();
351    loop {
352        let mut rest = number;
353        for (at, byte) in template.iter().enumerate() {
354            if *byte == filler {
355                candidate[at] = b'0' + u8::try_from(rest % 10).expect("a digit");
356                rest /= 10;
357            }
358        }
359        let qualified = qualify_name(interp, &candidate);
360        #[cfg(unix)]
361        if rustix::fs::stat(qualified.as_slice()).is_err() {
362            return qualified;
363        }
364        #[cfg(windows)]
365        match std::fs::symlink_metadata(os_path(&qualified)) {
366            Ok(_) => {}
367            Err(error) if error.kind() == std::io::ErrorKind::NotFound => return qualified,
368            Err(_) => return Vec::new(),
369        }
370        number = (number + 1) % limit;
371        if number == start {
372            return Vec::new();
373        }
374    }
