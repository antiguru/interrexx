19/// What one call looks for and how it reports what it finds, from the option
20/// letters.
21#[derive(Clone, Copy)]
22struct TreeOptions {
23    files: bool,
24    directories: bool,
25    recurse: bool,
26    /// `O`: the name alone, with no date, size or attributes.
27    names_only: bool,
28    /// `T`: the date as `YY/MM/DD/HH/MM`.
29    editable_time: bool,
30    /// `L`: the date as `YYYY-MM-DD HH:MM:SS`, over `T`.
31    long_time: bool,
32    /// `H`: the size twenty wide and never capped.
33    long_size: bool,
34    /// `I`: the final name matched without case.
35    #[cfg(unix)]
36    caseless: bool,
37}
38
39impl TreeOptions {
40    /// The letters in any case, each of `F`, `D` and
41    /// `B` replacing what an earlier one chose. `None` for any other letter.
42    fn parse(text: &[u8]) -> Option<TreeOptions> {
43        let mut options = TreeOptions {
44            files: true,
45            directories: true,
46            recurse: false,
47            names_only: false,
48            editable_time: false,
49            long_time: false,
50            long_size: false,
51            #[cfg(unix)]
52            caseless: false,
53        };
54        for letter in text {
55            match letter.to_ascii_uppercase() {
56                b'F' => (options.files, options.directories) = (true, false),
57                b'D' => (options.files, options.directories) = (false, true),
58                b'B' => (options.files, options.directories) = (true, true),
59                b'S' => options.recurse = true,
60                b'O' => options.names_only = true,
61                b'T' => options.editable_time = true,
62                b'L' => options.long_time = true,
63                b'H' => options.long_size = true,
64                #[cfg(unix)]
65                b'I' => options.caseless = true,
66                #[cfg(windows)]
67                b'I' => {}
68                _ => return None,
69            }
70        }
71        Some(options)
72    }
73}

75/// A 40.900 whose message `snprintf` formats into 256 bytes.
76fn user_defined(mut text: Vec<u8>) -> Failure {
77    text.truncate(OPTION_MESSAGE_LIMIT);
78    Raised::routine_user_defined(&text).into()
79}

81/// An attribute mask, cut at a NUL, is empty or up
82/// to five of `+`, `-` and `*`. Unix applies neither mask.
83fn check_mask(mask: Option<&[u8]>, position: usize) -> Result<(), Failure> {
84    let mask = mask.map_or(&[][..], c_string);
85    if mask.len() > 5 || mask.iter().any(|byte| !matches!(byte, b'+' | b'-' | b'*')) {
86        let mut text = format!(
87            "SysFileTree argument {position} must be 5 characters or less in length \
88             containing only '+', '-', or '*'; found \""
89        )
90        .into_bytes();
91        text.extend_from_slice(mask);
92        text.push(b'"');
93        return Err(user_defined(text));
94    }
95    Ok(())
96}

98/// `SysFileTree(filespec, stem [, options [, tattrib [, nattrib]]])`: the
99/// names matching `filespec`, into a stem or an array. The directory part of
100/// the spec is literal and only its final name may hold wildcards.
101///
102/// Errors after the stem is resolved still set its tail `0` to `0`, as
103/// `StemHandler`'s destructor does. The order entries are found in is the
104/// directory's own, which is not specified.
105pub(crate) fn file_tree(
106    interp: &mut Interp,
107    _name: &'static [u8],
108    args: &[Option<ObjRef>],
109) -> Result<ObjRef, Failure> {
110    check_arity(args, 5)?;
111    let spec = required_c_string(interp, args, 1)?;
112    let target = required(args, 2)?;
113    let options = optional_text(interp, args, 3)?;
114    let target_mask = optional_text(interp, args, 4)?;
115    let new_mask = optional_text(interp, args, 5)?;
116    let found = Found::resolve(interp, target, 2)?;
117
118    let options = match validate(
119        &spec,
120        options.as_deref(),
121        target_mask.as_deref(),
122        new_mask.as_deref(),
123    ) {
124        Ok(options) => options,
125        Err(failure) => {
126            found.complete(interp, 0);
127            return Err(failure);
128        }
129    };
130    let lines = platform::find(
131        interp,
132        &spec,
133        options,
134        target_mask.as_deref().map(c_string).unwrap_or_default(),
135        new_mask.as_deref().map(c_string).unwrap_or_default(),
136    );
137    let count = lines.len();
138    for (index, line) in lines.into_iter().enumerate() {
139        found.add(interp, index + 1, line)?;
140    }
141    found.complete(interp, count);
142    Ok(answer(interp, 0))
143}

145/// The arguments' checks in the order `TreeFinder`'s constructor makes them:
146/// the spec, the options, then the two masks.
147fn validate(
148    spec: &[u8],
149    options: Option<&[u8]>,
150    target_mask: Option<&[u8]>,
151    new_mask: Option<&[u8]>,
152) -> Result<TreeOptions, Failure> {
153    if spec.is_empty() {
154        return Err(Raised::routine_null_argument("SysFileTree", 1).into());
155    }
156    let options = match options.map(c_string) {
157        None => TreeOptions::parse(b"").expect("no letters is valid"),
158        Some([]) => return Err(Raised::routine_null_argument("SysFileTree", 3).into()),
159        Some(letters) => TreeOptions::parse(letters).ok_or_else(|| {
160            let mut text = b"SysFileTree options argument must be a combination of \
161                             F, D, B, S, T, L, I, O, or H; found \""
162                .to_vec();
163            text.extend_from_slice(letters);
164            text.push(b'"');
165            user_defined(text)
166        })?,
167    };
168    check_mask(target_mask, 4)?;
169    check_mask(new_mask, 5)?;
170    Ok(options)
171}
