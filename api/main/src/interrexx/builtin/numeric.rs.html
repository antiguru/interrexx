1//! The numeric builtins: `ABS`, `FORMAT`, `MAX`, `MIN`, `RANDOM`, `SIGN` and
2//! `TRUNC`.
3//! ```text
4//! max('a',1,3)   40.11    names MAX, argument 1 and the value a.
5//! max(1,'a',3)   40.11    names MAX, argument 2 and the value a.
6//! max(1,2,'a')   40.11    names MAX, argument 3 and the value a.
7//! ```
8//! ```rexx
9//! numeric digits 5 ; x = max(123456789,1) ; numeric digits 12 ; say x
10//!   -> 1.2346E+8, not 123456789
11//! numeric form engineering ; x = max(1e10,1) ; numeric form scientific ; say x
12//!   -> 10E+9, not 1E+10
13//! numeric digits 3 ; x = format(1.23456,,4) ; numeric digits 9 ; say x
14//!   -> 1.2300, which is text and could not have moved
15//! ```

25/// The `NUMERIC` settings this family reads, fetched once per call.
26struct Numeric {
27    digits: u64,
28    fuzz: u64,
29    form: Form,
30}
31
32fn current(interp: &Interp) -> Numeric {
33    let settings = &interp.activation().settings;
34    Numeric {
35        digits: settings.digits(),
36        fuzz: settings.fuzz(),
37        form: settings.form(),
38    }
39}

41/// Argument 1 as a number, or the builtin's own 40.11
42/// (`r[conv.method.number]`). The
43/// builtins round it without LOSTDIGITS, as ooTest's `base/keyword/LOSTDIGITS
44/// TESTBIFS` reads rexxref.
45fn target_number(
46    interp: &mut Interp,
47    name: &'static [u8],
48    args: Args<'_>,
49) -> Result<Number, Failure> {
50    let value = arg(args, 1).expect("check_arity admitted the required first argument");
51    match interp.to_number(value) {
52        Ok(number) => Ok(number),
53        Err(_) => {
54            let found = required_string(interp, args, 1);
55            Err(Raised::argument_not_a_number(name, 1, &found).into())
56        }
57    }
58}

60/// The range half of an optional non-negative argument, after
61/// [`super::whole_number`] has already done the type half: the builtin's own
62/// 40.13 naming the argument's `position` in the call.
63fn non_negative(name: &[u8], value: Option<i64>, position: usize) -> Result<Option<i64>, Failure> {
64    match value {
65        Some(value) if value < 0 => Err(Raised::argument_not_non_negative_call(
66            name,
67            position,
68            value.to_string().as_bytes(),
69        )
70        .into()),
71        other => Ok(other),
72    }
73}

75/// A width the result is padded out to, as the `u32` `interrexx-num` takes, having
76/// first asked the allocator for it. A width past `heap_limit` raises 5.0
77/// (r[env.invocation.heap-limit.raise]).
78fn padding_width(heap_limit: Option<usize>, value: i64) -> Result<u32, Failure> {
79    let width = u32::try_from(value).map_err(|_| Failure::from(Raised::system_resources()))?;
80    crate::admit_within(heap_limit, usize::cast_from(width))?;
81    fresh_buffer(usize::cast_from(width))?;
82    Ok(width)
83}

97/// [`abs`]'s answer once its target is a `Number`, shared with `String~abs`
98/// the way [`sign_of`] is shared with `String~sign`.
99pub(crate) fn abs_of(interp: &mut Interp, value: &Number) -> Result<ObjRef, Failure> {
100    let Numeric { digits, form, .. } = current(interp);
101    let rounded = rounded_target(value.abs(), digits)?;
102    Ok(interp.number(rounded, saturate(digits), form))
103}

129/// `String~MODULO`'s answer once its target and divisor are both in hand:
130/// the `//` remainder, with the divisor added back when that came out
131/// negative.
132pub(crate) fn modulo_over(
133    interp: &mut Interp,
134    value: &Number,
135    divisor: &Number,
136) -> Result<ObjRef, Failure> {
137    let Numeric { digits, form, .. } = current(interp);
138    let remainder = value
139        .div(divisor, digits, DivOp::Remainder)
140        .map_err(Raised::from)?;
141    let answer = if remainder.signum() < 0 {
142        remainder.add(divisor, digits).map_err(Raised::from)?
143    } else {
144        remainder
145    };
146    Ok(interp.number(answer, saturate(digits), form))
147}

163    let rounded = rounded_target(value.clone(), digits)?;

170/// `TRUNC(number, decimals)`.
171/// ```text
172/// trunc('AB.CD','V')   40.12   names TRUNC, argument 2 and the value V.
173/// trunc('AB.CD',-1)    40.11   names TRUNC, argument 1 and the value AB.CD.
174/// trunc(1.5,-1)        40.13   names TRUNC, argument 2 and the value -1.
175/// ```
176pub(crate) fn trunc(
177    interp: &mut Interp,
178    name: &'static [u8],
179    args: Args<'_>,
180) -> Result<ObjRef, Failure> {
181    let places = whole_number(interp, name, args, 2)?;
182    let value = target_number(interp, name, args)?;
183    trunc_of(interp, &value, non_negative(name, places, 2)?)
184}

186/// [`trunc`]'s answer once its target and its places are in hand, shared with
187/// `String~trunc`.
188pub(crate) fn trunc_of(
189    interp: &mut Interp,
190    value: &Number,
191    places: Option<i64>,
192) -> Result<ObjRef, Failure> {
193    let Numeric { digits, .. } = current(interp);
194    let value = rounded_target(value.clone(), digits)?;
195    let places = padding_width(interp.heap_limit, places.unwrap_or(0))?;
196    Ok(interp.integer_text(value.trunc(digits, places).into_bytes(), digits))
197}

199/// `FORMAT(number, before, after, expp, expt)`.
200pub(crate) fn format(
201    interp: &mut Interp,
202    name: &'static [u8],
203    args: Args<'_>,
204) -> Result<ObjRef, Failure> {
205    let before = whole_number(interp, name, args, 2)?;
206    let after = whole_number(interp, name, args, 3)?;
207    let expp = whole_number(interp, name, args, 4)?;
208    let expt = whole_number(interp, name, args, 5)?;
209    // **A short number laid out plainly, without a `Number` built**: no
210    // exponent arguments, a target whose digits fit a `Scaled`, and a
211    // result that is plain and short. The target converted, the widths'
212    // range checks are the next and last things that can fail.
213    if expp.is_none()
214        && expt.is_none()
215        && let Some(scaled) = interp
216            .scaled_operand(arg(args, 1).expect("check_arity admitted the required first argument"))
217    {
218        let before = non_negative(name, before, 2)?;
219        let after = non_negative(name, after, 3)?;
220        let mut out = [0u8; 48];
221        let width = |value: Option<i64>| value.map(u32::try_from).transpose().ok();
222        if let (Some(before), Some(after)) = (width(before), width(after))
223            && let Some(written) =
224                scaled.format_plain(current(interp).digits, before, after, &mut out)
225        {
226            return Ok(interp.text(&out[..written]));
227        }
228    }
229    let value = target_number(interp, name, args)?;
230
231    format_over(
232        interp,
233        &value,
234        non_negative(name, before, 2)?,
235        non_negative(name, after, 3)?,
236        non_negative(name, expp, 4)?,
237        non_negative(name, expt, 5)?,
238    )
239}

241/// [`format`]'s answer once its target and its four widths are in hand, shared
242/// with `String~format`.
243pub(crate) fn format_over(
244    interp: &mut Interp,
245    value: &Number,
246    before: Option<i64>,
247    after: Option<i64>,
248    expp: Option<i64>,
249    expt: Option<i64>,
250) -> Result<ObjRef, Failure> {
251    let Numeric { digits, form, .. } = current(interp);
252    let value = &rounded_target(value.clone(), digits)?;
253    // `before` and `after` are always materialised -- the interpreter's
254    // leading spaces and trailing decimal zeros are computed on every path
255    // -- so both are reserved from the allocator before anything is built.
256    let heap_limit = interp.heap_limit;
257    let before = before
258        .map(|width| padding_width(heap_limit, width))
259        .transpose()?;
260    let after = after
261        .map(|width| padding_width(heap_limit, width))
262        .transpose()?;
263
264    // `expt` is a *threshold* and never occupies a byte, so a value past
265    // `u32::MAX` saturates there rather than being refused: every exponent it
266    // is compared against is bounded by `MAX_EXPONENT` (999,999,999), so
267    // `u32::MAX` and nine quintillion decide every comparison identically.
268    // Measured, `format(1,,,,999999999999999999)` is `1`.
269    let expt = expt.map(|value| u32::try_from(value).unwrap_or(u32::MAX));
270
271    // `expp` is the one width the interpreter does not always write: it pads an
272    // exponent, and there is not always an exponent. So it is reserved only once
273    // `interrexx-num` says the field will be written, which is the same line
274    // `format` draws: at the identical width, `format(1,,,3000000000)` is `1` at
275    // rc 0 while `format(1,,,3000000000,0)` is error 5 at
276    // rc 251.
277    let displayed = value
278        .format_exponent(
279            digits,
280            form,
281            after,
282            expp.map(|width| u32::try_from(width).unwrap_or(u32::MAX)),
283            expt,
284        )
285        .is_some();
286    let expp = match expp {
287        None => None,
288        // Not a width at all: zero is the sentinel that suppresses
289        // exponential form, and it has to survive to `interrexx-num`.
290        Some(0) => Some(0),
291        Some(width) if displayed => Some(padding_width(heap_limit, width)?),
292        // A width that is never written changes nothing about the result, so
293        // it is dropped rather than narrowed. `interrexx-num` reads `expp` only
294        // through that zero test and through the exponential branch, and
295        // `format_exponent` has just said the second is not taken.
296        Some(_) => None,
297    };
298
299    let text = value
300        .format_with(digits, form, before, after, expp, expt)
301        .map_err(|error| Failure::from(Raised::from(error)))?;
302    Ok(interp.text_built(text.into_bytes()))
303}

307/// Which end of the range [`max_min`] is looking for.
308#[derive(Clone, Copy, PartialEq, Eq)]
309pub(crate) enum Extreme {
310    Max,
311    Min,
312}
313
314impl Extreme {
315    /// The comparison that makes a candidate the new answer, which honours
316    /// `FUZZ`.
317    fn op(self) -> CompareOp {
318        match self {
319            Extreme::Max => CompareOp::Greater,
320            Extreme::Min => CompareOp::Less,
321        }
322    }
323}

343/// The body both of them share.
344fn max_min(
345    interp: &mut Interp,
346    name: &'static [u8],
347    args: Args<'_>,
348    want: Extreme,
349) -> Result<ObjRef, Failure> {
350    let rest = args.values_from(2);
351    let best = target_number(interp, name, args)?;
352    let best_text = required_string(interp, args, 1);
353    let objects: Vec<Option<ObjRef>> = (0..rest.len()).map(|i| args.object(i + 2)).collect();
354    max_min_over(interp, name, &best, best_text, rest, &objects, want, false)
355}

357/// `MAX`'s and `MIN`'s answer, shared with `String~max` and `String~min`.
358/// Every number is rounded to DIGITS and the comparisons honour FUZZ. The
359/// target never raises LOSTDIGITS; an argument wider than DIGITS does, and
360/// so does each comparison under FUZZ, unless every value is written as a
361/// whole number and the target fits DIGITS (Deviation 44). An omitted
362/// argument is 40.5 for the function and 93.903 for the message, numbered by
363/// its position in the call.
364#[allow(
365    clippy::too_many_arguments,
366    reason = "the two callers differ in each of them"
367)]
368pub(crate) fn max_min_over(
369    interp: &mut Interp,
370    name: &'static [u8],
371    target: &Number,
372    target_text: Vec<u8>,
373    rest: &[Option<ObjRef>],
374    objects: &[Option<ObjRef>],
375    want: Extreme,
376    message: bool,
377) -> Result<ObjRef, Failure> {
378    let Numeric { digits, fuzz, form } = current(interp);
379    let mut best = rounded_target(target.clone(), digits)?;
380    let (converted, quiet) = extreme_arguments(interp, &target_text, rest, digits)?;
381    let rest = converted.as_ref();
382    let mut best_text = target_text;
383    for (index, slot) in rest.iter().enumerate() {
384        let Some(candidate) = slot else {
385            return Err(if message {
386                Raised::missing_method_argument(index + 1)
387            } else {
388                Raised::missing_argument(name, index + 2)
389            }
390            .into());
391        };
392        let candidate = interp.required_string_value(*candidate)?;
393        interp.roots.activity_mut().push_temp(candidate);
394        let found = interp.to_text(candidate).into_owned();
395        let Some(number) = interp.to_number(candidate).ok() else {
396            // The object rather than the conversion, which is every
397            // argument message's rule -- `whole_number`'s own doc has the
398            // measurement behind it.
399            let named = match objects.get(index).copied().flatten() {
400                Some(object) => interp.string_value_text(object)?,
401                None => found,
402            };
403            return Err(if message {
404                Raised::method_argument_not_a_number(index + 1, &named)
405            } else {
406                Raised::argument_not_a_number(name, index + 2, &named)
407            }
408            .into());
409        };
410        // The description is the number unrounded, rendered at DIGITS: `'
411        // 4444444444 '` names `4444444444` and `44444444440E-1` names
412        // `4.4444444440E+9`.
413        if !quiet {
414            interp.lostdigits_check_with(&number, |_| {
415                number.render_unrounded(digits, form).into_bytes()
416            })?;
417        }
418        let number = number.into_round(digits);
419        if !quiet {
420            interp.lostdigits_check_fuzz(
421                &number,
422                |_| number.render_unrounded(digits, form).into_bytes(),
423                &best,
424                |_| best.render_unrounded(digits, form).into_bytes(),
425                fuzz,
426            )?;
427        }
428        let takes_over = compare_decoded(
429            &found,
430            Some(&number),
431            &best_text,
432            Some(&best),
433            digits,
434            fuzz,
435            want.op(),
436        )
437        .map_err(|error| Failure::from(Raised::from(error)))?;
438        if takes_over {
439            best = number;
440            best_text = found;
441        }
442    }
443    Ok(interp.number(best, saturate(digits), form))
444}

446/// The arguments and their whole-number quietness, retaining ordinary values in place.
447fn extreme_arguments<'a>(
448    interp: &mut Interp,
449    target: &[u8],
450    rest: &'a [Option<ObjRef>],
451    digits: u64,
452) -> Result<(std::borrow::Cow<'a, [Option<ObjRef>]>, bool), Failure> {
453    let width =
454        u32::try_from(digits.min(u64::cast_from(WRITTEN_WHOLE_DIGITS))).expect("at most eighteen");
455    if !whole_written(target).is_some_and(|value| value.unsigned_abs() < 10u64.pow(width)) {
456        return Ok((std::borrow::Cow::Borrowed(rest), false));
457    }
458    for slot in rest {
459        let Some(value) = slot else {
460            return Ok((std::borrow::Cow::Borrowed(rest), false));
461        };
462        if needs_string(interp, *value) {
463            let converted = convert_extreme_arguments(interp, rest)?;
464            let quiet = written_whole(interp, target, &converted, digits);
465            return Ok((converted, quiet));
466        }
467        if whole_written(&interp.to_text(*value)).is_none() {
468            return Ok((std::borrow::Cow::Borrowed(rest), false));
469        }
470    }
471    Ok((std::borrow::Cow::Borrowed(rest), true))
472}

474/// Whether a value needs the required-string protocol before numeric parsing.
475fn needs_string(interp: &Interp, value: ObjRef) -> bool {
476    match value.decode() {
477        interrexx_core::Decoded::SmallInt(_) | interrexx_core::Decoded::Text(_) => false,
478        _ => !matches!(
479            interp.heap.get(value).map(|object| &object.body),
480            Some(interrexx_core::Body::Text { .. } | interrexx_core::Body::Num { .. })
481        ),
482    }
483}

485/// Converts the prefix needed to decide whole-number quietness, stopping when quietness is impossible.
486#[cold]
487fn convert_extreme_arguments<'a>(
488    interp: &mut Interp,
489    rest: &'a [Option<ObjRef>],
490) -> Result<std::borrow::Cow<'a, [Option<ObjRef>]>, Failure> {
491    let mut converted = std::borrow::Cow::Borrowed(rest);
492    for (at, slot) in rest.iter().enumerate() {
493        let Some(value) = slot else { break };
494        let text = interp.required_string_value(*value)?;
495        interp.roots.activity_mut().push_temp(text);
496        if text != *value {
497            converted.to_mut()[at] = Some(text);
498        }
499        if whole_written(&interp.to_text(text)).is_none() {
500            break;
501        }
502    }
503    Ok(converted)
504}

506/// The widest a whole number can be written and still count as one here.
507const WRITTEN_WHOLE_DIGITS: usize = 18;
508
509/// The value written as a whole number: an optional minus, then at most
510/// [`WRITTEN_WHOLE_DIGITS`] digits with no leading zero.
511fn whole_written(text: &[u8]) -> Option<i64> {
512    let digits = text.strip_prefix(b"-").unwrap_or(text);
513    if digits.is_empty() || digits.len() > WRITTEN_WHOLE_DIGITS {
514        return None;
515    }
516    if !digits.iter().all(u8::is_ascii_digit) || (digits[0] == b'0' && digits.len() > 1) {
517        return None;
518    }
519    std::str::from_utf8(text).ok()?.parse().ok()
520}

522/// Whether `MAX`/`MIN` stay quiet on LOSTDIGITS (Deviation 44): the target is
523/// written as a whole number that fits `digits`, and so is every argument.
524fn written_whole(interp: &mut Interp, target: &[u8], rest: &[Option<ObjRef>], digits: u64) -> bool {
525    let width =
526        u32::try_from(digits.min(u64::cast_from(WRITTEN_WHOLE_DIGITS))).expect("at most eighteen");
527    whole_written(target).is_some_and(|value| value.unsigned_abs() < 10u64.pow(width))
528        && rest
529            .iter()
530            .all(|slot| slot.is_some_and(|arg| whole_written(&interp.to_text(arg)).is_some()))
531}

535const RANDOM_FACTOR: u64 = 25_214_903_917;
536const RANDOM_ADDER: u64 = 11;

538/// `DefaultRandomMin`/`DefaultRandomMax`/`MaxRandomRange`.
539const DEFAULT_MIN: i64 = 0;
540const DEFAULT_MAX: i64 = 999;
541const MAX_RANGE: i64 = 999_999_999;

543/// `RANDOMIZE`, wrapping because the multiplication is on `uint64_t`.
544fn randomize(seed: u64) -> u64 {
545    seed.wrapping_mul(RANDOM_FACTOR).wrapping_add(RANDOM_ADDER)
546}

548/// `RANDOM(minimum, maximum, seed)`.
549pub(crate) fn random(
550    interp: &mut Interp,
551    name: &'static [u8],
552    args: Args<'_>,
553) -> Result<ObjRef, Failure> {
554    let minimum = whole_number(interp, name, args, 1)?;
555    let maximum = whole_number(interp, name, args, 2)?;
556    let seed = whole_number(interp, name, args, 3)?;
557
558    // The seed is validated and *applied* before the range is looked at, which
559    // is done first. A call that then fails its range check has still moved the
560    // stream on.
561    if let Some(seed) = seed
562        && seed < 0
563    {
564        let found = seed.to_string();
565        return Err(Raised::argument_not_non_negative_call(name, 3, found.as_bytes()).into());
566    }
567    let scrambled = next_seed(interp, seed);
568
569    // `RANDOM`'s three arguments do not mean the same thing in every
570    // combination, and the seed is what decides: with no maximum and no
571    // seed, a lone argument is the *maximum*, but with a seed it is the
572    // minimum. Measured, `random(0)` is always `0` while `random(1,,2)` --
573    // the same lone argument with a seed behind it -- answered `313`, which
574    // is above it.
575    let (mut low, high) = match (minimum, maximum, seed) {
576        (Some(low), None, None) => (DEFAULT_MIN, low),
577        (Some(low), None, Some(_)) => (low, DEFAULT_MAX),
578        (Some(low), Some(high), _) => (low, high),
579        (None, Some(high), _) => (DEFAULT_MIN, high),
580        (None, None, _) => (DEFAULT_MIN, DEFAULT_MAX),
581    };
582
583    // An argument the call omitted substitutes as the null string, because no
584    // object is passed: `random(-1)` reports `argument 1 ("-1")` and `argument
585    // 2 ("")`.
586    let written = |value: Option<i64>| value.map_or(Vec::new(), |v| v.to_string().into_bytes());
587    if high < low {
588        return Err(Raised::random_bounds_reversed(&written(minimum), &written(maximum)).into());
589    }
590    if high - low > MAX_RANGE {
591        return Err(Raised::random_range_too_wide(&written(minimum), &written(maximum)).into());
592    }
593
594    if low != high {
595        let spread = u64::try_from(high - low + 1).expect("the range check bounded this");
596        low += i64::try_from(reverse_bits(scrambled) % spread).expect("a value below the spread");
597    }
598    // **Text, not a number.** the minimum is answered as an integer object,
599    // whose spelling is its own decimal digits and is never re-rendered under a
600    // later `DIGITS` or `FORM`. So `numeric digits 3 ; say random(12345,12345)`
601    // is `12345`, where a value carrying the D15 pair would render `1.23E+4` --
602    // and `say random(12345,12345) + 0` at the same setting *is* `1.23E+4`,
603    // because that is the addition's own result rather than this one.
604    Ok(interp.text_built(low.to_string().into_bytes()))
605}

607/// Install a supplied seed, then advance the
608/// stream once and answer the new state.
609/// The state is the nearest top-level activation's, seeded on its first
610/// unseeded draw from the activity's generator.
611fn next_seed(interp: &mut Interp, seed: Option<i64>) -> u64 {
612    let held = match seed {
613        // "flipping all of the bits gives us a better spread", then thirteen
614        // scrambles, then the unconditional fourteenth below.
615        Some(seed) => {
616            let mut state = !seed.cast_unsigned();
617            for _ in 0..13 {
618                state = randomize(state);
619            }
620            Some(state)
621        }
622        None => interp
623            .top_level_activation_mut()
624            .and_then(|activation| activation.cold.as_deref()?.random_seed),
625    };
626    let state = randomize(held.unwrap_or_else(|| activation_seed(interp)));
627    if let Some(activation) = interp.top_level_activation_mut() {
628        activation.cold_mut().random_seed = Some(state);
629    }
630    state
631}

633/// A new activation's seed, advancing the
634/// activity's generator once. A splitmix64 step, so successive activations'
635/// streams are not shifted copies of one another.
636fn activation_seed(interp: &mut Interp) -> u64 {
637    if interp.activity.random_source.is_none() {
638        interp.activity.random_source = Some(initial_seed(interp));
639    }
640    let source = interp
641        .activity
642        .random_source
643        .as_mut()
644        .expect("the generator was seeded above");
645    *source = source.wrapping_add(0x9E37_79B9_7F4A_7C15);
646    let mut mixed = *source;
647    mixed = (mixed ^ (mixed >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
648    mixed = (mixed ^ (mixed >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
649    mixed ^ (mixed >> 31)
650}
651
652/// The generator's starting state: the random stream's draw in the
653/// simulation mode, else the clock's nanoseconds and the process id.
654fn initial_seed(interp: &mut Interp) -> u64 {
655    if let Some(seed) = interp.sim_random_seed() {
656        return seed;
657    }
658    let nanos = interp
659        .wall_now()
660        .duration_since(std::time::UNIX_EPOCH)
661        .map_or(0, |elapsed| elapsed.subsec_nanos());
662    u64::from(nanos) << 32 ^ u64::from(std::process::id())
663}

665/// The bit reversal done before taking the modulus: the seed's bits, most
666/// significant first.
667fn reverse_bits(seed: u64) -> u64 {
668    let mut work = 0u64;
669    let mut seed = seed;
670    for _ in 0..u64::BITS {
671        work = (work << 1) | (seed & 1);
672        seed >>= 1;
673    }
674    work
675}

677/// The `DIGITS` in force, narrowed the way every other value-producing site
678/// narrows it (`eval::saturate_digits`).
679fn saturate(digits: u64) -> u32 {
680    crate::eval::saturate_digits(digits)
681}
