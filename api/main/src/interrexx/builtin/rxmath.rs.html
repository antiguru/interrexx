1//! The `rxmath` extension library, run in Rust: `MathLoadFuncs`,
2//! `MathDropFuncs` and the sixteen `RxCalc` routines, as
3//! `docs/design/extensions.md` describes them.
4//!
5//! Every result is a double rendered by [`crate::dispatch::library::double_text`] at the
6//! routine's precision. The trigonometric routines reduce a degree or grade
7//! angle before any library call, and form the radian angle in the x87's
8//! extended precision, which [`extended`] carries out in integers.
9

22/// 88.922 where more arguments were passed than `arity`. An omitted argument at
23/// the end does not count: `RxCalcSqrt(4,,,,)` is `2`.
24pub(crate) fn check_arity(args: &[Option<ObjRef>], arity: usize) -> Result<(), Failure> {
25    let passed = args
26        .iter()
27        .rposition(Option::is_some)
28        .map_or(0, |at| at + 1);
29    if passed > arity {
30        return Err(Raised::too_many_external_arguments(arity).into());
31    }
32    Ok(())
33}

35/// The argument at `position`, 88.901 where it is omitted.
36pub(crate) fn required(args: &[Option<ObjRef>], position: usize) -> Result<ObjRef, Failure> {
37    args.get(position - 1)
38        .copied()
39        .flatten()
40        .ok_or_else(|| Raised::missing_native_argument(position).into())
41}

43/// A floating argument: a Rexx number, read as C's `strtod` reads its text,
44/// or exactly `nan`, `+infinity` or `-infinity`. 88.921 for anything else.
45pub(crate) fn double_argument(
46    interp: &mut Interp,
47    args: &[Option<ObjRef>],
48    position: usize,
49) -> Result<f64, Failure> {
50    let value = required(args, position)?;
51    if let Decoded::SmallInt(number) = value.decode() {
52        // A whole number converts to the nearest double, as C's does.
53        return Ok(f64::cast_lossy(number));
54    }
55    let text = interp.required_string_value(value)?;
56    let bytes = interp.to_text(text).into_owned();
57    if Number::parse_bytes(&bytes).is_some() {
58        return Ok(strtod(&bytes));
59    }
60    match &bytes[..] {
61        b"nan" => Ok(f64::NAN),
62        b"+infinity" => Ok(f64::INFINITY),
63        b"-infinity" => Ok(f64::NEG_INFINITY),
64        _ => {
65            let found = interp.native_found(value);
66            Err(Raised::native_argument_not_a_double(position, &found).into())
67        }
68    }
69}

71/// C's `strtod` over a valid Rexx number: leading white space skipped, then the
72/// longest prefix that is a decimal floating literal, and 0 where there is
73/// none. For example, `RxCalcPower(' -4', 1)` is `-4` and `RxCalcPower('- 4',
74/// 1)` is `0`, because a blank after the sign ends the literal before it
75/// starts.
76fn strtod(text: &[u8]) -> f64 {
77    let start = text
78        .iter()
79        .position(|byte| !byte.is_ascii_whitespace())
80        .unwrap_or(text.len());
81    let text = &text[start..];
82    let mut end = 0;
83    if matches!(text.first(), Some(b'+' | b'-')) {
84        end = 1;
85    }
86    let digits = |from: usize| {
87        text[from..]
88            .iter()
89            .take_while(|byte| byte.is_ascii_digit())
90            .count()
91    };
92    let whole = digits(end);
93    end += whole;
94    let mut fraction = 0;
95    if text.get(end) == Some(&b'.') {
96        fraction = digits(end + 1);
97        if whole + fraction > 0 {
98            end += 1 + fraction;
99        }
100    }
101    if whole + fraction == 0 {
102        return 0.0;
103    }
104    if matches!(text.get(end), Some(b'e' | b'E')) {
105        let mut at = end + 1;
106        if matches!(text.get(at), Some(b'+' | b'-')) {
107            at += 1;
108        }
109        let exponent = digits(at);
110        if exponent > 0 {
111            end = at + exponent;
112        }
113    }
114    let literal = std::str::from_utf8(&text[..end]).expect("an ASCII literal");
115    literal
116        .parse()
117        .unwrap_or_else(|_| unreachable!("a decimal literal parses: {literal}"))
118}

120/// The precision at `position`: a positive whole number, at most
121/// [`MAX_PRECISION`], or where omitted the caller's `NUMERIC DIGITS` capped
122/// the same way. 88.905 for anything that is not a positive whole number.
123fn precision(
124    interp: &mut Interp,
125    args: &[Option<ObjRef>],
126    position: usize,
127) -> Result<u64, Failure> {
128    let digits = match args.get(position - 1).copied().flatten() {
129        Some(value) => match super::rexxutil::stems::whole_value(interp, value, 1)? {
130            Some(given) => u64::try_from(given).unwrap_or(MAX_PRECISION),
131            None => {
132                let found = interp.native_found(value);
133                return Err(Raised::native_argument_not_positive(position, &found).into());
134            }
135        },
136        None => interp.activation().settings.digits(),
137    };
138    Ok(digits.min(MAX_PRECISION))
139}

141/// The unit an angle argument or an arc result is in.
142#[derive(Copy, Clone, PartialEq, Eq, Debug)]
143pub(crate) enum Unit {
144    Degrees,
145    Radians,
146    Grades,
147}
148
149impl Unit {
150    /// The angle of a full circle, or `None` for radians, which are never
151    /// reduced.
152    fn circle(self) -> Option<f64> {
153        match self {
154            Unit::Degrees => Some(360.0),
155            Unit::Grades => Some(400.0),
156            Unit::Radians => None,
157        }
158    }
159}

161/// The units argument at `position`: its first character, caselessly, `D`,
162/// `R` or `G`, degrees where omitted. 88.916 for anything else, the null
163/// string included.
164fn unit(interp: &mut Interp, args: &[Option<ObjRef>], position: usize) -> Result<Unit, Failure> {
165    let Some(value) = args.get(position - 1).copied().flatten() else {
166        return Ok(Unit::Degrees);
167    };
168    let Some(converted) = interp.blamed_string_conversion(value)? else {
169        return Err(Raised::native_argument_needs_a_string_value(position).into());
170    };
171    let text = interp.to_text(converted).into_owned();
172    match text.first().map(u8::to_ascii_uppercase) {
173        Some(b'D') => Ok(Unit::Degrees),
174        Some(b'R') => Ok(Unit::Radians),
175        Some(b'G') => Ok(Unit::Grades),
176        _ => {
177            let mut raised = Raised::not_one_of(position, b"D, R, or G", &text);
178            raised.delivery.lineless = true;
179            Err(raised.into())
180        }
181    }
182}

184/// `value` as the routine's answer, at `precision` significant digits.
185fn answer(interp: &mut Interp, value: f64, precision: u64) -> ObjRef {
186    let precision = usize::try_from(precision).unwrap_or(usize::MAX);
187    interp.text(&crate::dispatch::library::double_text(value, precision))
188}

190/// `MathLoadFuncs([version])`: with any argument, the library's banner on
191/// standard output. It registers nothing; loading the library did that.
192pub(crate) fn load_funcs(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<ObjRef, Failure> {
193    check_arity(args, 1)?;
194    if args.first().copied().flatten().is_some() {
195        interp.write_out(
196            concat!(
197                "rxmath (interrexx ",
198                env!("CARGO_PKG_VERSION"),
199                ") - REXX mathematical function library\n\n"
200            )
201            .as_bytes(),
202        );
203    }
204    Ok(interp.text(b""))
205}

276/// The four trigonometric functions of an angle.
277#[derive(Copy, Clone, PartialEq, Eq, Debug)]
278pub(crate) enum Trig {
279    Sin,
280    Cos,
281    Tan,
282    Cotan,
283}
284
285impl Trig {
286    /// The function of `rho`, in radians, with no reduction.
287    fn of(self, rho: f64) -> f64 {
288        match self {
289            Trig::Sin => rho.sin(),
290            Trig::Cos => rho.cos(),
291            Trig::Tan => rho.tan(),
292            Trig::Cotan => 1.0 / rho.tan(),
293        }
294    }
295}
296
297/// `function` of `angle` in `unit`.
298///
299/// A degree or grade angle is reduced first, so that a multiple of a quarter
300/// circle answers exactly: its magnitude modulo the circle picks a quadrant,
301/// the angle's offset from that quadrant's axis is converted to radians in
302/// extended precision, and the function comes from the sine, cosine or
303/// tangent of that offset with the quadrant's sign
304/// (`extensions-behaviour.md`, "Degrees" and "Grades").
305pub(crate) fn trig(function: Trig, angle: f64, unit: Unit) -> f64 {
306    let Some(circle) = unit.circle() else {
307        return function.of(angle);
308    };
309    let s = if angle < 0.0 { -1.0 } else { 1.0 };
310    let quarter = circle / 4.0;
311    let eighth = circle / 8.0;
312    let a = angle.abs() % circle;
313    let radians = |r: f64| extended::to_radians(r, circle / 2.0);
314    if a <= eighth {
315        let rho = radians(a);
316        match function {
317            Trig::Sin => s * rho.sin(),
318            Trig::Cos => rho.cos(),
319            Trig::Tan => s * rho.tan(),
320            Trig::Cotan => s / rho.tan(),
321        }
322    } else if a < quarter + eighth {
323        let rho = radians(quarter - a);
324        match function {
325            Trig::Sin => s * rho.cos(),
326            Trig::Cos => rho.sin(),
327            Trig::Tan => s / rho.tan(),
328            Trig::Cotan => s * rho.tan(),
329        }
330    } else if a <= 2.0 * quarter + eighth {
331        let rho = radians(2.0 * quarter - a);
332        match function {
333            Trig::Sin => s * rho.sin(),
334            Trig::Cos => -rho.cos(),
335            Trig::Tan => -s * rho.tan(),
336            Trig::Cotan => -s / rho.tan(),
337        }
338    } else if a < 3.0 * quarter + eighth {
339        let rho = radians(a - 3.0 * quarter);
340        match function {
341            Trig::Sin => -s * rho.cos(),
342            Trig::Cos => rho.sin(),
343            Trig::Tan => -s / rho.tan(),
344            Trig::Cotan => -s * rho.tan(),
345        }
346    } else {
347        let rho = radians(a - circle);
348        match function {
349            Trig::Sin => s * rho.sin(),
350            Trig::Cos => rho.cos(),
351            Trig::Tan => s * rho.tan(),
352            Trig::Cotan => s / rho.tan(),
353        }
354    }
355}

390/// `v`, an angle in radians, in `unit`: for degrees `v × 180` in double
391/// arithmetic, then divided by π in extended precision; for grades the same
392/// with 200.
393pub(crate) fn from_radians(v: f64, unit: Unit) -> f64 {
394    match unit.circle() {
395        None => v,
396        Some(circle) => extended::divide_by_pi(v * (circle / 2.0)),
397    }
398}
