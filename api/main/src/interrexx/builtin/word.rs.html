28/// The default word count `SUBWORD` and `DELWORD` use when their third
29/// argument is omitted.
30const ALL_REMAINING_WORDS: usize = usize::MAX;

32/// Whether `byte` separates words: blank or horizontal tab, and no other byte
33/// -- see the module doc for the 256-byte measurement.
34fn is_blank(byte: u8) -> bool {
35    byte == b' ' || byte == b'\t'
36}
37
38/// A scan over the words of a byte string.
39struct Words<'a> {
40    text: &'a [u8],
41    /// Where the next scan starts. A pointer and a remaining length would carry
42    /// the same state; one index says the same thing, since the remaining
43    /// length is always `text.len() - next`.
44    next: usize,
45    /// The word [`step`] last found, as a byte range of `text`.
46    word: Range<usize>,
47}
48
49impl<'a> Words<'a> {
50    fn new(text: &'a [u8]) -> Self {
51        Words {
52            text,
53            next: 0,
54            word: 0..0,
55        }
56    }
57
58    /// Advances to the next word, answering whether there was one.
59    fn step(&mut self) -> bool {
60        self.skip_blanks();
61        if self.next == self.text.len() {
62            return false;
63        }
64        let start = self.next;
65        while self.next < self.text.len() && !is_blank(self.text[self.next]) {
66            self.next += 1;
67        }
68        self.word = start..self.next;
69        true
70    }
71
72    /// Steps `count` times, answering whether every step found a word.
73    fn skip(&mut self, count: usize) -> bool {
74        (0..count).all(|_| self.step())
75    }
76
77    /// Advances past the blanks between the current word and the next.
78    fn skip_blanks(&mut self) {
79        while self.next < self.text.len() && is_blank(self.text[self.next]) {
80            self.next += 1;
81        }
82    }
83}

85/// Every word of `text`, in order.
86pub(crate) fn word_slices(text: &[u8]) -> Vec<&[u8]> {
87    let mut scan = Words::new(text);
88    let mut found = Vec::new();
89    while scan.step() {
90        found.push(&text[scan.word.clone()]);
91    }
92    found
93}

95/// `WORDS`'s body: how many words `text` holds.
96pub(crate) fn word_count(text: &[u8]) -> usize {
97    let mut scan = Words::new(text);
98    let mut count = 0usize;
99    while scan.step() {
100        count += 1;
101    }
102    count
103}

105/// `WORD`, `WORDINDEX` and `WORDLENGTH`'s body: where the `position`th
106/// (1-based) word of `text` lies, or `None` past the last.
107pub(crate) fn word_range(text: &[u8], position: usize) -> Option<Range<usize>> {
108    if position == 0 {
109        return Some(0..0);
110    }
111    let start = nth_word_start(text, position)?;
112    Some(start..first_blank_from(text, start))
113}

115/// One in the lowest bit of every byte lane of a `u64`.
116const LANES: u64 = u64::from_le_bytes([1; 8]);
117/// The high bit of every byte lane.
118const HIGH: u64 = LANES << 7;
119
120/// Eight bytes of `text` from `at`, as a little-endian word, with the lanes
121/// past the end of `text` read as blanks -- which neither start a word nor
122/// end one early, since nothing follows them.
123#[inline(always)]
124fn chunk_at(text: &[u8], at: usize) -> u64 {
125    match text.get(at..at + 8) {
126        Some(eight) => u64::from_le_bytes(eight.try_into().expect("eight bytes")),
127        None => {
128            let mut padded = [b' '; 8];
129            let rest = &text[at..];
130            padded[..rest.len()].copy_from_slice(rest);
131            u64::from_le_bytes(padded)
132        }
133    }
134}
135
136/// The high bit of each lane of `chunk` that holds `byte`, exactly: the
137/// carry-free form, so no lane's answer leaks into the next.
138#[inline(always)]
139fn lanes_equal(chunk: u64, byte: u8) -> u64 {
140    let x = chunk ^ (LANES * u64::from(byte));
141    !(((x & !HIGH).wrapping_add(!HIGH)) | x | !HIGH)
142}
143
144/// The high bit of each lane of `chunk` that [`is_blank`] answers true for.
145#[inline(always)]
146fn blank_lanes(chunk: u64) -> u64 {
147    lanes_equal(chunk, b' ') | lanes_equal(chunk, b'\t')
148}
149
150/// How many lanes have their high bit set in `lanes`, which has no other bit
151/// set. A multiply rather than `count_ones`, which is a library loop on the
152/// baseline target.
153#[inline(always)]
154fn count_lanes(lanes: u64) -> usize {
155    usize::try_from((lanes >> 7).wrapping_mul(LANES) >> 56).expect("at most eight lanes")
156}
157
158/// Where the `n`th (1-based, `n >= 1`) word of `text` starts, or `None`
159/// when it has fewer words. Eight bytes at a time: a word starts at a
160/// non-blank byte whose predecessor is blank, the byte before the text
161/// counting as blank, so the starts in a chunk are its non-blank lanes
162/// masked by its blank lanes shifted up one lane.
163fn nth_word_start(text: &[u8], n: usize) -> Option<usize> {
164    debug_assert!(n >= 1, "words are counted from 1");
165    let mut remaining = n;
166    // The high bit of the lowest lane: the byte before this chunk was blank.
167    let mut blank_before = 0x80u64;
168    let mut at = 0;
169    while at < text.len() {
170        let blanks = blank_lanes(chunk_at(text, at));
171        let starts = !blanks & HIGH & ((blanks << 8) | blank_before);
172        let found = count_lanes(starts);
173        if found >= remaining {
174            let mut starts = starts;
175            for _ in 1..remaining {
176                starts &= starts - 1;
177            }
178            return Some(at + usize::cast_from(starts.trailing_zeros() / 8));
179        }
180        remaining -= found;
181        blank_before = blanks >> 56;
182        at += 8;
183    }
184    None
185}

187/// The offset of the first blank of `text` at or after `from`, or its length
188/// when there is none.
189fn first_blank_from(text: &[u8], from: usize) -> usize {
190    let mut at = from;
191    while at < text.len() {
192        let blanks = blank_lanes(chunk_at(text, at));
193        if blanks != 0 {
194            return (at + usize::cast_from(blanks.trailing_zeros() / 8)).min(text.len());
195        }
196        at += 8;
197    }
198    text.len()
199}

201/// `SUBWORD`'s body: the slice of `text` from the `position`th (1-based) word
202/// to the end of the `count`th from there, empty past the last word or for a
203/// `count` of zero; a `count` of `None` reaches the last word.
204pub(crate) fn subword_range(text: &[u8], position: usize, count: Option<usize>) -> Range<usize> {
205    let count = count.unwrap_or(ALL_REMAINING_WORDS);
206    let mut scan = Words::new(text);
207    if count == 0 || !scan.skip(position) {
208        return 0..0;
209    }
210    let start = scan.word.start;
211    // Whether this reaches `count` words or runs out, `scan.word` is the
212    // last word taken either way, which is what ends the slice at a word
213    // rather than at the end of the string.
214    scan.skip(count - 1);
215    start..scan.word.end
216}

218/// `DELWORD`'s body: removes `count` words of `bytes` from the `position`th
219/// (1-based) together with the blanks after the last of them; a `count` of
220/// `None` removes every remaining word.
221pub(crate) fn delword_bytes(bytes: &mut Vec<u8>, position: usize, count: Option<usize>) {
222    let count = count.unwrap_or(ALL_REMAINING_WORDS);
223    let mut scan = Words::new(bytes);
224    if count == 0 || !scan.skip(position) {
225        return;
226    }
227    let front = scan.word.start;
228    // The blanks need only be asked for when the skip reached its count;
229    // unconditionally is the same thing here, because a [`Words::step`] that
230    // fails leaves the scan at the end of the string and there is nothing left
231    // for the skip to consume.
232    scan.skip(count - 1);
233    scan.skip_blanks();
234    let rest = scan.next;
235    bytes.drain(front..rest);
236}

249/// The word search both spellings run, `matches` the only difference.
250fn wordpos_with(
251    phrase: &[u8],
252    string: &[u8],
253    start: usize,
254    matches: impl Fn(&[u8], &[u8]) -> bool,
255) -> usize {
256    let needle = word_slices(phrase);
257    let haystack = word_slices(string);
258    // Both guards earn their place, and for different reasons. An empty phrase
259    // would otherwise match at every position, since the empty slice equals the
260    // empty needle -- the answer is 0. A phrase with more words than the string
261    // is what the subtraction below cannot survive. A `start` past the last
262    // candidate needs no guard of its own: the range is then simply empty.
263    if needle.is_empty() || needle.len() > haystack.len() {
264        0
265    } else {
266        (start..=haystack.len() - needle.len() + 1)
267            .find(|at| {
268                haystack[at - 1..at - 1 + needle.len()]
269                    .iter()
270                    .zip(&needle)
271                    .all(|(found, wanted)| matches(found, wanted))
272            })
273            .unwrap_or(0)
274    }
275}

298/// `WORD(string, n)`: the `n`th word, or the null string if there is no such
299/// word.
300pub(crate) fn word(interp: &mut Interp, name: &[u8], args: Args<'_>) -> Result<ObjRef, Failure> {
301    let position = position_of(name, 2, converted_position(interp, name, args, 2)?)?;
302    let value = arg(args, 1).expect("check_arity admitted the string");
303    let string = required_render(interp, args, 1);
304    let bytes = string.text(interp);
305    // **The same string asked about again is answered from its word table**
306    // ([`WordIndex`]), which is built the second time running it is asked
307    // about.
308    let (found, step) = match interp.word_index.lookup(value, position) {
309        Some(found) => (found, None),
310        None => {
311            let found = word_range(bytes, position).unwrap_or(0..0);
312            let table = interp.word_index.wants(value).then(|| word_table(bytes));
313            (found, Some(table))
314        }
315    };
316    // A word short enough for the handle is answered from the one borrow of
317    // the string; only a longer one reads it again past the buffer loan.
318    let inline = ObjRef::inline_text(&bytes[found.clone()]);
319    match step {
320        Some(Some(table)) => {
321            let text = matches!(
322                interp.heap.get(value).map(|object| &object.body),
323                Some(interrexx_core::Body::Text { .. })
324            );
325            interp.word_index.tabled(value, table.filter(|_| text));
326        }
327        Some(None) => interp.word_index.note(value),
328        None => {}
329    }
330    if let Some(inline) = inline {
331        return Ok(inline);
332    }
333    let mut out = interp.take_result_buffer();
334    out.extend_from_slice(&string.text(interp)[found]);
335    Ok(interp.text_built(out))
336}

338/// The most words [`WordIndex`] tables for one string; a string with more is
339/// scanned on every call, as it always was.
340const WORD_TABLE_LIMIT: usize = 4096;
341
342/// Every word of `text` as a byte range, or `None` past
343/// [`WORD_TABLE_LIMIT`] words.
344fn word_table(text: &[u8]) -> Option<Vec<(u32, u32)>> {
345    let mut scan = Words::new(text);
346    let mut table = Vec::new();
347    while scan.step() {
348        if table.len() == WORD_TABLE_LIMIT {
349            return None;
350        }
351        let (Ok(start), Ok(end)) = (u32::try_from(scan.word.start), u32::try_from(scan.word.end))
352        else {
353            return None;
354        };
355        table.push((start, end));
356    }
357    Some(table)
358}

360/// The word boundaries of one string `WORD` keeps being asked about:
361/// `word(list, k)` over a list held in a variable, for each `k`, scans the
362/// list from its start on every call without one.
363///
364/// **Keyed by the object**, which is a sound key only because a text body's
365/// bytes never change once it is built -- `Interp::concat_onto` grows only a
366/// string its chain's register alone holds, which no call can be passed --
367/// and because the whole index is dropped at every collection, before any
368/// slot it could name is handed out again. **Built on the second call in a
369/// row about the same string**, so that a program asking once about each of
370/// many strings never pays for a table it will not read.
371#[derive(Debug)]
372pub(crate) struct WordIndex {
373    /// The string the last call asked about, or `NIL`.
374    string: ObjRef,
375    table: WordTable,
376}
377
378#[derive(Debug)]
379enum WordTable {
380    /// Not built yet: the string has been asked about once.
381    Unbuilt,
382    /// Every word of the string.
383    Built(Vec<(u32, u32)>),
384    /// Never to be built: too many words, or not a text body.
385    Declined,
386}
387
388impl Default for WordIndex {
389    fn default() -> Self {
390        WordIndex {
391            string: ObjRef::NIL,
392            table: WordTable::Unbuilt,
393        }
394    }
395}
396
397impl WordIndex {
398    /// The `position`th (1-based) word of `string` from its table, where it
399    /// has one.
400    fn lookup(&self, string: ObjRef, position: usize) -> Option<Range<usize>> {
401        if string != self.string {
402            return None;
403        }
404        let WordTable::Built(table) = &self.table else {
405            return None;
406        };
407        Some(match position.checked_sub(1).and_then(|at| table.get(at)) {
408            Some(&(start, end)) => usize::cast_from(start)..usize::cast_from(end),
409            None => 0..0,
410        })
411    }
412
413    /// Whether a table should be built for `string`: it is the string the
414    /// call before asked about, and nothing has been decided for it yet.
415    fn wants(&self, string: ObjRef) -> bool {
416        string == self.string && string != ObjRef::NIL && matches!(self.table, WordTable::Unbuilt)
417    }
418
419    /// The table built for `string`, or `None` where it is declined.
420    fn tabled(&mut self, string: ObjRef, table: Option<Vec<(u32, u32)>>) {
421        self.string = string;
422        self.table = match table {
423            Some(table) => WordTable::Built(table),
424            None => WordTable::Declined,
425        };
426    }
427
428    /// Notes a call about `string` that built nothing.
429    fn note(&mut self, string: ObjRef) {
430        if string != self.string {
431            self.string = string;
432            self.table = WordTable::Unbuilt;
433        }
434    }
435
436    /// Forgets everything, at a collection.
437    pub(crate) fn clear(&mut self) {
438        *self = WordIndex::default();
439    }
440}

442/// `WORDINDEX(string, n)`: the 1-based byte offset the `n`th word starts at,
443/// or 0.
444pub(crate) fn word_index(
445    interp: &mut Interp,
446    name: &[u8],
447    args: Args<'_>,
448) -> Result<ObjRef, Failure> {
449    let position = position_of(name, 2, converted_position(interp, name, args, 2)?)?;
450    let string = required_render(interp, args, 1);
451
452    let index = word_range(string.text(interp), position).map_or(0, |word| word.start + 1);
453    Ok(interp.counted(index))
454}

469/// `SUBWORD(string, n [,length])`: `length` words from the `n`th, as a slice
470/// of the argument.
471pub(crate) fn subword(interp: &mut Interp, name: &[u8], args: Args<'_>) -> Result<ObjRef, Failure> {
472    let n = converted_position(interp, name, args, 2)?;
473    let requested = whole_number(interp, name, args, 3)?;
474    let position = position_of(name, 2, n)?;
475    let count = requested
476        .map(|value| length_of(name, 3, value))
477        .transpose()?;
478
479    let string = required_render(interp, args, 1);
480    let string = string.text(interp);
481    let found = subword_range(string, position, count);
482    let out = super::string::unchanged(interp, &string[found])?;
483    Ok(interp.text_built(out))
484}

486/// `DELWORD(string, n [,length])`: the argument with `length` words from the
487/// `n`th removed.
488pub(crate) fn delword(interp: &mut Interp, name: &[u8], args: Args<'_>) -> Result<ObjRef, Failure> {
489    let mut string = required_string(interp, args, 1);
490    let n = converted_position(interp, name, args, 2)?;
491    let requested = whole_number(interp, name, args, 3)?;
492    let position = position_of(name, 2, n)?;
493    let count = requested
494        .map(|value| length_of(name, 3, value))
495        .transpose()?;
496
497    delword_bytes(&mut string, position, count);
498    Ok(interp.text_built(string))
499}

501/// `WORDPOS(phrase, string [,start])`: which word of `string` begins a run
502/// matching `phrase`'s words, or 0.
503pub(crate) fn word_pos(
504    interp: &mut Interp,
505    name: &[u8],
506    args: Args<'_>,
507) -> Result<ObjRef, Failure> {
508    let start = match whole_number(interp, name, args, 3)? {
509        Some(value) => position_of(name, 3, value)?,
510        None => 1,
511    };
512
513    let found = {
514        let phrase = required_render(interp, args, 1);
515        let string = required_render(interp, args, 2);
516        wordpos_bytes(phrase.text(interp), string.text(interp), start)
517    };
518    Ok(interp.counted(found))
519}

582    /// Runs `name` over `arguments`, each `None` standing for an omitted
583    /// interior position, and answers the result's own bytes.
584    fn call(name: &[u8], arguments: &[Option<&[u8]>]) -> Result<Vec<u8>, Failure> {
585        let mut interp = Interp::new();
586        let args: Vec<_> = arguments
587            .iter()
588            .map(|argument| argument.map(|bytes| interp.text(bytes)))
589            .collect();
590        let result = dispatch(&mut interp, name, &args).expect("a builtin name")?;
591        Ok(interp.to_text(result).into_owned())
592    }

595    fn answer(name: &[u8], arguments: &[&[u8]]) -> Vec<u8> {
596        let arguments: Vec<_> = arguments.iter().map(|bytes| Some(*bytes)).collect();
597        call(name, &arguments).expect("this call succeeds")
598    }

600    /// The `(major, sub)` and substitutions of the condition `name` raises.
601    fn raised(name: &[u8], arguments: &[Option<&[u8]>]) -> (u16, u16, Vec<Vec<u8>>) {
602        let failure = call(name, arguments).expect_err("this call raises");
603        let Failure::Raised(raised) = failure else {
604            panic!("expected Raised, got {failure:?}");
605        };
606        (raised.number, raised.sub, raised.additional)
607    }

1115    /// The arity rows, at both ends and at the interior omission each one
1116    /// admits.
1117    #[test]
1118    fn the_arity_rows_are_the_oracles_own() {
1119        for (name, min, max) in [
1120            (b"DELWORD".as_slice(), 2usize, 3usize),
1121            (b"SUBWORD", 2, 3),
1122            (b"WORD", 2, 2),
1123            (b"WORDINDEX", 2, 2),
1124            (b"WORDLENGTH", 2, 2),
1125            (b"WORDPOS", 2, 3),
1126            (b"WORDS", 1, 1),
1127        ] {
1128            let short: Vec<Option<&[u8]>> = vec![Some(b"a b"); min - 1];
1129            assert_eq!(
1130                raised(name, &short),
1131                (40, 3, vec![name.to_vec(), min.to_string().into_bytes()]),
1132                "{} did not name its minimum",
1133                String::from_utf8_lossy(name)
1134            );
1135            let long: Vec<Option<&[u8]>> = vec![Some(b"1"); max + 1];
1136            assert_eq!(
1137                raised(name, &long),
1138                (40, 4, vec![name.to_vec(), max.to_string().into_bytes()]),
1139                "{} did not name its maximum",
1140                String::from_utf8_lossy(name)
1141            );
1142        }
1143        // An omission interior to the required positions is 40.5, and every
1144        // shape in this family that can reach it does.
1145        assert_eq!(
1146            raised(b"WORD", &[None, Some(b"1")]),
1147            (40, 5, vec![b"WORD".to_vec(), b"1".to_vec()])
1148        );
1149        for name in [b"DELWORD".as_slice(), b"SUBWORD", b"WORDPOS"] {
1150            assert_eq!(
1151                raised(name, &[Some(b"a b"), None, Some(b"1")]),
1152                (40, 5, vec![name.to_vec(), b"2".to_vec()]),
1153                "{} did not name its missing second argument",
1154                String::from_utf8_lossy(name)
1155            );
1156        }
1157    }
