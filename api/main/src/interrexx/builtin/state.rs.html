12/// The environment `ADDRESS()` names before any `ADDRESS` instruction has
13/// run, and after a bare `ADDRESS` swaps back to it.
14#[cfg(unix)]
15pub(crate) const DEFAULT_ENVIRONMENT: &[u8] = b"sh";
16#[cfg(not(unix))]
17pub(crate) const DEFAULT_ENVIRONMENT: &[u8] = b"CMD";

19/// The first byte of an option argument, upcased, or `None` when the
20/// argument was not supplied at all.
21fn option_letter(
22    interp: &mut Interp,
23    name: &[u8],
24    args: Args<'_>,
25    position: usize,
26    valid: &str,
27) -> Result<Option<u8>, Failure> {
28    let Some(text) = optional_string(interp, args, position) else {
29        return Ok(None);
30    };
31    match text.first() {
32        Some(byte) => Ok(Some(byte.to_ascii_uppercase())),
33        None => Err(Raised::argument_not_in_list(name, position, valid, &text).into()),
34    }
35}

37/// `ADDRESS()`: the environment a command clause would be sent to.
38/// ```text
39/// say address()   ->  sh
40/// address envA
41/// address envB
42/// say address()   ->  ENVB
43/// address ; say address()  ->  ENVA
44/// address ; say address()  ->  ENVB
45/// address ; say address()  ->  ENVA
46/// ```
47pub(crate) fn address(
48    interp: &mut Interp,
49    _name: &[u8],
50    _args: Args<'_>,
51) -> Result<ObjRef, Failure> {
52    let current = interp.activation().address.current.clone();
53    let bytes = match &current {
54        Some(name) => &name[..],
55        None => DEFAULT_ENVIRONMENT,
56    };
57    Ok(interp.text(bytes))
58}

85/// `QUEUED()`: how many lines the external data queue holds.
86pub(crate) fn queued(
87    interp: &mut Interp,
88    _name: &[u8],
89    _args: Args<'_>,
90) -> Result<ObjRef, Failure> {
91    // The queue exit's first pass (`BUILTIN(QUEUED)`).
92    if interp.exits.is_some()
93        && let Some(size) = interp.queue_size_exit()?
94    {
95        return Ok(interp.text(size.to_string().as_bytes()));
96    }
97    let Some(queue) = interp.local_route(b"STDQUE")? else {
98        let count = interp.queue.len();
99        return Ok(interp.text(count.to_string().as_bytes()));
100    };
101    let caller = interp.caller();
102    let answer = pinned!(
103        interp,
104        crate::pinning::PinKind::PullWrapper,
105        interp.send_message(queue, b"QUEUED", None, &[], caller)
106    )?;
107    Ok(answer.unwrap_or_else(|| interp.text(b"0")))
108}

113/// `GC()` / `GC('Force')`: whether a collection was run.
114pub(crate) fn gc(interp: &mut Interp, name: &[u8], args: Args<'_>) -> Result<ObjRef, Failure> {
115    let Some(text) = optional_string(interp, args, 1) else {
116        return Ok(crate::eval::logical(false));
117    };
118    if !matches!(text.first(), Some(b'f' | b'F')) {
119        return Err(Raised::argument_not_in_list(name, 1, GC_OPTIONS, &text).into());
120    }
121    // Really collect, the way a collect-and-uninit does, rather than report `.true`
122    // for something that did not happen. Safe at exactly the point every
123    // allocation is: this crate's own collect-stress mode
124    // (`Interp::alloc_with`) runs a collection immediately before every
125    // allocation, so a caller's values are already rooted by the time any
126    // builtin can be entered.
127    interp.collect_now();
128    // That is collect *and* run, and `GC('force')` is the one path that runs
129    // the finalizers inline rather than leaving them to a later safe point. For
130    // example, `o = .K~new` / `say 'b'` / `drop o` / `call gc 'force'` / `say
131    // 'c'` answers `a b uninit c`, five runs of five, where 200,000 unforced
132    // allocations in place of the `GC('force')` defer the same finalizer to
133    // termination.
134    if let Some(loud) = interp.run_ready_uninits().into_iter().next() {
135        return Err(Failure::Loud(Box::new(loud)));
136    }
137    Ok(crate::eval::logical(true))
138}

140/// `ERRORTEXT(n)`: the message for the error number or error code `n`, or the
141/// null string when the catalogue has no entry (`r[bif.errortext.n]`).
142///
143/// A whole number is an error number, 0-99. Anything else holding a period is
144/// an error code: the part before the first period is the error number and
145/// the part after it the subcode, 0-999, each a whole number as written, so
146/// `40.1` and `40.001` are both subcode 1 and `40.10` is subcode 10. A
147/// subcode's message keeps its `&n` places.
148pub(crate) fn errortext(
149    interp: &mut Interp,
150    name: &[u8],
151    args: Args<'_>,
152) -> Result<ObjRef, Failure> {
153    let (major, minor) = match whole_number(interp, name, args, 1) {
154        Ok(number) => {
155            let number = number.expect("check_arity admitted argument 1");
156            if !(0..=99).contains(&number) {
157                return Err(
158                    Raised::argument_out_of_range(name, 1, converted(number).as_bytes()).into(),
159                );
160            }
161            (number, 0)
162        }
163        Err(Failure::Raised(raised)) if (raised.number, raised.sub) == (40, 12) => {
164            let value = super::arg(args, 1).expect("check_arity admitted argument 1");
165            let text = interp.to_text(value).into_owned();
166            let Some(dot) = text.iter().position(|&byte| byte == b'.') else {
167                return Err(Failure::Raised(raised));
168            };
169            let part = |bytes: &[u8]| {
170                interrexx_num::Number::parse_bytes(bytes)
171                    .and_then(|number| number.whole_value(super::ARGUMENT_DIGITS))
172            };
173            match (part(&text[..dot]), part(&text[dot + 1..])) {
174                (Some(major @ 0..=99), Some(minor @ 0..=999)) => (major, minor),
175                _ => return Err(Raised::argument_not_an_error_code(name, 1, &text).into()),
176            }
177        }
178        Err(other) => return Err(other),
179    };
180    let text = u16::try_from(major)
181        .ok()
182        .zip(u16::try_from(minor).ok())
183        .and_then(|(major, minor)| interrexx_inventory::errors::lookup(major, minor))
184        .map(|message| message.text)
185        .unwrap_or_default();
186    Ok(interp.text(text.as_bytes()))
187}

189/// `SOURCELINE()`: the program's line count, or `SOURCELINE(n)`: line `n`.
190pub(crate) fn sourceline(
191    interp: &mut Interp,
192    name: &[u8],
193    args: Args<'_>,
194) -> Result<ObjRef, Failure> {
195    let source = &interp.activation().program.source;
196    // A restored program shows no source: measured, `SOURCELINE()` answers
197    // 0 and `SOURCELINE(1)` raises 40.34 naming 0 lines.
198    let lines = match source.is_traceable() {
199        true => source.line_count(),
200        false => 0,
201    };
202    let Some(number) = whole_number(interp, name, args, 1)? else {
203        return Ok(interp.text(lines.to_string().as_bytes()));
204    };
205    if number <= 0 {
206        return Err(Raised::argument_not_positive(name, 1, converted(number).as_bytes()).into());
207    }
208    let requested = usize::try_from(number).unwrap_or(usize::MAX);
209    if requested > lines {
210        return Err(Raised::sourceline_out_of_range(converted(number).as_bytes(), lines).into());
211    }
212    let line = interp
213        .activation()
214        .program
215        .source
216        .line(requested)
217        .expect("a line number within the count")
218        .to_vec();
219    Ok(interp.text_built(line))
220}

222/// `TRACE()` / `TRACE(setting)`: the setting in force, and optionally a new
223/// one.
224/// ```text
225/// trace value 5     24.901  a numeric request outside interactive debugging.
226/// say trace('5')    24.1    names the valid letters and the value 5.
227/// ```
228pub(crate) fn trace(interp: &mut Interp, _name: &[u8], args: Args<'_>) -> Result<ObjRef, Failure> {
229    let before = interp.trace_mode();
230    // **The builtin has no debug guard where the instruction does.**
231    // Measured, `call trace 'O'` from a program under `?R` ends debug and
232    // stops the tracing that a `trace off` *instruction* there would not.
233    if let Some(text) = optional_string(interp, args, 1) {
234        let request = crate::trace::parse_trace_request(&text)
235            .map_err(crate::trace::raised_invalid_trace_letter)?;
236        let merged = crate::trace::applied(before, &request);
237        interp.set_trace_mode(merged);
238        // A call with an argument ends the stop a typed line runs in
239        // (`r[env.debugging.input.trace]`); one without ends nothing.
240        interp.end_debug_stop();
241        if !before.debug && merged.debug {
242            // The clause that called the function is followed by a stop at
243            // once. The hot clause exit asks `clause_exits`, so raising it
244            // sends this clause to the exit that takes the stop.
245            interp.activity.debug_stop_owed = Some(interp.activation().id);
246            interp.clause_exits = true;
247        }
248    }
249    let mut answer = Vec::with_capacity(2);
250    if before.debug {
251        answer.push(b'?');
252    }
253    answer.push(before.letter);
254    Ok(interp.text_built(answer))
255}

257/// `ARG()`, `ARG(n)` and `ARG(n, option)`: the current routine's own
258/// argument list.
259/// ```text
260/// arg(2)        ''      an omitted position reads as the null string
261/// arg(2,'E')    0       ... and 'E'xists says so
262/// arg(2,'O')    1       'O'mitted is its complement
263/// arg(4,'E')    0       past the end is neither existing ...
264/// arg(4,'O')    1       ... nor supplied
265/// arg(2,'N')    ''      'N'ormal is `arg(n)` again
266/// ```
267/// ```text
268/// arg(1,'A')    ~size 3, ~items 2, ~dimension 1, ~hasIndex(2) 0
269/// arg(2,'A')    ~size 2, ~items 1, ~dimension 1
270/// arg(4,'A')    ~size 0,           ~dimension 0
271/// ```
272/// ```text
273/// arg('x','q')   40.12  argument 1 must be a whole number
274/// arg(,'')       40.5   argument 1 is required     (an empty option is still an option)
275/// arg(0,'')      40.14  argument 1 must be positive
276/// arg(1,'')      40.904 names argument 2, the valid letters AENO and the empty value
277/// ```
278pub(crate) fn arg(interp: &mut Interp, name: &[u8], args: Args<'_>) -> Result<ObjRef, Failure> {
279    const VALID: &str = "AENO";
280    let position = whole_number(interp, name, args, 1)?;
281    // Read but not validated here: `optional_string` is all that is done with
282    // it until the switch below.
283    let option = optional_string(interp, args, 2);
284    let Some(position) = position else {
285        // No position at all: any option, empty or not, is 40.5, and
286        // otherwise this is the count.
287        if option.is_some() {
288            return Err(Raised::missing_argument(name, 1).into());
289        }
290        let count = interp.activity.call_context.arguments.len();
291        return Ok(interp.text(count.to_string().as_bytes()));
292    };
293    if position <= 0 {
294        return Err(Raised::argument_not_positive(name, 1, converted(position).as_bytes()).into());
295    }
296    let index = usize::try_from(position).unwrap_or(usize::MAX);
297    let supplied = interp
298        .activity
299        .call_context
300        .arguments
301        .get(index - 1)
302        .cloned()
303        .flatten();
304    let letter = option
305        .as_ref()
306        .map(|text| text.first().copied().unwrap_or(0).to_ascii_uppercase());
307    match letter {
308        None | Some(b'N') => Ok(match supplied {
309            Some(argument) => argument,
310            None => interp.text(b""),
311        }),
312        Some(b'E') => Ok(interp.text(if supplied.is_some() { b"1" } else { b"0" })),
313        Some(b'O') => Ok(interp.text(if supplied.is_some() { b"0" } else { b"1" })),
314        Some(b'A') => {
315            let arguments = &interp.activity.call_context.arguments;
316            let slots: Vec<Option<ObjRef>> = if index == 1 {
317                arguments.to_vec()
318            } else if index > arguments.len() {
319                Vec::new()
320            } else {
321                arguments[index - 1..].to_vec()
322            };
323            // `new_array()` leaves the shape unset, while array construction
324            // with a count of zero fixes it -- and position 1 takes the second
325            // even when there are no arguments at all, because that is tested
326            // before the past-the-end arm.
327            let dimensions: Option<Box<[usize]>> =
328                (index == 1 && slots.is_empty()).then(|| Box::from([0].as_slice()));
329            let object = interp.alloc_charged(
330                BehaviourId::ARRAY,
331                Body::Array {
332                    slots: slots.into(),
333                    dimensions,
334                },
335            );
336            // The array's only root: nothing else names it between here and
337            // the caller storing it.
338            interp.roots.activity_mut().push_temp(object);
339            Ok(object)
340        }
341        Some(_) => {
342            let found = option.unwrap_or_default();
343            Err(Raised::argument_not_in_list(name, 2, VALID, &found).into())
344        }
345    }
346}

349pub(crate) const CONDITION_OPTIONS: &str = "ACDEIORS";

351/// `CONDITION()` and `CONDITION(option)`: what the handler now running was
352/// entered for.
353/// ```text
354///          SYNTAX (1/0)   NOVALUE      USER via SIGNAL   USER via CALL   no condition
355///  bare    SIGNAL         SIGNAL       SIGNAL            CALL            ''
356///  A       <Array 0>      .NIL         the ADDITIONAL    same            .NIL
357///  C       SYNTAX         NOVALUE      USER UC           same            ''
358///  D       ''             ZUNSETVAR    the DESCRIPTION   same            ''
359///  E       3              ''           ''                same            ''
360///  I       SIGNAL         SIGNAL       SIGNAL            CALL            ''
361///  O       <Directory 14> <Dir 9>      <Dir 10>          same            .NIL
362///  S       OFF            OFF          OFF               DELAY           ''
363/// ```
364/// ```text
365/// before rearm  I SIGNAL  S OFF
366/// signal on syntax name sh1
367/// after  rearm  I SIGNAL  S ON
368/// signal off syntax
369/// after  off    I SIGNAL  S OFF
370/// ```
371pub(crate) fn condition(
372    interp: &mut Interp,
373    name: &[u8],
374    args: Args<'_>,
375) -> Result<ObjRef, Failure> {
376    let style = option_letter(interp, name, args, 1, CONDITION_OPTIONS)?.unwrap_or(b'I');
377    // Resolved before the option is answered, because two of the options
378    // have a distinct answer when there is no condition at all and the rest
379    // share one.
380    let active: Option<TrappedCondition> = interp.activation().condition().cloned();
381    match (style, &active) {
382        // `.NIL` with nothing active, and not the null string the letters
383        // below answer -- measured, `say condition('O')` and `condition('A')`
384        // in a program that has trapped nothing both print `The NIL object`
385        // where `condition('C')` prints nothing at all.
386        (b'A', active) => {
387            let additional = active
388                .as_ref()
389                .and_then(|condition| condition.object)
390                .and_then(|object| interp.condition_entry(object, b"ADDITIONAL"));
391            Ok(additional.unwrap_or(ObjRef::NIL))
392        }
393        (b'O', active) => match active.as_ref().and_then(|condition| condition.object) {
394            Some(object) => interp.condition_copy(object),
395            None => Ok(ObjRef::NIL),
396        },
397        (b'R', _) => {
398            interp.activation_mut().replace_condition(None);
399            Ok(interp.text(b""))
400        }
401        (b'C' | b'D' | b'E' | b'I' | b'S', None) => Ok(interp.text(b"")),
402        // The condition object's entry as it stands now: a handler that holds
403        // the object may have changed it. An entry removed answers the null
404        // string; any other value is answered as it is.
405        (
406            b'C' | b'D' | b'I',
407            Some(TrappedCondition {
408                object: Some(object),
409                ..
410            }),
411        ) => {
412            let entry: &[u8] = match style {
413                b'C' => b"CONDITION",
414                b'D' => b"DESCRIPTION",
415                _ => b"INSTRUCTION",
416            };
417            match interp.condition_entry_held(*object, entry) {
418                Some(value) => Ok(value),
419                None => Ok(interp.text(b"")),
420            }
421        }
422        (b'C', Some(condition)) => {
423            let name = condition.name.to_vec();
424            Ok(interp.text_built(name))
425        }
426        (b'D', Some(condition)) => match &condition.description {
427            Some(description) => {
428                let description = description.clone();
429                Ok(interp.text_built(description))
430            }
431            None => Ok(interp.text(b"")),
432        },
433        (b'E', Some(condition)) => {
434            if condition.name.as_ref() == b"SYNTAX"
435                && let Some(object) = condition.object
436                && let Some(code) = interp.condition_entry_held(object, b"CODE")
437            {
438                let code = interp.unsent_string_value_text(code);
439                if let Some(dot) = code.iter().position(|&byte| byte == b'.') {
440                    return Ok(interp.text(&code[dot + 1..]));
441                }
442            }
443            Ok(interp.text(b""))
444        }
445        (b'I', Some(condition)) => {
446            let text: &[u8] = if condition.call { b"CALL" } else { b"SIGNAL" };
447            Ok(interp.text(text))
448        }
449        (b'S', Some(condition)) => {
450            // Live, not stored: the trap table is read by the condition's *own*
451            // name, so an `ANY` trap that caught a `SYNTAX` condition reports
452            // `OFF` even before it is removed.
453            let text: &[u8] = match interp.activation().traps.get(&condition.name) {
454                None => b"OFF",
455                Some(trap) if trap.delayed => b"DELAY",
456                Some(_) => b"ON",
457            };
458            Ok(interp.text(text))
459        }
460        _ => {
461            let found = optional_string(interp, args, 1).unwrap_or_default();
462            Err(Raised::argument_not_in_list(name, 1, CONDITION_OPTIONS, &found).into())
463        }
464    }
465}

467/// A converted argument as a 40.x range message spells it in `found "..."`:
468/// **the integer the conversion produced, never the value's own rendering.**
469/// ```text
470/// arg(0.0)                                found "0"          not "0.0"
471/// arg('+0')                               found "0"          not "+0"
472/// errortext(1e2)                          found "100"        not "1E2"
473/// errortext(' 100 ')                      found "100"        not " 100 "
474/// sourceline(1e1)                         ("10")             not ("1E1")
475/// numeric digits 3; errortext(999999+1)   found "1000000"    not "1.00E+6"
476/// ```
477fn converted(value: i64) -> String {
478    value.to_string()
479}

488    /// Runs `source` as a whole program and hands back the stdout it
489    /// produced, having first insisted the run ended cleanly.
490    fn output(source: &[u8]) -> String {
491        let outcome = run_program("/t.rex", source.to_vec(), Invocation::none());
492        assert_eq!(
493            outcome.exit_code,
494            0,
495            "stderr: {}",
496            String::from_utf8_lossy(&outcome.stderr)
497        );
498        String::from_utf8(outcome.stdout).expect("ASCII output")
499    }

501    /// Dispatches `name` with no arguments and renders the answer.
502    fn answer(interp: &mut Interp, name: &[u8]) -> Vec<u8> {
503        let result = dispatch(interp, name, &[])
504            .expect("a builtin name")
505            .expect("the call succeeds");
506        interp.to_text(result).into_owned()
507    }

509    /// Runs `source` expecting it to fail, and hands back the exit code and
510    /// the whole of stderr.
511    fn failure(source: &[u8]) -> (i32, String) {
512        let outcome = run_program("/t.rex", source.to_vec(), Invocation::none());
513        (
514            outcome.exit_code,
515            String::from_utf8(outcome.stderr).expect("ASCII stderr"),
516        )
517    }

519    fn raised(name: &[u8], arguments: &[&[u8]]) -> (u16, u16, Vec<Vec<u8>>) {
520        let mut interp = Interp::new();
521        let values: Vec<Option<interrexx_core::ObjRef>> = arguments
522            .iter()
523            .map(|bytes| Some(interp.text(bytes)))
524            .collect();
525        let failure = dispatch(&mut interp, name, &values)
526            .expect("a builtin name")
527            .expect_err("the call fails");
528        let Failure::Raised(condition) = failure else {
529            panic!("expected Raised, got {failure:?}");
530        };
531        (condition.number, condition.sub, condition.additional)
532    }
