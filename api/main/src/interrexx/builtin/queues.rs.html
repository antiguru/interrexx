1//! `RXQUEUE`: the current queue's name, and named queues through `.RexxQueue`
2//! (the `RXQUEUE` builtin).
3
4use interrexx_core::ObjRef;
5
6use super::datatype::{SymbolKind, classify};
7use super::{Args, arg, required_string};
8use crate::Interp;
9use crate::error::{Failure, Raised};

11/// The options `RXQUEUE` names in its 40.904.
12pub(crate) const OPTIONS: &str = "CDEGOS";
13
14/// `RXQUEUE(option [, name])`.
15pub(crate) fn rxqueue(interp: &mut Interp, name: &[u8], args: Args<'_>) -> Result<ObjRef, Failure> {
16    let option = required_string(interp, args, 1);
17    let queue_name = match arg(args, 2) {
18        Some(value) => Some(interp.string_value_text(value)?),
19        None => None,
20    };
21    let letter = option.first().map(u8::to_ascii_uppercase);
22    let required = |queue_name: Option<Vec<u8>>| match queue_name {
23        Some(queue_name) => {
24            if matches!(classify(&queue_name), SymbolKind::Bad) {
25                return Err(Failure::from(Raised::argument_not_a_symbol(
26                    name,
27                    2,
28                    &queue_name,
29                )));
30            }
31            Ok(queue_name)
32        }
33        None => Err(Raised::not_enough_arguments(name, 2).into()),
34    };
35    let (target, message, argument) = match letter {
36        Some(b'G') => {
37            if queue_name.is_some() {
38                return Err(Raised::too_many_arguments(name, 1).into());
39            }
40            (stdque(interp)?, b"GET".as_slice(), None)
41        }
42        Some(b'C') => {
43            let argument = match queue_name {
44                Some(_) => Some(required(queue_name)?),
45                None => None,
46            };
47            (queue_class(interp), b"CREATE".as_slice(), argument)
48        }
49        Some(b'S') => {
50            // The queue exit's pass at the new name, before it is checked (`BUILTIN(RXQUEUE)`).
51            let queue_name = match queue_name {
52                Some(given) if interp.exits.is_some() => {
53                    Some(interp.queue_name_exit(&given)?.unwrap_or(given))
54                }
55                other => other,
56            };
57            let argument = required(queue_name)?;
58            (stdque(interp)?, b"SET".as_slice(), Some(argument))
59        }
60        Some(b'O') => (
61            queue_class(interp),
62            b"OPEN".as_slice(),
63            Some(required(queue_name)?),
64        ),
65        Some(b'E') => (
66            queue_class(interp),
67            b"EXISTS".as_slice(),
68            Some(required(queue_name)?),
69        ),
70        Some(b'D') => (
71            queue_class(interp),
72            b"DELETE".as_slice(),
73            Some(required(queue_name)?),
74        ),
75        _ => return Err(Raised::argument_not_in_list(name, 1, OPTIONS, &option).into()),
76    };
77    let frame = interp.roots.activity_mut().push_frame();
78    let argument = argument.map(|text| {
79        let value = interp.text(&text);
80        interp.roots.activity_mut().push_temp(value);
81        value
82    });
83    let caller = interp.caller();
84    let arguments: &[Option<ObjRef>] = match argument {
85        Some(_) => &[argument],
86        None => &[],
87    };
88    let sent = pinned!(
89        interp,
90        crate::pinning::PinKind::PullWrapper,
91        interp.send_message(target, message, None, arguments, caller)
92    );
93    interp.roots.activity_mut().pop_frame(frame);
94    Ok(sent?.unwrap_or_else(|| interp.text(b"")))
95}

97/// `.local`'s `STDQUE`.
98fn stdque(interp: &mut Interp) -> Result<ObjRef, Failure> {
99    Ok(interp.local_route(b"STDQUE")?.unwrap_or(ObjRef::NIL))
100}

102/// The `REXX` package's `RexxQueue`.
103fn queue_class(interp: &mut Interp) -> ObjRef {
104    interp
105        .rexx_package_class(b"REXXQUEUE")
106        .unwrap_or(ObjRef::NIL)
107}
