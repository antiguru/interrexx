1//! Host lookups for `SockGetHostByName` and `SockGetHostByAddr`, as glibc's
2//! legacy `gethostbyname` and `gethostbyaddr` answer them for `AF_INET`
3//! (`extensions-behaviour.md`, "Host stems"). Those calls are not reachable
4//! without `unsafe`, so the two sources the host's `hosts: files dns` names
5//! are read here in their order: `/etc/hosts`, parsed as glibc's files
6//! backend parses it, with `multi` from `/etc/host.conf`, and then the
7//! resolver through `getaddrinfo` and `getnameinfo` (`dns-lookup`), which
8//! give the canonical name and the addresses but no aliases.
9
10#[cfg(unix)]
11use libc::{AI_CANONNAME, NI_NAMEREQD};
12#[cfg(unix)]
13use std::net::Ipv6Addr;
14use std::net::{IpAddr, Ipv4Addr, SocketAddr, SocketAddrV4};
15#[cfg(windows)]
16const AI_CANONNAME: i32 = windows_sys::Win32::Networking::WinSock::AI_CANONNAME.cast_signed();
17#[cfg(windows)]
18const NI_NAMEREQD: i32 = windows_sys::Win32::Networking::WinSock::NI_NAMEREQD.cast_signed();
19
20use super::inet_aton;
21
22/// What a lookup found.
23pub(super) struct Host {
24    pub(super) name: Vec<u8>,
25    pub(super) aliases: Vec<Vec<u8>>,
26    pub(super) addresses: Vec<Ipv4Addr>,
27}

29/// One line of the hosts file: its address, as an `AF_INET` lookup reads
30/// it, and its names, the official one first.
31#[cfg(unix)]
32struct Entry {
33    address: Ipv4Addr,
34    names: Vec<Vec<u8>>,
35}
36
37/// An address field as glibc's files backend reads it for `AF_INET`: a
38/// dotted quad, or an IPv6 address that is IPv4-mapped (its last four bytes)
39/// or the loopback `::1` (127.0.0.1). `None` skips the line.
40#[cfg(unix)]
41fn entry_address(text: &str) -> Option<Ipv4Addr> {
42    if let Ok(v4) = text.parse::<Ipv4Addr>() {
43        return Some(v4);
44    }
45    let v6 = text.parse::<Ipv6Addr>().ok()?;
46    if let Some(v4) = v6.to_ipv4_mapped() {
47        return Some(v4);
48    }
49    (v6 == Ipv6Addr::LOCALHOST).then_some(Ipv4Addr::LOCALHOST)
50}
51
52/// The hosts file's entries, in file order.
53#[cfg(unix)]
54fn entries() -> Vec<Entry> {
55    let Ok(text) = std::fs::read("/etc/hosts") else {
56        return Vec::new();
57    };
58    text.split(|byte| *byte == b'\n')
59        .filter_map(|line| {
60            let line = match line.iter().position(|byte| *byte == b'#') {
61                Some(comment) => &line[..comment],
62                None => line,
63            };
64            let mut fields = line
65                .split(u8::is_ascii_whitespace)
66                .filter(|field| !field.is_empty());
67            let address = std::str::from_utf8(fields.next()?).ok()?;
68            let address = entry_address(address)?;
69            let names: Vec<Vec<u8>> = fields.map(<[u8]>::to_vec).collect();
70            (!names.is_empty()).then_some(Entry { address, names })
71        })
72        .collect()
73}

75/// Whether `/etc/host.conf` (or `RESOLV_MULTI`) turns on `multi`, which
76/// makes a name lookup gather every matching hosts-file line.
77#[cfg(unix)]
78fn multi() -> bool {
79    if let Some(value) = std::env::var_os("RESOLV_MULTI") {
80        return value.eq_ignore_ascii_case("on");
81    }
82    let Ok(text) = std::fs::read_to_string("/etc/host.conf") else {
83        return false;
84    };
85    let mut on = false;
86    for line in text.lines() {
87        let mut words = line.split_whitespace();
88        if words
89            .next()
90            .is_some_and(|word| word.eq_ignore_ascii_case("multi"))
91        {
92            on = words
93                .next()
94                .is_some_and(|word| word.eq_ignore_ascii_case("on"));
95        }
96    }
97    on
98}

100/// `gethostbyname(name)` for `AF_INET`, `None` where it finds nothing.
101pub(super) fn by_name(name: &[u8]) -> Option<Host> {
102    // glibc's `__nss_hostname_digits_dots`: a name of digits and dots, not
103    // ending in a dot, is an address or nothing.
104    if name.first().is_some_and(u8::is_ascii_digit)
105        && name
106            .iter()
107            .all(|byte| byte.is_ascii_digit() || *byte == b'.')
108        && name.last() != Some(&b'.')
109    {
110        return inet_aton(name).map(|address| Host {
111            name: name.to_vec(),
112            aliases: Vec::new(),
113            addresses: vec![address],
114        });
115    }
116    #[cfg(unix)]
117    if let Some(found) = by_name_in_files(name) {
118        return Some(found);
119    }
120    // A name `inet_aton` reads that the digits-and-dots test let through
121    // (`0x7f.1`) is a name to the resolver, which `getaddrinfo` would read as
122    // an address instead.
123    if name.is_empty() || inet_aton(name).is_some() {
124        return None;
125    }
126    by_name_in_dns(name)
127}
128
129#[cfg(unix)]
130fn by_name_in_files(name: &[u8]) -> Option<Host> {
131    let matches = |entry: &Entry| {
132        entry
133            .names
134            .iter()
135            .any(|known| known.eq_ignore_ascii_case(name))
136    };
137    let mut found = entries().into_iter().filter(matches);
138    let first = found.next()?;
139    let mut names = first.names.into_iter();
140    let mut host = Host {
141        name: names.next().unwrap_or_default(),
142        aliases: names.collect(),
143        addresses: vec![first.address],
144    };
145    if multi() {
146        for entry in found {
147            let mut names = entry.names.into_iter();
148            if let Some(official) = names.next()
149                && official != host.name
150            {
151                host.aliases.push(official);
152            }
153            host.aliases.extend(names);
154            host.addresses.push(entry.address);
155        }
156    }
157    Some(host)
158}

160fn by_name_in_dns(name: &[u8]) -> Option<Host> {
161    let name = std::str::from_utf8(name).ok()?;
162    let hints = dns_lookup::AddrInfoHints {
163        flags: AI_CANONNAME,
164        address: dns_lookup::AddrFamily::Inet.into(),
165        socktype: dns_lookup::SockType::Stream.into(),
166        protocol: 0,
167    };
168    let answers = dns_lookup::getaddrinfo(Some(name), None, Some(hints)).ok()?;
169    let mut canonical = None;
170    let mut addresses = Vec::new();
171    for info in answers.flatten() {
172        if canonical.is_none() {
173            canonical = info.canonname.clone();
174        }
175        if let SocketAddr::V4(v4) = info.sockaddr {
176            addresses.push(*v4.ip());
177        }
178    }
179    if addresses.is_empty() {
180        return None;
181    }
182    Some(Host {
183        name: canonical.unwrap_or_else(|| name.to_owned()).into_bytes(),
184        aliases: Vec::new(),
185        addresses,
186    })
187}

189/// `gethostbyaddr(address, 4, AF_INET)`, `None` where it finds nothing.
190pub(super) fn by_addr(address: Ipv4Addr) -> Option<Host> {
191    #[cfg(unix)]
192    if let Some(entry) = entries().into_iter().find(|entry| entry.address == address) {
193        let mut names = entry.names.into_iter();
194        return Some(Host {
195            name: names.next().unwrap_or_default(),
196            aliases: names.collect(),
197            addresses: vec![address],
198        });
199    }
200    let socket = SocketAddr::V4(SocketAddrV4::new(address, 0));
201    let (name, _) = dns_lookup::getnameinfo(&socket, NI_NAMEREQD).ok()?;
202    if name.parse::<IpAddr>().is_ok() {
203        return None;
204    }
205    Some(Host {
206        name: name.into_bytes(),
207        aliases: Vec::new(),
208        addresses: vec![address],
209    })
