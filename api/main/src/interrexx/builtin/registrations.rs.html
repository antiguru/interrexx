1//! `RXFUNCADD`, `RXFUNCQUERY` and `RXFUNCDROP`: adding, querying and dropping a
2//! registered routine.
3//!
4//! A function a package library exports, or an internal package's, is callable without the
5//! registry. Any other library is a classic one: its function is registered with the service,
6//! and resolves to its `RexxFunctionHandler`.
7
8use interrexx_core::ObjRef;
9use interrexx_rxapi::message::code::OK;
10use interrexx_rxapi::message::{Registry, Reply, Request};

13use crate::Interp;

17use crate::internal_routines::{self, Package};
18use crate::rxapi_client;

20fn boolean(interp: &mut Interp, failed: bool) -> ObjRef {
21    interp.text(if failed { b"1" } else { b"0" })
22}
23
24impl Interp {
25    /// Whether a function `upper` names is loaded in this interpreter: an internal package's or
26    /// a loaded library's.
27    fn function_loaded(&self, upper: &[u8]) -> bool {
28        internal_routines::lookup(upper).is_some_and(|row| row.body.is_some())
29            || self.package_routine(upper).is_some()
30    }
31}

33/// `RXFUNCADD(name, module [, procedure])`: 0 where the function is callable, 1 otherwise.
34pub(crate) fn rxfuncadd(
35    interp: &mut Interp,
36    _name: &[u8],
37    args: Args<'_>,
38) -> Result<ObjRef, Failure> {
39    let name = required_string(interp, args, 1);
40    let upper = name.to_ascii_uppercase();
41    let module = required_string(interp, args, 2);
42    // r[native.loading.classic]: an omitted procedure is the name as it is
43    // spelled, which the lookup tries before its capitals.
44    let procedure = match arg(args, 3) {
45        Some(value) => interp.to_text(value).into_owned(),
46        None => name,
47    };
48    if interp.function_loaded(&upper) || interp.classic_functions.contains_key(&upper) {
49        return Ok(boolean(interp, false));
50    }
51    if let Some(package) = Package::named(&module) {
52        let found = internal_routines::package_routine(package, &upper).is_some();
53        return Ok(boolean(interp, !found));
54    }
55    match interp.resolve_library(&module) {
56        LibraryLoad::Loaded(_) => {
57            let loaded = interp.function_loaded(&upper);
58            return Ok(boolean(interp, !loaded));
59        }
60        LibraryLoad::Raised(failure) => return Err(failure),
61        // r[native.loading.version]: a first load of a library that needs a
62        // newer interpreter is refused here as everywhere else.
63        LibraryLoad::Version => return Err(Raised::library_version(&module).into()),
64        LibraryLoad::Missing => {}
65    }
66    let made = register_function(&upper, &module, &procedure);
67    let resolved = interp.resolve_classic_function(&upper);
68    if !resolved && made {
69        drop_function(&upper);
70    }
71    interp.routines_changed();
72    Ok(boolean(interp, !resolved))
73}

75/// Registers `name` as `library`'s `procedure` with the service, and answers
76/// whether this call made the registration, rather than finding one there.
77pub(crate) fn register_function(name: &[u8], library: &[u8], procedure: &[u8]) -> bool {
78    let Ok(client) = rxapi_client::client() else {
79        return false;
80    };
81    let reply = client.call(Request::Register {
82        registry: Registry::Function,
83        name: name.to_ascii_uppercase(),
84        library: library.to_vec(),
85        procedure: procedure.to_vec(),
86        user_area: None,
87        owner_only: true,
88    });
89    matches!(
90        reply,
91        Ok(Reply::Status(OK) | Reply::Registration { status: OK, .. })
92    )
93}

95/// Drops the registration of `name` that a call just made, when the function
96/// it names cannot be resolved: a failed registration leaves nothing
97/// registered (`r[bif.rxfuncadd.unknown-library]`).
98pub(crate) fn drop_function(name: &[u8]) {
99    let _ = registry_status(Request::Deregister {
100        registry: Registry::Function,
101        name: name.to_ascii_uppercase(),
102        library: Vec::new(),
103    });
104}

106/// `RXFUNCQUERY(name)`: 0 where the function is loaded here or registered, 1 otherwise.
107pub(crate) fn rxfuncquery(
108    interp: &mut Interp,
109    _name: &[u8],
110    args: Args<'_>,
111) -> Result<ObjRef, Failure> {
112    let upper = required_string(interp, args, 1).to_ascii_uppercase();
113    if interp.function_loaded(&upper)
114        || interp.classic_functions.contains_key(&upper)
115        || capi::registered_entry(Registry::Function, &upper).is_some()
116    {
117        return Ok(boolean(interp, false));
118    }
119    let status = registry_status(Request::QueryRegistration {
120        registry: Registry::Function,
121        name: upper,
122        library: Vec::new(),
123    });
124    Ok(boolean(interp, status != OK))
125}

127/// `RXFUNCDROP(name)`: 0 where the registration was removed, 1 otherwise.
128pub(crate) fn rxfuncdrop(
129    interp: &mut Interp,
130    _name: &[u8],
131    args: Args<'_>,
132) -> Result<ObjRef, Failure> {
133    let upper = required_string(interp, args, 1).to_ascii_uppercase();
134    interp.classic_functions.remove(&upper);
135    interp.routines_changed();
136    if capi::registered_entry(Registry::Function, &upper).is_some() {
137        let status = capi::deregister(Registry::Function, &upper, None);
138        return Ok(boolean(interp, status != OK));
139    }
140    let status = registry_status(Request::Deregister {
141        registry: Registry::Function,
142        name: upper,
143        library: Vec::new(),
144    });
145    Ok(boolean(interp, status != OK))
146}

148/// `request`'s status from a running service. With none, nothing is registered.
149fn registry_status(request: Request) -> u64 {
150    let Some(client) = rxapi_client::running_client() else {
151        return interrexx_rxapi::message::code::NOT_REGISTERED;
152    };
153    match client.call(request) {
154        Ok(Reply::Status(status) | Reply::Registration { status, .. }) => status,
155        Ok(_) => OK,
156        Err(status) => status,
157    }
158}
