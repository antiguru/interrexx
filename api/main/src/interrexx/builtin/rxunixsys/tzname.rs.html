1//! `SysGettzname1` and `SysGettzname2`: glibc's `tzname[0]` and `tzname[1]`
2//! after `tzset`, derived from the program's own `TZ` as glibc derives them.
3//!
4//! * `TZ` unset reads `/etc/localtime`; an empty `TZ` is the name
5//!   `Universal`; a leading `:` is dropped. A name not starting with `/` is
6//!   looked up under `TZDIR`, or `/usr/share/zoneinfo` where that is unset or
7//!   empty.
8//! * From a TZif file: walking the transitions from the last back, the first
9//!   standard type's designation and the first daylight type's; with no
10//!   transitions the first type's, for both. For example, `Asia/Tokyo` is `JST
11//!   JDT` and `Europe/Dublin` `IST GMT`.
12//! * Otherwise the text is a POSIX rule: `XYZ3` is `XYZ XYZ`, `XYZ3ABC` is
13//!   `XYZ ABC`, `XYZ` (no offset) is `XYZ` and the null string, and a name of
14//!   fewer than three letters that is not `<...>` is two null strings.
15//!
16//! Where `localtime` has been called since `TZ` last changed (by `TIME` or
17//! `DATE`, or a `SysStat` time), glibc's `tzname` holds the names of the rule
18//! in force then instead, and `tzset` with the same `TZ` does not reset it:
19//! `Asia/Tokyo` answers `JST JST` after `time()`. This answers as a program
20//! that has not.
21
22/// `(tzname[0], tzname[1])` for `tz`, the program's `TZ` (`None` where unset),
23/// with `tzdir` its `TZDIR`.
24pub(super) fn names(tz: Option<&[u8]>, tzdir: Option<&[u8]>) -> (Vec<u8>, Vec<u8>) {
25    let name: &[u8] = match tz {
26        None => b"/etc/localtime",
27        Some(b"") => b"Universal",
28        Some(text) => text.strip_prefix(b":").unwrap_or(text),
29    };
30    if let Some(names) = file_names(name, tzdir) {
31        return names;
32    }
33    rule_names(name)
34}

36/// The names a TZif file at `name` gives, or `None` where no file there
37/// reads as one.
38fn file_names(name: &[u8], tzdir: Option<&[u8]>) -> Option<(Vec<u8>, Vec<u8>)> {
39    if name.is_empty() {
40        return None;
41    }
42    let mut path = Vec::new();
43    if name.first() != Some(&b'/') {
44        let directory = match tzdir {
45            Some(dir) if !dir.is_empty() => dir,
46            _ => b"/usr/share/zoneinfo",
47        };
48        path.extend_from_slice(directory);
49        path.push(b'/');
50    }
51    path.extend_from_slice(name);
52    let bytes = std::fs::read(crate::paths::os_path(&path)).ok()?;
53    let zone = tz::TimeZone::from_tz_data(&bytes).ok()?;
54    let zone = zone.as_ref();
55    let types = zone.local_time_types();
56    let designation = |index: usize| {
57        types
58            .get(index)
59            .map(|kind| kind.time_zone_designation().as_bytes().to_vec())
60    };
61    let mut found: [Option<Vec<u8>>; 2] = [None, None];
62    for transition in zone.transitions().iter().rev() {
63        let index = transition.local_time_type_index();
64        let dst = usize::from(types.get(index).is_some_and(tz::LocalTimeType::is_dst));
65        if found[dst].is_none() {
66            found[dst] = designation(index);
67            if found[1 - dst].is_some() {
68                break;
69            }
70        }
71    }
72    let [standard, daylight] = found;
73    let standard = standard.or_else(|| designation(0)).unwrap_or_default();
74    let daylight = daylight.unwrap_or_else(|| standard.clone());
75    Some((standard, daylight))
76}

78/// The names glibc's `__tzset_parse_tz` leaves for `text`.
79fn rule_names(text: &[u8]) -> (Vec<u8>, Vec<u8>) {
80    let mut rest = text;
81    let Some(standard) = zone_name(&mut rest) else {
82        return (Vec::new(), Vec::new());
83    };
84    if !offset(&mut rest, true) {
85        return (standard, Vec::new());
86    }
87    if rest.is_empty() {
88        return (standard.clone(), standard);
89    }
90    let daylight = zone_name(&mut rest).unwrap_or_default();
91    (standard, daylight)
92}

94/// `parse_tzname`: three or more ASCII letters, or `<` three or more
95/// letters, digits, `+` and `-`, then `>`.
96fn zone_name(rest: &mut &[u8]) -> Option<Vec<u8>> {
97    let letters = rest
98        .iter()
99        .take_while(|byte| byte.is_ascii_alphabetic())
100        .count();
101    if letters >= 3 {
102        let name = rest[..letters].to_vec();
103        *rest = &rest[letters..];
104        return Some(name);
105    }
106    let quoted = rest.strip_prefix(b"<")?;
107    let length = quoted
108        .iter()
109        .take_while(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-'))
110        .count();
111    if quoted.get(length) != Some(&b'>') || length < 3 {
112        return None;
113    }
114    let name = quoted[..length].to_vec();
115    *rest = &quoted[length + 1..];
116    Some(name)
117}

119/// `parse_offset` for the standard rule: an optional sign and `hh[:mm[:ss]]`.
120/// `false` where no offset is there.
121fn offset(rest: &mut &[u8], standard: bool) -> bool {
122    let mut text = *rest;
123    if standard
124        && !matches!(text.first(), Some(byte) if *byte == b'+' || *byte == b'-' || byte.is_ascii_digit())
125    {
126        return false;
127    }
128    if matches!(text.first(), Some(b'+' | b'-')) {
129        text = &text[1..];
130    }
131    let mut consumed = 0;
132    for field in 0..3 {
133        let start = consumed + usize::from(field > 0);
134        if field > 0 && text.get(consumed) != Some(&b':') {
135            break;
136        }
137        let digits = text[start.min(text.len())..]
138            .iter()
139            .take_while(|byte| byte.is_ascii_digit())
140            .count();
141        if digits == 0 {
142            break;
143        }
144        consumed = start + digits;
145    }
146    if consumed == 0 {
147        return false;
148    }
149    *rest = &text[consumed..];
150    true
151}

153#[cfg(test)]
154mod tests {
155    use super::*;
156
157    fn rule(text: &str) -> (String, String) {
158        let (standard, daylight) = rule_names(text.as_bytes());
159        (
160            String::from_utf8(standard).unwrap(),
161            String::from_utf8(daylight).unwrap(),
162        )
163    }
164
165    /// For example, each as `TZ`, none of them a file under
166    /// `/usr/share/zoneinfo`.
167    #[test]
168    fn a_posix_rule_names_as_glibc_parses_it() {
169        let pair = |a: &str, b: &str| (a.to_owned(), b.to_owned());
170        assert_eq!(rule("XYZ3"), pair("XYZ", "XYZ"));
171        assert_eq!(rule("XYZ3ABC"), pair("XYZ", "ABC"));
172        assert_eq!(rule("XYZ3ABC,M3.2.0,M11.1.0"), pair("XYZ", "ABC"));
173        assert_eq!(rule("<+03>-3"), pair("+03", "+03"));
174        assert_eq!(rule("AB3"), pair("", ""));
175        assert_eq!(rule("XYZ"), pair("XYZ", ""));
176        assert_eq!(rule("XYZ+3:30"), pair("XYZ", "XYZ"));
177        assert_eq!(rule("XYZ-a"), pair("XYZ", ""));
178        assert_eq!(rule("Universal"), pair("Universal", ""));
179        assert_eq!(rule("Nowhere/X"), pair("Nowhere", ""));
180    }
