14/// `QUALIFY(name)`: the name resolved against the interpreter's current
15/// directory and reduced to one spelling, whether or not it exists (stream-name
16/// qualification).
17pub(crate) fn qualify(
18    interp: &mut Interp,
19    _name: &'static [u8],
20    args: Args<'_>,
21) -> Result<ObjRef, Failure> {
22    let text = required_string(interp, args, 1);
23    let qualified = qualify_name(interp, &text);
24    Ok(interp.text_built(qualified))
25}

27/// Stream-name qualification: `name` up to its first NUL, as the C string, with
28/// a leading `~` or `~user` resolved, made absolute against the interpreter's
29/// current directory and reduced by [`crate::paths::normalize_bytes`]. The null
30/// string where qualification fails: an empty name, a `~user` naming no account,
31/// a `~` with `HOME` unset, or a home directory that is not absolute.
32pub(crate) fn qualify_name(interp: &Interp, name: &[u8]) -> Vec<u8> {
33    let name = match name.iter().position(|byte| *byte == 0) {
34        Some(end) => &name[..end],
35        None => name,
36    };
37    if name.is_empty() {
38        return Vec::new();
39    }
40    if name.first() == Some(&b'~') {
41        return match expand_tilde(interp, name) {
42            Some(expanded) if crate::paths::os_path(&expanded).is_absolute() => {
43                crate::paths::normalize_bytes(&expanded, b"/")
44            }
45            _ => Vec::new(),
46        };
47    }
48    crate::paths::normalize_bytes(name, interp.cwd_bytes())
49}

51/// A leading `~` replaced by the environment's
52/// `HOME`, a leading `~user` by that account's home directory, and `None`
53/// where no account has that name or `HOME` is unset: an unset home is
54/// unresolved, as an unknown account's is.
55pub(crate) fn expand_tilde(interp: &Interp, name: &[u8]) -> Option<Vec<u8>> {
56    let rest = &name[1..];
57    if rest.is_empty() || rest[0] == b'/' {
58        let mut out = interp
59            .env_get(if cfg!(windows) {
60                b"USERPROFILE"
61            } else {
62                b"HOME"
63            })?
64            .to_vec();
65        out.extend_from_slice(rest);
66        return Some(out);
67    }
68    let slash = rest
69        .iter()
70        .position(|byte| *byte == b'/')
71        .unwrap_or(rest.len());
72    let mut out = passwd_home(&rest[..slash])?;
73    out.extend_from_slice(&rest[slash..]);
74    Some(out)
75}

77/// The home directory the system's user database gives the account `user`,
78/// found as `getpwnam` finds it, so an account that only an NSS source
79/// (LDAP, sssd, systemd-homed) holds is found too.
80#[cfg(unix)]
81pub(crate) fn passwd_home(user: &[u8]) -> Option<Vec<u8>> {
82    home_through(user, |name| {
83        nix::unistd::User::from_name(name)
84            .ok()
85            .flatten()
86            .map(|account| account.dir)
87    })
88}

90/// [`passwd_home`] over `lookup`, which answers the home directory of the
91/// account it is given the name of. A name that is not UTF-8 is no account's:
92/// `lookup` takes text, as the user database's own names are.
93#[cfg(unix)]
94fn home_through(
95    user: &[u8],
96    lookup: impl FnOnce(&str) -> Option<std::path::PathBuf>,
97) -> Option<Vec<u8>> {
98    use std::os::unix::ffi::OsStringExt as _;
99    let name = std::str::from_utf8(user).ok()?;
100    lookup(name).map(|home| home.into_os_string().into_vec())
101}

103/// `USERID()`: the effective user's login name, as the system's user database
104/// gives it for the effective uid, so an account that only an NSS source (LDAP,
105/// sssd, systemd-userdb) holds has its name too. An account the database does
106/// not know answers the environment's own `USER` or `LOGNAME` instead; the
107/// empty string is answered there.
108pub(crate) fn userid(
109    interp: &mut Interp,
110    _name: &'static [u8],
111    _args: Args<'_>,
112) -> Result<ObjRef, Failure> {
113    #[cfg(unix)]
114    if let Some(name) = name_through(nix::unistd::geteuid().as_raw(), |uid| {
115        nix::unistd::User::from_uid(nix::unistd::Uid::from_raw(uid))
116            .ok()
117            .flatten()
118            .map(|account| account.name)
119    }) {
120        return Ok(interp.text_built(name));
121    }
122    for fallback in [
123        b"USER".as_slice(),
124        b"LOGNAME".as_slice(),
125        #[cfg(windows)]
126        b"USERNAME".as_slice(),
127    ] {
128        if let Some(value) = interp.env_get(fallback) {
129            let value = value.to_vec();
130            return Ok(interp.text_built(value));
131        }
132    }
133    Ok(interp.text(b""))
134}

142/// `SETLOCAL()`: saves the directory and the environment, answering `.true`.
143pub(crate) fn setlocal(
144    interp: &mut Interp,
145    _name: &'static [u8],
146    _args: Args<'_>,
147) -> Result<ObjRef, Failure> {
148    let saved = interp.push_local_environment();
149    Ok(crate::eval::logical(saved))
150}

152/// `ENDLOCAL()`: restores the innermost saved pair, answering `.true` when there was
153/// one and `.false` when there was not, whether or not a `SETLOCAL` ran before
154/// (`r[bif.endlocal.none]`).
155pub(crate) fn endlocal(
156    interp: &mut Interp,
157    _name: &'static [u8],
158    _args: Args<'_>,
159) -> Result<ObjRef, Failure> {
160    let restored = interp.pop_local_environment();
161    Ok(crate::eval::logical(restored))
162}

164/// `DIRECTORY([new])`: a native routine of the `REXX` package, not a builtin
165/// which is why its arity errors are the
166/// 88.9xx family. With an argument it moves the interpreter's own current
167/// directory and answers the new one; a change it cannot make answers the null
168/// string and moves nothing. With none it answers where it is.
169pub(crate) fn directory(
170    interp: &mut Interp,
171    _name: &'static [u8],
172    args: &[Option<ObjRef>],
173) -> Result<ObjRef, Failure> {
174    if args.len() > 1 {
175        return Err(Raised::too_many_internal_arguments(1).into());
176    }
177    let Some(text) = rexxutil::optional_text(interp, args, 1)? else {
178        let here = interp.cwd_bytes().to_vec();
179        return Ok(interp.text_built(here));
180    };
181    if text.is_empty() {
182        return Ok(interp.text(b""));
183    }
184    let qualified = qualify_name(interp, &text);
185    let candidate = crate::paths::os_path(&qualified);
186    if !std::fs::metadata(candidate).is_ok_and(|meta| meta.is_dir()) || !can_enter(candidate) {
187        return Ok(interp.text(b""));
188    }
189    // `getcwd` answers the symlink-resolved path, so the answer is what the
190    // filesystem calls the directory rather than the way it was named.
191    let resolved = std::fs::canonicalize(candidate).unwrap_or_else(|_| candidate.to_path_buf());
192    let answer = resolved.as_os_str().as_encoded_bytes().to_vec();
193    interp.set_cwd(resolved);
194    Ok(interp.text_built(answer))
195}

197/// Whether the process may make `directory` its current one, which is what
198/// `chdir` asks: search permission, with the effective ids.
199#[cfg(unix)]
200fn can_enter(directory: &std::path::Path) -> bool {
201    rustix::fs::accessat(
202        rustix::fs::CWD,
203        directory,
204        rustix::fs::Access::EXEC_OK,
205        rustix::fs::AtFlags::EACCESS,
206    )
207    .is_ok()
208}

210/// The `FILESPEC` options, in the order the message lists them: `DELNP`.
211const FILESPEC_OPTIONS: &[u8] = b"DELNP";
212
213/// `FILESPEC(option, name)`: the piece of `name` that `option`'s first letter
214/// selects. Purely textual -- the file need
215/// not exist, and nothing here touches the file system. On unix the drive is
216/// always the null string.
217pub(crate) fn filespec(
218    interp: &mut Interp,
219    _name: &'static [u8],
220    args: &[Option<ObjRef>],
221) -> Result<ObjRef, Failure> {
222    if args.len() > 2 {
223        return Err(Raised::too_many_internal_arguments(2).into());
224    }
225    // Both are C strings, ending at a NUL, and the option is read for its
226    // first character only.
227    let option = rexxutil::required_text(interp, args, 1)?;
228    let spec = rexxutil::required_c_string(interp, args, 2)?;
229    let option = &option[..option.first().map_or(0, |first| usize::from(*first != 0))];
230    let letter = option.first().map(u8::to_ascii_uppercase);
231    let Some(letter) = letter.filter(|byte| FILESPEC_OPTIONS.contains(byte)) else {
232        return Err(Raised::argument_not_in_list(
233            b"FILESPEC",
234            1,
235            &String::from_utf8_lossy(FILESPEC_OPTIONS),
236            option,
237        )
238        .into());
239    };
240    let cut = spec.iter().rposition(|byte| *byte == b'/');
241    let answer: Vec<u8> = match letter {
242        // Unix has no drive letter, and the answer is the null string.
243        b'D' => Vec::new(),
244        b'L' | b'P' => match cut {
245            None => Vec::new(),
246            Some(at) => spec[..=at].to_vec(),
247        },
248        b'N' => match cut {
249            None => spec.clone(),
250            Some(at) => spec[at + 1..].to_vec(),
251        },
252        // What follows the last dot of the *name*, wherever that dot sits:
253        // measured, `filespec('E','/a/.hidden')` is `hidden`, not the null
254        // string. `.File~extension` has the leading-dot rule this does not.
255        b'E' => {
256            let name = match cut {
257                None => &spec[..],
258                Some(at) => &spec[at + 1..],
259            };
260            match name.iter().rposition(|byte| *byte == b'.') {
261                None => Vec::new(),
262                Some(at) => name[at + 1..].to_vec(),
263            }
264        }
265        other => unreachable!("{other} is not one of the options just checked"),
266    };
267    Ok(interp.text_built(answer))
268}

270/// `BEEP`'s bounds.
271const MIN_FREQUENCY: i64 = 37;
272const MAX_FREQUENCY: i64 = 32767;
273const MIN_DURATION: i64 = 0;
274const MAX_DURATION: i64 = 60000;
275
276/// `BEEP(frequency, duration)`: writes one bell byte and answers the null
277/// string. On this platform that reaches the terminal as a single `0x07` on
278/// standard output: `beep(440,10)` writes it and answers `''`.
279pub(crate) fn beep(
280    interp: &mut Interp,
281    _name: &'static [u8],
282    args: &[Option<ObjRef>],
283) -> Result<ObjRef, Failure> {
284    if args.len() > 2 {
285        return Err(Raised::too_many_internal_arguments(2).into());
286    }
287    // Finish checking the frequency before reading the duration.
288    check_beep_argument(interp, args, 1, "frequency", MIN_FREQUENCY, MAX_FREQUENCY)?;
289    check_beep_argument(interp, args, 2, "duration", MIN_DURATION, MAX_DURATION)?;
290    interp.write_out(&[0x07]);
291    Ok(interp.text(b""))
292}

294/// Checks the native whole number and uses the original argument in a range error.
295fn check_beep_argument(
296    interp: &mut Interp,
297    args: &[Option<ObjRef>],
298    position: usize,
299    name: &str,
300    minimum: i64,
301    maximum: i64,
302) -> Result<(), Failure> {
303    let value = rexxutil::required(args, position)?;
304    match rexxutil::stems::whole_value(interp, value, -MAX_WHOLENUMBER)? {
305        Some(number) if (minimum..=maximum).contains(&number) => Ok(()),
306        Some(_) => {
307            let found = interp.string_value_text(value)?;
308            Err(Raised::native_argument_out_of_range(name, minimum, maximum, &found).into())
309        }
310        None => {
311            let found = interp.string_value_text(value)?;
312            Err(rexxutil::marshalled(Raised::syntax(
313                88,
314                903,
315                vec![position.to_string().into_bytes(), found],
316            )))
317        }
318    }
319}

321/// `VALUE(name, [new], selector)`: the two external pools this crate answers.
322/// The empty selector names `.environment` itself; `ENVIRONMENT`, caselessly,
323/// names the process environment this interpreter holds. Any other selector is
324/// 40.914, which is also what is answered on unix for every selector the value
325/// function declines.
326pub(crate) fn value_selector(
327    interp: &mut Interp,
328    args: Args<'_>,
329    selector: &[u8],
330) -> Result<ObjRef, Failure> {
331    let name = required_string(interp, args, 1);
332    let new = arg(args, 2);
333    if selector.is_empty() {
334        return environment_directory(interp, &name, new);
335    }
336    if selector.eq_ignore_ascii_case(b"ENVIRONMENT") {
337        return process_environment(interp, &name, new);
338    }
339    // The value exit's pass at a selector nothing here knows (`BUILTIN(VALUE)`).
340    if interp.exits.is_some()
341        && let (Some(selector), Some(variable)) = (arg(args, 3), arg(args, 1))
342        && let Some(value) = interp.value_exit(selector, variable, new)?
343    {
344        return Ok(value.unwrap_or(ObjRef::NIL));
345    }
346    Err(Raised::syntax(40, 914, vec![selector.to_vec()]).into())
347}

349/// The `''` selector: `.environment`'s own entry, under the upper-cased name
350/// the directory stores. A name it does not hold reads as `.NAME` -- the same
351/// answer `.NAME` itself gives, because that is the path it takes.
352fn environment_directory(
353    interp: &mut Interp,
354    name: &[u8],
355    new: Option<ObjRef>,
356) -> Result<ObjRef, Failure> {
357    let upper = name.to_ascii_uppercase();
358    let mut dotted = Vec::with_capacity(upper.len() + 1);
359    dotted.push(b'.');
360    dotted.extend_from_slice(&upper);
361    let old = interp.dot_variable(&dotted)?;
362    // The write allocates, and `old` may be a fresh `.NAME` string nothing
363    // else holds.
364    interp.roots.activity_mut().push_temp(old);
365    if let Some(value) = new {
366        interp.set_directory_entry(EnvScope::Environment, &upper, value)?;
367    }
368    Ok(old)
369}

371/// The `ENVIRONMENT` selector: the interpreter's own environment. A name it
372/// does not hold reads as the null string; `.nil` removes; any other value is
373/// stored as its string. The name and the value are both cut at the first NUL,
374/// because `setenv` takes C strings and the read-back stops there.
375fn process_environment(
376    interp: &mut Interp,
377    name: &[u8],
378    new: Option<ObjRef>,
379) -> Result<ObjRef, Failure> {
380    let name = name.split(|byte| *byte == 0).next().unwrap_or_default();
381    let old = interp.env_get(name).unwrap_or_default().to_vec();
382    if let Some(value) = new {
383        if value == ObjRef::NIL {
384            interp.env_set(name, None);
385        } else {
386            let text = interp.to_text(value).into_owned();
387            let stored = match text.iter().position(|byte| *byte == 0) {
388                Some(at) => text[..at].to_vec(),
389                None => text,
390            };
391            interp.env_set(name, Some(stored));
392        }
393    }
394    Ok(interp.text_built(old))
395}

397#[cfg(test)]
398mod tests {
399    use std::path::PathBuf;
400
401    use super::{home_through, name_through, passwd_home};

403    /// **A home directory comes from the user database lookup, not from
404    /// `/etc/passwd`**: an account only an NSS source holds, which no file of
405    /// this host lists, is found through the lookup it is given, and an account
406    /// the lookup does not know is not found. The adjacent success is the
407    /// real lookup, which finds `root` where `/etc/passwd` lists it.
408    #[test]
409    fn a_home_directory_comes_from_the_user_database_lookup() {
410        let ldap_only = |name: &str| (name == "ldapuser").then(|| PathBuf::from("/home/ldap/u"));
411        assert_eq!(
412            home_through(b"ldapuser", ldap_only),
413            Some(b"/home/ldap/u".to_vec())
414        );
415        assert_eq!(home_through(b"nobodyhere", ldap_only), None);
416        assert_eq!(home_through(b"ldap\xffuser", ldap_only), None);
417        let passwd = std::fs::read_to_string("/etc/passwd").expect("the passwd file");
418        let listed = passwd
419            .lines()
420            .find_map(|line| line.strip_prefix("root:"))
421            .and_then(|rest| rest.split(':').nth(4))
422            .map(|home| home.as_bytes().to_vec());
423        assert_eq!(passwd_home(b"root"), listed);
424        assert_eq!(passwd_home(b"no-such-account-zq"), None);
425    }
426
427    /// **A login name comes from the user database lookup, not from
428    /// `/etc/passwd`**: an account only an NSS source holds, which no file of
429    /// this host lists, is named through the lookup it is given, and a uid the
430    /// lookup does not know has no name. The adjacent success is the running
431    /// user, whom `USERID()` names as the real lookup does.
432    #[test]
433    fn a_login_name_comes_from_the_user_database_lookup() {
434        let ldap_only = |uid: u32| (uid == 424_242).then(|| "ldapuser".to_string());
435        assert_eq!(name_through(424_242, ldap_only), Some(b"ldapuser".to_vec()));
436        assert_eq!(name_through(7, ldap_only), None);
437        let running = nix::unistd::User::from_uid(nix::unistd::geteuid())
438            .expect("the lookup runs")
439            .map(|account| account.name);
440        let source = b"say userid()\n";
441        let outcome = crate::run_program("/t.rex", source.to_vec(), crate::Invocation::none());
442        assert_eq!(outcome.exit_code, 0);
443        let said = String::from_utf8_lossy(&outcome.stdout).into_owned();
444        match running {
445            Some(name) => assert_eq!(said, format!("{name}\n")),
446            None => assert_eq!(said.matches('\n').count(), 1),
447        }
448    }
449
450    /// **A `~` or `~/x` with `HOME` unset qualifies to the null string**: the
451    /// home is unresolved, as an unknown account's is (SF #2182, r13616;
452    /// `corpus/oracle-crashes.txt` entry 48, dropped). The adjacent success is
453    /// the same names with `HOME` set.
454    #[test]
455    fn a_tilde_with_home_unset_qualifies_without_a_home() {
456        let source = b"call value 'HOME', '/h', 'ENVIRONMENT'\n\
457            say qualify('~') qualify('~/x')\n\
458            'unset HOME'\n\
459            say '['qualify('~')']['qualify('~/x')']'\n";
460        let outcome = crate::run_program("/t.rex", source.to_vec(), crate::Invocation::none());
461        assert_eq!(outcome.exit_code, 0);
462        assert_eq!(String::from_utf8_lossy(&outcome.stdout), "/h /h/x\n[][]\n");
463    }
464}
