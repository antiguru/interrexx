1//! Where an `rxsock` call's wait for its sockets runs while its activity is
2//! parked: the platform-neutral face of it.
3//!
4//! A wait is a list of descriptors with what each is waited for, an
5//! optional deadline, and a way to post its end to the interpreter that
6//! owns it ([`OnEnd`]). [`register`] hands it to the process's one reactor
7//! thread, which runs it with every other interpreter's waits; the answer is
8//! a [`Wait`] whose [`Wait::cancel`] (or drop) ends it early. Each
9//! registration's end is posted **exactly once**: when one of its
10//! descriptors answers it, when its deadline passes, or when it is
11//! cancelled.
12//!
13//! Behind it:
14//!
15//! - Linux: an `epoll` reactor ([`epoll`]), started on the first
16//!   registration.
17//! - Elsewhere: no reactor yet ([`reactor_available`] answers `false`), and
18//!   the caller runs the wait on a pool thread of its own, polling the
19//!   sockets with a [`Canceller`]'s descriptor that ends it early. `kqueue`
20//!   and Windows reactors go in beside `epoll`.
21

27/// on the reactor's thread, so it must not block.
28pub(crate) type OnEnd = Box<dyn FnOnce() + Send>;

30/// Whether this platform has a reactor that [`register`] can use.
31pub(crate) fn reactor_available() -> bool {
32    imp::available()
33}

35/// Starts the reactor where this platform has one, when the `rxsock`
36/// library is loaded rather than at the first wait: its descriptors are
37/// then open before a program's first look at its own, and a count of open
38/// descriptors taken around a socket's use does not see them appear.
39pub(crate) fn prepare() {
40    imp::available();

43/// Hands a wait for `polls` until `deadline` to the reactor, which calls
44/// `on_end` once it is over. Answers `on_end` back where there is no reactor
45/// to take it, or it cannot be started, so the caller can wait another way.
46pub(crate) fn register(
47    polls: &[PollRequest],
48    deadline: Option<Instant>,
49    on_end: OnEnd,
50) -> Result<Wait, OnEnd> {
51    imp::register(polls, deadline, on_end).map(Wait)
52}

71/// A descriptor a waiting thread polls beside its sockets, and the means to
72/// make it readable: an `eventfd` where there is one, and a pipe elsewhere.
73pub(crate) struct Canceller {
74    #[cfg(any(target_os = "linux", target_os = "android"))]
75    event: std::os::fd::OwnedFd,
76    #[cfg(all(unix, not(any(target_os = "linux", target_os = "android"))))]
77    pipe: (std::os::fd::OwnedFd, std::os::fd::OwnedFd),
78    #[cfg(windows)]
79    sockets: (std::net::UdpSocket, std::net::UdpSocket),
80}

82impl Canceller {
83    /// A new canceller, or `None` where no descriptor can be had.
84    pub(crate) fn new() -> Option<Canceller> {
85        #[cfg(any(target_os = "linux", target_os = "android"))]
86        {
87            let event = rustix::event::eventfd(
88                0,
89                rustix::event::EventfdFlags::CLOEXEC | rustix::event::EventfdFlags::NONBLOCK,
90            )
91            .ok()?;
92            Some(Canceller { event })
93        }
94        #[cfg(all(unix, not(any(target_os = "linux", target_os = "android"))))]
95        {
96            let (read, write) = crate::host::pipe().ok()?;
97            let (read, write): (std::os::fd::OwnedFd, std::os::fd::OwnedFd) =
98                (read.into(), write.into());
99            rustix::fs::fcntl_setfl(&write, rustix::fs::OFlags::NONBLOCK).ok()?;
100            Some(Canceller {
101                pipe: (read, write),
102            })
103        }
104        #[cfg(windows)]
105        {
106            use std::net::{Ipv4Addr, UdpSocket};
107            let read = UdpSocket::bind((Ipv4Addr::LOCALHOST, 0)).ok()?;
108            let write = UdpSocket::bind((Ipv4Addr::LOCALHOST, 0)).ok()?;
109            read.connect(write.local_addr().ok()?).ok()?;
110            write.connect(read.local_addr().ok()?).ok()?;
111            write.set_nonblocking(true).ok()?;
112            Some(Canceller {
113                sockets: (read, write),
114            })
115        }
116    }

120        use rustix::fd::AsFd as _;

124        }

135    /// Makes [`Canceller::polled`] readable. A full counter or pipe is
136    /// already readable, so a refused write changes nothing.
137    pub(crate) fn signal(&self) {
138        #[cfg(any(target_os = "linux", target_os = "android"))]
139        let _ = rustix::io::write(&self.event, &1u64.to_ne_bytes());
140        #[cfg(all(unix, not(any(target_os = "linux", target_os = "android"))))]
141        let _ = rustix::io::write(&self.pipe.1, &[1u8]);
142        #[cfg(windows)]
143        let _ = self.sockets.1.send(&[1]);
144    }

179/// The `epoll` reactor: one thread per process, started on the first
180/// registration and never ended, serving every interpreter's waits.
181///
182/// **One `epoll` registration per descriptor**, with the union of what its
183/// waiters wait for, level-triggered: an `epoll` instance takes a
184/// descriptor once (`EEXIST` on a second add), and two activities can wait
185/// on one socket. The interest is modified as waiters come and go and the
186/// descriptor deleted when none is left. A descriptor closed while
187/// registered leaves the interest list silently and its number can be
188/// reused, so a modify's `ENOENT` adds it again and an add's `EEXIST`
189/// modifies it.
190///
191/// An event ends each waiter whose interest it answers, by
192/// [`super::rxsock::answers`]'s rule. `epoll` reports hang-up and error
193/// whatever was asked, and those answer no wait for priority data alone, so
194/// a descriptor whose event answers none of its waiters would end every
195/// `epoll_wait` at once. Such a descriptor is **muted** -- deleted from the
196/// interest list -- and added again [`MUTED_RECHECK`] later, since its
197/// socket may yet be connected.
198///
199/// Registrations and cancellations reach the thread through a command
200/// queue and an `eventfd` that wakes it; its `epoll_wait` timeout is the
201/// earliest deadline or muted descriptor's recheck.
202#[cfg(any(target_os = "linux", target_os = "android"))]
203mod epoll {
204    use super::{OnEnd, PollRequest};
205    use crate::builtin::rxsock::answers;
206    use crate::builtin::rxsock_sys::PollFlags;
207    use crate::builtin::rxsock_sys::with_socket;
208    use interrexx_rxapi::reserved::{Reserved, reserve};
209    use rustc_hash::FxHashMap;
210    use rustix::event::epoll::{self, EventData, EventFlags};
211    use std::collections::BTreeSet;
212    use std::os::fd::OwnedFd;
213    use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
214    use std::sync::{Arc, Mutex, OnceLock};
215    use std::time::{Duration, Instant};

217    /// How long a muted descriptor stays out of the interest list.
218    const MUTED_RECHECK: Duration = Duration::from_millis(50);
219
220    /// The data the wake descriptor is registered under: no descriptor
221    /// number reaches it.
222    const WAKE: u64 = u64::MAX;
223
224    /// The most events one `epoll_wait` takes.
225    const EVENTS: usize = 256;

227    /// The process's reactor, or `None` where it could not be started.
228    static REACTOR: OnceLock<Option<Reactor>> = OnceLock::new();

230    /// What the registering side holds of the reactor.
231    struct Reactor {
232        commands: Mutex<Vec<Command>>,
233        wake: Arc<Reserved<OwnedFd>>,
234        next_id: AtomicU64,
235        /// Set when the thread has stopped on an error it cannot recover
236        /// from, after it has ended every wait it held.
237        stopped: AtomicBool,
238    }
239
240    /// The command queue, whatever a panicking holder left in it: a
241    /// command is whole once pushed.
242    fn lock(commands: &Mutex<Vec<Command>>) -> std::sync::MutexGuard<'_, Vec<Command>> {
243        commands
244            .lock()
245            .unwrap_or_else(std::sync::PoisonError::into_inner)
246    }

248    enum Command {
249        Register {
250            id: u64,
251            polls: Vec<PollRequest>,
252            deadline: Option<Instant>,
253            on_end: OnEnd,
254        },
255        Cancel(u64),
256    }

269    impl Reactor {
270        fn send(&self, command: Command) {
271            lock(&self.commands).push(command);
272            let _ = rustix::io::write(&**self.wake, &1u64.to_ne_bytes());
273        }
274    }

276    pub(super) fn available() -> bool {
277        reactor().is_some()
278    }

280    pub(super) fn register(
281        polls: &[PollRequest],
282        deadline: Option<Instant>,
283        on_end: OnEnd,
284    ) -> Result<Wait, OnEnd> {
285        let Some(reactor) = reactor() else {
286            return Err(on_end);
287        };
288        if reactor.stopped.load(Ordering::Acquire) {
289            return Err(on_end);
290        }
291        let id = reactor.next_id.fetch_add(1, Ordering::Relaxed);
292        reactor.send(Command::Register {
293            id,
294            polls: polls.to_vec(),
295            deadline,
296            on_end,
297        });
298        Ok(Wait { id, reactor })
299    }

301    /// The reactor, started on first use.
302    fn reactor() -> Option<&'static Reactor> {
303        REACTOR.get_or_init(start).as_ref()
304    }

306    fn start() -> Option<Reactor> {
307        // Both moved above the numbers a program's own descriptors take,
308        // as the interpreter's other descriptors are, so that a socket's
309        // number does not depend on whether the reactor runs.
310        let epoll = reserve(epoll::create(epoll::CreateFlags::CLOEXEC).ok()?);
311        let wake = Arc::new(reserve(
312            rustix::event::eventfd(
313                0,
314                rustix::event::EventfdFlags::CLOEXEC | rustix::event::EventfdFlags::NONBLOCK,
315            )
316            .ok()?,
317        ));
318        epoll::add(&*epoll, &**wake, EventData::new_u64(WAKE), EventFlags::IN).ok()?;
319        let drained = Arc::clone(&wake);
320        std::thread::Builder::new()
321            .name("rxsock reactor".into())
322            // Its frames are shallow; the default would reserve megabytes
323            // of address space for nothing.
324            .stack_size(64 * 1024)
325            .spawn(move || run(&epoll, &drained))
326            .ok()?;
327        Some(Reactor {
328            commands: Mutex::new(Vec::new()),
329            wake,
330            next_id: AtomicU64::new(0),
331            stopped: AtomicBool::new(false),
332        })
333    }

335    /// One registered wait, as the thread holds it.
336    struct Waiter {
337        /// Each descriptor once, with everything waited for on it.
338        fds: Vec<PollRequest>,
339        deadline: Option<Instant>,
340        on_end: OnEnd,
341    }

343    /// One descriptor, as the thread holds it.
344    #[derive(Default)]
345    struct Watched {
346        waiters: Vec<u64>,
347        /// The interest it is registered with, or `None` while it is not in
348        /// the interest list.
349        registered: Option<EventFlags>,
350        /// When a muted descriptor is added again.
351        muted_until: Option<Instant>,
352        /// Whether its registration is to be made again even where the
353        /// interest has not changed.
354        recheck: bool,
355    }

357    #[derive(Default)]
358    struct State {
359        waiters: FxHashMap<u64, Waiter>,
360        watched: FxHashMap<i32, Watched>,
361        deadlines: BTreeSet<(Instant, u64)>,
362        mutes: BTreeSet<(Instant, i32)>,
363    }

365    /// The thread: takes commands, waits, ends what is over.
366    fn run(epoll: &OwnedFd, wake: &OwnedFd) {
367        let mut state = State::default();
368        let mut events: Vec<epoll::Event> = Vec::with_capacity(EVENTS);
369        loop {
370            take_commands(epoll, &mut state);
371            let now = crate::sync::now();
372            state.expire(epoll, now);
373            let next = [
374                state.deadlines.first().map(|(at, _)| *at),
375                state.mutes.first().map(|(at, _)| *at),
376            ]
377            .into_iter()
378            .flatten()
379            .min();
380            let timeout = next.map(|at| {
381                let left = at.saturating_duration_since(now);
382                rustix::time::Timespec {
383                    tv_sec: i64::try_from(left.as_secs()).unwrap_or(i64::MAX),
384                    tv_nsec: i64::from(left.subsec_nanos()),
385                }
386            });
387            events.clear();
388            match epoll::wait(
389                epoll,
390                rustix::buffer::spare_capacity(&mut events),
391                timeout.as_ref(),
392            ) {
393                Ok(_) => {}
394                Err(rustix::io::Errno::INTR) => continue,
395                Err(_) => {
396                    stop(&mut state);
397                    return;
398                }
399            }
400            for event in &events {
401                let data = event.data.u64();
402                if data == WAKE {
403                    let mut count = [0u8; 8];
404                    let _ = rustix::io::read(wake, &mut count);
405                    continue;
406                }
407                let Ok(fd) = i32::try_from(data) else {
408                    continue;
409                };
410                state.event(epoll, fd, event.flags);
411            }
412        }
413    }

415    /// Ends every wait after an error the thread cannot go on from, and
416    /// refuses registrations from then on.
417    fn stop(state: &mut State) {
418        if let Some(Some(reactor)) = REACTOR.get() {
419            reactor.stopped.store(true, Ordering::Release);
420            for command in std::mem::take(&mut *lock(&reactor.commands)) {
421                if let Command::Register { on_end, .. } = command {
422                    on_end();
423                }
424            }
425        }
426        for (_, waiter) in state.waiters.drain() {
427            (waiter.on_end)();
428        }
429    }
430
431    fn take_commands(epoll: &OwnedFd, state: &mut State) {
432        let Some(Some(reactor)) = REACTOR.get() else {
433            return;
434        };
435        let commands = std::mem::take(&mut *lock(&reactor.commands));
436        for command in commands {
437            match command {
438                Command::Register {
439                    id,
440                    polls,
441                    deadline,
442                    on_end,
443                } => state.add(epoll, id, &polls, deadline, on_end),
444                Command::Cancel(id) => state.end(epoll, id),
445            }
446        }
447    }

449    /// What `epoll` is asked to report for a wait for `flags`. Hang-up and
450    /// error it reports unasked.
451    fn interest(flags: PollFlags) -> EventFlags {
452        let mut interest = EventFlags::empty();
453        if flags.contains(PollFlags::IN) {
454            interest |= EventFlags::IN;
455        }
456        if flags.contains(PollFlags::OUT) {
457            interest |= EventFlags::OUT;
458        }
459        if flags.contains(PollFlags::PRI) {
460            interest |= EventFlags::PRI;
461        }
462        interest
463    }
464
465    /// What `epoll` reported, as `poll` would have.
466    fn reported(flags: EventFlags) -> PollFlags {
467        let mut revents = PollFlags::empty();
468        for (event, poll) in [
469            (EventFlags::IN, PollFlags::IN),
470            (EventFlags::OUT, PollFlags::OUT),
471            (EventFlags::PRI, PollFlags::PRI),
472            (EventFlags::ERR, PollFlags::ERR),
473            (EventFlags::HUP, PollFlags::HUP),
474        ] {
475            if flags.contains(event) {
476                revents |= poll;
477            }
478        }
479        revents
480    }

482    impl State {
483        fn add(
484            &mut self,
485            epoll: &OwnedFd,
486            id: u64,
487            polls: &[PollRequest],
488            deadline: Option<Instant>,
489            on_end: OnEnd,
490        ) {
491            let mut fds: Vec<PollRequest> = Vec::with_capacity(polls.len());
492            for &(fd, flags) in polls {
493                match fds.iter_mut().find(|(seen, _)| *seen == fd) {
494                    Some((_, union)) => *union |= flags,
495                    None => fds.push((fd, flags)),
496                }
497            }
498            if let Some(deadline) = deadline {
499                self.deadlines.insert((deadline, id));
500            }
501            self.waiters.insert(
502                id,
503                Waiter {
504                    fds: fds.clone(),
505                    deadline,
506                    on_end,
507                },
508            );
509            for (fd, _) in fds {
510                let watched = self.watched.entry(fd).or_default();
511                watched.waiters.push(id);
512                // The number may name a new socket since it was registered:
513                // a closed one leaves the interest list silently. Telling
514                // `epoll` again costs a call per wait and finds out.
515                watched.recheck = watched.registered.is_some();
516                // A descriptor that cannot be registered (closed, or not
517                // one `epoll` takes) ends its waiters, as a `poll` of it
518                // would have reported it at once.
519                if !self.update(epoll, fd) {
520                    self.end_waiters_of(epoll, fd);
521                }
522                if !self.waiters.contains_key(&id) {
523                    break;
524                }
525            }
526        }
527
528        /// Ends wait `id`, if it has not ended, and posts its end.
529        fn end(&mut self, epoll: &OwnedFd, id: u64) {
530            let Some(waiter) = self.waiters.remove(&id) else {
531                return;
532            };
533            if let Some(deadline) = waiter.deadline {
534                self.deadlines.remove(&(deadline, id));
535            }
536            for (fd, _) in &waiter.fds {
537                if let Some(watched) = self.watched.get_mut(fd) {
538                    watched.waiters.retain(|&other| other != id);
539                }
540                self.update(epoll, *fd);
541            }
542            (waiter.on_end)();
543        }
544
545        fn end_waiters_of(&mut self, epoll: &OwnedFd, fd: i32) {
546            let ids = self
547                .watched
548                .get(&fd)
549                .map(|watched| watched.waiters.clone())
550                .unwrap_or_default();
551            for id in ids {
552                self.end(epoll, id);
553            }
554        }
555
556        /// Brings `fd`'s registration in line with its waiters, and answers
557        /// whether it could be.
558        fn update(&mut self, epoll: &OwnedFd, fd: i32) -> bool {
559            let Some(watched) = self.watched.get_mut(&fd) else {
560                return true;
561            };
562            if watched.waiters.is_empty() {
563                if watched.registered.is_some() {
564                    // A closed descriptor has already left the list.
565                    let _ = with_socket(fd, |socket| {
566                        epoll::delete(epoll, socket).map_err(|error| error.raw_os_error())
567                    });
568                }
569                if let Some(until) = watched.muted_until {
570                    self.mutes.remove(&(until, fd));
571                }
572                self.watched.remove(&fd);
573                return true;
574            }
575            if watched.muted_until.is_some() {
576                return true;
577            }
578            let wanted = watched
579                .waiters
580                .iter()
581                .filter_map(|id| self.waiters.get(id))
582                .flat_map(|waiter| waiter.fds.iter())
583                .filter(|(other, _)| *other == fd)
584                .fold(EventFlags::empty(), |union, (_, flags)| {
585                    union | interest(*flags)
586                });
587            if watched.registered == Some(wanted) && !std::mem::take(&mut watched.recheck) {
588                return true;
589            }
590            let data = EventData::new_u64(u64::try_from(fd).unwrap_or(WAKE));
591            let registered = watched.registered;
592            let done = with_socket(fd, |socket| {
593                let first = match registered {
594                    Some(_) => epoll::modify(epoll, socket, data, wanted),
595                    None => epoll::add(epoll, socket, data, wanted),
596                };
597                match first {
598                    Err(rustix::io::Errno::NOENT) => epoll::add(epoll, socket, data, wanted),
599                    Err(rustix::io::Errno::EXIST) => epoll::modify(epoll, socket, data, wanted),
600                    other => other,
601                }
602                .map_err(|error| error.raw_os_error())
603            });
604            let watched = self.watched.get_mut(&fd).expect("looked up above");
605            match done {
606                Ok(()) => {
607                    watched.registered = Some(wanted);
608                    watched.recheck = false;
609                    true
610                }
611                Err(_) => {
612                    watched.registered = None;
613                    false
614                }
615            }
616        }
617
618        /// `epoll` reported `flags` for `fd`: ends each waiter they answer,
619        /// and mutes `fd` where they answer none.
620        fn event(&mut self, epoll: &OwnedFd, fd: i32, flags: EventFlags) {
621            let revents = reported(flags);
622            let Some(watched) = self.watched.get(&fd) else {
623                return;
624            };
625            let answered: Vec<u64> = watched
626                .waiters
627                .iter()
628                .copied()
629                .filter(|id| {
630                    self.waiters.get(id).is_some_and(|waiter| {
631                        waiter
632                            .fds
633                            .iter()
634                            .any(|(other, wanted)| *other == fd && answers(*wanted, revents))
635                    })
636                })
637                .collect();
638            if answered.is_empty() {
639                if !revents.is_empty() {
640                    self.mute(epoll, fd);
641                }
642                return;
643            }
644            for id in answered {
645                self.end(epoll, id);
646            }
647        }
648
649        fn mute(&mut self, epoll: &OwnedFd, fd: i32) {
650            let Some(watched) = self.watched.get_mut(&fd) else {
651                return;
652            };
653            if watched.registered.take().is_some() {
654                let _ = with_socket(fd, |socket| {
655                    epoll::delete(epoll, socket).map_err(|error| error.raw_os_error())
656                });
657            }
658            let until = crate::sync::now() + MUTED_RECHECK;
659            watched.muted_until = Some(until);
660            self.mutes.insert((until, fd));
661        }
662
663        /// Ends the waits whose deadline has passed and adds back the muted
664        /// descriptors whose recheck is due.
665        fn expire(&mut self, epoll: &OwnedFd, now: Instant) {
666            while let Some(&(at, id)) = self.deadlines.first() {
667                if at > now {
668                    break;
669                }
670                self.deadlines.remove(&(at, id));
671                self.end(epoll, id);
672            }
673            while let Some(&(at, fd)) = self.mutes.first() {
674                if at > now {
675                    break;
676                }
677                self.mutes.remove(&(at, fd));
678                if let Some(watched) = self.watched.get_mut(&fd) {
679                    watched.muted_until = None;
680                }
681                if !self.update(epoll, fd) {
682                    self.end_waiters_of(epoll, fd);
683                }
684            }
685        }
686    }
