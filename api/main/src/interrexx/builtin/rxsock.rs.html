1//! The `rxsock` extension library, run in Rust: the IPv4 socket routines
2//! `docs/design/extensions.md` describes.
3//!
4//! A socket is its descriptor number. Each call borrows the number for the call
5//! ([`rxsock_sys::with_socket`]) and makes its OS calls through `rustix` or
6//! `nix` on the borrow; the options neither names go through [`rxsock_sys`]'s
7//! raw `getsockopt` and `setsockopt`. Most routines set `errno` and `h_errno` in
8//! the caller's variables ([`report`]); the `errno` is also kept on the activity
9//! for `SockSock_Errno`.
10//!
11//! `SockAccept`, `SockConnect`, `SockRecv`, `SockRecvFrom` and `SockSelect`
12//! wait without holding up the other activities: a call that would block
13//! parks its activity and hands its sockets and deadline to the process's
14//! socket reactor ([`rxsock_wait`]), one thread serving every wait of every
15//! interpreter, and the OS call is made once the reactor has posted that
16//! the sockets are ready or the deadline has passed. A halt or the
17//! interpreter's end cancels the registration, so the wait ends at once.
18//! Where the platform has no reactor, a pool thread `poll`s the sockets
19//! together with a cancel descriptor instead ([`wait_off_baton`]); where no
20//! thread can be had either (the pool configured inline, or the simulation
21//! mode), the activity parks on a short sleep whose re-test polls the
22//! sockets ([`wait_again`]).

152/// Sets the caller's `errno` to `errno`'s text and `h_errno` to 0, and keeps
153/// a failure's `errno` for `SockSock_Errno`; a success leaves that as it
154/// was, as C's `errno` is left.
155fn report(interp: &mut Interp, errno: i32) {
156    if errno != 0 {
157        interp.activity.rxsock_errno = errno;
158    }
159    let text = interp.text_built(errno_text(errno));
160    interp.set_context_variable(b"ERRNO", text);
161    let zero = number(interp, 0);
162    interp.set_context_variable(b"H_ERRNO", zero);
163}

165/// A call's answer: its value with `errno` 0, or -1 with its `errno`.
166fn answer(interp: &mut Interp, result: Os<i64>) -> ObjRef {
167    let value = match result {
168        Ok(value) => {
169            report(interp, 0);
170            value

172        Err(errno) => {
173            report(interp, errno);
174            -1
175        }
176    };
177    number(interp, value)
178}

271/// `text` with leading and trailing blanks removed.
272fn trimmed(text: &[u8]) -> &[u8] {
273    let start = text
274        .iter()
275        .position(|byte| *byte != b' ')
276        .unwrap_or(text.len());
277    let end = text
278        .iter()
279        .rposition(|byte| *byte != b' ')
280        .map_or(start, |at| at + 1);
281    &text[start..end]
282}

284/// C's `atoi`: leading white space, a sign and the decimal digits after it,
285/// 0 where there are none; wrapped to 32 bits as C's `int` overflow is.
286fn atoi(text: &[u8]) -> i32 {
287    let mut rest = text;
288    while let Some((first, tail)) = rest.split_first()
289        && first.is_ascii_whitespace()
290    {
291        rest = tail;
292    }
293    let negative = rest.first() == Some(&b'-');
294    if matches!(rest.first(), Some(b'+' | b'-')) {
295        rest = &rest[1..];
296    }
297    let mut value: i64 = 0;
298    for digit in rest.iter().take_while(|byte| byte.is_ascii_digit()) {
299        value = value.wrapping_mul(10).wrapping_add(i64::from(digit - b'0'));
300    }
301    if negative {
302        value = value.wrapping_neg();
303    }
304    // C's int wraps.
305    i32::cast_truncate(value)
306}

321    prefix: Vec<u8>,
322}

324/// The stem `value` names: a stem object itself, or a string naming one of
325/// the caller's stems, `name.tail` prefixing each element with `TAIL`.
326/// `None` where the string names no stem.
327fn stem_of(interp: &mut Interp, value: ObjRef, position: usize) -> Result<Option<Stem>> {
328    if interp.is_stem_object(value) {
329        return Ok(Some(Stem {
330            object: value,
331            prefix: Vec::new(),
332        }));
333    }
334    let Some(text) = interp.blamed_string_conversion(value)? else {
335        return Err(crate::error::Raised::native_argument_needs_a_string_value(position).into());
336    };
337    let mut name = interp.to_text(text).to_ascii_uppercase();
338    if let Some(end) = name.iter().position(|byte| *byte == 0) {
339        name.truncate(end);
340    }
341    let (stem_name, prefix) = match name.iter().position(|byte| *byte == b'.') {
342        Some(dot) => (name[..=dot].to_vec(), name[dot + 1..].to_vec()),
343        None => (name.clone(), Vec::new()),
344    };
345    if stem_name.is_empty() {
346        return Ok(None);
347    }
348    Ok(interp
349        .context_stem_named(&stem_name)
350        .map(|object| Stem { object, prefix }))
351}

353/// A required stem argument.
354fn stem_argument(

364    fn key(&self, element: &[u8]) -> Vec<u8> {
365        let mut key = self.prefix.clone();

414/// `inet_aton`: one to four dotted parts, each decimal, octal (`0`) or hex
415/// (`0x`), the last filling the bytes the others leave; trailing white space
416/// ends the address. `None` for anything else.
417fn inet_aton(text: &[u8]) -> Option<Ipv4Addr> {
418    let end = text
419        .iter()
420        .position(u8::is_ascii_whitespace)
421        .unwrap_or(text.len());
422    let text = &text[..end];
423    let mut parts = Vec::new();
424    for part in text.split(|byte| *byte == b'.') {
425        if !part.first().is_some_and(u8::is_ascii_digit) {
426            return None;
427        }
428        let (digits, radix) = match part {
429            [b'0', b'x' | b'X', rest @ ..] => (rest, 16),
430            [b'0', rest @ ..] if !rest.is_empty() => (rest, 8),
431            _ => (part, 10),
432        };
433        let mut value: u64 = 0;
434        for byte in digits {
435            let digit = char::from(*byte).to_digit(radix)?;
436            value = value
437                .checked_mul(u64::from(radix))?
438                .checked_add(u64::from(digit))?;
439            if value > u64::from(u32::MAX) {
440                return None;
441            }
442        }
443        parts.push(value);
444    }
445    let (last, leading) = parts.split_last()?;
446    if leading.len() > 3 || leading.iter().any(|part| *part > 0xff) {
447        return None;
448    }
449    let room = 32 - 8 * leading.len();
450    if room < 32 && *last >> room != 0 {
451        return None;
452    }
453    let mut value = *last;
454    for (index, part) in leading.iter().enumerate() {
455        value |= part << (24 - 8 * index);
456    }
457    u32::try_from(value).ok().map(Ipv4Addr::from)
458}

460/// An `ADDR` value: `INADDR_ANY` caselessly, else `inet_addr`'s reading,
461/// 255.255.255.255 where that fails.
462fn parse_addr(text: &[u8]) -> Ipv4Addr {
463    if text.eq_ignore_ascii_case(b"INADDR_ANY") {
464        return Ipv4Addr::UNSPECIFIED;
465    }
466    inet_aton(text).unwrap_or(Ipv4Addr::BROADCAST)
467}

469/// The address a stem holds, all zero where any element is unset.
470fn read_address(interp: &mut Interp, stem: &Stem) -> Address {
471    let (Some(family), Some(port), Some(addr)) = (
472        stem.get(interp, b"FAMILY"),
473        stem.get(interp, b"PORT"),
474        stem.get(interp, b"ADDR"),
475    ) else {
476        return Address::ZERO;
477    };
478    let family = trimmed(&family);
479    // C's u_short.
480    let family = if family.eq_ignore_ascii_case(b"AF_INET") {
481        2
482    } else {
483        u16::cast_truncate(atoi(family).cast_unsigned())
484    };
485    // C's htons.
486    let port = u16::cast_truncate(atoi(trimmed(&port)).cast_unsigned());
487    Address {
488        family,
489        port,
490        addr: parse_addr(trimmed(&addr)),
491    }
492}

501/// Writes a socket address an OS call answered: an IPv4 one as itself, any
502/// other with its family and zeros.
503fn write_socket_address(
504    interp: &mut Interp,
505    stem: &Stem,
506    address: Option<rustix::net::SocketAddrAny>,
507) {
508    let unspecified = SocketAddrV4::new(Ipv4Addr::UNSPECIFIED, 0);
509    match address {
510        Some(any) => {
511            let family = i64::from(any.address_family().as_raw());
512            let v4 = SocketAddrV4::try_from(any).unwrap_or(unspecified);
513            write_address(interp, stem, family, v4);
514        }
515        None => write_address(interp, stem, 0, unspecified),
516    }
517}

519/// How an inet socket takes `address` for `bind`, `connect` or `sendto`.
520enum Target {
521    Inet(SocketAddrV4),
522    /// `AF_UNSPEC`, which `connect` reads as a disconnect, `bind` accepts with
523    /// the any address, and `sendto` reads as `AF_INET`.
524    Unspec(SocketAddrV4),
525    /// Any other family, which an inet socket refuses with `EAFNOSUPPORT`.
526    Other,
527}
528
529fn target(address: Address) -> Target {
530    match address.family {
531        2 => Target::Inet(address.v4()),
532        0 => Target::Unspec(address.v4()),
533        _ => Target::Other,
534    }

551/// What waits for a socket wait off the baton: the token its end is posted

560/// How a [`Helper`] is told to stop.
561enum Stop {
562    /// A registration with the process's reactor.
563    Reactor(rxsock_wait::Wait),
564    /// A pool thread of its own, polling the canceller's descriptor.
565    Thread(Arc<rxsock_wait::Canceller>),
566}
567
568impl Stop {
569    fn cancel(&self) {
570        match self {
571            Stop::Reactor(wait) => wait.cancel(),
572            Stop::Thread(canceller) => canceller.signal(),
573        }
574    }
575}

577impl Drop for Helper {
578    fn drop(&mut self) {
579        self.stop.cancel();
580    }

616/// The readiness of `polls` now, each entry's received events; `None` where
617/// any descriptor is invalid.
618fn poll_now(polls: &[PollRequest]) -> Option<Vec<PollFlags>> {
619    rxsock_sys::poll(polls.iter().copied(), Some(Duration::ZERO), None)
620        .map(|(_, readiness)| readiness)
621        .ok()
622}

624/// Whether `revents` answers a wait for `events`, as `select` reads it:
625/// hang-up and error count as readable and error as writable.
626pub(crate) fn answers(events: PollFlags, revents: PollFlags) -> bool {
627    let readable = PollFlags::IN | PollFlags::HUP | PollFlags::ERR;
628    let writable = PollFlags::OUT | PollFlags::ERR;
629    (events.contains(PollFlags::IN) && revents.intersects(readable))
630        || (events.contains(PollFlags::OUT) && revents.intersects(writable))
631        || (events.contains(PollFlags::PRI) && revents.contains(PollFlags::PRI))
632        || revents.contains(PollFlags::NVAL)
633}
634
635/// Whether a wait for `polls` is over: any of them ready, or any invalid.
636fn any_ready(polls: &[PollRequest]) -> bool {
637    match poll_now(polls) {
638        Some(revents) => polls
639            .iter()
640            .zip(revents)
641            .any(|((_, events), revents)| answers(*events, revents)),
642        None => true,
643    }
644}

646/// Whether the running activity's socket wait is over: its deadline has
647/// passed or its sockets are ready. A ready listening socket can lose its
648/// connection to another accept, in this process or another, before this
649/// one runs: the accept is tried here without blocking, and the wait goes on
650/// where it finds nothing.
651fn wait_over(interp: &mut Interp) -> bool {
652    let now = interp.now();
653    let Some(wait) = interp.activity.socket_wait.as_mut() else {
654        return true;
655    };
656    if wait.deadline.is_some_and(|deadline| now >= deadline) {
657        return true;
658    }
659    if !any_ready(&wait.polls) {
660        return false;
661    }
662    let Pending::Accept { fd, accepted, .. } = &mut wait.op else {
663        return true;
664    };
665    match try_accept(*fd) {
666        Err(EAGAIN) => false,
667        result => {
668            *accepted = Some(result);
669            true
670        }
671    }
672}

674/// The running activity's socket wait, re-tested at its sleep's end: the
675/// next poll's time while it goes on, `None` once its sockets are ready or
676/// its deadline has passed.
677pub(crate) fn wait_again(interp: &mut Interp) -> Option<Instant> {
678    if wait_over(interp) {
679        return None;
680    }
681    let now = interp.now();
682    let deadline = interp.activity.socket_wait.as_ref()?.deadline;
683    let next = now + WAIT_POLL;
684    Some(deadline.map_or(next, |deadline| deadline.min(next)))
685}

687/// The running activity's socket wait, re-tested once the reactor or its
688/// pool thread has posted its end: `None` once it is over, or the reason to
689/// park again for, off the baton where that can be had and on a polling
690/// sleep where not.
691pub(crate) fn block_again(interp: &mut Interp) -> Option<ParkReason> {
692    if wait_over(interp) {
693        return None;
694    }
695    if start_helper(interp) {
696        return Some(ParkReason::Socket);
697    }
698    wait_again(interp).map(|deadline| ParkReason::Sleep { deadline })
699}

732/// Starts waiting off the baton for the running activity's socket wait,
733/// and answers whether that was done: with the process's reactor where it
734/// has one, and on a pool thread of its own where not. Neither in the
735/// simulation mode, whose clock no other thread may read, nor where the
736/// pool runs work inline or no thread or descriptor can be had.
737fn start_helper(interp: &mut Interp) -> bool {
738    if interp.sim.is_some() {
739        return false;
740    }
741    let Some(wait) = interp.activity.socket_wait.as_ref() else {
742        return false;
743    };
744    let (polls, deadline) = (wait.polls.clone(), wait.deadline);
745    let registered = if rxsock_wait::reactor_available() {
746        interp.register_socket_wait(&polls, deadline)
747    } else {
748        None
749    };
750    let stop = match registered {
751        Some((token, registered)) => Some((token, Stop::Reactor(registered))),
752        None => start_thread(interp, polls, deadline),
753    };
754    let Some((token, stop)) = stop else {
755        return false;
756    };
757    if let Some(wait) = interp.activity.socket_wait.as_mut() {
758        wait.helper = Some(Helper {
759            token,
760            ended: false,
761            stop,
762        });
763    }
764    true
765}

767/// A pool thread polling `polls` until `deadline`, where there is no
768/// reactor: the token its end is posted under and how it is stopped.
769fn start_thread(
770    interp: &mut Interp,
771    polls: Vec<PollRequest>,
772    deadline: Option<Instant>,
773) -> Option<(u64, Stop)> {
774    let canceller = Arc::new(rxsock_wait::Canceller::new()?);
775    let Ok(Some(worker)) = interp.pool.reserve() else {
776        return None;
777    };
778    let polled = Arc::clone(&canceller);
779    let token = interp.exit_for_socket_wait(
780        worker,
781        Box::new(move || wait_off_baton(&polls, deadline, &polled)),
782    );
783    Some((token, Stop::Thread(canceller)))
784}

786/// A pool thread's wait for `polls`: until any is ready or invalid, until
787/// `deadline`, or until `cancel` is written. A signal's interruption polls
788/// again for what is left.
789///
790/// `poll` reports hang-up and error whatever was asked, and those answer no
791/// wait for priority data alone, so an exception-set entry in that state
792/// would end every poll at once without ending the wait. Such an entry is
793/// left out of the polls that follow, and looked at again every
794/// [`MUTED_RECHECK`], since its socket may yet be connected.
795fn wait_off_baton(
796    polls: &[PollRequest],
797    deadline: Option<Instant>,
798    cancel: &rxsock_wait::Canceller,
799) {
800    let mut muted = vec![false; polls.len()];
801    loop {
802        let now = crate::sync::now();
803        if deadline.is_some_and(|deadline| now >= deadline) {
804            return;
805        }
806        let any_muted = muted.contains(&true);
807        let left = match (deadline, any_muted) {
808            (Some(deadline), true) => Some((deadline - now).min(MUTED_RECHECK)),
809            (Some(deadline), false) => Some(deadline - now),
810            (None, true) => Some(MUTED_RECHECK),
811            (None, false) => None,
812        };
813        let live: Vec<usize> = (0..polls.len()).filter(|&i| !muted[i]).collect();
814        let requests = live.iter().map(|&i| polls[i]);
815        let polled = rxsock_sys::poll(requests, left, Some(cancel.polled()));
816        let (ready, revents) = match polled {
817            Err(EINTR) => continue,
818            Err(_) => return,
819            Ok(answer) => answer,
820        };
821        if ready == 0 {
822            // The deadline is tested above; a recheck's timeout unmutes.
823            muted.fill(false);
824            continue;
825        }
826        let Some((cancelled, entries)) = revents.split_last() else {
827            return;
828        };
829        if !cancelled.is_empty() {
830            return;
831        }
832        let mut answered = false;
833        for (&i, revents) in live.iter().zip(entries) {
834            if answers(polls[i].1, *revents) {
835                answered = true;
836            } else if !revents.is_empty() {
837                muted[i] = true;
838            }
839        }
840        if answered {
841            return;
842        }
843    }
844}

846/// Parks the running call until `wait`'s sockets are ready or its deadline
847/// passes, then answers what [`resume`] does; `stems` are the stems the
848/// finishing call writes, kept rooted meanwhile.
849fn park(interp: &mut Interp, wait: SocketWait, stems: &[Option<ObjRef>]) -> Result<ObjRef> {
850    let receiver = interp.new_array(stems);
851    interp.activity.socket_wait = Some(Box::new(wait));
852    let reason = if start_helper(interp) {
853        ParkReason::Socket
854    } else {
855        let now = interp.now();
856        let next = now + WAIT_POLL;
857        let deadline = interp
858            .activity
859            .socket_wait
860            .as_ref()
861            .and_then(|wait| wait.deadline);
862        ParkReason::Sleep {
863            deadline: deadline.map_or(next, |deadline| deadline.min(next)),
864        }
865    };
866    let parked = interp.park_rust_routine(reason, resume, receiver);
867    if interp.activity.native_park.is_none() {
868        interp.activity.socket_wait = None;
869    }
870    parked
871}

881/// A waiting call's continuation: the OS call it waited to make, or, where a
882/// halt ended the wait, -1 with `EINTR`.
883fn resume(interp: &mut Interp, receiver: ObjRef) -> Result<Option<ObjRef>> {
884    interp.activity.asleep = None;
885    let halted = std::mem::take(&mut interp.activity.woken_by_halt);
886    let Some(wait) = interp.activity.socket_wait.take() else {
887        return Ok(Some(number(interp, -1)));
888    };
889    let stem = |interp: &Interp, index: usize, prefix: Option<Vec<u8>>| {
890        prefix.and_then(|prefix| {
891            kept_stem(interp, receiver, index).map(|object| Stem { object, prefix })
892        })
893    };
894    let answered = match wait.op {
895        _ if halted => {
896            if let Pending::Recv { var, .. } = &wait.op {
897                let empty = interp.text(b"");
898                interp.set_context_variable(var, empty);
899            }
900            answer(interp, Err(EINTR))
901        }
902        Pending::Accept {
903            fd,
904            stem: prefix,
905            lost,
906            accepted,
907        } => {
908            // At the deadline, the one try answers `EAGAIN` where nothing came:
909            // `SO_RCVTIMEO` 500 then `SockAccept` gives -1 with `errno` 11
910            // after half a second.
911            let accepted = accepted.unwrap_or_else(|| try_accept(fd));
912            let target = stem(interp, 0, prefix);
913            finish_accept(interp, accepted, target.as_ref(), lost)
914        }
915        Pending::Connect { fd, to } => {
916            // Not writable is the deadline: `SO_SNDTIMEO` 500 then a
917            // `SockConnect` that cannot complete gives -1 with `errno` 115
918            // after half a second, and the connection goes on.
919            let result = if any_ready(&[(fd, rxsock_sys::CONNECT_WAIT_EVENTS)]) {
920                with_socket(fd, |socket| finish_connect(socket, to))
921            } else {
922                Err(EINPROGRESS)
923            };
924            code(interp, result)
925        }
926        Pending::Recv {
927            fd,
928            var,
929            len,
930            flags,
931            from,
932        } => {
933            let target = stem(interp, 0, from);
934            #[cfg(unix)]
935            let flags = flags | RecvFlags::DONTWAIT;
936            finish_recv(interp, fd, &var, len, flags, target.as_ref())
937        }
938        #[cfg(windows)]
939        Pending::Send {
940            fd,
941            bytes,
942            flags,
943            target,
944        } => {
945            let result = with_socket(fd, |socket| {
946                send_now(socket, &bytes, flags, target.as_ref())
947            });
948            answer(interp, result)
949        }
950        Pending::Select { sets } => {
951            let mut sets = sets;
952            for (index, set) in sets.iter_mut().enumerate() {
953                if let Some(set) = set {
954                    set.stem.object = kept_stem(interp, receiver, index).unwrap_or(ObjRef::NIL);
955                }
956            }
957            finish_select(interp, &sets)
958        }
959    };
960    Ok(Some(answered))
961}

1007/// `SockSocket(domain, type, protocol)`.
1008pub(crate) fn socket_new(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<ObjRef> {
1009    check_arity(args, 3)?;
1010    let domain = c_string(interp, args, 1)?;
1011    let kind = c_string(interp, args, 2)?;
1012    let protocol = c_string(interp, args, 3)?;
1013    if !trimmed(&domain).eq_ignore_ascii_case(b"AF_INET") {
1014        return Err(failed(interp));
1015    }
1016    let kind = match trimmed(&kind).to_ascii_uppercase().as_slice() {
1017        b"SOCK_STREAM" => rustix::net::SocketType::STREAM,
1018        b"SOCK_DGRAM" => rustix::net::SocketType::DGRAM,
1019        b"SOCK_RAW" => rustix::net::SocketType::RAW,
1020        _ => return Err(failed(interp)),
1021    };
1022    let protocol = match trimmed(&protocol).to_ascii_uppercase().as_slice() {
1023        b"IPPROTO_UDP" => Some(rustix::net::ipproto::UDP),
1024        b"IPPROTO_TCP" => Some(rustix::net::ipproto::TCP),
1025        b"0" => None,
1026        _ => return Err(failed(interp)),
1027    };
1028    #[cfg(windows)]
1029    if let Err(error) = rxsock_sys::ensure_started() {
1030        return Ok(answer(interp, Err(error)));
1031    }
1032    let made = rustix::net::socket(rustix::net::AddressFamily::INET, kind, protocol)
1033        .and_then(|socket| {
1034            rxsock_sys::prepare_socket(socket.as_fd())?;
1035            let fd = socket.into_raw_fd();
1036            #[cfg(windows)]
1037            let fd = rxsock_sys::publish_socket(fd, false);
1038            Ok(fd)
1039        })
1040        .map_err(os);
1041    Ok(answer_socket(interp, made))
1042}

1083/// `SockConnect(s, addrstem)`. A blocking socket connects without blocking
1084/// and the call waits for the outcome.
1085pub(crate) fn connect(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<ObjRef> {
1086    check_arity(args, 2)?;
1087    let fd = socket_arg(interp, args)?;
1088    let Some(stem) = stem_argument(interp, args, 2)? else {
1089        return Ok(number(interp, 0));
1090    };
1091    let address = read_address(interp, &stem);
1092    // Ok(Some(to)) for a connection under way, which the call waits for.
1093    let begun = with_socket(fd, |socket| {
1094        let v4 = match target(address) {
1095            Target::Inet(v4) => v4,
1096            Target::Unspec(_) => {
1097                return rustix::net::connect_unspec(socket)
1098                    .map(|()| None)
1099                    .map_err(os);
1100            }
1101            Target::Other => return Err(EAFNOSUPPORT),
1102        };
1103        if !blocks(socket) {
1104            return rustix::net::connect(socket, &v4).map(|()| None).map_err(os);
1105        }
1106        start_connect(socket, v4)
1107    });
1108    match begun {
1109        Ok(Some(to)) => {
1110            let wait = SocketWait {
1111                polls: vec![(fd, rxsock_sys::CONNECT_WAIT_EVENTS)],
1112                deadline: socket_deadline(interp, fd, rustix::net::sockopt::Timeout::Send),
1113                helper: None,
1114                op: Pending::Connect { fd, to },
1115            };
1116            park(interp, wait, &[])
1117        }
1118        result => Ok(code(interp, result.map(|_| ()))),
1119    }
1120}

1172pub(crate) fn accept(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<ObjRef> {
1173    check_arity(args, 2)?;
1174    let fd = socket_arg(interp, args)?;
1175    let (stem, lost) = match args.get(1).copied().flatten() {
1176        Some(value) => match stem_of(interp, value, 2)? {
1177            Some(stem) => (Some(stem), false),
1178            None => (None, true),
1179        },
1180        None => (None, false),
1181    };
1182    let blocking = with_socket(fd, |socket| Ok(blocks(socket))).unwrap_or(false);
1183    // A blocking socket's accept is tried without blocking once it is
1184    // ready, and waits again where another accept took the connection.
1185    let accepted = match blocking && !any_ready(&[(fd, PollFlags::IN)]) {
1186        true => None,
1187        false => match try_accept(fd) {
1188            Err(EAGAIN) if blocking => None,
1189            result => Some(result),
1190        },
1191    };
1192    let Some(accepted) = accepted else {
1193        let wait = SocketWait {
1194            polls: vec![(fd, PollFlags::IN)],
1195            deadline: socket_deadline(interp, fd, rustix::net::sockopt::Timeout::Recv),
1196            helper: None,
1197            op: Pending::Accept {
1198                fd,
1199                stem: stem.as_ref().map(|stem| stem.prefix.clone()),
1200                lost,
1201                accepted: None,
1202            },
1203        };
1204        let kept = [stem.as_ref().map(|stem| stem.object)];
1205        return park(interp, wait, &kept);
1206    };
1207    Ok(finish_accept(interp, accepted, stem.as_ref(), lost))
1208}

1210/// An accepted connection and its peer's address.
1211struct Accepted {

1213    address: Option<rustix::net::SocketAddrAny>,

1218impl Accepted {

1220        let fd = self.socket.into_raw_fd();
1221        #[cfg(windows)]
1222        let fd = rxsock_sys::publish_socket(fd, self.nonblocking);

1224    }

1227/// `accept` on the socket numbered `fd` without blocking: a blocking socket
1228/// is made non-blocking for the call. The socket it answers has the mode the
1229/// host's `accept` gives it for the listening socket's own mode, which the
1230/// call's toggle does not change. A socket it cannot ready is closed and the
1231/// failure answered.
1232#[cfg(unix)]
1233fn try_accept(fd: SocketId) -> Os<Accepted> {
1234    with_socket(fd, |socket| {
1235        let flags = rustix::fs::fcntl_getfl(socket).map_err(os)?;
1236        let toggled = !flags.contains(rustix::fs::OFlags::NONBLOCK);
1237        if toggled {
1238            rustix::fs::fcntl_setfl(socket, flags | rustix::fs::OFlags::NONBLOCK).map_err(os)?;
1239        }
1240        let accepted = rustix::net::acceptfrom(socket).map_err(os);
1241        if toggled {
1242            let _ = rustix::fs::fcntl_setfl(socket, flags);
1243        }
1244        let (new, address) = accepted?;
1245        // A host whose accepted socket takes the listening socket's mode
1246        // gave it the toggle's `O_NONBLOCK`, which the program never set.
1247        if toggled && crate::host::ACCEPT_INHERITS_NONBLOCK {
1248            blocking(&new).map_err(os)?;
1249        }
1250        rxsock_sys::prepare_socket(new.as_fd()).map_err(os)?;
1251        Ok(Accepted {
1252            socket: new,
1253            address,
1254        })
1255    })
1256}

1258#[cfg(windows)]
1259fn try_accept(fd: SocketId) -> Os<Accepted> {
1260    with_socket(fd, |socket| {
1261        let toggled = blocks(socket);
1262        if toggled {
1263            rustix::io::ioctl_fionbio(socket, true).map_err(os)?;
1264        }
1265        let accepted = rustix::net::acceptfrom(socket).map_err(os);
1266        if toggled {
1267            rustix::io::ioctl_fionbio(socket, false).map_err(os)?;
1268        }
1269        let (new, address) = accepted?;
1270        rxsock_sys::prepare_socket(new.as_fd()).map_err(os)?;
1271        rustix::io::ioctl_fionbio(new.as_fd(), !toggled).map_err(os)?;
1272        Ok(Accepted {
1273            socket: new,
1274            address,
1275            nonblocking: !toggled,
1276        })
1277    })
1278}

1280/// Makes `socket` blocking. A socket `accept` answers starts blocking on
1281/// Linux; Apple's systems and the BSDs give it the listening socket's
1282/// `O_NONBLOCK`, which [`try_accept`] sets for its call.
1283#[cfg(unix)]
1284fn blocking(socket: &OwnedFd) -> rustix::io::Result<()> {
1285    let flags = rustix::fs::fcntl_getfl(socket)?;
1286    if flags.contains(rustix::fs::OFlags::NONBLOCK) {
1287        rustix::fs::fcntl_setfl(socket, flags - rustix::fs::OFlags::NONBLOCK)?;
1288    }
1289    Ok(())
1290}

1317/// An accept's answer: the new socket's number, or -1. With a stem it gets
1318/// the peer's address, even on failure. Where the stem argument named no
1319/// stem (`lost`), the connection's descriptor is lost and the call answers
1320/// 0 without setting `errno`.
1321fn finish_accept(
1322    interp: &mut Interp,
1323    accepted: Os<Accepted>,
1324    stem: Option<&Stem>,
1325    lost: bool,
1326) -> ObjRef {
1327    if lost {
1328        if let Ok(accepted) = accepted {
1329            let _ = accepted.into_parts();
1330        }
1331        return number(interp, 0);
1332    }
1333    let (result, peer) = match accepted {
1334        Ok(accepted) => {
1335            let (socket, peer) = accepted.into_parts();
1336            (Ok(socket), peer)
1337        }
1338        Err(errno) => (Err(errno), None),
1339    };
1340    if let Some(stem) = stem {
1341        write_socket_address(interp, stem, peer);
1342    }
1343    answer_socket(interp, result)
1344}

1375/// Fills a host stem from `host`.
1376fn write_host(interp: &mut Interp, stem: &Stem, host: &hosts::Host) {
1377    stem.set_text(interp, b"NAME", &host.name);
1378    let aliases = i64::try_from(host.aliases.len()).unwrap_or(i64::MAX);
1379    stem.set_number(interp, b"ALIAS.0", aliases);
1380    for (index, alias) in host.aliases.iter().enumerate() {
1381        stem.set_text(interp, format!("ALIAS.{}", index + 1).as_bytes(), alias);
1382    }
1383    stem.set_text(interp, b"ADDRTYPE", b"AF_INET");
1384    if let Some(first) = host.addresses.first() {
1385        stem.set_text(interp, b"ADDR", first.to_string().as_bytes());
1386    }
1387    let addresses = i64::try_from(host.addresses.len()).unwrap_or(i64::MAX);
1388    stem.set_number(interp, b"ADDR.0", addresses);
1389    for (index, address) in host.addresses.iter().enumerate() {
1390        stem.set_text(
1391            interp,
1392            format!("ADDR.{}", index + 1).as_bytes(),
1393            address.to_string().as_bytes(),
1394        );
1395    }
1396}

1400    let (found, errno) = match found {

1516/// The flags a flag word list names, caselessly, unknown words ignored.
1517fn flag_words(text: &[u8], known: &[(&str, u32)]) -> u32 {
1518    text.split(u8::is_ascii_whitespace)
1519        .filter(|word| !word.is_empty())
1520        .filter_map(|word| {
1521            known
1522                .iter()
1523                .find(|(name, _)| word.eq_ignore_ascii_case(name.as_bytes()))
1524                .map(|(_, bit)| *bit)
1525        })
1526        .fold(0, |flags, bit| flags | bit)

1649/// Winsock has no per-call `MSG_DONTWAIT`. Keep the current mode when it is
1650/// already nonblocking and otherwise toggle it around the single I/O call.
1651#[cfg(windows)]
1652fn try_without_blocking<T>(socket: BorrowedFd<'_>, call: impl FnOnce() -> Os<T>) -> Os<T> {
1653    if !blocks(socket) {
1654        return call();
1655    }
1656    rustix::io::ioctl_fionbio(socket, true).map_err(os)?;
1657    let result = call();
1658    rustix::io::ioctl_fionbio(socket, false).map_err(os)?;
1659    result

1662fn send_on_socket(
1663    socket: BorrowedFd<'_>,
1664    bytes: &[u8],
1665    flags: SendFlags,
1666    target: Option<&Target>,
1667) -> Os<i64> {
1668    let flags = flags | rxsock_sys::no_sigpipe_flags(socket);
1669    let count = match target {
1670        None => rustix::net::send(socket, bytes, flags),
1671        Some(Target::Inet(v4) | Target::Unspec(v4)) => {
1672            rustix::net::sendto(socket, bytes, flags, v4)
1673        }
1674        Some(Target::Other)
1675            if rustix::net::sockopt::socket_type(socket) == Ok(rustix::net::SocketType::STREAM) =>
1676        {

1678        }
1679        Some(Target::Other) => return Err(EAFNOSUPPORT),
1680    };

1682        .map(|count| i64::try_from(count).unwrap_or(i64::MAX))
1683        .map_err(os)
1684}

1702fn receive(interp: &mut Interp, args: &[Option<ObjRef>], from: bool) -> Result<ObjRef> {
1703    let fd = socket_arg(interp, args)?;
1704    let var = c_string(interp, args, 2)?;
1705    let len = int(interp, args, 3)?;
1706    let (flags, stem) = if from {
1707        required(args, 4)?;
1708        match args.get(4).copied().flatten() {
1709            Some(value) => {
1710                let flags = c_string(interp, args, 4)?;
1711                (flags, stem_of(interp, value, 5)?)
1712            }
1713            None => (Vec::new(), stem_argument(interp, args, 4)?),
1714        }
1715    } else {
1716        (optional_string(interp, args, 4)?.unwrap_or_default(), None)
1717    };
1718    let Ok(len) = usize::try_from(len) else {
1719        return Err(failed(interp));
1720    };
1721    if from && stem.is_none() {
1722        return Ok(number(interp, 0));
1723    }
1724    let flags = recv_flags(&flags);
1725    // A stream socket that is not connected, a listening one included, fails
1726    // at once with ENOTCONN (r[rxsock.sockets.stream]) instead of waiting for
1727    // data that can never come. Linux's poll reports a socket never connected
1728    // as hung up, which ends its wait at once; Apple's does not, so there it
1729    // is asked here.
1730    let waits = !flags.contains(RecvFlags::OOB)
1731        && with_socket(fd, |socket| {
1732            Ok(blocks(socket) && !unconnected_stream(socket))
1733        })
1734        .unwrap_or(false)
1735        && !any_ready(&[(fd, PollFlags::IN)]);
1736    if waits {
1737        let deadline = socket_deadline(interp, fd, rustix::net::sockopt::Timeout::Recv);
1738        let wait = SocketWait {
1739            polls: vec![(fd, PollFlags::IN)],
1740            deadline,
1741            helper: None,
1742            op: Pending::Recv {
1743                fd,
1744                var,
1745                len,
1746                flags,
1747                from: stem.as_ref().map(|stem| stem.prefix.clone()),
1748            },
1749        };
1750        let kept = [stem.as_ref().map(|stem| stem.object)];
1751        return park(interp, wait, &kept);
1752    }
1753    Ok(finish_recv(interp, fd, &var, len, flags, stem.as_ref()))
1754}

1756fn finish_recv(
1757    interp: &mut Interp,
1758    fd: SocketId,
1759    var: &[u8],
1760    len: usize,
1761    flags: RecvFlags,
1762    from: Option<&Stem>,
1763) -> ObjRef {
1764    let mut buffer = vec![0; len.min(RECV_MOST)];
1765    let received = with_socket(fd, |socket| {
1766        let receive = || match from {
1767            Some(_) => rustix::net::recvfrom(socket, &mut buffer[..], flags)
1768                .map(|(_, count, address)| (count, address))
1769                .map_err(os),
1770            None => rustix::net::recv(socket, &mut buffer[..], flags)
1771                .map(|(_, count)| (count, None))
1772                .map_err(os),
1773        };
1774        #[cfg(windows)]
1775        {
1776            try_without_blocking(socket, receive)
1777        }
1778        #[cfg(unix)]
1779        {
1780            let mut receive = receive;
1781            receive()
1782        }
1783    });
1784    let (result, address) = match received {
1785        Ok((count, address)) => {
1786            buffer.truncate(count.min(buffer.len()));
1787            (Ok(i64::try_from(count).unwrap_or(i64::MAX)), address)
1788        }
1789        Err(errno) => {
1790            buffer.clear();
1791            (Err(errno), None)
1792        }
1793    };
1794    let value = interp.text_built(buffer);
1795    interp.roots.activity_mut().push_temp(value);
1796    interp.set_context_variable(var, value);
1797    if let Some(stem) = from
1798        && result.is_ok()
1799    {
1800        write_socket_address(interp, stem, address);
1801    }
1802    answer(interp, result)
1803}

1872/// `SockSetSockOpt(s, level, option, value)`.
1873pub(crate) fn set_sock_opt(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<ObjRef> {

1875    let fd = socket_arg(interp, args)?;
1876    let level = c_string(interp, args, 2)?;
1877    let option = c_string(interp, args, 3)?;
1878    let value = c_string(interp, args, 4)?;

1881        None => return Ok(answer(interp, Err(ENOPROTOOPT))),

1892            || u16::cast_truncate(atoi(words.next().unwrap_or_default()).cast_unsigned());
1893        (half(), half())
1894    } else {
1895        (0, 0)
1896    };
1897    let result = rxsock_sys::set_option(fd, option, numeric, linger);
1898    Ok(code(interp, result))

1901/// `SockIoctl(s, cmd, arg)`.
1902pub(crate) fn ioctl(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<ObjRef> {
1903    check_arity(args, 3)?;
1904    let fd = socket_arg(interp, args)?;
1905    let command = c_string(interp, args, 2)?;
1906    let value = required(args, 3)?;
1907    if command.eq_ignore_ascii_case(b"FIONBIO") {
1908        let on = match value.decode() {
1909            Decoded::SmallInt(number) => i32::try_from(number).ok(),
1910            _ => {
1911                let text = c_string(interp, args, 3)?;
1912                whole(&text).and_then(|number| i32::try_from(number).ok())
1913            }
1914        };
1915        let Some(on) = on else {
1916            return Err(failed(interp));
1917        };
1918        let result = rxsock_sys::set_nonblocking(fd, on != 0);
1919        return Ok(code(interp, result));
1920    }
1921    if command.eq_ignore_ascii_case(b"FIONREAD") {
1922        let var = c_string(interp, args, 3)?;
1923        let read = rxsock_sys::available_bytes(fd);
1924        let count = *read.as_ref().unwrap_or(&0);
1925        let count = number(interp, i128::from(count));
1926        interp.roots.activity_mut().push_temp(count);
1927        interp.set_context_variable(&var, count);
1928        return Ok(code(interp, read.map(|_| ())));
1929    }
1930    Ok(number(interp, -1))
1931}

1935/// One of `SockSelect`'s sets: its stem and the sockets it lists.
1936#[derive(Clone)]
1937struct Set {
1938    stem: Stem,
1939    sockets: Vec<SocketId>,
1940}

1942/// The set an argument names, `None` where it takes no part: omitted,
1943/// naming no stem, or with a count that is not a whole number. 40.1 for an
1944/// entry that is not a whole number (Deviation 88).
1945fn set_of(interp: &mut Interp, args: &[Option<ObjRef>], position: usize) -> Result<Option<Set>> {
1946    let Some(value) = args.get(position - 1).copied().flatten() else {
1947        return Ok(None);
1948    };
1949    // For example, a name with a tail (`'q.x'`) takes no part, whatever `Q.0`
1950    // or `Q.X0` hold.
1951    let Some(stem) = stem_of(interp, value, position)?.filter(|stem| stem.prefix.is_empty()) else {
1952        return Ok(None);
1953    };
1954    let Some(count) = stem.get(interp, b"0").and_then(|count| whole(&count)) else {
1955        return Ok(None);
1956    };
1957    let mut sockets = Vec::new();
1958    for index in 1..=count.max(0) {
1959        let entry = stem.get(interp, index.to_string().as_bytes());
1960        let Some(fd) = entry.as_deref().and_then(parse_socket) else {
1961            return Err(failed(interp));
1962        };
1963        sockets.push(fd);
1964    }
1965    Ok(Some(Set { stem, sockets }))
1966}

1968const SET_EVENTS: [PollFlags; 3] = [PollFlags::IN, PollFlags::OUT, PollFlags::PRI];
1969
1970/// Whether `select` looks at each entry of `sets`, per set
1971/// ([`crate::host::SelectTable`]): a negative entry or one past the
1972/// descriptor table is never ready and no error, while a closed one inside it
1973/// answers -1 with `EBADF`. Taken once per call, so the polls and the walk of
1974/// their answers agree on every entry.
1975#[cfg(unix)]
1976fn reached(sets: &[Option<Set>; 3]) -> [Vec<bool>; 3] {
1977    let table = crate::host::SelectTable::now();
1978    sets.each_ref().map(|set| {
1979        set.as_ref().map_or_else(Vec::new, |set| {
1980            set.sockets.iter().map(|fd| table.reaches(*fd)).collect()
1981        })
1982    })
1983}
1984
1985#[cfg(windows)]
1986fn reached(sets: &[Option<Set>; 3]) -> [Vec<bool>; 3] {
1987    sets.each_ref().map(|set| {
1988        set.as_ref()
1989            .map_or_else(Vec::new, |set| vec![true; set.sockets.len()])
1990    })
1991}

1993/// The polls for the entries of `sets` that `reached` says `select` looks at.
1994fn select_polls(sets: &[Option<Set>; 3], reached: &[Vec<bool>; 3]) -> Vec<PollRequest> {
1995    sets.iter()
1996        .zip(SET_EVENTS)
1997        .zip(reached)
1998        .filter_map(|((set, events), reached)| set.as_ref().map(|set| (set, events, reached)))
1999        .flat_map(|(set, events, reached)| {
2000            set.sockets
2001                .iter()
2002                .zip(reached)
2003                .filter(|(_, reached)| **reached)
2004                .map(move |(fd, _)| (*fd, events))
2005        })
2006        .collect()
2007}

2009/// `SockSelect([reads], [writes], [excepts] [, timeout])`.
2010pub(crate) fn select(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<ObjRef> {
2011    check_arity(args, 4)?;
2012    let reads = set_of(interp, args, 1)?;
2013    let writes = set_of(interp, args, 2)?;
2014    let excepts = set_of(interp, args, 3)?;
2015    let timeout = match args.get(3).copied().flatten() {
2016        Some(_) => Some(double_argument(interp, args, 4)?),
2017        None => None,
2018    };
2019    let sets = [reads, writes, excepts];
2020    let polls = select_polls(&sets, &reached(&sets));
2021    let ready = any_ready(&polls);
2022    if ready || timeout.is_some_and(|seconds| seconds <= 0.0) {
2023        return Ok(finish_select(interp, &sets));
2024    }
2025    let deadline = timeout.map(|seconds| {
2026        interp.now() + Duration::try_from_secs_f64(seconds).unwrap_or(Duration::MAX / 4)
2027    });
2028    let kept: Vec<Option<ObjRef>> = sets
2029        .iter()
2030        .map(|set| set.as_ref().map(|set| set.stem.object))
2031        .collect();
2032    let wait = SocketWait {
2033        polls,
2034        deadline,
2035        helper: None,
2036        op: Pending::Select { sets },
2037    };
2038    park(interp, wait, &kept)
2039}

2041fn finish_select(interp: &mut Interp, sets: &[Option<Set>; 3]) -> ObjRef {
2042    let reached = reached(sets);
2043    let polls = select_polls(sets, &reached);
2044    let Some(revents) = poll_now(&polls) else {
2045        return answer(interp, Err(EBADF));
2046    };
2047    let mut revents = revents.into_iter();
2048    let mut count = 0;
2049    let mut kept: Vec<Option<Vec<SocketId>>> = Vec::new();
2050    for ((set, events), reached) in sets.iter().zip(SET_EVENTS).zip(&reached) {
2051        let Some(set) = set else {
2052            kept.push(None);
2053            continue;
2054        };
2055        let mut ready = Vec::new();
2056        let mut counted = Vec::new();
2057        for (fd, reached) in set.sockets.iter().copied().zip(reached) {
2058            if !reached {
2059                continue;
2060            }
2061            let revents = revents.next().unwrap_or_else(PollFlags::empty);
2062            if answers(events, revents) {
2063                ready.push(fd);
2064                if !counted.contains(&fd) {
2065                    counted.push(fd);
2066                }
2067            }
2068        }
2069        count += counted.len();
2070        kept.push(Some(ready));
2071    }
2072    if count > 0 {
2073        for (set, ready) in sets.iter().zip(kept) {
2074            let (Some(set), Some(ready)) = (set, ready) else {
2075                continue;
2076            };
2077            let total = i64::try_from(ready.len()).unwrap_or(i64::MAX);
2078            set.stem.set_number(interp, b"0", total);
2079            for (index, fd) in ready.into_iter().enumerate() {
2080                set.stem
2081                    .set_number(interp, (index + 1).to_string().as_bytes(), i128::from(fd));
2082            }
2083        }
2084    }
2085    answer(interp, Ok(i64::try_from(count).unwrap_or(i64::MAX)))
2086}
