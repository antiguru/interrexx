1//! The process's one connection to the API service, opened on first need and kept for the
2//! process's life, so a registration it makes stays owned by it.
3
4use std::path::{Path, PathBuf};
5use std::sync::OnceLock;
6
7use interrexx_rxapi::client::{Session, connect};
8use interrexx_rxapi::message::code::NO_SERVICE;
9use interrexx_rxapi::message::wire::PROTOCOL_VERSION;
10use interrexx_rxapi::message::{Reply, Request};
11#[cfg(unix)]
12use interrexx_rxapi::transport::unix::socket_path;
13use interrexx_rxapi::transport::{CancelToken, Transport};
14use interrexx_rxapi::transport::{PlatformClient, service_dir};

16pub(crate) use interrexx_rxapi::client::{SESSION_VARIABLE, session_value};

18/// The connection and the session queue it serves.
19pub(crate) struct ApiClient {
20    transport: PlatformClient,
21    session: u64,
22}
23
24/// The connection, or the failure every later call answers.
25static CLIENT: OnceLock<Result<ApiClient, u64>> = OnceLock::new();
26
27/// The service directory and the daemon to start in it, where the process named them.
28static SERVICE: OnceLock<(PathBuf, PathBuf)> = OnceLock::new();
29
30/// Points this process's connection at the service in `dir`, started from `daemon`, instead of
31/// the one `REXX_API_DIR` or the defaults name: for an embedding, such as a test harness, whose
32/// programs must not reach the user's service. `false` where the process already chose one.
33pub fn use_api_service(dir: &Path, daemon: &Path) -> bool {
34    CLIENT.get().is_none()
35        && SERVICE
36            .set((dir.to_path_buf(), daemon.to_path_buf()))
37            .is_ok()
38}
39
40/// The service directory this process uses.
41fn chosen_dir() -> PathBuf {
42    match SERVICE.get() {
43        Some((dir, _)) => dir.clone(),
44        None => service_dir(),
45    }
46}
47
48/// This process's session queue.
49pub(crate) fn session() -> Session {
50    static SESSION: OnceLock<Session> = OnceLock::new();
51    *SESSION.get_or_init(Session::of_process)
52}
53
54pub(crate) fn has_shared_session() -> bool {
55    session().inherited()
56}
57
58/// This process's session handle.
59pub(crate) fn session_handle() -> u64 {
60    session().handle()
61}
62
63/// The process's client, connecting and starting the service on first use. A failure is the
64/// API code, and stays the answer for the process's life.
65pub(crate) fn client() -> Result<&'static ApiClient, u64> {
66    CLIENT
67        .get_or_init(|| connect_on_host(connect_client))
68        .as_ref()
69        .map_err(|status| *status)
70}
71
72fn connect_on_host<T>(connect: impl FnOnce() -> T) -> T {
73    #[cfg(unix)]
74    {
75        crate::signal::blocked(connect)
76    }
77    #[cfg(windows)]
78    {
79        connect()
80    }
81}
82
83/// A new connection, attached to this process's session queue.
84fn connect_client() -> Result<ApiClient, u64> {
85    if !connections_allowed() {
86        return Err(NO_SERVICE);
87    }
88    let transport = match SERVICE.get() {
89        Some((dir, daemon)) => {
90            PlatformClient::connect_or_start_with(dir, daemon).map_err(|error| {
91                interrexx_rxapi::client::report_refusal(&error);
92                error.status()
93            })?
94        }
95        None => connect()?,
96    };
97    let client = ApiClient {
98        transport,
99        session: session_handle(),
100    };
101    client.call(Request::CreateSessionQueue {
102        session: client.session,
103    })?;
104    Ok(client)
105}
106
107/// The process's client where it is connected or a service is running to connect to, starting
108/// none. An absent service is not remembered as a failure.
109pub(crate) fn running_client() -> Option<&'static ApiClient> {
110    if let Some(client) = CLIENT.get() {
111        return client.as_ref().ok();
112    }
113    if !connections_allowed() {
114        return None;
115    }
116    let dir = chosen_dir();
117    #[cfg(unix)]
118    if !socket_path(&dir, PROTOCOL_VERSION).exists() {
119        return None;
120    }
121    let transport =
122        connect_on_host(|| PlatformClient::connect_version(&dir, PROTOCOL_VERSION)).ok()?;
123    let client = ApiClient {
124        transport,
125        session: session_handle(),
126    };
127    client
128        .call(Request::CreateSessionQueue {
129            session: client.session,
130        })
131        .ok()?;
132    let _ = CLIENT.set(Ok(client));
133    CLIENT.get()?.as_ref().ok()
134}
135
136/// Whether this process may reach a service: a unit test may only through a private
137/// directory, so none reaches the user's own.
138fn connections_allowed() -> bool {
139    !cfg!(test) || SERVICE.get().is_some() || std::env::var_os("REXX_API_DIR").is_some()
140}
141
142impl ApiClient {
143    pub(crate) fn session(&self) -> u64 {
144        self.session
145    }
146
147    /// One request's reply, or `RXAPI_NORXAPI` when the service is gone.
148    pub(crate) fn call(&self, request: Request) -> Result<Reply, u64> {
149        self.transport.call(request).map_err(|_| NO_SERVICE)
150    }
151
152    /// As [`ApiClient::call`], ended early by `token`.
153    pub(crate) fn call_cancellable(
154        &self,
155        request: Request,
156        token: &CancelToken,
157    ) -> Result<Reply, u64> {
158        self.transport
159            .call_cancellable(request, token)
160            .map_err(|_| NO_SERVICE)
161    }
162
163    /// Ends the call `token` started, from any thread.
164    pub(crate) fn cancel(&self, token: &CancelToken) {
165        let _ = self.transport.cancel(token);
166    }
167}
