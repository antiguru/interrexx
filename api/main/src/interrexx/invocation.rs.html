1//! What a command line supplies to a top-level program, as
2//! [`run_program`](crate::run_program)'s third parameter.
3//! ```text
4//! rexx p.rex            ->  arg() = 0, arg(1) omitted
5//! rexx p.rex a b c      ->  arg() = 1, arg(1) = "a b c"
6//! rexx p.rex "a b c"    ->  arg() = 1, arg(1) = "a b c"
7//! rexx p.rex a,b,c      ->  arg() = 1, arg(1) = "a,b,c"   (commas do not split)
8//! rexx p.rex ""         ->  arg() = 1, arg(1) = ""        (empty, but PRESENT)
9//! ```
10//! ```text
11//! rexx p.rex     ->  <P>   -- nothing bound, so `p` reads as its own name
12//! rexx p.rex ""  ->  <>    -- bound to the null string
13//! ```
14//! ```text
15//! ""      ""        ->  ""      (0)   -- not " " (1)
16//! ""      "x"       ->  "x"     (1)   -- not " x" (2)
17//! "x"     ""        ->  "x "    (2)
18//! "" "" "x"         ->  "x"     (1)
19//! "x" "" "y"        ->  "x  y"  (4)
20//! " " "x"           ->  "  x"   (3)   -- a blank word IS non-empty
21//! " x"              ->  " x"    (2)   -- nothing is stripped
22//! "a  b"  "c"       ->  "a  b c" (6)  -- internal spacing survives
23//! ```

29/// What a command line supplied to the program being run.
30pub struct Invocation {
31    /// The one argument string, or `None` when the command line supplied no
32    /// argument. See the module doc for why the two are different states and
33    /// how each is observable.
34    argument: Option<Vec<u8>>,
35    /// Where `.input` reads its lines from.
36    input: ProgramInput,
37    /// How long the run may take before the interpreter abandons it, or
38    /// `None` for no bound at all -- which is what every shipped caller
39    /// passes. See [`Invocation::with_deadline`].
40    deadline: Option<Duration>,
41    /// The most body bytes the heap may hold, or `None` to take
42    /// [`crate::HEAP_LIMIT_ENV`]'s value. See [`Invocation::with_heap_limit`].
43    heap_limit: Option<usize>,
44    /// The directory the interpreter resolves relative paths against, or
45    /// `None` to take the process's. A harness running interpreters on threads
46    /// sets it, because the process's own directory is shared.
47    directory: Option<std::path::PathBuf>,
48    /// The environment the interpreter reads and writes, or `None` to copy the
49    /// process's at start.
50    environment: Option<Vec<(Vec<u8>, Vec<u8>)>>,
51    /// Whether each standard descriptor -- input, output, error, in that
52    /// order -- is transient: a terminal, a pipe or a character device rather
53    /// than a regular file.
54    ///
55    /// **A property of the embedding, not of this process.** The interpreter
56    /// never writes to a descriptor -- `SAY` appends to a buffer the caller
57    /// drains -- so the only honest source for this is whoever owns the
58    /// descriptors. Positioning a transient stream is 93.958 and
59    /// `QUERY STREAMTYPE` answers `TRANSIENT`, both Rexx-visible, so reading
60    /// the host's own `/proc/self/fd` here would make one program answer
61    /// differently under a binary, a test harness and a library embedding.
62    /// Defaults to all transient, which is what a terminal is.
63    standard_transient: [bool; 3],
64    /// Where the buffered output goes when the program is about to block on a
65    /// live standard input, or `None` for a caller that reads the whole of it
66    /// out of [`Outcome`](crate::Outcome) at the end.
67    sinks: Option<Sinks>,
68    /// The command-line words as separate strings, which `.SYSCARGS` answers.
69    ///
70    /// **Kept beside `argument` rather than derived from it**, because the
71    /// join is not invertible: measured, `1 "2 3" 4` gives one argument string
72    /// `1 2 3 4` and a `.SYSCARGS` of `1`, `2 3`, `4`, and nothing in the
73    /// string says where the second word ended. The two are independent
74    /// afterwards -- appending to `.SYSCARGS` leaves `ARG(1)` unchanged.
75    words: Vec<Vec<u8>>,
76    /// The index `.SYSCARGS` holds its first word at, or `None` for a run
77    /// whose `.local` has no `SYSCARGS` entry.
78    words_base: Option<usize>,
79    /// The `::OPTIONS` directive `.Package~defaultOptions` is sent before the
80    /// program is installed, and the override count sent after it.
81    default_options: Option<(Vec<u8>, i64)>,
82    /// The register frame arena's block size.
83    frame_block: FrameBlock,
84    switch_mode: Option<SwitchMode>,
85    /// What an embedding's `RexxStart` passes in place of the command line's
86    /// one string: each argument, `None` an omitted one.
87    arguments: Option<Vec<Option<Vec<u8>>>>,
88    /// How the program is called, `PARSE SOURCE`'s second word.
89    start: StartType,
90    /// The initial `ADDRESS` environment, or `None` for the platform's.
91    address: Option<Vec<u8>>,
92    /// The system exits an embedding's `RexxStart` names: each a function code and the name
93    /// of a registered exit handler.
94    exits: Vec<(std::ffi::c_int, Vec<u8>)>,
95    /// The thread an embedding's `RexxStart` was called on, which `RexxSetHalt` and
96    /// `RexxSetTrace` name to reach this run.
97    classic_thread: Option<usize>,
98    /// Whether a host outside the interpreter runs the program through the API, which turns off
99    /// the refusal of unending waits (Deviation 20).
100    hosted: bool,
101}

112/// Where a test sets `SLICE` itself in place of the timer (the
113/// deterministic switch mode).
114#[derive(Clone, Debug, PartialEq)]
115pub enum SwitchMode {
116    /// At the program's `k`-th clause, counting from 1.

124/// What an embedding that owns real descriptors hands the interpreter's two
125/// output buffers to as the program runs.
126///
127/// **The interpreter never writes to a descriptor**: it appends to two
128/// buffers, and without sinks nothing is written until the caller drains
129/// [`Outcome`](crate::Outcome). With them, each buffer is handed over before
130/// a read, before the thread idles, before a command starts, and as
131/// [`Sinks::stdout_terminal`] says between those, which is the buffering of C
132/// stdio and Python 3 (Deviation 40). The rest stays for the `Outcome`.
133pub struct Sinks {
134    /// Takes what `SAY` and `.STDOUT` have written since the last handover,
135    /// answering the error that kept it from the descriptor, which the
136    /// stream writes to standard output then answer.
137    ///
138    /// **`Send`, because the interpreter runs on a thread of its own**
139    /// (`on_interpreter_thread`), which is where the sized stack it needs
140    /// comes from.
141    pub stdout: Box<dyn FnMut(&[u8]) -> std::io::Result<()> + Send>,
142    /// The same for the trace sink, which is `.STDERR`'s descriptor too.
143    pub stderr: Box<dyn FnMut(&[u8]) -> std::io::Result<()> + Send>,
144    /// Whether `stdout` writes to a terminal, which is handed each line as it
145    /// ends. Any other `stdout` is handed its buffer once it passes 64 KiB,
146    /// and `stderr` is handed each line whatever it writes to.
147    pub stdout_terminal: bool,
148    /// Whether `stdout` is handed each line as it ends whatever it writes to,
149    /// as a terminal is: what [`unbuffered_requested`](crate::unbuffered_requested)
150    /// answers, or `rexx -u` (r[instr.say.deviation.40]).
151    pub unbuffered: bool,
152    /// Whether `stdout` and `stderr` write this process's own standard output
153    /// and error. A command whose output or error is not redirected then writes
154    /// that descriptor itself, once the buffers have been handed over;
155    /// otherwise its writes are read back and handed over with the
156    /// interpreter's own.
157    pub own_descriptors: bool,
158}

160/// Where `.input` -- the position `PULL`, `PARSE PULL` and `PARSE LINEIN` all
161/// advance -- reads its lines from.
162pub enum ProgramInput {
163    /// Nothing to read: every line read answers the null string.
164    Nothing,
165    /// The process's own standard input, read one line at a time.
166    Stdin,
167    /// A fixed buffer of bytes, read one line at a time -- what a test uses to
168    /// supply input deterministically.
169    Bytes(Vec<u8>),
170}

409/// The one argument string a list of command-line words becomes, or `None`
410/// when the list is empty.
411pub fn join_command_line<I, W>(words: I) -> Invocation
412where
413    I: IntoIterator<Item = W>,
414    W: AsRef<[u8]>,
415{
416    let mut joined: Option<Vec<u8>> = None;
417    let mut kept: Vec<Vec<u8>> = Vec::new();
418    for word in words {
419        kept.push(word.as_ref().to_vec());
420        let buffer = joined.get_or_insert_with(Vec::new);
421        // The blank goes in only when something is already accumulated, which
422        // is what makes a leading empty word contribute neither text nor
423        // separator. `buffer` having been created by `get_or_insert_with`
424        // above is what keeps the *presence* of an argument independent of
425        // whether any word had bytes in it.
426        if !buffer.is_empty() {
427            buffer.push(b' ');
428        }
429        buffer.extend_from_slice(word.as_ref());
430    }
431    Invocation {
432        argument: joined,
433        words: kept,
434        ..Invocation::none()
435    }
436}

439mod tests {

442    /// The module doc's own measured table, row for row.
443    #[test]
444    fn command_line_words_join_the_way_the_oracle_joins_them() {
445        let cases: &[(&[&str], Option<&[u8]>)] = &[
446            (&[], None),
447            (&[""], Some(b"")),
448            (&["", ""], Some(b"")),
449            (&["", "x"], Some(b"x")),
450            (&["x", ""], Some(b"x ")),
451            (&["", "", "x"], Some(b"x")),
452            (&["x", "", "y"], Some(b"x  y")),
453            (&[" ", "x"], Some(b"  x")),
454            (&[" x"], Some(b" x")),
455            (&["x "], Some(b"x ")),
456            (&["a", "b", "c"], Some(b"a b c")),
457            (&["a b c"], Some(b"a b c")),
458            (&["a  b", "c"], Some(b"a  b c")),
459            (&["a,b,c"], Some(b"a,b,c")),
460            (&["x ", "y"], Some(b"x  y")),
461        ];
462        for (words, expected) in cases {
463            let joined = join_command_line(*words).into_parts().argument;
464            assert_eq!(
465                joined.as_deref(),
466                *expected,
467                "joining {words:?} gave {:?}",
468                joined.as_ref().map(|b| String::from_utf8_lossy(b))
469            );
470        }
471    }

473    /// The absent/empty split, on its own, because it is the one this type
474    /// exists for and the table above could satisfy it by accident.
475    #[test]
476    fn no_words_is_absent_and_one_empty_word_is_present() {
477        assert!(
478            join_command_line(Vec::<&str>::new())
479                .into_parts()
480                .argument
481                .is_none()
482        );
483        assert_eq!(
484            join_command_line([""]).into_parts().argument.as_deref(),
485            Some(&b""[..])
486        );
487        assert!(Invocation::none().into_parts().argument.is_none());
488        assert_eq!(
489            Invocation::with_argument(Vec::new())
490                .into_parts()
491                .argument
492                .as_deref(),
493            Some(&b""[..])
494        );
495    }
