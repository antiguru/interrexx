7/// Where a `TraceObject` went.
8enum Delivered {
9    /// To the `.TRACEOUTPUT` destination, by `LINEOUT`.
10    Sent,
11    /// Nowhere, under `.TraceObject~option` `P`.
12    KeptBack,
13    /// To standard error, as the line.
14    ToStandardError,
15    /// To standard error, as the object's string.
16    ToStandardErrorAs(Vec<u8>),
17}

20    /// What `.local` holds for one of the route names, or `None` when it holds
21    /// nothing.
22    ///
23    /// **Not [`Interp::dot_variable`]**, which is what a `.NAME` in a program
24    /// resolves through: on a miss that answers the name *as text*, and `SAY`
25    /// sent to the string `".OUTPUT"` would be a wrong answer. The routes need
26    /// the plain question -- is there an entry -- which is what
27    /// `directory_lookup` answers, and it is reused rather than reopened so the
28    /// seam still has one read chokepoint. An owed entry answers `None`.
29    ///
30    /// `.local` only: measured, stream resolution and the say output both
31    /// read `.local` and never `.environment`.
32    pub(crate) fn local_route(&mut self, name: &[u8]) -> Result<Option<ObjRef>, Failure> {
33        match self.directory_lookup(&[EnvScope::Local], name, &env_seam::Access::Direct)? {
34            hash::DirectoryEntry::Found(found) => Ok(Some(found)),
35            hash::DirectoryEntry::Owed | hash::DirectoryEntry::Absent => Ok(None),
36        }
37    }

42        self.route_generation = self.route_generation.wrapping_add(1);

45    /// Where `SAY` sends, or `None` to write straight to the buffer.
46    ///
47    /// **The decision is cached, not either half of it**: deciding afresh is a
48    /// probe of `.local` and a walk of the monitors' destination queues, 764
49    /// instructions a line between them, and caching one half would leave most
50    /// of that standing.
51    pub(crate) fn output_route(&mut self) -> Result<Option<ObjRef>, Failure> {
52        if let Some((generation, cached)) = self.output_route
53            && generation == self.route_generation
54        {
55            // A writer this crate forgot to count would otherwise be a wrong
56            // answer no corpus program could see, because none of them
57            // redirect `.OUTPUT`.
58            #[cfg(debug_assertions)]
59            {
60                let fresh = unshared!(self, self.output_route_uncached())?;
61                assert_eq!(
62                    cached, fresh,
63                    "the cached SAY route is stale, so some writer does not bump \
64                     Interp::route_generation"
65                );
66            }
67            return Ok(cached);
68        }
69        let fresh = self.output_route_uncached()?;
70        self.output_route = Some((self.route_generation, fresh));
71        Ok(fresh)
72    }

74    /// [`Interp::output_route`] with no cache in front of it.
75    fn output_route_uncached(&mut self) -> Result<Option<ObjRef>, Failure> {
76        match self.local_route(b"OUTPUT")? {
77            Some(route)
78                if route != ObjRef::NIL && !self.route_ends_at(route, self.bootstrap_stdout) =>
79            {
80                Ok(Some(route))
81            }
82            _ => Ok(None),
83        }

86    /// Whether `route` still ends at `terminal`, following each monitor's
87    /// destination queue head (`Monitor`'s `UNKNOWN` forwards to
88    /// `destination~peek`).
89    ///
90    /// A route that ends at the stream the bundle minted writes the same bytes
91    /// whether it is delivered or written straight to the buffer, and the
92    /// direct write is the one that runs no Rexx. That matters twice over.
93    /// Delivering re-enters the interpreter mid-clause, which perturbs the
94    /// argument stack, `PROCEDURE` permission and a `SELECT`'s own search --
95    /// measured, all three. And it is not cheap: measured, delivering every
96    /// `SAY` costs 16,360 instructions a line, which is `sayloop` at 9.63x.
97    pub(crate) fn route_ends_at(&mut self, route: ObjRef, terminal: Option<ObjRef>) -> bool {
98        let Some(terminal) = terminal else {
99            return false;
100        };
101        let Some(monitor_class) = self.rexx_package_class(b"MONITOR") else {
102            return false;
103        };
104        let Some(array_class) = self.classes().lookup("Array") else {
105            return false;
106        };
107        let mut at = route;
108        // The chain the mint builds is `TRACEOUTPUT` over `ERROR` over
109        // `STDERR`, and a program may leave it shorter or longer; the bound
110        // is what keeps a cycle from spinning here.
111        for _ in 0..8 {
112            if at == terminal {
113                return true;
114            }
115            let Some(queue) = self.pool_entry(at, monitor_class, b"DESTINATION") else {
116                return false;
117            };
118            let store = match self.array_slots(queue) {
119                Some(_) => queue,
120                None => match self.pool_entry(queue, array_class, b"ITEMS") {
121                    Some(store) => store,
122                    None => return false,
123                },
124            };
125            let Some(Some(head)) = self.array_slots(store).and_then(<[_]>::first).copied() else {
126                return false;
127            };
128            at = head;
129        }
130        false
131    }

133    /// What `name` holds in `owner`'s pool for `scope`, or `None`.
134    fn pool_entry(&self, owner: ObjRef, scope: ObjRef, name: &[u8]) -> Option<ObjRef> {
135        match self.heap.get(owner).map(|object| &object.body) {
136            Some(Body::Instance { pools, .. }) => pools.get(scope, name),

141    /// One finished trace line, delivered to `.TRACEOUTPUT` as a `TraceObject`.
142    ///
143    /// `start` is where the line begins in the buffer: a `.STDERR` `CHAROUT`
144    /// leaves an unterminated fragment there, so the line cannot be found by
145    /// scanning back for a newline. A route that cannot take the line leaves it
146    /// on stdout, which is `corpus/oracle-crashes.txt` entry 11's licensed
147    /// answer for a route the reference run dies on.
148    ///
149    /// Code that shows no source traces nothing through here
150    /// ([`Interp::untraceable_code`]).
151    // Inlined into `loop_advance`, it cost sayloop +0.31% against +0.14%.
152    #[inline(never)]
153    pub(crate) fn route_trace_line(&mut self, start: usize) {
154        if self.untraceable_code() {
155            self.trace.truncate(start);
156            return;
157        }
158        self.route_trace_line_as(start, false);
159    }

161    /// [`Interp::route_trace_line`] for a `>I>` (`entry`) or `<I<` line, whose
162    /// `TraceObject` names the caller's frame for `>I>`. Standard error is
163    /// handed over after it. Not gated on the source: these are written for a
164    /// restored program too. Answers whether delivery exhausted the stack.
165    pub(crate) fn route_trace_line_as(&mut self, start: usize, entry: bool) -> bool {
166        let exhausted = self.deliver_trace_line_as(start, entry);
167        self.hand_over_stderr();
168        exhausted
169    }

171    /// Trace output: every line becomes a `TraceObject`, which is sent to
172    /// `.TRACEOUTPUT`, writes to standard error where that is the error
173    /// stream, or keeps back under `.TraceObject~option` `P`. Without the
174    /// class the line goes to standard error as it is.
175    fn deliver_trace_line_as(&mut self, start: usize, entry: bool) -> bool {
176        let variable = self.traced_variable.take();
177        if self.activity.routing_trace || self.activity.delivering {
178            self.trace.truncate(start);
179            return false;
180        }
181        if self.trace_failure_queued() {
182            self.trace.truncate(start);
183            return false;
184        }
185        let Some(class) = self.rexx_package_class(b"TRACEOBJECT") else {
186            return false;
187        };
188        // Only the line end the formatter appended: measured, a traced clause
189        // ending in `'0a'x` is delivered with it.
190        let mut line = self.trace.split_off(start);
191        let ended = line.last() == Some(&b'\n');
192        if ended {
193            line.pop();
194        }
195        self.activity.routing_trace = true;
196        self.deliveries.push(crate::activity::Delivery::default());
197        // **The nested-send protocol, not a bare send**, and around the whole
198        // delivery because `NEW`, each `PUT` and the `LINEOUT` all run Rexx
199        // clauses. A clause boundary clears the shared argument stack, so
200        // delivering on the caller's own stack wipes the arguments an
201        // enclosing call has already pushed: measured, `zz = length('abcd')`
202        // under `trace i` refuses with `call_op_off_its_node`, where a call
203        // with no arguments is untouched.
204        // The permission a traced clause holds for `PROCEDURE` is the
205        // activity's, and the clauses the delivery runs spend it: measured,
206        // `p: procedure` under `trace i` with a Rexx destination was 17.1.
207        let permitted = (
208            self.activity.procedure_permitted,
209            self.activity.region_procedure_permitted,
210        );
211        // A failure the delivery leaves no trace of must not leave its levels
212        // either: they would join the report the line may be part of.
213        let temps = self.roots.activity_mut().push_frame();
214        let outer = self.copy_failure_levels();
215        let (values, mark) = self.take_value_buffer();
216        let delivered = pinned!(
217            self,
218            crate::pinning::PinKind::TraceWrapper,
219            self.deliver_trace_line(class, &line, entry, variable)
220        );
221        self.give_value_buffer(values, mark);
222        (
223            self.activity.procedure_permitted,
224            self.activity.region_procedure_permitted,
225        ) = permitted;
226        self.activity.routing_trace = false;
227        self.deliveries.pop();
228        let to_standard_error = delivered.is_err() && self.routes_to_standard_error();
229        let queued =
230            delivered.is_err() && !to_standard_error && self.running_activation().is_some();
231        if !queued {
232            self.restore_failure_levels(outer);
233            self.roots.activity_mut().pop_frame(temps);
234        }
235        match delivered {
236            Ok(Delivered::Sent | Delivered::KeptBack) => {}
237            Ok(Delivered::ToStandardError) => {
238                self.trace.extend_from_slice(&line);
239                if ended {
240                    self.trace.push(b'\n');
241                }
242            }
243            Ok(Delivered::ToStandardErrorAs(shown)) => {
244                self.trace.extend_from_slice(&shown);
245                if ended {
246                    self.trace.push(b'\n');
247                }
248            }
249            // A failure before the line reached a destination of the
250            // program's own, such as making the `TraceObject` with the stack
251            // exhausted, leaves the line to the error stream.
252            Err(_) if to_standard_error => {
253                self.trace.extend_from_slice(&line);
254                if ended {
255                    self.trace.push(b'\n');
256                }
257            }
258            Err(failure) => {
259                let exhausted =
260                    matches!(&failure, Failure::Raised(raised) if raised.delivery.stack_exhausted);
261                self.queue_trace_failure(failure, &line);
262                return exhausted;
263            }
264        }
265        false
266    }

278    /// Whether a trace line's failed delivery is waiting to be raised in the
279    /// running activation, which traces nothing more until it is.
280    pub(crate) fn trace_failure_queued(&self) -> bool {
281        let Some(running) = self.running_activation() else {
282            return false;
283        };
284        let here = running.id;
285        self.activity
286            .pending_traps
287            .iter()
288            .any(|pending| pending.trace_failure.is_some() && pending.activation == here)
289    }

291    /// The failure [`Interp::queue_trace_failure`] queued for the running
292    /// activation, taken off the queue.
293    #[cold]
294    #[inline(never)]
295    pub(crate) fn take_trace_failure(&mut self) -> Option<Failure> {
296        let here = self.running_activation()?.id;
297        let at =
298            self.activity.pending_traps.iter().position(|pending| {
299                pending.trace_failure.is_some() && pending.activation == here
300            })?;
301        self.activity
302            .pending_traps
303            .remove(at)
304            .and_then(|pending| pending.trace_failure)
305    }

307    /// The failed `LINEOUT` of a trace delivery: a condition the destination
308    /// raised leaves the line on stdout, as ooRexx does since r13424 (SF
309    /// #2149), and the failure is raised in the traced activation at its
310    /// clause's end; a refusal of this crate's writes no line. With no activation running the
311    /// line is all there is, as `corpus/oracle-crashes.txt` entry 11 licenses
312    /// (oracle defect).
313    /// A refusal is never dropped: an activation that ends before a clause end
314    /// raises it passes it to its caller ([`Interp::carry_trace_refusals`]).
315    #[cold]
316    #[inline(never)]
317    fn queue_trace_failure(&mut self, failure: Failure, line: &[u8]) {
318        let refusal = !matches!(failure, Failure::Raised(_));
319        let Some(running) = self.running_activation() else {
320            self.write_out_line(line);
321            if refusal {
322                self.trace_refusal.get_or_insert(failure);
323            }
324            return;
325        };
326        let activation = running.id;
327        if refusal {
328            // So that its end runs `Interp::end_cold`, which carries the
329            // refusal on.
330            self.activation_mut().cold_mut();
331        } else {
332            self.write_out_line(line);
333        }
334        self.activity.pending_traps.push_back(crate::PendingTrap {
335            condition: b"SYNTAX".as_slice().into(),
336            rc: None,
337            description: None,
338            object: None,
339            activation,
340            queued_during_delivery: true,
341            request: false,
342            fragment_depth: self.activity.fragment_depth,
343            trace_failure: Some(failure),
344        });
345    }

347    /// Moves each refusal [`Interp::queue_trace_failure`] queued for the
348    /// activation now ending onto its caller, which raises it at its own
349    /// clause's end, or with no caller onto the activity, which ends with it
350    /// ([`Interp::trace_refusal`]).
351    #[cold]
352    #[inline(never)]
353    pub(crate) fn carry_trace_refusals(&mut self) {
354        let Some(ending) = self.running_activation().map(|activation| activation.id) else {
355            return;
356        };
357        let caller = self
358            .activity
359            .suspended
360            .last()
361            .map(|activation| activation.id);
362        let depth = self.activity.fragment_depth;
363        let mut index = 0;
364        while index < self.activity.pending_traps.len() {
365            let pending = &mut self.activity.pending_traps[index];
366            let refused = pending.activation == ending
367                && pending
368                    .trace_failure
369                    .as_ref()
370                    .is_some_and(|failure| !matches!(failure, Failure::Raised(_)));
371            if !refused {
372                index += 1;
373                continue;
374            }
375            if let Some(caller) = caller {
376                pending.activation = caller;
377                pending.fragment_depth = depth;
378                index += 1;
379                continue;
380            }
381            let failure = self
382                .activity
383                .pending_traps
384                .remove(index)
385                .and_then(|pending| pending.trace_failure);
386            if let Some(failure) = failure {
387                self.trace_refusal.get_or_insert(failure);
388            }
389        }
390    }

392    /// Each line of an error report, delivered the way a trace line is: one
393    /// `LINEOUT` per line -- three for an uncaught `1/0` -- and not one for the
394    /// report.
395    pub(crate) fn write_trace_report(&mut self, report: &[u8]) {
396        let body = report.strip_suffix(b"\n").unwrap_or(report);
397        for line in body.split(|&byte| byte == b'\n') {
398            self.write_trace_report_line(line);
399        }
400    }

402    /// [`Interp::write_trace_report`] over the report's own lines, each
403    /// delivered whole, with any line end it holds.
404    ///
405    /// **The levels the failure left are cleared first.** The lines are
406    /// already rendered and hold all they report, and nothing reads the levels
407    /// after the report, so delivering a line has no levels left to copy.
408    ///
409    /// If delivery of an exhausted-stack report also exhausts the stack,
410    /// finish the report on stdout (r[env.objects.error.report]).
411    pub(crate) fn write_trace_report_lines(&mut self, lines: &[Vec<u8>], stack_exhausted: bool) {
412        self.clear_failure_levels();
413        let mut direct = false;
414        for line in lines {
415            if direct {
416                self.write_out_line(line);
417            } else {
418                direct = self.write_trace_report_line(line) && stack_exhausted;
419            }
420        }
421    }

423    fn write_trace_report_line(&mut self, line: &[u8]) -> bool {
424        if self.untraceable_code() {
425            return false;
426        }
427        let start = self.trace.len();
428        self.trace.extend_from_slice(line);
429        self.trace.push(b'\n');
430        self.route_trace_line_as(start, false)
431    }

433    /// The `TraceObject` carrying `line` and where it was traced, then its
434    /// delivery.
435    fn deliver_trace_line(
436        &mut self,
437        class: ObjRef,
438        line: &[u8],
439        entry: bool,
440        variable: Option<crate::trace::TracedVariable>,
441    ) -> Result<Delivered, Failure> {
442        let waiting = self.activity.waiting_traced.take();
443        let thread = self.activity_number();
444        let invocation = self.invocation_of(0).unwrap_or(0);
445        let levels = crate::dispatch::context::live_levels(self);
446        let frame = match levels.first() {
447            Some(&level) => Some(self.frame_record(level)?),
448            None => None,
449        };
450        let native = match (entry, levels.first()) {
451            (true, Some(crate::dispatch::context::LiveLevel::Activation(0))) => {
452                self.entering_native_frame()
453            }
454            _ => None,
455        };
456        let caller = match (entry, native, levels.get(1)) {
457            (false, ..) => None,
458            (true, Some(frame), _) => Some(CallerFrame::Of(frame)),
459            (true, None, Some(&level)) => Some(CallerFrame::Of(self.frame_record(level)?)),
460            (true, None, None) => Some(CallerFrame::Spawner),
461        };
462        let method = match (levels.first(), self.frame_at(0)) {
463            (Some(crate::dispatch::context::LiveLevel::Activation(0)), Some(activation))
464                if activation.entry == crate::activation::Entry::Method =>
465            {
466                activation.method_identity.as_ref().map(|identity| {
467                    (
468                        crate::guards::GuardKey {
469                            object: identity.receiver,
470                            scope: identity.scope,
471                        },
472                        activation.flags.guarded(),
473                        activation.flags.reserved(),
474                    )
475                })
476            }
477            _ => None,
478        };
479
480        // **A call tail of the delivery's own**, so that a primitive method
481        // the delivery sends parks its tail above every tail the traced code
482        // parked: a `<I<` is delivered after its call tail is popped, at the
483        // depth where the sender's primitive `NEW` waits.
484        self.activity.call_tails.push(crate::run::CallTail::method(
485            crate::CallContext::default(),
486            self.save_clause_state(),
487            0,
488            None,
489        ));
490        let delivered = (|| -> Result<Delivered, Failure> {
491            let caller_ref = self.caller();
492            let Some(object) = self.send_message(class, b"NEW", None, &[], caller_ref)? else {
493                return Err(Failure::Raised(Box::new(crate::error::Raised::syntax(
494                    97,
495                    1,
496                    vec![b"TraceObject".to_vec()],
497                ))));
498            };
499            self.roots.activity_mut().push_temp(object);
500            let traced = self.text(line);
501            self.trace_object_put(object, b"TRACELINE", traced)?;
502            let interpreter = self.counted(1);
503            self.trace_object_put(object, b"INTERPRETER", interpreter)?;
504            let thread = self.counted(usize::cast_from(thread));
505            self.trace_object_put(object, b"THREAD", thread)?;
506            let invocation = self.counted(usize::cast_from(invocation));
507            self.trace_object_put(object, b"INVOCATION", invocation)?;
508            let stack_frame = match frame {
509                Some(frame) => self.frame_table(Some(frame))?,
510                None => ObjRef::NIL,
511            };
512            self.trace_object_put(object, b"STACKFRAME", stack_frame)?;
513            if let Some(caller) = caller {
514                let table = match caller {
515                    CallerFrame::Of(frame) => self.frame_table(Some(frame))?,
516                    CallerFrame::Spawner => self.spawner_table()?,
517                };
518                self.trace_object_put(object, b"CALLERSTACKFRAME", table)?;
519            }
520            if let Some((key, guarded, reserved)) = method {
521                let pool = self.activities.guards.pool_number(key);
522                let pool = self.counted(usize::cast_from(pool));
523                self.trace_object_put(object, b"ATTRIBUTEPOOL", pool)?;
524                self.trace_object_put(object, b"ISGUARDED", crate::eval::logical(guarded))?;
525                let count = self.guard_count(key);
526                let count = self.counted(usize::cast_from(count));
527                self.trace_object_put(object, b"SCOPELOCKCOUNT", count)?;
528                self.trace_object_put(object, b"HASSCOPELOCK", crate::eval::logical(reserved))?;
529                self.trace_object_put(object, b"RECEIVER", key.object)?;
530                if let Some(waiting) = waiting {
531                    self.trace_object_put(object, b"ISWAITING", crate::eval::logical(waiting))?;
532                }
533            }
534            if let Some(variable) = variable {
535                self.trace_object_variable(object, variable)?;
536            }
537            let caller_ref = self.caller();
538            let notify = self
539                .send_message(class, b"NOTIFY", None, &[], caller_ref)?
540                .unwrap_or(ObjRef::NIL);
541            if notify != ObjRef::NIL {
542                let caller_ref = self.caller();
543                self.send_message(notify, b"APPEND", None, &[Some(object)], caller_ref)?;
544            }
545            // The I/O exit's first pass.
546            if self.exits.is_some() && self.trace_exit(line)? {
547                return Ok(Delivered::KeptBack);
548            }
549            // The option the object was made under.
550            let index = self.text(b"OPTION");
551            self.roots.activity_mut().push_temp(index);
552            let caller_ref = self.caller();
553            let option = self
554                .send_message(object, b"[]", None, &[Some(index)], caller_ref)?
555                .unwrap_or(ObjRef::NIL);
556            let letter = if option == ObjRef::NIL {
557                b'N'
558            } else {
559                self.to_text(option).first().copied().unwrap_or(b'N')
560            };
561            if letter == b'P' {
562                return Ok(Delivered::KeptBack);
563            }
564            if self.routes_to_standard_error() {
565                // Even option N can have a custom MAKESTRING formatter.
566                let caller_ref = self.caller();
567                let shown = self
568                    .send_message(object, b"MAKESTRING", None, &[], caller_ref)?
569                    .unwrap_or(ObjRef::NIL);
570                let shown = self.to_text(shown).to_vec();
571                return Ok(Delivered::ToStandardErrorAs(shown));
572            }
573            let Some(route) = self.local_route(b"TRACEOUTPUT")? else {
574                return Ok(Delivered::ToStandardError);
575            };
576            let floor =
577                Some(self.activity.suspended.len() + usize::from(self.activity.running.is_some()));
578            if let Some(delivery) = self.deliveries.last_mut() {
579                delivery.floor = floor;
580            }
581            self.deliver_on_own_activity(route, object)
582        })();
583        self.activity.call_tails.pop();
584        delivered
585    }

587    /// `route~lineout(object)`, sent on an activity of its own that runs as
588    /// part of the running one (`r[instr.trace.delivery.owner]`), and waited
589    /// for here. The delivery runs before any other activity that is ready,
590    /// so no other line comes between. With no Rust frame of the running
591    /// activity below it, the destination can wait for another activity and
592    /// give up its slices like any other code (`r[instr.trace.delivery.others]`).
593    /// Its failure comes back as it was raised, with the levels it unwound.
594    fn deliver_on_own_activity(
595        &mut self,
596        route: ObjRef,
597        object: ObjRef,
598    ) -> Result<Delivered, Failure> {
599        let Some(class) = self.rexx_package_class(b"MESSAGE") else {
600            return Err(crate::Loud::environment_symbol(b".MESSAGE").into());
601        };
602        let name = self.text(crate::dispatch::LINEOUT);
603        self.roots.activity_mut().push_temp(name);
604        let form = self.text(b"I");
605        self.roots.activity_mut().push_temp(form);
606        let caller_ref = self.caller();
607        let message = self
608            .send_message(
609                class,
610                b"NEW",
611                None,
612                &[Some(route), Some(name), Some(form), Some(object)],
613                caller_ref,
614            )?
615            .unwrap_or(ObjRef::NIL);
616        self.roots.activity_mut().push_temp(message);
617        let me = self.running_activity();
618        let delivery = crate::dispatch::start_trace_delivery(self, message)?;
619        self.activities.delivery_owners.insert(delivery, me);
620        self.activities.handoff = Some(delivery);
621        let caller_ref = self.caller();
622        let waited = self.send_message(message, b"WAIT", None, &[], caller_ref);
623        self.activities.delivery_owners.remove(&delivery);
624        if self.activities.handoff == Some(delivery) {
625            self.activities.handoff = None;
626        }
627        // The delivery's clauses spent the countdown: a pending reserve of
628        // the traced method is served at its next clause.
629        self.clause_countdown = 1;
630        waited?;
631        if let Some(failure) = self.activity.delivery_failure.take() {
632            return Err(failure);
633        }
634        Ok(Delivered::Sent)
635    }

637    /// The `VARIABLE` entry of a `>V>` or `>=>` line's `TraceObject`: a
638    /// `StringTable` of `NAME`, `VALUE` and `ASSIGNMENT`.
639    fn trace_object_variable(
640        &mut self,
641        object: ObjRef,
642        variable: crate::trace::TracedVariable,
643    ) -> Result<(), Failure> {
644        let table = crate::dispatch::hash::new_string_table(self)?;
645        self.roots.activity_mut().push_temp(table);
646        self.trace_object_put(object, b"VARIABLE", table)?;
647        let name = match &variable.name {
648            Some(name) => self.text(name),
649            None => ObjRef::NIL,
650        };
651        self.roots.activity_mut().push_temp(name);
652        crate::dispatch::hash::directory_put(self, table, b"NAME", name)?;
653        let value = variable.value.unwrap_or(ObjRef::NIL);
654        crate::dispatch::hash::directory_put(self, table, b"VALUE", value)?;
655        crate::dispatch::hash::directory_put(
656            self,
657            table,
658            b"ASSIGNMENT",
659            crate::eval::logical(variable.assignment),
660        )
661    }

663    /// `PUT` of `value` under `name` on the `TraceObject` being built. A
664    /// `StringTable` keeps its entries in a bucket table, so the put goes
665    /// through the message; `t[i] = v` sends `t~"[]="(v, i)`, so the value
666    /// leads.
667    fn trace_object_put(
668        &mut self,
669        object: ObjRef,
670        name: &[u8],
671        value: ObjRef,
672    ) -> Result<(), Failure> {
673        self.roots.activity_mut().push_temp(value);
674        let index = self.text(name);
675        self.roots.activity_mut().push_temp(index);
676        let caller = self.caller();
677        self.send_message(object, b"PUT", None, &[Some(value), Some(index)], caller)?;
678        Ok(())
679    }

681    /// The `StackFrame` of `level` and its executable (`.nil` where this
682    /// crate has no object for it), rooted as temporaries.
683    pub(crate) fn frame_record(
684        &mut self,
685        level: crate::dispatch::context::LiveLevel,
686    ) -> Result<(ObjRef, ObjRef), Failure> {
687        let frame = crate::dispatch::context::build_live_frame(self, level)?;
688        self.roots.activity_mut().push_temp(frame);
689        let executable = match level {
690            crate::dispatch::context::LiveLevel::Activation(depth) => {
691                crate::dispatch::context::executable_at(self, depth).unwrap_or(ObjRef::NIL)
692            }
693            _ => ObjRef::NIL,
694        };
695        self.roots.activity_mut().push_temp(executable);
696        Ok((frame, executable))
697    }

699    /// A `StringTable` of `frame`'s entries, empty for none.
700    fn frame_table(&mut self, frame: Option<(ObjRef, ObjRef)>) -> Result<ObjRef, Failure> {
701        let table = crate::dispatch::hash::new_string_table(self)?;
702        self.roots.activity_mut().push_temp(table);
703        let Some((frame, executable)) = frame else {
704            return Ok(table);
705        };
706        for name in [
707            b"ARGUMENTS".as_slice(),
708            b"INVOCATION",
709            b"LINE",
710            b"NAME",
711            b"TRACELINE",
712            b"TYPE",
713        ] {
714            self.frame_entry_into(table, frame, name)?;
715        }
716        crate::dispatch::hash::directory_put(self, table, b"EXECUTABLE", executable)?;
717        self.frame_entry_into(table, frame, b"TARGET")?;
718        Ok(table)
719    }

721    /// What `frame` answers to `name`, put in `table` under `name`.
722    fn frame_entry_into(
723        &mut self,
724        table: ObjRef,
725        frame: ObjRef,
726        name: &[u8],
727    ) -> Result<(), Failure> {
728        let caller = self.caller();
729        let value = self
730            .send_message(frame, name, None, &[], caller)?
731            .unwrap_or(ObjRef::NIL);
732        self.roots.activity_mut().push_temp(value);
733        crate::dispatch::hash::directory_put(self, table, name, value)
734    }

736    /// The table of the frame that started this activity, with the spawner's
737    /// number as its `THREAD`, built once; `.nil` for main.
738    fn spawner_table(&mut self) -> Result<ObjRef, Failure> {
739        match self.activity.spawner {
740            None => Ok(ObjRef::NIL),
741            Some(crate::activity::Spawner::Table(table)) => Ok(table),
742            Some(crate::activity::Spawner::Frame { thread, frame }) => {
743                let table = self.frame_table(frame)?;
744                let thread = self.counted(usize::cast_from(thread));
745                self.roots.activity_mut().push_temp(thread);
746                crate::dispatch::hash::directory_put(self, table, b"THREAD", thread)?;
747                self.activity.spawner = Some(crate::activity::Spawner::Table(table));
748                Ok(table)
749            }
750        }
751    }

753    /// What a spawned activity records of the running one for its
754    /// `CALLERSTACKFRAME`: its number and the frame of the innermost level.
755    pub(crate) fn spawner_of_running(&mut self) -> Result<crate::activity::Spawner, Failure> {
756        let thread = self.activity_number();
757        let frame = match crate::dispatch::context::live_levels(self).first() {
758            Some(&level) => Some(self.frame_record(level)?),
759            None => None,
760        };
761        Ok(crate::activity::Spawner::Frame { thread, frame })
762    }
763}
