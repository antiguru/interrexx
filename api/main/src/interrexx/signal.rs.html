1//! The signal handlers. A process entry point that starts Rexx for a user
2//! installs them once: SIGINT and SIGTERM where the action is the default or
3//! ignored, SIGHUP only where it is the default, never over a handler,
4//! without `SA_RESTART`; SIGPIPE is ignored (`:146-148`). A handler sets
5//! [`PENDING`] and writes one byte to the timer's wake socket, nothing else;
6//! the timer thread serves it. These statics are the signal half of the
7//! timer's wake source.
8
9#![allow(unsafe_code)]
10
11use std::os::fd::RawFd;
12use std::sync::Once;
13use std::sync::atomic::{AtomicBool, AtomicI32, Ordering};

15/// The signals that halt every live interpreter, each with whether it is
16/// installed over an ignored action too.
17const HALTING: [(libc::c_int, bool); 3] = [
18    (libc::SIGINT, true),
19    (libc::SIGTERM, true),
20    (libc::SIGHUP, false),
21];
22
23/// A signal arrived that the timer has not yet served.
24static PENDING: AtomicBool = AtomicBool::new(false);
25
26/// The nonblocking write end of the timer's wake socket, which lives as
27/// long as the process.
28static WAKE: AtomicI32 = AtomicI32::new(-1);
29
30static INSTALL: Once = Once::new();

32/// The calling thread's `pthread_t`, the `thread_id_t` the classic API names threads by.
33pub(crate) fn current_thread() -> usize {
34    // SAFETY: `pthread_self` has no preconditions.
35    let thread = unsafe { libc::pthread_self() };
36    // A `uintptr_t` on Apple's targets, an `unsigned long` elsewhere: as wide
37    // as a pointer either way.
38    #[cfg(target_vendor = "apple")]
39    {
40        thread
41    }
42    #[cfg(not(target_vendor = "apple"))]
43    {
44        usize::try_from(thread).expect("a pthread_t is as wide as a pointer")
45    }
46}

57/// Flushes the C library's output streams, so that what a C program embedding the interpreter
58/// wrote through stdio before a program's output lands before it, also where standard output
59/// is a file and stdio holds it back.
60pub(crate) fn flush_c_streams() {
61    // SAFETY: `fflush(NULL)` flushes every open output stream, each under its own lock.
62    unsafe {
63        libc::fflush(std::ptr::null_mut());
64    }
65}

67/// Installs the handlers, once per process, with `wake` as the socket they
68/// write to.
69pub(crate) fn install(wake: RawFd) {
70    INSTALL.call_once(|| {
71        WAKE.store(wake, Ordering::Release);
72        for (signal, over_ignored) in HALTING {
73            install_where_unset(signal, over_ignored);
74        }
75        // SAFETY: `signal` with `SIG_IGN` installs no code of ours.
76        unsafe {
77            libc::signal(libc::SIGPIPE, libc::SIG_IGN);
78        }
79    });
80}

82/// Installs [`handler`] for `signal` where its action is the default, or
83/// ignored where `over_ignored`.
84fn install_where_unset(signal: libc::c_int, over_ignored: bool) {
85    // SAFETY: both `sigaction` structs are zeroed plain data, which is a
86    // valid value for every field, and each pointer passed is to a live
87    // local or null. The handler installed is async-signal-safe (see its
88    // own note).
89    unsafe {
90        let mut previous: libc::sigaction = std::mem::zeroed();
91        if libc::sigaction(signal, std::ptr::null(), &raw mut previous) != 0 {
92            return;
93        }
94        let unset = previous.sa_sigaction == libc::SIG_DFL
95            || over_ignored && previous.sa_sigaction == libc::SIG_IGN;
96        if !unset {
97            return;
98        }
99        let mut action: libc::sigaction = std::mem::zeroed();
100        action.sa_sigaction = handler_address(handler);
101        libc::sigfillset(&raw mut action.sa_mask);
102        action.sa_flags = 0;
103        libc::sigaction(signal, &raw const action, std::ptr::null_mut());
104    }
105}

107extern "C" fn handler(_: libc::c_int) {
108    PENDING.store(true, Ordering::Release);
109    let wake = WAKE.load(Ordering::Acquire);
110    // SAFETY: async-signal-safe: an atomic load and store, `write(2)` and
111    // `errno`'s own location, which the handler saves and restores for the
112    // code it interrupted. No allocation, no lock, no interpreter state.
113    // `wake` is a nonblocking socket that is never closed, so the write
114    // neither blocks nor reaches a reused descriptor; a full socket already
115    // holds a wake.
116    unsafe {
117        let errno = errno_location();
118        let saved = *errno;
119        libc::write(wake, [0u8].as_ptr().cast(), 1);
120        *errno = saved;
121    }
122}

145/// The halting signals as a set.
146fn halting() -> libc::sigset_t {
147    // SAFETY: the set is initialised by `sigemptyset` before it is added to,
148    // through a pointer to a live local.
149    unsafe {
150        let mut set: libc::sigset_t = std::mem::zeroed();
151        libc::sigemptyset(&raw mut set);
152        for (signal, _) in HALTING {
153            libc::sigaddset(&raw mut set, signal);
154        }
155        set
156    }
157}
158
159/// Changes this thread's mask by `how` with `set`, answering the mask before.
160fn mask(how: libc::c_int, set: &libc::sigset_t) -> libc::sigset_t {
161    // SAFETY: both pointers are to live values, and a zeroed `sigset_t` is a
162    // valid place for the old mask.
163    unsafe {
164        let mut before: libc::sigset_t = std::mem::zeroed();
165        libc::pthread_sigmask(how, set, &raw mut before);
166        before
167    }
168}

170/// Blocks the halting signals in this thread, so the kernel delivers them
171/// only to a thread that takes them.
172pub(crate) fn block() {
173    mask(libc::SIG_BLOCK, &halting());
174}

176/// Unblocks the halting signals in this thread where the handlers are
177/// installed: for the interpreter's own thread, which runs Rexx and drains
178/// what a blocking wait posts, so a handler it takes has run before it
179/// files that post.
180pub(crate) fn receive() {
181    if INSTALL.is_completed() {
182        mask(libc::SIG_UNBLOCK, &halting());
183    }
184}

186/// Runs `wait` with the halting signals unblocked in this thread where the
187/// handlers are installed, and the thread's mask restored after: a signal
188/// then interrupts the wait, and its handler has run before the wait
189/// returns.
190///
191/// The kernel hands a signal sent to the process to one thread that takes
192/// it, which may be another interpreter's: a terminal's Ctrl-C can end the
193/// wait's child before that thread has run the handler, and this
194/// interpreter would then find no signal arrived and run on past the
195/// clause. Blocking and unblocking once more before the restore has the
196/// kernel deliver a signal still pending for the process here, so its
197/// handler has run before the wait returns.
198pub(crate) fn unblocked<R>(wait: impl FnOnce() -> R) -> R {
199    struct Restore(libc::sigset_t);
200    impl Drop for Restore {
201        fn drop(&mut self) {
202            let halting = halting();
203            mask(libc::SIG_BLOCK, &halting);
204            mask(libc::SIG_UNBLOCK, &halting);
205            mask(libc::SIG_SETMASK, &self.0);
206        }
207    }
208    if !INSTALL.is_completed() {
209        return wait();
210    }
211    let _restore = Restore(mask(libc::SIG_UNBLOCK, &halting()));
212    wait()
213}

215/// Runs `start` with the halting signals blocked in this thread and the
216/// thread's mask restored after, so a thread `start` spawns, such as the API
217/// connection's reader, never takes them.
218pub(crate) fn blocked<R>(start: impl FnOnce() -> R) -> R {
219    struct Restore(libc::sigset_t);
220    impl Drop for Restore {
221        fn drop(&mut self) {
222            mask(libc::SIG_SETMASK, &self.0);
223        }
224    }
225    let _restore = Restore(mask(libc::SIG_BLOCK, &halting()));
226    start()
227}

229/// The signal that interrupts the thread running a native call under
230/// simulation: its handler does nothing, so a system
231/// call it interrupts returns `EINTR`, and it halts nothing.
232#[cfg(any(target_os = "linux", target_os = "android"))]
233fn interrupting() -> libc::c_int {
234    libc::SIGRTMIN() + 3
235}
236
237#[cfg(not(any(target_os = "linux", target_os = "android")))]
238fn interrupting() -> libc::c_int {
239    libc::SIGURG
240}

246/// A thread [`Interrupter::interrupt`] signals with [`interrupting`].
247pub(crate) struct Interrupter(libc::pthread_t);
248
249// SAFETY: a `pthread_t` is a handle, valid on any thread while the thread it
250// names lives; an `Interrupter`'s user signals only while that thread waits
251// for it.
252unsafe impl Send for Interrupter {}
253
254impl Interrupter {
255    /// This thread, with the do-nothing handler installed once per process
256    /// where the signal's action is the default.
257    pub(crate) fn here() -> Interrupter {
258        INTERRUPT_INSTALL.call_once(|| {
259            // SAFETY: as `install_where_unset`; the handler does nothing.
260            unsafe {
261                let mut previous: libc::sigaction = std::mem::zeroed();
262                if libc::sigaction(interrupting(), std::ptr::null(), &raw mut previous) != 0
263                    || previous.sa_sigaction != libc::SIG_DFL
264                {
265                    return;
266                }
267                let mut action: libc::sigaction = std::mem::zeroed();
268                action.sa_sigaction = handler_address(interrupted);
269                libc::sigfillset(&raw mut action.sa_mask);
270                action.sa_flags = 0;
271                libc::sigaction(interrupting(), &raw const action, std::ptr::null_mut());
272            }
273        });
274        // SAFETY: `pthread_self` has no precondition.
275        Interrupter(unsafe { libc::pthread_self() })
276    }
277
278    /// Signals the thread, ending a system call it is blocked in.
279    pub(crate) fn interrupt(&self) {
280        // SAFETY: the thread is alive (see the `Send` impl).
281        unsafe {
282            libc::pthread_kill(self.0, interrupting());
283        }
284    }
285}

287/// `signal(number, SIG_IGN)` where `ignore`, else `signal(number, SIG_DFL)`,
288/// for `rxunixsys`'s `SysSignal`: the `errno` of a refusal. It replaces any
289/// action, the halting signals' handlers included, so a SIGINT after
290/// `SysSignal(2, 'I')` is ignored and after `SysSignal(2, 'D')` ends the
291/// process.
292pub(crate) fn set_disposition(number: i32, ignore: bool) -> Result<(), i32> {
293    let action = if ignore { libc::SIG_IGN } else { libc::SIG_DFL };
294    // SAFETY: `SIG_IGN` and `SIG_DFL` install no code; `signal` checks the
295    // number itself.
296    let previous = unsafe { libc::signal(number, action) };
297    if previous == libc::SIG_ERR {
298        return Err(last_errno());
299    }
300    Ok(())
301}
302
303/// `kill(pid, number)` with the raw signal number, the real-time signals
304/// included, for `rxunixsys`'s `SysKill`: the `errno` of a refusal.
305pub(crate) fn send(pid: i32, number: i32) -> Result<(), i32> {
306    // SAFETY: `kill` takes plain integers and has no precondition.
307    if unsafe { libc::kill(pid, number) } != 0 {
308        return Err(last_errno());
309    }
310    Ok(())
311}

313/// The `errno` the call before left.
314fn last_errno() -> i32 {
315    std::io::Error::last_os_error()
316        .raw_os_error()
317        .unwrap_or(libc::EINVAL)
318}

320/// Whether a signal arrived since the last call.
321pub(crate) fn take_pending() -> bool {
322    PENDING
323        .compare_exchange(true, false, Ordering::AcqRel, Ordering::Acquire)
324        .is_ok()
325}

331    /// Running a program installs no handler: only a process entry point

344    /// Set in the process a test runs its case in alone.

347    /// Whether this is the process the test `name` runs its case in: a halt
348    /// reaches every live interpreter in the process. Elsewhere it runs that
349    /// process and asserts it passed.
350    fn alone(name: &str) -> bool {
351        if std::env::var_os(ALONE).is_some() {
352            return true;
353        }
354        let output = std::process::Command::new(std::env::current_exe().expect("this binary"))
355            .args(["--exact", name, "--test-threads=1"])
356            .env(ALONE, "1")
357            .output()
358            .expect("the case's process");
359        let stdout = String::from_utf8_lossy(&output.stdout);
360        assert!(output.status.success(), "{stdout}");
361        assert!(stdout.contains("1 passed"), "{stdout}");
362        false
363    }

365    /// A signal whose handler ran as a command's child ended halts that
366    /// command's clause, before the timer thread posts it. The flag is set
367    /// with no wake written, so the timer thread does not take it.
368    #[test]
369    fn a_signal_pending_at_a_commands_end_halts_it() {
370        if !alone("signal::tests::a_signal_pending_at_a_commands_end_halts_it") {
371            return;
372        }
373        // Off the baton, and redirected, which waits on it.
374        for command in [
375            "address system 'sleep 0.5'",
376            "address system 'sleep 0.5' with output stem o.",
377        ] {
378            let raiser = std::thread::spawn(|| {
379                std::thread::sleep(std::time::Duration::from_millis(200));
380                super::PENDING.store(true, super::Ordering::Release);
381            });
382            let outcome = crate::run_program(
383                "/no/such/dir/t.rex",
384                format!("say 'a'\n{command}\nsay 'b' rc\n").into_bytes(),
385                crate::Invocation::none(),
386            );
387            raiser.join().expect("the raiser");
388            assert_eq!(String::from_utf8_lossy(&outcome.stdout), "a\n", "{command}");
389            assert_eq!(
390                String::from_utf8_lossy(&outcome.stderr),
391                expand(&format!(

393                     Error 4 running /no/such/dir/t.rex line 2:  ⟨4⟩\n\
394                     Error 4.1:  ⟨4.1¦HALT⟩\n"

397        }
