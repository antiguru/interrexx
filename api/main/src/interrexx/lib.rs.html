367/// The least step of a heap limit (see [`heap_limit_step`]), so that a heap
368/// at a tiny limit does not collect at every allocation.
369const HEAP_LIMIT_STEP_FLOOR: usize = 64 << 10;

371/// How far past a heap limit of `limit` the heap may grow before the next
372/// collection, and how much further than the last raise it must grow before
373/// it is raised again: a sixteenth of the limit, and no less than
374/// [`HEAP_LIMIT_STEP_FLOOR`] (r[env.invocation.heap-limit.raise]).
375fn heap_limit_step(limit: usize) -> usize {
376    (limit / 16).max(HEAP_LIMIT_STEP_FLOOR)
377}

449/// How deep evaluation went and how much stack it took to get there.
450#[derive(Copy, Clone, Debug, Default)]
451pub struct StackSpan {
452    /// The deepest `eval` recursion the run reached. Zero if it never
453    /// evaluated an expression at all, and zero also for a run whose only
454    /// values came from ops that do not enter `eval` -- see this type's own doc
455    /// comment.
456    pub max_depth: usize,
457    /// Stack bytes between the first `eval` level and the deepest one, both
458    /// on the chain that reached `max_depth`. Meaningless unless `max_depth`
459    /// is above 1, which is what `bytes_per_frame` checks before dividing.
460    pub bytes: usize,
461}
462
463impl StackSpan {
464    /// Stack bytes one further level of `eval` costs, or `None` when the run
465    /// never recursed and there is nothing to divide by.
466    pub fn bytes_per_frame(self) -> Option<f64> {
467        (self.max_depth > 1)
468            .then(|| f64::cast_lossy(self.bytes) / f64::cast_lossy(self.max_depth - 1))
469    }
470}

1013/// Names an expression form in **bounded** text, for a loud failure to quote.
1014fn form_name(expr: ExprView<'_>) -> String {
1015    let name = match expr {
1016        ExprView::Literal(_) => "a literal",
1017        ExprView::Constant(_) => "a constant symbol",
1018        ExprView::Variable(_) => "a simple variable",
1019        ExprView::Stem(_) => "a stem",
1020        ExprView::Compound(_) => "a compound variable",
1021        ExprView::DotVariable(_) => "an environment symbol",
1022        // The two operator forms name the operator, because "a dyadic
1023        // operator is not implemented" does not tell a reader which one to
1024        // go and implement. Each asks its own operator type for the spelling,
1025        // which is where the canonical bytes live and what the trace line for
1026        // that operator carries.
1027        ExprView::Prefix { op, .. } => {
1028            return format!("the prefix operator `{}`", op.spelling());
1029        }
1030        ExprView::Binary { op, .. } => {
1031            return format!("the operator `{}`", op.spelling());
1032        }
1033        ExprView::Call { .. } => "a function call",
1034        ExprView::QualifiedCall { .. } => "a namespace-qualified call",
1035        ExprView::ClassResolver { .. } => "a namespace-qualified class lookup",
1036        ExprView::Message { .. } => "a message send",
1037        ExprView::List(_) => "a parenthesised list",
1038        ExprView::Logical(_) => "a comma list in a condition",
1039        ExprView::VariableReference { .. } => "a variable reference",
1040    };
1041    name.to_string()
1042}

1109                }),

2743    /// Parses `text`, the program `resolved` names, and runs it as the external routine `name`
2744    /// with `arguments`, merging what it makes public into the caller's package.
2745    fn enter_program_text(
2746        &mut self,
2747        name: &[u8],
2748        resolved: &[u8],
2749        text: Vec<u8>,
2750        arguments: Vec<Option<ObjRef>>,
2751        call_type: CallType,
2752    ) -> Result<Option<ObjRef>, Failure> {
2753        let parsed = Rc::new(self.parse_file(text, resolved)?);
2754        let caller = self
2755            .running_activation()
2756            .map(|activation| activation.program_id);
2757        let program_id = self.load_program(Rc::clone(&parsed));
2758        // Its own path, so the callee's `PARSE SOURCE`, `~package~name` and
2759        // traceback name the file rather than the program that called it.
2760        self.required_paths.insert(program_id, resolved.into());
2761        let address = self.activation().address.clone();
2762        let settings = self.activation().settings.clone();
2763        let saved = std::mem::replace(
2764            &mut self.activity.call_context,
2765            CallContext {
2766                name: Rc::from(name),
2767                arguments: Rc::from(arguments),
2768                receiver: None,
2769            },
2770        );
2771        // Both packages are built before the call, the callee's rooted as a
2772        // temp of this frame: once the callee returns, nothing else reaches
2773        // its package before the merge below reads it.
2774        let merge = match caller {
2775            Some(caller) => {
2776                let into = self.package_object(Package::Program(caller))?;
2777                self.roots.activity_mut().push_temp(into);
2778                Some((into, self.package_object(Package::Program(program_id))?))
2779            }
2780            None => None,
2781        };

2783        self.activity.call_context = saved;
2784        let value = outcome?;
2785        // What the callee made public becomes the caller's, and transitively
2786        // what the callee itself required.
2787        if let Some((into, from)) = merge {
2788            if let Some(value) = value {
2789                self.roots.activity_mut().push_temp(value);
2790            }
2791            self.merge_package(into, from);
2792        }
2793        Ok(value)
2794    }

2889    /// Installs `program`'s directives and runs its main body, for a program
2890    /// already registered under `program_id`.
2891    ///
2892    /// `address` is the environment pair the body starts with, `None` for one
2893    /// that starts at the platform default: a file reached by a **call**
2894    /// inherits its caller's, and measured, nothing else does.
2895    /// `caller_settings` is what `::OPTIONS NUMERIC INHERIT` inherits from,
2896    /// `None` where there is no call site above the body.
2897    fn run_loaded(
2898        &mut self,
2899        program: Rc<Program>,
2900        program_id: ProgramId,
2901        call_type: CallType,
2902        address: Option<crate::activation::AddressState>,
2903        caller_settings: Option<Settings>,
2904    ) -> Result<Option<ObjRef>, Failure> {
2905        // **Before the first clause, and its failures print nothing on
2906        // stdout.** So `say 'main ran'` followed by `::requires
2907        // 'no_such_file_zz.rex'` is 43.901 at rc 213 with stdout EMPTY, and
2908        // `::class foo subclass zzznotaclass` is 98.909 at rc 158, likewise
2909        // empty. A program whose directives all install runs its main body
2910        // exactly as one with no directives does. A called program's level is a
2911        // `ROUTINE` frame named as it was called, whose line while its
2912        // directives install is the one installing.
2913        let called = matches!(call_type, CallType::Subroutine | CallType::Function);
2914        // As `Interp::begin_call` takes it: this body runs on a nested
2915        // driver, so recursion through programs is bounded by the same count.
2916        let nested = self.running_activation().is_some();
2917        let probe = 0u8;
2918        if nested
2919            && (self.nesting_depth() >= crate::run::MAX_ACTIVATION_DEPTH
2920                || self.stack_exhausted(&raw const probe))
2921        {
2922            return Err(Raised::insufficient_stack().into());
2923        }
2924        let outer = self.install_caller.replace(caller_settings.clone());
2925        let outer_call = self.install_call.replace(call_type);
2926        // The activation this installs ahead of keeps the manager its package
2927        // had before the install.
2928        self.installing_managers.push((program_id, None));
2929        let installed = self.install_directives(program_id, &program);
2930        let kept = self
2931            .installing_managers
2932            .iter()
2933            .rposition(|(installing, _)| *installing == program_id)
2934            .and_then(|row| self.installing_managers.remove(row).1);
2935        if let Some(Some(manager)) = kept {
2936            self.roots.activity_mut().push_temp(manager);
2937        }
2938        self.install_caller = outer;
2939        self.install_call = outer_call;
2940        if let Err(failure) = installed {
2941            if called {
2942                let arguments = Rc::clone(&self.activity.call_context.arguments);
2943                let name = Rc::clone(&self.activity.call_context.name);
2944                self.capture_site_frame(&failure, b"ROUTINE", &name, &arguments, program_id);
2945                self.seal_site_level();
2946            } else if call_type == CallType::Requires {
2947                self.seal_package_level(&failure, program_id);
2948            }
2949            return Err(failure);
2950        }
2951
2952        // **`::OPTIONS NOPROLOG` suppresses the leading code section of a
2953        // package a `::REQUIRES` loaded, and of nothing else** -- the prologue
2954        // run installs and stops where the prologue is disabled, and the
2955        // top-level program never goes through it. For example, the same file
2956        // prints its first clause when it is the program and does not when
2957        // another file requires it, its public routines and classes reachable
2958        // either way.
2959        if call_type == CallType::Requires
2960            && self
2961                .options_of(program_id)
2962                .is_some_and(|options| options.suppress_prolog)
2963        {
2964            return Ok(None);
2965        }
2966        if call_type == CallType::Requires {
2967            return self.run_prologue(program, program_id, caller_settings, kept);
2968        }
2969        self.run_main(
2970            program,
2971            program_id,
2972            call_type,
2973            address,
2974            caller_settings,
2975            kept,
2976        )
2977    }

3119        // not only the first: a called external program and a `::REQUIRES` prologue

3169        };

3180            }
3181            .map(Ended::value),
3182        };

3189            exit
3190        };

3192            self.capture_activation_frame(failure);

3196                Ok(value) | Err(Failure::Exited(value)) => *value,

3905    /// What a collection that found `live` body bytes owes a heap limit of
3906    /// `limit`: the next collection falls due before the heap can pass the
3907    /// limit by more than its [`heap_limit_step`], and a heap past it raises
3908    /// 5.0 at the next clause boundary of a user program, unless the last
3909    /// raise was for a heap that could hold within a step of this one
3910    /// (`heap_raise_above`). The figure a
3911    /// collection finds is recomputed from the survivors, so the bound does
3912    /// not drift however long the run.
3913    #[cold]
3914    fn bound_by_heap_limit(&mut self, limit: usize, live: usize) {
3915        let step = heap_limit_step(limit);
3916        let headroom = limit.saturating_sub(live).max(step);
3917        self.bytes_due = self.bytes_due.min(headroom);
3918        if live <= limit {
3919            self.heap_raise_above = 0;
3920        } else if live > self.heap_raise_above {
3921            self.heap_over_limit = true;
3922            self.clause_exits = true;
3923        }
3924    }

3933        self.heap_raise_above = self.heap.body_bytes_bound().saturating_add(step);
3934    }

4186/// (r[env.invocation.heap-limit]): decimal digits, optionally followed by `K`, `M` or

4192/// What a [`HEAP_LIMIT_ENV`] value says: `Ok(None)` for the empty string,
4193/// `Ok(Some(bytes))` for a size it names, and otherwise `Err` with the
4194/// message that refuses it.
4195pub fn parse_heap_limit(value: &[u8]) -> Result<Option<usize>, String> {
4196    if value.is_empty() {
4197        return Ok(None);
4198    }
4199    let (digits, shift) = match value.split_last() {
4200        Some((b'K' | b'k', digits)) => (digits, 10),
4201        Some((b'M' | b'm', digits)) => (digits, 20),
4202        Some((b'G' | b'g', digits)) => (digits, 30),
4203        _ => (value, 0),
4204    };
4205    std::str::from_utf8(digits)
4206        .ok()
4207        .filter(|text| !text.is_empty() && text.bytes().all(|byte| byte.is_ascii_digit()))
4208        .and_then(|text| text.parse::<usize>().ok())
4209        .and_then(|count| count.checked_mul(1 << shift))
4210        .map(Some)
4211        .ok_or_else(|| {
4212            format!(
4213                "{HEAP_LIMIT_ENV} must be empty or decimal digits, optionally followed by K, M \
4214                 or G, naming a size this host can address, not `{}`",
4215                String::from_utf8_lossy(value)
4216            )
4217        })
4218}

4459/// Every body of `text`, compiled to a chunk and rendered as text: what the
4460/// `rexx-ir` binary prints.
4461pub fn render_ir(text: Vec<u8>, setting: &[u8]) -> Result<String, String> {
4462    let mode = trace::mode_from_setting(setting).map_err(|byte| {
4463        format!(
4464            "not a TRACE setting: {} (at {:?})",
4465            String::from_utf8_lossy(setting),
4466            char::from(byte)
4467        )
4468    })?;
4469    let trace = trace::ChunkTrace::of(mode);
4470    let program = interrexx_parse::parse_program(text).map_err(|error| format!("{error:?}"))?;
4471
4472    let mut out = String::new();
4473    for (body, what, kind) in ir_bodies(&program) {
4474        out.push_str(&format!("=== {what} ===\n"));
4475        let plan = plan::Plan::build(body, &program.symbols, Some(&program.source), kind);
4476        match ir::compile(body, &plan, trace) {
4477            Ok(chunk) => out.push_str(&ir::render_annotated(&chunk, body, &program.source)),
4478            Err(error) => out.push_str(&format!("(refused: {error:?})\n")),
4479        }
4480    }
4481    Ok(out)
4482}

4510/// Every body [`render_ir`] compiles, in source order, each with the name it is
4511/// printed under.
4512fn ir_bodies(
4513    program: &interrexx_parse::Program,
4514) -> Vec<(&interrexx_parse::CodeBody, String, BodyKind)> {
4515    use interrexx_parse::DirectiveKind;
4516
4517    let mut out = vec![(&program.main, "main".to_string(), BodyKind::Plain)];
4518    for directive in &program.directives {
4519        let (body, what, kind) = match &directive.kind {
4520            DirectiveKind::Method(method) => (method.body.as_ref(), "::METHOD", BodyKind::Method),
4521            DirectiveKind::Attribute(attribute) => {
4522                (attribute.body.as_ref(), "::ATTRIBUTE", BodyKind::Method)
4523            }
4524            DirectiveKind::Routine(routine) => {
4525                (routine.body.as_ref(), "::ROUTINE", BodyKind::Plain)
4526            }
4527            DirectiveKind::Annotate(_)
4528            | DirectiveKind::Class(_)
4529            | DirectiveKind::Constant(_)
4530            | DirectiveKind::Options(_)
4531            | DirectiveKind::Requires(_)
4532            | DirectiveKind::Resource(_) => (None, "", BodyKind::Plain),
4533        };
4534        if let Some(body) = body {
4535            out.push((body, what.to_string(), kind));
4536        }
4537    }
4538    out
4539}

4541/// Reports what a body run after the main one failed with: a refusal on its
4542/// own line and in the exit status, a condition as its traceback; the status
4543/// stays the main body's otherwise.
4544fn report_late_failures(
4545    interp: &mut Interp,
4546    failures: Vec<(Failure, Vec<FailureSite>)>,
4547    path: &[u8],
4548    exit_code: &mut i32,
4549) {
4550    for (failure, mut sites) in failures {
4551        match failure {
4552            // A started activity answers `Ok` for this variant, exactly as a
4553            // send does; the arm is what makes this match exhaustive and
4554            // nothing else.
4555            Failure::Exited(_) => {}
4556            // The guard below the `UNINIT` sweep is what reports this, for
4557            // the reason the main body's own arm gives.
4558            Failure::Deadline => {}
4559            Failure::Loud(loud) => {
4560                interp
4561                    .trace
4562                    .extend_from_slice(format!("interrexx: {}\n", loud.message).as_bytes());
4563                *exit_code = NOT_IMPLEMENTED_EXIT;

4565            Failure::Slice => {
4566                let loud = Loud::scheduler_inconsistency("a slice outside a root driver");

4568                    .trace
4569                    .extend_from_slice(format!("interrexx: {}\n", loud.message).as_bytes());
4570                *exit_code = NOT_IMPLEMENTED_EXIT;
4571            }
4572            Failure::Raised(raised) => {
4573                if sites.is_empty() {
4574                    sites.push(FailureSite::Clause {
4575                        line: 0,
4576                        text: b"<no failing clause recorded>".to_vec(),

4581                    path,
4582                    sites: &sites,
4583                };

4588            }
4589        }
4590    }
4591}

4600/// Runs `body` on a thread with `stack` bytes of stack and answers its
4601/// outcome as soon as it has one. An interpreter `body` hands back still has
4602/// calls in flight: it stays on that thread, serving them, and is dropped
4603/// once they have ended ([`Interp::wind_down`]), while the caller goes on.
4604fn on_thread_of<T: Send + 'static>(
4605    stack: usize,
4606    body: impl FnOnce() -> (T, Option<Interp>) + Send + 'static,
4607) -> T {
4608    let (deliver, delivered) = std::sync::mpsc::channel();
4609    #[cfg(test)]
4610    let family = libraries::round_gate::family();
4611    let interpreter = std::thread::Builder::new()
4612        .name("rexx-interp".to_string())
4613        .stack_size(stack)
4614        .spawn(move || {
4615            #[cfg(test)]
4616            libraries::round_gate::started_by(family);
4617            signal::receive();
4618            let (outcome, kept) = body();
4619            let _ = deliver.send(outcome);
4620            free_ended();
4621            if let Some(interp) = kept {
4622                interp.wind_down();
4623            }
4624        })
4625        .expect("spawning the interpreter thread");
4626    match delivered.recv() {
4627        Ok(outcome) => outcome,
4628        Err(_) => match interpreter.join() {
4629            Err(panic) => std::panic::resume_unwind(panic),
4630            Ok(()) => unreachable!("the interpreter thread ended without an outcome"),
4631        },
4632    }
4633}

4779/// [`execute_as`], answering the interpreter with the outcome where a call is
4780/// still in flight: a pool thread still running a call ends it under the
4781/// baton, through this interpreter's state, which therefore outlives the
4782/// run until [`Interp::wind_down`] has served that end.
4783fn execute_kept(
4784    path: &[u8],
4785    text: Vec<u8>,
4786    entry: Entry<'_>,
4787    invocation: Invocation,
4788    stack: Option<usize>,
4789) -> (Outcome, Option<Interp>) {
4790    let probe = 0u8;
4791    let base = (&raw const probe).addr();
4792    // `rexxc` translates its input as source, an image included: `rexxc`
4793    // answers 13.1 at the `@` of a `.rxo` it is given.
4794    let source = if matches!(entry, Entry::Translate(_)) {
4795        interrexx_parse::ProgramSource::new(text, interrexx_parse::SourceKind::Program)
4796    } else {
4797        match compiled::program_source(text, path) {
4798            Ok(source) => source,
4799            Err(raised) => return (unrestorable_outcome(raised), None),
4800        }
4801    };
4802    // A rejected source that passed an `EXTERNAL`'s resolution point needs an
4803    // interpreter to resolve it in, which `Interp::raise_parse_failure` does.
4804    let parsed = if let Entry::Translate(Some(required_level)) = entry {
4805        interrexx_parse::program_from_with_language_level(source).map(|(program, level)| {
4806            required_level.set(level);
4807            program
4808        })
4809    } else {
4810        interrexx_parse::program_from(source)
4811    };
4812    let program = match parsed {
4813        Ok(program) => Ok(program),
4814        Err(rejected) if rejected.binds.is_empty() => {
4815            return (parse_failure_outcome(path, &rejected), None);
4816        }
4817        Err(rejected) => Err(rejected),
4818    };
4819
4820    let mut interp = Interp::new();
4821    if let Some(stack) = stack {
4822        interp.measure_stack(base, stack);
4823    }
4824    interp.program_path = path.to_vec();
4825    if matches!(entry, Entry::CollectEveryAlloc) {
4826        interp.enable_stress_collect();
4827    }
4828    // The command line is the top-level program's caller, so what it supplied
4829    // goes into the same `call_context` a `CALL` fills -- see that field's own
4830    // doc for what reads it and for the three measured invocations that tell
4831    // "no argument" from "one empty argument" apart.
4832    interp.activity.call_context.name = Rc::from(path);
4833    let parts = invocation.into_parts();
4834    interp
4835        .roots
4836        .activity_mut()
4837        .set_frame_block(parts.frame_block);
4838    let (argument, deadline, switch_mode) = (parts.argument, parts.deadline, parts.switch_mode);
4839    let (arguments, start, address) = (parts.arguments, parts.start, parts.address);
4840    let exits = parts.exits;
4841    // Reachable by `RexxSetHalt` and `RexxSetTrace` through either thread while it runs.
4842    interp.trace_switchable = parts.classic_thread.is_some();
4843    let _classic = parts.classic_thread.map(|caller| {
4844        embedding::ClassicRun::register([caller, signal::current_thread()], interp.timer.inbox())
4845    });
4846    // The host's native code may end a wait nothing in the program can.
4847    if parts.hosted {
4848        interp.note_outside_host();
4849    }
4850    interp.input = Input::new(parts.input);
4851    interp.standard_transient = parts.standard_transient;
4852    if let Some(sinks) = &parts.sinks {
4853        interp.set_buffering(sinks.stdout_terminal, sinks.unbuffered);
4854    }
4855    interp.sinks = parts.sinks;
4856    interp.command_words = parts.words;
4857    interp.command_words_base = parts.words_base;
4858    let default_options = parts.default_options;
4859    if let Some(directory) = parts.directory {
4860        interp.cwd = directory;
4861    }
4862    if let Some(environment) = parts.environment {
4863        interp.adopt_environment(environment);
4864    }
4865    // A seed the environment names is recorded in the simulation's own
4866    // configuration, so that its trace header and report name it.
4867    let mut switch_mode = switch_mode;
4868    if let Some(SwitchMode::Sim(config)) = &mut switch_mode
4869        && let Ok(Some(seed)) = named_hash_seed(&interp.env)
4870    {
4871        config.knobs.hash = Some(seed);
4872    }
4873    // So is the handle key, where a replay could not make it again; it is
4874    // set below, before the bootstrap mints the first handles.
4875    let sim_key = match &mut switch_mode {
4876        Some(SwitchMode::Sim(config)) => match sim::reserve_handle_key(config) {
4877            Ok(live) => Some(live),
4878            Err(refusal) => return (switch_mode_refused(&refusal), None),
4879        },
4880        _ => None,
4881    };
4882    let sim_config = match &switch_mode {
4883        Some(SwitchMode::Sim(config)) => Some(config),
4884        _ => None,
4885    };
4886    // Before the bootstrap, which hashes the first strings: a store holds
4887    // hashes made under the seed of its time, so the seed never changes once
4888    // one is made.
4889    interp.hash_seed = hash_seed_of(&interp.env, sim_config);
4890    interp.warn_of_refused_hash_seed();
4891    // Before the bootstrap too, which mints the first handles: a simulated
4892    // run's handles follow its seed.
4893    if let Some(live) = sim_key {
4894        interp.set_handle_key(live.key());
4895        interp.sim_key = Some(live);
4896    }
4897    interp.set_heap_limit(parts.heap_limit);
4898    // Armed here rather than in `Interp::new`, and after the parse, so that
4899    // what it bounds is the running of this program. `Interp::bootstrap_library`
4900    // below runs clauses of its own and is inside the bound, which is what a
4901    // caller asking for a bounded run wants: a bootstrap that did not finish
4902    // is a run that did not finish.
4903    interp.deadline = deadline.map(crate::clause::Deadline::starting_now);
4904    // **The library bootstrap runs before the command line's own program is
4905    // installed or run, and before its argument string exists.** The program
4906    // has already been *parsed* above, which is where a syntax error is
4907    // reported from and is why that report does not wait on this. The
4908    // library's classes have to be
4909    // in `.environment` by the time the program's first clause runs --
4910    // `Interp::bootstrap_library` carries why that is at start rather than on
4911    // demand -- and running it first leaves nothing of this program's to keep
4912    // reachable across the allocation it does.
4913    let result = interp.bootstrap_library().and_then(|()| {
4914        if let Some(arguments) = &arguments {
4915            let mut values = Vec::with_capacity(arguments.len());
4916            for argument in arguments {
4917                values.push(argument.as_deref().map(|bytes| {
4918                    let value = interp.text(bytes);
4919                    // Rooted as the one argument below is.
4920                    interp.roots.activity_mut().push_temp(value);
4921                    value
4922                }));
4923            }
4924            interp.activity.call_context.arguments = Rc::from(values);
4925        } else if let Some(argument) = argument {
4926            let value = interp.text(&argument);
4927            // Rooted with a `push_temp` taken before `run`, which is what makes it
4928            // outlive every clause: `Op::Clause`'s region truncates the
4929            // temporaries stack back to a watermark it takes on entry, and every
4930            // such watermark sits above this push. This is the same mechanism
4931            // `Interp::invoke_call` uses to keep a call's own arguments reachable
4932            // (`run/call.rs`, the `push_temp(argument.value())` beside the argument
4933            // list it builds); `call_context` itself is not walked by the
4934            // collector, so without this the value is unreachable the first time
4935            // anything allocates.
4936            interp.roots.activity_mut().push_temp(value);
4937            interp.activity.call_context.arguments = Rc::from(&[Some(value)][..]);
4938        }
4939        if let Some((directive, count)) = default_options {
4940            interp.send_default_options(&directive, count)?;
4941        }
4942        if !exits.is_empty() {
4943            interp.install_exits(&exits, &[], false);
4944        }
4945        // After the bootstrap, so `SwitchMode::AtClause` counts this program's
4946        // clauses alone.
4947        if let Some(mode) = switch_mode {
4948            interp.set_switch_mode(mode);
4949        }
4950        match (entry, program) {
4951            (_, Err(rejected)) => Err(interp.raise_parse_failure(&rejected, None)),
4952            (Entry::Translate(_), Ok(program)) => interp.translate(program),
4953            (Entry::Run | Entry::CollectEveryAlloc, Ok(program)) => {
4954                interp.run_as_started(program, start, address)
4955            }
4956        }
4957    });
4958    // The whole echo stack, innermost first: the levels `seal_site_level`
4959    // already closed, then the level that was still unwinding when the
4960    // condition reached the top. See `Activity::failure_sites` for why the two
4961    // are separate fields, and `Raised::report` for what the order means.
4962    let mut failure_sites = std::mem::take(&mut interp.activity.failure_sites);
4963    failure_sites.extend(interp.activity.failure_site.take());
4964    // A propagated SYNTAX condition shows its object's `TRACEBACK`, every line
4965    // by its string value (`r[instr.raise.propagate.traceback]`).
4966    // Read while the activity still holds the object, which roots it: the
4967    // read allocates. The echo lines are replaced and the position is not:
4968    // the major line still names the clause the condition was raised again
4969    // in (`r[instr.raise.propagate]`).
4970    if let Some(object) = interp.activity.traceback_object
4971        && matches!(&result, Err(Failure::Raised(raised)) if raised.condition == "SYNTAX")
4972        && let Ok(lines) = interp.traceback_lines(object)
4973    {
4974        let position = failure_sites
4975            .iter()
4976            .find(|site| site.positions())
4977            .and_then(|site| {
4978                let line = site.line()?;
4979                let name = site.reported_name().map(<[u8]>::to_vec);
4980                Some(FailureSite::Reraised { line, name })
4981            });
4982        failure_sites = lines
4983            .into_iter()
4984            .map(|text| FailureSite::Rendered { text })
4985            .chain(position)
4986            .collect();
4987    }
4988    interp.activity.traceback_object = None;
4989    // `exit_code_for` needs `&mut interp` (`to_number` fills a lazy cache),
4990    // so this has to run before `interp.trace`/`interp.out` move out of
4991    // `interp` below -- a partial move of one field ends `interp`'s usability
4992    // as a whole value, and every other call above this one only reads or
4993    // takes a single field, never the whole struct.
4994    let refused_main = matches!(result, Err(Failure::Loud(_) | Failure::Slice));
4995    let error = match &result {
4996        Err(Failure::Raised(raised)) => Some(i32::from(raised.number)),
4997        _ => None,
4998    };
4999    let returned = match &result {
5000        Ok(Some(value)) | Err(Failure::Exited(Some(value))) => {
5001            interp.string_value_text(*value).ok()
5002        }
5003        _ => None,
5004    };
5005    let mut exit_code = match result {
5006        // `Failure::Exited` is not a failure -- it is `EXIT` (or falling off
5007        // the routine's own end) reached through `NodeKind::Call`'s
5008        // expression form, tunnelled here through `Err`/`?` only because
5009        // `eval`'s own return type has no `Flow` to carry it through instead
5010        // (`Failure::Exited`'s own doc, `error.rs`, has the full argument).
5011        // Treated exactly like an ordinary `Ok(value)`: same exit-code rule,
5012        // no stderr report, because it is not one.
5013        Ok(value) | Err(Failure::Exited(value)) => interp.exit_code_for(value),
5014        Err(Failure::Loud(loud)) => {
5015            interp
5016                .trace
5017                .extend_from_slice(format!("interrexx: {}\n", loud.message).as_bytes());
5018            NOT_IMPLEMENTED_EXIT
5019        }
5020        Err(Failure::Slice) => {
5021            let loud = Loud::scheduler_inconsistency("a slice outside a root driver");
5022            interp
5023                .trace
5024                .extend_from_slice(format!("interrexx: {}\n", loud.message).as_bytes());
5025            NOT_IMPLEMENTED_EXIT
5026        }
5027        Err(Failure::Raised(raised)) => {
5028            // `run_activation` records the site on the way out. An empty
5029            // stack here would mean a condition escaped without passing an
5030            // instruction loop, which nothing in this crate can do; it
5031            // renders visibly rather than panicking, on the error path's
5032            // standing rule that a reportable condition must never become a
5033            // crash.
5034            if failure_sites.is_empty() {
5035                failure_sites.push(FailureSite::Clause {
5036                    line: 0,
5037                    text: b"<no failing clause recorded>".to_vec(),
5038                    indent: 0,
5039                });
5040            }
5041            let site = ClauseSite {
5042                path,
5043                sites: &failure_sites,
5044            };
5045            interp.write_trace_report_lines(
5046                &raised.report_lines(&site),
5047                raised.delivery.stack_exhausted,
5048            );
5049            raised.exit_code()
5050        }
5051        // No report and no status here: the run may still have activities
5052        // and `UNINIT`s to abandon, and the one place that says a deadline
5053        // fired is the guard below them.
5054        Err(Failure::Deadline) => 0,
5055    };
5056
5057    // Main's dispatch has ended.
5058    let mut refused = interp.run_ready_uninits();
5059
5060    // Termination waits for every activity; a refusal in main ends the run
5061    // without the wait.
5062    let started = if refused_main {
5063        interp.take_late_failures()
5064    } else {
5065        interp.run_started_to_end()
5066    };
5067    report_late_failures(&mut interp, started, path, &mut exit_code);
5068
5069    // The last-chance uninit pass, reached from interpreter termination --
5070    // after everything the program and its activities do, and reached whatever
5071    // the program's own outcome was. For example, a program whose main body
5072    // raises 42.3 still prints its class `UNINIT` and exits 214, and one ending
5073    // `exit 7` prints it and exits 7. An activity those `UNINIT`s start is not
5074    // waited for.
5075    refused.extend(interp.terminate());
5076    refused.extend(interp.sim_finish());
5077    for loud in refused {
5078        interp
5079            .trace
5080            .extend_from_slice(format!("interrexx: {}\n", loud.message).as_bytes());
5081        exit_code = NOT_IMPLEMENTED_EXIT;
5082    }
5083
5084    // **The one place a deadline becomes an answer**, below everything that
5085    // runs clauses, so that a run abandoned part-way cannot report the status
5086    // its main body happened to reach. It is here rather than in the three arms
5087    // above because one of the paths between them discards failures on purpose:
5088    // `Interp::run_one_uninit` throws away a raised condition and an `EXIT`
5089    // because the dispatcher does, and a deadline reaching it would otherwise
5090    // vanish. `Interp::deadline_expired` reads the flag the check itself set,
5091    // not the clock, so a run that finished inside its bound is untouched
5092    // however narrow the margin was.
5093    if interp.deadline_expired() {
5094        interp.trace.extend_from_slice(DEADLINE_REPORT);
5095        exit_code = DEADLINE_EXIT;
5096    }
5097
5098    // Read after the activities above, so a collection or a refused chunk
5099    // inside one is counted: `run_program_collect_every_alloc` decides that its
5100    // mode ran from `collections`.
5101    let stack = interp.stack_span();
5102    let collections = interp.heap.collections_performed() - interp.collections_before_program;
5103    let chunks_refused = interp.chunks_refused;
5104    #[cfg(test)]
5105    crate::dispatch::library::note_callback_takes(interp.baton.takes());
5106    #[cfg(test)]
5107    crate::scheduler::note_abandoned_held(&interp.activities);
5108
5109    let outcome = Outcome {
5110        exit_code,
5111        result: returned,
5112        error,
5113        stdout: std::mem::take(&mut interp.out),
5114        stderr: std::mem::take(&mut interp.trace),
5115        stack,
5116        collections,
5117        #[cfg(test)]
5118        peak_body_bytes: interp.heap.peak_body_bytes(),
5119        #[cfg(test)]
5120        failure_level_roots: interp.failure_level_roots,
5121        chunks_refused,
5122        sim: interp.sim_report(),
5123        #[cfg(feature = "pinning")]
5124        pinning: interp.pinning.take(&interp.activity.pins),
5125        #[cfg(feature = "sharing")]
5126        sharing: {
5127            let [bootstrap, program] = interp.heap.sharing_counts();
5128            SharingReport { bootstrap, program }
5129        },
5130    };
5131    if interp.activities.in_flight() {
5132        return (outcome, Some(interp));
5133    }
5134    let ballast = Ballast {
5135        _object_model: interp.object_model.take(),
5136        _programs: std::mem::take(&mut interp.programs),
5137    };
5138    drop(interp);
5139    ENDED.with(|ended| *ended.borrow_mut() = Some(ballast));
5140    (outcome, None)
5141}
