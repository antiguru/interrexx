31/// How many macrospace and registration changes this module has made, which an interpreter of
32/// this process compares when a native call ends, so that a call site does not keep a
33/// resolution made before one ([`crate::Interp::sync_api_changes`]).
34static CHANGES: AtomicU64 = AtomicU64::new(0);
35
36/// [`CHANGES`] now.
37pub(crate) fn changes() -> u64 {
38    CHANGES.load(Ordering::Acquire)
39}
40
41/// Answers `status`, counting a change that may move what a name resolves to.
42fn changed(status: u64) -> u64 {
43    CHANGES.fetch_add(1, Ordering::AcqRel);
44    status
45}

47/// An entry-point registration.
48struct Registered {
49    registry: Registry,
50    name: Vec<u8>,
51    entry: ClassicEntry,
52    user_area: [u8; 16],
53}
54
55/// This process's entry-point registrations, in registration order.
56static ENTRY_POINTS: Mutex<Vec<Registered>> = Mutex::new(Vec::new());
57
58fn entry_points() -> MutexGuard<'static, Vec<Registered>> {
59    ENTRY_POINTS
60        .lock()
61        .expect("the registrations' critical sections cannot panic")
62}
63
64fn find(entries: &[Registered], registry: Registry, name: &[u8]) -> Option<usize> {
65    entries
66        .iter()
67        .position(|entry| entry.registry == registry && entry.name.eq_ignore_ascii_case(name))
68}

70/// `request`'s reply, connecting and starting the service on first use.
71fn call(request: Request) -> Result<Reply, u64> {
72    rxapi_client::client()?.call(request)
73}
74
75fn status_of(reply: &Result<Reply, u64>) -> u64 {
76    match reply {
77        Ok(
78            Reply::Status(status)
79            | Reply::QueueCreated { status, .. }
80            | Reply::QueueOpened { status, .. }
81            | Reply::QueueCount { status, .. }
82            | Reply::Pulled { status, .. }
83            | Reply::Registration { status, .. }
84            | Reply::MacroPosition { status, .. }
85            | Reply::MacroImage { status, .. }
86            | Reply::Macros { status, .. },
87        ) => *status,
88        Ok(Reply::Cancelled) => NO_SERVICE,
89        Err(status) => *status,
90    }
91}

97/// Where `name` is the session queue, has an interpreter whose native call this thread runs
98/// move the session queue it still holds in process to the service first, as it does before a
99/// command, so that this operation and the program see one queue.
100fn attach_session(name: Option<&[u8]>) {
101    if name.is_none_or(|name| name.eq_ignore_ascii_case(b"SESSION")) {
102        interrexx_api::ffi::classic::share_session_queue();
103    }
104}

123/// `RexxCreateQueue`: the status, the name to hand back and the duplicate flag, each `None` where
124/// the caller's output stays untouched.
125pub fn create_queue(requested: Option<&[u8]>, size: usize) -> (u64, Option<Vec<u8>>, Option<bool>) {
126    let reply = call(Request::CreateQueue {
127        session: session(),
128        requested: owned(requested),
129        size: u64::cast_from(size),
130    });
131    match reply {
132        Ok(Reply::QueueCreated {
133            status,
134            name,
135            duplicate,
136        }) => (status, name, duplicate),
137        reply => (status_of(&reply), None, None),
138    }
139}

224/// `RexxRegister*Exe`: `RXSUBCOM_DUP` where this process already has `name`, the first entry
225/// staying.
226pub fn register_entry(
227    registry: Registry,
228    name: &[u8],
229    entry: ClassicEntry,
230    user_area: Option<[u8; 16]>,
231) -> u64 {
232    let mut entries = entry_points();
233    if find(&entries, registry, name).is_some() {
234        return DUPLICATE;
235    }
236    entries.push(Registered {
237        registry,
238        name: name.to_vec(),
239        entry,
240        user_area: user_area.unwrap_or_default(),
241    });
242    changed(OK)
243}

245/// The entry point this process registered as `name`.
246pub fn registered_entry(registry: Registry, name: &[u8]) -> Option<ClassicEntry> {
247    let entries = entry_points();
248    find(&entries, registry, name).map(|at| entries[at].entry)
249}

251/// `RexxQuery*`: the status and the user area. A `None` library looks at this process's entry
252/// points first; an empty one goes to the service by name alone.
253pub fn query_registration(
254    registry: Registry,
255    name: &[u8],
256    library: Option<&[u8]>,
257) -> (u64, Option<[u8; 16]>) {
258    if library.is_none() {
259        let entries = entry_points();
260        if let Some(at) = find(&entries, registry, name) {
261            return (OK, Some(entries[at].user_area));
262        }
263    }
264    match call(Request::QueryRegistration {
265        registry,
266        name: name.to_vec(),
267        library: library.unwrap_or_default().to_vec(),
268    }) {
269        Ok(Reply::Registration { status, registered }) => {
270            (status, registered.map(|registered| registered.user_area))
271        }
272        reply => (status_of(&reply), None),
273    }
274}

276/// `RexxDeregister*`. A `None` library removes this process's entry point first.
277pub fn deregister(registry: Registry, name: &[u8], library: Option<&[u8]>) -> u64 {
278    if library.is_none() {
279        let mut entries = entry_points();
280        if let Some(at) = find(&entries, registry, name) {
281            entries.remove(at);
282            return changed(OK);
283        }
284    }
285    changed(status_of(&call(Request::Deregister {
286        registry,
287        name: name.to_vec(),
288        library: library.unwrap_or_default().to_vec(),
289    })))
290}

292/// The handler `name` resolves to in `registry`, as `RexxResolveSubcom` and its siblings find
293/// it: this process's entry point, else the registration's library loaded in this process and
294/// its procedure found, first as registered and then uppercased, with `search` tried before the
295/// loader's own path and `cwd` after it. `Ok(None)` for a registration naming no library; `RXSUBCOM_NOTREG` where
296/// nothing is registered or the library or procedure is not found. `start` lets the lookup start
297/// a service; without it, no running service means nothing is registered.
298pub fn resolve(
299    registry: Registry,
300    name: &[u8],
301    search: &[std::path::PathBuf],
302    cwd: &std::path::Path,
303    start: bool,
304) -> Result<Option<ClassicEntry>, u64> {
305    match registration(registry, name, start)? {
306        Registration::Entry(entry) => Ok(Some(entry)),
307        Registration::Library { library, procedure } => {
308            interrexx_api::load::classic_procedure(&library, &procedure, search, cwd)
309                .map(Some)
310                .ok_or(NOT_REGISTERED)
311        }
312        Registration::Unbound => Ok(None),
313    }
314}

316/// Where a registration [`registration`] found points.
317pub enum Registration {
318    /// An entry point of this process's.
319    Entry(ClassicEntry),
320    /// A procedure of a library, which is not loaded yet.
321    Library {
322        library: Vec<u8>,
323        procedure: Vec<u8>,
324    },
325    /// A registration naming no library.
326    Unbound,
327}
328
329/// The registration `name` has in `registry`, as [`resolve`] looks it up, before any library is
330/// loaded: this process's entry point, else the service's registration. `RXSUBCOM_NOTREG` where
331/// nothing is registered. `start` lets the lookup start a service.
332pub fn registration(registry: Registry, name: &[u8], start: bool) -> Result<Registration, u64> {
333    if let Some(entry) = registered_entry(registry, name) {
334        return Ok(Registration::Entry(entry));
335    }
336    let client = match start {
337        true => rxapi_client::client()?,
338        false => rxapi_client::running_client().ok_or(NOT_REGISTERED)?,
339    };
340    let reply = client.call(Request::QueryRegistration {
341        registry,
342        name: name.to_vec(),
343        library: Vec::new(),
344    });
345    let registered = match reply {
346        Ok(Reply::Registration {
347            status: OK,
348            registered: Some(registered),
349        }) => registered,
350        reply => {
351            return Err(match status_of(&reply) {
352                OK => NOT_REGISTERED,
353                status => status,
354            });
355        }
356    };
357    if registered.library.is_empty() {
358        return Ok(Registration::Unbound);
359    }
360    Ok(Registration::Library {
361        library: registered.library,
362        procedure: registered.procedure,
363    })
364}

379/// `RexxAddMacro`: `file` translated in this process, relative to its current directory.
380pub fn add_macro(name: &[u8], file: &[u8], position: u64) -> u64 {
381    // Before the file is read: a position neither before nor after is refused
382    // whatever the file holds (Deviation 185).
383    if !interrexx_rxapi::message::macro_position::is_valid(position) {
384        return MACRO_INVALID_POSITION;
385    }
386    let Some(image) = macros::translate(file, name) else {
387        return macros::SOURCE_NOT_FOUND;
388    };
389    changed(status_of(&call(Request::AddMacro {
390        name: name.to_vec(),
391        image,
392        position,
393    })))
394}

426/// `RexxSaveMacroSpace`: the macros `names` lists, or every one, written to `file`. A failure
427/// leaves no file.
428pub fn save_macros(names: Option<Vec<Vec<u8>>>, file: &[u8]) -> u64 {
429    match call(Request::SaveMacros { names }) {
430        Ok(Reply::Macros {
431            status: OK,
432            macros: saved,
433        }) => macros::write_file(file, &interrexx_rxapi::macro_file::write(&saved)),
434        reply => status_of(&reply),
435    }
436}

438/// `RexxLoadMacroSpace`: the macros `names` lists, or every one, from `file`. Macros read before a
439/// failure are loaded.
440pub fn load_macros(names: Option<Vec<Vec<u8>>>, file: &[u8]) -> u64 {
441    let Ok(bytes) = std::fs::read(crate::paths::os_path(file)) else {
442        return MACRO_FILE_ERROR;
443    };
444    let (status, read) = interrexx_rxapi::macro_file::read(&bytes);
445    if read.is_empty() {
446        return status;
447    }
448    let loaded = status_of(&call(Request::LoadMacros {
449        names,
450        macros: read,
451    }));
452    changed(if status == OK { loaded } else { status })
453}
