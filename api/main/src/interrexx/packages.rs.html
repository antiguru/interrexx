1//! The Rexx packages shipped inside this crate, which a `::REQUIRES` finds
2//! after every search route has missed: the rewritten packages of the
3//! extension libraries this crate runs in Rust.
4
5/// One shipped package.
6pub(crate) struct Shipped {
7    /// The file name a `::REQUIRES` names it by.
8    pub(crate) name: &'static str,
9    /// The package's source, from `packages/`.
10    pub(crate) source: &'static [u8],
11}
12
13impl Shipped {
14    /// The path the package is known by: its name in the interpreter's own
15    /// directory, where its copy is installed, or the bare name where that
16    /// directory is unknown.
17    pub(crate) fn path(&self) -> Vec<u8> {
18        match std::env::current_exe()
19            .ok()
20            .and_then(|exe| exe.parent().map(|dir| dir.join(self.name)))
21        {
22            Some(path) => path.as_os_str().as_encoded_bytes().to_vec(),
23            None => self.name.as_bytes().to_vec(),
24        }
25    }
26}

52/// The shipped package a `::REQUIRES` of `name` finds: its file name, or that
53/// name without the `.cls` the search appends, as written or in lower case,
54/// as the file search tries them.
55pub(crate) fn lookup(name: &[u8]) -> Option<&'static Shipped> {
56    let lower = name.to_ascii_lowercase();
57    SHIPPED.iter().find(|package| {
58        let full = package.name.as_bytes();
59        let bare = full.strip_suffix(b".cls").unwrap_or(full);
60        [name, &lower[..]]
61            .iter()
62            .any(|candidate| *candidate == full || *candidate == bare)
63    })
64}
