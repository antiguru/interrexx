40impl Interp {
41    /// Installs `manager` on `package`, or clears it when the setter was sent
42    /// no argument.
43    ///
44    /// `Method~setSecurityManager` and `Routine~setSecurityManager` both set
45    /// it on the package rather than on the executable, so all three setters
46    /// write the same place.
47    ///
48    /// Every activation already running `package`'s code keeps the manager it
49    /// started under, as it is copied into the activation's settings when it
50    /// begins.
51    pub(crate) fn install_security_manager(&mut self, package: ObjRef, manager: Option<ObjRef>) {
52        if let Some(previous) = self.package_data(package).map(|data| data.security_manager) {
53            self.keep_security_manager(package, previous);
54        }
55        self.security_armed |= manager.is_some();
56        if let Some(data) = self.package_data_mut(package) {
57            data.security_manager = manager;
58        }
59    }
60
61    /// Gives every live activation of `package`'s code that does not yet
62    /// keep a manager of its own `previous` to keep.
63    /// A package installing before its own activation runs keeps it too, for
64    /// that activation and the rest of the install.
65    #[cold]
66    fn keep_security_manager(&mut self, package: ObjRef, previous: Option<ObjRef>) {
67        for row in 0..self.installing_managers.len() {
68            let (program, kept) = self.installing_managers[row];
69            if kept.is_none() && self.program_package(program) == Some(package) {
70                self.installing_managers[row].1 = Some(previous);
71            }
72        }
73        let mut programs = Vec::new();
74        self.activity.running_programs(&mut programs);
75        self.activities.running_programs(&mut programs);
76        programs.retain(|&program| self.program_package(program) == Some(package));
77        if programs.is_empty() {
78            return;
79        }
80        let mut keep = |activation: &mut crate::activation::Activation| {
81            if programs.contains(&activation.program_id)
82                && activation
83                    .cold
84                    .as_deref()
85                    .is_none_or(|cold| cold.security_manager.is_none())
86            {
87                activation.cold_mut().security_manager = Some(previous);
88            }
89        };
90        self.activity.for_each_activation_mut(&mut keep);
91        self.activities.for_each_activation_mut(&mut keep);
92    }

94    /// The manager the running activation started under. There is no
95    /// instance-level manager here, so that arm's fallback is `None`, as it
96    /// is with no running activation.
97    #[inline]
98    pub(crate) fn effective_security_manager(&self) -> Option<ObjRef> {
99        if !self.security_armed {
100            return None;
101        }
102        self.armed_security_manager()
103    }

105    /// [`Interp::effective_security_manager`] once any manager has been
106    /// installed. A started message's activity runs no activation, and no
107    /// manager is found for it.
108    #[inline(never)]
109    fn armed_security_manager(&self) -> Option<ObjRef> {
110        let activation = self.activity.running.as_deref()?;
111        if let Some(kept) = activation
112            .cold
113            .as_deref()
114            .and_then(|cold| cold.security_manager)
115        {
116            return kept;
117        }
118        self.program_data(activation.program_id)?.security_manager
119    }

121    /// The manager the running code's package has now, which its `LOCAL` and
122    /// `ENVIRONMENT` checkpoints ask: the class lookup reads the package's
123    /// own, not the activation's.
124    #[inline]
125    pub(crate) fn package_security_manager(&self) -> Option<ObjRef> {
126        if !self.security_armed {
127            return None;
128        }
129        self.armed_package_security_manager()
130    }

132    /// [`Interp::package_security_manager`] once any manager has been
133    /// installed.
134    #[inline(never)]
135    fn armed_package_security_manager(&self) -> Option<ObjRef> {
136        let program = self.activity.running.as_deref()?.program_id;
137        self.program_data(program)?.security_manager
138    }

140    /// The manager `program`'s own installing activation runs under, which is
141    /// the one its `::REQUIRES` directives consult: the package's when its
142    /// install began.
143    pub(crate) fn installing_security_manager(&self, program: ProgramId) -> Option<ObjRef> {
144        if !self.security_armed {
145            return None;
146        }
147        if let Some(kept) = self.kept_installing_manager(program) {
148            return kept;
149        }
150        self.program_data(program)?.security_manager
151    }
152
153    /// The manager `program` had when its install, still running, began,
154    /// where one has been set on it since.
155    pub(crate) fn kept_installing_manager(&self, program: ProgramId) -> Option<Option<ObjRef>> {
156        self.installing_managers
157            .iter()
158            .rev()
159            .find(|(installing, _)| *installing == program)
160            .and_then(|(_, kept)| *kept)
161    }

163    /// One checkpoint.
164    ///
165    /// Answers the info directory, rooted as a temp of the caller's frame,
166    /// when the manager handled the checkpoint, and `None` when it did not or
167    /// when no manager is installed. A manager
168    /// that returns nothing is 91.999 and one that returns anything but `0`
169    /// or `1` is 34.903, both raised here rather than by the caller.
170    pub(crate) fn security_check(
171        &mut self,
172        message: &[u8],
173        entries: &[(&[u8], ObjRef)],
174    ) -> Result<Option<ObjRef>, Failure> {
175        let manager = self.effective_security_manager();
176        self.security_check_by(manager, message, entries)
177    }
178
179    /// [`Interp::security_check`] against `manager` rather than the running
180    /// activation's.
181    pub(crate) fn security_check_by(
182        &mut self,
183        manager: Option<ObjRef>,
184        message: &[u8],
185        entries: &[(&[u8], ObjRef)],
186    ) -> Result<Option<ObjRef>, Failure> {
187        let Some(manager) = manager else {
188            return Ok(None);
189        };
190        let frame = self.roots.activity_mut().push_frame();
191        let outcome = self.security_send(manager, message, entries);
192        self.roots.activity_mut().pop_frame(frame);
193        if let Ok(Some(directory)) = outcome {
194            self.roots.activity_mut().push_temp(directory);
195        }
196        outcome
197    }
198
199    /// [`Interp::security_check`]'s body, so that the root frame it runs
200    /// under is popped on the raising paths too.
201    fn security_send(
202        &mut self,
203        manager: ObjRef,
204        message: &[u8],
205        entries: &[(&[u8], ObjRef)],
206    ) -> Result<Option<ObjRef>, Failure> {
207        let directory = self.security_arguments(entries)?;
208        self.roots.activity_mut().push_temp(directory);
209        let caller = self.caller();
210        let answer = pinned!(
211            self,
212            crate::pinning::PinKind::SecurityManager,
213            self.send_message(manager, message, None, &[Some(directory)], caller)
214        )?;
215        let Some(answer) = answer else {
216            return Err(Raised::no_result(message).into());
217        };
218        self.roots.activity_mut().push_temp(answer);
219        if self.truth(answer, Raised::authorization_not_logical)? {
220            Ok(Some(directory))
221        } else {
222            Ok(None)
223        }
224    }

226    /// The info directory one checkpoint hands its manager, filled by the
227    /// directory's own put rather than a sent `PUT`.
228    fn security_arguments(&mut self, entries: &[(&[u8], ObjRef)]) -> Result<ObjRef, Failure> {
229        let directory = crate::dispatch::hash::new_directory(self, 0)?;
230        self.roots.activity_mut().push_temp(directory);
231        let frame = self.roots.activity_mut().push_frame();
232        for (name, value) in entries {
233            crate::dispatch::hash::directory_put(self, directory, name, *value)?;
234        }
235        self.roots.activity_mut().pop_frame(frame);
236        Ok(directory)
237    }

239    /// The `ARGUMENTS` array a `METHOD` or `CALL` checkpoint carries, holes
240    /// and all -- `new_array(count, arguments)`, which copies the argument
241    /// list as it stands.
242    pub(crate) fn security_arguments_array(&mut self, args: &[Option<ObjRef>]) -> ObjRef {
243        let array = self.alloc_charged(BehaviourId::ARRAY, Body::array(args.to_vec()));
244        self.roots.activity_mut().push_temp(array);
245        array
246    }

248    /// One entry of an info directory the manager has answered, or `None` where it
249    /// left the name unset. The lookup is an exact-string one and not `ENTRY`'s
250    /// upcasing one. An entry the manager set to `.nil` is `Some(.nil)`: the lookup
251    /// answers the object stored.
252    pub(crate) fn security_entry(
253        &mut self,
254        directory: ObjRef,
255        name: &[u8],
256    ) -> Result<Option<ObjRef>, Failure> {
257        let index = self.text(name);
258        self.roots.activity_mut().push_temp(index);
259        let answer = crate::dispatch::hash::store_get(self, directory, index)?;
260        if let Some(value) = answer {
261            self.roots.activity_mut().push_temp(value);
262        }
263        Ok(answer)
264    }

266    /// `Method~setProtected` on the method `object` stands for, which the
267    /// dispatch seam then asks the manager about.
268    pub(crate) fn make_method_protected(&mut self, object: ObjRef) {
269        let Some(method) = self.unshared_identity(object) else {
270            return;
271        };
272        let package = self
273            .source_package(self.installed_executable_source(method))
274            .unwrap_or(Package::Rexx);
275        match self.special_methods.get_mut(method) {
276            Some(existing) => existing.protected = true,
277            None => self.special_methods.insert(
278                method,
279                AccessScope {
280                    access: Access::Default,
281                    protected: true,
282                    package,
283                },
284            ),
285        }
286    }

288    /// Whether the method `object` stands for is a private primitive:
289    /// `Object`'s `run`, `setMethod` and `unsetMethod`, whose access scope is
290    /// filed when the object model is installed.
291    pub(crate) fn installed_is_private(&self, object: ObjRef) -> bool {
292        self.executable_sources
293            .get(&object)
294            .and_then(|record| record.installed)
295            .and_then(|method| self.special_methods.get(method))
296            .is_some_and(|scope| scope.access == Access::Private)
297    }

299    /// Whether the method `object` stands for is protected: a protected primitive of the
300    /// class table, or one `setProtected` was sent to.
301    pub(crate) fn installed_is_protected(&self, object: ObjRef) -> bool {
302        self.executable_sources
303            .get(&object)
304            .and_then(|record| record.installed)
305            .and_then(|method| self.special_methods.get(method))
306            .is_some_and(|scope| scope.protected)
307    }
