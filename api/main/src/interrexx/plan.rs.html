19/// A loaded program's identity: a slot of [`Programs`] in the low half and
20/// the generation it was issued under in the high half. Never reused, so an
21/// id whose program was freed misses.
22#[derive(Copy, Clone, PartialEq, Eq, Hash, Debug)]
23pub(crate) struct ProgramId(u64);
24
25impl std::fmt::Display for ProgramId {
26    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
27        write!(f, "{}.{}", self.slot(), self.generation())
28    }
29}
30
31impl ProgramId {
32    #[inline]
33    fn new(slot: u32, generation: u32) -> ProgramId {
34        ProgramId(u64::from(generation) << 32 | u64::from(slot))
35    }
36
37    /// The slot this id names.
38    #[inline]
39    fn slot(self) -> u32 {
40        u32::cast_truncate(self.0)
41    }
42
43    #[inline]
44    fn generation(self) -> u32 {
45        u32::try_from(self.0 >> 32).expect("the high half")
46    }
47
48    /// The id the `slot`th program loaded into a fresh interpreter gets.
49    #[cfg(test)]
50    pub(crate) fn nth(slot: u32) -> ProgramId {
51        ProgramId::new(slot, 0)
52    }
53
54    /// The slot this id names.
55    #[cfg(test)]
56    pub(crate) fn slot_index(self) -> u32 {
57        self.slot()
58    }
59}

61/// A package object's hold on its program. The program is freed at the first
62/// collection after the last hold is dropped.
63#[derive(Clone, Debug)]
64pub(crate) struct ProgramHold(#[allow(dead_code)] Rc<ProgramId>);
65
66/// One slot of [`Programs`]: the id it answers to and that id's program.
67/// A freed slot holds an id not yet issued.
68struct ProgramSlot {
69    id: ProgramId,
70    program: Option<Rc<Program>>,
71}
72
73/// What stands for one slot's package.
74#[derive(Default)]
75struct PackageSlot {
76    /// The object standing for the program's package, once built.
77    package: Option<ObjRef>,
78    /// The holds of every package object over this program, once one is
79    /// built. A program no package object was built for is never freed.
80    holds: Option<std::rc::Weak<ProgramId>>,
81}

83/// Every program the loader has issued an id for, and what stands for each
84/// one's package. A freed slot is reused under the next generation.
85#[derive(Default)]
86pub(crate) struct Programs {
87    slots: Vec<ProgramSlot>,
88    packages: Vec<PackageSlot>,
89    free: Vec<u32>,
90    live: usize,
91    /// Every id [`Programs::insert`] issued, in order.
92    #[cfg(test)]
93    pub(crate) issued: Vec<ProgramId>,
94    /// Every id [`Programs::free_unheld`] freed, in order.
95    #[cfg(test)]
96    pub(crate) freed: Vec<ProgramId>,
97}
98
99impl Programs {
100    /// Files `program` under a fresh id.
101    pub(crate) fn insert(&mut self, program: Rc<Program>) -> ProgramId {
102        self.live += 1;
103        let id = if let Some(slot) = self.free.pop() {
104            let held = &mut self.slots[usize::cast_from(slot)];
105            held.program = Some(program);
106            held.id
107        } else {
108            let slot = u32::try_from(self.slots.len()).expect("fewer than 2^32 program slots");
109            let id = ProgramId::new(slot, 0);
110            self.slots.push(ProgramSlot {
111                id,
112                program: Some(program),
113            });
114            self.packages.push(PackageSlot::default());
115            id
116        };
117        #[cfg(test)]
118        self.issued.push(id);
119        id
120    }
121
122    /// The program `id` names, or `None` once it has been freed.
123    #[inline]
124    pub(crate) fn get(&self, id: ProgramId) -> Option<&Rc<Program>> {
125        let held = self.slots.get(usize::cast_from(id.slot()))?;
126        if held.id != id {
127            return None;
128        }
129        held.program.as_ref()
130    }
131
132    /// How many programs are loaded.
133    pub(crate) fn len(&self) -> usize {
134        self.live
135    }
136
137    /// `id`'s package slot, or `None` once the program is freed.
138    #[inline]
139    fn package_slot(&self, id: ProgramId) -> Option<&PackageSlot> {
140        self.get(id)?;
141        self.packages.get(usize::cast_from(id.slot()))
142    }
143
144    /// The object standing for `id`'s package, or `None` before one is built
145    /// and once the program is freed.
146    #[inline]
147    pub(crate) fn package(&self, id: ProgramId) -> Option<ObjRef> {
148        self.package_slot(id)?.package
149    }
150
151    /// Every built package object standing for a loaded program.
152    pub(crate) fn packages(&self) -> impl Iterator<Item = ObjRef> + '_ {
153        self.packages.iter().filter_map(|held| held.package)
154    }
155
156    /// Records `package` as the object standing for `id`'s package, and
157    /// answers the hold it keeps on the program. `None` for a freed `id`.
158    pub(crate) fn set_package(&mut self, id: ProgramId, package: ObjRef) -> Option<ProgramHold> {
159        self.get(id)?;
160        let held = &mut self.packages[usize::cast_from(id.slot())];
161        let hold = Rc::new(id);
162        held.package = Some(package);
163        held.holds = Some(Rc::downgrade(&hold));
164        Some(ProgramHold(hold))
165    }
166
167    /// Frees every program whose package objects have all been dropped, and
168    /// answers their ids.
169    pub(crate) fn free_unheld(&mut self) -> Vec<ProgramId> {
170        let mut freed = Vec::new();
171        for (slot, (held, package)) in self.slots.iter_mut().zip(&mut self.packages).enumerate() {
172            if held.program.is_none()
173                || package
174                    .holds
175                    .as_ref()
176                    .is_none_or(|holds| holds.strong_count() > 0)
177            {
178                continue;
179            }
180            freed.push(held.id);
181            #[cfg(test)]
182            self.freed.push(held.id);
183            held.program = None;
184            *package = PackageSlot::default();
185            self.live -= 1;
186            // A slot whose generation would reach the last one is retired, so
187            // that the id it holds is never issued.
188            let next = held.id.generation() + 1;
189            held.id = ProgramId::new(
190                u32::try_from(slot).expect("fewer than 2^32 program slots"),
191                next,
192            );
193            if next < u32::MAX {
194                self.free.push(held.id.slot());
195            }
196        }
197        freed
198    }
199}

250/// A compound's name, split into the pieces `Interp::tail_key` joins.
251#[derive(Debug, PartialEq, Eq)]
252pub(crate) struct CompoundName {
253    /// The stem half, its trailing period included (`compound_parts`' own
254    /// convention).
255    pub(crate) stem: Box<[u8]>,
256    /// The slot the plan holding this entry bound `stem` to, so that finding
257    /// the stem object at a reference costs an index rather than a hash of the

260    /// The tail pieces in source order.
261    pub(crate) tails: Box<[TailPiece]>,

264impl CompoundName {
265    /// Splits one compound-shaped interned spelling.
266    pub(crate) fn split(name: &str) -> CompoundName {
267        let (stem, tails) = compound_parts(name);
268        CompoundName {
269            stem: stem.as_bytes().into(),
270            stem_at: None,
271            tails: tails
272                .into_iter()
273                .map(|tail| match tail {
274                    Tail::Constant(piece) => TailPiece::Constant(piece.as_bytes().into()),
275                    Tail::Variable(piece) => TailPiece::Variable {
276                        name: piece.as_bytes().into(),
277                        at: None,
278                    },
279                })
280                .collect(),
281        }
282    }

285/// A table by `SymbolId`, covering the ids from `first` on. A body's table
286/// interns its own symbols over the keyword spellings every table shares
287/// (`SymbolTable::base_len`), so a body's own ids start well above zero: the
288/// table starts at `base_len`, sized up front, and moves `first` down only for
289/// a body that names a keyword spelling as a variable (`value = 1`). An
290/// `INTERPRET` fragment's plan is then sized by the fragment, not by the
291/// keyword set, and a lookup stays one subtraction and one bounds check.
292#[derive(Debug, Default)]
293pub(crate) struct BySymbol<T> {
294    first: usize,
295    items: Vec<T>,
296}
297
298impl<T: Default> BySymbol<T> {
299    /// Every entry the default, for every id `symbols` has handed out.
300    fn for_table(symbols: &SymbolTable) -> Self {
301        let first = symbols.base_len();
302        BySymbol {
303            first,
304            items: std::iter::repeat_with(T::default)
305                .take(symbols.len() - first)
306                .collect(),
307        }
308    }
309
310    /// The entry for `id`, or `None` outside the table.
311    #[inline]
312    pub(crate) fn get(&self, id: SymbolId) -> Option<&T> {
313        // An id below `first` wraps to past the end.
314        self.items.get(id.index().wrapping_sub(self.first))
315    }
316
317    /// The entry for `id`, moving `first` down to it if it is below. Panics
318    /// on an id past the table, which is an id from a different table.
319    fn entry(&mut self, id: SymbolId) -> &mut T {
320        if id.index() < self.first {
321            let added = self.first - id.index();
322            self.items
323                .splice(0..0, std::iter::repeat_with(T::default).take(added));
324            self.first = id.index();
325        }
326        &mut self.items[id.index() - self.first]
327    }
328
329    /// Every entry, in id order.
330    #[cfg(test)]
331    pub(crate) fn iter(&self) -> impl Iterator<Item = &T> {
332        self.items.iter()
333    }
334
335    /// Every entry, in id order.
336    fn iter_mut(&mut self) -> impl Iterator<Item = &mut T> {
337        self.items.iter_mut()
338    }
339}

341/// One body's variable-resolution plan, built by one upfront pass at first
342/// execution.
343#[derive(Debug, Default)]
344pub(crate) struct Plan {
345    pub(crate) names: interrexx_core::NameMap<Box<[u8]>, usize>,
346    /// The slots `build` reserved for the names the interpreter assigns on its
347    /// own -- `RESULT` after a call that returns a value, `SIGL` on a transfer.
348    /// Cached because the alternative is `Interp::slot_of`, which hashes the
349    /// name; setting `RESULT` cost 236 user instructions per call against 18
350    /// for a slot write.
351    pub(crate) result_slot: Option<usize>,
352    pub(crate) sigl_slot: Option<usize>,
353    /// A method body's `SELF` and `SUPER`, which every send binds.
354    pub(crate) self_slot: Option<usize>,
355    pub(crate) super_slot: Option<usize>,
356    /// The slot per `SymbolId` of the body's own table, `None` where this
357    /// pass bound the symbol none.
358    pub(crate) by_symbol: BySymbol<Option<usize>>,
359    /// The walk's own flag, and scratch: `note_instruction` clears it when the
360    /// instruction it is walking can change the `TRACE` setting in force, and
361    /// [`Plan::build`] sets it again in front of each one. It means nothing
362    /// once that walk has finished.
363    keeps_the_trace_setting: bool,
364    /// What each instruction does to the `TRACE` setting in force, by index --
365    /// the optimizing function `crate::ir::trace_flow` runs its forward
366    /// analysis over, and the whole of what lets a compiled stream decide at
367    /// compile time which value echoes it carries rather than gating each one.
368    trace_events: Box<[TraceEvent]>,
369    /// The static clause indent of every instruction in this body, by index.
370    pub(crate) indents: Box<[usize]>,
371    /// The 1-based source line every instruction in this body sits on, by
372    /// index -- empty for a body built without a source, which is the
373    /// fragment case `build`'s own parameter documents.
374    pub(crate) lines: Box<[usize]>,
375    /// How a compound-shaped symbol this body names splits, by
376    /// `SymbolId::index`, `None` where this pass recorded nothing for it.
377    pub(crate) compounds: BySymbol<Option<CompoundName>>,
378    /// Whether this is a method body whose first instruction past an `EXPOSE`
379    /// is a `TRACE` naming a label-tracing letter, whose `>I>` is traced before
380    /// the method's guard lock is reserved.
381    pub(crate) traces_entry_early: bool,
382    /// Per slot, where the name an `EXPOSE` bound it to last sat in an
383    /// object's pools. On the plan rather than on the binding so that it
384    /// outlives the activation: every object of a class keeps its pools in
385    /// one order, so the place a method body found a name on one receiver is
386    /// where the next receiver holds it too. Sized to the plan's own slots; a
387    /// slot past them, or a fragment's, goes without.
388    pub(crate) pool_hints: Box<[interrexx_core::PoolHint]>,
389    /// The spelling an `EXPOSE` binds each symbol under, made once and shared
390    /// by every binding of it.
391    exposed_names: BySymbol<std::cell::OnceCell<Rc<[u8]>>>,
392    literal_parses: Vec<(usize, Vec<(u32, interrexx_core::ObjRef)>)>,
393    last_chunk: LastChunk,
394}

396/// Whether `body`'s first instruction past an `EXPOSE` is a `TRACE` naming a
397/// label-tracing letter.
398fn traces_entry_early(body: &CodeBody) -> bool {
399    let mut instructions = body
400        .instructions
401        .iter()
402        .map(|instruction| &instruction.kind);
403    let first = match instructions.next() {
404        Some(InstructionKind::Expose { .. }) => instructions.next(),
405        first => first,
406    };
407    matches!(
408        first,
409        Some(InstructionKind::Trace(interrexx_parse::Trace::Setting(bytes)))
410            if crate::trace::parse_trace_request(bytes)
411                .is_ok_and(|request| request.setting.is_some_and(|mode| mode.labels))
412    )
413}

466            Some(line) => {

501    /// Walks `body` once and returns a finished table, built by one
502    /// upfront pass rather than populated lazily one name at a time.
503    pub(crate) fn build(
504        body: &CodeBody,
505        symbols: &SymbolTable,
506        source: Option<&ProgramSource>,
507        kind: BodyKind,
508    ) -> Plan {
509        // `compounds` is sized here rather than filled as names arrive,
510        // because `bind`/`note_compound_name` write into it by id and an id
511        // is only an index into a table of that table's own length.
512        let mut plan = Plan {
513            compounds: BySymbol::for_table(symbols),
514            // Sized here for the reason `compounds` above is, and indexed the
515            // same way: `bind` writes it by `SymbolId::index()`, and an id is
516            // only an index into a table of that table's own length.
517            by_symbol: BySymbol::for_table(symbols),
518            exposed_names: BySymbol::for_table(symbols),
519            ..Plan::default()
520        };
521        let mut trace_events = Vec::with_capacity(body.instructions.len());
522        for (index, instruction) in body.instructions.iter().enumerate() {
523            // Set again in front of each instruction and read back after it,
524            // because `note_instruction` only ever clears the flag: the walk
525            // would otherwise report every instruction after the first
526            // retracing one as retracing too.
527            plan.keeps_the_trace_setting = true;
528            plan.note_instruction(body, &instruction.kind, symbols);
529            if let InstructionKind::Parse(parse) = &instruction.kind
530                && let Some(writes) =
531                    crate::parse_template::literal_writes(body, parse, &plan, symbols)
532            {
533                plan.literal_parses.push((index, writes));
534            }
535            trace_events.push(match (plan.keeps_the_trace_setting, &instruction.kind) {
536                (true, _) => TraceEvent::Keeps,
537                (false, InstructionKind::Trace(setting)) => {
538                    crate::trace::literal_trace_event(setting)
539                }
540                (false, _) => TraceEvent::Unknown,
541            });
542        }
543        plan.trace_events = trace_events.into_boxed_slice();
544        // **`RESULT`, `RC` and `SIGL` get a slot whether or not the body
545        // mentions them**. Registered after the body's own names rather than
546        // before them, because only being *in* the plan matters here and taking
547        // the low indices would renumber every slot this crate's tests pin.
548        plan.result_slot = Some(plan.slot_for(b"RESULT"));
549        plan.slot_for(b"RC");
550        plan.sigl_slot = Some(plan.slot_for(b"SIGL"));
551        // **`SELF` and `SUPER` for the same reason, in a method body only.**
552        // `Interp::begin_method` binds both on every send through
553        // `Interp::slot_of`, whose third source grows the frame and records
554        // the name in `Activation::extra` -- so a method body that never
555        // mentions either name paid two boxed keys and two map inserts per
556        // send, which is exactly the cost the paragraph above describes for
557        // `RESULT`. Measured on `bench-programs/dispatch.rex`, `slot_of`
558        // reached from that binding was the largest single caller of `extra`.
559        if kind == BodyKind::Method {
560            plan.self_slot = Some(plan.slot_for(b"SELF"));
561            plan.super_slot = Some(plan.slot_for(b"SUPER"));
562            plan.traces_entry_early = traces_entry_early(body);
563        }
564        plan.pool_hints = std::iter::repeat_with(interrexx_core::PoolHint::default)
565            .take(plan.names.len())
566            .collect();
567        plan.indents = crate::run::all_indents(&body.instructions);
568        if let Some(source) = source {
569            plan.lines = body
570                .instructions
571                .iter()
572                .map(|instruction| source.line_of(instruction.clause_span.start))
573                .collect();
574        }
575        plan
576    }

578    pub(crate) fn literal_parse(&self, index: usize) -> Option<&[(u32, interrexx_core::ObjRef)]> {
579        let at = self
580            .literal_parses
581            .binary_search_by_key(&index, |(at, _)| *at)
582            .ok()?;
583        Some(&self.literal_parses[at].1)
584    }

586    /// Registers every name one instruction's own fields could read -- the
587    /// exhaustive match `build`'s doc comment describes, factored out to
588    /// its own function because the match itself, covering all
589    /// thirty-nine `InstructionKind` variants, does not fit inside a loop
590    /// body and stay readable.
591    fn note_instruction(&mut self, body: &CodeBody, kind: &InstructionKind, symbols: &SymbolTable) {
592        match kind {
593            InstructionKind::Assignment { target, value } => {
594                self.note(body.expr(*target), symbols);
595                self.note(body.expr(*value), symbols);
596            }
597            InstructionKind::Message { term, value } => {
598                self.note(body.expr(*term), symbols);
599                self.note_opt(value.as_ref().map(|e| body.expr(*e)), symbols);
600            }
601            InstructionKind::Command { expression }
602            | InstructionKind::Push { expression }
603            | InstructionKind::Queue { expression }
604            | InstructionKind::Say { expression }
605            | InstructionKind::Return { expression }
606            | InstructionKind::Exit { expression }
607            | InstructionKind::Reply { expression }
608            | InstructionKind::Numeric { expression, .. } => {
609                self.note_opt(expression.as_ref().map(|e| body.expr(*e)), symbols)
610            }
611            InstructionKind::Interpret { expression } | InstructionKind::Options { expression } => {
612                self.keeps_the_trace_setting = false;
613                self.note(body.expr(*expression), symbols);
614            }
615            InstructionKind::Do(loop_) | InstructionKind::Loop(loop_) => {
616                self.note_loop(body, loop_, symbols);
617            }
618            InstructionKind::If { condition, .. } | InstructionKind::When { condition, .. } => {
619                self.note(body.expr(*condition), symbols);
620            }
621            InstructionKind::WhenCase { values, .. } => {
622                for value in values {
623                    self.note(body.expr(*value), symbols);
624                }
625            }
626            InstructionKind::Select { case, .. } => {
627                self.note_opt(case.as_ref().map(|e| body.expr(*e)), symbols)
628            }
629            InstructionKind::Catch(handler) => {
630                self.note_opt(handler.variable.as_ref().map(|e| body.expr(*e)), symbols);
631                for condition in &handler.conditions {
632                    if let Some(interrexx_parse::CatchCode::Variable(name)) = condition.code {
633                        self.note_variable_ref(&VariableRef::Direct(name), symbols);
634                    }
635                }
636            }
637            // No expression and no data variable: `Select`'s own `label`
638            // and `Leave`/`Iterate`/`End`'s `name` are *block* labels
639            // matched against `LEAVE`/`ITERATE`/`END`, never read through
640            // `slot_of` -- the same distinction `note_loop` draws for a
641            // `DO`/`LOOP`'s own `label`, below.
642            InstructionKind::Label { .. }
643            | InstructionKind::Then
644            | InstructionKind::Else { .. }
645            | InstructionKind::Otherwise
646            | InstructionKind::Leave { .. }
647            | InstructionKind::Iterate { .. }
648            | InstructionKind::End { .. }
649            | InstructionKind::Nop => {}
650            InstructionKind::Drop { variables }
651            | InstructionKind::Expose { variables }
652            | InstructionKind::Procedure { variables } => {
653                for variable in variables {
654                    self.note_variable_ref(variable, symbols);
655                }
656            }
657            InstructionKind::Parse(parse)
658            | InstructionKind::Arg(parse)
659            | InstructionKind::Pull(parse) => self.note_parse(body, parse, symbols),
660            InstructionKind::Call(call) => self.note_call(body, call, symbols),
661            InstructionKind::Signal(signal) => {
662                if let Signal::Value(expr) = &**signal {
663                    self.note(body.expr(*expr), symbols);
664                }
665                // `Label`/`Trap` name a label or a condition, neither a
666                // data variable.
667            }
668            InstructionKind::Guard(guard) => {
669                self.note_opt(guard.condition.as_ref().map(|e| body.expr(*e)), symbols)
670            }
671            InstructionKind::Forward(forward) => {
672                self.note_opt(forward.to.as_ref().map(|e| body.expr(*e)), symbols);
673                self.note_opt(forward.message.as_ref().map(|e| body.expr(*e)), symbols);
674                self.note_opt(forward.class.as_ref().map(|e| body.expr(*e)), symbols);
675                self.note_opt(forward.arguments.as_ref().map(|e| body.expr(*e)), symbols);
676                if let Some(items) = &forward.array {
677                    self.note_args(body.args(items), symbols);
678                }
679            }
680            InstructionKind::Raise(raise) => {
681                self.note_opt(raise.rc.as_ref().map(|e| body.expr(*e)), symbols);
682                self.note_opt(raise.description.as_ref().map(|e| body.expr(*e)), symbols);
683                self.note_opt(raise.additional.as_ref().map(|e| body.expr(*e)), symbols);
684                if let Some(items) = &raise.array {
685                    self.note_args(body.args(items), symbols);
686                }
687                if let Some(result) = &raise.result {
688                    self.note_opt(result.value.as_ref().map(|e| body.expr(*e)), symbols);
689                }
690            }
691            InstructionKind::Use(use_) => match &**use_ {
692                Use::Arg { targets, .. } => {
693                    for target in targets.iter().flatten() {
694                        self.note(body.expr(target.target), symbols);
695                        self.note_opt(target.default.as_ref().map(|e| body.expr(*e)), symbols);
696                    }
697                }
698                Use::Local { variables } => {
699                    for variable in variables {
700                        self.note_variable_ref(variable, symbols);
701                    }
702                }
703            },
704            InstructionKind::Address(address) => {
705                self.note_opt(address.dynamic.as_ref().map(|e| body.expr(*e)), symbols);
706                self.note_opt(address.command.as_ref().map(|e| body.expr(*e)), symbols);
707                if let Some(io) = &address.io {
708                    for redirection in [&io.input, &io.output, &io.error] {
709                        match redirection {
710                            // `STEM name.`: the same single, complete,
711                            // interned symbol a bare `ExprView::Stem` read
712                            // is, so it gets the same treatment `note`
713                            // gives one, id and all.
714                            Redirection::Stem(id) => self.bind(*id, symbols.name(*id)),
715                            Redirection::Stream(expr) | Redirection::Using(expr) => {
716                                self.note(body.expr(*expr), symbols);
717                            }
718                            Redirection::Default | Redirection::Normal => {}
719                        }
720                    }
721                }
722            }
723            InstructionKind::Trace(trace) => {
724                self.keeps_the_trace_setting = false;
725                if let Trace::Value(expr) = trace {
726                    self.note(body.expr(*expr), symbols);
727                }
728            }
729        }
730    }

732    /// A `DO`/`LOOP` header's own names: `COUNTER`/control variables (bound,
733    /// since each is one complete, interned symbol with its own id, the
734    /// same treatment `note` gives a bare `Variable`) and every expression
735    /// its kind and its trailing `WHILE`/`UNTIL` carry.
736    fn note_loop(&mut self, body: &CodeBody, loop_: &Loop, symbols: &SymbolTable) {
737        if let Some(counter) = loop_.counter {
738            self.bind(counter, symbols.name(counter));
739        }
740        match &loop_.kind {
741            LoopKind::Simple | LoopKind::Forever => {}
742            LoopKind::Count(count) => self.note_opt(count.as_ref().map(|e| body.expr(*e)), symbols),
743            LoopKind::Controlled(controlled) => {
744                self.bind(controlled.control, symbols.name(controlled.control));
745                self.note(body.expr(controlled.initial), symbols);
746                self.note_opt(controlled.to.as_ref().map(|e| body.expr(*e)), symbols);
747                self.note_opt(controlled.by.as_ref().map(|e| body.expr(*e)), symbols);
748                self.note_opt(
749                    controlled.for_count.as_ref().map(|e| body.expr(*e)),
750                    symbols,
751                );
752            }
753            LoopKind::Over {
754                control,
755                target,
756                for_count,
757            } => {
758                self.bind(*control, symbols.name(*control));
759                self.note(body.expr(*target), symbols);
760                self.note_opt(for_count.as_ref().map(|e| body.expr(*e)), symbols);
761            }
762            LoopKind::With {
763                index,
764                item,
765                target,
766                for_count,
767            } => {
768                if let Some(index) = index {
769                    self.bind(*index, symbols.name(*index));
770                }
771                if let Some(item) = item {
772                    self.bind(*item, symbols.name(*item));
773                }
774                self.note(body.expr(*target), symbols);
775                self.note_opt(for_count.as_ref().map(|e| body.expr(*e)), symbols);
776            }
777        }
778        if let Some(conditional) = &loop_.conditional {
779            self.note(body.expr(conditional.condition), symbols);
780        }
781    }

783    /// A `PARSE`/`ARG`/`PULL` instruction's own names: `PARSE VAR name`'s
784    /// source variable (read from, so bound the same way any read is), any
785    /// expression a `VALUE` source or a trigger's pattern carries, and
786    /// every template target -- already expressions (`ast.rs`'s own doc: a
787    /// dropped `.` placeholder aside, a target is a `Variable`/`Stem`/
788    /// `Compound` node), so `note` alone is enough for them.
789    fn note_parse(&mut self, body: &CodeBody, parse: &Parse, symbols: &SymbolTable) {
790        match &parse.source {
791            ParseSource::Var(id) => self.bind(*id, symbols.name(*id)),
792            ParseSource::Value(expr) => {
793                self.note_opt(expr.as_ref().map(|e| body.expr(*e)), symbols)
794            }
795            ParseSource::Arg
796            | ParseSource::LineIn
797            | ParseSource::Pull
798            | ParseSource::Source
799            | ParseSource::Version => {}
800        }
801        for trigger in parse.template.iter().flatten() {
802            self.note_opt(trigger.value.as_ref().map(|e| body.expr(*e)), symbols);
803            self.note_args(body.args(&trigger.targets), symbols);
804        }
805    }

807    /// A `CALL`'s own names: a dynamic target and every argument. `Named`'s
808    /// own `name`/`literal` and `Qualified`'s `namespace`/`name` are a
809    /// routine name and a namespace, resolved by their own search, never
810    /// through `slot_of`; `Trap` names a condition, not a variable.
811    fn note_call(&mut self, body: &CodeBody, call: &Call, symbols: &SymbolTable) {
812        match call {
813            Call::Named { name, args, .. } => {
814                if names_trace(name) {
815                    self.keeps_the_trace_setting = false;
816                }
817                self.note_args(body.args(args), symbols);
818            }
819            Call::Qualified { name, args, .. } => {
820                if names_trace(symbols.name(*name).as_bytes()) {
821                    self.keeps_the_trace_setting = false;
822                }
823                self.note_args(body.args(args), symbols);
824            }
825            Call::Dynamic { target, args } => {
826                // The target is a run-time value, so this cannot be read here.
827                self.keeps_the_trace_setting = false;
828                self.note(body.expr(*target), symbols);
829                self.note_args(body.args(args), symbols);
830            }
831            Call::Trap(_) => {}
832        }
833    }

835    /// One `Drop`/`Expose`/`Procedure`/`Use Local` target.
836    fn note_variable_ref(&mut self, var_ref: &VariableRef, symbols: &SymbolTable) {
837        let (VariableRef::Direct(id) | VariableRef::Indirect(id)) = *var_ref;
838        let name = symbols.name(id);
839        if name.contains('.') {
840            self.note_compound_name(id, name);
841        } else {
842            self.bind(id, name);
843        }
844    }

846    /// Registers every name a compound-shaped spelling's decomposition
847    /// touches: the stem itself, and any tail piece that is a variable
848    /// -- a constant piece (a bare digit run, or the empty piece a
849    /// trailing period leaves) names nothing. `name` is either a `Compound`
850    /// expression's own interned spelling, or a `Drop`/`Expose`/
851    /// `Procedure`/`Use Local` target's, once `note_variable_ref` has
852    /// already established it is compound- or stem-shaped.
853    fn note_compound_name(&mut self, id: SymbolId, name: &str) {
854        let mut entry = CompoundName::split(name);
855        // The stem's slot, kept for the same reason each piece's is: a
856        // compound reference reads or writes the stem object through it, so
857        // finding it costs an index into the frame instead of hashing the
858        // stem's bytes at every reference.
859        entry.stem_at = Some(self.slot_for(&entry.stem));
860        for piece in &mut entry.tails {
861            if let TailPiece::Variable { name, at } = piece {
862                // The slot this pass was already computing and dropping. It
863                // is kept on the piece so that resolving the piece at a
864                // reference costs an index into the frame instead of hashing
865                // the name: measured by `perf record` over
866                // `samples/rexxcps.rex` at `5dc12a403`, before this work
867                // (`REXX_ENGINE=ir`, `count=100`/`averaging=100`, 999 Hz),
868                // hashing a `&[u8]` was 4.43% of self time with SipHash's own
869                // `write` at a further 3.29% and `Interp::slot_of` at 1.72%.
870                // The commit belongs with the figures: this function is what
871                // takes that work out, so re-profiling at head will not find
872                // them.
873                *at = Some(self.slot_for(name));
874            }
875        }
876        *self.compounds.entry(id) = Some(entry);
877    }

879    /// Assigns slots to every variable `expr` names, in source order.
880    fn note(&mut self, expr: ExprRef<'_>, symbols: &SymbolTable) {
881        match expr.kind() {
882            ExprView::Variable(id) | ExprView::Stem(id) => {
883                self.bind(id, symbols.name(id));
884            }
885            ExprView::Compound(id) => self.note_compound_name(id, symbols.name(id)),
886            // A literal, a constant (its value is its own spelling, never a
887            // variable) and a `.name` environment symbol (resolved through
888            // the class/environment lookup, never `slot_of`) name nothing.
889            // A namespace:name class resolver names a class, not a
890            // variable, either.
891            ExprView::Literal(_)
892            | ExprView::Constant(_)
893            | ExprView::DotVariable(_)
894            | ExprView::ClassResolver { .. } => {}
895            ExprView::Prefix { operand, .. } => self.note(operand, symbols),
896            ExprView::Binary { left, right, .. } => {
897                self.note(left, symbols);
898                self.note(right, symbols);
899            }
900            ExprView::Call { target, args } => {
901                let named = match target {
902                    Target::Symbol(id) => names_trace(symbols.name(id).as_bytes()),
903                    Target::Literal(bytes) => names_trace(bytes),
904                };
905                if named {
906                    self.keeps_the_trace_setting = false;
907                }
908                self.note_args(args, symbols);
909            }
910            ExprView::QualifiedCall { name, args, .. } => {
911                if names_trace(symbols.name(name).as_bytes()) {
912                    self.keeps_the_trace_setting = false;
913                }
914                self.note_args(args, symbols);
915            }
916            ExprView::Message {
917                target,
918                super_class,
919                args,
920                ..
921            } => {
922                self.note(target, symbols);
923                if let Some(super_class) = super_class {
924                    self.note(super_class, symbols);
925                }
926                self.note_args(args, symbols);
927            }
928            ExprView::List(items) => self.note_args(items, symbols),
929            ExprView::Logical(items) => {
930                for item in items.iter().flatten() {
931                    self.note(item, symbols);
932                }
933            }
934            ExprView::VariableReference { symbol, .. } => self.bind(symbol, symbols.name(symbol)),
935        }
936    }

956    /// Binds `name` to a slot, and `id` to the same one.
957    fn bind(&mut self, id: SymbolId, name: &str) {
958        let slot = self.slot_for(name.as_bytes());
959        *self.by_symbol.entry(id) = Some(slot);
960        if name.contains('.') {
961            self.compounds.entry(id).get_or_insert_with(|| {
962                let mut entry = CompoundName::split(name);
963                if shape_of(name.as_bytes()) == NameShape::Stem {
964                    entry.stem_at = Some(slot);
965                }
966                entry
967            });
968        }
969    }

1000    /// The plan for one body, from the cache or built and cached on
1001    /// `Interp` rather than on the body, because an `Rc<Program>` gives
1002    /// shared immutable access and nothing can be written into a `CodeBody`
1003    /// reached through one).
1004    pub(crate) fn plan_for(
1005        &mut self,
1006        key: BodyKey,
1007        body: &CodeBody,
1008        symbols: &SymbolTable,
1009        source: &ProgramSource,
1010    ) -> Rc<Plan> {
1011        if let Some((last, plan)) = &self.last_plan
1012            && *last == key
1013        {
1014            return Rc::clone(plan);
1015        }
1016        let plan = match self.plans.get(&key) {
1017            Some(plan) => Rc::clone(plan),
1018            None => {
1019                let plan = Rc::new(Plan::build(
1020                    body,
1021                    symbols,
1022                    Some(source),
1023                    self.body_kind(key),
1024                ));
1025                self.plans.insert(key, Rc::clone(&plan));
1026                plan
1027            }
1028        };
1029        self.last_plan = Some((key, Rc::clone(&plan)));
1030        plan
1031    }

1033    /// Whether the body `key` names is entered as a method.
1034    fn body_kind(&self, key: BodyKey) -> BodyKind {
1035        let Some(index) = key.directive else {
1036            return BodyKind::Plain;
1037        };
1038        let kind = self
1039            .programs
1040            .get(key.program)
1041            .and_then(|program| program.directives.get(index))
1042            .map(|directive| &directive.kind);
1043        match kind {
1044            Some(DirectiveKind::Method(_) | DirectiveKind::Attribute(_)) => BodyKind::Method,
1045            _ => BodyKind::Plain,
1046        }
1047    }

1049    /// The chunk for one body **under one trace setting**, from the cache or
1050    /// compiled and cached, keyed as plans are plus the trace setting.
1051    pub(crate) fn chunk_for(
1052        &mut self,
1053        key: BodyKey,
1054        trace: ChunkTrace,
1055        body: &CodeBody,
1056        plan: &Plan,
1057    ) -> Option<Rc<crate::ir::Chunk>> {
1058        if let Some(chunk) = plan.last_chunk(key, trace) {
1059            return Some(chunk);
1060        }
1061        let chunk = match self.chunks.get(&(key, trace)) {
1062            Some(chunk) => Rc::clone(chunk),
1063            None => match self.compile_chunk(body, plan, trace) {
1064                Ok(chunk) => {
1065                    let chunk = Rc::new(chunk);
1066                    self.chunks.insert((key, trace), Rc::clone(&chunk));
1067                    chunk
1068                }
1069                Err(_) => {
1070                    self.chunks_refused += 1;
1071                    return None;
1072                }
1073            },
1074        };
1075        *plan.last_chunk.0.borrow_mut() = Some((key, trace, Rc::clone(&chunk)));
1076        Some(chunk)
1077    }

1079    /// `body` compiled under `trace`, settling each clause's trace setting at compile time
1080    /// unless something outside the body can switch it between two of its clauses: a clause
1081    /// exit, or another thread's `RexxSetTrace` or `SetTrace`.
1082    pub(crate) fn compile_chunk(
1083        &self,
1084        body: &CodeBody,
1085        plan: &Plan,
1086        trace: ChunkTrace,
1087    ) -> Result<crate::ir::Chunk, crate::ir::ChunkTooLarge> {
1088        if self.clause_exits || self.trace_switchable {
1089            crate::ir::compile_unsettled(body, plan, trace)
1090        } else {
1091            crate::ir::compile(body, plan, trace)
1092        }
1093    }

1095    /// The first `SetTrace` served on the instance (`r[native.threads.trace]`), or the first
1096    /// `SetThreadTrace`: from here on every body compiles unsettled, and each cached chunk is
1097    /// retired, so a body compiles again the next time it is entered. A frame running a
1098    /// retired chunk keeps it to the end of that run, which is what
1099    /// `r[native.threads.trace.deviation.166]` declares.
1100    #[cold]
1101    pub(crate) fn make_trace_switchable(&mut self) {
1102        if self.trace_switchable {
1103            return;
1104        }
1105        self.trace_switchable = true;
1106        // Frames hold their chunk by `Rc`, so taking it out of the cache frees nothing a
1107        // frame runs. The resources a chunk holds go when its program does.
1108        self.retired_chunks
1109            .extend(self.chunks.drain().map(|((key, _), chunk)| (key, chunk)));
1110        for plan in self.plans.values() {
1111            plan.forget_last_chunk();
1112        }
1113        if let Some((_, plan)) = &self.last_plan {
1114            plan.forget_last_chunk();
1115        }
1116    }

1118    /// The slot `name` resolves to in the current frame, allocating one if it
1119    /// resolves to none.
1120    pub(crate) fn bound_slot_of(&self, name: &[u8]) -> Option<usize> {
1121        let activation = self.activation();
1122        activation
1123            .plan
1124            .slot_of(name)
1125            .or_else(|| activation.extra.get(name))
1126    }

1128    pub(crate) fn slot_of(&mut self, name: &[u8]) -> usize {
1129        let activation = self.activation();
1130        if let Some(slot) = activation.plan.slot_of(name) {
1131            return slot;
1132        }
1133        if let Some(slot) = activation.extra.get(name) {
1134            return slot;
1135        }
1136        let frame = activation.frame;
1137        let slot = self.roots.activity_mut().grow_slots_of(frame, 1);
1138        self.activation_mut().extra.insert(name.into(), slot);
1139        if let Some(auto) =
1140            (self.activation().cold.as_deref()).and_then(|cold| cold.auto_expose.clone())
1141        {
1142            self.auto_expose_slot(&auto, slot, name.into());
1143        }
1144        slot
1145    }

1147    /// Resolves a fragment's own `SymbolId`s to slots in the **enclosing**
1148    /// frame. Out of line: inlined into `Interp::run_fragment`, the register
1149    /// allocation of its loops moves with unrelated changes to the interpreter.
1150    #[inline(never)]
1151    pub(crate) fn fragment_plan(&mut self, fragment: &Fragment) -> Plan {
1152        // The same upfront pass, run against the fragment's own body, which
1153        // numbers its names 0..n in walk order. Those numbers are local to the
1154        // fragment and mean nothing to the enclosing frame; the loop below is
1155        // what translates them.
1156        // `BodyKind::Plain` whatever body the `INTERPRET` sits in: this plan
1157        // is a numbering of the *fragment's* names, and every name in it is
1158        // resolved against the enclosing frame below. `SELF` and `SUPER` are
1159        // already bound there when that frame is a method's, so asking for
1160        // them here would allocate nothing and change nothing -- and when it
1161        // is not a method's, it would grow the enclosing frame by two slots
1162        // that nothing writes.
1163        let local = Plan::build(&fragment.body, &fragment.symbols, None, BodyKind::Plain);
1164
1165        // Walk order, recovered from the local numbering rather than from
1166        // iterating the map, because a `HashMap`'s order varies run to run and
1167        // the enclosing frame's slots would then be allocated in a different
1168        // order each time. Nothing observable depends on that order today,
1169        // which is the reason to fix it now rather than after something does.
1170        let mut by_local: Vec<&[u8]> = vec![b""; local.len()];
1171        for (name, slot) in &local.names {
1172            by_local[*slot] = name;
1173        }
1174        let enclosing: Vec<usize> = by_local.iter().map(|name| self.slot_of(name)).collect();
1175
1176        // **The same plan with every slot it names moved into the enclosing
1177        // frame**, which is what lets a fragment compile at all: a chunk's
1178        // `Op::Load` and `Op::Store` carry plan slots, and a fragment's own
1179        // numbering means nothing in the frame those ops write.
1180        let mut compiled = local;
1181        compiled.names = compiled
1182            .names
1183            .iter()
1184            .map(|(name, slot)| (name.clone(), enclosing[*slot]))
1185            .collect();
1186        for entry in compiled.by_symbol.iter_mut() {
1187            *entry = entry.map(|local_slot| enclosing[local_slot]);
1188        }
1189        // Prepared writes use the same local numbering and must move with
1190        // the ordinary stores. Fall back to the template walk if a caller
1191        // slot cannot fit in the prepared op's u32 operand.
1192        compiled.literal_parses.retain_mut(|(_, writes)| {
1193            writes.iter_mut().all(|(slot, _)| {
1194                let local_slot = usize::cast_from(*slot);
1195                let Ok(enclosing_slot) = u32::try_from(enclosing[local_slot]) else {
1196                    return false;
1197                };
1198                *slot = enclosing_slot;
1199                true
1200            })
1201        });
1202        compiled.result_slot = compiled.result_slot.map(|slot| enclosing[slot]);
1203        compiled.sigl_slot = compiled.sigl_slot.map(|slot| enclosing[slot]);
1204        compiled.self_slot = compiled.self_slot.map(|slot| enclosing[slot]);
1205        compiled.super_slot = compiled.super_slot.map(|slot| enclosing[slot]);
1206        for entry in compiled.compounds.iter_mut().flatten() {
1207            entry.stem_at = None;
1208            for tail in entry.tails.iter_mut() {
1209                if let TailPiece::Variable { at, .. } = tail {
1210                    *at = None;
1211                }
1212            }
1213        }
1214
1215        compiled
1216    }
1217}

1219/// Whether a call names the `TRACE()` builtin, which sets the running
1220/// activation's own `TRACE` setting where every other call cannot.
1221fn names_trace(name: &[u8]) -> bool {
1222    name.eq_ignore_ascii_case(b"TRACE")
1223}
