1//! An interpreter's pool of threads, which run native calls and blocking
2//! operations while the baton's holder runs other activities.
3
4use std::sync::{Arc, Condvar, Mutex, PoisonError};
5use std::thread::{JoinHandle, ThreadId};

7/// A pool thread's stack: the interpreter thread's, since a callback runs
8/// Rexx code there, and the translator's and the collector's depth limits
9/// are measured against that size.
10pub(crate) const POOL_STACK_BYTES: usize = crate::INTERPRETER_STACK_BYTES;
11
12/// The most threads an interpreter's pool keeps. A job that finds every
13/// thread busy gets a thread beyond the bound, which ends once the pool is
14/// back within it, so a blocking call never waits for a busy thread.
15pub(crate) const POOL_BOUND: usize = 64;
16
17/// What a pool thread runs.
18pub(crate) type Job = Box<dyn FnOnce() + Send>;
19
20/// Runs `job`, posting a panic it ends in to `inbox` for the baton's holder
21/// to panic with, before the baton lent to this thread goes back: a lend
22/// is not given back as a panic unwinds ([`crate::island::Lent`]), and the
23/// lender drains the inbox before it runs a clause.
24pub(crate) fn posting_panics(
25    inbox: &crate::timer::Inbox<super::Posted>,
26    baton: &crate::island::InterpBaton,
27    job: impl FnOnce(),
28) {
29    if let Err(payload) = std::panic::catch_unwind(std::panic::AssertUnwindSafe(job)) {
30        // Widens the window a lend given back while the panic unwinds would
31        // open, for a test to see.
32        #[cfg(test)]
33        #[expect(
34            clippy::disallowed_methods,
35            reason = "widens the real pool thread panic window for the test"
36        )]
37        std::thread::sleep(std::time::Duration::from_millis(100));
38        inbox.post(super::Posted::Panicked(payload));
39        if baton.lent().is_some() {
40            baton.give_back();
41        }
42    }
43}

93impl Pool {
94    /// An empty pool that keeps at most `bound` threads, each with `stack`
95    /// bytes of stack.
96    pub(crate) fn new(stack: usize, bound: usize) -> Pool {
97        let pool = Pool {
98            shared: Arc::new(Shared {
99                returned: Condvar::new(),
100                state: Mutex::new(State {
101                    idle: Vec::new(),
102                    threads: Vec::new(),
103                    closed: false,
104                    bound,
105                    mode: Mode::Normal,
106                    #[cfg(test)]
107                    fixed: false,
108                }),
109            }),
110            stack,
111        };
112        #[cfg(test)]
113        LAST_POOL.with_borrow_mut(|last| *last = Some(Arc::downgrade(&pool.shared)));
114        pool
115    }
116
117    /// The stack each thread gets.
118    pub(crate) fn stack(&self) -> usize {
119        self.stack
120    }
121
122    /// In simulation, only opted-in native calls may reserve a worker.
123    pub(crate) fn simulate(&self, native: bool) {
124        lock(&self.shared.state).mode = if native {
125            Mode::SimNative
126        } else {
127            Mode::SimInline
128        };
129    }
130
131    /// How many threads the pool holds, idle or running a job.
132    #[cfg(test)]
133    pub(crate) fn threads(&self) -> usize {
134        lock(&self.shared.state).threads.len()
135    }
136
137    /// Spawns no thread beyond the bound from now on, so a job that finds
138    /// every thread busy gets none.
139    #[cfg(test)]
140    pub(crate) fn fix_at_bound(&self) {
141        lock(&self.shared.state).fixed = true;
142    }
143
144    /// An idle thread, or a new one, for one job: beyond the bound where
145    /// every thread is busy. `Ok(None)` selects configured inline execution.
146    /// An OS thread creation failure is returned without running the job.
147    pub(crate) fn reserve(&self) -> Result<Option<Worker>, crate::Failure> {
148        self.reserve_for(false)
149    }
150
151    /// Reserves a native-call worker where simulation permits it.
152    pub(crate) fn reserve_native(&self) -> Result<Option<Worker>, crate::Failure> {
153        self.reserve_for(true)
154    }
155
156    /// Waits for a completed simulation worker to become idle or retire.
157    /// Its answer can arrive before the pool has finished returning it.
158    pub(crate) fn wait_for_return(&self, worker: ThreadId) {
159        let state = lock(&self.shared.state);
160        drop(
161            self.shared
162                .returned
163                .wait_while(state, |state| {
164                    !state.closed
165                        && state.threads.iter().any(|(id, _)| *id == worker)
166                        && !state.idle.iter().any(|idle| idle.thread == worker)
167                })
168                .unwrap_or_else(PoisonError::into_inner),
169        );
170    }
171
172    fn reserve_for(&self, native: bool) -> Result<Option<Worker>, crate::Failure> {
173        let mut state = lock(&self.shared.state);
174        if state.bound == 0
175            || state.mode == Mode::SimInline
176            || (state.mode == Mode::SimNative && !native)
177        {
178            return Ok(None);
179        }
180        if let Some(worker) = state.idle.pop() {
181            return Ok(Some(worker));
182        }
183        // Every thread busy: a new one beyond the bound, since a command or
184        // native call run inline instead holds the baton while it waits, and
185        // what it waits for may be another activity's to do.
186        #[cfg(test)]
187        if state.fixed && state.threads.len() >= state.bound {
188            return Ok(None);
189        }
190        let mailbox = Arc::new(Mailbox {
191            mail: Mutex::new(None),
192            arrived: Condvar::new(),
193        });
194        let spawned = std::thread::Builder::new()
195            .name("rexx-pool".to_string())
196            .stack_size(self.stack)
197            .spawn({
198                let (shared, mailbox) = (Arc::clone(&self.shared), Arc::clone(&mailbox));
199                move || work(&shared, &mailbox)
200            })
201            .map_err(|_| spawn_failed())?;
202        let thread = spawned.thread().id();
203        #[cfg(test)]
204        SPAWNED.with(|spawned| spawned.set(spawned.get() + 1));
205        #[cfg(test)]
206        if state.threads.len() >= state.bound {
207            BEYOND.with(|beyond| beyond.set(beyond.get() + 1));
208        }
209        state.threads.push((thread, spawned));
210        Ok(Some(Worker { thread, mailbox }))
211    }
212}

214impl Drop for Pool {
215    /// Ends the idle threads and waits for them; a thread still running a
216    /// job ends once it has run it.
217    fn drop(&mut self) {
218        let (idle, mut threads) = {
219            let mut state = lock(&self.shared.state);
220            state.closed = true;
221            (
222                std::mem::take(&mut state.idle),
223                std::mem::take(&mut state.threads),
224            )
225        };
226        for worker in idle {
227            worker.send(Mail::End);
228            if let Some(at) = threads.iter().position(|(id, _)| *id == worker.thread) {
229                let (_, handle) = threads.swap_remove(at);
230                let _ = handle.join();
231            }
232        }
233    }
234}

236impl Worker {
237    /// The thread's identity, which the baton is lent to.
238    pub(crate) fn thread(&self) -> ThreadId {
239        self.thread
240    }
241
242    /// Runs `job` on the thread.
243    pub(crate) fn run(self, job: Job) {
244        self.send(Mail::Run(job));
245    }
246
247    fn send(&self, mail: Mail) {
248        *lock(&self.mailbox.mail) = Some(mail);
249        self.mailbox.arrived.notify_one();
250    }
251}

291/// A pool thread: runs each job it is sent, then waits idle for the next,
292/// or ends where the pool holds more threads than its bound.
293fn work(shared: &Arc<Shared>, mailbox: &Arc<Mailbox>) {
294    crate::signal::block();
295    loop {
296        let mail = {
297            let mut mail = lock(&mailbox.mail);
298            loop {
299                if let Some(arrived) = mail.take() {
300                    break arrived;
301                }
302                mail = mailbox
303                    .arrived
304                    .wait(mail)
305                    .unwrap_or_else(PoisonError::into_inner);
306            }
307        };
308        let Mail::Run(job) = mail else {
309            return;
310        };
311        job();
312        let mut state = lock(&shared.state);
313        if state.closed {
314            return;
315        }
316        if state.threads.len() > state.bound {
317            let me = std::thread::current().id();
318            if let Some(at) = state.threads.iter().position(|(id, _)| *id == me) {
319                // Dropping the handle detaches this thread, which ends here.
320                state.threads.swap_remove(at);
321            }
322            if state.mode == Mode::SimNative {
323                shared.returned.notify_all();
324            }
325            return;
326        }
327        state.idle.push(Worker {
328            thread: std::thread::current().id(),
329            mailbox: Arc::clone(mailbox),
330        });
331        if state.mode == Mode::SimNative {
332            shared.returned.notify_all();
333        }
334    }
335}
