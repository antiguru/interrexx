12/// `RXSUBCOM_ERROR`.
13const SUBCOM_ERROR: u16 = 1;
14/// `RXSUBCOM_FAILURE`.
15const SUBCOM_FAILURE: u16 = 2;

17impl Interp {
18    /// The handler `name` resolves to in `registry`, as [`capi::resolve`] finds it, with
19    /// this interpreter's library search. A library it loads counts as an external native
20    /// library ([`Interp::load_classic_procedure`]).
21    pub(crate) fn resolve_classic(
22        &mut self,
23        registry: Registry,
24        name: &[u8],
25        start: bool,
26    ) -> Result<Option<ClassicEntry>, u64> {
27        match capi::registration(registry, name, start)? {
28            capi::Registration::Entry(entry) => Ok(Some(entry)),
29            capi::Registration::Library { library, procedure } => self
30                .load_classic_procedure(&library, &procedure)
31                .map(Some)
32                .ok_or(capi::NOT_REGISTERED),
33            capi::Registration::Unbound => Ok(None),
34        }
35    }

37    /// The entry point `procedure` names in the classic library `library`, loaded as
38    /// [`interrexx_api::load::classic_procedure`] loads it. Once it has loaded, waits are no
39    /// longer refused as unending ([`Interp::refuses_unending_waits`]).
40    pub(crate) fn load_classic_procedure(
41        &mut self,
42        library: &[u8],
43        procedure: &[u8],
44    ) -> Option<ClassicEntry> {
45        let entry = interrexx_api::load::classic_procedure(
46            library,
47            procedure,
48            &self.library_search,
49            &self.cwd,
50        )?;
51        self.note_external_library();
52        Some(entry)
53    }

55    /// The classic handler `name` names, kept from an earlier call or resolved now. A lookup
56    /// starts no service.
57    pub(crate) fn classic_function(&mut self, name: &[u8]) -> Option<ClassicEntry> {
58        let upper = name.to_ascii_uppercase();
59        if let Some(entry) = self.classic_functions.get(&upper) {
60            return Some(*entry);
61        }
62        let entry = self
63            .resolve_classic(Registry::Function, &upper, false)
64            .ok()
65            .flatten()?;
66        self.classic_functions.insert(upper, entry);
67        Some(entry)
68    }
69
70    /// Resolves `name` as `RxFuncAdd` does once it has registered it, starting a service if
71    /// needed, and keeps the handler. Whether it resolved.
72    pub(crate) fn resolve_classic_function(&mut self, upper: &[u8]) -> bool {
73        match self.resolve_classic(Registry::Function, upper, true) {
74            Ok(Some(entry)) => {
75                self.classic_functions.insert(upper.to_vec(), entry);
76                true
77            }
78            _ => false,
79        }
80    }

82    /// Calls the classic function `name` with `arguments` as strings. A non-zero
83    /// return is 40.1, its traceback line the routine's built-in routine line.
84    pub(crate) fn call_classic_function(
85        &mut self,
86        name: &[u8],
87        arguments: &[Option<ObjRef>],
88    ) -> Result<Option<ObjRef>, Failure> {
89        let Some(entry) = self.classic_function(name) else {
90            return Err(Raised::routine_not_found(name).into());
91        };
92        self.call_classic_entry(name, arguments, |name, borrowed, queue| {
93            entry.call_function(name, borrowed, queue)
94        })
95    }

97    /// Calls a classic `RexxFunctionHandler` through `call` as the function
98    /// `name`, as [`Interp::call_classic_function`] calls a registered one: a
99    /// row of a package's routine table of the classic style is called the
100    /// same way (r[native.loading.classic-row]). `call` gets the name, the
101    /// arguments as strings and the queue's name.
102    pub(crate) fn call_classic_entry(
103        &mut self,
104        name: &[u8],
105        arguments: &[Option<ObjRef>],
106        call: impl FnOnce(&[u8], &[Option<&[u8]>], &[u8]) -> Option<ClassicResult<usize>>,
107    ) -> Result<Option<ObjRef>, Failure> {
108        let mut texts = Vec::with_capacity(arguments.len());
109        for argument in arguments {
110            texts.push(match argument {
111                Some(value) => Some(self.string_value_text(*value)?),
112                None => None,
113            });
114        }
115        let queue = self.current_queue_name()?;
116        let borrowed: Vec<Option<&[u8]>> = texts.iter().map(Option::as_deref).collect();
117        // r[native.loading.classic-call]: the handler gets the function's name
118        // in capitals, however the call spelled it.
119        let upper = name.to_ascii_uppercase();
120        let called =
121            self.run_classic_handler(&upper, arguments, || call(&upper, &borrowed, &queue))?;
122        if called.rc != 0 {
123            self.blame_native_routine(name, None, arguments, crate::RoutineObject::OfCode, None);
124            return Err(Raised::routine_failed(name).into());
125        }
126        Ok(called.value.map(|bytes| self.text(&bytes)))
127    }

129    /// The name `RXQUEUE('Get')` answers. The `GET` it sends runs Rexx, under a pinned frame:
130    /// a slice of the activity cannot end inside it.
131    pub(crate) fn current_queue_name(&mut self) -> Result<Vec<u8>, Failure> {
132        let Some(queue) = self.local_route(b"STDQUE")? else {
133            return Ok(b"SESSION".to_vec());
134        };
135        let caller = self.caller();
136        let sent = pinned!(
137            self,
138            crate::pinning::PinKind::PullWrapper,
139            self.send_message(queue, b"GET", None, &[], caller)
140        );
141        match sent? {
142            Some(name) => self.string_value_text(name),
143            None => Ok(b"SESSION".to_vec()),
144        }
145    }

147    /// The classic subcommand handler registered for `environment`, kept once resolved: a
148    /// handler dropped afterwards stays usable here. A lookup starts no service.
149    pub(crate) fn classic_subcom(&mut self, environment: &[u8]) -> Option<ClassicEntry> {
150        let upper = environment.to_ascii_uppercase();
151        if let Some(entry) = self.classic_subcoms.get(&upper) {
152            return Some(*entry);
153        }
154        let registered = self
155            .subcom_aliases
156            .get(&upper)
157            .cloned()
158            .unwrap_or_else(|| environment.to_vec());
159        let entry = self
160            .resolve_classic(Registry::Subcom, &registered, false)
161            .ok()
162            .flatten()?;
163        self.classic_subcoms.insert(upper, entry);
164        Some(entry)
165    }

167    /// A non-zero return code is `RC`, else the result string, else 0; the flags raise
168    /// `FAILURE` or `ERROR`. A redirected command is 98.921.
169    pub(crate) fn run_classic_command(
170        &mut self,
171        entry: ClassicEntry,
172        environment: &[u8],
173        command: &[u8],
174        redirected: bool,
175    ) -> Result<CommandOutcome, Failure> {
176        if redirected {
177            return Err(Raised::redirection_not_supported(environment).into());
178        }
179        let (called, flags) =
180            self.run_classic_handler(environment, &[], || Some(entry.call_subcom(command)))?;
181        let text = match (called.rc, called.value) {
182            (0, Some(value)) => value,
183            (0, None) => b"0".to_vec(),
184            (rc, _) => rc.to_string().into_bytes(),
185        };
186        let object = self.text(&text);
187        let status = if flags & SUBCOM_FAILURE != 0 {
188            ReturnStatus::Failure
189        } else if flags & SUBCOM_ERROR != 0 {
190            ReturnStatus::Error
191        } else {
192            ReturnStatus::Normal
193        };
194        Ok(CommandOutcome {
195            shown: None,
196            rc: whole_value(&text).unwrap_or(0),
197            status,
198            supplied: Some(Supplied { object, text }),
199            condition: None,
200        })
201    }
