1//! The interpreter's state on another OS thread: a pool thread reaches it only through what the baton's
2//! holder lends it, and only while it holds the baton.

11/// The baton every interpreter's state is reached under.
12pub(crate) type InterpBaton = crate::baton::Baton<Island>;
13
14/// A value of the interpreter's that moves to another OS thread.
15pub(crate) struct Islanded<T>(T);
16
17/// What an [`Islanded`] may carry: each payload is listed here, and its
18/// argument in the `Send` grant below.
19pub(crate) trait IslandPayload {}
20
21impl IslandPayload for NonNull<Interp> {}
22
23impl IslandPayload for (Box<OffBaton>, interrexx_api::ffi::ThreadContext) {}
24
25// SAFETY: D-U2. The island is reachable only through one root pointer and the
26// baton, and a thread does no refcount operation and no interior access on an
27// island value (derives a borrow, clones or drops an `Rc`, reads or writes a
28// `Cell`) off the baton. A value made here is made on the baton, moves without
29// being touched, and is taken out only by `Islanded::take`, which requires the
30// baton; so no two threads ever touch it at once, which is the property its
31// `Rc` and `Cell` interior needs. A call run where it is made keeps a
32// `ThreadContext` clone across the release, made before the release and
33// dropped after the reacquire; a pool thread's clone moves here and is dropped
34// under a lend. `Sync` is not granted. Per payload: the root pointer
35// `NonNull<Interp>` is dereferenced only by the thread it is lent to, while
36// its lender waits: by `Lent::interp`, and by `HostRef`'s `Deref` and
37// `DerefMut` (`interrexx-api/src/ffi.rs`) for a callback, which receives it through
38// `Island::host`. A native call's `(Box<OffBaton>, ThreadContext)` is taken
39// out by `PooledCall::run` under a lend, and the context, an `Rc`, is dropped
40// under the recall's lend. The box holds no `Rc` or `Cell` (its parts are
41// `ObjRef` words, byte buffers, stub pointers and an `Arc`): it goes back on
42// the activity's record under that lend, or, for a call abandoned while it
43// ran, drops after the lend is given back.
44unsafe impl<T: IslandPayload> Send for Islanded<T> {}
45
46impl<T> Islanded<T> {
47    /// `value`, made by the baton's holder.
48    pub(crate) fn new(value: T, baton: &InterpBaton) -> Islanded<T> {
49        assert!(baton.held_here(), "an island value moved off the baton");
50        Islanded(value)
51    }
52
53    /// The value, for the baton's holder.
54    pub(crate) fn take(self, baton: &InterpBaton) -> T {
55        assert!(baton.held_here(), "an island value taken off the baton");
56        self.0
57    }
58}

60/// The interpreter's root pointer, which the baton's holder lends with it.
61pub(crate) type Island = Islanded<NonNull<Interp>>;
62
63impl Clone for Island {
64    fn clone(&self) -> Island {
65        *self
66    }
67}
68
69impl Copy for Island {}

71impl Island {
72    /// The root pointer of `interp`, which its holder lends while it waits.
73    pub(crate) fn of(interp: &mut Interp) -> Island {
74        Islanded(NonNull::from(interp))
75    }
76
77    /// The interpreter as the API reaches it.
78    pub(crate) fn host(self) -> NonNull<dyn interrexx_api::values::Host> {
79        self.0
80    }
81}

83/// The baton lent to this thread, given back when this drops.
84pub(crate) struct Lent<'b> {
85    baton: &'b InterpBaton,
86    island: Island,
87}
88
89impl<'b> Lent<'b> {
90    /// Waits until the baton is lent to this thread.
91    pub(crate) fn wait(baton: &'b InterpBaton) -> Lent<'b> {
92        Lent {
93            island: baton.await_lend(),
94            baton,
95        }
96    }
97
98    /// The interpreter, for as long as this lend.
99    pub(crate) fn interp(&mut self) -> &mut Interp {
100        // SAFETY: the lender derived the pointer from its own `&mut Interp`
101        // and waits in `Baton::lend`, touching nothing, until this thread
102        // gives the baton back, which only this value's drop does. The answer
103        // borrows `self` uniquely, so it ends before that drop and no second
104        // borrow is live beside it.
105        unsafe { self.island.0.as_mut() }
106    }
107}
108
109impl Drop for Lent<'_> {
110    /// Gives the baton back, except as a panic unwinds: the pool thread's
111    /// job posts the panic first, so the lender drains it before it runs a
112    /// clause.
113    fn drop(&mut self) {
114        if !std::thread::panicking() {
115            self.baton.give_back();
116        }
117    }
118}

120/// Waits until the baton is lent to this thread, which has posted a recall
121/// for it. Where the baton is vacated by a native call running off it
122/// ([`crate::baton::Baton::vacate`]) and nobody serves that recall, this
123/// thread claims the vacancy and hands it to a pool thread it starts to run
124/// the interpreter until the call returns, which serves the recall. Where
125/// the pool gives none, the vacancy is freed again and this thread waits for
126/// the vacating thread to serve it. `None` where `unless_closed` and the
127/// interpreter is gone first, so nothing will lend the baton ([`Closing`]).
128pub(crate) fn wait_served(baton: &InterpBaton, unless_closed: bool) -> Option<Lent<'_>> {
129    let mut claim = true;
130    loop {
131        match baton.await_lend_or_vacancy(claim, unless_closed) {
132            crate::baton::Awaited::Lent(island) => return Some(Lent { baton, island }),
133            crate::baton::Awaited::Closed => return None,
134            crate::baton::Awaited::Vacancy(island) => {
135                // SAFETY: the vacating thread derived the pointer from its
136                // own `&mut Interp` ([`Island::of`]) before it vacated, and
137                // touches the interpreter again only once it has reclaimed
138                // the vacancy, which waits while this thread holds it by its
139                // claim. No other thread holds the baton meanwhile. The borrow
140                // ends before the vacancy is handed on or freed below.
141                let started = unsafe { island.0.as_ref() }.start_vacancy_server();
142                match started {
143                    Some(server) => baton.hand_vacancy(server),
144                    None => {
145                        baton.free_vacancy();
146                        claim = false;
147                    }
148                }
149            }
150        }
151    }
152}

154/// Closes the baton it holds when it drops, as the interpreter holding it
155/// does: a thread still waiting for a lend, a callback whose recall nothing
156/// served, wakes and is refused.
157pub(crate) struct Closing(pub(crate) crate::sync::Arc<InterpBaton>);
158
159impl Drop for Closing {
160    fn drop(&mut self) {
161        self.0.close();
162    }
163}

165// The interpreter moves to another thread only as an `Islanded` payload, and a
166// frame not at all (static_assertions' `assert_not_impl_any`).
167trait AmbiguousIfSend<A> {
168    fn some_item() {}
169}
170impl<T: ?Sized> AmbiguousIfSend<()> for T {}
171impl<T: ?Sized + Send> AmbiguousIfSend<u8> for T {}
172const _: fn() = || {
173    let _ = <Interp as AmbiguousIfSend<_>>::some_item;
174    let _ = <interrexx_core::RegFrame<'static> as AmbiguousIfSend<_>>::some_item;
175};

197        let baton = InterpBaton::new();
198        baton.acquire();
199        let value = Islanded::new(1, &baton);
200        std::thread::scope(|scope| {
201            scope
202                .spawn(|| value.take(&baton))
203                .join()
204                .unwrap_or_else(|payload| std::panic::resume_unwind(payload))
205        });
