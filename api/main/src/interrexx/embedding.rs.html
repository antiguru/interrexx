51/// Each message of `interrexx_inventory::errors::MESSAGES` as a C string, in its order; built once
52/// and kept, since `RexxGetErrorMessage` answers a pointer the caller never frees.
53fn message_texts() -> &'static [std::ffi::CString] {
54    static TEXTS: std::sync::OnceLock<Vec<std::ffi::CString>> = std::sync::OnceLock::new();
55    TEXTS.get_or_init(|| {
56        interrexx_inventory::errors::MESSAGES
57            .iter()
58            .map(|message| std::ffi::CString::new(message.text).unwrap_or_default())
59            .collect()
60    })

63/// The text of the message whose code (major times 1000 plus minor) is
64/// `code`.
65pub fn error_message(code: i32) -> Option<&'static std::ffi::CStr> {
66    let at = interrexx_inventory::errors::MESSAGES
67        .iter()
68        .position(|message| i32::from(message.major) * 1000 + i32::from(message.sub) == code)?;
69    message_texts().get(at).map(std::ffi::CString::as_c_str)

78/// A program `RexxStart` or a translation reads: its name, its text, and whether the text is
79/// an in-store source whose image the caller is given back.
80struct Loaded {
81    path: Vec<u8>,
82    text: Vec<u8>,
83    image_wanted: bool,
84}
85
86/// Reads the program `name` and `instore` give: a file, an image from an earlier call (its
87/// source, where given too, is not read again), an in-store source, or, with neither, the
88/// macro `name`. A refusal is the report written and the negated error.
89fn load(name: &[u8], instore: Option<Instore>) -> Result<Loaded, (Vec<u8>, i32)> {
90    let Some(instore) = instore else {
91        return match launcher::load(name) {
92            Ok((found, text)) => Ok(Loaded {
93                path: found.into_os_string().into_encoded_bytes(),
94                text,
95                image_wanted: false,
96            }),
97            Err((report, _)) => Err((report, UNREADABLE)),
98        };
99    };
100    let unreadable = || (launcher::program_unreadable(name).0, UNREADABLE);
101    match instore {
102        Instore {
103            image: Some(image), ..
104        } => Ok(Loaded {
105            path: name.to_vec(),
106            text: image,
107            image_wanted: false,
108        }),
109        Instore {
110            source: Some(source),
111            image: None,
112        } => Ok(Loaded {
113            path: name.to_vec(),
114            text: source,
115            image_wanted: true,
116        }),
117        Instore {
118            source: None,
119            image: None,
120        } => match crate::builtin::rexxutil::macros::macro_image(name) {
121            Some(text) => Ok(Loaded {
122                path: name.to_vec(),
123                text,
124                image_wanted: false,
125            }),
126            None => Err(unreadable()),
127        },
128    }
129}

131/// Writes a run's output to this process's standard output and error, after what the C
132/// program wrote through stdio.
133fn write_outcome(outcome: &crate::Outcome) {
134    crate::signal::flush_c_streams();
135    let _ = std::io::stdout().write_all(&outcome.stdout);
136    let _ = std::io::stdout().flush();
137    let _ = std::io::stderr().write_all(&outcome.stderr);
138}

140/// Runs `start` in an interpreter of its own, its output on this process's standard output
141/// and error, and answers as `RexxStart` does: 0, the result and, where it is a whole
142/// number a `short` holds, its value; or the negated error number of the condition that
143/// ended it. A single command argument with a leading blank loses it, and a single command
144/// argument `//T` translates the program without running it, leaving the `short` alone.
145pub fn rexx_start(start: Start) -> Started {
146    // The interpreter takes the halting signals, so a SIGINT during the run halts the program
147    // rather than ending the process, and the handlers stay installed once it returns:
148    // `RexxStart` of `call SysSleep 5` answers -4 with Error 4.1 reported, or the trapped
149    // halt's result.
150    crate::timer::install_signals();
151    let Start {
152        name,
153        instore,
154        mut arguments,
155        environment,
156        start,
157        exits,
158    } = start;
159    let command = start == StartType::Command;
160    let translate_only =
161        command && matches!(&arguments[..], [Some(only)] if only.eq_ignore_ascii_case(b"//T"));
162    let loaded = match load(&name, instore) {
163        Ok(loaded) => loaded,
164        Err((report, code)) => {
165            let _ = std::io::stderr().write_all(&report);
166            return Started {
167                code,
168                result: None,
169                short: (!translate_only).then(|| i16::try_from(code).unwrap_or(0)),
170                image: None,
171            };
172        }
173    };
174    // An in-store source's image is saved once it has translated, before it
175    // runs, so a run that fails still leaves it.
176    let image = if loaded.image_wanted {
177        crate::translate_program_with_image(&loaded.path, loaded.text.clone(), Some(false)).1
178    } else {
179        None
180    };
181    if translate_only {
182        let outcome = crate::translate_program(&loaded.path, loaded.text);
183        write_outcome(&outcome);
184        let code = outcome.error.map_or(0, |number| -number);
185        return Started {
186            code,
187            result: None,
188            short: None,
189            image: image.filter(|_| code == 0),
190        };
191    }
192    if command
193        && let [Some(only)] = &mut arguments[..]
194        && only.len() > 1
195        && only[0] == b' '
196    {
197        only.remove(0);
198    }
199    let mut invocation = Invocation::none()
200        .with_input(ProgramInput::Stdin)
201        .with_sinks(Sinks {
202            // After what the C program wrote through stdio, in that order.
203            stdout: Box::new(|bytes| {
204                crate::signal::flush_c_streams();
205                let mut stdout = std::io::stdout();
206                stdout.write_all(bytes).and_then(|()| stdout.flush())
207            }),
208            stderr: Box::new(|bytes| {
209                crate::signal::flush_c_streams();
210                std::io::stderr().write_all(bytes)
211            }),
212            stdout_terminal: std::io::stdout().is_terminal(),
213            unbuffered: crate::unbuffered_requested(),
214            own_descriptors: true,
215        })
216        .with_arguments(arguments)
217        .with_start(start)
218        .with_exits(exits)
219        .with_classic_thread(crate::signal::current_thread())
220        .with_outside_host();
221    if let Some(environment) = environment {
222        invocation = invocation.with_address(environment);
223    }
224    let outcome = run_program_named(&loaded.path, loaded.text, invocation);
225    write_outcome(&outcome);
226    if let Some(number) = outcome.error {
227        // The `rc` is the negated error number too.
228        return Started {
229            code: -number,
230            result: None,
231            short: Some(i16::try_from(-number).unwrap_or(0)),
232            image,
233        };
234    }
235    let short = outcome
236        .result
237        .as_deref()
238        .and_then(|text| std::str::from_utf8(text).ok())
239        .and_then(|text| text.trim().parse::<i16>().ok())
240        .unwrap_or(0);
241    Started {
242        code: 0,
243        result: outcome.result,
244        short: Some(short),
245        image,
246    }
247}

249/// `RexxTranslateProgram` and `RexxCompileProgram`: the program file `input` translated and,
250/// where `output` names a file, its image written there, encoded as `encode` says. An error
251/// is reported and answers its negated number.
252pub fn translate_file(input: &[u8], output: Option<&[u8]>, encode: bool) -> i32 {
253    let loaded = match load(input, None) {
254        Ok(loaded) => loaded,
255        Err((report, code)) => {
256            let _ = std::io::stderr().write_all(&report);
257            return code;
258        }
259    };
260    let (outcome, image) =
261        crate::translate_program_with_image(&loaded.path, loaded.text, output.map(|_| encode));
262    write_outcome(&outcome);
263    if let Some(number) = outcome.error {
264        return -number;
265    }
266    let Some(output) = output else {
267        return 0;
268    };
269    let path = crate::paths::os_path(output);
270    if std::fs::write(
271        path,
272        image.expect("successful output translation yields an image"),
273    )
274    .is_err()
275    {
276        let _ = std::fs::remove_file(path);
277        let (report, _) = launcher::output_unwritable(output);
278        let _ = std::io::stderr().write_all(&report);
279        return UNREADABLE;
280    }
281    0
282}

284/// `RexxTranslateInstoreProgram`: `source`, named `name`, translated, and its image; an error

286pub fn translate_instore(name: &[u8], source: Vec<u8>) -> Result<Vec<u8>, i32> {

304/// A `RexxStart` run registered as reachable by `RexxSetHalt` and `RexxSetTrace`, until this

308impl ClassicRun {

324fn classic_inbox(thread: usize) -> Option<ClassicInbox> {

331/// One interpreter's native calls on one thread: whether any is in flight, and when the
332/// innermost of them was entered.
333///
334/// Only the thread the entry is for writes it, so entering and leaving a call costs no
335/// read-modify-write and takes no lock; another thread reads it only to halt the call, which
336/// races with the call ending however it is read.
337struct OtherEntry {
338    inbox: ClassicInbox,
339    /// The calls in flight.
340    in_flight: AtomicUsize,
341    /// The thread's [`OtherLocal::entered`] count when the innermost call in flight was
342    /// entered, which orders this interpreter's calls against the other interpreters' on the
343    /// thread.
344    entered: AtomicU64,
345}
346
347/// The [`OtherEntry`]s of one thread, reachable by `RexxSetHalt` from any thread while the
348/// thread lives.
349struct OtherThread {
350    thread: usize,
351    entries: std::sync::Mutex<Vec<Arc<OtherEntry>>>,
352}
353
354/// Every live thread that has made a native call, with its entries.
355fn other_threads() -> std::sync::MutexGuard<'static, Vec<Arc<OtherThread>>> {
356    static THREADS: std::sync::Mutex<Vec<Arc<OtherThread>>> = std::sync::Mutex::new(Vec::new());
357    THREADS
358        .lock()
359        .unwrap_or_else(std::sync::PoisonError::into_inner)
360}
361
362/// This thread's side of its [`OtherThread`]: the entries without the lock, and the count
363/// that orders its calls. Dropped when the thread ends, which takes the thread off the list.
364struct OtherLocal {
365    shared: Arc<OtherThread>,
366    entries: std::cell::RefCell<Vec<Arc<OtherEntry>>>,
367    entered: std::cell::Cell<u64>,
368}
369
370impl OtherLocal {
371    fn new() -> OtherLocal {
372        let shared = Arc::new(OtherThread {
373            thread: crate::signal::current_thread(),
374            entries: std::sync::Mutex::new(Vec::new()),
375        });
376        other_threads().push(Arc::clone(&shared));
377        OtherLocal {
378            shared,
379            entries: std::cell::RefCell::new(Vec::new()),
380            entered: std::cell::Cell::new(0),
381        }
382    }
383
384    /// The index of the entry for `inbox`'s interpreter, made where this thread has none.
385    /// A new entry takes the place of one whose interpreter is gone, whose inbox only the
386    /// entry holds, so the list stays as long as the interpreters this thread calls for; an
387    /// entry with a call in flight keeps its place.
388    fn entry(&self, inbox: &ClassicInbox) -> usize {
389        let mut entries = self.entries.borrow_mut();
390        if let Some(at) = entries
391            .iter()
392            .position(|entry| Arc::ptr_eq(&entry.inbox, inbox))
393        {
394            return at;
395        }
396        let gone = |entry: &Arc<OtherEntry>| {
397            entry.in_flight.load(Ordering::Relaxed) == 0 && Arc::strong_count(&entry.inbox) == 1
398        };
399        let entry = Arc::new(OtherEntry {
400            inbox: Arc::clone(inbox),
401            in_flight: AtomicUsize::new(0),
402            entered: AtomicU64::new(0),
403        });
404        let mut shared = self
405            .shared
406            .entries
407            .lock()
408            .unwrap_or_else(std::sync::PoisonError::into_inner);
409        shared.retain(|entry| !gone(entry));
410        shared.push(Arc::clone(&entry));
411        match entries.iter().position(gone) {
412            Some(at) => {
413                entries[at] = entry;
414                at
415            }
416            None => {
417                entries.push(entry);
418                entries.len() - 1
419            }
420        }
421    }
422}
423
424impl Drop for OtherLocal {
425    fn drop(&mut self) {
426        other_threads().retain(|thread| !Arc::ptr_eq(thread, &self.shared));
427    }
428}

430thread_local! {
431    static OTHER_LOCAL: OtherLocal = OtherLocal::new();
432}

434/// A native call in flight on this thread, reachable by `RexxSetHalt` from another thread
435/// until this is dropped, which is on this thread.
436pub(crate) struct OtherRun {
437    /// The place of this thread's entry for the call's interpreter, which keeps it while a
438    /// call is in flight; `None` where the thread is ending and has no list.
439    entry: Option<usize>,
440    /// The entry's `entered` before this call, which its end puts back.
441    outer: u64,
442    /// Dropped on the thread that made it, whose list `entry` indexes.
443    _thread: std::marker::PhantomData<*const ()>,
444}
445
446impl OtherRun {
447    /// Registers the native call this thread is entering, of the interpreter whose inbox is
448    /// `inbox`.
449    pub(crate) fn enter(inbox: &ClassicInbox) -> OtherRun {
450        let entered = OTHER_LOCAL.try_with(|local| {
451            let at = local.entry(inbox);
452            let entered = local.entered.get() + 1;
453            local.entered.set(entered);
454            let entries = local.entries.borrow();
455            let held = &entries[at];
456            let outer = held.entered.load(Ordering::Relaxed);
457            held.entered.store(entered, Ordering::Relaxed);
458            let in_flight = held.in_flight.load(Ordering::Relaxed);
459            held.in_flight.store(in_flight + 1, Ordering::Release);
460            (at, outer)
461        });
462        OtherRun {
463            entry: entered.as_ref().ok().map(|(at, _)| *at),
464            outer: entered.map_or(0, |(_, outer)| outer),
465            _thread: std::marker::PhantomData,
466        }
467    }
468}
469
470impl Drop for OtherRun {
471    fn drop(&mut self) {
472        let Some(at) = self.entry else {
473            return;
474        };
475        let _ = OTHER_LOCAL.try_with(|local| {
476            let entries = local.entries.borrow();
477            let held = &entries[at];
478            held.entered.store(self.outer, Ordering::Relaxed);
479            let in_flight = held.in_flight.load(Ordering::Relaxed);
480            held.in_flight.store(in_flight - 1, Ordering::Release);
481        });
482    }
483}

485/// The inbox of the interpreter whose native call is innermost on `thread`, if one is in
486/// flight there.
487fn other_inbox(thread: usize) -> Option<ClassicInbox> {
488    let threads = other_threads();
489    let on = threads.iter().find(|on| on.thread == thread)?;
490    let entries = on
491        .entries
492        .lock()
493        .unwrap_or_else(std::sync::PoisonError::into_inner);
494    entries
495        .iter()
496        .filter(|entry| entry.in_flight.load(Ordering::Acquire) != 0)
497        .max_by_key(|entry| entry.entered.load(Ordering::Relaxed))
498        .map(|entry| Arc::clone(&entry.inbox))
499}

501/// `RexxSetHalt`: the run on `thread` halted at its next clause; `false` where no run is in
502/// flight there. A `thread` no `RexxStart` run is on that is this thread halts the activity
503/// whose native call this thread runs, as a command handler that names its own thread reaches
504/// it (hostemu's `HI`); a native call on another thread has its interpreter halted as a
505/// `RexxStart` run is (the halt reaches a native function's activity from a thread the
506/// function started).
507pub fn classic_halt(thread: usize) -> bool {
508    let current = crate::signal::current_thread();
509    let inbox = classic_inbox(thread)
510        .or_else(|| (thread != current).then(|| other_inbox(thread)).flatten());
511    let Some(inbox) = inbox else {
512        return thread == current && interrexx_api::ffi::classic::halt_calling();
513    };
514    inbox.post_halt();
515    true
516}

518/// `RexxSetTrace` and `RexxResetTrace`: the external trace of the run on `thread`, on or
519/// off; `false` where no run is in flight there.
520pub fn classic_trace(thread: usize, on: bool) -> bool {
521    let Some(inbox) = classic_inbox(thread) else {
522        return false;
523    };
524    inbox.post(crate::timer::Posted::Trace(on));
525    inbox.requests().set(crate::timer::BOUNDARY);
526    true
527}

558/// Where an instance has got to, under [`Shared::state`].
559#[derive(Default)]
560struct State {
561    /// The root thread context, once the root activity waits in it.
562    context: Option<usize>,
563    /// `Terminate` was called.
564    terminating: bool,
565    /// The interpreter thread has ended.
566    ended: bool,
567}

569/// One instance, as the threads that use it share it.
570#[derive(Default)]
571struct Shared {
572    state: std::sync::Mutex<State>,
573    changed: std::sync::Condvar,
574    inbox: std::sync::OnceLock<crate::sync::Arc<crate::timer::Inbox<crate::timer::Posted>>>,
575    /// The address of the interpreter's instance, which the instance members
576    /// name it by.
577    instance: std::sync::OnceLock<usize>,
578    /// What the interpreter thread left unwritten, once it has ended.
579    outcome: std::sync::Mutex<Option<crate::Outcome>>,
580}
581
582impl Shared {
583    fn state(&self) -> std::sync::MutexGuard<'_, State> {
584        self.state
585            .lock()
586            .unwrap_or_else(std::sync::PoisonError::into_inner)
587    }
588
589    fn wait<'a>(
590        &self,
591        state: std::sync::MutexGuard<'a, State>,
592    ) -> std::sync::MutexGuard<'a, State> {
593        self.changed
594            .wait(state)
595            .unwrap_or_else(std::sync::PoisonError::into_inner)
596    }
597}

599/// The live instances, by instance address.

603fn instances() -> std::sync::MutexGuard<'static, Vec<(usize, std::sync::Arc<Shared>)>> {
604    INSTANCES
605        .lock()
606        .unwrap_or_else(std::sync::PoisonError::into_inner)
607}
608
609fn instance(address: usize) -> Option<std::sync::Arc<Shared>> {
610    instances()
611        .iter()
612        .find(|(at, _)| *at == address)
613        .map(|(_, shared)| std::sync::Arc::clone(shared))
614}
615
616/// Instances not yet waiting in their root call, by the interpreter thread's
617/// id, which [`wait_in_root`] finds its own in.
618static STARTING: std::sync::Mutex<Vec<(usize, std::sync::Arc<Shared>)>> =
619    std::sync::Mutex::new(Vec::new());
620
621/// Holds an instance's [`State::ended`] to be set when it drops, as the
622/// interpreter thread leaves its body, normally or by unwinding.
623struct Ending(std::sync::Arc<Shared>);
624
625impl Drop for Ending {
626    fn drop(&mut self) {
627        self.0.state().ended = true;
628        self.0.changed.notify_all();
629    }
630}
631
632/// Has the root call of each instance panic once `Terminate` has woken it.
633#[cfg(test)]
634static PANIC_IN_ROOT: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
635
636/// `RexxCreateInterpreter`: an interpreter of its own on a thread of its own,
637/// whose root activity waits in a native call that the calling thread, now
638/// attached to it, acts as; answers the root thread context's address, or
639/// `None` where the interpreter did not start.
640pub fn create_instance(options: InstanceOptions) -> Option<usize> {
641    interrexx_api::ffi::set_embedder(&EMBEDDER);
642    // The instance takes the halting signals, which reach every live
643    // interpreter; the caller's mask is its own.
644    crate::timer::install_signals();
645    let shared = std::sync::Arc::new(Shared::default());
646    let starting = std::sync::Arc::clone(&shared);
647    // Marks the instance ended however the thread leaves, a panic included:
648    // the panic is reported by the hook, and `Terminate` returns.
649    let ending = Ending(std::sync::Arc::clone(&shared));
650    // This thread waits for the instance to start, so the instance must not
651    // wait for a library hook this thread is running.
652    let lineage = crate::libraries::lineage();
653    let spawned = std::thread::Builder::new()
654        .name("rexx-interp".to_string())
655        .stack_size(crate::INTERPRETER_STACK_BYTES)
656        .spawn(move || {
657            crate::libraries::adopt(lineage);
658            crate::signal::receive();
659            let (outcome, kept) = crate::run_instance(
660                options,
661                Box::new(move |context, instance, inbox| {
662                    let _ = starting.inbox.set(inbox);
663                    let _ = starting.instance.set(instance);
664                    STARTING
665                        .lock()
666                        .unwrap_or_else(std::sync::PoisonError::into_inner)
667                        .push((context, starting));
668                }),
669            );
670            *ending
671                .0
672                .outcome
673                .lock()
674                .unwrap_or_else(std::sync::PoisonError::into_inner) = Some(outcome);
675            drop(ending);
676            if let Some(interp) = kept {
677                interp.wind_down();
678            }
679        })
680        .ok()?;
681    drop(spawned);
682    let mut state = shared.state();
683    while state.context.is_none() && !state.ended {
684        state = shared.wait(state);
685    }
686    let context = state.context;
687    drop(state);
688    match context {
689        Some(context) => {
690            interrexx_api::ffi::attach_current_thread(context);
691            Some(context)
692        }
693        None => {
694            end(&shared);
695            None
696        }
697    }
698}
699
700/// The root call's body, on the thread running it: the instance is live until
701/// `Terminate`. A thread attached and idle is not waited for, as `terminate`
702/// removes inactive activities; one inside a callback holds the baton the
703/// call's end needs, so the instance ends after it.
704pub(crate) fn wait_in_root(context: usize) {
705    let shared = {
706        let mut starting = STARTING
707            .lock()
708            .unwrap_or_else(std::sync::PoisonError::into_inner);
709        let Some(at) = starting.iter().position(|(at, _)| *at == context) else {
710            return;
711        };
712        starting.remove(at).1
713    };
714    let address = *shared
715        .instance
716        .get()
717        .expect("the interpreter thread sets the instance before its root waits here");
718    instances().push((address, std::sync::Arc::clone(&shared)));
719    let mut state = shared.state();
720    state.context = Some(context);
721    shared.changed.notify_all();
722    while !state.terminating {
723        state = shared.wait(state);
724    }
725    drop(state);
726    #[cfg(test)]
727    if PANIC_IN_ROOT.load(std::sync::atomic::Ordering::SeqCst) {
728        panic!("the instance's root call panicked");
729    }
730}
731
732/// Waits for the instance's interpreter thread to end its run, and writes
733/// what it left unwritten. A call still in flight then is served on that
734/// thread, which ends after it.
735fn end(shared: &Shared) {
736    let mut state = shared.state();
737    while !state.ended {
738        state = shared.wait(state);
739    }
740    drop(state);
741    let outcome = shared
742        .outcome
743        .lock()
744        .unwrap_or_else(std::sync::PoisonError::into_inner)
745        .take();
746    if let Some(outcome) = outcome {
747        let _ = std::io::stdout().write_all(&outcome.stdout);
748        let _ = std::io::stdout().flush();
749        let _ = std::io::stderr().write_all(&outcome.stderr);
750    }
751}
752
753/// The [`interrexx_api::ffi::Embedder`] the instance members reach.
754struct Embedder;
755
756static EMBEDDER: Embedder = Embedder;
757
758impl interrexx_api::ffi::Embedder for Embedder {
759    fn terminate(&self, address: usize) -> bool {
760        let Some(shared) = instance(address) else {
761            return false;
762        };
763        {
764            let mut state = shared.state();
765            if state.terminating {
766                return false;
767            }
768            state.terminating = true;
769            shared.changed.notify_all();
770        }
771        end(&shared);
772        instances().retain(|(at, _)| *at != address);
773        true
774    }
775
776    fn halt(&self, address: usize) -> bool {
777        let Some(shared) = instance(address) else {
778            return false;
779        };
780        if let Some(inbox) = shared.inbox.get() {
781            inbox.post_halt();
782        }
783        true
784    }
785
786    fn set_trace(&self, address: usize, on: bool) -> bool {
787        let Some(shared) = instance(address) else {
788            return false;
789        };
790        if let Some(inbox) = shared.inbox.get() {
791            inbox.post_trace_all(on);
792        }
793        true
794    }
795}
