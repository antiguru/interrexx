1//! The extension libraries this crate runs in Rust: `rxmath`, `rxregexp`,
2//! `rxsock` and `rxunixsys` (`docs/design/extensions.md`).
3//!
4//! A program sees each as a shared library: [`Interp::resolve_library`] answers a
5//! library whose tables name the routines and methods with no address
6//! ([`interrexx_api::load::names_only`]) before any `dlopen`, so `::REQUIRES ...
7//! LIBRARY`, `RxFuncAdd`, `Package~loadLibrary`, `EXTERNAL` directives and
8//! `loadExternalRoutine`/`loadExternalMethod` bind, merge and report it as any
9//! library. Where one of its rows is called, its body here runs in place of a
10//! native call.
11//!
12//! [`Interp::resolve_library`]: crate::Interp::resolve_library
13
14use interrexx_core::ObjRef;
15
16use crate::{Failure, Interp};
17
18/// A routine body: the interpreter and the call's arguments.
19pub(crate) type RoutineBody = fn(&mut Interp, &[Option<ObjRef>]) -> Result<ObjRef, Failure>;
20
21/// A method body: the interpreter, the receiver, the method's scope and the
22/// arguments.
23pub(crate) type MethodBody =
24    fn(&mut Interp, ObjRef, ObjRef, &[Option<ObjRef>]) -> Result<Option<ObjRef>, Failure>;
25
26/// One library.
27pub(crate) struct RustLibrary {
28    /// The library name, matched byte for byte as a shared library's is.
29    name: &'static str,
30    routines: &'static [(&'static str, RoutineBody)],
31    methods: &'static [(&'static str, MethodBody)],
32    /// What loading the library prepares in the process, before any of its
33    /// rows runs.
34    opened: fn(),
35}
36
37impl RustLibrary {
38    /// The library a load of this name answers.
39    pub(crate) fn open(&self) -> interrexx_api::load::Library {
40        let routines: Vec<&str> = self.routines.iter().map(|(name, _)| *name).collect();
41        let methods: Vec<&str> = self.methods.iter().map(|(name, _)| *name).collect();
42        (self.opened)();
43        interrexx_api::load::names_only(self.name, &routines, &methods)
44    }
45}

195/// The library `name` names, or `None` for a name this crate leaves to the
196/// shared-library search.
197pub(crate) fn library(name: &[u8]) -> Option<&'static RustLibrary> {
198    LIBRARIES
199        .iter()
200        .find(|library| library.name.as_bytes() == name)
201}
202
203/// The body of a row of `library`, a library [`RustLibrary::open`] answered,
204/// named `row`; `None` for any other library or row.
205fn of<B: Copy>(
206    library: &interrexx_api::load::Library,
207    row: &[u8],
208    table: fn(&RustLibrary) -> &'static [(&'static str, B)],
209) -> Option<B> {
210    let held = LIBRARIES
211        .iter()
212        .find(|candidate| library.name() == Some(candidate.name.as_bytes()))?;
213    table(held)
214        .iter()
215        .find(|(name, _)| name.as_bytes() == row)
216        .map(|(_, body)| *body)
217}
218
219/// The body of `library`'s routine row named `row`, where that row carries no
220/// address and the library is one of these.
221pub(crate) fn routine_body(
222    library: &interrexx_api::load::Library,
223    row: &[u8],
224) -> Option<RoutineBody> {
225    of(library, row, |held| held.routines)
226}
227
228/// The body of `library`'s method row named `row`, as [`routine_body`].
229pub(crate) fn method_body(
230    library: &interrexx_api::load::Library,
231    row: &[u8],
232) -> Option<MethodBody> {
233    of(library, row, |held| held.methods)
234}
