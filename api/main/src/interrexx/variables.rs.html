12    /// The variable slot `slot` of `frame` names: the frame's own storage,
13    /// unless an `EXPOSE` in this activation bound that slot to a pool on the
14    /// receiving object.
15    #[inline(always)]
16    pub(super) fn variable(&self, frame: SlotFrame, slot: usize) -> Option<ObjRef> {
17        if self.activation_exposes(frame) {
18            return self.exposed_variable(frame, slot);
19        }
20        self.roots.frame_slot(frame, slot)
21    }

23    /// Assigns the variable slot `slot` of `frame` names.
24    #[inline(always)]
25    /// `pub(crate)` for `crate::ir::Op::Store`'s own fast path, which writes
26    /// a simple target's slot without going through `assign_evaluated`.
27    pub(crate) fn set_variable(&mut self, frame: SlotFrame, slot: usize, value: ObjRef) {
28        if self.activation_exposes(frame) {
29            return self.set_exposed_variable(frame, slot, value);
30        }
31        self.roots.set_frame_slot(frame, slot, value);
32    }

34    /// Returns the variable slot `slot` of `frame` names to the uninitialised
35    /// state, which is what `DROP` does. For example, a class method that
36    /// exposes `v`, assigns it and drops it leaves a later `expose v` reading
37    /// the derived name `V`.
38    #[inline(always)]
39    pub(super) fn clear_variable(&mut self, frame: SlotFrame, slot: usize) {
40        if self.activation_exposes(frame) {
41            return self.clear_exposed_variable(frame, slot);
42        }
43        self.roots.clear_frame_slot(frame, slot);
44    }

46    /// The cell slot `slot` of `frame` resolves to, moving the variable into a
47    /// fresh one if it has none yet, so that a reference to it survives the
48    /// frame. `name` is the variable's own name, which a fresh cell keeps.
49    pub(crate) fn promote_slot(&mut self, frame: SlotFrame, slot: usize, name: &[u8]) -> SlotRef {
50        let target = self.roots.activity().slot_ref(frame, slot);
51        if target.cell().is_some() {
52            return target;
53        }
54        let cell = self.roots.new_cell();
55        let object = self.alloc_with(BehaviourId::OBJECT, Body::Cell(cell, name.into()));
56        self.roots.promote(frame, slot, object, cell)
57    }

59    /// The name of the variable that owns the cell slot `slot` of `frame`
60    /// resolves to, following every alias, or `None` when it resolves to frame
61    /// storage.
62    pub(crate) fn cell_owner_name(&self, frame: SlotFrame, slot: usize) -> Option<&[u8]> {
63        let cell = self.roots.activity().slot_ref(frame, slot).cell()?;
64        match &self.heap.get(cell)?.body {
65            Body::Cell(_, name) => Some(name),
66            _ => None,
67        }
68    }

70    /// Whether the running activation has bound any name at all to a scope
71    /// pool over `frame` -- the whole of what the three accessors above test
72    /// before taking the frame, and the reason each of them is two functions.
73    #[inline(always)]
74    fn activation_exposes(&self, frame: SlotFrame) -> bool {
75        let activation = self.activation();
76        activation.exposed.is_some() && activation.frame == frame
77    }

79    /// [`Interp::variable`] for a **named** activation rather than the running
80    /// one -- what `RexxContext~variables` reads a suspended context's pool
81    /// through.
82    pub(crate) fn variable_in(&self, activation: &Activation, slot: usize) -> Option<ObjRef> {
83        let frame = activation.frame;
84        match Interp::exposure_in(activation, frame, slot) {
85            Some(var) => self
86                .pools_of(var.owner)
87                .and_then(|pools| pools.get(var.scope, &var.name)),
88            None => self.roots.frame_slot_of(frame, slot),
89        }
90    }

92    /// [`Interp::variable`]'s exposed half. A slot the list does not name is
93    /// still an ordinary local, so this falls back rather than answering
94    /// unset.
95    #[cold]
96    #[inline(never)]
97    fn exposed_variable(&self, frame: SlotFrame, slot: usize) -> Option<ObjRef> {
98        let activation = self.activation();
99        match Interp::exposure_in(activation, frame, slot) {
100            Some(var) => {
101                let pools = self.pools_of(var.owner)?;
102                match activation.plan.pool_hints.get(slot) {
103                    Some(hint) => pools.get_hinted(var.scope, &var.name, hint),
104                    None => pools.get(var.scope, &var.name),
105                }
106            }
107            None => self.roots.frame_slot(frame, slot),
108        }
109    }

111    /// [`Interp::set_variable`]'s exposed half.
112    #[cold]
113    #[inline(never)]
114    fn set_exposed_variable(&mut self, frame: SlotFrame, slot: usize, value: ObjRef) {
115        // Field by field rather than through `Interp::exposure`, because the
116        // name borrowed out of the activation has to stay live across the
117        // `&mut self.heap` below; disjoint fields borrow independently where a
118        // method taking `&self` would not.
119        let activation = self
120            .activity
121            .running
122            .as_deref()
123            .expect("an activation is running");
124        let Some(var) = Interp::exposure_in(activation, frame, slot) else {
125            self.roots.set_frame_slot(frame, slot, value);
126            return;
127        };
128        let object = Interp::pool_object(&mut self.heap, var.owner);
129        if object.watched() {
130            return self.store_watched_exposed(frame, slot, Some(value));
131        }
132        let pools = Interp::pools_in(object);
133        match activation.plan.pool_hints.get(slot) {
134            Some(hint) => pools.set_hinted(var.scope, &var.name, value, hint),
135            None => pools.set(var.scope, &var.name, value),
136        }
137    }

139    /// [`Interp::clear_variable`]'s exposed half.
140    #[cold]
141    #[inline(never)]
142    fn clear_exposed_variable(&mut self, frame: SlotFrame, slot: usize) {
143        let activation = self
144            .activity
145            .running
146            .as_deref()
147            .expect("an activation is running");
148        let Some(var) = Interp::exposure_in(activation, frame, slot) else {
149            self.roots.clear_frame_slot(frame, slot);
150            return;
151        };
152        let object = Interp::pool_object(&mut self.heap, var.owner);
153        if object.watched() {
154            return self.store_watched_exposed(frame, slot, None);
155        }
156        Interp::pools_in(object).clear(var.scope, &var.name);
157    }

159    /// [`Interp::set_pool_variable`] for each of `entries` in turn, with each
160    /// name a copy of the caller's own.
161    pub(super) fn set_pool_variables(
162        &mut self,
163        owner: ObjRef,
164        scope: ObjRef,
165        entries: &[(&std::rc::Rc<[u8]>, ObjRef)],
166    ) {
167        let object = Interp::pool_object(&mut self.heap, owner);
168        if object.watched() {
169            for &(name, value) in entries {
170                self.set_pool_variable(owner, scope, name, value);
171            }
172            return;
173        }
174        Interp::pools_in(object).set_all(scope, entries);
175    }

177    /// Assigns one name in one scope's pool on `owner`, outside any
178    /// activation's exposure list -- what a generated `::ATTRIBUTE` setter
179    /// writes through.
180    pub(super) fn set_pool_variable(
181        &mut self,
182        owner: ObjRef,
183        scope: ObjRef,
184        name: &[u8],
185        value: ObjRef,
186    ) {
187        let object = Interp::pool_object(&mut self.heap, owner);
188        if object.watched() {
189            return self.store_watched(owner, scope, name, Some(value));
190        }
191        Interp::pools_in(object).set(scope, name, value);
192    }

194    /// [`Interp::set_pool_variable`] for a name the caller bound from a copy
195    /// it keeps: the entry is found by identity
196    /// ([`interrexx_core::ScopePools::set_shared`]), and by name where it is
197    /// not or where a `GUARD WHEN` has watched the object.
198    #[inline]
199    pub(super) fn set_pool_variable_shared(
200        &mut self,
201        owner: ObjRef,
202        scope: ObjRef,
203        name: &std::rc::Rc<[u8]>,
204        value: ObjRef,
205    ) {
206        let object = Interp::pool_object(&mut self.heap, owner);
207        if !object.watched() && Interp::pools_in(object).set_shared(scope, name, value) {
208            return;
209        }
210        self.set_pool_variable(owner, scope, name, value);
211    }

213    /// The value an accessor's or `DELEGATE`'s name holds in `scope`'s pool on
214    /// `owner`: a simple variable's value or derived name, the stem (made if
215    /// the pool has none), or one tail of that stem.
216    pub(crate) fn pool_value(&mut self, owner: ObjRef, scope: ObjRef, name: &[u8]) -> ObjRef {
217        match shape_of(name) {
218            NameShape::Simple => match self
219                .pools_of(owner)
220                .and_then(|pools| pools.get(scope, name))
221            {
222                Some(value) => value,
223                None => self.text(name),
224            },
225            NameShape::Stem => self.pool_stem(owner, scope, name),
226            NameShape::Compound => {
227                let (stem_name, key) = direct_compound(name);
228                let stem = self.pool_stem(owner, scope, stem_name);
229                self.stem_object_compound(stem, key)
230            }
231        }
232    }

234    /// Assigns an accessor's name in `scope`'s pool on `owner`: a stem takes a
235    /// stem as it is and wraps anything else as a new stem's default, and a
236    /// compound writes one tail of the pool's stem.
237    pub(crate) fn set_pool_value(
238        &mut self,
239        owner: ObjRef,
240        scope: ObjRef,
241        name: &[u8],
242        value: ObjRef,
243    ) {
244        match shape_of(name) {
245            NameShape::Simple => self.set_pool_variable(owner, scope, name, value),
246            NameShape::Stem => {
247                let stem = self.stem_assignment_value(name, value);
248                self.set_pool_variable(owner, scope, name, stem);
249            }
250            NameShape::Compound => {
251                let (stem_name, key) = direct_compound(name);
252                let stem = self.pool_stem(owner, scope, stem_name);
253                self.stem_object_write(stem, key, value);
254            }
255        }
256    }

258    /// The stem `scope`'s pool on `owner` holds under `stem_name`, made empty
259    /// if it holds none.
260    fn pool_stem(&mut self, owner: ObjRef, scope: ObjRef, stem_name: &[u8]) -> ObjRef {
261        if let Some(stem) = self
262            .pools_of(owner)
263            .and_then(|pools| pools.get(scope, stem_name))
264        {
265            return stem;
266        }
267        let stem = self.empty_stem(stem_name);
268        self.set_pool_variable(owner, scope, stem_name, stem);
269        stem
270    }

273    /// uninitialised state.
274    pub(super) fn clear_pool_variable(&mut self, owner: ObjRef, scope: ObjRef, name: &[u8]) {
275        let object = Interp::pool_object(&mut self.heap, owner);
276        if object.watched() {
277            return self.store_watched(owner, scope, name, None);
278        }
279        Interp::pools_in(object).clear(scope, name);

282    /// `owner`, whose pools a variable lives in.
283    ///
284    /// # Panics
285    ///
286    /// Where `owner` is not live.
287    fn pool_object(heap: &mut interrexx_core::Heap, owner: ObjRef) -> &mut interrexx_core::Object {
288        heap.get_mut(owner)

293    ///
294    /// # Panics
295    ///
296    /// Where `object` is not a `Body::Instance`.
297    fn pools_in(object: &mut interrexx_core::Object) -> &mut interrexx_core::ScopePools {

304    /// A store of `value`, or a `DROP` where it is `None`, to the exposed
305    /// variable of slot `slot` of `frame`, on an object a `GUARD WHEN` has
306    /// watched.
307    #[cold]
308    #[inline(never)]
309    fn store_watched_exposed(&mut self, frame: SlotFrame, slot: usize, value: Option<ObjRef>) {
310        let Some(var) = self.exposure(frame, slot) else {
311            return;
312        };
313        let (owner, scope, name) = (var.owner, var.scope, std::rc::Rc::clone(&var.name));
314        self.store_watched(owner, scope, &name, value);
315    }

317    /// A store of `value`, or a `DROP` where it is `None`, of `name` in
318    /// `scope`'s pool on `owner`, an object a `GUARD WHEN` has watched: every
319    /// watcher of the variable is posted (on a set and on a drop). The notifier
320    /// then yields, which hands its kernel lock only to an activity already
321    /// queued for it; a woken watcher that is not one runs at the next switch,
322    /// so no switch is asked for here.
323    #[cold]
324    #[inline(never)]
325    fn store_watched(&mut self, owner: ObjRef, scope: ObjRef, name: &[u8], value: Option<ObjRef>) {
326        let pools = Interp::pools_in(Interp::pool_object(&mut self.heap, owner));
327        match value {
328            Some(value) => pools.set(scope, name, value),
329            None => pools.clear(scope, name),
330        }
331        let watchers = self.activities.guards.watchers(owner, scope, name).to_vec();
332        let mark = self.activities.ready.len();
333        for watcher in watchers {
334            self.post_guard(watcher);
335        }
336        if self.sim.is_some() {
337            self.sim_order_event(mark);
338        }
339    }

341    /// A tail of `stem` was assigned or dropped: every activity that watches
342    /// an exposed stem variable holding it is posted, as a store to the
343    /// variable posts it (r[conc.locks.when.stem]). Costs one test where
344    /// no `GUARD WHEN` watches a stem.
345    #[inline]
346    pub(crate) fn stem_changed(&mut self, stem: ObjRef) {
347        if self.activities.guards.watches_a_stem() {
348            self.wake_stem_watchers(stem);
349        }
350    }
351
352    #[cold]
353    #[inline(never)]
354    fn wake_stem_watchers(&mut self, stem: ObjRef) {
355        let mut watchers = Vec::new();
356        for (owner, scope, name, who) in self.activities.guards.stem_watches() {
357            let held = self
358                .pools_of(owner)
359                .and_then(|pools| pools.get(scope, &name));
360            if held == Some(stem) {
361                watchers.extend(who);
362            }
363        }
364        let mark = self.activities.ready.len();
365        for watcher in watchers {
366            self.post_guard(watcher);
367        }
368        if self.sim.is_some() {
369            self.sim_order_event(mark);
370        }
371    }

378    /// [`Interp::exposure`] over one activation, so that a caller holding a
379    /// `&mut` borrow of another `Interp` field can still ask.
380    fn exposure_in(activation: &Activation, frame: SlotFrame, slot: usize) -> Option<&InstanceVar> {
381        if activation.frame != frame {
382            return None;
383        }
384        activation.exposed.as_deref()?.find(slot)
385    }

395    /// The value of a `(name)` selector in a variable list, read as an
396    /// expression reads the symbol, with its `NOVALUE` check and its `TRACE
397    /// I` lines.
398    pub(super) fn read_list_selector(
399        &mut self,
400        code: &Code<'_>,
401        id: SymbolId,
402    ) -> Result<ObjRef, Failure> {
403        let read = match shape_of(code.symbols.name(id).as_bytes()) {
404            NameShape::Simple => SymbolRead::Simple,
405            NameShape::Stem => SymbolRead::Stem,
406            NameShape::Compound => SymbolRead::Compound,
407        };
408        let value = self.read_symbol(code, None, read, id, None)?;
409        self.echo_symbol_read(code, id, value)?;
410        Ok(value)
411    }

418    /// Reads a variable, by the slot the plan already resolved its id to, or
419    /// by `at` when a compiler resolved the same thing earlier.
420    pub(crate) fn read_at(
421        &mut self,
422        code: &Code<'_>,
423        id: SymbolId,
424        at: Option<usize>,
425    ) -> (ObjRef, Novalue) {
426        let slot = match at {
427            Some(slot) => slot,
428            None => match code.slot_for(id) {
429                Some(slot) => slot,
430                None => self.slot_of(code.symbols.name(id).as_bytes()),
431            },
432        };
433        let frame = self.activation().frame;
434        match self.variable(frame, slot) {
435            Some(value) => (value, Novalue::Set),
436            None => (self.unset_read(code, id), Novalue::Unset),
437        }
438    }

456    /// What an uninitialised read yields: the derived name, which for a
457    /// simple variable is its own upcased spelling, as
458    /// [`Interp::novalue_answer`] answers it.
459    #[cold]
460    #[inline(never)]
461    fn unset_read(&mut self, code: &Code<'_>, id: SymbolId) -> ObjRef {
462        let derived = code.symbols.name(id).as_bytes();
463        let derived = self.text(derived);
464        self.novalue_answer(derived)
465    }

468/// A compound name's stem, its period included, and its tail: everything after the
469/// first period, taken literally, as the parser builds it for an attribute or
470/// `DELEGATE` name.
471fn direct_compound(name: &[u8]) -> (&[u8], &[u8]) {
472    let dot = name
473        .iter()
474        .position(|&b| b == b'.')
475        .expect("NameShape::Compound guarantees at least one period");
476    name.split_at(dot + 1)
