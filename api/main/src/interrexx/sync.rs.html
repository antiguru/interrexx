30/// Locks `mutex`. A panic that poisoned it left its state whole: every
31/// critical section here writes its fields before anything can panic.
32pub(crate) fn lock<T>(mutex: &Mutex<T>) -> MutexGuard<'_, T> {
33    mutex.lock().unwrap_or_else(PoisonError::into_inner)
34}

36/// Waits on `condvar`, giving up `guard` meanwhile.
37pub(crate) fn wait<'a, T>(condvar: &Condvar, guard: MutexGuard<'a, T>) -> MutexGuard<'a, T> {
38    condvar.wait(guard).unwrap_or_else(PoisonError::into_inner)
39}

41/// The timer thread's wake source: a socket pair and, on Linux, a timerfd.
42/// The timer thread polls the socket's read end and the timerfd, armed for
43/// its next deadline; on other unix targets the deadline is the poll's
44/// timeout. Registration changes and the signal handlers each write one
45/// byte to the socket's nonblocking write end. The socket is never closed,
46/// so a byte written while no timer thread reads stays until one does.
47#[cfg(all(unix, not(all(loom, test))))]
48pub(crate) struct Wake {
49    reader: Reserved<std::os::unix::net::UnixStream>,
50    writer: Reserved<std::os::unix::net::UnixStream>,
51    #[cfg(any(target_os = "linux", target_os = "android"))]
52    timer: Reserved<std::os::fd::OwnedFd>,
53}
54
55#[cfg(all(unix, not(all(loom, test))))]
56impl Wake {
57    pub(crate) fn new() -> Wake {
58        let (reader, writer) = crate::host::unix_stream_pair().expect("the timer's wake socket");
59        for end in [&reader, &writer] {
60            end.set_nonblocking(true)
61                .expect("a nonblocking wake socket");
62        }
63        // Out of reach of a program's `SockClose(4)`: the signal handler
64        // writes to `writer` by number for the life of the process.
65        Wake {
66            reader: reserve(reader),
67            writer: reserve(writer),
68            #[cfg(any(target_os = "linux", target_os = "android"))]
69            timer: reserve(
70                rustix::time::timerfd_create(
71                    rustix::time::TimerfdClockId::Monotonic,
72                    rustix::time::TimerfdFlags::NONBLOCK | rustix::time::TimerfdFlags::CLOEXEC,
73                )
74                .expect("the timer's deadline timer"),
75            ),
76        }
77    }
78
79    /// Wakes the timer. A full socket already holds a wake.
80    pub(crate) fn notify(&self) {
81        use std::io::Write;
82        let _ = (&*self.writer).write(&[0]);
83    }
84
85    /// Blocks until a wake, `timeout`, or an interrupted poll, taking every
86    /// wake written. The caller recomputes its deadline on every return.
87    pub(crate) fn wait(&self, timeout: Option<Duration>) {
88        use rustix::event::{PollFd, PollFlags, poll};
89        use std::io::Read;
90        if timeout == Some(Duration::ZERO) {
91            return;
92        }
93        #[cfg(any(target_os = "linux", target_os = "android"))]
94        {
95            use rustix::time::{Itimerspec, TimerfdTimerFlags, Timespec, timerfd_settime};
96            let zero = Timespec {
97                tv_sec: 0,
98                tv_nsec: 0,
99            };
100            let deadline = Itimerspec {
101                it_interval: zero,
102                it_value: timeout
103                    .and_then(|timeout| Timespec::try_from(timeout).ok())
104                    .unwrap_or(zero),
105            };
106            let _ = timerfd_settime(&self.timer, TimerfdTimerFlags::empty(), &deadline);
107            let mut polled = [
108                PollFd::new(&self.reader, PollFlags::IN),
109                PollFd::new(&self.timer, PollFlags::IN),
110            ];
111            let _ = poll(&mut polled, None);
112            let _ = rustix::io::read(&self.timer, &mut [0u8; 8]);
113        }
114        #[cfg(not(any(target_os = "linux", target_os = "android")))]
115        {
116            // `poll(2)`'s timeout is a C int of milliseconds, so a wait is
117            // capped at a day and the caller recomputes after it.
118            let timeout = timeout
119                .map(|timeout| timeout.min(Duration::from_secs(86_400)))
120                .and_then(|timeout| rustix::time::Timespec::try_from(timeout).ok());
121            let _ = poll(
122                &mut [PollFd::new(&self.reader, PollFlags::IN)],
123                timeout.as_ref(),
124            );
125        }
126        let _ = (&*self.reader).read(&mut [0u8; 64]);
127    }
128
129    /// Installs the signal handlers, writing to this socket.
130    pub(crate) fn install_signals(&self) {
131        use std::os::fd::AsRawFd;
132        crate::signal::install(self.writer.as_raw_fd());
133    }
134
135    /// Whether a signal arrived since the last call.
136    pub(crate) fn take_signal(&self) -> bool {
137        crate::signal::take_pending()
138    }
139}

141#[cfg(all(windows, not(all(loom, test))))]
142pub(crate) struct Wake {
143    pending: Mutex<bool>,
144    arrived: Condvar,
145}
146
147#[cfg(all(windows, not(all(loom, test))))]
148impl Wake {
149    pub(crate) fn new() -> Self {
150        Self {
151            pending: Mutex::new(false),
152            arrived: Condvar::new(),
153        }
154    }
155
156    pub(crate) fn notify(&self) {
157        *lock(&self.pending) = true;
158        self.arrived.notify_one();
159    }
160
161    pub(crate) fn wait(&self, timeout: Option<Duration>) {
162        if timeout == Some(Duration::ZERO) {
163            return;
164        }
165        let pending = lock(&self.pending);
166        let mut pending = match timeout {
167            Some(timeout) => {
168                self.arrived
169                    .wait_timeout_while(pending, timeout, |pending| !*pending)
170                    .unwrap_or_else(PoisonError::into_inner)
171                    .0
172            }
173            None => self
174                .arrived
175                .wait_while(pending, |pending| !*pending)
176                .unwrap_or_else(PoisonError::into_inner),
177        };
178        *pending = false;
179    }
180
181    pub(crate) fn install_signals(&'static self) {
182        crate::signal::install(self);
183    }
184
185    pub(crate) fn take_signal(&self) -> bool {
186        crate::signal::take_pending()
187    }
188}

190/// The model's wake source: a wake flag under a lock, set by a notify and
191/// taken by a wait, and the signal flag as the handler and the timer use it,
192/// a lock-free atomic. Timeouts are not modelled, so a wait waits for a
193/// notify.
194#[cfg(all(loom, test))]
195pub(crate) struct Wake {
196    woken: Mutex<bool>,
197    arrived: Condvar,
198    signalled: loom::sync::atomic::AtomicBool,
199}
200
201#[cfg(all(loom, test))]
202impl Wake {
203    pub(crate) fn new() -> Wake {
204        Wake {
205            woken: Mutex::new(false),
206            arrived: Condvar::new(),
207            signalled: loom::sync::atomic::AtomicBool::new(false),
208        }
209    }
210
211    pub(crate) fn notify(&self) {
212        *lock(&self.woken) = true;
213        self.arrived.notify_one();
214    }
215
216    pub(crate) fn wait(&self, _timeout: Option<Duration>) {
217        let mut woken = lock(&self.woken);
218        while !*woken {
219            woken = wait(&self.arrived, woken);
220        }
221        *woken = false;
222    }
223
224    /// What a signal handler does: the pending flag, then the wake.
225    pub(crate) fn signal(&self) {
226        self.signalled.store(true, Ordering::Release);
227        self.notify();
228    }
229
230    /// The shipped take, `signal::take_pending`'s.
231    pub(crate) fn take_signal(&self) -> bool {
232        self.signalled
233            .compare_exchange(true, false, Ordering::AcqRel, Ordering::Acquire)
234            .is_ok()
235    }
236}

238/// The timer thread's clock, which slices and wakes on real time.
239#[cfg(not(all(loom, test)))]

245    Instant::now()
246}

248/// The model clock: the instant the model execution began, plus what
249/// [`advance`] added.
250#[cfg(all(loom, test))]
251pub(crate) fn now() -> Instant {
252    let (began, elapsed) = &*CLOCK;
253    *began + Duration::from_nanos(elapsed.load(std::sync::atomic::Ordering::SeqCst))
254}
255
256/// Moves the model clock on by `by`.
257#[cfg(all(loom, test))]
258pub(crate) fn advance(by: Duration) {
259    let nanos = u64::try_from(by.as_nanos()).expect("a model advance fits in u64 nanoseconds");
260    CLOCK
261        .1
262        .fetch_add(nanos, std::sync::atomic::Ordering::SeqCst);
263}

265#[cfg(all(loom, test))]
266loom::lazy_static! {
267    static ref CLOCK: (Instant, std::sync::atomic::AtomicU64) =
268        (Instant::now(), std::sync::atomic::AtomicU64::new(0));
269}

272#[expect(clippy::disallowed_methods, reason = "these tests time real waits")]
273mod tests {
274    use super::Wake;

277    /// A timed wait with nothing written ends at its timeout: the best of
278    /// three 300 ms waits is under 1 ms late where a timerfd holds the
279    /// deadline, and under 5 ms where the poll's timeout, in whole
280    /// milliseconds, does. Measured at niceness 5, a socket read timeout was
281    /// 15 to 24 ms late and a poll timeout about 1.5 ms.
282    #[test]
283    fn a_timed_wait_ends_close_to_its_timeout() {
284        let wake = Wake::new();
285        let timeout = Duration::from_millis(300);
286        let late = (0..3)
287            .map(|_| {
288                let began = Instant::now();
289                wake.wait(Some(timeout));
290                let waited = began.elapsed();
291                assert!(waited >= timeout, "a wait ended early, after {waited:?}");
292                waited - timeout
293            })
294            .min()
295            .expect("three waits");
296        let bound = if cfg!(any(target_os = "linux", target_os = "android")) {
297            Duration::from_millis(1)
298        } else {
299            Duration::from_millis(5)
300        };
301        assert!(late < bound, "{late:?} late");
302    }
303
304    /// A wake written before the wait ends it at once, and the wait takes
305    /// it, so the next wait runs to its timeout.
306    #[test]
307    fn a_written_wake_ends_one_wait() {
308        let wake = Wake::new();
309        wake.notify();
310        let began = Instant::now();
311        wake.wait(Some(Duration::from_secs(5)));
312        assert!(began.elapsed() < Duration::from_secs(1));
313        let began = Instant::now();
314        wake.wait(Some(Duration::from_millis(50)));
315        assert!(began.elapsed() >= Duration::from_millis(50));
316    }
