1//! Where a `::REQUIRES` name is looked for: the routes it is searched over,
2//! the extensions appended to it, and the name the file is then known by.
3
4/// The extension `RESOLVE_REQUIRES` tries ahead of every other, and the one
5/// thing that makes a `::REQUIRES` search differ from an external call's.
6const REQUIRES_EXTENSION: &str = ".cls";
7
8/// The extensions appended, in order.
9const DEFAULT_EXTENSIONS: [&str; 2] = [".REX", ".rex"];
10
11/// The current directory's entry, which `SysSearchPath` adds as this literal
12/// rather than as an expanded path.
13const CURRENT_DIRECTORY: &str = ".";
14
15/// The path entries a `::REQUIRES` inside a package at `program_dir` is
16/// searched over, in order: the requiring program's own directory, the current
17/// directory, `REXX_PATH`, then `PATH`.
18#[cfg(all(test, unix))]
19pub(crate) fn search_entries(
20    program_dir: Option<&str>,
21    rexx_path: Option<&str>,
22    sys_path: Option<&str>,
23) -> Vec<String> {
24    search_entries_with(program_dir, None, rexx_path, sys_path)
25}
26
27/// [`search_entries`] with an embedding instance's `EXTERNAL_CALL_PATH`,
28/// searched after the current directory and before `REXX_PATH`.
29pub(crate) fn search_entries_with(
30    program_dir: Option<&str>,
31    call_path: Option<&str>,
32    rexx_path: Option<&str>,
33    sys_path: Option<&str>,
34) -> Vec<String> {
35    let mut entries: Vec<String> = Vec::new();
36    let mut push = |entry: &str| {
37        if !entry.is_empty() {
38            entries.push(entry.to_string());
39        }
40    };
41    if let Some(dir) = program_dir {
42        push(dir);
43    }
44    push(CURRENT_DIRECTORY);
45    for path in [call_path, rexx_path, sys_path].into_iter().flatten() {
46        #[cfg(unix)]
47        for entry in path.split(':') {
48            push(entry);
49        }
50        #[cfg(windows)]
51        {
52            let bytes = narrow(path);
53            if let Some(path) = crate::paths::os_str(&bytes) {
54                for entry in std::env::split_paths(path) {
55                    push(&widen(entry.as_os_str().as_encoded_bytes()));
56                }
57            }
58        }
59    }
60    entries
61}

63#[cfg(unix)]
64const SEPARATORS: &[char] = &['/'];
65#[cfg(windows)]
66const SEPARATORS: &[char] = &['/', '\\'];

68/// Every path a `::REQUIRES` of `name` tries, in the order it tries them.
69#[cfg(all(test, unix))]
70pub(crate) fn candidates(
71    name: &str,
72    entries: &[String],
73    parent_extension: Option<&str>,
74    requires: bool,
75) -> Vec<Vec<String>> {
76    candidates_with(name, entries, parent_extension, requires, &[])
77}

79/// [`candidates`] with an embedding instance's `EXTERNAL_CALL_EXTENSIONS`,
80/// tried before the default extensions, which follow where they are not among
81/// them.
82pub(crate) fn candidates_with(
83    name: &str,
84    entries: &[String],
85    parent_extension: Option<&str>,
86    requires: bool,
87    instance_extensions: &[String],
88) -> Vec<Vec<String>> {
89    // **ASCII, because `strlower` is.** A name with a non-ASCII letter lowers
90    // to itself here and takes one pass, where Unicode lowercasing would invent
91    // a second spelling that is never tried.
92    let lower = name.to_ascii_lowercase();
93    // `primitiveSearchName`'s own `iterations`: the second pass exists only
94    // where lowercasing changes the name.
95    let spellings: &[&str] = if lower == name {
96        &[name]
97    } else {
98        &[name, &lower]
99    };
100
101    let mut extensions: Vec<Option<&str>> = Vec::new();
102    if has_extension(name) {
103        extensions.push(None);
104    } else {
105        if requires {
106            extensions.push(Some(REQUIRES_EXTENSION));
107        }
108        extensions.extend(parent_extension.map(Some));
109        extensions.extend(instance_extensions.iter().map(|ext| Some(ext.as_str())));
110        extensions.extend(
111            DEFAULT_EXTENSIONS
112                .iter()
113                .filter(|ext| !instance_extensions.iter().any(|own| own == *ext))
114                .map(|ext| Some(*ext)),
115        );
116        extensions.push(None);
117    }
118
119    let mut out: Vec<Vec<String>> = Vec::new();
120    for extension in extensions {
121        for spelling in spellings {
122            let candidate = format!("{spelling}{}", extension.unwrap_or(""));
123            // Qualified enough to name a file on its own, so the path is not
124            // searched at all, as the name search does, and the path search
125            // repeats the test for the same reason.
126            if has_directory(&candidate) {
127                out.push(vec![candidate]);
128                continue;
129            }
130            out.push(
131                entries
132                    .iter()
133                    .map(|entry| {
134                        format!(
135                            "{}{}{candidate}",
136                            entry.trim_end_matches(SEPARATORS),
137                            std::path::MAIN_SEPARATOR
138                        )
139                    })
140                    .collect(),
141            );
142        }
143    }
144    out
145}

147/// What one candidate path's `stat` found.
148#[derive(Copy, Clone, PartialEq, Eq, Debug)]
149pub(crate) enum Found {
150    /// A regular file. The search ends here.
151    Regular,
152    /// Something else that exists -- a directory, a device, a socket.
153    Other,
154    /// Nothing at that path.
155    Missing,
156}
157
158/// The first regular file among `groups`, in order, or `None` when no group
159/// holds one.
160///
161/// **A group is abandoned whole the moment one of its paths exists and is not
162/// a regular file.** `searchPath` returns false there rather than trying its
163/// remaining entries, so a directory named like the target hides a real file
164/// further along the path -- measured, a `main/zdir.rex` directory makes a
165/// call find nothing even though `cwd/zdir.rex` is a file. The next
166/// spelling-and-extension group still runs.
167pub(crate) fn first_regular(
168    groups: &[Vec<String>],
169    mut probe: impl FnMut(&str) -> Found,
170) -> Option<String> {
171    for group in groups {
172        for candidate in group {
173            match probe(candidate) {
174                Found::Regular => return Some(candidate.clone()),
175                Found::Other => break,
176                Found::Missing => {}
177            }
178        }
179    }
180    None
181}

183/// The directory `path` names, trailing separator included, or `None` for a
184/// name with no directory part.
185pub(crate) fn program_directory(path: &str) -> Option<&str> {
186    let first = first_char_len(path)?;
187    path[first..]
188        .rfind(SEPARATORS)
189        .map(|at| &path[..=first + at])
190}
191
192/// The extension `path` ends with, leading dot included, or `None` for a name
193/// whose last component carries none.
194pub(crate) fn program_extension(path: &str) -> Option<&str> {
195    let first = first_char_len(path)?;
196    let at = first + path[first..].rfind(|ch| ch == '.' || SEPARATORS.contains(&ch))?;
197    path[at..].starts_with('.').then(|| &path[at..])
198}
199
200/// The length of `path`'s first char, the byte the backwards scans stop before,
201/// or `None` for an empty path.
202fn first_char_len(path: &str) -> Option<usize> {
203    path.chars().next().map(char::len_utf8)
204}
205
206/// Whether `name`'s last component carries an extension.
207fn has_extension(name: &str) -> bool {
208    program_extension(name).is_some()
209}
210
211/// Whether `name` is checked directly rather than searched for along the
212/// path: an absolute name or one from the home directory. `./name` and
213/// `../name` are searched for like any other relative name
214/// (`r[env.search.names]`).
215fn has_directory(name: &str) -> bool {
216    name.starts_with('~')
217        || name.starts_with(SEPARATORS)
218        || (cfg!(windows) && name.as_bytes().get(1) == Some(&b':'))
219}
220
221/// The file a search for `name` finds, normalised against `cwd`, or `None`
222/// when no route holds one. `program` is the searching program's path, `None`
223/// for a program with no caller.
224///
225/// **Every input is bytes and every candidate is `stat`ed as bytes.** Each
226/// byte is carried through the `str` functions above as the `char` of the
227/// same value, so a path that is not ASCII is never re-encoded, and the
228/// lowercased candidate keeps every byte above 127, as `strlower` does.
229pub(crate) fn search(
230    program: Option<&[u8]>,
231    name: &[u8],
232    rexx_path: Option<&[u8]>,
233    sys_path: Option<&[u8]>,
234    cwd: &[u8],
235    requires: bool,
236) -> Option<Vec<u8>> {
237    search_with(
238        program,
239        name,
240        &Instance::default(),
241        rexx_path,
242        sys_path,
243        cwd,
244        requires,
245    )
246}

248/// An embedding instance's own search settings: `EXTERNAL_CALL_PATH` and
249/// `EXTERNAL_CALL_EXTENSIONS`.
250#[derive(Default, Clone, Debug)]
251pub(crate) struct Instance {
252    pub(crate) path: Option<Vec<u8>>,
253    pub(crate) extensions: Vec<Vec<u8>>,
254}
255
256/// [`search`] with an embedding instance's settings.
257pub(crate) fn search_with(
258    program: Option<&[u8]>,
259    name: &[u8],
260    instance: &Instance,
261    rexx_path: Option<&[u8]>,
262    sys_path: Option<&[u8]>,
263    cwd: &[u8],
264    requires: bool,
265) -> Option<Vec<u8>> {
266    let program = program.map(widen);
267    let program = program.as_deref();
268    let entries = search_entries_with(
269        program.and_then(program_directory),
270        instance.path.as_deref().map(widen).as_deref(),
271        rexx_path.map(widen).as_deref(),
272        sys_path.map(widen).as_deref(),
273    );
274    let extension = program.and_then(program_extension);
275    let extensions: Vec<String> = instance.extensions.iter().map(|ext| widen(ext)).collect();
276    let groups = candidates_with(&widen(name), &entries, extension, requires, &extensions);
277    let located = |candidate: &str| crate::paths::normalize_bytes(&narrow(candidate), cwd);
278    let hit = first_regular(&groups, |candidate| {
279        match std::fs::metadata(crate::paths::os_path(&located(candidate))) {
280            Ok(meta) if meta.is_file() => Found::Regular,
281            Ok(_) => Found::Other,
282            Err(_) => Found::Missing,
283        }
284    })?;
285    Some(located(&hit))
286}
287
288/// `bytes` with each byte as the `char` of the same value.
289fn widen(bytes: &[u8]) -> String {
290    bytes.iter().copied().map(char::from).collect()
291}
292
293/// [`widen`]'s inverse. Every candidate [`search`] builds is widened bytes
294/// and ASCII, so no char is above 255; one that were would become `?` and
295/// name a file the search does not find.
296fn narrow(text: &str) -> Vec<u8> {
297    text.chars()
298        .map(|c| u8::try_from(c).unwrap_or(b'?'))
299        .collect()
300}

464    /// A path that exists and is not a regular file abandons the rest of its
465    /// own spelling and extension, and only that group.
466    #[test]
467    fn a_non_regular_hit_abandons_the_rest_of_its_group() {
468        let groups = vec![
469            vec!["/main/zdir.rex".to_string(), "/cwd/zdir.rex".to_string()],
470            vec!["/main/zdir".to_string(), "/cwd/zdir".to_string()],

474        assert_eq!(
475            first_regular(&groups, |path| match path {
476                "/main/zdir.rex" => Found::Other,
477                "/cwd/zdir.rex" | "/cwd/zdir" => Found::Regular,

484        assert_eq!(

486                "/cwd/zdir.rex" | "/cwd/zdir" => Found::Regular,

488            })
489            .as_deref(),
490            Some("/cwd/zdir.rex")
491        );
492        assert_eq!(first_regular(&groups, |_| Found::Missing), None);

541#[cfg(all(test, windows))]
542mod windows_tests {
543    use super::*;
544
545    #[test]
546    fn searches_a_quoted_path_entry_and_accepts_a_qualified_name() {
547        let cwd = std::env::temp_dir();
548        let folder = cwd.join(format!("interrexx-{};search", std::process::id()));
549        std::fs::create_dir_all(&folder).expect("create search directory");
550        let script = folder.join("échantillon.rex");
551        std::fs::write(&script, b"say 2 + 2").expect("write Rexx fixture");
552        let path = std::env::join_paths([cwd.join("missing"), folder.clone()]).expect("PATH");
553        let cwd = cwd.as_os_str().as_encoded_bytes();
554        let qualified = script.as_os_str().as_encoded_bytes();
555        let found = search(
556            None,
557            "échantillon".as_bytes(),
558            Some(path.as_encoded_bytes()),
559            None,
560            cwd,
561            false,
562        );
563        let direct = search(None, qualified, None, None, cwd, false);
564        let found = found.and_then(|name| std::fs::canonicalize(crate::paths::os_path(&name)).ok());
565        let expected = std::fs::canonicalize(&script).expect("canonicalize fixture");
566        std::fs::remove_dir_all(&folder).expect("remove search directory");
567        assert_eq!(found, Some(expected));
568        assert_eq!(direct.as_deref(), Some(qualified));
569    }
