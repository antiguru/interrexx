6/// **Every [`Op::TraceClause`] is the first op of a [`Op::Clause`] region**,
7/// which is both halves of that op's own contract at once.
8pub(super) fn assert_trace_ops_open_a_clause_region(ops: &[Op]) {
9    for (at, op) in ops.iter().enumerate() {
10        if !matches!(op, Op::TraceClause { .. }) {
11            continue;
12        }
13        let opens_here = at
14            .checked_sub(1)
15            .and_then(|before| ops.get(before))
16            .is_some_and(|before| matches!(before, Op::Clause { end, .. } | Op::CallingClause { end, .. } if usize::cast_from(*end) > at));
17        assert!(
18            opens_here,
19            "the trace op at {at} is not the first op of a Clause region, so it echoes against \
20             another clause's indent or after a value line it has to precede"
21        );
22    }
23}

25/// **Every [`Op::TraceLiteral`] sits immediately behind the [`Op::Const`] or
26/// [`Op::LoadConstant`] whose own register it reads**, which is both halves of
27/// that op's contract at once.
28pub(super) fn assert_literal_echoes_follow_their_load(ops: &[Op]) {
29    for (at, op) in ops.iter().enumerate() {
30        let Op::TraceLiteral { src } = op else {
31            continue;
32        };
33        let loads_it = at
34            .checked_sub(1)
35            .and_then(|before| ops.get(before))
36            .is_some_and(|before| {
37                matches!(
38                    before,
39                    Op::Const { dst, .. } | Op::LoadConstant { dst, .. } if dst == src
40                )
41            });
42        assert!(
43            loads_it,
44            "the literal echo at {at} does not follow the load of the register it reads, so it \
45             echoes a value that op did not put there"
46        );
47    }
48}

50/// **Every [`Op::TraceRead`] sits immediately behind the [`Op::Load`] it
51/// echoes**, reading that op's register and repeating its symbol and its read
52/// kind -- all three halves of that op's contract at once.
53pub(super) fn assert_read_echoes_follow_their_load(ops: &[Op]) {
54    for (at, op) in ops.iter().enumerate() {
55        let Op::TraceRead { symbol, read, src } = op else {
56            continue;
57        };
58        let loads_it = at
59            .checked_sub(1)
60            .and_then(|before| ops.get(before))
61            .is_some_and(|before| {
62                matches!(
63                    before,
64                    Op::Load {
65                        symbol: loaded,
66                        read: kind,
67                        dst,
68                        ..
69                    } if loaded == symbol && kind == read && dst == src
70                )
71            });
72        assert!(
73            loads_it,
74            "the read echo at {at} does not follow the load of the symbol and register it \
75             names, so it echoes a value or a name that op did not put there"
76        );
77    }
78}

80/// **Every [`Op::TraceOperator`] sits immediately behind the [`Op::Arith`] or
81/// [`Op::Binary`] it echoes**, reading that op's destination register and
82/// repeating its operator.
83pub(super) fn assert_operator_echoes_follow_their_op(ops: &[Op]) {
84    for (at, op) in ops.iter().enumerate() {
85        let Op::TraceOperator { op: echoed, src } = op else {
86            continue;
87        };
88        let computes_it = at
89            .checked_sub(1)
90            .and_then(|before| ops.get(before))
91            .is_some_and(|before| {
92                matches!(
93                    before,
94                    Op::Arith { op: applied, dst, .. } | Op::Binary { op: applied, dst, .. }
95                        if applied == echoed && dst == src
96                )
97            });
98        assert!(
99            computes_it,
100            "the operator echo at {at} does not follow the operation whose operator and register \
101             it names, so it echoes a value or a sign that op did not put there"
102        );
103    }
104}

130/// **Every [`Op::TraceFunction`] sits immediately behind the [`Op::CallExpr`]
131/// it echoes**, reading that op's destination register and repeating its whole
132/// address.
133pub(super) fn assert_call_echoes_follow_their_op(ops: &[Op]) {
134    for (at, op) in ops.iter().enumerate() {
135        let Op::TraceFunction {
136            index: echoed,
137            slot: echoed_slot,
138            path: echoed_path,
139            src,
140        } = op
141        else {
142            continue;
143        };
144        let runs_it = at
145            .checked_sub(1)
146            .and_then(|before| ops.get(before))
147            .is_some_and(|before| {
148                match before {
149                    Op::CallExpr {
150                        index,
151                        slot,
152                        path,
153                        dst,
154                        ..
155                    } => {
156                        index == echoed && slot == echoed_slot && path == echoed_path && dst == src
157                    }
158                    // No `index` of its own to compare -- `Op::CallArgs`'s own
159                    // doc has why it carries none, and this check is what
160                    // gives it the echo's instead.
161                    Op::CallArgs {
162                        slot, path, dst, ..
163                    }
164                    | Op::Reference { slot, path, dst }
165                    | Op::ClassAt { slot, path, dst } => {
166                        slot == echoed_slot && path == echoed_path && dst == src
167                    }
168                    _ => false,
169                }
170            });
171        assert!(
172            runs_it,
173            "the function echo at {at} does not follow the call whose address and register it \
174             names, so it echoes a value that call did not produce or reads a node that call \
175             did not run"
176        );
177    }
178}

180/// **Every [`Op::TraceKeyword`] is immediately followed by the
181/// [`Op::LoopHeaderValue`] that files the value it echoes**, reading that op's
182/// register and naming that op's role.
183pub(super) fn assert_keyword_echoes_precede_their_value(ops: &[Op]) {
184    for (at, op) in ops.iter().enumerate() {
185        let Op::TraceKeyword { role: echoed, src } = op else {
186            continue;
187        };
188        let files_it = ops.get(at + 1).is_some_and(|next| {
189            matches!(
190                next,
191                Op::LoopHeaderValue { role, src: filed } if role == echoed && filed == src
192            )
193        });
194        assert!(
195            files_it,
196            "the keyword echo at {at} is not in front of the header value it names, so it echoes \
197             a value or a keyword that op does not file"
198        );
199    }
200}

202/// Every [`Op::Exec`] region holds nothing but its own [`Op::Clause`], the
203/// optional [`Op::TraceClause`] between them, and the `Exec` itself.
204pub(super) fn assert_exec_regions_hold_nothing_else(ops: &[Op]) {
205    for (at, op) in ops.iter().enumerate() {
206        let (Op::Clause { end, .. } | Op::CallingClause { end, .. }) = op else {
207            continue;
208        };
209        let inside = &ops[at + 1..usize::cast_from(*end).min(ops.len())];
210        if !inside.iter().any(|op| matches!(op, Op::Exec { .. })) {
211            continue;
212        }
213        for (offset, op) in inside.iter().enumerate() {
214            assert!(
215                matches!(op, Op::Exec { .. } | Op::TraceClause { .. }),
216                "op {} sits in the Exec region of clause {at}, which may hold                  nothing but its echo and the Exec itself",
217                at + 1 + offset
218            );
219        }
220    }
221}

223/// **Every index-bearing op inside a [`Op::Clause`] region names that region's
224/// own clause.**
225pub(super) fn assert_region_ops_name_their_clause(ops: &[Op]) {
226    for (at, op) in ops.iter().enumerate() {
227        let (Op::Clause { index, end }
228        | Op::CallingClause { index, end }
229        | Op::LoopTest { index, end }) = op
230        else {
231            continue;
232        };
233        for (inside, op) in ops[at + 1..usize::cast_from(*end).min(ops.len())]
234            .iter()
235            .enumerate()
236        {
237            let named = match op {
238                Op::TraceClause { index }
239                | Op::EvalExpr { index, .. }
240                | Op::Store { index, .. }
241                | Op::Say { index, .. }
242                | Op::Return { index, .. }
243                | Op::Queue { index, .. }
244                | Op::WhenTest { index, .. }
245                | Op::Call { index, .. }
246                | Op::CallNamed { index, .. }
247                | Op::Message { index }
248                | Op::Expose { index }
249                | Op::Exec { index }
250                | Op::Escape { index }
251                | Op::CallExpr { index, .. }
252                | Op::TraceFunction { index, .. }
253                | Op::ConditionJump { index, .. }
254                | Op::LoopRun { index }
255                | Op::LoopNext { index }
256                | Op::LoopTested { index, .. }
257                | Op::SelectCase { index, .. }
258                | Op::ExecValue { index, .. }
259                | Op::ExecWith { index, .. }
260                | Op::ReplyCheck { index }
261                | Op::RaisePart { index, .. }
262                | Op::ForwardPart { index, .. }
263                | Op::Signal { index, .. }
264                | Op::ParseBegin { index, .. }
265                | Op::ParseRun { index, .. } => Some(*index),
266                Op::TraceKeyword { .. }
267                | Op::LoopHeaderValue { .. }
268                | Op::Clause { .. }
269                | Op::CallingClause { .. }
270                | Op::LoopTest { .. }
271                | Op::Send { .. }
272                | Op::SendWith { .. }
273                | Op::SendScope { .. }
274                | Op::Reference { .. }
275                | Op::ClassAt { .. }
276                | Op::List { .. }
277                | Op::EndBranch
278                | Op::EndWhen
279                | Op::EnterWhen { .. }
280                | Op::EnterOtherwise { .. }
281                | Op::Const { .. }
282                | Op::LoadConstant { .. }
283                | Op::TraceLiteral { .. }
284                | Op::Load { .. }
285                | Op::TraceRead { .. }
286                | Op::Arith { .. }
287                | Op::Binary { .. }
288                | Op::TraceOperator { .. }
289                | Op::Prefix { .. }
290                | Op::TracePrefix { .. }
291                | Op::Jump { .. }
292                | Op::PushArg { .. }
293                | Op::TraceArgument { .. }
294                | Op::CallArgs { .. }
295                | Op::UseArg(..)
296                | Op::ParseStore { .. }
297                | Op::JumpUnless { .. } => None,
298            };
299            assert!(
300                named.is_none_or(|named| named == *index),
301                "the op at {} names instruction {} inside the region of clause {index}, so the \
302                 driver would hand it the wrong instruction",
303                at + 1 + inside,
304                named.unwrap_or_default()
305            );
306        }
307    }
