253pub(crate) enum FrameKind {
254    Select(SelectFrame),
255    /// Boxed so that the stack's element stays near a `SelectFrame`'s width:
256    /// a loop's state holds the header's `Number`s and is several times that.

279        Frame {
280            op_end: frame.op_end,
281            start: frame.start,
282            end: frame.end,
283            kind: FrameKind::Select(frame),
284        }

287    /// The frame over the labelled block at `index` whose `END` is

293        label: Option<SymbolId>,

301                label,

304        })
305    }

309            op_end: u32::MAX,

431/// The ops of one clause region, `$ops`, in order, as `Op::Clause`'s arm
432/// runs them from `$from`. The region leaves through `break $region` with the
433/// op the counter goes to, `break $cold` with its `Result<RegionEnd,
434/// Failure>`, or, where `$top` holds, `break $park` with the [`Park`] of a call
435/// op that entered its callee.
436macro_rules! region_ops {
437    (
438        $self:ident,
439        $code:ident,
440        $chunk:ident,
441        $registers:ident,
442        $source:ident,
443        $clause:ident,
444        $index:ident,
445        $stale:ident,
446        $header:ident,
447        $end:ident,
448        $ops:ident,
449        $from:expr,
450        $top:expr,
451        $cold:lifetime,
452        $region:lifetime,
453        $park:lifetime
454    ) => {
455        let mut remaining = $ops.iter();
456        'region_loop: while let Some(region_op) = remaining.next() {
457            let deliver: Deliver = 'entered: {
458            match region_op {
459                // **No gate**: this op exists only in a chunk
460                // compiled under a setting that echoes, which is
461                // the decision. `stale` is the one thing that
462                // can withdraw it, and then the clause unit has
463                // already asked the current setting instead.
464                Op::TraceClause { $index } => {
465                    debug_assert_names_the_clause(
466                        $code,
467                        *$index,
468                        $clause,
469                        "TraceClause",
470                    );
471                    if !$stale {
472                        #[cfg(test)]
473                        count_trace_op_echo();
474                        // The indent this clause's own entry
475                        // computed, read back rather than
476                        // recomputed: `enter_stepped_clause`
477                        // sets this field to `printed_indent`
478                        // for the clause it is opening and
479                        // nothing between there and here writes
480                        // it, so the two engines cannot come to
481                        // print an echo at two different indents
482                        // for one clause.
483                        let indent =
484                            $self.activity.clause_state.current_value_indent;
485                        $self.echo_compiled_clause($source, $clause, indent);
486                        // A failed delivery abandons the clause it echoes.
487                        if !$self.activity.pending_traps.is_empty()
488                            && let Some(failure) = $self.take_trace_failure()
489                        {
490                            break $cold Err(failure);
491                        }
492                    }
493                }
494                // **The one thing this does that `EvalExpr`
495                // does not is start from the site's kept
496                // resolution.** The argument loop, the `>A>`
497                // lines, the activation bookkeeping and the
498                // three `Ended` arms are the same functions
499                // `eval.rs` calls on the same node.
500                Op::CallExpr {
501                    $index,
502                    slot,
503                    path,
504                    site,
505                    dst,
506                } => {
507                    debug_assert!(
508                        $chunk.holds_register(*dst),
509                        "op writes register {dst} outside the region the chunk \
510                     reserved"
511                    );
512                    debug_assert_names_the_clause(
513                        $code, *$index, $clause, "CallExpr",
514                    );
515                    if $top {
516                        match $self.begin_call_expr($code, $chunk, $clause, *slot, *path, *site)
517                        {
518                            Ok(Started::Ran(value)) => $registers.set(*dst, value),
519                            Ok(Started::Entered) => {
520                            break 'entered Deliver::Register(*dst);
521                            }
522                            Err(failure) => break $cold Err(failure),
523                        }
524                        continue 'region_loop;
525                    }
526                    match $self.run_call_expr($code, $chunk, $clause, *slot, *path, *site) {
527                        Ok(value) => {
528                            $registers.set(*dst, value);
529                        }
530                        Err(failure) => break $cold Err(failure),
531                    }
532                }
533                // **Every one of these bodies is behind
534                // a call.** What they do is small, but
535                // what they *contain* is not -- a `Vec`
536                // push carries its own growth path and an
537                // argument echo builds a rendering -- and
538                // inlining either into the driver's frame
539                // is paid by every op in the stream.
540                // Measured with the bodies written out
541                // here, `varlookup` -- which never
542                // executes one -- retired 5.97% more
543                // instructions, `compound` 4.14%.
544                Op::PushArg { src } => {
545                    $self.push_call_arg($registers, *src);
546                }
547                Op::TraceArgument { src } => {
548                    if !$self.tracing_intermediates() {
549                        continue 'region_loop;
550                    }
551                    if $self.trace_call_arg_op($registers, *src) {
552                        break $cold Err($self.take_held_trace_failure());
553                    }
554                }
555                Op::CallArgs {
556                    slot,
557                    path,
558                    site,
559                    argc,
560                    dst,
561                } => {
562                    debug_assert!(
563                        $chunk.holds_register(*dst),
564                        "op writes register {dst} outside the region the chunk \
565                     reserved"
566                    );
567                    // **The body is behind a call and not
568                    // written here**, which is layout
569                    // rather than style: a call's
570                    // resolution, its argument run and its
571                    // outcome are wide, and inlining them
572                    // into the driver's own frame raises
573                    // the register pressure every *other*
574                    // op then pays. Measured: with this
575                    // body inline, `varlookup` -- which
576                    // calls nothing at all -- retired 4.9%
577                    // more instructions.
578                    if $top {
579                        match $self.begin_call_args(
580                            $code, $chunk, $clause, *slot, *path, *site, *argc,
581                        ) {
582                            Ok(Started::Ran(value)) => $registers.set(*dst, value),
583                            Ok(Started::Entered) => {
584                            break 'entered Deliver::Register(*dst);
585                            }
586                            Err(failure) => break $cold Err(failure),
587                        }
588                        continue 'region_loop;
589                    }
590                    match $self.run_call_args(
591                        $code, $chunk, $clause, *slot, *path, *site, *argc,
592                    ) {
593                        Ok(value) => {
594                            $registers.set(*dst, value);
595                        }
596                        Err(failure) => break $cold Err(failure),
597                    }
598                }
599                // The `>F>` line the op above owes, emitted
600                // behind it because `eval`'s own hook is
601                // post-order and this is the same line.
602                Op::TraceFunction {
603                    $index,
604                    slot,
605                    path,
606                    src,
607                } => {
608                    debug_assert_names_the_clause(
609                        $code,
610                        *$index,
611                        $clause,
612                        "TraceFunction",
613                    );
614                    // **The gate in front of the descent, not
615                    // only inside `trace_intermediate`.** That
616                    // function returns immediately under the
617                    // same condition, so this changes no
618                    // output; what it changes is that an
619                    // untraced run walks no path to reach a
620                    // node it is not going to print.
621                    if !$self.tracing_intermediates() {
622                        continue 'region_loop;
623                    }
624                    let value = $registers.get(*src);
625                    if let Err(failure) =
626                        $self.trace_intermediate_op($code, $clause, *slot, *path, value)
627                    {
628                        break $cold Err(failure);
629                    }
630                }
631                // A send whose receiver and arguments ops of this region
632                // computed. Its body is behind a call for the reason
633                // `Op::CallArgs`'s arm gives.
634                Op::Send { recv, dst, .. } | Op::SendWith { recv, dst, .. } => {
635                    debug_assert!(
636                        $chunk.holds_register(*recv) && $chunk.holds_register(*dst),
637                        "op reads or writes a register outside the region the chunk \
638                     reserved"
639                    );
640                    if $top {
641                        match $self.begin_send_op($chunk, $registers, region_op) {
642                            Ok(Started::Ran(value)) => $registers.set(*dst, value),
643                            Ok(Started::Entered) => {
644                            break 'entered Deliver::Send;
645                            }
646                            Err(failure) => break $cold Err(failure),
647                        }
648                        continue 'region_loop;
649                    }
650                    if let Err(failure) =
651                        $self.run_send_op($chunk, $registers, region_op)
652                    {
653                        break $cold Err(failure);
654                    }
655                }
656                Op::Reference { slot, path, dst } => {
657                    debug_assert!(
658                        $chunk.holds_register(*dst),

669                        "op writes register {dst} outside the region the chunk reserved"

699                        $chunk.holds_register(*dst),
700                        "op writes register {dst} outside the region the chunk \
701                     reserved"
702                    );

704                        $code, *$index, $clause, "EvalExpr",
705                    );

707                        $self,
708                        crate::pinning::PinKind::TreeEval,
709                        $self.eval_chunk_expr($code, $clause, *slot)
710                    ) {
711                        Ok(value) => value,
712                        Err(failure) => break $cold Err(failure),
713                    };
714                    $registers.set(*dst, value);
715                }
716                // **A native expression op**:
717                // the literal's value, built from the chunk's
718                // own interned bytes through the same
719                // `Interp::literal` that `eval_node`'s `Literal`
720                // arm calls, with `eval.rs` not entered at all.
721                // It emits nothing -- `Op::TraceLiteral` below
722                // is the line that loading a literal owes.
723                Op::Const { dst, konst } => {
724                    debug_assert!(
725                        $chunk.holds_register(*dst),
726                        "op writes register {dst} outside the region the chunk \
727                     reserved"
728                    );
729                    // **Built once for the whole run.** The
730                    // value is interned into the chunk on the
731                    // first execution and read from it after
732                    // that, which is what stops this op
733                    // allocating a fresh object for a literal
734                    // longer than the handle carries inline --
735                    // 560,000 times on the pinned `rexxcps`
736                    // before this. `Chunk::interned`'s own doc
737                    // has why the shared value is safe and why
738                    // `NIL` is the empty state.
739                    let mut value = $chunk.interned_konst(*konst);
740                    if value == ObjRef::NIL {
741                        let Some(bytes) = $chunk.konst(*konst) else {
742                            break $cold Err(
743                                Loud::constant_out_of_range().into()
744                            );
745                        };
746                        #[cfg(test)]
747                        count_const_build();

749                        debug_assert_ne!(
750                            value,
751                            ObjRef::NIL,
752                            "a constant interned to the handle that means \
753                         'not built yet', so it would be rebuilt on every \
754                         execution and the cache would be dead code"
755                        );
756                        $chunk.remember_konst(*konst, value);
757                    }
758                    $registers.set(*dst, value);
759                }
760                // A constant symbol's own value: its upcased
761                // spelling, built through the same
762                // `Interp::literal` `Op::Const` above uses and
763                // read out of the symbol table `code` carries,
764                // which is what `eval_node`'s own `Constant`
765                // arm does. It emits nothing -- the
766                // `Op::TraceLiteral` below is the line it owes
767                // too, because both print `>L>`.
768                Op::LoadConstant { symbol, dst } => {
769                    debug_assert!(
770                        $chunk.holds_register(*dst),
771                        "op writes register {dst} outside the region the chunk \
772                     reserved"
773                    );
774                    // Interned exactly as `Op::Const`'s value
775                    // is, and keyed by the symbol's own id --
776                    // so the symbol table is read once for the
777                    // whole run rather than on every execution.
778                    let mut value = $chunk.interned_symbol(*symbol);
779                    if value == ObjRef::NIL {
780                        #[cfg(test)]
781                        count_load_constant_build();
782                        value = $self.interned_literal(
783                            $code.symbols.name(*symbol).as_bytes(),
784                        );
785                        debug_assert_ne!(
786                            value,
787                            ObjRef::NIL,
788                            "a constant symbol interned to the handle that means \
789                         'not built yet', so it would be rebuilt on every \
790                         execution and the cache would be dead code"
791                        );
792                        $chunk.remember_symbol(*symbol, value);
793                    }
794                    $registers.set(*dst, value);
795                }
796                // The `>L>` line of one literal. **Its own op**,
797                // because the load emits nothing and `eval.rs`
798                // emits this as a side effect of evaluating --
799                // so a promoted clause with no such op drops the
800                // line while every line after it still matches.
801                Op::TraceLiteral { src } => {
802                    debug_assert!(
803                        $chunk.holds_register(*src),
804                        "op reads register {src} outside the region the chunk \
805                     reserved"
806                    );
807                    // **The gate in front of the register read,
808                    // not only inside `echo_literal`.** That
809                    // function returns immediately under the
810                    // same condition, so this changes no output;
811                    // what it changes is that an untraced run
812                    // does not reach for a value it is not going
813                    // to print, and a register read is a load
814                    // per echo op in a stream that
815                    // carries one behind every literal, read
816                    // and operator. Measured over this arm and
817                    // the other value-echo arms carrying the
818                    // same gate, with the read moved behind it:
819                    // `bench-programs/varlookup.rex` 23.751 to
820                    // 21.737 billion user instructions and the
821                    // pinned `rexxcps` 11.427 to 11.060
822                    // billion.
823                    if !$self.tracing_intermediates() {
824                        continue 'region_loop;
825                    }
826                    let value = $registers.get(*src);
827                    $self.echo_literal(value);
828                }
829                // **The second native expression op**: one bare
830                // symbol's own value, through the same
831                // `Interp::read_symbol` that `eval_node`'s
832                // `Variable`/`Stem`/`Compound` arms enter, with
833                // `eval.rs` itself not entered at all. It emits
834                // nothing -- `Op::TraceRead` below is what
835                // reading a symbol owes.
836                Op::Load {
837                    symbol,
838                    read,
839                    at,
840                    dst,
841                } => {
842                    debug_assert!(
843                        $chunk.holds_register(*dst),
844                        "op writes register {dst} outside the region the chunk \
845                     reserved"
846                    );
847                    let at = at.resolved();
848                    // **The tripwire for a chunk run against a
849                    // plan that is not the one it was compiled
850                    // from.** `at` was read out of that plan's
851                    // `by_symbol`, which is the map `Code::slot_for`
852                    // reads, so a mismatch here is two
853                    // different plans for one body -- and it
854                    // would read somebody else's slot rather
855                    // than fail, which is a wrong value found by
856                    // chasing it.
857                    debug_assert!(
858                        at.is_none()
859                            || matches!(read, SymbolRead::Environment)
860                            || $code.slot_for(*symbol) == at,
861                        "a compiled read names a slot this body's plan does not \
862                     give its symbol"
863                    );
864                    // **The simple read is resolved here rather
865                    // than through `read_symbol`, and that is a
866                    // measurement.** A simple read is a slot
867                    // index; reaching it through the shared
868                    // function costs an out-of-line call, a
869                    // match on `SymbolRead`, a tuple return and
870                    // a `Result` return, per read. Measured with
871                    // the marginal method -- a body run at N and
872                    // 2N iterations, differenced -- `z = a` costs
873                    // 479 user instructions per execution through
874                    // `read_symbol` and 431 resolved here, while
875                    // `z = 1` is unmoved at 367, which is the
876                    // control saying the change reached the read
877                    // and nothing else.
878                    let value = match read {
879                        SymbolRead::Simple => {
880                            let (value, novalue) = match at {
881                                Some(slot) => {
882                                    $self.read_slot($code, *symbol, slot)
883                                }
884                                None => $self.read_at($code, *symbol, None),
885                            };
886                            if let Err(failure) =
887                                $self.novalue_check(novalue, value)
888                            {
889                                break $cold Err(failure);
890                            }
891                            value
892                        }
893                        SymbolRead::Stem
894                        | SymbolRead::Compound
895                        | SymbolRead::CompoundShared
896                        | SymbolRead::Environment => {
897                            match $self.read_symbol($code, Some($chunk), *read, *symbol, at) {
898                                Ok(value) => value,
899                                Err(failure) => break $cold Err(failure),
900                            }
901                        }
902                    };
903                    $registers.set(*dst, value);
904                }
905                // The `>V>` line one read owes, and the `>C>`
906                // line in front of it when the read is a
907                // compound. **Its own op**, because the load
908                // emits nothing and `eval.rs` emits these as a
909                // side effect of evaluating -- so a promoted
910                // clause with no such op drops them while every
911                // line after them still matches.
912                Op::TraceRead { symbol, src, .. } => {
913                    debug_assert!(
914                        $chunk.holds_register(*src),
915                        "op reads register {src} outside the region the chunk \
916                     reserved"
917                    );

919                    // for `Op::TraceLiteral`'s own reason.
920                    if !$self.tracing_intermediates() {
921                        continue 'region_loop;
922                    }
923                    let value = $registers.get(*src);
924                    if $self.echo_symbol_read_op($code, *symbol, value) {
925                        break $cold Err($self.take_held_trace_failure());
926                    }
927                }

929                // operator applied to two registers, through
930                // the same `Interp::arith_small_int` and
931                // `Interp::arith_general` that
932                // `eval_arithmetic` enters, with `eval.rs`
933                // itself not entered at all -- for the operands
934                // either, which is what the ops in front of
935                // this one are. It emits nothing --
936                // `Op::TraceOperator` below is what applying an
937                // operator owes.
938                Op::Arith {
939                    op,
940                    hint,
941                    lhs,
942                    rhs,
943                    dst,
944                } => {
945                    debug_assert!(
946                        $chunk.holds_register(*lhs)
947                            && $chunk.holds_register(*rhs),
948                        "op reads registers {lhs}/{rhs} outside the region the \
949                     chunk reserved"
950                    );
951                    debug_assert!(
952                        $chunk.holds_register(*dst),
953                        "op writes register {dst} outside the region the chunk \
954                     reserved"
955                    );
956                    // **Both read before either is written**,
957                    // which is what makes `lhs == dst` -- the
958                    // shape a chain compiles to -- safe.
959                    let left = $registers.get(*lhs);
960                    let right = $registers.get(*rhs);
961                    // The operands are rooted by the registers
962                    // they came from, which is what
963                    // `arith_general` requires of a caller and
964                    // is why no frame is pushed here where
965                    // `eval_arithmetic` pushes one: it has to
966                    // root values held in Rust locals across
967                    // the evaluation of the operand after them,
968                    // and this op's operands were rooted before
969                    // it ran.
970                    let mut quick = None;
971                    if $chunk.tries_small_int(*hint) {
972                        quick = $self.arith_small_int(*op, left, right);
973                        // The one state change a site makes,
974                        // and it makes it at most once: a site
975                        // that has fallen through skips the
976                        // attempt from here on, so the store
977                        // never repeats and the line the table
978                        // sits on is not dirtied again.
979                        if quick.is_none() {
980                            $chunk.saw_general(*hint);
981                        }
982                    } else {
983                        #[cfg(test)]
984                        count_arith_hint_skip();
985                    }
986                    let value = match quick {
987                        Some(value) => value,
988                        None => match $self.arith_general(*op, left, right) {
989                            Ok(value) => value,
990                            Err(failure) => break $cold Err(failure),
991                        },
992                    };
993                    $registers.set(*dst, value);
994                }

996                // concatenation, comparison or logical
997                // operator applied to two registers, through
998                // the same `Interp::apply_binary` that
999                // `eval_node`'s own binary arm enters, with
1000                // `eval.rs` itself not entered at all -- for
1001                // the operands either, which is what the ops in
1002                // front of this one are. It emits nothing --
1003                // `Op::TraceOperator` below is what applying an
1004                // operator owes.
1005                Op::Binary {
1006                    op,
1007                    lhs,
1008                    rhs,
1009                    dst,
1010                    text,
1011                    onto,

1014                        $chunk.holds_register(*lhs)

1021                        "op writes register {dst} outside the region the chunk \
1022                     reserved"
1023                    );

1028                    let right = $registers.get(*rhs);
1029                    // The operands are rooted by the registers
1030                    // they came from, which is what
1031                    // `apply_binary` requires of a caller and
1032                    // is why no frame is pushed here where
1033                    // `eval_node`'s own arm pushes one: it has
1034                    // to root values held in Rust locals across
1035                    // the evaluation of the operand after them,
1036                    // and this op's operands were rooted before
1037                    // it ran.
1038                    // **A chain appends to the string it is building**
1039                    // where the op before it built one: `onto` says the
1040                    // left operand came from that op, and the comparison
1041                    // says it is still the string that op made rather than

1043                    let applied = if *onto && left == $self.fresh_concat {

1045                    } else {
1046                        $self.apply_binary_op(*op, *text, left, right)
1047                    };

1049                        Ok(value) => value,
1050                        Err(failure) => break $cold Err(failure),
1051                    };
1052                    $registers.set(*dst, value);
1053                }
1054                // The `>O>` line one operator owes. **Its own
1055                // op**, because the operation emits nothing and
1056                // `eval.rs` emits this as a side effect of
1057                // evaluating -- so a promoted clause with no
1058                // such op drops the line while every line after
1059                // it still matches.
1060                Op::TraceOperator { op, src } => {
1061                    debug_assert!(
1062                        $chunk.holds_register(*src),
1063                        "op reads register {src} outside the region the chunk \
1064                     reserved"
1065                    );
1066                    // The gate in front of the register read,
1067                    // for `Op::TraceLiteral`'s own reason.
1068                    if !$self.tracing_intermediates() {
1069                        continue 'region_loop;
1070                    }
1071                    let value = $registers.get(*src);
1072                    $self.echo_operator(*op, value);
1073                }
1074                // **A prefix operator**: `+`, `-` or `\`
1075                // applied to one register, through the same
1076                // `Interp::apply_prefix` that
1077                // `Interp::eval_prefix` enters, with `eval.rs`
1078                // itself not entered at all -- for the operand
1079                // either, which is what the ops in front of
1080                // this one are. It emits nothing --
1081                // `Op::TracePrefix` below is what applying an
1082                // operator owes.
1083                Op::Prefix { op, src, dst } => {
1084                    debug_assert!(
1085                        $chunk.holds_register(*src),
1086                        "op reads register {src} outside the region the chunk \
1087                     reserved"
1088                    );
1089                    debug_assert!(
1090                        $chunk.holds_register(*dst),
1091                        "op writes register {dst} outside the region the chunk \
1092                     reserved"
1093                    );
1094                    // **Read before the destination is
1095                    // written**, which is what makes `src ==
1096                    // dst` -- the shape a prefix compiles to --
1097                    // safe.
1098                    let value = $registers.get(*src);
1099                    // The operand is rooted by the register it
1100                    // came from, which is what `apply_prefix`
1101                    // requires of a caller and is why no frame
1102                    // is pushed here where `eval_prefix` pushes
1103                    // one: it has to root a value held in a
1104                    // Rust local across the operator's own
1105                    // allocation, and this op's operand was
1106                    // rooted before it ran.
1107                    let value = match $self.apply_prefix(*op, value) {
1108                        Ok(value) => value,
1109                        Err(failure) => break $cold Err(failure),
1110                    };

1112                }
1113                // The `>P>` line one prefix operator owes.

1174                            "a compiled write names a slot this body's plan does not give \

1176                        );

1181                            &$clause.kind

1186                        };

1201                // for the same reason `Op::Store` goes through
1202                // `assign_evaluated`.
1203                // **`SIGNAL`, all three forms**, each doing
1204                // what `step`'s own arm does and nothing else:

1206                // `Flow::Signal` and end the region, the trap
1207                // form edits the activation's table and falls

1218                        (interrexx_parse::Signal::Label(name), None) => {

1220                                Ok(flow) => flow,
1221                                Err(failure) => break $cold Err(failure),
1222                            }
1223                        }
1224                        (interrexx_parse::Signal::Value(_), Some(register)) => {

1232                                Ok(flow) => flow,
1233                                Err(failure) => break $cold Err(failure),

1236                        (interrexx_parse::Signal::Trap(trap), None) => {

1255                // a line or run Rexx code and so ends the region as a

1891                }
1892                Op::EnterWhen { .. } => {

1896                    break $cold Err(

1898                    );
1899                }

1902                }

1905                }
1906            }

1918/// The opening half of an [`Op::Clause`] or [`Op::CallingClause`] arm: the
1919/// clause `$op_index` names entered, and the region's state assigned to the
1920/// idents given.
1921macro_rules! open_clause {
1922    (
1923        $self:ident,
1924        $code:ident,
1925        $chunk:ident,
1926        $source:ident,
1927        $pc:ident,
1928        $op_index:ident,
1929        $op_end:ident,
1930        $granting:ident,
1931        $granting_instance:expr,
1932        $grants:expr,
1933        $clause_pc:ident,
1934        $index:ident,
1935        $region_end:ident,
1936        $clause:ident,
1937        $stale:ident,
1938        $debugging:ident,
1939        $entry:ident,
1940        $from:ident,
1941        $registers:ident
1942    ) => {
1943        if $granting_instance && !$granting {
1944            return Ok(Exit::At($pc));
1945        }
1946        // Counted ahead of everything the clause's opening does, so a slice
1947        // that ends here leaves it to be opened afresh from `$pc`.
1948        let counted = match $self.count_clause_against_deadline($grants) {
1949            Ok(counted) => counted,
1950            Err(Failure::Slice) => return Ok(Exit::Slice($pc)),
1951            Err(failure) => return Err(failure),
1952        };
1953        #[cfg(test)]
1954        count_clause_op_entry();
1955        // Where this region starts, which is where `=` at an
1956        // interactive-debug pause sends the counter back to.
1957        $clause_pc = $pc;
1958        $index = usize::cast_from(*$op_index);
1959        $region_end = *$op_end;
1960        // **The instruction this whole region names**, fetched once
1961        // and read by every index-bearing op inside the region
1962        // rather than each resolving its own `index` against the
1963        // body, which is a bounds-checked lookup of the same
1964        // instruction per op that would do it.
1965        // `compile::invariants::assert_region_ops_name_their_clause` is what
1966        // makes the two the same instruction by checking rather
1967        // than by assuming, and [`debug_assert_names_the_clause`]
1968        // is the same check per op in debug.
1969        $clause = match $code.body.instructions.get($index) {
1970            Some(found) => found,
1971            None => return Err(Loud::chunk_map_too_short().into()),
1972        };
1973        if $granting {
1974            $self.grant_procedure_permission($clause);
1975            // **A label leaves the permission alone**, so a label
1976            // reading `false` here is not the permission being
1977            // spent: `sub: procedure expose zg` grants at the
1978            // `PROCEDURE`, one clause after the label.
1979            $granting = $self.activity.procedure_permitted
1980                || matches!($clause.kind, InstructionKind::Label { .. });
1981        }
1982        // **Whether the setting in force is still the one this
1983        // chunk's trace ops were emitted for**, and the whole of
1984        // what makes a compiled-in emission decision safe.
1985        // **One read of the setting answers both questions.**
1986        // Asking `self` again after the clause for the debug flag
1987        // cost `bench-programs/emptyloop.rex` 1.52% and
1988        // `dispatch.rex` 1.14% in `instructions:u`, measured.
1989        let sink = $self.chunk_trace();
1990        // **The analysis, checked at the clause it answered for.**
1991        // A `Known` answer is a claim that this clause always runs
1992        // under exactly that setting, and the claim is what a
1993        // compile-time emission decision rests on; a missing
1994        // control-flow edge shows up here as a mismatch rather
1995        // than as a program that silently stops tracing.
1996        // A silenced activation is excluded because
1997        // [`Interp::traced_mode`] answers `OFF` under it by
1998        // design, which is not the program's own setting. So is a
1999        // run another thread's `SetTrace` has reached
2000        // (`r[native.threads.trace]`), or a `SetThreadTrace`: a frame
2001        // that was running a settled chunk then runs it to its end
2002        // under the setting switched from outside, which the claim
2003        // did not foresee (`r[native.threads.trace.deviation.166]`).
2004        #[cfg(debug_assertions)]
2005        if !$self.silenced()
2006            && !$self.trace_switchable
2007            && let Some(crate::ir::trace_flow::Setting::Known(claimed)) = $chunk.setting_at($index)
2008        {
2009            debug_assert_eq!(
2010                claimed, sink,
2011                "the trace analysis answered {claimed:?} for instruction {}, and \
2012         the setting in force when it ran is not that one",
2013                $index
2014            );
2015        }
2016        $stale = $chunk.trace().clause_echoes() != sink.clause_echoes();
2017        $debugging = sink.debugging();
2018        // **`stale` moves the clause echo from the stream back to
2019        // the run-time gate, in both directions at once.** The
2020        // region's own [`Op::TraceClause`] is skipped and
2021        // the clause opening asks the current setting, so a chunk compiled
2022        // to echo under a setting that no longer does prints nothing,
2023        // and one compiled silent under a setting that now echoes
2024        // prints the line the setting now asks for. The two have
2025        // to be one decision: doing only the first would leave a
2026        // `TRACE R` inside a body invisible to every promoted
2027        // clause after it, and only the second would leave `TRACE
2028        // N` unable to switch one off.
2029        let opening = $chunk.opening_at($index);
2030        // Every register at or above the clause's top is dead as it opens.
2031        $registers.clear_from(opening.register_top);
2032        // **The clause unit, entered by its two halves rather than
2033        // by its closure form**, which is what puts the region's
2034        // ops in this function's own frame instead of a callee's.
2035        // There is one implementation of the clause boundary --
2036        // `Interp::enter_stepped_clause` and
2037        // `Interp::leave_stepped_clause`, which
2038        // `crate::ir::Op::Clause` is itself defined in
2039        // terms of -- so a promoted clause and an unpromoted one
2040        // discharge the same list from the same code.
2041        $entry = $self.enter_stepped_clause(
2042            $code,
2043            $index,
2044            $clause,
2045            $source,
2046            (opening.tabled, opening.position),
2047            counted,
2048        );
2049        $from = if $stale && $self.open_with_runtime_trace($source, $clause) {
2050            // Route the refusal through the region's existing cold exit.
2051            u32::MAX
2052        } else {
2053            $pc + 1
2054        };
2055        // Taken on entry exactly as `step` takes it, because a
2056        // promoted clause is a clause and the permission is spent
2057        // by whichever clause the activation granted it to.
2058        // Outside a body the driver runs nothing has granted it, so
2059        // there is nothing to take.
2060        $self.activity.region_procedure_permitted = if $grants {
2061            std::mem::take(&mut $self.activity.procedure_permitted)
2062        } else {
2063            debug_assert!(
2064                !$self.activity.procedure_permitted,
2065                "a permission granted outside a driven body"
2066            );
2067            false
2068        };
2069    };
2070}

2072/// The rest of an [`Op::Clause`] or [`Op::CallingClause`] arm, from the
2073/// region's state: its ops from `$from`, and the clause boundary. Continues
2074/// `$ops_label` or returns where the arm does, and otherwise answers the
2075/// `Flow` and the op it is settled from.
2076macro_rules! clause_region {
2077    (
2078        $self:ident,
2079        $code:ident,
2080        $chunk:ident,
2081        $registers:ident,
2082        $source:ident,
2083        $pc:ident,
2084        $clause_pc:ident,
2085        $index:ident,
2086        $clause:ident,
2087        $stale:ident,
2088        $debugging:ident,
2089        $entry:ident,
2090        $end:ident,
2091        $from:ident,
2092        $header:expr,
2093        $top:expr,
2094        $ops_label:lifetime
2095    ) => {
2096        // The ops of this clause region, `[from, end)`, and where
2097        // they leave the counter.
2098        'arm: {
2099            let park: Park = 'park: {
2100                let ran: Result<RegionEnd, Failure> = 'cold: {
2101                    // **The region answers an op index and nothing else.**
2102                    // Where a clause leaves the counter is the whole of what
2103                    // an ordinary one has to say, and carrying that in a
2104                    // `Result<RegionEnd, Failure>` costs a 24-byte value
2105                    // built and moved per clause. The two answers that do
2106                    // need one -- a `Flow` the enclosing range settles, and a
2107                    // failure -- leave through `'cold` instead, so the
2108                    // discriminant rides the program counter on the path
2109                    // every clause takes and the value exists only on the
2110                    // paths that have something to put in it.
2111                    let next: u32 = 'region: {
2112                        // A `DO`/`LOOP` header's values, accumulated across this
2113                        // region's own ops because they are not `ObjRef`s and so
2114                        // have no register to live in: a bound is a `Number` and
2115                        // a budget is a count.
2116                        let mut header: Option<LoopHeaderValues> = $header;
2117                        let Some(ops) = $chunk.ops_in($from, $end) else {
2118                            if $from == u32::MAX
2119                                && let Some(failure) = $self.take_trace_failure()
2120                            {
2121                                break 'cold Err(failure);
2122                            }
2123                            break 'cold Err(Loud::chunk_map_too_short().into());
2124                        };
2125                        region_ops!(
2126                            $self, $code, $chunk, $registers, $source, $clause, $index, $stale,
2127                            header, $end, ops, $from, $top, 'cold, 'region, 'park
2128                        );
2129                        $end
2130                    };
2131                    // **The hot exit, and the whole point of the split.**
2132                    // `leave_clause`'s own fast path is this same question,
2133                    // so asking it here reaches the same answer without
2134                    // building the value that answer would travel in.
2135                    // `finish_plain_clause` discharges what is left of the
2136                    // boundary.
2137                    // **Interactive debug takes the cold exit**, whose pause
2138                    // comes before the boundary that delivers what the clause
2139                    // queued. Asked of the setting and not of `stale`:
2140                    // staleness heals -- the chunk is recompiled under the
2141                    // setting now in force -- so a skip count set at a pause
2142                    // would stop running down after the first clause.
2143                    // Measured: `trace -2` then suppressed every later clause
2144                    // instead of two.
2145                    if $self.activity.pending_traps.is_empty()
2146                        && !$debugging
2147                        && !$self.clause_exits
2148                    {
2149                        $self.finish_plain_clause($entry);
2150                        $pc = next;
2151                        continue $ops_label;
2152                    }
2153                    Ok(RegionEnd::At(next))
2154                };
2155                let ran = if $debugging || $self.activity.debug_stop_owed.is_some() {
2156                    $self.debug_pause_in_region(ran, $code, $index, $clause, $source, $clause_pc)
2157                } else {
2158                    ran
2159                };
2160                if ran.is_err() {
2161                    $self.blame_failed_test($code, $chunk, $clause_pc, $source);
2162                }
2163                break 'arm match $self
2164                    .leave_stepped_clause($entry, $code, $index, $clause, $source, ran)?
2165                {
2166                    ClauseOutcome::Ran(region) => match region? {
2167                        // A promoted clause produces no `Flow` of its own:
2168                        // where it leaves the counter *is* its answer, which
2169                        // is what a jump op is for. Only its boundary can
2170                        // end the activation, and that is the `Ended` below.
2171                        RegionEnd::At(next) => {
2172                            $pc = next;
2173                            continue $ops_label;
2174                        }
2175                        // Settled against this range from the op past the
2176                        // region, which is where an absorbed `Flow::Next`
2177                        // continues -- the same position `pc + 1` is for an
2178                        // op that runs one clause and no more.
2179                        RegionEnd::Flowed(flow) => (flow, $end),
2180                    },
2181                    ClauseOutcome::Ended(exit) => (Flow::Exit(exit.value()), $clause_pc),
2182                };
2183            };
2184            $self.park_call(park, $entry, $clause_pc, $end, $index, $stale, $debugging);
2185            return Ok(Exit::Parked);
2186        }
2187    };
2188}

2560    /// [`Interp::run_call_named`] up to the point its callee's body would
2561    /// run, and the indent its `RESULT` is settled at.

2564        &mut self,
2565        code: &Code<'_>,
2566        clause: &Instruction,
2567        chunk: &Chunk,
2568        site: u16,
2569        argc: u16,
2570    ) -> Result<(Started<Ended>, usize), Failure> {
2571        let InstructionKind::Call(call) = &clause.kind else {
2572            return Err(Loud::call_op_off_its_node().into());
2573        };
2574        let Some(mark) = self
2575            .activity
2576            .value_buffer
2577            .len()
2578            .checked_sub(usize::from(argc))
2579        else {
2580            return Err(Loud::call_op_off_its_node().into());
2581        };
2582        let (name, literal) = match &**call {
2583            Call::Named { name, literal, .. } => (name, literal),
2584            Call::Qualified {
2585                namespace, name, ..
2586            } => {
2587                let namespace = code.symbols.name(*namespace).as_bytes();
2588                let name = code.symbols.name(*name).as_bytes();

2590                let base_indent = self.activity.clause_state.current_value_indent;
2591                let started = self.begin_subroutine_over_pushed_args(resolved, name, mark)?;
2592                return Ok((started, base_indent));
2593            }
2594            _ => return Err(Loud::call_op_off_its_node().into()),

2596        // The site's own kept answer and the resolution when it has none --
2597        // `Op::Call`'s arm has the whole argument, and this is the same site
2598        // table read the same way. The name comes off the instruction here
2599        // rather than off a node, which is why this op needs no address where
2600        // `Op::CallArgs` does.
2601        let generation = self.routine_generation;
2602        let (resolved, keep) = match chunk.resolved_call(site, generation) {
2603            Some(resolved) => {
2604                #[cfg(test)]
2605                count_call_site_hit();
2606                (resolved, None)
2607            }
2608            None => {
2609                let resolved = self.resolve_call(name, !*literal)?;
2610                (
2611                    resolved,
2612                    self.missed_site(chunk, site, resolved, generation),
2613                )
2614            }
2615        };
2616        // Captured before the callee runs, which overwrites
2617        // `current_value_indent` with its own clauses'.
2618        let base_indent = self.activity.clause_state.current_value_indent;
2619        if let Some(keep) = keep {
2620            let started = self.begin_kept_subroutine(chunk, site, resolved, name, mark, keep)?;
2621            return Ok((started, base_indent));
2622        }
2623        let started = self.begin_subroutine_over_pushed_args(resolved, name, mark)?;
2624        Ok((started, base_indent))
2625    }

2867    /// answered `sent` owes, delivered as `form` says.
2868    #[inline(always)]
2869    fn deliver_answer(
2870        &mut self,
2871        chunk: &Chunk,
2872        site: u32,
2873        form: SendForm,
2874        sent: Option<ObjRef>,
2875    ) -> Result<ObjRef, Failure> {

4582    /// Runs `chunk`'s ops from op `at` until one of them produces a `Flow`
4583    /// that `[start, end]` does not absorb. Under `TOP` the range is a body
4584    /// [`Interp::drive`] runs, and a call op parks its clause region for its
4585    /// callee; `grant` is the op of the clause the first-instruction
4586    /// permission goes to, and `resume` a parked region whose callee has
4587    /// ended and whose answer is delivered, run on first.
4588    #[expect(
4589        clippy::too_many_arguments,
4590        reason = "every argument is a value each caller already holds"
4591    )]
4592    #[inline(always)]
4593    fn ops_loop<const GRANTING: bool, const TOP: bool>(
4594        &mut self,
4595        code: &Code<'_>,
4596        chunk: &Chunk,
4597        registers: RegFrame<'_>,
4598        at: u32,
4599        start: usize,
4600        end: usize,
4601        source: Option<&ProgramSource>,
4602        base: usize,
4603        grant: u32,
4604        mut resume: Option<ParkedCall>,
4605    ) -> Result<Exit, Failure> {
4606        let Some(stop) = chunk.op_at(end) else {
4607            return Err(Loud::chunk_map_too_short().into());
4608        };
4609        // **The stream cut to this range's own bound, taken once.** The loop
4610        // below continues only while `pc < stop`, and this slice is `stop`
4611        // long, so every read of it under that guard is in range already and
4612        // the per-op check goes. A stream shorter than `stop` is the same
4613        // fault the per-op read used to report, found here instead.
4614        let Some(full) = chunk.ops_upto(stop) else {
4615            return Err(Loud::chunk_map_too_short().into());
4616        };
4617        // Cut short at the clause the permission goes to, so the bound check
4618        // every op already makes is what finds that clause too.
4619        let mut stream = full.get(..usize::cast_from(grant)).unwrap_or(full);
4620        let depth = self.activation_depth();
4621        // Whether the permission is still worth asking about. It is granted to
4622        // this activation's first instruction and cleared by the one after it;
4623        // the first clause that finds it spent hands the rest of the range to
4624        // the instance that never asks.
4625        let mut granting = GRANTING;
4626        // A parked region to resume starts the counter past `stop`, so the
4627        // bound check every op already makes is what finds it.
4628        let mut pc = if resume.is_some() { u32::MAX } else { at };
4629        'ops: loop {
4630            let (flow, next) = 'step: {
4631                // An `Op::CallingClause` region's state, opened here or resumed:
4632                // assigned in place on each path rather than built as one
4633                // value, which the region's hot path would then copy.
4634                let clause_pc;
4635                let index;
4636                let region_end;
4637                let clause;
4638                let stale;
4639                let debugging;
4640                let entry;
4641                let from;
4642                // A resumed region's `DO`/`LOOP` header values so far.
4643                let header: Option<Box<LoopHeaderValues>>;
4644                'calling: {
4645                    // **A branch that runs out exactly at this range's own end.**
4646                    // `stop` is `Chunk::op_of`'s entry for `end`, which is where the
4647                    // branch's own `Op::EndWhen` sits -- so the op is one past what
4648                    // this range drives and the boundary owes the close itself.
4649                    if usize::cast_from(pc) >= stream.len() {
4650                        if let Some(parked) = resume.take() {
4651                            clause = match code.body.instructions.get(parked.index) {
4652                                Some(clause) => clause,
4653                                None => return Err(Loud::chunk_map_too_short().into()),
4654                            };
4655                            clause_pc = parked.clause_pc;
4656                            index = parked.index;
4657                            region_end = parked.end;
4658                            stale = parked.stale;
4659                            debugging = parked.debugging;
4660                            entry = parked.entry;
4661                            header = parked.header;
4662                            from = parked.at;
4663                            break 'calling;
4664                        }
4665                        if stream.len() < full.len() {
4666                            stream = full;
4667                            if pc == grant {
4668                                self.activity.procedure_permitted = std::mem::take(
4669                                    &mut self.activation_mut().first_instruction_pending,
4670                                );
4671                            }
4672                            continue 'ops;
4673                        }
4674                        match self.leave_ended_select_branch(
4675                            code, chunk, registers, base, pc, start, end, source,
4676                        )? {
4677                            BranchEnd::At(target) => {
4678                                pc = target;
4679                                continue 'ops;
4680                            }
4681                            BranchEnd::Escaped(other) => return Ok(Exit::Flow(other)),
4682                            BranchEnd::None => return Ok(Exit::Flow(Flow::Next)),
4683                        }
4684                    }
4685                    // **The tripwire for the op that replaced the check that used to
4686                    // stand here.** A frame is overdue only at the position its own
4687                    // `Op::EndWhen` occupies; anywhere else means a branch outlived
4688                    // the op that closes it, which is a `SELECT` running on into what
4689                    // follows it and is silent in any program whose branches happen to
4690                    // agree.
4691                    //
4692                    // **A labelled block's frame is overdue from the boundaries of the
4693                    // `IF`s that end in front of its `END`** too: they run inside the
4694                    // block, ahead of the `Op::EndWhen` that closes it.
4695                    #[cfg(debug_assertions)]
4696                    if !matches!(stream.get(usize::cast_from(pc)), Some(Op::EndWhen)) {
4697                        debug_assert!(
4698                            !(self.activity.frames.len() > base
4699                                && self.activity.frames.last().is_some_and(|frame| {
4700                                    pc >= frame.op_end
4701                                        && !(matches!(frame.kind, FrameKind::Block(_))
4702                                            && matches!(
4703                                                stream.get(usize::cast_from(pc)),
4704                                                Some(Op::EndBranch)
4705                                            ))
4706                                })),
4707                            "the innermost SELECT frame ended before op {pc}, which is not its own EndWhen"
4708                        );
4709                    }
4710                    let op = &stream[usize::cast_from(pc)];
4711                    match op {
4712                        Op::Clause {
4713                            index: op_index,
4714                            end: op_end,
4715                        } => {
4716                            let clause_pc;
4717                            let index;
4718                            let region_end;
4719                            let clause;
4720                            let stale;
4721                            let debugging;
4722                            let entry;
4723                            let from;
4724                            open_clause!(
4725                                self, code, chunk, source, pc, op_index, op_end, granting,
4726                                GRANTING, TOP, clause_pc, index, region_end, clause, stale,
4727                                debugging, entry, from, registers
4728                            );
4729                            let end = region_end;
4730                            break 'step clause_region!(
4731                                self, code, chunk, registers, source, pc, clause_pc, index, clause,
4732                                stale, debugging, entry, end, from, None, TOP, 'ops
4733                            );
4734                        }
4735                        Op::CallingClause {
4736                            index: op_index,
4737                            end: op_end,
4738                        } => {
4739                            open_clause!(
4740                                self, code, chunk, source, pc, op_index, op_end, granting,
4741                                GRANTING, TOP, clause_pc, index, region_end, clause, stale,
4742                                debugging, entry, from, registers
4743                            );
4744                            header = None;
4745                        }
4746                        // **A jump past this range's end is loud rather than a
4747                        // stop.** `absorb` cannot check it -- a jump target is an op
4748                        // index and absorption is decided in instruction space -- so
4749                        // without this the loop's own `pc < stop` reads an escaping
4750                        // jump as "the range completed" and answers `Flow::Next`,
4751                        // which is a construct silently finishing where it should have
4752                        // propagated. Landing exactly on `stop` *is* completion, which
4753                        // is what a branch-end jump at a range boundary does, so the
4754                        // comparison is strict. A backward jump out of the range is
4755                        // not checked and is not emitted: it would re-run ops inside
4756                        // the range, which is a wrong answer rather than a silent one,
4757                        // and checking it costs a second `op_at` on the hot path.
4758                        Op::Jump { target } => {
4759                            if *target > stop {
4760                                return Err(Loud::jump_out_of_range().into());
4761                            }
4762                            pc = *target;
4763                            continue 'ops;
4764                        }
4765                        // The boundary wrapper around an
4766                        // `IF`'s whole arm runs, which a flattened construct has no
4767                        // wrapper to run. `Interp::end_promoted_branch`'s doc comment
4768                        // has the program that says it is not a spare one.
4769                        Op::EndBranch => {
4770                            break 'step (self.end_promoted_branch(Flow::Next)?, pc + 1);
4771                        }
4772                        Op::EndWhen => {
4773                            match self.leave_ended_select_branch(
4774                                code, chunk, registers, base, pc, start, end, source,
4775                            )? {
4776                                BranchEnd::At(target) => {
4777                                    pc = target;
4778                                    continue 'ops;
4779                                }
4780                                BranchEnd::Escaped(other) => return Ok(Exit::Flow(other)),
4781                                // The scan's own landing place when the branch this
4782                                // ends was never entered.
4783                                BranchEnd::None => break 'step (Flow::Next, pc + 1),
4784                            }
4785                        }
4786                        Op::EnterWhen { select, when } => {
4787                            let frame = self.when_frame(
4788                                code,
4789                                chunk,
4790                                usize::cast_from(*select),
4791                                usize::cast_from(*when),
4792                            )?;
4793                            self.activity.frames.push(Frame::select(frame));
4794                            pc += 1;
4795                            continue 'ops;
4796                        }
4797                        Op::EnterOtherwise { select } => {
4798                            let frame =
4799                                self.otherwise_frame(code, chunk, usize::cast_from(*select))?;
4800                            self.activity.frames.push(Frame::select(frame));
4801                            pc += 1;
4802                            continue 'ops;
4803                        }
4804                        // A loop's header test region, which a pass
4805                        // boundary jumped to: counted and opened here, like
4806                        // a clause, though its echo, line and indent are the
4807                        // boundary's. The condition's ops and `LoopTested`
4808                        // run as the region's ops.
4809                        Op::LoopTest { .. } => {
4810                            match self.loop_test_entry(code, chunk, source, pc, TOP)? {
4811                                TestEntry::Region(parked) => {
4812                                    resume = Some(parked);
4813                                    pc = u32::MAX;
4814                                    continue 'ops;
4815                                }
4816                                TestEntry::Done(flow) => {
4817                                    self.activity.frames.pop();
4818                                    break 'step (flow, pc);
4819                                }
4820                                TestEntry::Slice => return Ok(Exit::Slice(pc)),
4821                            }
4822                        }
4823                        // The bottom of a flattened pass, reached by the
4824                        // body falling out of its last clause into the `END`'s own op.
4825                        Op::LoopNext { index } => {
4826                            if self.activity.frames.len() <= base
4827                                || !matches!(
4828                                    self.activity.frames.last().map(|f| &f.kind),
4829                                    Some(FrameKind::Loop)
4830                                )
4831                            {
4832                                return Err(Loud::op_not_driven("LoopNext").into());
4833                            }
4834                            debug_assert_eq!(
4835                                self.activity.flat_top.as_ref().map(|flat| flat.do_index),
4836                                Some(usize::cast_from(*index)),
4837                                "a LoopNext op ended a pass of a loop other than the one it names"
4838                            );
4839                            // Compiled passes, once the loop is hot: the
4840                            // same boundary and clauses, run until one of
4841                            // them needs the driver.
4842                            #[cfg(feature = "jit")]
4843                            if !GRANTING
4844                                && chunk.jit.hot(*index)
4845                                && stream.len() == full.len()
4846                                && let Some(action) = self.jit_passes(
4847                                    code,
4848                                    chunk,
4849                                    registers,
4850                                    source,
4851                                    pc,
4852                                    *index,
4853                                    TOP,
4854                                    (base, start, end),
4855                                )
4856                            {
4857                                match action {
4858                                    jit_host::JitAction::At(target) => {
4859                                        pc = target;
4860                                        continue 'ops;
4861                                    }
4862                                    jit_host::JitAction::Return(exit) => return exit,
4863                                }
4864                            }
4865                            // The pass's header clause, counted here so a
4866                            // slice that ends at it leaves this op to run
4867                            // afresh.
4868                            let counted = match self.count_clause_against_deadline(TOP) {
4869                                Ok(counted) => counted,
4870                                Err(Failure::Slice) => return Ok(Exit::Slice(pc)),
4871                                Err(failure) => return Err(failure),
4872                            };
4873                            match self.flat_loop_step_top(
4874                                code,
4875                                source,
4876                                chunk,
4877                                registers,
4878                                Flow::Next,
4879                                |_| Ok(counted),
4880                            )? {
4881                                crate::run::FlatStep::Body(op_body) => {
4882                                    pc = op_body;
4883                                    continue 'ops;
4884                                }
4885                                crate::run::FlatStep::Done(flow) => {
4886                                    self.activity.frames.pop();
4887                                    break 'step (flow, pc);
4888                                }
4889                            }
4890                        }
4891                        // The ops below are only meaningful inside a `Clause` region,
4892                        // which the `Op::Clause` arm above walks: reaching one here
4893                        // means a jump landed in the middle of a region rather than on
4894                        // its `Clause`. Loud rather than a panic, which is this crate's
4895                        // standing rule for a state the type system admits and the
4896                        // compiler does not produce.
4897                        // Every op that belongs inside a clause region, which this
4898                        // loop does not drive -- `undriven_op_name` has why they are
4899                        // one arm rather than one each.
4900                        _ => return Err(Loud::op_not_driven(undriven_op_name(op)).into()),
4901                    }
4902                };
4903                let end = region_end;
4904                clause_region!(
4905                    self, code, chunk, registers, source, pc, clause_pc, index, clause, stale,
4906                    debugging, entry, end, from, header.map(|header| *header), TOP, 'ops
4907                )
4908            };
4909            // **Per clause, not per escaping flow.** A clause that left the
4910            // activation stack changed makes this loop's `code` describe a
4911            // frame it is no longer running, and every op after it is stepped
4912            // against the wrong body -- so the check has to sit where a clause
4913            // finishes rather than where control leaves this function.
4914            debug_assert_eq!(
4915                self.activation_depth(),
4916                depth,
4917                "a clause left the activation stack changed, so this loop's `code` and its `pc` \
4918                 no longer describe the same frame"
4919            );
4920            match self.settle(code, chunk, registers, base, flow, next, start, end, source)? {
4921                Settled::At(target) => pc = target,
4922                Settled::Escaped(other) => return Ok(Exit::Flow(other)),
4923            }
4924        }
4925    }

4927    /// Drops every frame this level opened, and the loop state each open loop
4928    /// frame stands for.

4933            let Some(frame) = self.activity.frames.pop() else {
4934                return;

4939                // raised, which loses one box to the allocator and no state.
4940                if let Some(flat) = self.activity.flat_top.take() {

4954    /// Where `flow` leaves the counter, once every open frame and then the
4955    /// range itself have had their say.
4956    #[expect(
4957        clippy::too_many_arguments,
4958        reason = "two callers inside one loop, and every argument is a value that loop holds"
4959    )]
4960    #[inline(always)]
4961    fn settle(
4962        &mut self,
4963        code: &Code<'_>,
4964        chunk: &Chunk,
4965        registers: RegFrame<'_>,
4966        base: usize,
4967        mut flow: Flow,
4968        next: u32,
4969        start: usize,
4970        end: usize,
4971        source: Option<&ProgramSource>,
4972    ) -> Result<Settled, Failure> {
4973        // `Flow::Next` settles at `next` whatever is open: `absorb`
4974        // answers `Advance` for it against every range, and every arm below
4975        // turns `Advance` into `At(next)`. So the walk over the open frames is
4976        // skipped for the one flow that every clause of every body produces,
4977        // which is what makes a frame affordable to hold open across a loop's
4978        // whole body rather than only across a `WHEN`'s branch.
4979        if matches!(flow, Flow::Next) {
4980            return Ok(Settled::At(next));
4981        }
4982        loop {
4983            // Copied out rather than held, because the `Escaped` arm pops the
4984            // frame this came from. **`base` and not emptiness** is what ends
4985            // the walk: the frames below it are another level's, and this one's
4986            // own range has the next say.
4987            let Some((frame_start, frame_end)) = (self.activity.frames.len() > base)
4988                .then(|| self.activity.frames.last().map(|f| (f.start, f.end)))
4989                .flatten()
4990            else {
4991                return Ok(match absorb(flow, start, end) {
4992                    Absorbed::Advance => Settled::At(next),
4993                    Absorbed::Resume(target) => Settled::At(op_at(chunk, target)?),
4994                    Absorbed::Escaped(other) => Settled::Escaped(other),
4995                });
4996            };
4997            match absorb(flow, frame_start, frame_end) {
4998                Absorbed::Advance => return Ok(Settled::At(next)),
4999                // `target` can be the branch's own `end`, which is one past its
5000                // last instruction: `op_at` answers this frame's `op_end`
5001                // there, and the loop's own check closes the frame on the next
5002                // pass rather than a second rule doing it here.
5003                Absorbed::Resume(target) => return Ok(Settled::At(op_at(chunk, target)?)),
5004                Absorbed::Escaped(other) => {
5005                    match self
5006                        .activity
5007                        .frames
5008                        .pop()
5009                        .expect("the check above just observed one")
5010                        .kind
5011                    {
5012                        FrameKind::Select(frame) => {
5013                            flow = self.leave_branch(code, &frame, other)?;
5014                        }
5015                        FrameKind::Block(frame) => {
5016                            let resume = SelectResume {
5017                                done: frame.resume,
5018                                left: frame.resume,
5019                            };
5020                            flow =
5021                                self.leave_select(code, frame.index, frame.label, resume, other)?;
5022                        }
5023                        // A `LEAVE`/`ITERATE` that reached this
5024                        // loop, decided by the same `do_body_outcome` the
5025                        // nested form hands the answering `Flow` to. An
5026                        // `ITERATE` this loop consumes puts the frame back and
5027                        // resumes at the body's first op -- the same place
5028                        // `Op::LoopNext` resumes a pass that fell through.
5029                        FrameKind::Loop => {
5030                            match self
5031                                .flat_loop_step_escaped(code, chunk, registers, source, other)?
5032                            {
5033                                crate::run::FlatStep::Body(op_body) => {
5034                                    self.activity.frames.push(Frame {
5035                                        op_end: u32::MAX,
5036                                        start: frame_start,
5037                                        end: frame_end,
5038                                        kind: FrameKind::Loop,
5039                                    });
5040                                    return Ok(Settled::At(op_body));
5041                                }
5042                                crate::run::FlatStep::Done(escape) => flow = escape,
5043                            }
5044                        }
5045                    }
5046                }
5047            }
5048        }
5049    }

5051    /// What a `SELECT` does with a `Flow` that left one of its branches:
5052    /// `select_escape` and `leave_select`, exactly what `step`'s own `Select`
5053    /// arm and `Op::EnterOtherwise` do with the same `Flow`, and then the boundary
5054    /// wrapper around the whole arm runs.
5055    fn leave_branch(
5056        &mut self,
5057        code: &Code<'_>,
5058        frame: &SelectFrame,
5059        flow: Flow,
5060    ) -> Result<Flow, Failure> {
5061        let flow = match frame.branch {
5062            Branch::When => match select_escape(frame.otherwise, flow) {
5063                // Still inside this `SELECT`: `EnterOtherwise` at the target's
5064                // own entry opens the next branch's frame, and this `WHEN`'s
5065                // branch did not finish -- control was redirected out of it,
5066                // so the end-of-branch boundary below is not owed.
5067                SelectEscape::Otherwise(target) => return Ok(Flow::Goto(target)),
5068                SelectEscape::Forward(flow) => flow,
5069            },
5070            // `Interp::leave_otherwise` is the whole of leaving this branch,
5071            // shared with `Op::EnterOtherwise`: the escape elevation is restored
5072            // and `leave_select` decides where control goes. **And no
5073            // end-of-branch boundary follows it**, because `OTHERWISE`'s
5074            // branch ends at the `END`, a real instruction with a boundary of
5075            // its own -- measured, a handler queued by the last clause of an
5076            // `OTHERWISE` and re-queued by its own handler is delivered at the
5077            // `END`'s line, not at the branch's.
5078            Branch::Otherwise => {
5079                return self.leave_otherwise(
5080                    code,
5081                    frame.select,
5082                    frame.label,
5083                    frame.end,
5084                    frame.select_end,
5085                    flow,
5086                );
5087            }
5088        };
5089        let flow = self.leave_select(code, frame.select, frame.label, frame.resume, flow)?;
5090        // A matched `WHEN`'s branch is one closed with a synthetic instruction,
5091        // so it owes that instruction's boundary.
5092        self.end_promoted_branch(flow)
5093    }
