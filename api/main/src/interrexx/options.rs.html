13/// The six conditions `::OPTIONS` can turn from a condition into a SYNTAX
14/// error, in `subDirectives[]` order.
15const ESCALATABLE: [&[u8]; 6] = [
16    b"ERROR",
17    b"FAILURE",
18    b"LOSTDIGITS",
19    b"NOSTRING",
20    b"NOTREADY",
21    b"NOVALUE",
22];
23
24/// The settings one file's `::OPTIONS` directives accumulate.
25#[derive(Clone, Debug, Default)]
26pub(crate) struct PackageOptions {
27    /// `DIGITS`, `FUZZ` and `FORM`: the numeric settings every activation of
28    /// this package's code starts from, and the defaults a bare `NUMERIC
29    /// DIGITS`/`FUZZ`/`FORM` resets to.
30    pub(crate) numeric: Settings,
31    /// `TRACE`, or `None` for a package that named none. `None` is not
32    /// `TraceMode::NORMAL`: a package that says nothing leaves the
33    /// activation's own starting mode alone.
34    pub(crate) trace: Option<TraceMode>,
35    /// `NUMERIC INHERIT`, which makes a `::ROUTINE` or `::METHOD` start from
36    /// the numeric settings in force at the call site instead of from
37    /// [`PackageOptions::numeric`].
38    pub(crate) numeric_inherit: bool,
39    /// Which of [`ESCALATABLE`] this package raises as a SYNTAX error, which
40    /// every activation of its code starts from and can then turn off.
41    pub(crate) syntax: ConditionSyntax,
42    /// `NOPROLOG`, stored negated so that a package naming no `::OPTIONS` has
43    /// its prologue run, which the prologue run reads.
44    pub(crate) suppress_prolog: bool,
45    /// Which options a `::OPTIONS` directive named, as opposed to which are
46    /// in force, which `Package~options('X')` renders and which the
47    /// class-level override mechanism consults before it writes.
48    explicit: Explicit,
49}

51/// The `::OPTIONS` subkeywords a package's directives actually named, in the
52/// order `Package~options` writes them.
53#[derive(Copy, Clone, PartialEq, Eq, Debug, Default)]
54struct Explicit {
55    named: [bool; EXPLICIT_WORDS.len()],
56}
57
58/// The subkeyword each [`Explicit`] flag stands for, in rendering order.
59const EXPLICIT_WORDS: [&str; 13] = [
60    "DIGITS",
61    "FORM",
62    "FUZZ",
63    "NUMERIC",
64    "ERROR",
65    "FAILURE",
66    "LOSTDIGITS",
67    "NOSTRING",
68    "NOTREADY",
69    "NOVALUE",
70    "NOPROLOG",
71    "PROLOG",
72    "TRACE",
73];
74
75impl Explicit {
76    /// Whether the directives named `word`, an [`EXPLICIT_WORDS`] entry.
77    fn named(self, word: &str) -> bool {
78        EXPLICIT_WORDS
79            .iter()
80            .position(|candidate| *candidate == word)
81            .is_some_and(|which| self.named[which])
82    }
83
84    /// The blank-delimited subkeyword list, empty for a package whose
85    /// directives named none: a file with no `::OPTIONS` renders the null
86    /// string.
87    fn to_text(self) -> Vec<u8> {
88        let named: Vec<&str> = EXPLICIT_WORDS
89            .iter()
90            .zip(self.named)
91            .filter(|(_, named)| *named)
92            .map(|(word, _)| *word)
93            .collect();
94        named.join(" ").into_bytes()
95    }
96}

98/// Which conditions one activation raises as a SYNTAX error rather than as a
99/// condition.
100#[derive(Copy, Clone, PartialEq, Eq, Debug, Default)]
101pub(crate) struct ConditionSyntax {
102    escalated: [bool; 6],
103}
104
105impl ConditionSyntax {
106    /// `::OPTIONS <which> SYNTAX`/`CONDITION`, with `ALL` writing all six.
107    fn set(&mut self, which: ConditionOption, syntax: bool) {
108        match which {
109            ConditionOption::All => self.escalated = [syntax; 6],
110            ConditionOption::Error => self.escalated[0] = syntax,
111            ConditionOption::Failure => self.escalated[1] = syntax,
112            ConditionOption::LostDigits => self.escalated[2] = syntax,
113            ConditionOption::NoString => self.escalated[3] = syntax,
114            ConditionOption::NotReady => self.escalated[4] = syntax,
115            ConditionOption::NoValue => self.escalated[5] = syntax,
116        }
117    }
118
119    /// Whether an untrapped `condition` is a SYNTAX error here.
120    pub(crate) fn raises(self, condition: &[u8]) -> bool {
121        ESCALATABLE
122            .iter()
123            .position(|name| *name == condition)
124            .is_some_and(|which| self.escalated[which])
125    }
126
127    /// `SIGNAL ON`/`OFF` and `CALL ON`/`OFF` of `condition`, each of which
128    /// turns off the escalation the package asked for -- for the named
129    /// condition, or for all six when the name is `ANY`.
130    pub(crate) fn disable_for(&mut self, condition: &[u8], signal: bool, novalue_trapped: bool) {
131        let any = condition == b"ANY";
132        let names = |name: &[u8]| any || condition == name;
133        if signal && names(b"NOVALUE") && !novalue_trapped {
134            self.escalated[5] = false;
135        }
136        if names(b"ERROR") {
137            self.escalated[0] = false;
138        }
139        if names(b"FAILURE") {
140            self.escalated[1] = false;
141        }
142        if signal && names(b"LOSTDIGITS") {
143            self.escalated[2] = false;
144        }
145        if signal && names(b"NOSTRING") {
146            self.escalated[3] = false;
147        }
148        if names(b"NOTREADY") {
149            self.escalated[4] = false;
150        }
151    }
152}

154impl PackageOptions {
155    /// Folds one `::OPTIONS` option into these settings.
156    pub(crate) fn apply(&mut self, option: &PackageOption) -> Result<(), SettingsError> {
157        let mut names = |word: &str| {
158            let which = EXPLICIT_WORDS
159                .iter()
160                .position(|candidate| *candidate == word)
161                .expect("every name passed here is an EXPLICIT_WORDS entry");
162            self.explicit.named[which] = true;
163            // `PROLOG` and `NOPROLOG` clear each other: the directive parser
164            // keeps one of the two flags.
165            let other = match word {
166                "PROLOG" => Some("NOPROLOG"),
167                "NOPROLOG" => Some("PROLOG"),
168                _ => None,
169            };
170            if let Some(other) = other
171                && let Some(other) = EXPLICIT_WORDS
172                    .iter()
173                    .position(|candidate| *candidate == other)
174            {
175                self.explicit.named[other] = false;
176            }
177        };
178        match option {
179            PackageOption::Digits(_) => names("DIGITS"),
180            PackageOption::Fuzz(_) => names("FUZZ"),
181            PackageOption::Form(_) => names("FORM"),
182            PackageOption::Trace(_) => names("TRACE"),
183            PackageOption::Condition { which, .. } => match which {
184                ConditionOption::All => {
185                    for name in ESCALATABLE {
186                        names(&String::from_utf8_lossy(name));
187                    }
188                }
189                ConditionOption::Error => names("ERROR"),
190                ConditionOption::Failure => names("FAILURE"),
191                ConditionOption::LostDigits => names("LOSTDIGITS"),
192                ConditionOption::NoString => names("NOSTRING"),
193                ConditionOption::NotReady => names("NOTREADY"),
194                ConditionOption::NoValue => names("NOVALUE"),
195            },
196            PackageOption::Prolog(enabled) => names(if *enabled { "PROLOG" } else { "NOPROLOG" }),
197            PackageOption::NumericInherit(_) => names("NUMERIC"),
198        }
199        match option {
200            PackageOption::Digits(digits) => self.numeric.set_digits(u64::cast_from(*digits))?,
201            PackageOption::Fuzz(fuzz) => self.numeric.set_fuzz(u64::cast_from(*fuzz))?,
202            PackageOption::Form(form) => self.numeric.set_form(match form {
203                OptionsForm::Scientific => Form::Scientific,
204                OptionsForm::Engineering => Form::Engineering,
205            }),
206            PackageOption::Trace(setting) => {
207                self.trace =
208                    Some(mode_from_setting(setting).expect(
209                        "interrexx-parse's check_trace_setting already validated this byte",
210                    ));
211            }
212            PackageOption::Condition { which, syntax } => self.syntax.set(*which, *syntax),
213            PackageOption::Prolog(enabled) => self.suppress_prolog = !*enabled,
214            PackageOption::NumericInherit(inherit) => self.numeric_inherit = *inherit,
215        }
216        Ok(())
217    }

219    /// Whether `NOSTRING` is one of the escalated ones, which
220    /// `Interp::install_directives` asks in order to arm the required-string
221    /// protocol.
222    pub(crate) fn escalates_nostring(&self) -> bool {
223        self.syntax.raises(b"NOSTRING")
224    }

227    /// asks in order to arm the arithmetic path's own gate.
228    pub(crate) fn escalates_lostdigits(&self) -> bool {
229        self.syntax.raises(b"LOSTDIGITS")

232    /// These settings written back as the `::OPTIONS` directive that would
233    /// produce them, which `Package~options` answers when sent no option
234    /// name.
235    pub(crate) fn to_options_string(&self, trace: Option<TraceMode>) -> Vec<u8> {
236        let condition = |which: &[u8]| {
237            if self.syntax.raises(which) {
238                "SYNTAX"
239            } else {
240                "CONDITION"
241            }
242        };
243        format!(
244            "::OPTIONS DIGITS {} FORM {} FUZZ {} NUMERIC {} ERROR {} FAILURE {} LOSTDIGITS {} \
245NOSTRING {} NOTREADY {} NOVALUE {} {} TRACE {}",
246            self.numeric.digits(),
247            form_word(self.numeric.form()),
248            self.numeric.fuzz(),
249            if self.numeric_inherit {
250                "INHERIT"
251            } else {
252                "NOINHERIT"
253            },
254            condition(b"ERROR"),
255            condition(b"FAILURE"),
256            condition(b"LOSTDIGITS"),
257            condition(b"NOSTRING"),
258            condition(b"NOTREADY"),
259            condition(b"NOVALUE"),
260            if self.suppress_prolog {
261                "NOPROLOG"
262            } else {
263                "PROLOG"
264            },
265            trace.map_or("?n/a?", trace_word),
266        )
267        .into_bytes()
268    }

270    /// What `Package~options(name)` answers for `which`: the value a read
271    /// returns and a write returns as the previous one. `trace` is the
272    /// setting `TRACE` renders.
273    pub(crate) fn value_of(&self, which: OptionName, trace: Option<TraceMode>) -> Vec<u8> {
274        let condition = |which: &[u8]| {
275            if self.syntax.raises(which) {
276                "SYNTAX"
277            } else {
278                "CONDITION"
279            }
280        };
281        let text = match which {
282            OptionName::All => {
283                return ESCALATABLE
284                    .iter()
285                    .map(|name| format!("{} {}", String::from_utf8_lossy(name), condition(name)))
286                    .collect::<Vec<_>>()
287                    .join(" ")
288                    .into_bytes();
289            }
290            OptionName::Digits => return self.numeric.digits().to_string().into_bytes(),
291            OptionName::Fuzz => return self.numeric.fuzz().to_string().into_bytes(),
292            OptionName::Condition(which) => condition(ESCALATABLE[which]),
293            OptionName::Form => form_word(self.numeric.form()),
294            OptionName::NumericInherit => {
295                if self.numeric_inherit {
296                    "INHERIT"
297                } else {
298                    "NOINHERIT"
299                }
300            }
301            OptionName::Prolog => {
302                if self.suppress_prolog {
303                    "NOPROLOG"
304                } else {
305                    "PROLOG"
306                }
307            }
308            OptionName::Initial | OptionName::Reset | OptionName::SetOptions => {
309                return self.to_options_string(trace);
310            }
311            OptionName::Trace => trace.map_or("?n/a?", trace_word),
312            OptionName::Explicit => return self.explicit.to_text(),
313        };
314        text.as_bytes().to_vec()
315    }

317    /// The setting switch's arms that take a value's first letter, as
318    /// `Package~options` does. `Err` is the list the 93.914 for any other
319    /// letter names.
320    pub(crate) fn set_by_letter(
321        &mut self,
322        which: OptionName,
323        letter: u8,
324    ) -> Result<(), &'static str> {
325        let letter = letter.to_ascii_uppercase();
326        match which {
327            OptionName::All | OptionName::Condition(_) => {
328                if !matches!(letter, b'S' | b'C') {
329                    return Err("\"C[ondition] or S[yntax]\"");
330                }
331                let syntax = letter == b'S';
332                match which {
333                    OptionName::Condition(which) => self.syntax.escalated[which] = syntax,
334                    _ => self.syntax.escalated = [syntax; 6],
335                }
336            }
337            OptionName::Form => match letter {
338                b'E' => self.numeric.set_form(Form::Engineering),
339                b'S' => self.numeric.set_form(Form::Scientific),
340                _ => return Err("\"E[ngineering] or S[cientific]\""),
341            },
342            OptionName::NumericInherit => match letter {
343                b'I' => self.numeric_inherit = true,
344                b'N' => self.numeric_inherit = false,
345                _ => return Err("\"I[nherit] or N[oInherit]\""),
346            },
347            OptionName::Prolog => match letter {
348                b'P' => self.suppress_prolog = false,
349                b'N' => self.suppress_prolog = true,
350                _ => return Err("\"N[oprolog] or P[rolog]\""),
351            },
352            _ => unreachable!("only the letter-valued options are set here"),
353        }
354        Ok(())
355    }

357    /// The package-settings override: every setting this package's own
358    /// `::OPTIONS` did not name, taken from `defaults`. Only `PROLOG` guards
359    /// the prologue setting: a package naming `NOPROLOG` takes it too, as one
360    /// flag is read.
361    pub(crate) fn take_defaults(&mut self, defaults: &PackageOptions) {
362        let named = |word: &str| self.explicit.named(word);
363        let pick = |word: &str, own: u64, other: u64| if named(word) { own } else { other };
364        let digits = pick("DIGITS", self.numeric.digits(), defaults.numeric.digits());
365        let fuzz = pick("FUZZ", self.numeric.fuzz(), defaults.numeric.fuzz());
366        let form = if named("FORM") {
367            self.numeric.form()
368        } else {
369            defaults.numeric.form()
370        };
371        let inherit = named("NUMERIC");
372        let prolog = named("PROLOG");
373        let trace = named("TRACE");
374        let conditions: Vec<bool> = ESCALATABLE
375            .iter()
376            .map(|name| named(&String::from_utf8_lossy(name)))
377            .collect();
378        self.numeric = Settings::from_parts(digits, fuzz, form);
379        if !inherit {
380            self.numeric_inherit = defaults.numeric_inherit;
381        }
382        for (which, named) in conditions.into_iter().enumerate() {
383            if !named {
384                self.syntax.escalated[which] = defaults.syntax.escalated[which];
385            }
386        }
387        if !prolog {
388            self.suppress_prolog = defaults.suppress_prolog;
389        }
390        if !trace {
391            self.trace = defaults.trace;
392        }
393    }

405/// One `Package~options` option name, by the letters `Package~options`
406/// switches on.
407#[derive(Copy, Clone, PartialEq, Eq, Debug)]
408pub(crate) enum OptionName {
409    /// `A[ll]`, which only a write accepts.
410    All,
411    Digits,
412    /// One of [`ESCALATABLE`], by its index there.
413    Condition(usize),
414    Form,
415    Fuzz,
416    /// `I[nitialOptions]`.
417    Initial,
418    /// `NU[meric]`.
419    NumericInherit,
420    Prolog,
421    /// `R[esetOptions]`.
422    Reset,
423    /// `S[etOptions]`, whose value is an `::OPTIONS` directive.
424    SetOptions,
425    Trace,
426    /// `X`, the options the package's directives named.
427    Explicit,
428}
429
430impl OptionName {
431    /// The option `name` names, or `None` for one none of the arms accept.
432    pub(crate) fn parse(name: &[u8]) -> Option<OptionName> {
433        let upper: Vec<u8> = name.to_ascii_uppercase();
434        Some(match (upper.first()?, upper.get(1), upper.get(2)) {
435            (b'A', _, _) => OptionName::All,
436            (b'D', _, _) => OptionName::Digits,
437            (b'E', _, _) => OptionName::Condition(0),
438            (b'F', Some(b'A'), _) => OptionName::Condition(1),
439            (b'F', Some(b'O'), _) => OptionName::Form,
440            (b'F', Some(b'U'), _) => OptionName::Fuzz,
441            (b'I', _, _) => OptionName::Initial,
442            (b'L', _, _) => OptionName::Condition(2),
443            (b'N', Some(b'U'), _) => OptionName::NumericInherit,
444            (b'N', Some(b'O'), Some(b'S')) => OptionName::Condition(3),
445            (b'N', Some(b'O'), Some(b'T')) => OptionName::Condition(4),
446            (b'N', Some(b'O'), Some(b'V')) => OptionName::Condition(5),
447            (b'P', _, _) => OptionName::Prolog,
448            (b'R', _, _) => OptionName::Reset,
449            (b'S', _, _) => OptionName::SetOptions,
450            (b'T', _, _) => OptionName::Trace,
451            (b'X', _, _) => OptionName::Explicit,
452            _ => return None,
453        })
454    }
455}

491impl Interp {
492    /// `package`'s settings for a change, whose first change enters what
493    /// `~options('I')` answers from then on.
494    pub(crate) fn package_settings_mut(&mut self, package: ObjRef) -> &mut PackageOptions {
495        self.options_armed = true;
496        let data = self
497            .package_data_mut(package)
498            .expect("a package object holds package fields");
499        let settings = data.options.get_or_insert_default();
500        data.initial_options.get_or_insert_with(|| settings.clone());
501        settings
502    }
503
504    /// The latches `Interp::install_directives` arms for an escalated
505    /// `NOSTRING` or `LOSTDIGITS`, armed for a change that escalates one.
506    pub(crate) fn arm_option_latches(&mut self, package: ObjRef) {
507        let Some(settings) = self
508            .package_data(package)
509            .and_then(|data| data.options.as_ref())
510        else {
511            return;
512        };
513        let (nostring, lostdigits) = (
514            settings.escalates_nostring(),
515            settings.escalates_lostdigits(),
516        );
517        self.reqstr_armed |= nostring;
518        self.lostdigits_armed |= lostdigits;
519    }
520
521    /// The package-settings override, which every compile of a package runs
522    /// and an `INTERPRET` does not. A restored program is not compiled, so it
523    /// neither takes the defaults nor spends a count: measured, `rexx -o
524    /// "digits 5"` on an image answers 9, and a source it requires 5.
525    pub(crate) fn override_package_settings(&mut self, id: ProgramId) -> Result<(), Failure> {
526        if self.option_writes.count == 0
527            || self
528                .programs
529                .get(id)
530                .is_some_and(|program| !program.source.is_traceable())
531        {
532            return Ok(());
533        }
534        // No latch to arm: the defaults are an `::OPTIONS` directive's,
535        // whose own install armed any they escalate.
536        let defaults = self.option_writes.defaults.clone();
537        let package = self.package_object(Package::Program(id))?;
538        self.package_settings_mut(package).take_defaults(&defaults);
539        self.option_writes.count = self.option_writes.count.wrapping_sub(1);
540        Ok(())
541    }
542
543    /// The settings `text`, a single `::OPTIONS` directive, gives a routine
544    /// compiled from it. That compile is a package's like any other, so it
545    /// takes the override too.
546    pub(crate) fn settings_from_directive(
547        &mut self,
548        text: &[u8],
549    ) -> Result<PackageOptions, Failure> {
550        let mut colon = false;
551        for &byte in text.iter().take_while(|byte| **byte != 0) {
552            if matches!(byte, b' ' | b'\t') {
553                continue;
554            }
555            if !colon {
556                if byte != b':' {
557                    return Err(Raised::method_user_defined(
558                        "argument is not an ::OPTIONS directive",
559                    )
560                    .into());
561                }
562                colon = true;
563                continue;
564            }
565            let hint = match byte {
566                b';' => "\";\" (end of clause)",
567                b'\r' => "\"0d\"x (CR)",
568                b'\n' => "\"0a\"x (LF)",
569                _ => continue,
570            };
571            return Err(Raised::method_user_defined(&format!(
572                "argument must not contain a semi-colon, CR, or LF, found: {hint}"
573            ))
574            .into());
575        }
576        let name = b"PackageSettingsFromOptionsDirectiveString";
577        let parsed = self.parse_file(text.to_vec(), name)?;
578        let mut compiled = None;
579        self.install_executable(parsed, name, true, None, false, |_, id| compiled = Some(id))?;
580        Ok(compiled
581            .and_then(|id| self.program_data(id)?.options.clone())
582            .unwrap_or_default())
583    }

591    /// `ALL` writes every one of the six and nothing outside them, asserted
592    /// over [`ESCALATABLE`] rather than over a list written here -- a name
593    /// added to that array with no arm in `ConditionSyntax::set` fails this
594    /// instead of passing unnoticed.

597        let mut options = PackageOptions::default();
598        for name in ESCALATABLE {
599            assert!(
600                !options.syntax.raises(name),
601                "a fresh package escalates {}",
602                String::from_utf8_lossy(name)
603            );
604        }
605        options
606            .apply(&PackageOption::Condition {
607                which: ConditionOption::All,
608                syntax: true,
609            })
610            .expect("a condition option makes no numeric check");
611        for name in ESCALATABLE {
612            assert!(
613                options.syntax.raises(name),
614                "`::OPTIONS ALL SYNTAX` left {} unescalated",
615                String::from_utf8_lossy(name)
616            );
617        }
618        for name in [b"HALT".as_slice(), b"SYNTAX", b"USER MINE", b""] {
619            assert!(
620                !options.syntax.raises(name),
621                "`::OPTIONS ALL SYNTAX` escalated {}, which has no `::OPTIONS` spelling",
622                String::from_utf8_lossy(name)
623            );
624        }
625    }

627    /// Each of the six is written on its own, and writing one leaves the
628    /// other five where they were. Without this an `apply` that indexed one
629    /// slot for two conditions would still pass the `ALL` test above.
630    #[test]
631    fn each_condition_option_writes_only_its_own_slot() {
632        for (which, name) in each_condition() {
633            let mut options = PackageOptions::default();
634            options
635                .apply(&PackageOption::Condition {
636                    which,
637                    syntax: true,
638                })
639                .expect("a condition option makes no numeric check");
640            for other in ESCALATABLE {
641                assert_eq!(
642                    options.syntax.raises(other),
643                    other == name,
644                    "setting {} moved {}",
645                    String::from_utf8_lossy(name),
646                    String::from_utf8_lossy(other)
647                );
648            }
649        }
650    }
651
652    /// `SIGNAL ON <c>` turns off exactly `<c>`'s escalation and `SIGNAL ON
653    /// ANY` turns off all six, while the `CALL` forms leave the three
654    /// conditions no `CALL ON` can resume from.
655    #[test]
656    fn a_trap_clause_disables_the_escalation_its_own_condition_asked_for() {
657        let uncallable = [b"LOSTDIGITS".as_slice(), b"NOSTRING", b"NOVALUE"];
658        for (_, name) in each_condition() {
659            for (spelling, signal) in [(b"ANY".as_slice(), true), (name, true), (b"ANY", false)] {
660                let mut syntax = all_escalated();
661                syntax.disable_for(spelling, signal, false);
662                for other in ESCALATABLE {
663                    let cleared = (spelling == b"ANY" || other == name)
664                        && (signal || !uncallable.contains(&other));
665                    assert_eq!(
666                        syntax.raises(other),
667                        !cleared,
668                        "{} {} left {} wrong",
669                        if signal { "SIGNAL ON" } else { "CALL ON" },
670                        String::from_utf8_lossy(spelling),
671                        String::from_utf8_lossy(other)
672                    );
673                }
674            }
675        }
676    }
677
678    /// `trapOff`'s own extra guard: a `NOVALUE` or `ANY` trap still in the
679    /// table after the removal leaves `NOVALUE`'s escalation alone, where
680    /// every other condition's is cleared regardless.
681    #[test]
682    fn a_remaining_novalue_trap_keeps_that_one_escalation() {
683        let mut syntax = all_escalated();
684        syntax.disable_for(b"NOVALUE", true, true);
685        assert!(
686            syntax.raises(b"NOVALUE"),
687            "a `NOVALUE` trap left in the table did not keep the escalation"
688        );
689        let mut syntax = all_escalated();
690        syntax.disable_for(b"NOVALUE", true, false);
691        assert!(
692            !syntax.raises(b"NOVALUE"),
693            "with no trap left the escalation must go"
694        );
695    }
696
697    /// The six, paired with the name [`ESCALATABLE`] holds for each.
698    fn each_condition() -> [(ConditionOption, &'static [u8]); 6] {
699        [
700            (ConditionOption::Error, b"ERROR".as_slice()),
701            (ConditionOption::Failure, b"FAILURE"),
702            (ConditionOption::LostDigits, b"LOSTDIGITS"),
703            (ConditionOption::NoString, b"NOSTRING"),
704            (ConditionOption::NotReady, b"NOTREADY"),
705            (ConditionOption::NoValue, b"NOVALUE"),
706        ]
707    }
708
709    /// `::OPTIONS ALL SYNTAX`, as a starting point for the clearing tests.
710    fn all_escalated() -> ConditionSyntax {
711        let mut options = PackageOptions::default();
712        options
713            .apply(&PackageOption::Condition {
714                which: ConditionOption::All,
715                syntax: true,
716            })
717            .expect("a condition option makes no numeric check");
718        options.syntax
719    }
