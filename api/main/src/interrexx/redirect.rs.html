1//! `ADDRESS ... WITH`: the permanent per-environment configuration a program
2//! sets, and the per-command context saying where one command's three streams
3//! go.
4//!
5//! A configuration is what the parser built and what an `ADDRESS env WITH ...`
6//! stores under the environment name; a context is built at **issue** time by
7//! evaluating that configuration's sources and targets in the variable context
8//! then in force.

20/// One stored `ADDRESS ... WITH` configuration and the code its names were
21/// parsed in. Its stems and expressions name symbols of that code's own table,
22/// so a configuration an `INTERPRET` stored is evaluated through the fragment
23/// it came from, against whatever frame issues the command -- as a
24/// configuration holds its own retrievers. Without the origin the fragment's
25/// ids were read in the program's table, which named another variable entirely.
26pub(crate) struct StoredIo {
27    pub(crate) io: AddressIo,
28    /// `None` for the program this activation runs, which every activation
29    /// sharing the table runs too: an internal call inherits it, and an
30    /// `INTERPRET` runs in the same activation.
31    pub(crate) fragment: Option<Rc<Fragment>>,
32}

34/// Which of a command's three streams a redirection names -- the word its
35/// `>K>` line carries and the one its errors are worded for.
36#[derive(Copy, Clone, PartialEq, Eq)]
37enum Stream {
38    Input,
39    Output,
40    Error,
41}
42
43impl Stream {
44    fn keyword(self) -> &'static str {
45        match self {
46            Stream::Input => "INPUT",
47            Stream::Output => "OUTPUT",
48            Stream::Error => "ERROR",
49        }
50    }
51}

53/// One command's resolved redirection.
54pub(crate) struct IoContext {
55    /// Every line the command reads, gathered **before it starts** but after
56    /// the security manager's `COMMAND` checkpoint has let it through
57    /// ([`IoContext::gather_input`]). `readInputBuffered` gathers before the
58    /// command starts, and that is what lets one object be a command's input
59    /// and its output at once: the doc's `sort` example reads the array
60    /// before the target empties it.
61    input: RefCell<Input>,
62    output: Option<Target>,
63    error: Option<Target>,
64    /// The output target that is also the input source, which is buffered: it
65    /// is neither emptied nor written until the command has ended. The
66    /// output's, else the error's.
67    deferred: Option<Stream>,
68    /// The output's and the error's next stem tail, from
69    /// [`IoContext::begin`] on.
70    next: [Cell<usize>; 2],
71    /// Output bytes not yet a whole line, or all of a deferred output.
72    carry: RefCell<Vec<u8>>,
73}
74
75/// A command's redirected input: what it names until the lines are read, and
76/// the lines once they are.
77enum Input {
78    /// No input is redirected.
79    None,
80    /// The source's object or name, resolved but not yet read.
81    Pending(Pending),
82    Ready(Vec<Vec<u8>>),
83}
84
85/// An input source resolved at issue time and read later.
86#[derive(Clone)]
87enum Pending {
88    /// Nothing to read: a default or normal source.
89    Empty,
90    /// `WITH INPUT STEM`: the stem object.
91    Stem(ObjRef),
92    /// `WITH INPUT USING`: the object, which decides what it means.
93    Using(ObjRef),
94    /// `WITH INPUT STREAM`: the name, opened and closed around the read.
95    Stream(Vec<u8>),
96}
97
98/// Where an input source reads from, as the buffering test compares it with an
99/// output target: a stem object, a stream object or a stream name. Any other
100/// source is never an output target.
101enum Origin {
102    Stem(ObjRef),
103    Stream(ObjRef),
104    Name(Vec<u8>),
105    Other,
106}
107
108/// Where one redirected output stream's lines end up. Every write happens at
109/// [`IoContext::finish`] rather than as the bytes arrive, which is what keeps
110/// a target that is also the input source readable until the child has run.
111enum Target {
112    /// A stem object, whose tails this rewrites.
113    Stem { object: ObjRef, append: bool },
114    /// An `OrderedCollection`: `EMPTY`, then one `APPEND` per line, so the
115    /// class's own methods decide what that means.
116    Collection { object: ObjRef, append: bool },
117    /// A `RexxQueue`, one `QUEUE` per line.
118    Queue { object: ObjRef },
119    /// An open stream, one `LINEOUT` per line.
120    ///
121    /// `qualified` is `Some` for a `STREAM name` target, which this opened and
122    /// therefore closes, and `None` for a `USING` stream object, which the
123    /// program owns and which stays open -- measured, such an object's
124    /// `~state` is still `READY` after the command while its bytes are on
125    /// disk.
126    ///
127    /// `emptied_late` marks a `REPLACE` of the file the input reads: it is
128    /// opened without emptying it where it is resolved, so that an open that
129    /// fails still keeps the command from running, and emptied only where
130    /// its deferred lines are written, after the input has been read.
131    Stream {
132        object: ObjRef,
133        qualified: Option<Vec<u8>>,
134        emptied_late: bool,
135    },
136}

138impl Target {
139    /// Whether this target is the input source `origin`, which
140    /// `needsBuffering` decides by kind and then by object, or for a named
141    /// stream by name.
142    fn reads_from(&self, origin: &Origin) -> bool {
143        match (self, origin) {
144            (Target::Stem { object, .. }, Origin::Stem(source)) => object == source,
145            (
146                Target::Stream {
147                    qualified: Some(name),
148                    ..
149                },
150                Origin::Name(source),
151            ) => name == source,
152            (
153                Target::Stream {
154                    object,
155                    qualified: None,
156                    ..
157                },
158                Origin::Stream(source),
159            ) => object == source,
160            _ => false,
161        }
162    }
163
164    fn object(&self) -> ObjRef {
165        match self {
166            Target::Stem { object, .. }
167            | Target::Collection { object, .. }
168            | Target::Queue { object }
169            | Target::Stream { object, .. } => *object,
170        }
171    }
172
173    /// Whether two targets are one destination.
174    ///
175    /// **Two named streams compare on the qualified name, not on identity**, --
176    /// `g.txt` with `g.txt` and `h.txt` with `./h.txt` each collapse to a
177    /// single file the two streams interleave into.
178    fn same_target(&self, other: &Target) -> bool {
179        match (self, other) {
180            (
181                Target::Stream {
182                    qualified: Some(one),
183                    ..
184                },
185                Target::Stream {
186                    qualified: Some(two),
187                    ..
188                },
189            ) => one == two,
190            (
191                Target::Stream {
192                    qualified: Some(_), ..
193                },
194                _,
195            )
196            | (
197                _,
198                Target::Stream {
199                    qualified: Some(_), ..
200                },
201            ) => false,
202            _ => self.object() == other.object(),
203        }
204    }
205}

213    /// What a child reads on standard input: every line with its terminator,
214    /// the last one included -- measured, `'cat -A' with input using 'x'`
215    /// prints `x$`.
216    pub(crate) fn input_bytes(&self) -> Option<Vec<u8>> {
217        let input = self.input.borrow();
218        let Input::Ready(lines) = &*input else {
219            return None;
220        };
221        let mut buffer = Vec::new();
222        for line in lines {
223            buffer.extend_from_slice(line);
224            buffer.push(b'\n');
225        }
226        Some(buffer)
227    }

229    /// Every input line, gathered before the command started, or `None`
230    /// with no input redirected.
231    pub(crate) fn input_lines(&self) -> Option<Vec<Vec<u8>>> {
232        match &*self.input.borrow() {
233            Input::Ready(lines) => Some(lines.clone()),
234            _ => None,
235        }
236    }

238    /// Reads the input source, once, where the command is about to run. The
239    /// security manager's `COMMAND` checkpoint comes first, and a manager
240    /// that takes the command reads nothing from its source.
241    pub(crate) fn gather_input(&self, interp: &mut Interp) -> Result<(), Failure> {
242        let pending = match &*self.input.borrow() {
243            Input::Pending(pending) => pending.clone(),
244            _ => return Ok(()),
245        };
246        let lines = match pending {
247            Pending::Empty => Vec::new(),
248            Pending::Stem(object) => interp.stem_input_lines(object)?,
249            Pending::Using(object) => interp.using_input_lines(object)?,
250            Pending::Stream(name) => {
251                let stream = interp.open_named_stream(&name, b"READ", true)?;
252                let lines = interp.read_stream_lines(stream)?;
253                let caller = interp.caller();
254                interp.send_message(stream, b"CLOSE", None, &[], caller)?;
255                lines
256            }
257        };
258        *self.input.borrow_mut() = Input::Ready(lines);
259        Ok(())
260    }

280    /// Whether both output streams resolved to one object, in which case the
281    /// child writes them down a single pipe and they interleave in the order
282    /// it wrote them -- measured, `echo out1; echo err1 1>&2; echo out2` with
283    /// both streams on one stem fills it with `out1`, `err1`, `out2`.
284    pub(crate) fn shares_one_target(&self) -> bool {
285        match (&self.output, &self.error) {
286            (Some(output), Some(error)) => output.same_target(error),
287            _ => false,
288        }
289    }

291    /// Before the command runs: a `REPLACE` stem is emptied with `stem.0` set
292    /// to 0, an `APPEND` one reads its `stem.0`, and a `REPLACE` collection
293    /// is sent `EMPTY`. A deferred target and an error target shared with the
294    /// output are left for later.
295    pub(crate) fn begin(&self, interp: &mut Interp) -> Result<(), Failure> {
296        if let Some(target) = &self.output
297            && self.deferred != Some(Stream::Output)
298        {
299            self.begin_target(interp, target, 0)?;
300        }
301        if let Some(target) = &self.error
302            && !self.shares_one_target()
303            && self.deferred != Some(Stream::Error)
304        {
305            self.begin_target(interp, target, 1)?;
306        }
307        Ok(())
308    }

310    fn begin_target(&self, interp: &mut Interp, target: &Target, at: usize) -> Result<(), Failure> {
311        let next = pinned!(
312            interp,
313            crate::pinning::PinKind::RedirectWrapper,
314            interp.begin_target(target)
315        )?;
316        self.next[at].set(next);
317        Ok(())
318    }

320    /// Hands the whole lines of `bytes`, standard output's next piece, to the
321    /// output target as they arrive, and keeps a trailing part line for the
322    /// next piece. A deferred output keeps everything.
323    ///
324    /// A line the target raises on ends this piece: its later lines and its
325    /// part line are dropped, as the unwind out of the buffer write drops them,
326    /// and the next piece is offered as usual.
327    pub(crate) fn write_output(&self, interp: &mut Interp, bytes: &[u8]) -> Result<(), Failure> {
328        let Some(target) = &self.output else {
329            return Ok(());
330        };
331        let lines = {
332            let mut carry = self.carry.borrow_mut();
333            carry.extend_from_slice(bytes);
334            if self.deferred == Some(Stream::Output) {
335                return Ok(());
336            }
337            whole_lines(&mut carry)
338        };
339        let written = self.write_lines(interp, target, 0, &lines);
340        if written.is_err() {
341            self.carry.borrow_mut().clear();
342        }
343        written
344    }

346    /// Hands what the child wrote to the targets once it has ended, in this
347    /// order: `out`, standard output's last piece, then standard error, then
348    /// the cleanup, which writes the output's last part line, or all of a
349    /// deferred output, and closes each named stream.
350    ///
351    /// `held` is the condition a target raised on an earlier piece. A condition
352    /// raised here replaces it ([`Interp::write_holding_last`]); one standing
353    /// once standard error has been offered is raised again at the command's
354    /// clause and answered, and the cleanup does not run, as it is raised
355    /// before the cleanup.
356    ///
357    /// Under [`IoContext::shares_one_target`] the caller has already merged
358    /// both streams into `out` and `err` is empty.
359    pub(crate) fn finish(
360        &self,
361        interp: &mut Interp,
362        out: &[u8],
363        err: &[u8],
364        held: Option<Failure>,
365    ) -> Result<(), Failure> {
366        let mut held = held;
367        interp.write_holding_last(&mut held, |interp| self.write_output(interp, out))?;
368        let error = self.error.as_ref().filter(|_| !self.shares_one_target());
369        if let Some(target) = error {
370            interp.write_holding_last(&mut held, |interp| {
371                if self.deferred == Some(Stream::Error) {
372                    self.begin_target(interp, target, 1)?;
373                }
374                self.write_lines(interp, target, 1, &split_lines(err))
375            })?;
376        }
377        if let Some(held) = held {
378            // Raised again at the command's clause, which the report names.
379            interp.reraise_failure_levels();
380            return Err(held);
381        }
382        if let Some(target) = &self.output {
383            if self.deferred == Some(Stream::Output) {
384                self.begin_target(interp, target, 0)?;
385            }
386            let rest = split_lines(&std::mem::take(&mut *self.carry.borrow_mut()));
387            self.write_lines(interp, target, 0, &rest)?;
388            self.write_deferred_again(interp, &rest)?;
389        }
390        self.close_named(interp)
391    }
392
393    /// The lines of a deferred output, written once more to an error target
394    /// it shares: measured, a stem that is the input, the output and the error
395    /// ends up holding everything the command wrote twice over, the second
396    /// time after the first, and so does a file.
397    fn write_deferred_again(&self, interp: &mut Interp, lines: &[Vec<u8>]) -> Result<(), Failure> {
398        if self.deferred != Some(Stream::Output) || !self.shares_one_target() {
399            return Ok(());
400        }
401        let Some(error) = &self.error else {
402            return Ok(());
403        };
404        self.write_lines(interp, error, 0, lines)
405    }

407    /// Closes each named stream this redirection opened, an error target
408    /// that shares the output's file among them, which is a stream of its own.
409    fn close_named(&self, interp: &mut Interp) -> Result<(), Failure> {
410        for target in self.output.iter().chain(&self.error) {
411            if let Target::Stream {
412                object,
413                qualified: Some(_),
414                ..
415            } = *target
416            {
417                let caller = interp.caller();
418                interp.send_message(object, b"CLOSE", None, &[], caller)?;
419            }
420        }
421        Ok(())
422    }

424    /// Hands each redirected stream's remaining lines to its target, then
425    /// closes each named stream. Under [`IoContext::shares_one_target`]
426    /// every line is in `out`.
427    pub(crate) fn finish_lines(
428        &self,
429        interp: &mut Interp,
430        out: &[Vec<u8>],
431        err: &[Vec<u8>],
432    ) -> Result<(), Failure> {
433        if let Some(target) = &self.output {
434            if self.deferred == Some(Stream::Output) {
435                self.begin_target(interp, target, 0)?;
436            }
437            self.write_lines(interp, target, 0, out)?;
438            self.write_deferred_again(interp, out)?;
439        }
440        // **One object for both streams is written once.** Everything the
441        // command produced is in `out` and `err` is empty.
442        let error = self.error.as_ref().filter(|_| !self.shares_one_target());
443        if let Some(target) = error {
444            if self.deferred == Some(Stream::Error) {
445                self.begin_target(interp, target, 1)?;
446            }
447            self.write_lines(interp, target, 1, err)?;
448        }
449        self.close_named(interp)
450    }
451
452    fn write_lines(
453        &self,
454        interp: &mut Interp,
455        target: &Target,
456        at: usize,
457        lines: &[Vec<u8>],
458    ) -> Result<(), Failure> {
459        if lines.is_empty() {
460            return Ok(());
461        }
462        let next = pinned!(
463            interp,
464            crate::pinning::PinKind::RedirectWrapper,
465            interp.write_target(target, self.next[at].get(), lines)
466        )?;
467        self.next[at].set(next);
468        Ok(())
469    }
470}

472/// The whole lines at the front of `carry`, taken out of it, each without its
473/// terminator as [`split_lines`] reads one; the part line after the last `\n`
474/// stays.
475fn whole_lines(carry: &mut Vec<u8>) -> Vec<Vec<u8>> {
476    let Some(last) = carry.iter().rposition(|byte| *byte == b'\n') else {
477        return Vec::new();
478    };
479    let rest = carry.split_off(last + 1);
480    let lines = split_lines(carry);
481    *carry = rest;
482    lines
483}

485/// One captured stream as lines.
486///
487/// `\n` ends a line and a `\r` immediately before it is part of that one
488/// terminator; a `\r` anywhere else is data, as a NUL always is; and a
489/// trailing remainder with no terminator is a line of its own. Measured:
490/// `printf "a\nb"` gives two, `printf "a\r\nb\r\n"` gives two whose first is
491/// one byte long, `printf "a\0b\n"` gives one of three bytes, and
492/// `printf ""` gives none.
493pub(crate) fn split_lines(bytes: &[u8]) -> Vec<Vec<u8>> {
494    let mut lines = Vec::new();
495    let mut at = 0;
496    while at < bytes.len() {
497        match bytes[at..].iter().position(|byte| *byte == b'\n') {
498            Some(offset) => {
499                let mut end = at + offset;
500                if end > at && bytes[end - 1] == b'\r' {
501                    end -= 1;
502                }
503                lines.push(bytes[at..end].to_vec());
504                at += offset + 1;
505            }
506            None => {
507                lines.push(bytes[at..].to_vec());
508                break;
509            }
510        }
511    }
512    lines
513}

515/// One stream's effective redirection and the option that came with it.
516///
517/// **The local configuration decides only where it says something.**
518/// `NORMAL` is a reset rather than silence, a local target replaces the
519/// permanent one, and a stream the local configuration does not mention keeps
520/// the permanent target -- measured, a permanent `with output stem o.` and a
521/// one-off `with error stem e.` fill both. The option travels with whichever
522/// configuration supplied the target, as it does in `createOutputTarget`.
523/// The answer's `bool` is whether the local configuration supplied it, which
524/// decides the code its names are read in.
525fn merge<'a>(
526    local: Option<(&'a Redirection, OutputOption)>,
527    global: Option<(&'a Redirection, OutputOption)>,
528) -> Option<(&'a Redirection, OutputOption, bool)> {
529    match local {
530        Some((Redirection::Normal, _)) => None,
531        Some((Redirection::Default, _)) | None => match global {
532            Some((Redirection::Default | Redirection::Normal, _)) | None => None,
533            Some((found, option)) => Some((found, option, false)),
534        },
535        Some((found, option)) => Some((found, option, true)),
536    }
537}

552    /// `addIOConfig`: makes `io` the configuration for `environment`.
553    ///
554    /// The table is cloned first when this activation inherited one, which is
555    /// `checkIOConfigTable`'s lazy copy and what keeps a callee's write off
556    /// its caller.
557    pub(crate) fn add_io_config(&mut self, environment: &[u8], io: Rc<StoredIo>) {
558        let upper = environment.to_ascii_uppercase().into_boxed_slice();
559        let table = self
560            .activation_mut()
561            .io_configs
562            .get_or_insert_with(|| Rc::new(IoConfigs::default()));
563        Rc::make_mut(table).insert(upper, io);
564    }

566    /// `setAddress`' second half: the `WITH` configuration an `ADDRESS`
567    /// instruction carried becomes the one stored under the name it just set.
568    ///
569    /// An instruction with no `WITH` stores nothing and leaves whatever that
570    /// name already holds in place.
571    /// `code` is the code the instruction sits in: the configuration keeps
572    /// the fragment it came from, where it came from one.
573    pub(crate) fn store_io_config(
574        &mut self,
575        code: &Code<'_>,
576        environment: &[u8],
577        address: &interrexx_parse::Address,
578    ) {
579        if let Some(io) = &address.io {
580            let fragment = self
581                .activity
582                .fragments
583                .iter()
584                .rev()
585                .filter_map(|level| level.fragment.as_ref())
586                .find(|fragment| std::ptr::eq(&fragment.symbols, code.symbols))
587                .cloned();
588            let stored = StoredIo {
589                io: io.as_ref().clone(),
590                fragment,
591            };
592            self.add_io_config(environment, Rc::new(stored));
593        }
594    }

596    /// `resolveAddressIOConfig`: the context one command runs under, or
597    /// `None` where neither the environment nor the instruction redirects
598    /// anything.
599    ///
600    /// Sources and targets are evaluated here, at issue time and in the
601    /// variable context in force -- so a permanent configuration inherited by
602    /// a callee fills the **callee's** own stem, measured. The order is
603    /// `INPUT`, `OUTPUT`, `ERROR`, which their `>K>` lines show.
604    pub(crate) fn io_context(
605        &mut self,
606        code: &Code<'_>,
607        environment: &[u8],
608        local: Option<&AddressIo>,
609    ) -> Result<Option<IoContext>, Failure> {
610        let stored = self.io_config(environment);
611        if stored.is_none() && local.is_none() {
612            return Ok(None);
613        }
614        // The stored configuration's own code, where it is not `code`: the
615        // fragment it came from, planned against this frame as an
616        // `INTERPRET` plans one, or the program when `code` is a fragment.
617        let program = Rc::clone(&self.activation().program);
618        let program_plan = Rc::clone(&self.activation().plan);
619        let program_body = crate::activation::body_of(&program, self.activation().body);
620        let fragment = stored.as_ref().and_then(|stored| stored.fragment.clone());
621        let fragment_planned = match &fragment {
622            Some(fragment) if !std::ptr::eq(&fragment.symbols, code.symbols) => {
623                Some(self.fragment_plan(fragment))
624            }
625            _ => None,
626        };
627        let stored_code = match (&fragment, &fragment_planned, program_body) {
628            (Some(fragment), Some(plan), _) => Some(Code {
629                body: &fragment.body,
630                symbols: &fragment.symbols,
631                plan: Some(plan),
632            }),
633            (None, _, Some(body)) if !std::ptr::eq(&program.symbols, code.symbols) => Some(Code {
634                body,
635                symbols: &program.symbols,
636                plan: Some(&program_plan),
637            }),
638            _ => None,
639        };
640        let stored_code = stored_code.as_ref().unwrap_or(code);
641        let code_for = |local: bool| if local { code } else { stored_code };
642        let global = stored.as_deref().map(|stored| &stored.io);
643        let input = merge(
644            local.map(|io| (&io.input, OutputOption::Default)),
645            global.map(|io| (&io.input, OutputOption::Default)),
646        );
647        let output = merge(
648            local.map(|io| (&io.output, io.output_option)),
649            global.map(|io| (&io.output, io.output_option)),
650        );
651        let error = merge(
652            local.map(|io| (&io.error, io.error_option)),
653            global.map(|io| (&io.error, io.error_option)),
654        );
655        let (input, origin) = match input {
656            Some((source, _, local)) => {
657                let (pending, origin) = self.resolve_input(code_for(local), source)?;
658                (Input::Pending(pending), origin)
659            }
660            None => (Input::None, Origin::Other),
661        };
662        let input_name = match &origin {
663            Origin::Name(name) => Some(name.as_slice()),
664            _ => None,
665        };
666        let output = match output {
667            Some((target, option, local)) => Some(self.output_target(
668                code_for(local),
669                target,
670                option,
671                Stream::Output,
672                input_name,
673            )?),
674            None => None,
675        };
676        let error = match error {
677            Some((target, option, local)) => Some(self.output_target(
678                code_for(local),
679                target,
680                option,
681                Stream::Error,
682                input_name,
683            )?),
684            None => None,
685        };
686        let deferred = if output
687            .as_ref()
688            .is_some_and(|target| target.reads_from(&origin))
689        {
690            Some(Stream::Output)
691        } else if error
692            .as_ref()
693            .is_some_and(|target| target.reads_from(&origin))
694        {
695            Some(Stream::Error)
696        } else {
697            None
698        };
699        Ok(Some(IoContext {
700            input: RefCell::new(input),
701            output,
702            error,
703            deferred,
704            next: [Cell::new(1), Cell::new(1)],
705            carry: RefCell::new(Vec::new()),
706        }))
707    }

709    /// What a redirected input reads from, resolved but not read, and where it
710    /// reads.
711    fn resolve_input(
712        &mut self,
713        code: &Code<'_>,
714        source: &Redirection,
715    ) -> Result<(Pending, Origin), Failure> {
716        Ok(match source {
717            // `merge` answers `None` for both, so neither reaches here.
718            Redirection::Default | Redirection::Normal => (Pending::Empty, Origin::Other),
719            Redirection::Stem(id) => {
720                let object = self.redirect_stem(code, *id, Stream::Input)?;
721                (Pending::Stem(object), Origin::Stem(object))
722            }
723            Redirection::Using(expression) => {
724                let object = self.eval(code, code.body.expr(*expression))?;
725                self.roots.activity_mut().push_temp(object);
726                self.trace_redirect_target(Stream::Input, object)?;
727                let origin = if self.is_stem_object(object) {
728                    Origin::Stem(object)
729                } else if self.is_input_stream_object(object) {
730                    Origin::Stream(object)
731                } else {
732                    Origin::Other
733                };
734                (Pending::Using(object), origin)
735            }
736            Redirection::Stream(expression) => {
737                let name =
738                    self.redirect_stream_name(code, code.body.expr(*expression), Stream::Input)?;
739                (Pending::Stream(name.clone()), Origin::Name(name))
740            }
741        })
742    }

744    /// A stem read as input: `stem.1` through `stem.<stem.0>`.
745    ///
746    /// A tail with no value of its own contributes the stem's **default**,
747    /// and with no default its own derived name -- measured, a stem holding
748    /// `IN.1` and `IN.3` with `IN.0 = 3` feeds `one`, `IN.` and `three`,
749    /// which is `getFullElement`'s ordinary read.
750    fn stem_input_lines(&mut self, object: ObjRef) -> Result<Vec<Vec<u8>>, Failure> {
751        let Some(count) = self.redirect_stem_count(object)? else {
752            let name = self.stem_object_name(object);
753            return Err(Raised::stem_without_a_size(&name).into());
754        };
755        let mut lines = Vec::with_capacity(count);
756        for index in 1..=count {
757            let key = index.to_string().into_bytes();
758            let value = self.stem_object_read(object, &key);
759            lines.push(self.string_value_text(value)?);
760        }
761        Ok(lines)
762    }

764    /// `WITH INPUT USING expr`, whose object decides what it means.
765    ///
766    /// The order is `createInputSource`'s own: a String is one line; a stem
767    /// is read as `STEM` reads one; an Array contributes its items and skips
768    /// its holes; anything else is asked for an array and is 98.924 if it
769    /// does not answer one -- measured, `.nil` and `42` are the two ends of
770    /// that, `42` being a String.
771    fn using_input_lines(&mut self, object: ObjRef) -> Result<Vec<Vec<u8>>, Failure> {
772        if is_string(self, object) {
773            return Ok(vec![self.string_value_text(object)?]);
774        }
775        if self.is_stem_object(object) {
776            return self.stem_input_lines(object);
777        }
778        if self.is_input_stream_object(object) {
779            return self.read_stream_lines(object);
780        }
781        if let Some(path) = self.file_object_path(object)? {
782            let stream = self.open_named_stream(&path, b"READ", true)?;
783            let lines = self.read_stream_lines(stream)?;
784            let caller = self.caller();
785            self.send_message(stream, b"CLOSE", None, &[], caller)?;
786            return Ok(lines);
787        }
788        let array = match self.array_slots_of(object) {
789            Some(_) => object,
790            None => {
791                let caller = self.caller();
792                let selector = self.text(b"ARRAY");
793                self.roots.activity_mut().push_temp(selector);
794                let answered = self
795                    .send_message(object, b"REQUEST", None, &[Some(selector)], caller)?
796                    .unwrap_or(ObjRef::NIL);
797                self.roots.activity_mut().push_temp(answered);
798                answered
799            }
800        };
801        let Some(slots) = self.array_slots_of(array) else {
802            let shown = self.string_value_text(object)?;
803            return Err(Raised::address_input_source(&shown).into());
804        };
805        let mut lines = Vec::new();
806        // **An item that is not a string follows the string argument rule**:
807        // `REQUEST('STRING')` (so `MAKESTRING`) first, then `STRING`.
808        for item in slots.into_iter().flatten() {
809            let converted = self.required_string_value(item)?;
810            lines.push(self.string_value_text(converted)?);
811        }
812        Ok(lines)
813    }

815    /// `WITH OUTPUT`/`ERROR`, whose target is resolved but not yet written.
816    fn output_target(
817        &mut self,
818        code: &Code<'_>,
819        target: &Redirection,
820        option: OutputOption,
821        stream: Stream,
822        input_name: Option<&[u8]>,
823    ) -> Result<Target, Failure> {
824        let append = option == OutputOption::Append;
825        match target {
826            // `merge` answers `None` for both, so neither reaches here.
827            Redirection::Default | Redirection::Normal => {
828                unreachable!("a stream `merge` answered `None` for reached the target builder")
829            }
830            Redirection::Stem(id) => {
831                let object = self.redirect_stem(code, *id, stream)?;
832                Ok(Target::Stem { object, append })
833            }
834            Redirection::Using(expression) => {
835                let object = self.eval(code, code.body.expr(*expression))?;
836                self.roots.activity_mut().push_temp(object);
837                self.trace_redirect_target(stream, object)?;
838                self.using_output_target(object, option, input_name)
839            }
840            Redirection::Stream(expression) => {
841                let name = self.redirect_stream_name(code, code.body.expr(*expression), stream)?;
842                self.named_output_target(name, append, input_name)
843            }
844        }
845    }

847    /// The file `qualified` opened as an output target, emptied now for a
848    /// `REPLACE` unless it is the file the input reads.
849    fn named_output_target(
850        &mut self,
851        qualified: Vec<u8>,
852        append: bool,
853        input_name: Option<&[u8]>,
854    ) -> Result<Target, Failure> {
855        let emptied_late = !append && input_name == Some(qualified.as_slice());
856        let mode: &[u8] = match append || emptied_late {
857            true => b"WRITE APPEND",
858            false => b"WRITE REPLACE",
859        };
860        let object = self.open_named_stream(&qualified, mode, false)?;
861        Ok(Target::Stream {
862            object,
863            qualified: Some(qualified),
864            emptied_late,
865        })
866    }

868    /// `WITH OUTPUT USING expr`, in `createOutputTarget`'s own order: a stem,
869    /// then a stream-shaped object, then an `OrderedCollection`, and 98.996
870    /// for anything else.
871    fn using_output_target(
872        &mut self,
873        object: ObjRef,
874        option: OutputOption,
875        input_name: Option<&[u8]>,
876    ) -> Result<Target, Failure> {
877        let append = option == OutputOption::Append;
878        // **Whether an option was written matters here, not just which one.**
879        // Neither `REPLACE` nor `APPEND` means anything to an object that is
880        // not a file, and both are refused rather than ignored.
881        let optioned = option != OutputOption::Default;
882        if self.is_stem_object(object) {
883            return Ok(Target::Stem { object, append });
884        }
885        if self.is_rexx_queue(object) {
886            if optioned {
887                return Err(Raised::queue_target_option().into());
888            }
889            return Ok(Target::Queue { object });
890        }
891        if self.is_output_stream_object(object) {
892            if optioned {
893                return Err(Raised::stream_target_option().into());
894            }
895            return Ok(Target::Stream {
896                object,
897                qualified: None,
898                emptied_late: false,
899            });
900        }
901        if let Some(path) = self.file_object_path(object)? {
902            return self.named_output_target(path, append, input_name);
903        }
904        if self.is_ordered_collection(object) {
905            return Ok(Target::Collection { object, append });
906        }
907        let shown = self.string_value_text(object)?;
908        Err(Raised::address_output_target(&shown).into())
909    }

944    /// `init` of one target, answering the stem tail its first line goes to.
945    ///
946    /// Emptying a stem drops every tail and **keeps the stem's own default**,
947    /// measured: a stem with `o. = 'dflt'` and `o.7` set answers `dflt` for
948    /// `o.7` again after one line of output. An `APPEND` stem with no `stem.0`
949    /// of its own is emptied too -- a default is not a size count, measured
950    /// with both `drop p.` and `q. = 'q'`.
951    fn begin_target(&mut self, target: &Target) -> Result<usize, Failure> {
952        match *target {
953            Target::Stem { object, append } => {
954                if append && let Some(count) = self.redirect_stem_count(object)? {
955                    return Ok(count + 1);
956                }
957                self.stem_object_clear(object);
958                let zero = self.counted(0);
959                self.stem_object_write(object, b"0", zero);
960                Ok(1)
961            }
962            Target::Collection { object, append } => {
963                if !append {
964                    let caller = self.caller();
965                    self.send_message(object, b"EMPTY", None, &[], caller)?;
966                }
967                Ok(1)
968            }
969            Target::Stream {
970                object,
971                ref qualified,
972                emptied_late: true,
973            } => {
974                let caller = self.caller();
975                self.send_message(object, b"CLOSE", None, &[], caller)?;
976                let option = self.text(b"WRITE REPLACE");
977                self.roots.activity_mut().push_temp(option);
978                let caller = self.caller();
979                let answer = self
980                    .send_message(object, b"OPEN", None, &[Some(option)], caller)?
981                    .unwrap_or(ObjRef::NIL);
982                let answered = self.string_value_text(answer)?;
983                if answered != b"READY:" {
984                    let qualified = qualified.as_deref().unwrap_or_default();
985                    return Err(Raised::stream_not_writeable(qualified, &answered).into());
986                }
987                Ok(1)
988            }
989            // Opened, and emptied for `REPLACE`, where it was resolved.
990            Target::Stream { .. } | Target::Queue { .. } => Ok(1),
991        }
992    }

994    /// Writes `lines` to one target, a stem's from tail `next` on with
995    /// `stem.0` after each, answering the next tail.
996    fn write_target(
997        &mut self,
998        target: &Target,
999        mut next: usize,
1000        lines: &[Vec<u8>],
1001    ) -> Result<usize, Failure> {
1002        for line in lines {
1003            let value = self.text(line);
1004            self.roots.activity_mut().push_temp(value);
1005            match *target {
1006                Target::Stem { object, .. } => {
1007                    let key = next.to_string().into_bytes();
1008                    self.stem_object_write(object, &key, value);
1009                    let count = self.counted(next);
1010                    self.stem_object_write(object, b"0", count);
1011                }
1012                Target::Collection { object, .. } => {
1013                    let caller = self.caller();
1014                    self.send_message(object, b"APPEND", None, &[Some(value)], caller)?;
1015                }
1016                Target::Stream { object, .. } => {
1017                    let caller = self.caller();
1018                    self.send_message(object, b"LINEOUT", None, &[Some(value)], caller)?;
1019                }
1020                Target::Queue { object } => {
1021                    let caller = self.caller();
1022                    self.send_message(object, b"QUEUE", None, &[Some(value)], caller)?;
1023                }
1024            }
1025            next += 1;
1026        }
1027        Ok(next)
1028    }

1046    /// A `.Stream` opened for one redirection.
1047    ///
1048    /// **The open happens before the command runs**, and its failure is
1049    /// 98.999 for reading or 98.920 for writing, each carrying the qualified
1050    /// name and the `ERROR:n` the open answered. Measured: a command whose
1051    /// output stream cannot be opened never spawns, and a file it would have
1052    /// written is absent afterwards.
1053    fn open_named_stream(
1054        &mut self,
1055        qualified: &[u8],
1056        mode: &[u8],
1057        reading: bool,
1058    ) -> Result<ObjRef, Failure> {
1059        let Some(class) = self.rexx_package_class(b"STREAM") else {
1060            // Only before the library program has installed `Stream`, which is
1061            // the library bootstrap's own step.
1062            return Err(Loud::environment_symbol(b".STREAM").into());
1063        };
1064        let argument = self.text(qualified);
1065        self.roots.activity_mut().push_temp(argument);
1066        let caller = self.caller();
1067        let object = self
1068            .send_message(class, b"NEW", None, &[Some(argument)], caller)?
1069            .ok_or_else(|| Failure::from(Raised::no_result(b"NEW")))?;
1070        self.roots.activity_mut().push_temp(object);
1071        let option = self.text(mode);
1072        self.roots.activity_mut().push_temp(option);
1073        let caller = self.caller();
1074        let answer = self
1075            .send_message(object, b"OPEN", None, &[Some(option)], caller)?
1076            .unwrap_or(ObjRef::NIL);
1077        let answered = self.string_value_text(answer)?;
1078        if answered != b"READY:" {
1079            let raised = match reading {
1080                true => Raised::stream_not_readable(qualified, &answered),
1081                false => Raised::stream_not_writeable(qualified, &answered),
1082            };
1083            return Err(raised.into());
1084        }
1085        Ok(object)
1086    }

1088    /// Every line a stream still holds, read as `StreamObjectInputSource`
1089    /// reads: `LINEIN` until a send of it raises a condition, `NOTREADY` at
1090    /// the end of a stream.
1091    ///
1092    /// **The condition decides, not the answer.** A blank line answers `''`
1093    /// and so does the read past the end -- measured over a file of `l1`,
1094    /// `l2`, an empty line and `l4`, four reads answer with `state` `READY`
1095    /// throughout and only the fifth is `NOTREADY`, so stopping on an empty
1096    /// answer would swallow every line after a blank one.
1097    fn read_stream_lines(&mut self, stream: ObjRef) -> Result<Vec<Vec<u8>>, Failure> {
1098        let mut lines = Vec::new();
1099        loop {
1100            let dispatcher = self.running_activation().map(|activation| activation.id);
1101            let outer = std::mem::replace(&mut self.activity.input_dispatch, dispatcher);
1102            let outer_trapped = std::mem::take(&mut self.activity.input_dispatch_trapped);
1103            let caller = self.caller();
1104            let answered = self.send_message(stream, b"LINEIN", None, &[], caller);
1105            let trapped =
1106                std::mem::replace(&mut self.activity.input_dispatch_trapped, outer_trapped);
1107            self.activity.input_dispatch = outer;
1108            let value = match answered {
1109                Ok(Some(value)) => value,
1110                // `readBuffered` stops at a `LINEIN` that answers nothing.
1111                Ok(None) => return Ok(lines),
1112                Err(Failure::Raised(raised)) if raised.condition == "SYNTAX" => {
1113                    self.activity.input_dispatch_syntax = Some(Failure::Raised(raised));
1114                    return Ok(lines);
1115                }
1116                Err(failure) => return Err(failure),
1117            };
1118            // A condition the send raised ends the input, and the line it
1119            // answered with is not one.
1120            if trapped {
1121                return Ok(lines);
1122            }
1123            lines.push(self.string_value_text(value)?);
1124        }
1125    }

1157    /// One tail's own value, with no fallback to the stem's default and none
1158    /// to a derived name. A tombstone answers `None`, as an absent tail does.
1159    pub(crate) fn stem_object_tail(&self, object: ObjRef, key: &[u8]) -> Option<ObjRef> {
1160        let Body::Stem { tails, exposed, .. } = &self.heap.get(object)?.body else {
1161            return None;

1164        if let Some(home) = crate::stem::exposed_home(exposed, key) {
1165            return self.tail_value(home, key);
1166        }

1168    }

1170    /// One tail as `getFullElement` answers it: its own value, else the
1171    /// stem's default.
1172    ///
1173    /// **A stem with no default answers its own name, with no tail appended**
1174    /// -- measured, a stem holding `IN.1` and `IN.3` under `IN.0 = 3` feeds a
1175    /// command `one`, `IN.` and `three`. That is not the `IN.2` an ordinary
1176    /// `say in.2` derives: this read goes through the stem object, whose
1177    /// unset value is the name it was created under.
1178    fn stem_object_read(&mut self, object: ObjRef, key: &[u8]) -> ObjRef {
1179        let resolved = {
1180            let Some(found) = self.heap.get(object) else {
1181                return ObjRef::NIL;
1182            };
1183            let Body::Stem {
1184                default,
1185                tails,
1186                exposed,
1187                ..
1188            } = &found.body
1189            else {
1190                return ObjRef::NIL;
1191            };
1192            match crate::stem::exposed_home(exposed, key) {
1193                Some(home) => self.tail_value(home, key),
1194                None => match tails.get(key) {
1195                    Some(value) => value,
1196                    None => *default,
1197                },
1198            }
1199        };
1200        match resolved {
1201            Some(value) => value,
1202            None => {
1203                let name = self.stem_object_name(object);
1204                self.text(&name)
1205            }
1206        }
1207    }

1209    /// Writes one tail of a stem object, keeping the insertion ordinal a tail
1210    /// already has -- the order `allIndexes` answers in depends on it.
1211    pub(crate) fn stem_object_write(&mut self, object: ObjRef, key: &[u8], value: ObjRef) {
1212        let Some(found) = self.heap.get_mut(object) else {
1213            return;
1214        };
1215        let Body::Stem { tails, exposed, .. } = &mut found.body else {
1216            return;
1217        };
1218        if let Some(home) = crate::stem::exposed_home(exposed, key) {
1219            self.set_tail(home, key, Some(value));
1220            return;
1221        }
1222        tails.set(self.hash_seed, key, Some(value));
1223    }

1225    /// Drops every tail **in place**, leaving the stem's default and the
1226    /// variable's binding alone.
1227    ///
1228    /// Not `stem_drop`, which allocates a fresh object and rebinds the
1229    /// variable for the stem aliasing rule: here the object is the one the
1230    /// redirection resolved to and the program's own name for it may not even
1231    /// be in scope.
1232    fn stem_object_clear(&mut self, object: ObjRef) {
1233        let Some(found) = self.heap.get_mut(object) else {
1234            return;
1235        };
1236        let Body::Stem { tails, exposed, .. } = &mut found.body else {
1237            return;
1238        };
1239        tails.clear();
1240        *exposed = None;
1241    }

1244    fn stem_object_name(&self, object: ObjRef) -> Vec<u8> {
1245        match self.heap.get(object).map(|found| &found.body) {
1246            Some(Body::Stem { name, .. }) => name.to_vec(),
1247            _ => Vec::new(),
1248        }
1249    }

1301    /// `isInstanceOf` against a class the `REXX` package defines, answering
1302    /// `false` where that class is not defined at all.
1303    pub(crate) fn is_instance_of_rexx_class(&mut self, value: ObjRef, upper: &[u8]) -> bool {
1304        let Some(class) = self.rexx_package_class(upper) else {
1305            return false;
1306        };
1307        let Some(of) = self.class_of_value(value) else {
1308            return false;
1309        };
1310        self.classes().is_a(of, class)
1311    }

1314/// Whether `value` is a String.
1315///
1316/// Every string-valued primitive is one, a whole number in the handle's own
1317/// tag included -- `receiver_kind` answers `Primitive::String` for a literal
1318/// and `Primitive::SmallInt` for a small integer, and both take the `String`
1319/// class. Measured, `with input using 42` feeds one line rather than being
1320/// asked for an array.
1321fn is_string(interp: &Interp, value: ObjRef) -> bool {
1322    match value.decode() {
1323        Decoded::Text(_) | Decoded::SmallInt(_) => true,
1324        Decoded::Heap { .. } => matches!(
1325            interp.heap.get(value).map(|found| &found.body),
1326            Some(Body::Text { .. } | Body::Num { .. })
1327        ),
1328        Decoded::Nil => false,
1329    }
1330}

1332/// `text` as a non-negative whole number, or `None` for anything else --
1333/// what a `stem.0` has to be. Any spelling of a whole number counts (`1.0`,
1334/// `1e0`, `+1`, `' 1 '`), bounded by the argument rule's 18 digits whatever
1335/// the NUMERIC DIGITS setting.
1336fn whole_number(text: &[u8]) -> Option<usize> {
1337    let number = interrexx_num::Number::parse_bytes(text)?;
1338    let whole = number.whole_value(interrexx_num::ARGUMENT_DIGITS)?;
1339    usize::try_from(whole).ok()
