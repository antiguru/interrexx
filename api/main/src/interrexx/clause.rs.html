8/// A wall-clock bound on a whole run, honoured at the clause boundary and
9/// unset by default.
10pub(crate) struct Deadline {
11    /// When the run must stop.
12    at: Instant,
13    /// Whether the clock has already been found past [`Deadline::at`].
14    expired: bool,
15}
16
17impl Deadline {
18    /// Clauses between two visits to [`Interp::countdown_reached`], with or
19    /// without a deadline.
20    pub(crate) const CLAUSES_PER_CHECK: u32 = 1024;
21
22    /// A bound `limit` from now.
23    #[expect(clippy::disallowed_methods, reason = "the run's deadline")]
24    pub(crate) fn starting_now(limit: Duration) -> Deadline {
25        Deadline {
26            at: Instant::now() + limit,
27            expired: false,
28        }
29    }
30}

36/// Every piece of state `Op::Clause`'s region sets fresh, unconditionally, on
37/// **every** instruction it steps -- and so every field a caller pushing a
38/// nested activation (`Interp::invoke_call`, `run/call.rs`) must save before the
39/// callee runs and restore after it returns, because the callee's own
40/// `Op::Clause`'s region calls overwrite these exactly as the caller's own
41/// next clause would.
42pub(crate) struct ClauseState {
43    /// The indent (`static_indent`, spaces already doubled) an intermediate
44    /// value line traces at right now -- the one piece of state `eval`'s own
45    /// single insertion point needs that `eval`'s signature does not otherwise
46    /// carry, a persistent field every value or variable trace call reads,
47    /// never threaded as a parameter through evaluation. Set once per traced
48    /// clause, at whichever call site already computed that clause's own indent
49    /// for the `*-*` echo or a `>K>` line -- `run.rs`'s own doc comments name
50    /// each site.
51    pub(crate) current_value_indent: usize,
52    /// Whether the clause now running had instruction tracing in force **when
53    /// it began**, which is the only thing an assignment's `>>>` echo is
54    /// allowed to consult.
55    pub(crate) instructions_traced_at_entry: bool,
56    /// Instance SetTrace enabled result tracing while this clause was open.
57    /// A CALL already in progress does not gain a result echo.
58    pub(crate) instance_trace_started: bool,
59    /// The line the clause currently being stepped starts at -- **`SIGL`'s** own value, one
60    /// control transfer away from being read, and the exact analogue of
61    /// `current_value_indent` just above: `Interp::invoke_call` (`CALL`, and
62    /// `NodeKind::Call`'s expression form through `eval_call`, `eval.rs`) and `SIGNAL`'s
63    /// own two `step` arms all need "which line is this transfer's own", and `eval_call`
64    /// reaches `invoke_call` from arbitrarily deep inside an expression tree with no
65    /// `source`/ `instruction` of its own to compute it from -- threading either through
66    /// `eval`/`eval_node`'s entire recursive call graph is exactly the "every arm in
67    /// `eval.rs`" retrofit `current_value_indent`'s own doc comment already declined for
68    /// the identical reason. Set unconditionally by `Op::Clause`'s region, via
69    /// `clause_line`, which honours `clause_line_override` the same way `clause_site` does
70    /// -- so a `SIGNAL`/`CALL` fired from inside an `INTERPRET` fragment reads the
71    /// *enclosing* `INTERPRET` clause's own line, as an interpret-created activation
72    /// delegates a `SIGNAL` to its parent rather than setting `SIGL` itself, so what ends
73    /// up in `SIGL` is the parent's own currently-executing instruction -- the `INTERPRET`
74    /// clause -- and this field reproduces that observable answer (`run_fragment` runs
75    /// inside the creating activation, not a nested one of its own).
76    current_clause_line: usize,
77    /// The index, in the running activation's own body, of the clause being
78    /// stepped -- the one piece `StackFrame~traceLine` needs that the two
79    /// fields above do not carry, since a clause's *text* is its
80    /// instruction's `clause_span` and nothing else records which
81    /// instruction is running (`Activation::pc` stands on the enclosing
82    /// construct, per [`crate::activation::ClauseSnapshot`]).
83    pub(crate) current_clause_index: usize,
84}
85

87    /// The current clause's line -- `SIGL`'s own value.
88    pub(crate) fn line(&self) -> usize {
89        self.current_clause_line
90    }
91
92    /// The clause index [`ClauseState::current_clause_index`] holds.
93    pub(crate) fn clause_index(&self) -> usize {
94        self.current_clause_index
95    }

97    /// The zero state `Interp::new` starts from.
98    pub(crate) fn new() -> ClauseState {
99        ClauseState {
100            current_value_indent: 0,
101            current_clause_line: 0,
102            current_clause_index: 0,
103            // `false` is the state an interpreter with nothing running is in,

129/// What a clause resolved to, for the one thing a boundary needs from it:
130/// a value that has to stay rooted while a `CALL ON` handler runs.
131pub(crate) trait ClauseValue {
132    /// The value to root across a delivered handler, if any.
133    fn rooted(&self) -> Option<ObjRef>;
134}
135
136impl ClauseValue for Flow {
137    /// `flow` may carry an `ObjRef` whose one-clause temps frame is already
138    /// popped, and the handler is a nested activation that allocates, so the
139    /// value is rooted across it. Measured with a negative control: without
140    /// the `push_temp` in `in_clause` a `Flow::Return` and a `Flow::Exit`
141    /// program each panic on `a live value` under collect-on-every-
142    /// allocation.
143    fn rooted(&self) -> Option<ObjRef> {
144        match self {
145            Flow::Return(Some(value)) | Flow::Exit(Some(value)) => Some(*value),
146            _ => None,
147        }
148    }
149}
150
151impl ClauseValue for () {
152    /// A loop header, a `WHILE`/`UNTIL` re-test and a `SELECT CASE`
153    /// expression all produce no `Flow` at all, so there is nothing whose
154    /// only root was this clause's frame.
155    fn rooted(&self) -> Option<ObjRef> {
156        None
157    }
158}
159
160impl ClauseValue for bool {
161    /// An `IF`'s or a `WHEN`'s condition: the answer is a Rexx logical
162    /// value, already consumed into a `bool` by `eval_condition`, so no
163    /// `ObjRef` escapes this clause.
164    fn rooted(&self) -> Option<ObjRef> {
165        None
166    }
167}

169/// A clause boundary that is open: [`Interp::enter_clause`] makes one and
170/// [`Interp::leave_clause`] spends it.
171#[must_use]
172pub(crate) struct ClauseEntry(());
173
174/// How [`Interp::in_clause`] finished.
175pub(crate) enum ClauseOutcome<T> {
176    /// The clause ran, successfully or not, and here is what it produced.
177    Ran(Result<T, Failure>),
178    /// A `CALL ON` handler ran at this clause's boundary and ended the whole
179    /// program.
180    Ended(HandlerExit),
181}
182
183/// A delivered `CALL ON` handler ended the program with `EXIT`.
184pub(crate) struct HandlerExit(Option<ObjRef>);
185
186impl HandlerExit {
187    /// The value the handler exited with.
188    pub(crate) fn value(self) -> Option<ObjRef> {
189        self.0
190    }
191
192    /// The only constructor, and it is the invariant rather than a wrapper
193    /// around it: an `Ended::Returned` answers `None`, because a handler that
194    /// *returned* resumes the interrupted clause and has not ended anything.
195    /// A caller that routes the wrong variant here therefore gets "carry on",
196    /// which is the correct behaviour, rather than a `RETURN` silently
197    /// rendered as an `EXIT` -- the shape re-review 3's NEW-6 named.
198    pub(crate) fn from_ended(ended: Ended) -> Option<HandlerExit> {
199        match ended {
200            Ended::Exited(value) => Some(HandlerExit(value)),
201            Ended::Returned(_) => None,
202        }
203    }
204}

207    /// Runs one Rexx clause: `line` is the clause's own line, `body` is the
208    /// whole of the clause.
209    #[inline(always)]
210    pub(crate) fn in_clause<T: ClauseValue>(
211        &mut self,
212        line: usize,
213        body: impl FnOnce(&mut Self) -> Result<T, Failure>,
214    ) -> Result<ClauseOutcome<T>, Failure> {
215        let counted = self.count_clause_against_deadline(false)?;
216        self.in_counted_clause(line, counted, body)
217    }
218
219    /// [`Interp::in_clause`] for a clause already `counted`.
220    #[inline(always)]
221    pub(crate) fn in_counted_clause<T: ClauseValue>(
222        &mut self,
223        line: usize,
224        counted: DeadlineCounted,
225        body: impl FnOnce(&mut Self) -> Result<T, Failure>,
226    ) -> Result<ClauseOutcome<T>, Failure> {
227        let entry = self.enter_clause(line, counted);
228        let ran = body(self);
229        self.leave_clause(entry, ran)
230    }

232    /// Opens the clause at `line`: everything [`Interp::in_clause`] does before
233    /// the clause's own work runs.
234    #[inline(always)]
235    pub(crate) fn enter_clause(&mut self, line: usize, counted: DeadlineCounted) -> ClauseEntry {
236        let DeadlineCounted(()) = counted;
237        // **The argument stack is empty at every clause boundary**, and this
238        // is what keeps that true when a clause is abandoned part-way through
239        // pushing a call's run: a trapped `NOVALUE` on a second argument
240        // leaves the first standing, and nothing later would take it. Every
241        // ordinary call removes its own run, so this stores a length that is
242        // already zero. Measured: 200,000 abandoned pushes grew the process
243        // by 3.3 MB without it.
244        self.activity.value_buffer.clear();
245        self.activity.clause_state.instance_trace_started = false;
246        // **Tripwire.** A condition queued by
247        // this activation's clause at line L that is still waiting when a
248        // clause at a *different* line begins means some construct resolved
249        // an instruction inside its own `step` without ending its header
250        // clause first. The handler would then report `SIGL` for the wrong clause.
251        // Keyed as `deliver_pending_traps` keys a delivery: a condition queued
252        // in an enclosing fragment's queue waits for that fragment's boundary.
253        debug_assert!(
254            self.activity.clause_state.current_clause_line == line
255                || !self.activity.pending_traps.iter().any(|pending| {
256                    !pending.queued_during_delivery
257                        && pending.activation == self.activation().id
258                        && pending.fragment_depth == self.activity.fragment_depth
259                }),
260            "a clause at line {} began while a condition queued by this activation's clause at \
261             line {} was still waiting: some construct ran an instruction inside its own step \
262             without ending its header clause first",
263            line,
264            self.activity.clause_state.current_clause_line
265        );
266        self.activity.clause_state.current_clause_line = line;
267        ClauseEntry(())
268    }

295    /// Counts one clause against this run's deadline, and answers the proof
296    /// [`Interp::enter_clause`] needs. Only a clause a root driver can leave
297    /// `yields` its activity's slice.
298    #[inline(always)]
299    pub(crate) fn count_clause_against_deadline(
300        &mut self,
301        yields: bool,
302    ) -> Result<DeadlineCounted, Failure> {
303        self.clause_countdown -= 1;
304        if self.clause_countdown == 0 {
305            self.countdown_reached(yields)?;
306        }
307        Ok(DeadlineCounted(()))
308    }

313    pub(crate) fn uncount_clause(&mut self, counted: DeadlineCounted) {
314        let DeadlineCounted(()) = counted;
315        self.clause_countdown += 1;

318    /// Reads the clock and either ends the run or reloads the countdown and
319    /// serves the requests pending.
320    #[cold]
321    #[inline(never)]
322    #[expect(clippy::disallowed_methods, reason = "the run's deadline")]
323    fn countdown_reached(&mut self, yields: bool) -> Result<(), Failure> {
324        #[cfg(test)]
325        COUNTDOWN_VISITS.with(|visits| visits.set(visits.get() + 1));
326        if let Some(deadline) = &mut self.deadline
327            && (deadline.expired || Instant::now() >= deadline.at)
328        {
329            // Reloaded to 1, so the very next clause lands here again and
330            // fails too. A run that has outlived its bound does not get to
331            // resume because a caller swallowed one failure.
332            deadline.expired = true;
333            self.clause_countdown = 1;
334            return Err(Failure::Deadline);
335        }
336        self.clause_countdown = Deadline::CLAUSES_PER_CHECK;
337        self.serve_requests(yields)
338    }

340    /// Stops the timer's slices while this thread idles, and has the next
341    /// clause decide again whether they are wanted.
342    ///
343    /// A visit of the clause countdown arms the slices while work is in
344    /// flight off the baton, so that an activity its end readies gets its
345    /// turn; only a switch with nothing in flight disarmed them. An idle
346    /// interpreter runs no activity a slice could end, yet stayed armed for
347    /// as long as anything was in flight, and the timer thread woke every
348    /// slice (24 ms) to set a request nobody read: with a hundred activities
349    /// parked in socket waits, about 42 wakes a second of an otherwise idle
350    /// process (#591). Fewer showed none only because their clauses did not
351    /// add up to a countdown visit while a wait was in flight.
352    fn disarm_while_idle(&mut self) {
353        self.timer.disarm();
354        self.clause_countdown = 1;
355    }

357    /// Blocks this thread until `due` or a post to the inbox, whose
358    /// completions it files, or fails where the run's deadline comes first.
359    #[expect(clippy::disallowed_methods, reason = "the run's deadline")]
360    pub(crate) fn idle_until(&mut self, due: Instant) -> Result<(), Failure> {
361        if self.sim.is_some() {
362            return self.sim_idle_until(due);
363        }
364        self.disarm_while_idle();
365        self.hand_over_output();
366        if let Some(deadline) = &self.deadline
367            && (deadline.expired || deadline.at <= due)
368        {
369            let at = deadline.at;
370            // A post ends the idle early; a wake with nothing posted may be
371            // one left from an earlier idle, so the clock decides.
372            while !self.deadline_expired() && Instant::now() < at {
373                let posted = self.timer.idle_until(at);
374                if !posted.is_empty() {
375                    self.file_completions(posted, crate::scheduler::DrainAt::Stopped);
376                    return Ok(());
377                }
378            }
379            return Err(self.expire_deadline());
380        }
381        let posted = self.timer.idle_until(due);
382        self.file_completions(posted, crate::scheduler::DrainAt::Stopped);
383        Ok(())
384    }

386    /// Blocks this thread until something is posted to the inbox, whose
387    /// completions it files, or fails where the run's deadline comes first.
388    pub(crate) fn idle_for_posts(&mut self) -> Result<(), Failure> {
389        if self.sim.is_some() {
390            if self.sim_native_pool() {
391                if self.deadline_passed() {
392                    return Err(self.expire_deadline());
393                }
394                if self.sim_deliver_native(None) {
395                    return self.sim_breached();
396                }
397            }
398            return Err(self.sim_unpostable_wait());
399        }
400        if let Some(deadline) = &self.deadline {
401            return self.idle_until(deadline.at);
402        }
403        self.disarm_while_idle();
404        self.hand_over_output();
405        let posted = self.timer.idle();
406        self.file_completions(posted, crate::scheduler::DrainAt::Stopped);
407        Ok(())
408    }

416            .is_some_and(|deadline| deadline.expired || Instant::now() >= deadline.at)
417    }

420    pub(crate) fn expire_deadline(&mut self) -> Failure {
421        if let Some(deadline) = &mut self.deadline {
422            deadline.expired = true;
423        }
424        self.clause_countdown = 1;
425        Failure::Deadline

435    /// Closes the clause `entry` opened, around `ran` -- everything
436    /// [`Interp::in_clause`] does once the clause's own work has run.
437    #[inline(always)]
438    pub(crate) fn leave_clause<T: ClauseValue>(
439        &mut self,
440        entry: ClauseEntry,
441        ran: Result<T, Failure>,
442    ) -> Result<ClauseOutcome<T>, Failure> {
443        let ClauseEntry(()) = entry;
444        self.clause_boundary(ran)
445    }

447    /// The boundary [`Interp::leave_clause`] runs, for a clause whose entry
448    /// is held by a region that closes without one: a loop's header test,
449    /// whose region closes as its `DO` instruction's does.
450    #[inline(always)]
451    pub(crate) fn clause_boundary<T: ClauseValue>(
452        &mut self,
453        ran: Result<T, Failure>,
454    ) -> Result<ClauseOutcome<T>, Failure> {
455        let Ok(value) = &ran else {
456            // A clause that is unwinding never reached a boundary, so it
457            // delivers nothing.
458            return Ok(ClauseOutcome::Ran(ran));
459        };
460        // The halt and trace exits, ahead of the conditions the boundary
461        // delivers; their failure is the clause's.
462        if self.clause_exits {
463            if let Some(value) = value.rooted() {
464                self.roots.activity_mut().push_temp(value);
465            }
466            if let Err(failure) = self.run_clause_exits() {
467                return Ok(ClauseOutcome::Ran(Err(failure)));
468            }
469            // The heap passed its limit at the last collection
470            // (r[env.invocation.heap-limit.raise]): raised as a halt is, at the
471            // end of the clause of a user program that was running.
472            if self.heap_over_limit && self.running_user_program() {
473                self.heap_limit_raised();
474                return Err(crate::error::Raised::system_resources().into());
475            }
476        }
477        if self.activity.pending_traps.is_empty() {
478            return Ok(ClauseOutcome::Ran(ran));
479        }
480        if let Some(value) = value.rooted() {
481            self.roots.activity_mut().push_temp(value);
482        }
483        match self.deliver_pending_traps()? {
484            Some(exit) => Ok(ClauseOutcome::Ended(exit)),
485            None => Ok(ClauseOutcome::Ran(ran)),
486        }
487    }

512    /// Makes the clause on `line` the running one for an activation with no

543    /// An instance SetTrace request can be delivered while a native CALL is
