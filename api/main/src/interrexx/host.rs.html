1//! What differs from one host system to the next. Each concern is one
2//! function with an implementation per host, chosen by `cfg`, so a port adds
3//! an implementation beside the others and no caller changes. The raw calls
4//! some hosts need, where no safe crate offers them, are in [`sys`].
5//!
6//! The hosts are Linux and Apple's systems. On Linux the `cfg` arms say
7//! `linux` or `android`, and the other arm of a pair is Apple's; a new host
8//! adds its own arm to each pair and its name to the check below.
9
10#[cfg(not(any(target_os = "linux", target_os = "android", target_vendor = "apple")))]
11compile_error!(
12    "interrexx builds for Linux and Apple's systems only; a new host adds its implementations to \
13     host.rs"
14);

30/// The flags of a send on `socket` that keep it from raising `SIGPIPE`. No
31/// send can say so here, so the socket is set to answer `EPIPE` before every
32/// send, since it may be one the interpreter did not make and an embedding
33/// program may have restored `SIGPIPE`. Apple's kernel refuses the option on
34/// a socket whose connection has ended, with `EINVAL`; the send then goes
35/// ahead, so that it answers the host's own failure. A socket the interpreter
36/// made had the option set when it was made ([`prepare_socket`]), so only a
37/// socket from elsewhere, already ended, in a process that does not ignore
38/// `SIGPIPE`, can still raise it.
39#[cfg(not(any(target_os = "linux", target_os = "android")))]
40pub(crate) fn no_sigpipe_flags(socket: BorrowedFd<'_>) -> SendFlags {
41    let _ = rustix::net::sockopt::set_socket_nosigpipe(socket, true);
42    SendFlags::empty()
43}

101/// The process's descriptor table as `select` sees it, taken once for a
102/// call: `select` looks at an open descriptor always, and at a closed one
103/// where it lies inside the table, which makes `select` answer `EBADF`. One
104/// past the table, and a negative number, is never ready and no error.
105pub(crate) struct SelectTable {
106    /// The table's size, `FDSize` in `/proc/self/status`.
107    #[cfg(any(target_os = "linux", target_os = "android"))]
108    size: i32,
109}
110
111#[cfg(any(target_os = "linux", target_os = "android"))]
112impl SelectTable {
113    /// The table as it is now.
114    pub(crate) fn now() -> SelectTable {
115        let size = std::fs::read_to_string("/proc/self/status")
116            .ok()
117            .and_then(|status| {
118                status
119                    .lines()
120                    .find_map(|line| line.strip_prefix("FDSize:"))
121                    .and_then(|size| size.trim().parse().ok())
122            })
123            .unwrap_or(i32::MAX);
124        SelectTable { size }
125    }
126
127    /// Whether `select` looks at the descriptor numbered `fd`.
128    pub(crate) fn reaches(&self, fd: i32) -> bool {
129        (0..self.size).contains(&fd)
130    }
131}
132
133/// Apple's kernel does not say how large its table has grown, so a closed
134/// descriptor is put to `select` itself; one at or above `FD_SETSIZE`, which
135/// that `select` refuses with `EINVAL`, is taken to be past the table.
136#[cfg(target_vendor = "apple")]
137impl SelectTable {
138    /// The table as it is now.
139    pub(crate) fn now() -> SelectTable {
140        SelectTable {}
141    }
142
143    /// Whether `select` looks at the descriptor numbered `fd`.
144    #[expect(
145        clippy::unused_self,
146        reason = "the table is read when it is asked, not when it is taken"
147    )]
148    pub(crate) fn reaches(&self, fd: i32) -> bool {
149        // `select` first: a number closed when it asks and opened again before
150        // the second question is then open, and reached, either way. Only a
151        // close racing the call itself can differ from `select`, as it can in C.
152        fd >= 0 && (sys::select_rejects(fd) || sys::is_open(fd))
153    }
154}

156/// How many bytes a read of `file` finds at once, where the host counts them
157/// in a way to trust; `None` where it does not, and the caller then asks a
158/// `poll`.
159#[cfg(any(target_os = "linux", target_os = "android"))]
160pub(crate) fn bytes_waiting(file: BorrowedFd<'_>) -> Option<u64> {
161    rustix::io::ioctl_fionread(file).ok()
162}
163
164/// How many bytes a read of `file` finds at once, where the host counts them
165/// in a way to trust; `None` where it does not, and the caller then asks a
166/// `poll`. Apple's `FIONREAD` on a FIFO open for both reading and writing
167/// counts 0 with bytes waiting, so a 0 there is not trusted.
168#[cfg(not(any(target_os = "linux", target_os = "android")))]
169pub(crate) fn bytes_waiting(file: BorrowedFd<'_>) -> Option<u64> {
170    let count = rustix::io::ioctl_fionread(file).ok()?;
171    let fifo = || {
172        rustix::fs::fstat(file).is_ok_and(|stat| {
173            rustix::fs::FileType::from_raw_mode(stat.st_mode) == rustix::fs::FileType::Fifo
174        })
175    };
176    (count > 0 || !fifo()).then_some(count)
177}

258/// A descriptor that becomes readable once `child` has ended, and stays
259/// readable; it reaps nothing. `None` where the host cannot say, and the
260/// caller then waits for the child's pipes to close instead.
261#[cfg(any(target_os = "linux", target_os = "android"))]
262pub(crate) fn exit_watch(child: &Child) -> Option<OwnedFd> {
263    let pid = rustix::process::Pid::from_raw(i32::try_from(child.id()).ok()?)?;
264    rustix::process::pidfd_open(pid, rustix::process::PidfdFlags::empty()).ok()
265}

267/// A descriptor that becomes readable once `child` has ended, and stays
268/// readable; it reaps nothing. A `kqueue` holding the child's exit event,
269/// which stays pending since nothing collects it. A child that has already
270/// ended cannot be registered, and is answered by a pipe that holds a byte;
271/// so is one that the kernel registered while it was ending, which may never
272/// post the event, and which `waitid` without reaping finds ended once the
273/// registration is in place. Where `waitid` cannot say (a process that lets
274/// its children be reaped unasked), the registration stands.
275#[cfg(target_vendor = "apple")]
276pub(crate) fn exit_watch(child: &Child) -> Option<OwnedFd> {
277    use nix::errno::Errno;
278    use nix::sys::event::{EvFlags, EventFilter, FilterFlag, KEvent, Kqueue};
279    let queue = creating(|| {
280        let queue = Kqueue::new().ok()?;
281        rustix::io::fcntl_setfd(&queue, rustix::io::FdFlags::CLOEXEC).ok()?;
282        Some(queue)
283    })?;
284    let exit = KEvent::new(
285        usize::try_from(child.id()).ok()?,
286        EventFilter::EVFILT_PROC,
287        EvFlags::EV_ADD,
288        FilterFlag::NOTE_EXIT,
289        0,
290        0,
291    );
292    let ended = || {
293        let pid = rustix::process::Pid::from_raw(i32::try_from(child.id()).ok()?)?;
294        let options = rustix::process::WaitIdOptions::EXITED
295            | rustix::process::WaitIdOptions::NOHANG
296            | rustix::process::WaitIdOptions::NOWAIT;
297        Some(
298            rustix::process::waitid(rustix::process::WaitId::Pid(pid), options)
299                .ok()?
300                .is_some(),
301        )
302    };
303    match queue.kevent(&[exit], &mut [], None) {
304        Ok(_) if ended() != Some(true) => Some(queue.into()),
305        Ok(_) | Err(Errno::ESRCH) => {
306            let (read, write) = pipe().ok()?;
307            rustix::io::write(&write, &[1]).ok()?;
308            Some(read.into())
309        }
310        Err(_) => None,
311    }
312}

320    /// A send on a socket the interpreter did not make is kept from raising
321    /// `SIGPIPE` all the same: by the send's own flag on Linux, and by the
322    /// option the send path sets on the socket elsewhere.
323    #[test]
324    fn a_send_on_any_socket_is_kept_from_raising_sigpipe() {
325        let (socket, _peer) = std::os::unix::net::UnixStream::pair().expect("a socket pair");
326        let flags = no_sigpipe_flags(socket.as_fd());
327        #[cfg(any(target_os = "linux", target_os = "android"))]
328        assert_eq!(flags, rustix::net::SendFlags::NOSIGNAL);
329        #[cfg(not(any(target_os = "linux", target_os = "android")))]
330        {
331            assert_eq!(flags, rustix::net::SendFlags::empty());
332            assert_eq!(
333                rustix::net::sockopt::socket_nosigpipe(&socket),
334                Ok(true),
335                "SO_NOSIGPIPE"
336            );
337        }
338    }

340    /// A child that has ended but is not reaped is watched by a descriptor
341    /// that is readable at once, and the watch leaves it to be reaped.
342    #[test]
343    fn an_ended_child_is_watched_as_ended() {
344        use rustix::process::{Pid, WaitId, WaitIdOptions};
345        let mut child = std::process::Command::new("true")
346            .spawn()
347            .expect("true runs");
348        let pid = Pid::from_raw(i32::try_from(child.id()).expect("a pid")).expect("a pid");
349        let not_reaping = WaitIdOptions::EXITED | WaitIdOptions::NOWAIT;
350        rustix::process::waitid(WaitId::Pid(pid), not_reaping).expect("the child ends");
351        let watch = super::exit_watch(&child).expect("a watch");
352        let mut fds = [rustix::event::PollFd::new(
353            &watch,
354            rustix::event::PollFlags::IN,
355        )];
356        let zero = rustix::event::Timespec {
357            tv_sec: 0,
358            tv_nsec: 0,
359        };
360        assert_eq!(rustix::event::poll(&mut fds, Some(&zero)), Ok(1));
361        assert!(child.wait().expect("the child is reaped").success());
362    }

364    /// A descriptor just closed lies inside the table, which grew to hold it,
365    /// so `select` looks at it (and answers `EBADF`); a negative one never.
366    #[test]
367    fn select_looks_at_a_descriptor_just_closed() {
368        let (read, write) = rustix::pipe::pipe().expect("a pipe");
369        let highest = read.as_raw_fd().max(write.as_raw_fd());
370        let table = SelectTable::now();
371        assert!(table.reaches(read.as_raw_fd()));
372        drop((read, write));
373        let table = SelectTable::now();
374        assert!(table.reaches(highest), "closed descriptor {highest}");
375        assert!(!table.reaches(-1));
376    }

378    /// A count of 0 is trusted where the host's count is, so a terminal's end
379    /// of file is never read to learn there is no data: on Linux for every
380    /// device, and on Apple's systems for all but a FIFO, which a pipe is
381    /// there. A count above 0 is trusted everywhere.
382    #[test]
383    fn an_empty_count_is_trusted_but_for_an_apple_fifo() {
384        let (read, write) = rustix::pipe::pipe().expect("a pipe");
385        let empty_pipe = if cfg!(any(target_os = "linux", target_os = "android")) {
386            Some(0)
387        } else {
388            None
389        };
390        assert_eq!(bytes_waiting(read.as_fd()), empty_pipe);
391        rustix::io::write(&write, b"abc").expect("a write");
392        assert_eq!(bytes_waiting(read.as_fd()), Some(3));
393
394        let (socket, _peer) = std::os::unix::net::UnixStream::pair().expect("a socket pair");
395        assert_eq!(bytes_waiting(socket.as_fd()), Some(0));
396    }

398    /// Apple's `FIONREAD` on a FIFO open for reading and writing counts 0
399    /// with bytes waiting, so an empty count there is not trusted.
400    #[cfg(target_vendor = "apple")]
401    #[test]
402    fn a_fifo_count_of_zero_is_not_trusted_on_apple() {
403        use std::hash::BuildHasher as _;
404        let unique = std::hash::RandomState::new().hash_one(std::process::id());
405        let path = std::env::temp_dir().join(format!("host-fifo-{unique:016x}"));
406        // Neither `rustix` nor this crate's `nix` features make a FIFO on Apple's systems.
407        let made = std::process::Command::new("mkfifo")
408            .arg(&path)
409            .status()
410            .expect("mkfifo runs");
411        assert!(made.success(), "a FIFO");
412        let fifo = rustix::fs::open(&path, rustix::fs::OFlags::RDWR, rustix::fs::Mode::empty())
413            .expect("the FIFO, open both ways");
414        let _ = std::fs::remove_file(&path);
415        assert_eq!(bytes_waiting(fifo.as_fd()), None);
416    }
