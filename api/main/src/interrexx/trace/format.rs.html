6/// The visible-output shape of the current `TRACE` setting, and whether
7/// interactive debug is on. There is no field for separate per-kind pause
8/// flags: a pause follows every clause the letter in force traced, so
9/// [`TraceMode::debug`] and the letter decide it together.
10#[derive(Copy, Clone, PartialEq, Eq, Debug)]
11pub(crate) struct TraceMode {
12    /// `TRACE_PREFIX_CLAUSE` (`*-*`): every stepped instruction's own clause
13    /// is echoed.
14    pub(crate) all: bool,
15    /// `TRACE_PREFIX_RESULT`/`_KEYWORD` (`>>>`/`>K>`): a traced instruction's
16    /// own top-level computed value is shown once.
17    pub(crate) results: bool,
18    /// Every other value prefix (`>L>`/`>V>`/`>O>`/`>P>`/`>C>`) plus
19    /// `>=>` (`TRACE_PREFIX_ASSIGNMENT`): every step of *evaluating* a
20    /// traced instruction's expression is shown, not only its final value.
21    pub(crate) intermediates: bool,
22    /// A `LABEL` clause is echoed, with the ordinary `*-*` prefix. A `LABEL`
23    /// clause traces through the label path and nothing else, and that path's
24    /// gate is the labels flag -- which the `A`, `R` and `I` settings set as
25    /// well as `L`. So this is `true` under `A`/`R`/`I` too, where `all`
26    /// already covers it, and the only mode it decides anything in is `L`.
27    pub(crate) labels: bool,
28    /// A command clause echoes `*-*` and its evaluated string as `>>>`,
29    /// before the command runs. Measured under `TRACE C`, where a command
30    /// that succeeds still shows both and nothing else in the program shows
31    /// anything.
32    pub(crate) commands: bool,
33    /// A command that raised `ERROR` echoes `*-*` and `>>>` **after** it has
34    /// run, where the setting did not already echo them.
35    pub(crate) errors: bool,
36    /// The same for `FAILURE`. Set under `TRACE N`, which is why a failing
37    /// command traces itself in a program carrying no `TRACE` instruction at
38    /// all.
39    pub(crate) failures: bool,
40    /// The byte this setting renders as, and so the byte `TRACE()` answers.
41    pub(crate) letter: u8,
42    /// The `?` prefix: every traced clause is followed by a pause. `TRACE()`
43    /// answers it as a `?` before the letter.
44    pub(crate) debug: bool,
45}
46
47impl TraceMode {
48    /// Every flag clear, and the one setting under which a failing command
49    /// shows nothing.
50    pub(crate) const OFF: TraceMode = TraceMode {
51        all: false,
52        results: false,
53        intermediates: false,
54        labels: false,
55        commands: false,
56        errors: false,
57        failures: false,
58        letter: b'O',
59        debug: false,
60    };
61    /// `TRACE N`, and the setting every activation starts under. The default is
62    /// normal tracing **and failures**, so this is not [`TraceMode::OFF`] with
63    /// another letter: a command that raises `FAILURE` traces itself here.
64    pub(crate) const NORMAL: TraceMode = TraceMode {
65        failures: true,
66        letter: b'N',
67        ..TraceMode::OFF
68    };
69    /// `TRACE C`: the single instruction type, so a command clause echoes and
70    /// nothing else in the program does.
71    const COMMANDS: TraceMode = TraceMode {
72        commands: true,
73        letter: b'C',
74        ..TraceMode::OFF
75    };
76    /// `TRACE E`, which sets the failures flag beside the errors flag --
77    /// "just errors includes failures".
78    const ERRORS: TraceMode = TraceMode {
79        errors: true,
80        failures: true,
81        letter: b'E',
82        ..TraceMode::OFF
83    };
84    /// `TRACE F`: failures and, unlike [`TraceMode::ERRORS`], not errors.
85    /// Measured -- `exit 127` traces and `exit 3` does not.
86    const FAILURES: TraceMode = TraceMode {
87        failures: true,
88        letter: b'F',
89        ..TraceMode::OFF
90    };
91    /// `TRACE L` (which resets every other flag and sets the labels flag
92    /// alone). The one mode where `labels` decides anything: every executed
93    /// `LABEL` clause echoes and nothing else does.
94    /// `tests/trace_oracle/trace_labels.rex` carries the probed routes and
95    /// says why they are examples rather than an enumeration.
96    pub(super) const LABELS: TraceMode = TraceMode {
97        all: false,
98        results: false,
99        intermediates: false,
100        labels: true,
101        commands: false,
102        errors: false,
103        failures: false,
104        letter: b'L',
105        debug: false,
106    };
107    /// `TRACE A`: every clause echoes, but the results flag is deliberately
108    /// omitted, because `mode_from_setting` needs a real answer for the letter,
109    /// not a silent fallback that would misreport `TRACE A x = 1` as producing
110    /// no `*-*` line at all.
111    pub(super) const ALL: TraceMode = TraceMode {
112        all: true,
113        results: false,
114        intermediates: false,
115        labels: true,
116        // The `A` setting carries the commands flag and neither of the other two, so
117        // a command's `>>>` here comes from this flag rather than from `results`.
118        commands: true,
119        errors: false,
120        failures: false,
121        letter: b'A',
122        debug: false,
123    };
124    /// `TRACE R`. Also the half of the external-trace setting that is not the debug
125    /// flag, which is what `RXTRACE=ON` starts a program under.
126    pub(crate) const RESULTS: TraceMode = TraceMode {
127        all: true,
128        results: true,
129        intermediates: false,
130        labels: true,
131        commands: true,
132        errors: false,
133        failures: false,
134        letter: b'R',
135        debug: false,
136    };
137    /// `TRACE I`.
138    pub(crate) const INTERMEDIATES: TraceMode = TraceMode {
139        all: true,
140        results: true,
141        intermediates: true,
142        labels: true,
143        commands: true,
144        errors: false,
145        failures: false,
146        letter: b'I',
147        debug: false,
148    };
149
150    /// The one setting that refuses the debug flag. The letter is the flag:
151    /// the off setting clears every other one.
152    pub(crate) fn is_off(self) -> bool {
153        self.letter == b'O'
154    }

157/// Everything `crate::ir::compile` is allowed to read out of a [`TraceMode`],
158/// and so everything a compiled chunk's identity depends on.
159#[derive(Copy, Clone, PartialEq, Eq, Hash, Debug)]
160pub(crate) struct ChunkTrace(u8);
161
162impl ChunkTrace {
163    /// [`TraceMode::all`]: every stepped clause echoes.
164    const CLAUSES: u8 = 1;
165    /// [`TraceMode::labels`]: a `LABEL` clause echoes.
166    const LABELS: u8 = 2;
167    /// [`TraceMode::intermediates`]: a literal, a read, an operator's result,
168    /// an argument and a call's result each echo a line of their own.
169    const INTERMEDIATES: u8 = 4;
170
171    /// [`TraceMode::results`]: a traced instruction's own top-level computed
172    /// value echoes, as `>>>` or -- for a loop header -- as `>K>`.
173    const RESULTS: u8 = 8;
174
175    /// [`TraceMode::commands`]: a command clause echoes. Part of a chunk's
176    /// identity because it decides whether the chunk carries the echo op at
177    /// all, which is the same reason [`ChunkTrace::CLAUSES`] is.
178    const COMMANDS: u8 = 16;
179
180    /// [`TraceMode::debug`]: every traced clause is followed by a pause.
181    /// Part of a chunk's identity so that entering debug makes every chunk
182    /// stale, which routes the clause echo back through the run-time gate --
183    /// the pause rides that decision instead of costing the compiled path a
184    /// branch of its own.
185    const DEBUG: u8 = 32;
186
187    /// What `compile` reads out of the setting in force.
188    #[inline(always)]
189    pub(crate) fn of(mode: TraceMode) -> ChunkTrace {
190        // **One byte and not three `bool` fields, which is a measurement.**
191        // `Interp::run_ops`'s staleness check compares a whole `ChunkTrace`
192        // once per promoted clause, so the type's width is on that path: as
193        // three fields the comparison cost `bench-programs/emptyloop.rex`
194        // 0.723% and the pinned `rexxcps` 0.372% against two, where packed it
195        // costs neither.
196        ChunkTrace(
197            (u8::from(mode.all) * ChunkTrace::CLAUSES)
198                | (u8::from(mode.debug) * ChunkTrace::DEBUG)
199                | (u8::from(mode.labels) * ChunkTrace::LABELS)
200                | (u8::from(mode.intermediates) * ChunkTrace::INTERMEDIATES)
201                | (u8::from(mode.results) * ChunkTrace::RESULTS)
202                | (u8::from(mode.commands) * ChunkTrace::COMMANDS),
203        )
204    }
205
206    /// Whether interactive debug is on, which is what puts a line typed at a
207    /// pause -- and so a setting no reading of the source fixes -- in reach.
208    #[inline(always)]
209    pub(crate) fn debugging(self) -> bool {
210        self.0 & ChunkTrace::DEBUG != 0
211    }
212
213    /// Whether a `*-*` line prints for a clause of this kind.
214    #[inline(always)]
215    pub(crate) fn intermediates(self) -> bool {
216        self.0 & ChunkTrace::INTERMEDIATES != 0
217    }
218
219    /// Whether a top-level computed value echoes -- `>>>` and the loop
220    /// header's `>K>` -- which is what decides whether a chunk carries the op
221    /// that emits the latter.
222    #[inline(always)]
223    pub(crate) fn results(self) -> bool {
224        self.0 & ChunkTrace::RESULTS != 0
225    }
226
227    /// This setting with [`ChunkTrace::INTERMEDIATES`] and
228    /// [`ChunkTrace::RESULTS`] masked off, which is what `Interp::run_ops`
229    /// compares its chunk against per promoted clause.
230    #[inline(always)]
231    pub(crate) fn clause_echoes(self) -> ChunkTrace {
232        ChunkTrace(self.0 & (ChunkTrace::CLAUSES | ChunkTrace::LABELS | ChunkTrace::COMMANDS))
233    }
234
235    #[inline(always)]
236    pub(crate) fn echoes(self, is_label: bool, is_command: bool) -> bool {
237        self.0 & ChunkTrace::CLAUSES != 0
238            || (self.0 & ChunkTrace::LABELS != 0 && is_label)
239            || (self.0 & ChunkTrace::COMMANDS != 0 && is_command)
240    }
241
242    /// [`ChunkTrace::of`] applied to [`TraceMode::OFF`]: no bit set.
243    pub(crate) const OFF: ChunkTrace = ChunkTrace(0);
244}

246/// A [`TraceMode`], whether a line typed at a debug pause runs, and the
247/// [`ChunkTrace`] the trace sink obeys under them -- the mode's packing, or
248/// [`ChunkTrace::OFF`] while the pause runs -- so that all are paid where the
249/// setting or the pause changes rather than at every clause that reads them.
250///
251/// **The byte cannot drift from the mode**: [`TraceCache::of`] is the only way
252/// to build one, and it derives the byte.
253#[derive(Copy, Clone, Debug)]
254pub(crate) struct TraceCache {
255    mode: TraceMode,
256    chunk: ChunkTrace,
257    paused: bool,
258}
259
260impl TraceCache {
261    #[inline(always)]
262    pub(crate) fn of(mode: TraceMode, paused: bool) -> TraceCache {
263        TraceCache {
264            mode,
265            chunk: if paused {
266                ChunkTrace::OFF
267            } else {
268                ChunkTrace::of(mode)
269            },
270            paused,
271        }
272    }
273
274    /// The setting the trace sink obeys: [`TraceMode::OFF`] while a line
275    /// typed at a debug pause runs, else [`TraceCache::mode`].
276    #[inline(always)]
277    pub(crate) fn traced(self) -> TraceMode {
278        if self.paused {
279            TraceMode::OFF
280        } else {
281            self.mode
282        }
283    }
284
285    #[inline(always)]
286    pub(crate) fn mode(self) -> TraceMode {
287        self.mode
288    }
289
290    #[inline(always)]
291    pub(crate) fn chunk(self) -> ChunkTrace {
292        self.chunk
293    }
294}

296/// What one instruction does to the `TRACE` setting in force -- the
297/// optimizing function of `crate::ir::trace_flow`'s forward analysis, one
298/// entry per instruction in [`crate::plan::Plan::trace_events`].
299#[derive(Copy, Clone, PartialEq, Eq, Debug)]
300pub(crate) enum TraceEvent {
301    /// The setting leaves this instruction as it arrived.
302    Keeps,
303    /// The setting this instruction puts in force, fixed by its own text.
304    Sets(ChunkTrace),
305    /// The instruction can put a setting in force that no reading of the
306    /// source fixes.
307    Unknown,
308}
309
310/// What a `TRACE` instruction whose setting is a literal puts in force,
311/// following `Interp::exec_trace`'s own arms.
312pub(crate) fn literal_trace_event(setting: &Trace) -> TraceEvent {
313    let mode = match setting {
314        Trace::Default => TraceMode::NORMAL,
315        Trace::Setting(bytes) => {
316            let Ok(request) = parse_trace_request(bytes) else {
317                return TraceEvent::Unknown;
318            };
319            // A setting made only of `?`s toggles the flag on whatever is
320            // already in force, so its answer is the incoming pool rather
321            // than this instruction's text.
322            if request.setting.is_none() {
323                return TraceEvent::Unknown;
324            }
325            applied(TraceMode::OFF, &request)
326        }
327        // `TRACE n` raises 24.901 outside a pause, and `TRACE VALUE`
328        // computes its setting at run time.
329        Trace::Skip(_) | Trace::Value(_) => return TraceEvent::Unknown,
330    };
331    // A line typed at an interactive-debug pause changes the setting from
332    // outside the body, which no analysis of the body can see.
333    if mode.debug {
334        return TraceEvent::Unknown;
335    }
336    TraceEvent::Sets(ChunkTrace::of(mode))
337}

339/// What one `TRACE` setting asks for: any number of `?`s and then one letter,
340/// case-insensitively, ignoring everything after it.
341pub(crate) struct TraceRequest {
342    /// The letter's own mode, or `None` for a setting that named no letter.
343    pub(crate) setting: Option<TraceMode>,
344    /// How many `?`s it carried. An even number asks for no change.
345    pub(crate) toggles: usize,
346}
347
348/// [`TraceRequest`] for one setting, or the byte that is not a letter.
349pub(crate) fn parse_trace_request(bytes: &[u8]) -> Result<TraceRequest, u8> {
350    let toggles = bytes.iter().take_while(|byte| **byte == b'?').count();
351    let letter = bytes.get(toggles);
352    let setting = match letter {
353        None => None,
354        Some(byte) => Some(mode_of_letter(*byte)?),
355    };
356    Ok(TraceRequest { setting, toggles })
357}
358
359/// The setting in force after `request` is applied to `current`.
360///
361/// **A setting naming a letter replaces the debug flag; one made only of
362/// `?`s toggles it and keeps the letter.** Measured: `trace ?r` traces
363/// results *and* pauses, `trace ?` from a pause ends debug and leaves `R` in
364/// force, and a bare `trace` ends debug and sets Normal.
365///
366/// `OFF` is the exception in both halves: the debug flag is skipped when the
367/// letter is `O`, and a bare toggle only while the setting in force is not
368/// `OFF`. So no spelling of `TRACE` reaches interactive debug from `OFF`, which
369/// is what `base/keyword`'s `TRACE::test_trace_?o` asserts.
370pub(crate) fn applied(current: TraceMode, request: &TraceRequest) -> TraceMode {
371    match (request.setting, request.toggles) {
372        (Some(mode), toggles) => TraceMode {
373            debug: toggles % 2 == 1 && !mode.is_off(),
374            ..mode
375        },
376        (None, 0) => TraceMode::NORMAL,
377        (None, toggles) => TraceMode {
378            debug: if toggles % 2 == 1 && !current.is_off() {
379                !current.debug
380            } else {
381                current.debug
382            },
383            ..current
384        },
385    }
386}

388/// The answer for a caller with no setting in force to merge against: the
389/// letter alone, with `?` recorded as the debug flag. An empty string is the
390/// silent normal setting, and one made only of `?`s toggles the debug flag on
391/// the default `NORMAL` setting, as `trace ?` does at the start of a program
392/// (`r[dir.options.deviation.103]`).
393pub(crate) fn mode_from_setting(bytes: &[u8]) -> Result<TraceMode, u8> {
394    let toggles = bytes.iter().take_while(|byte| **byte == b'?').count();
395    let answer = |mode: TraceMode| {
396        Ok(TraceMode {
397            debug: toggles % 2 == 1,
398            ..mode
399        })
400    };
401    for &byte in bytes {
402        if byte == b'?' {
403            continue;
404        }
405        let mode = mode_of_letter(byte)?;
406        // Setting `OFF` is unconditional, so `::options trace ?o` is `OFF`
407        // and not a paused `OFF`, exactly as the instruction is. The all-`?`
408        // fallback below is not covered by this: what it stands in for is a
409        // bare debug toggle, not setting `OFF`.
410        if mode.is_off() {
411            return Ok(mode);
412        }
413        return answer(mode);
414    }
415    answer(TraceMode::NORMAL)
416}

418/// The mode one `TRACE` letter names, case-insensitively, or the byte itself
419/// when it names none.
420fn mode_of_letter(byte: u8) -> Result<TraceMode, u8> {
421    match byte.to_ascii_uppercase() {
422        b'A' => Ok(TraceMode::ALL),
423        b'R' => Ok(TraceMode::RESULTS),
424        b'I' => Ok(TraceMode::INTERMEDIATES),
425        // `L` echoes every executed `LABEL` clause -- see `TraceMode::labels`.
426        b'L' => Ok(TraceMode::LABELS),
427        // `C`/`E`/`F`/`N`/`O`: every letter `check_trace_setting` accepts is
428        // recognised here. Each of the four besides `O` decides whether a
429        // command clause echoes, so none of them is a setting this crate has
430        // nothing to show for.
431        b'C' => Ok(TraceMode::COMMANDS),
432        b'E' => Ok(TraceMode::ERRORS),
433        b'F' => Ok(TraceMode::FAILURES),
434        b'N' => Ok(TraceMode::NORMAL),
435        b'O' => Ok(TraceMode::OFF),
436        _ => Err(byte),
437    }
438}

440/// The same precision `interrexx-parse`'s own `TRACE_DIGITS` uses for the
441/// skip-count forms (`trace 5`, `trace -3`) -- duplicated as a constant
442/// rather than imported, because `interrexx-parse::instruction::TRACE_DIGITS` is
443/// private to that module and this crate cannot reach it. Not a second
444/// number-parsing *algorithm*: `whole_number` (`interrexx-parse::convert`) is
445/// `Number::parse(text)?.whole_value(digits)`, the same two `interrexx-num`
446/// calls below, so only the digits bound is restated, not the arithmetic.
447const TRACE_DIGITS: usize = 9;
448
449/// Whether `text` (a `TRACE VALUE` expression's rendered result) is a whole
450/// number under `TRACE_DIGITS` -- the same question `interrexx-parse`'s parser
451/// asks of `TRACE`'s own literal-number form, asked again here because
452/// `TRACE VALUE`'s text is not known until run time and never passes
453/// through the parser's own check at all. Measured: `trace value 5` raises
454/// 24.901 exactly like `trace 5` (this task's own report has the
455/// transcript), so a numeric-looking `VALUE` result is a skip count, not a
456/// setting string, even though nothing parsed it as one.
457pub(crate) fn is_whole_number(text: &[u8]) -> bool {
458    std::str::from_utf8(text)
459        .ok()
460        .and_then(Number::parse)
461        .and_then(|n| n.whole_value(TRACE_DIGITS))
462        .is_some()
463}

465/// 24.901, a numeric `TRACE` request outside interactive debugging --
466/// unconditional, regardless of the number (measured: `trace
467/// 0` raises it exactly like `trace 5`), because this runtime has no
468/// interactive debugging at all for a nonzero skip count to be valid *from*.
469/// No substitution value, matching the catalogue's `(24, 901)` entry.
470pub(crate) fn raised_numeric_trace_interactive_only() -> Raised {
471    Raised::syntax(24, 901, Vec::new())
472}

474/// 24.1, an invalid `TRACE` request letter, the insert the letter --
475/// `mode_from_setting`'s `Err` case, reachable only through `TRACE VALUE` (a

487/// Appends `indent` spaces to `out` -- the one place every formatter below
488/// applies the clause indent, so a caller never repeats
489/// `std::iter::repeat_n(b' ', indent)` at five call sites.
490fn push_indent(out: &mut Vec<u8>, indent: usize) {
491    out.extend(std::iter::repeat_n(b' ', indent));
492}

494/// Runs the display rule over the trace line that starts at `line_start`, which
495/// is the last thing each formatter below does.
496pub(super) fn make_displayable(out: &mut [u8], line_start: usize) {
497    crate::error::displayable(&mut out[line_start..]);
498}
499
500/// The widest indent a `*-*` clause echo ever prints, in spaces.
501pub(crate) const MAX_CLAUSE_INDENT: usize = 40;
502
503/// `TRACE_PREFIX_CLAUSE` (`*-*`): `line`'s own clause, `text`, indented by
504/// `indent` spaces (`static_indent`), clamped at [`MAX_CLAUSE_INDENT`].
505pub(crate) fn push_clause(out: &mut Vec<u8>, line: usize, indent: usize, text: &[u8]) {
506    let line_start = out.len();
507    out.extend_from_slice(format!("{line:>6} *-* ").as_bytes());
508    push_indent(out, indent.min(MAX_CLAUSE_INDENT));
509    out.extend_from_slice(text);
510    out.push(b'\n');
511    make_displayable(out, line_start);
512}

514/// `TRACE_PREFIX_RESULT` (`>>>`) or `TRACE_PREFIX_LITERAL`/`_VARIABLE`/
515/// `_PREFIX` when called with the matching `prefix` and no tag: an
516/// untagged, quoted value alone -- `push_prefixed_blanks` builds the header
517/// and its own fixed-plus-indent gap, this just quotes `value` after it.
518pub(crate) fn push_value(out: &mut Vec<u8>, prefix: &str, indent: usize, value: &[u8]) {
519    let line_start = out.len();
520    push_prefixed_blanks(out, prefix, indent);
521    push_quoted(out, value);
522    make_displayable(out, line_start);
523}
524
525/// The shared header every formatter in this module builds first: 7 blanks
526/// (the unused six-wide line-number field plus its trailing space,
527/// `PREFIX_OFFSET = 7`), then `prefix`, then a further blank run out to
528/// where the quoted value or tag starts.
529fn push_prefixed_blanks(out: &mut Vec<u8>, prefix: &str, indent: usize) {
530    debug_assert_eq!(prefix.len(), 3, "every trace prefix is exactly 3 bytes");
531    out.extend(std::iter::repeat_n(b' ', 7));
532    out.extend_from_slice(prefix.as_bytes());
533    push_indent(out, 3 + indent);
534}

536/// `"value"`, quoted -- the trailing half of every value-bearing trace line.
537fn push_quoted(out: &mut Vec<u8>, value: &[u8]) {
538    out.push(b'"');
539    out.extend_from_slice(value);
540    out.push(b'"');
541    out.push(b'\n');
542}

544/// `TRACE_PREFIX_VARIABLE`/`_ASSIGNMENT`/`_KEYWORD` (`>V>`/`>=>`/`>K>`), the
545/// `tag marker "value"` shape. `quote_tag` controls whether `tag` itself is
546/// wrapped in quotes (a keyword result quotes it, `>K> "TO" => "2"`; a variable
547/// read and an assignment do not, `>V> X => "2"` and `>=> X <= "2"`, no quotes
548/// around the variable name).
549pub(crate) fn push_tagged(
550    out: &mut Vec<u8>,
551    prefix: &str,
552    indent: usize,
553    quote_tag: bool,
554    tag: &[u8],
555    marker: &str,
556    value: &[u8],
557) {
558    let line_start = out.len();
559    push_prefixed_blanks(out, prefix, indent);
560    if quote_tag {
561        push_quoted_tag(out, tag);
562    } else {
563        out.extend_from_slice(tag);
564    }
565    out.extend_from_slice(marker.as_bytes());
566    push_quoted(out, value);
567    make_displayable(out, line_start);
568}

570/// `"tag"` with no trailing newline -- `push_tagged`'s own quoted-tag case,
571/// split out because `push_quoted` always terminates its own line and a tag
572/// is never the last thing on one.
573fn push_quoted_tag(out: &mut Vec<u8>, tag: &[u8]) {
574    out.push(b'"');
575    out.extend_from_slice(tag);
576    out.push(b'"');
577}

579/// `TRACE_PREFIX_OPERATOR`/`_PREFIX` (`>O>`/`>P>`): the shape `"op" =>
580/// "value"` -- always a quoted tag (measured: `>O> "+" => "2"`, unlike
581/// `>V>`/`>=>`), so this is `push_tagged` with `quote_tag` fixed to `true`
582/// and the arithmetic/comparison/logical operator's own spelling as the tag.
583pub(crate) fn push_operator(
584    out: &mut Vec<u8>,
585    prefix: &str,
586    indent: usize,
587    op: &[u8],
588    value: &[u8],
589) {
590    push_tagged(out, prefix, indent, true, op, " => ", value);
591}
