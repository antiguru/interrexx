1//! `.input`: one line position, shared by every construct that reads a line.
2//! ```text
3//! 1=<line-A> 2=<line-B> 3=<line-C> 4=<line-D>
4//! ```
5//! ```text
6//! 1=<qentry-one> 2=<line-A> 3=<qentry-two> 4=<line-B>
7//! ```
8//! ```text
9//! " pad \n"          ->  " pad "   (5)   leading and trailing blanks kept
10//! "\n"               ->  ""        (0)
11//! "last-no-newline"  ->  the whole 15 bytes, at end of file
12//! "crlf\r\n"         ->  "crlf"    (4)   the CR is not data
13//! "a\rb\n"           ->  "a\rb"    (3)   a CR elsewhere IS data
14//! "x\r\r\n"          ->  "x\r"     (2)   exactly one CR removed, not both
15//! "y\r"              ->  "y\r"     (2)   at end of file, no pair to collapse
16//! "a\x00b\n"         ->  "a\x00b"  (3)   NUL is data
17//! ```

28/// `.input`'s position: the one line cursor every input construct advances.
29pub(crate) struct Input {
30    source: Source,
31    /// Whether a read has already found the end of the input. **What parts a
32    /// live source from a drained one**, which is a distinction seekability
33    /// alone does not make: `.stdin~chars` over a pipe answers `1` before
34    /// anything is read and `0` once it is drained.
35    exhausted: bool,
36    /// Whether the line count has been asked for already. Measured on an
37    /// unmeasurable input: `.stdin~lines` answers `1` the first time and `0`
38    /// every time after, with no read in between, where `.stdin~chars` keeps
39    /// answering `1`. `LINES('N')` is that first ask too.
40    lines_asked: bool,
41}
42
43enum Source {
44    /// Nothing to read, ever. Distinct from `Bytes` over an empty buffer only
45    /// in costing no allocation; both answer `None` on the first read.
46    Nothing,
47    /// The process's standard input, read a chunk at a time off the baton,
48    /// where a halt can abandon the wait ([`Interp::fill_stdin`]).
49    Stdin(Stdin),
50    /// The process's standard input when it is a regular file.
51    File(FileConsole),
52    Bytes(Cursor<Vec<u8>>),
53}
54
55/// What standard input has delivered and not been read yet.
56#[derive(Default)]
57struct Stdin {
58    buffered: Vec<u8>,
59    /// A chunk read found the end of the input, or failed.
60    ended: bool,
61    /// A chunk read is in flight, perhaps one a halt abandoned.
62    reading: bool,
63}
64
65/// How much one read of standard input asks for: `std`'s own buffer size,
66/// which a read of at least that size bypasses.
67const STDIN_CHUNK: usize = 8 * 1024;
68
69/// Reads one chunk of standard input: empty at its end or on an error.
70pub(crate) fn read_stdin_chunk() -> Vec<u8> {
71    let mut chunk = vec![0u8; STDIN_CHUNK];
72    let read = loop {
73        match std::io::stdin().lock().read(&mut chunk) {
74            Err(error) if error.kind() == std::io::ErrorKind::Interrupted => {}
75            read => break read.unwrap_or(0),
76        }
77    };
78    chunk.truncate(read);
79    chunk
80}

82/// A standard input that is a regular file: a persistent stream with one read
83/// position, which CHARIN's start and LINEIN's line move (rexxref
84/// funct.xml:1314-1322 and :3080-3094), and whose counts are taken from that
85/// position (iostrms.xml:155-165), with [`crate::lines`]'s walkers and line
86/// cache on an [`interrexx_core::OpenFile`]. The descriptor's offset is the read
87/// position whenever another process may read from it: it is put there before
88/// a command starts and when the console is dropped, and taken from there
89/// when a command has returned.
90struct FileConsole {
91    open: interrexx_core::OpenFile,
92    /// Bytes read ahead, from character position `buffered_at`.
93    buffered: Vec<u8>,
94    buffered_at: u64,
95    /// The size the line count was taken at; a changed size drops the count.
96    counted_size: Option<u64>,
97    /// The descriptor's offset where this console's own reads, walks and
98    /// syncs last left it; any other offset was moved by a command.
99    left_at: std::cell::Cell<Option<u64>>,
100}
101
102impl FileConsole {
103    /// Standard input, when it is a regular file.
104    #[cfg(unix)]
105    fn open() -> Option<FileConsole> {
106        use std::os::fd::AsFd;
107        let stdin = std::io::stdin();
108        FileConsole::over(std::fs::File::from(
109            stdin.as_fd().try_clone_to_owned().ok()?,
110        ))
111    }
112
113    #[cfg(windows)]
114    fn open() -> Option<FileConsole> {
115        Self::over(crate::host::standard_file(0).ok()?)
116    }
117
118    /// `file` as a console, positioned at its descriptor's offset; `None`
119    /// unless it is a regular file.
120    fn over(mut file: std::fs::File) -> Option<FileConsole> {
121        use std::io::Seek;
122        if !file.metadata().ok()?.file_type().is_file() {
123            return None;
124        }
125        let offset = file.stream_position().unwrap_or(0);
126        let mut open = interrexx_core::OpenFile::new(file, 1, false, false);
127        open.read_position = as_position(offset + 1);
128        Some(FileConsole {
129            open,
130            buffered: Vec::new(),
131            buffered_at: 0,
132            counted_size: None,
133            left_at: std::cell::Cell::new(Some(offset)),
134        })
135    }
136
137    /// Records where the console's own use of the descriptor left its offset.
138    fn note_offset(&self) {
139        use std::io::Seek;
140        self.left_at.set(self.open.file().stream_position().ok());
141    }
142
143    /// The read position, as `read_from` takes it.
144    fn position(&self) -> u64 {
145        offset_of(self.open.read_position)
146    }
147
148    /// Makes `buffered` hold the character at the read position, answering
149    /// whether the file has one.
150    fn fill(&mut self) -> bool {
151        let at = self.position();
152        let end = self.buffered_at + u64::cast_from(self.buffered.len());
153        if at >= self.buffered_at && at < end {
154            return true;
155        }
156        self.buffered = read_from(self.open.flushed(), at, STDIN_CHUNK).unwrap_or_default();
157        self.buffered_at = at;
158        self.note_offset();
159        !self.buffered.is_empty()
160    }
161
162    /// The buffered bytes from the read position on, after [`FileConsole::fill`].
163    fn ahead(&self) -> &[u8] {
164        let from = usize::try_from(self.position().saturating_sub(self.buffered_at)).unwrap_or(0);
165        self.buffered.get(from..).unwrap_or_default()
166    }
167
168    /// One line with its terminator, or nothing at the end of the file.
169    fn read_line(&mut self) -> Vec<u8> {
170        let mut line = Vec::new();
171        while self.fill() {
172            let ahead = self.ahead();
173            let found = ahead.iter().position(|byte| *byte == b'\n');
174            let take = found.map_or(ahead.len(), |at| at + 1);
175            line.extend_from_slice(&ahead[..take]);
176            self.advance(take, i64::from(found.is_some()));
177            if found.is_some() {
178                break;
179            }
180        }
181        line
182    }
183
184    /// Moves the read position over `taken` characters holding `line_ends`
185    /// line ends. A line mark at the old position moves with it, so a count
186    /// after a loop of reads walks nothing; its debug digest is dropped, as
187    /// a mark moved without a walk cannot vouch for one.
188    fn advance(&mut self, taken: usize, line_ends: i64) {
189        let from = self.open.read_position;
190        self.open.read_position += memory_count(taken);
191        if let Some((mark, line)) = self.open.line_mark
192            && mark == from
193        {
194            self.open.line_mark = Some((self.open.read_position, line + line_ends));
195            self.open.line_mark_digest = None;
196        }
197    }
198
199    /// Up to `wanted` bytes from the read position, short only at the end of
200    /// the file.
201    fn read_bytes(&mut self, wanted: usize) -> Vec<u8> {
202        let mut out = Vec::new();
203        while out.len() < wanted && self.fill() {
204            let ahead = self.ahead();
205            let take = ahead.len().min(wanted - out.len());
206            out.extend_from_slice(&ahead[..take]);
207            let line_ends = ahead[..take].iter().filter(|byte| **byte == b'\n').count();
208            self.advance(take, memory_count(line_ends));
209        }
210        out
211    }
212
213    /// CHARS: the characters from the read position to the end of the file.
214    fn chars_left(&self) -> u64 {
215        let left = as_position(size_of(self.open.file())) - (self.open.read_position - 1);
216        u64::try_from(left).unwrap_or(0)
217    }
218
219    /// LINES Normal: whether any character remains.
220    fn has_data(&mut self) -> bool {
221        self.fill()
222    }
223
224    /// LINES Count: the lines from the read position on, a last line with no
225    /// line end included, from the line cache. Nothing here writes the file,
226    /// so the count is dropped when the file's size changes instead.
227    fn lines_left(&mut self) -> u64 {
228        let size = size_of(self.open.file());
229        if self.open.read_position > as_position(size) {
230            return 0;
231        }
232        match self.counted_size {
233            Some(counted_at) if counted_at == size => {}
234            _ => {
235                self.open.drop_line_total();
236                self.open.line_total_digest = None;
237                self.counted_size = Some(size);
238            }
239        }
240        let position = self.open.read_position;
241        let left = lines_after(&mut self.open, position).unwrap_or(0);
242        self.note_offset();
243        left
244    }
245
246    /// CHARIN's start: moves the read position to character `position`,
247    /// counted from 1, answering `false` past the end of the file, where
248    /// nothing moves.
249    fn seek_char(&mut self, position: u64) -> bool {
250        if position == 0 || position > size_of(self.open.file()) {
251            return false;
252        }
253        self.open.read_position = as_position(position);
254        true
255    }
256
257    /// LINEIN's line: moves the read position to the start of line `line`,
258    /// or to the end of the file when it has fewer lines.
259    fn seek_line(&mut self, line: u64) {
260        let (_, at) = crate::lines::seek_line(&mut self.open, as_position(line), 0);
261        self.open.read_position = at;
262        self.note_offset();
263    }
264
265    /// Puts the descriptor's offset at the read position where the console's
266    /// own reads moved it, for whatever reads the descriptor next. Where they
267    /// did not, the offset is where the last command left it, which is the
268    /// read position already.
269    fn sync_offset(&self) {
270        use std::io::{Seek, SeekFrom};
271        let want = self.position().saturating_sub(1);
272        if self
273            .open
274            .file()
275            .stream_position()
276            .is_ok_and(|at| at != want)
277        {
278            let _ = self.open.file().seek(SeekFrom::Start(want));
279        }
280        self.note_offset();
281    }
282
283    /// Takes the read position from the descriptor's offset, where a command
284    /// that read the descriptor left it. The line mark and the read-ahead go.
285    /// An offset where the console itself left it means no command read, and
286    /// nothing moves.
287    fn adopt_offset(&mut self) {
288        use std::io::Seek;
289        let Ok(offset) = self.open.file().stream_position() else {
290            return;
291        };
292        if self.left_at.get() == Some(offset) {
293            return;
294        }
295        self.left_at.set(Some(offset));
296        self.open.read_position = as_position(offset + 1);
297        self.open.line_mark = None;
298        self.open.line_mark_digest = None;
299        self.buffered.clear();
300    }
301}

303impl Drop for FileConsole {
304    fn drop(&mut self) {
305        self.sync_offset();

309impl Input {
310    /// Whether this is the process's own standard input, whose reads hand
311    /// the buffered output over first.
312    fn is_live(&self) -> bool {
313        matches!(self.source, Source::Stdin(_) | Source::File(_))
314    }
315
316    /// Whether standard input needs a chunk more before the next read can
317    /// answer: a whole line where `line`, else any byte.
318    fn stdin_wants(&self, line: bool) -> bool {
319        let Source::Stdin(stdin) = &self.source else {
320            return false;
321        };
322        let ready = match line {
323            true => stdin.buffered.contains(&b'\n'),
324            false => !stdin.buffered.is_empty(),
325        };
326        !ready && !stdin.ended
327    }
328
329    /// Starts a chunk read where none is in flight, answering whether one
330    /// was started.
331    fn start_stdin_read(&mut self) -> bool {
332        match &mut self.source {
333            Source::Stdin(stdin) if !stdin.reading => {
334                stdin.reading = true;
335                true
336            }
337            _ => false,
338        }
339    }
340
341    /// A halt abandoned a read: the counts answer `0` until the next read, as
342    /// they do after an interrupted read.
343    fn interrupted(&mut self) {
344        self.exhausted = true;
345    }
346
347    /// A read of standard input follows, which decides the counts afresh.
348    fn reading_stdin(&mut self) {
349        if matches!(self.source, Source::Stdin(_)) {
350            self.exhausted = false;
351        }
352    }
353
354    /// Files a chunk standard input delivered: its end where empty.
355    pub(crate) fn receive(&mut self, chunk: Vec<u8>) {
356        if let Source::Stdin(stdin) = &mut self.source {
357            stdin.reading = false;
358            if chunk.is_empty() {
359                stdin.ended = true;
360            }
361            stdin.buffered.extend(chunk);
362        }
363    }

365    pub(crate) fn new(input: ProgramInput) -> Input {
366        Input {
367            source: match input {
368                ProgramInput::Nothing => Source::Nothing,
369                ProgramInput::Stdin => match FileConsole::open() {
370                    Some(file) => Source::File(file),
371                    None => Source::Stdin(Stdin::default()),
372                },
373                ProgramInput::Bytes(bytes) => Source::Bytes(Cursor::new(bytes)),
374            },
375            exhausted: false,
376            lines_asked: false,
377        }
378    }
379
380    /// The next line, or `None` once there is nothing left to read.
381    fn read_line(&mut self) -> Option<Vec<u8>> {
382        let mut line = Vec::new();
383        let read = match &mut self.source {
384            Source::Nothing => {
385                self.exhausted = true;
386                return None;
387            }
388            Source::Stdin(stdin) => {
389                let end = match stdin.buffered.iter().position(|byte| *byte == b'\n') {
390                    Some(at) => at + 1,
391                    None => stdin.buffered.len(),
392                };
393                line.extend(stdin.buffered.drain(..end));
394                Ok(end)
395            }
396            Source::File(file) => {
397                line = file.read_line();
398                Ok(line.len())
399            }
400            Source::Bytes(cursor) => cursor.read_until(b'\n', &mut line),
401        };
402        match read {
403            // Zero bytes with no error is end of input; a read error answers
404            // the same way, per this module's doc.
405            Ok(0) | Err(_) => {
406                self.exhausted = true;
407                return None;
408            }
409            Ok(_) => {}
410        }
411        // The terminator is not part of the line, and a `\r` immediately
412        // before it is not either -- but a `\r` anywhere else is data, and a
413        // final line with no newline has no pair to collapse. All three are
414        // measured in the module doc's table.
415        if line.last() == Some(&b'\n') {
416            line.pop();
417            if line.last() == Some(&b'\r') {
418                line.pop();
419            }
420        }
421        Some(line)
422    }
423
424    /// Up to `wanted` bytes, answering what is **already there** rather than
425    /// waiting for the rest. For example, `.stdin~charin(,50)` against a pipe
426    /// carrying six bytes and still open answers those six immediately, so
427    /// filling the request would hang the interpreter on a live descriptor.
428    /// Empty once there is nothing left.
429    fn read_bytes(&mut self, wanted: usize) -> Vec<u8> {
430        let mut buffer = vec![0u8; wanted];
431        let read = match &mut self.source {
432            Source::Nothing => {
433                self.exhausted = true;
434                return Vec::new();
435            }
436            Source::Stdin(stdin) => {
437                let count = wanted.min(stdin.buffered.len());
438                buffer[..count].copy_from_slice(&stdin.buffered[..count]);
439                stdin.buffered.drain(..count);
440                Ok(count)
441            }
442            Source::File(file) => return file.read_bytes(wanted),
443            Source::Bytes(cursor) => cursor.read(&mut buffer),
444        };
445        let filled = match read {
446            Ok(0) | Err(_) => {
447                self.exhausted = true;
448                0
449            }
450            Ok(count) => count,
451        };
452        buffer.truncate(filled);
453        buffer
454    }
455
456    /// The lines left to read, counted the way [`Input::read_line`] would
457    /// split them, when the source can be measured without consuming it.
458    /// `None` when it cannot, matching [`Input::remaining`]. Without `count`
459    /// a file console answers 1 or 0 for whether anything is left, with no
460    /// scan (rexxref funct.xml:3296-3306).
461    fn remaining_lines(&mut self, count: bool) -> Option<u64> {
462        if let Source::File(file) = &mut self.source {
463            return Some(match count {
464                true => file.lines_left(),
465                false => u64::from(file.has_data()),
466            });
467        }
468        if self.exhausted {
469            return Some(0);
470        }
471        if let Source::Stdin(_) = &self.source {
472            return self.stdin_waiting().map(u64::from);
473        }
474        let asked = std::mem::replace(&mut self.lines_asked, true);
475        match &self.source {
476            // No input answers `1` once and `0` after, which is where this
477            // parts from [`Input::remaining`].
478            Source::Nothing if asked => Some(0),
479            Source::Nothing | Source::Stdin(_) | Source::File(_) => None,
480            Source::Bytes(cursor) => {
481                let at = usize::try_from(cursor.position()).unwrap_or(usize::MAX);
482                let left = cursor.get_ref().get(at..).unwrap_or_default();
483                if left.is_empty() {
484                    return Some(0);
485                }
486                // A final line with no terminator still counts, which is why
487                // this is not simply the newline count.
488                let terminators =
489                    u64::cast_from(left.iter().filter(|byte| **byte == b'\n').count());
490                Some(match left.last() {
491                    Some(b'\n') => terminators,
492                    _ => terminators + 1,
493                })
494            }
495        }
496    }

498    /// Whether data waits on a live standard input, found without blocking:
499    /// what is buffered, else a `poll` with no wait, and when it reports the
500    /// descriptor readable, a read that cannot block takes what is there into
501    /// the buffer (rexxref iostrms.xml:189-193). `None` while a read is
502    /// already in flight, when the state cannot be learned.
503    fn stdin_waiting(&mut self) -> Option<bool> {
504        #[cfg(unix)]
505        use rustix::event::{PollFd, PollFlags, Timespec, poll};
506        let Source::Stdin(stdin) = &self.source else {
507            return None;
508        };
509        if !stdin.buffered.is_empty() {
510            return Some(true);
511        }
512        if stdin.ended {
513            return Some(false);
514        }
515        if stdin.reading {
516            return None;
517        }
518        #[cfg(unix)]
519        {
520            let zero = Timespec {
521                tv_sec: 0,
522                tv_nsec: 0,
523            };
524            let descriptor = std::io::stdin();
525            let mut polled = [PollFd::new(&descriptor, PollFlags::IN)];
526            match poll(&mut polled, Some(&zero)) {
527                Ok(_) if polled[0].revents().is_empty() => return Some(false),
528                Ok(_) => {}
529                Err(_) => return None,
530            }
531        }
532        #[cfg(windows)]
533        {
534            let file = crate::host::standard_file(0).ok()?;
535            if crate::host::bytes_waiting(&file)? == 0 {
536                return Some(false);
537            }
538        }
539        let chunk = read_stdin_chunk();
540        self.receive(chunk);
541        let Source::Stdin(stdin) = &self.source else {
542            return None;
543        };
544        Some(!stdin.buffered.is_empty())
545    }
546
547    /// The bytes left to read, when that is knowable without consuming them,
548    /// and `None` when it is not -- a live standard input is a stream, not a
549    /// length. For example, `.stdin~chars` answers the true remaining count for
550    /// a redirected regular file, `1` for a pipe nothing has drained, and `0`
551    /// for either once a read has found the end, which is what `exhausted`
552    /// carries.
553    fn remaining(&mut self) -> Option<u64> {
554        if let Source::File(file) = &self.source {
555            return Some(file.chars_left());
556        }
557        if self.exhausted {
558            return Some(0);
559        }
560        if let Source::Stdin(_) = &self.source {
561            return self.stdin_waiting().map(u64::from);
562        }
563        match &self.source {
564            // Live but empty, exactly like a `/dev/null` descriptor: `1` is
565            // answered there until a read drains it and `0` after -- so an
566            // unread `Nothing` is not yet a zero.
567            Source::Nothing => None,
568            Source::Stdin(_) | Source::File(_) => None,
569            Source::Bytes(cursor) => {
570                Some(u64::cast_from(cursor.get_ref().len()).saturating_sub(cursor.position()))
571            }
572        }
573    }
574}

576impl Interp {
577    /// One line for `PULL` and `PARSE PULL`: the queue's head if the queue has
578    /// one, and otherwise [`Interp::linein_line`].
579    ///
580    /// **The queue is asked first and the route is not asked at all when it
581    /// answers** -- measured, a program with one line pushed reads it and the
582    /// destination sees a single `LINEIN`, for the second read.
583    pub(crate) fn pull_line(&mut self) -> Result<Vec<u8>, Failure> {
584        // The queue exit's first pass: a handler that returned no line answers `.nil`,
585        // read as its string.
586        if self.exits.is_some()
587            && let Some(line) = self.pull_exit()?
588        {
589            return Ok(match line {
590                Some(line) => line,
591                None => self.unsent_string_value_text(ObjRef::NIL),
592            });
593        }
594        let Some(queue) = self.local_route(b"STDQUE")? else {
595            return match self.queue.pop() {
596                Some(line) => Ok(line),
597                None => self.linein_line(),
598            };
599        };
600        let caller = self.caller();
601        let answer = pinned!(
602            self,
603            crate::pinning::PinKind::PullWrapper,
604            self.send_message(queue, b"PULL", None, &[], caller)
605        )?;
606        match answer {
607            Some(value) if value != ObjRef::NIL => self.input_answer_text(Some(value)),
608            _ => self.linein_line(),
609        }
610    }
611
612    /// One line for `PARSE LINEIN`: always the route, never the queue.
613    ///
614    /// **A message, not a descriptor read.** `LINEIN` is sent with no arguments
615    /// to whatever `.local` holds under `INPUT`, so a program that redirects
616    /// the destination answers the instruction. End of input raises `NOTREADY`
617    /// for the same reason: the untouched route ends at `.STDIN`, whose own
618    /// `LINEIN` raises it -- `PARSE LINEIN` and `PARSE PULL` on empty input are
619    /// both rc 9 under a trap, and answer no empty line.
620    ///
621    /// A missing or `.nil` entry reads the descriptor instead, which is this
622    /// crate's licensed answer where the reference run dies (oracle defect,
623    /// `corpus/oracle-crashes.txt` entry 11b).
624    pub(crate) fn linein_line(&mut self) -> Result<Vec<u8>, Failure> {
625        // The I/O exit's first pass.
626        if self.exits.is_some()
627            && let Some(line) = self.terminal_read_exit()?
628        {
629            return Ok(line);
630        }
631        let route = self.local_route(b"INPUT")?;
632        let Some(route) = route.filter(|route| *route != ObjRef::NIL) else {
633            return Ok(self.input_line()?.unwrap_or_default());
634        };
635        let caller = self.caller();
636        let answer = pinned!(
637            self,
638            crate::pinning::PinKind::PullWrapper,
639            self.send_message(route, crate::dispatch::LINEIN, None, &[], caller)
640        )?;
641        self.input_answer_text(answer)
642    }

644    /// Input instructions and debug reads use the required string conversion.
645    /// A `LINEIN` with no result or a `.nil` result supplies an empty line.
646    pub(crate) fn input_answer_text(&mut self, answer: Option<ObjRef>) -> Result<Vec<u8>, Failure> {
647        let Some(value) = answer.filter(|value| *value != ObjRef::NIL) else {
648            return Ok(Vec::new());
649        };
650        let value = self.required_string_value(value)?;
651        Ok(self.to_text(value).into_owned())
652    }

654    /// The reader behind `.STDIN~LINEIN`, keeping end of input apart from an

680    /// Reads standard input until it holds a line where `line`, else a byte, or
681    /// has ended: each chunk off the baton, with this thread idle until it is
682    /// posted or a signal's halt arrives. A halt abandons the wait, whose chunk
683    /// is filed when it comes, and answers `false`: the read answers nothing,
684    /// as an interrupted read does. Other posts wait for the next drain, held
685    /// here so that an idle waits for a new post.
686    fn fill_stdin(&mut self, line: bool) -> Result<bool, Failure> {
687        let mut kept = std::collections::VecDeque::new();
688        let filled = self.fill_stdin_keeping(line, &mut kept);
689        self.timer.requeue(kept);
690        if matches!(filled, Ok(true)) {
691            self.input.reading_stdin();
692        }
693        filled
694    }
695
696    fn fill_stdin_keeping(
697        &mut self,
698        line: bool,
699        kept: &mut std::collections::VecDeque<crate::scheduler::Posted>,
700    ) -> Result<bool, Failure> {
701        while self.input.stdin_wants(line) {
702            if self.input.start_stdin_read() {
703                let worker = match self.pool.reserve() {
704                    Ok(Some(worker)) => worker,
705                    Ok(None) => {
706                        self.input.receive(read_stdin_chunk());
707                        continue;
708                    }
709                    Err(failure) => {
710                        // Only this unstarted read's marker changes. Partial
711                        // input, EOF and descriptor ownership remain intact.
712                        if let Source::Stdin(stdin) = &mut self.input.source {
713                            stdin.reading = false;
714                        }
715                        return Err(failure);
716                    }
717                };
718                let inbox = self.timer.inbox();
719                let baton = crate::sync::Arc::clone(&self.baton);
720                worker.run(Box::new(move || {
721                    crate::scheduler::posting_panics(&inbox, &baton, || {
722                        let chunk = crate::signal::unblocked(read_stdin_chunk);
723                        inbox.post(crate::scheduler::Posted::Input(chunk));
724                    });
725                }));
726            }
727            let mut halted = false;
728            for post in self.timer.idle() {
729                match post {
730                    crate::scheduler::Posted::Input(chunk) => self.input.receive(chunk),
731                    crate::scheduler::Posted::Halt => halted = true,
732                    other => kept.push_back(other),
733                }
734            }
735            if halted {
736                self.input.interrupted();
737                self.halt_all();
738                return Ok(false);
739            }
740        }
741        Ok(true)
742    }

744    /// Hands the two output buffers to the embedding's [`crate::Sinks`],
745    /// which is what puts a prompt in front of the reader before the read
746    /// that waits for their answer.
747    ///
748    /// Does nothing for an embedding that installed none, and nothing for a
749    /// source that cannot wait -- so no differential run's output moves.
750    fn hand_over_before_read(&mut self) {
751        if self.input.is_live() {
752            self.hand_over_output();
753        }
754    }

756    /// Hands the two output buffers to the embedding's [`crate::Sinks`], if
757    /// it installed any: before a read, before this thread idles and before
758    /// a command starts.
759    pub(crate) fn hand_over_output(&mut self) {
760        self.hand_over_stdout();
761        self.hand_over_stderr();
762    }
763
764    /// [`Interp::hand_over_output`] for standard output alone. A refusal is
765    /// kept in [`Interp::sink_errors`].
766    pub(crate) fn hand_over_stdout(&mut self) {
767        if let Some(sinks) = &mut self.sinks
768            && !self.out.is_empty()
769        {
770            if let Err(error) = (sinks.stdout)(&self.out) {
771                self.sink_errors[0] = Some(error.raw_os_error().unwrap_or(0));
772            }
773            self.out.clear();
774        }
775    }
776
777    /// [`Interp::hand_over_output`] for standard error alone.
778    pub(crate) fn hand_over_stderr(&mut self) {
779        if let Some(sinks) = &mut self.sinks
780            && !self.trace.is_empty()
781        {
782            if let Err(error) = (sinks.stderr)(&self.trace) {
783                self.sink_errors[1] = Some(error.raw_os_error().unwrap_or(0));
784            }
785            self.trace.clear();
786        }
787    }
