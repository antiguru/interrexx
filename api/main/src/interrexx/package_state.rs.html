13/// One package's fields. A copy is a field copy holding the same program.
14#[derive(Clone, Debug, Default)]
15pub(crate) struct PackageData {
16    /// The program the package was compiled from, `None` for the REXX
17    /// package. Every copy of a package holds the same one.
18    pub(crate) program: Option<ProgramId>,
19    /// What keeps [`PackageData::program`] loaded, `None` for the REXX
20    /// package. Every copy holds it.
21    pub(crate) hold: Option<ProgramHold>,
22    /// What `::OPTIONS` directives and `Package~options` writes left on the
23    /// package, `None` where neither has.
24    pub(crate) options: Option<PackageOptions>,
25    /// The package's initial settings, entered at the first change.
26    pub(crate) initial_options: Option<PackageOptions>,
27    /// The security manager the package carries, which the
28    /// `setSecurityManager` setters write.
29    pub(crate) security_manager: Option<ObjRef>,
30    /// Every `::ROUTINE` the program installs and every routine `~addRoutine`
31    /// files, by upcased name.
32    pub(crate) routines: FxHashMap<Box<[u8]>, MergedRoutine>,
33    /// Whether a program has written the package's `.ROUTINES` table, whose
34    /// entries routine lookups read in place of [`PackageData::routines`]
35    /// from then on.
36    pub(crate) routines_written: bool,
37    /// `publicRoutines`: the subset of [`PackageData::routines`] a
38    /// `::ROUTINE ... PUBLIC` filed.
39    pub(crate) public_routines: FxHashMap<Box<[u8]>, MergedRoutine>,
40    /// `mergedPublicRoutines`: every `::REQUIRES ... LIBRARY`'s library
41    /// routines, then each `::REQUIRES`'s public and imported routines, a name
42    /// keeping its first entry (the library and required merges).
43    pub(crate) merged_public_routines: FxHashMap<Box<[u8]>, MergedRoutine>,
44    /// `installedClasses`: the classes the package's `::CLASS` directives and
45    /// `~addClass` installed, by uppercased name.
46    pub(crate) classes: NameMap<Box<[u8]>, ObjRef>,
47    /// `installedPublicClasses`, a second table beside
48    /// [`PackageData::classes`] and not a flag on its entries.
49    pub(crate) public_classes: NameMap<Box<[u8]>, ObjRef>,
50    /// `mergedPublicClasses`, merged alongside the routines above.
51    pub(crate) merged_public_classes: NameMap<Box<[u8]>, ObjRef>,
52    /// The packages `::REQUIRES ... NAMESPACE` directives registered, under
53    /// the upcased qualifier.
54    pub(crate) namespaces: FxHashMap<Box<[u8]>, ObjRef>,
55    /// The `Directory` `Package~local` answers, built on first ask.
56    pub(crate) local: Option<ObjRef>,
57    /// `loadedPackages`, in the order they were added.
58    pub(crate) imports: Vec<ObjRef>,
59    /// `parentPackage`: the package a routine or class lookup that misses
60    /// this one's own tables walks next.
61    pub(crate) parent: Option<ObjRef>,
62    /// The `Routine` object standing for the program's main section.
63    pub(crate) main_routine: Option<ObjRef>,
64    /// The `.METHODS`, `.ROUTINES` and `.RESOURCES` tables, by
65    /// [`PackageTable::index`], each built on first ask.
66    pub(crate) tables: [Option<ObjRef>; 3],
67    /// The value each `::CONSTANT` accessor answers, by declaring directive.
68    pub(crate) constants: FxHashMap<usize, ObjRef>,
69    /// The package of each other program's `::ROUTINE` a call site of this
70    /// program keeps, with how many sites keep one of its routines: the
71    /// instruction holds the routine it keeps ([`Interp::hold_site_program`]).
72    pub(crate) site_held: FxHashMap<ObjRef, u32>,
73}

75impl PackageFields for PackageData {
76    fn trace(&self, out: &mut Vec<ObjRef>) {
77        let PackageData {
78            program: _,
79            hold: _,
80            options: _,
81            initial_options: _,
82            security_manager,
83            // Each `Routine` object here is in the package's `.ROUTINES`
84            // table too, which `~addRoutine` writes.
85            routines: _,
86            routines_written: _,
87            public_routines,
88            merged_public_routines,
89            classes,
90            public_classes,
91            merged_public_classes,
92            namespaces,
93            local,
94            imports,
95            parent,
96            main_routine,
97            tables,
98            constants,
99            site_held,
100        } = self;
101        out.extend(*security_manager);
102        // The `Routine` objects `~addRoutine` filed as themselves, which the
103        // table does not follow into these copies, and the package of each
104        // `::ROUTINE`, as the merged `Routine` object holds it.
105        for table in [public_routines, merged_public_routines] {
106            out.extend(table.values().filter_map(|merged| match merged {
107                MergedRoutine::Installed(_, object)
108                | MergedRoutine::LibraryObject(_, object)
109                | MergedRoutine::InternalObject(_, object) => Some(*object),
110                _ => None,
111            }));
112        }
113        // A package pins the classes it declares: a `::CLASS` class survives a
114        // forced collection.
115        for table in [classes, public_classes, merged_public_classes] {
116            out.extend(table.values().copied());
117        }
118        out.extend(namespaces.values().copied());
119        out.extend(*local);
120        out.extend(imports.iter().copied());
121        out.extend(*parent);
122        out.extend(*main_routine);
123        out.extend(tables.iter().flatten().copied());
124        out.extend(constants.values().copied());
125        out.extend(site_held.keys().copied());
126    }
127
128    fn clone_fields(&self) -> Box<dyn PackageFields> {
129        Box::new(self.clone())
130    }
131}

133impl PackageData {
134    /// A fresh package's fields, over `program`.
135    pub(crate) fn state(program: Option<ProgramId>) -> PackageState {
136        PackageState::new(Box::new(PackageData {
137            program,
138            ..PackageData::default()
139        }))
140    }
141}

143impl Interp {
144    /// The object standing for `program`'s package, or `None` before one has
145    /// been built.
146    #[inline]
147    pub(crate) fn program_package(&self, program: ProgramId) -> Option<ObjRef> {
148        self.programs.package(program)
149    }
150
151    /// `package`'s fields, or `None` for an object that is not a package.
152    #[inline]
153    pub(crate) fn package_data(&self, package: ObjRef) -> Option<&PackageData> {
154        self.heap
155            .get(package)?
156            .body
157            .package()?
158            .downcast_ref::<PackageData>()
159    }
160
161    /// [`Interp::package_data`] for a caller that changes them. Every such
162    /// write moves [`Interp::class_lookup_stamp`], because a package's class
163    /// tables and parent are what a `.NAME` read's memo was answered from.
164    #[inline]
165    pub(crate) fn package_data_mut(&mut self, package: ObjRef) -> Option<&mut PackageData> {
166        self.class_lookup_stamp += 1;
167        self.heap
168            .get_mut(package)?
169            .body
170            .package_mut()?
171            .downcast_mut::<PackageData>()
172    }
173
174    /// The fields of `program`'s package, or `None` before it has been built.
175    #[inline]
176    pub(crate) fn program_data(&self, program: ProgramId) -> Option<&PackageData> {
177        self.package_data(self.program_package(program)?)
178    }
179
180    /// The fields of `package`'s object, built first where it does not exist
181    /// yet. May allocate, and so collect.
182    pub(crate) fn package_fields_mut(
183        &mut self,
184        package: Package,
185    ) -> Result<&mut PackageData, crate::error::Failure> {
186        let object = self.package_object(package)?;
187        Ok(self
188            .package_data_mut(object)
189            .expect("package_object builds a package body"))
190    }
191
192    /// The fields of `program`'s package, built first where it does not
193    /// exist yet. May allocate, and so collect.
194    pub(crate) fn program_fields_mut(
195        &mut self,
196        program: ProgramId,
197    ) -> Result<&mut PackageData, crate::error::Failure> {
198        self.package_fields_mut(Package::Program(program))
199    }
200
201    /// `package` and its fields, then each package up its parent chain.
202    pub(crate) fn package_chain(
203        &self,
204        package: Option<ObjRef>,
205    ) -> impl Iterator<Item = (ObjRef, &PackageData)> + '_ {
206        let first = package.and_then(|package| Some((package, self.package_data(package)?)));
207        // A parent is a package that existed before its child's program was
208        // compiled, so the programs along a chain strictly decrease and the
209        // chain cannot close; the bound costs less than that argument.
210        std::iter::successors(first, |(_, data)| {
211            let parent = data.parent?;
212            Some((parent, self.package_data(parent)?))
213        })
214        .take(self.programs.len() + 1)
215    }
216
217    /// Whether `package` is the REXX package's own object, which
218    /// `checkRexxPackage` refuses writes to. A copy of it is not: the test is
219    /// on the original object, and a copy is a new one.
220    pub(crate) fn is_the_rexx_package(&self, package: ObjRef) -> bool {
221        self.rexx_package == Some(package)
222    }
223
224    /// Which package a package object's code belongs to, or `None` for an
225    /// object that is not a package.
226    pub(crate) fn which_package(&self, package: ObjRef) -> Option<Package> {
227        Some(match self.package_data(package)?.program {
228            Some(program) => Package::Program(program),
229            None => Package::Rexx,
230        })
231    }
