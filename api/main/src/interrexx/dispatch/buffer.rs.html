328/// `MutableBuffer~new(string, size, ...)`: an instance carrying the string and
329/// the capacities derived from the arguments: `default_size` is the second
330/// argument or `DEFAULT_BUFFER_LENGTH`, and the capacity is that or the
331/// string's length, whichever is larger.
332pub(super) fn native_mutable_buffer_new(
333    interp: &mut Interp,
334    _cleared: Cleared,
335    receiver: ObjRef,
336    args: &[Option<ObjRef>],
337) -> Result<super::NativeStarted, Failure> {
338    let class = class_receiver(interp, receiver)?;
339    let mut initial = interp.take_result_buffer();
340    if let Some(text) = args.first().copied().flatten() {
341        let text = required_string_argument(interp, text, 1)?;
342        initial.extend_from_slice(&interp.to_text(text));
343    }
344    let default_size = optional_length_argument(interp, args, 1)?.unwrap_or(BUFFER_DEFAULT_LENGTH);
345    let capacity = default_size.max(initial.len());
346    interp.admit_bytes(capacity)?;
347    let mut bytes = Vec::new();
348    bytes
349        .try_reserve_exact(capacity)
350        .map_err(|_| Failure::from(Raised::system_resources()))?;
351    bytes.extend_from_slice(&initial);
352    interp.give_result_buffer(initial);
353    let object = new_instance(interp, class)?;
354    let Some(Object {
355        body: Body::Instance { native, .. },
356        ..
357    }) = interp.heap.get_mut(object)
358    else {
359        unreachable!("new_instance allocates a Body::Instance")
360    };
361    *native = Some(Box::new(interrexx_core::NativeState::Buffer(BufferState {
362        bytes: bytes.into(),
363        capacity,
364        default_size,
365    })));
366    interp.charge_growth(capacity);
367    let kept = args.len().saturating_sub(2);
368    let rest: Vec<Option<ObjRef>> = args.iter().take(kept).copied().collect();
369    super::begin_init(interp, object, &rest)
370}

467/// Every argument a required string, appended in turn; answers the receiver.
468fn native_mutable_buffer_append(
469    interp: &mut Interp,
470    _cleared: Cleared,
471    receiver: ObjRef,
472    args: &[Option<ObjRef>],
473) -> Result<Option<ObjRef>, Failure> {
474    if args.is_empty() {
475        return Err(Raised::missing_method_argument(1).into());
476    }
477    for (index, argument) in args.iter().enumerate() {
478        let Some(argument) = *argument else {
479            return Err(Raised::missing_method_argument(index + 1).into());
480        };
481        let argument = required_string_argument(interp, argument, index + 1)?;
482        let mut piece = interp.take_result_buffer();
483        piece.extend_from_slice(&interp.to_text(argument));
484        let state = grow_buffer(interp, receiver, piece.len())?;
485        state.bytes.extend_from_slice(&piece);
486        interp.give_result_buffer(piece);
487    }
488    Ok(Some(receiver))
489}

526/// `MutableBuffer~setBufferSize`, whose rule [`BufferState::set_buffer_size`]
527/// carries; answers the receiver.
528fn native_mutable_buffer_setbuffersize(
529    interp: &mut Interp,
530    _cleared: Cleared,
531    receiver: ObjRef,
532    args: &[Option<ObjRef>],
533) -> Result<Option<ObjRef>, Failure> {
534    let size = required_length_argument(interp, args, 0)?;
535    let state = buffer_state_mut(interp, receiver)?;
536    let before = state.capacity;
537    let grown = state
538        .set_buffer_size(size)
539        .map_err(|_| Failure::from(Raised::system_resources()))?;
540    let shrunk = before.saturating_sub(state.capacity);
541    interp.charge_growth(grown);
542    interp.heap.release_body_bytes(shrunk);
543    Ok(Some(receiver))
544}

981/// A fresh `Array` holding one string per element of `pieces`.
982pub(super) fn array_of_texts(interp: &mut Interp, pieces: Vec<Vec<u8>>) -> Result<ObjRef, Failure> {
983    let frame = interp.roots.activity_mut().push_frame();
984    let mut slots = Vec::with_capacity(pieces.len());
985    for piece in pieces {
986        let value = interp.text_built(piece);
987        interp.roots.activity_mut().push_temp(value);
988        slots.push(Some(value));
989    }
990    let object = interp.alloc_charged(
991        BehaviourId::ARRAY,
992        Body::Array {
993            dimensions: None,
994            slots: slots.into(),
995        },
996    );
997    interp.roots.activity_mut().pop_frame(frame);
998    interp.roots.activity_mut().push_temp(object);
999    let caller = interp.caller();
1000    interp.send_message(object, INIT, None, &[], caller)?;
1001    Ok(object)
1002}

1261/// The rest of `match`'s arguments once `start` is inside the receiver: an
1262/// explicit offset past `other` is `.false` before the length is looked at, and a
1263/// length past `other` is `.false`: `~match(1, 'abc', 4, -1)` is `.false`. `None` is
1264/// those two answers; `Some` is the region of `other` to compare.
1265pub(super) fn match_region_arguments(
1266    interp: &mut Interp,
1267    args: &[Option<ObjRef>],
1268    other_len: usize,
1269) -> Result<Option<(usize, usize)>, Failure> {
1270    let offset = optional_position_argument(interp, args, 2)?;
1271    if offset.is_some_and(|offset| offset > other_len) {
1272        return Ok(None);
1273    }
1274    let offset = offset.unwrap_or(1);
1275    let length = optional_length_argument(interp, args, 3)?.unwrap_or(other_len + 1 - offset);
1276    if length == 0 || offset.saturating_add(length) - 1 > other_len {
1277        return Ok(None);
1278    }
1279    Ok(Some((offset, length)))
1280}

1282/// [`match_region_arguments`]' comparison, once the receiver's bytes are in
1283/// hand: the two regions are compared, and `matches` is what separates the two
1284/// spellings.
1285pub(super) fn match_region_over(
1286    haystack: &[u8],
1287    start: usize,
1288    other: &[u8],
1289    region: Option<(usize, usize)>,
1290    matches: fn(&[u8], &[u8]) -> bool,
1291) -> bool {
1292    let Some((offset, length)) = region else {
1293        return false;
1294    };
1295    haystack
1296        .get(start - 1..(start - 1).saturating_add(length))
1297        .is_some_and(|region| matches(region, &other[offset - 1..offset - 1 + length]))
1298}

1386/// The receiver's state with room for `added` more bytes
1387/// ([`BufferState::ensure_capacity`], whose refusal is 5.1), the growth charged
1388/// before the state is answered.
1389fn grow_buffer(
1390    interp: &mut Interp,
1391    receiver: ObjRef,
1392    added: usize,
1393) -> Result<&mut BufferState, Failure> {
1394    interp.admit_bytes(added)?;
1395    let state = buffer_state_mut(interp, receiver)?;
1396    let grown = state
1397        .ensure_capacity(added)
1398        .map_err(|_| Failure::from(Raised::system_resources()))?;
1399    // Charged here, not by the caller: a caller's later step can fail, and a
1400    // growth it never charged would leave the live figure short for the
1401    // buffer's life.
1402    interp.charge_growth(grown);
1403    buffer_state_mut(interp, receiver)
1404}

1528/// How much of the receiver `replaceAt` overwrites, and how long the answer
1529/// is: `(replaced, final_length)`.
1530pub(super) fn replace_at_plan(
1531    contents: usize,
1532    begin: usize,
1533    length: Option<usize>,
1534    new_len: usize,
1535) -> (usize, usize) {
1536    let mut replaced = length.unwrap_or(new_len);
1537    if begin > contents {
1538        replaced = 0;
1539    } else if begin.saturating_add(replaced) > contents {
1540        replaced = contents - begin;
1541    }
1542    let kept = if begin > contents { begin } else { contents };
1543    (replaced, kept - replaced + new_len)
1544}
