1//! The hash store's layout in its receiver's pool, and the primitives that
2//! read and write it.
3//!
4//! A store is a compact table. Its entries sit in dense arrays in the order
5//! they were added, and a sparse slot index, open-addressed, maps a hash to
6//! the entries filed under it. A new entry goes at the end of the entry
7//! arrays; a removed one leaves a hole there. When the entry arrays fill or
8//! holes dominate them, the store is rebuilt into fresh arrays that keep the
9//! live entries in their order and re-slot them from each entry's cached hash,
10//! so a rebuild sends nothing. Iteration walks the entry arrays, which is what
11//! makes every enumeration answer insertion order.
12//!
13//! **The live entries of one code are one chain**, filed at one slot: the
14//! slot names the oldest of them, each links to the next, oldest first, and
15//! the oldest also names the newest, so an append joins its chain without
16//! walking it. Duplicates of one index share a code, so however many a
17//! `Relation` or `Bag` holds, a lookup reaches the oldest at the chain's head
18//! and an append or a removal of the oldest passes over no other. Distinct
19//! indexes whose hashes fold alike share a chain too, and a lookup compares
20//! them by `==` in the chain's order. A slot whose chain has lost its last
21//! entry is marked deleted.
22//!
23//! **Within one set of arrays, a position holds at most one entry in its
24//! life**: entries are only appended, a hole is never filled again, and every
25//! operation that would start over at position zero (a rebuild, `EMPTY`)
26//! installs fresh arrays. A caller that kept a position across a send can
27//! therefore trust it while the arrays are the same and the position is live.
28

98/// A slot no entry has been filed in since the slots were built: a probe
99/// that reaches one has seen every entry of its hash.
100const EMPTY: i64 = 0;
101/// A slot whose chain lost its last entry: a probe passes over it, and a new
102/// chain may take it once the probe has found no chain of its code.
103const DELETED: i64 = -1;

105/// The fields each entry has in the codes array of a linked store, at
106/// `position * FIELDS`: its [`CODE`], the position of the [`NEXT`] entry of its
107/// chain, and, for the oldest entry of a chain, the position of the chain's
108/// [`TAIL`]. The links reuse the codes array rather than a pool entry of their
109/// own, so that the store stays the names that `copy`, a rebuild and `EMPTY`
110/// already treat as one: a copy duplicates the links with the codes, and fresh
111/// arrays start with none.
112///
113/// **A store is linked only once it needs to be.** While no two of its
114/// entries share a code every chain is one entry long, so its codes array
115/// holds the code alone, one field per entry, and a table of distinct keys
116/// pays nothing for links. The first append that joins an existing chain
117/// gives the store a linked codes array ([`relink`]), and nothing else; a
118/// rebuild keeps the layout it finds, and `EMPTY` installs an unlinked one.
119/// Which layout an array has is read off its length: a linked one is at
120/// least as long as the slot index, an unlinked one shorter.
121const FIELDS: usize = 3;
122/// The entry's [`code_of`], `None` for a hole.
123const CODE: usize = 0;
124/// The next entry of the chain, `None` for its newest.
125const NEXT: usize = 1;
126/// At a chain's oldest entry, the chain's newest; `None` elsewhere.
127const TAIL: usize = 2;

129/// The fewest slots a store has.
130const MINIMUM_SLOTS: usize = 8;

132/// The most slots a store is sized at.
133const MAXIMUM_SLOTS: usize = 1 << 31;

135/// The smallest used span worth rebuilding on removal.
136const MINIMUM_COMPACT_SPAN: usize = 128;

138/// Used and live both fit in 31 bits, since an entry array holds fewer than
139/// MAXIMUM_SLOTS entries. Use the entire signed SmallInt range to keep the
140/// counts together in one pool entry.
141#[derive(Clone, Copy, PartialEq, Eq, Debug)]
142struct Counts {
143    used: usize,
144    live: usize,
145}
146
147impl Counts {
148    const BITS: u32 = 31;
149    const MASK: i64 = (1 << Self::BITS) - 1;
150    const BIAS: i64 = 1 << (2 * Self::BITS - 1);
151
152    fn encode(self) -> ObjRef {
153        debug_assert!(self.live <= self.used && self.used < MAXIMUM_SLOTS);
154        small((count_value(self.used) << Self::BITS | count_value(self.live)) - Self::BIAS)
155    }
156
157    fn decode(value: ObjRef) -> Option<Self> {
158        let Decoded::SmallInt(value) = value.decode() else {
159            return None;
160        };
161        let packed = value + Self::BIAS;
162        Some(Self {
163            used: stored_count(packed >> Self::BITS),
164            live: stored_count(packed & Self::MASK),
165        })
166    }
167}

169/// A position or count of a store as the small integer it is kept as. Both
170/// are below [`MAXIMUM_SLOTS`].
171fn count_value(count: usize) -> i64 {
172    i64::try_from(count).expect("below MAXIMUM_SLOTS")
173}

175/// A position or count [`count_value`] kept, back as a `usize`.
176fn stored_count(value: i64) -> usize {
177    usize::try_from(value).expect("a position or count this store kept")
178}

180/// The entry capacity of a store of `slots` slots: two thirds, so that the
181/// slots, which hold at most one mark per used entry position, never fill
182/// and every probe ends at an empty one.
183fn capacity_of(slots: usize) -> usize {
184    slots * 2 / 3
185}
186
187/// The slot count of a store sized to hold `entries` entries.
188fn slots_for(entries: usize) -> usize {
189    let mut slots = MINIMUM_SLOTS;
190    while capacity_of(slots) < entries && slots < MAXIMUM_SLOTS {
191        slots *= 2;
192    }
193    slots
194}

196/// The capacity a store is built at when nothing asks for more, which
197/// [`slots_for`] gives the fewest slots.
198pub(super) const MINIMUM_ENTRIES: usize = 0;

200/// The capacity a rebuild gives a store holding `live` entries in `used`
201/// positions: room for as many appends again before the next rebuild, or,
202/// when more of the positions are holes than entries, for three times as
203/// many. The capacity is a multiple of `live` that only `used` chooses, so
204/// the number of removals a store has seen does not grow it: a store that
205/// churns through few live entries stays small however many it has removed.
206fn regrown(live: usize, used: usize) -> usize {
207    // A store whose removals keep pace with its appends would otherwise be
208    // rebuilt after as many appends as it has entries, and a rebuild costs
209    // far more than an append: the larger room spreads it over more appends.
210    if used.saturating_sub(live) > live {
211        live.saturating_mul(4)
212    } else {
213        live.saturating_mul(2)
214    }
215}

217/// The form a hash takes in an entry's cache and in the probe sequence:
218/// mixed, so that the low bits that pick the first slot depend on every bit
219/// of the hash, and cut to what a SmallInt holds. Equal hashes give equal
220/// codes, which is all a comparison of codes relies on.
221pub(super) fn code_of(hash: u64) -> u64 {
222    let mixed = hash.wrapping_mul(0x9E37_79B9_7F4A_7C15);
223    (mixed ^ (mixed >> 29)) & SMALL_INT_MAX.unsigned_abs()
224}

226/// The slots a code probes, in order: a perturbed sequence over a power of
227/// two slots, which takes every bit of the code into account and, once the
228/// perturbation has shifted out, visits every slot.
229#[derive(Clone, Copy)]
230pub(super) struct Sequence {
231    mask: usize,
232    at: usize,
233    perturb: u64,
234}
235
236impl Sequence {
237    pub(super) fn new(code: u64, slots: usize) -> Sequence {
238        let mask = slots - 1;
239        Sequence {
240            mask,
241            at: usize::cast_truncate(code) & mask,
242            perturb: code,
243        }
244    }
245
246    /// The slot this step of the sequence reads.
247    pub(super) fn at(&self) -> usize {
248        self.at
249    }
250
251    pub(super) fn advance(&mut self) {
252        self.perturb >>= 5;
253        self.at = self
254            .at
255            .wrapping_mul(5)
256            .wrapping_add(1)
257            .wrapping_add(usize::cast_truncate(self.perturb))
258            & self.mask;
259    }
260}

262/// What one slot holds.
263#[derive(Clone, Copy, PartialEq, Eq, Debug)]
264pub(super) enum Mark {
265    Empty,
266    Deleted,
267    Entry(usize),
268}

270/// The mark a slot array holds at `slot`. A slot array that is not one
271/// reads as empty everywhere.
272pub(super) fn mark_in(slots: &[Option<ObjRef>], slot: usize) -> Mark {
273    match slots.get(slot).copied().flatten().map(ObjRef::decode) {
274        Some(Decoded::SmallInt(DELETED)) => Mark::Deleted,
275        Some(Decoded::SmallInt(value)) if value > EMPTY => Mark::Entry(stored_count(value) - 1),
276        _ => Mark::Empty,
277    }
278}

280/// A store's codes array, read in its layout.
281#[derive(Clone, Copy)]
282pub(super) struct Codes<'a> {
283    fields: &'a [Option<ObjRef>],
284    /// Fields per entry: [`FIELDS`] when linked, else one.
285    stride: usize,
286}
287
288impl<'a> Codes<'a> {
289    /// The codes array `fields` of a store with `slot_count` slots.
290    #[inline]
291    pub(super) fn new(fields: &'a [Option<ObjRef>], slot_count: usize) -> Codes<'a> {
292        // An unlinked array has one field per entry position, fewer than the
293        // slots. A linked one has [`FIELDS`], at least as many as the slots.
294        let stride = if fields.len() >= slot_count {
295            FIELDS
296        } else {
297            1
298        };
299        Codes { fields, stride }
300    }
301
302    #[inline]
303    pub(super) fn linked(&self) -> bool {
304        self.stride == FIELDS
305    }
306
307    #[inline]
308    fn field(&self, position: usize, field: usize) -> Option<i64> {
309        if field >= self.stride {
310            return None;
311        }
312        match self
313            .fields
314            .get(position * self.stride + field)
315            .copied()
316            .flatten()
317            .map(ObjRef::decode)
318        {
319            Some(Decoded::SmallInt(value)) if value >= 0 => Some(value),
320            _ => None,
321        }
322    }
323
324    /// The code cached for the entry at a position, `None` for a hole.
325    #[inline]
326    pub(super) fn code(&self, position: usize) -> Option<u64> {
327        self.field(position, CODE).map(i64::cast_unsigned)
328    }
329
330    /// The entry after the one at `position` in its chain, `None` for the
331    /// newest.
332    #[inline]
333    pub(super) fn next(&self, position: usize) -> Option<usize> {
334        self.field(position, NEXT).map(stored_count)
335    }
336
337    /// The newest entry of the chain whose oldest is at `head`.
338    #[inline]
339    fn tail(&self, head: usize) -> usize {
340        self.field(head, TAIL).map_or(head, stored_count)
341    }
342
343    /// The positions of the chain whose oldest entry is at `head`, oldest
344    /// first.
345    pub(super) fn chain(self, head: usize) -> impl Iterator<Item = usize> + 'a {
346        std::iter::successors(Some(head), move |&position| self.next(position))
347    }
348}

350/// Where the chain of `code` is: its slot and the position of its oldest
351/// entry. When the store holds no entry of that code, `Err` with the slot a
352/// new chain takes: the first deleted one the probe passed, or the empty one
353/// it ended at.
354#[inline]
355pub(super) fn chain_in(
356    slots: &[Option<ObjRef>],
357    codes: Codes<'_>,
358    code: u64,
359) -> Result<(usize, usize), usize> {
360    let mut sequence = Sequence::new(code, slots.len());
361    let mut free = None;
362    loop {
363        match mark_in(slots, sequence.at()) {
364            Mark::Empty => return Err(free.unwrap_or(sequence.at())),
365            Mark::Deleted => {
366                free.get_or_insert(sequence.at());
367            }
368            Mark::Entry(head) => {
369                if codes.code(head) == Some(code) {
370                    return Ok((sequence.at(), head));
371                }
372            }
373        }
374        sequence.advance();
375    }
376}

378/// Where a probe found the chain of a code: the slot that files it, or, when
379/// there was none, the slot a new chain takes; and whether the store is
380/// linked. One word, the slot above the linked bit, since every probe
381/// answers one and most never use it.
382#[derive(Clone, Copy)]
383pub(super) struct Place(usize);
384
385impl Place {
386    #[inline]
387    pub(super) fn of(chain: Result<(usize, usize), usize>, codes: Codes<'_>) -> Place {
388        let (Ok((slot, _)) | Err(slot)) = chain;
389        Place(slot << 1 | usize::from(codes.linked()))
390    }
391
392    #[inline]
393    fn slot(self) -> usize {
394        self.0 >> 1
395    }
396
397    #[inline]
398    fn stride(self) -> usize {
399        if self.0 & 1 == 1 { FIELDS } else { 1 }
400    }
401}

403/// A store's codes array and slot index, read together.
404#[inline]
405pub(super) fn codes_of<'a>(
406    interp: &'a Interp,
407    store: &Store,
408) -> Result<(&'a [Option<ObjRef>], Codes<'a>), Failure> {
409    let slots = array_slots(interp, store.slots)?;
410    let codes = Codes::new(array_slots(interp, store.codes)?, slots.len());
411    Ok((slots, codes))
412}

420/// A hash store's arrays and its count, read out together.
421#[derive(Clone, Copy, PartialEq, Eq, Debug)]
422pub(super) struct Store {
423    pub(super) indexes: ObjRef,
424    pub(super) items: ObjRef,
425    pub(super) codes: ObjRef,
426    pub(super) slots: ObjRef,
427    /// Entry positions used, holes included: where the next entry goes.
428    pub(super) used: usize,
429    live: usize,

432impl Store {
433    /// Entry positions the arrays have. Read off the indexes array when
434    /// asked, since only an append and a relink need it.
435    pub(super) fn capacity(&self, interp: &Interp) -> Result<usize, Failure> {
436        Ok(array_slots(interp, self.indexes)?.len())
437    }
438
439    /// Whether `other` is these same arrays, whatever their count: the same
440    /// positions holding the same entries. The codes array is left out,
441    /// since [`relink`] replaces it alone and moves no entry.
442    pub(super) fn same_arrays(&self, other: &Store) -> bool {
443        self.indexes == other.indexes && self.items == other.items && self.slots == other.slots
444    }
445
446    /// Whether an in-flight probe can continue at the same position. Removing
447    /// another entry changes the count and links, but keeps the positions.
448    pub(super) fn same_layout(&self, other: &Store) -> bool {
449        self.same_arrays(other) && self.codes == other.codes && self.used == other.used
450    }
451}

496    capacity: usize,

498    build(interp, receiver, half, &[], capacity, false)
499}

501/// Builds a store holding `entries` -- (index, item, code) in their order --
502/// sized for at least `capacity`, and binds it to `receiver` in place of
503/// whatever it had. An unlinked one is built only for entries of distinct
504/// codes, which is what an unlinked store holds.
505fn build(
506    interp: &mut Interp,
507    receiver: ObjRef,
508    half: Half,
509    entries: &[(ObjRef, Option<ObjRef>, u64)],
510    capacity: usize,
511    linked: bool,
512) -> Store {
513    let slot_count = slots_for(capacity.max(entries.len()));
514    let capacity = capacity_of(slot_count);
515    let stride = if linked { FIELDS } else { 1 };
516    let scope = hash_scope(interp);
517    let mut index_slots = vec![None; capacity];
518    let mut item_slots = vec![None; capacity];
519    let mut code_slots = vec![None; capacity * stride];
520    let mut marks = vec![Some(small(EMPTY)); slot_count];
521    for (position, &(index, item, code)) in entries.iter().enumerate() {
522        index_slots[position] = Some(index);
523        item_slots[position] = item;
524        code_slots[position * stride + CODE] = Some(small(code.cast_signed()));
525        if !linked {
526            // Every code is a chain of its own: the entry takes the first
527            // empty slot of its sequence.
528            debug_assert!(
529                chain_in(&marks, Codes::new(&code_slots, slot_count), code).is_err(),
530                "an unlinked store holds one entry per code"
531            );
532            let mut sequence = Sequence::new(code, slot_count);
533            while mark_in(&marks, sequence.at()) != Mark::Empty {
534                sequence.advance();
535            }
536            marks[sequence.at()] = Some(small(count_value(position) + 1));
537            continue;
538        }
539        let codes = Codes::new(&code_slots, slot_count);
540        match chain_in(&marks, codes, code) {
541            Ok((_, head)) => {
542                let tail = codes.tail(head);
543                code_slots[tail * stride + NEXT] = Some(small(count_value(position)));
544                code_slots[head * stride + TAIL] = Some(small(count_value(position)));
545            }
546            Err(slot) => marks[slot] = Some(small(count_value(position) + 1)),
547        }
548    }
549    // Each array is rooted as soon as it exists: the next allocation may
550    // collect. The entries copied in stay reachable through the arrays still
551    // bound until the new ones replace them.
552    let indexes = interp.alloc_charged(BehaviourId::ARRAY, Body::array(index_slots));
553    interp.roots.activity_mut().push_temp(indexes);
554    let items = interp.alloc_charged(BehaviourId::ARRAY, Body::array(item_slots));
555    interp.roots.activity_mut().push_temp(items);
556    let codes = interp.alloc_charged(BehaviourId::ARRAY, Body::array(code_slots));
557    interp.roots.activity_mut().push_temp(codes);
558    let slots = interp.alloc_charged(BehaviourId::ARRAY, Body::array(marks));
559    interp.roots.activity_mut().push_temp(slots);
560    let used = entries.len();
561    let counts = Counts { used, live: used }.encode();
562    let [n_indexes, n_items, n_codes, n_slots, n_used] = pool_names(interp, half).clone();
563    // A capacity [`initialize_store`] left pending goes, so that the half's
564    // entries are bound together in their order, as [`shared_run`] reads them.
565    if pool_variable(interp, receiver, scope, half.indexes).is_none()
566        && pool_variable(interp, receiver, scope, half.slots).is_some()
567    {
568        interp.clear_pool_variable(receiver, scope, half.slots);
569    }
570    interp.set_pool_variables(
571        receiver,
572        scope,
573        &[
574            (&n_indexes, indexes),
575            (&n_items, items),
576            (&n_codes, codes),
577            (&n_slots, slots),
578            (&n_used, counts),
579        ],
580    );
581    bump_store_generation(interp);
582    Store {
583        indexes,
584        items,
585        codes,
586        slots,
587        used,
588        live: used,
589    }
590}

592/// The receiver's store, or `None` when it has none yet. Builds nothing, which
593/// [`store_in`] does.
594pub(super) fn read_store(
595    interp: &mut Interp,
596    receiver: ObjRef,
597    half: Half,
598) -> Result<Option<Store>, Failure> {
599    let parts = read_parts(interp, receiver);
600    let found = if half == CONTENTS {
601        parts.contents
602    } else {
603        parts.methods
604    };
605    Ok(found.store())
606}

614    /// The slot array, or before the store is built the capacity
615    /// [`keep_capacity`] left for it.
616    pub(super) slots: Option<ObjRef>,
617    counts: Option<Counts>,
618}

620impl Found {
621    /// The store these entries make, or `None` when any is missing.
622    pub(super) fn store(self) -> Option<Store> {
623        let (Some(indexes), Some(items), Some(codes), Some(slots), Some(counts)) = (
624            self.indexes,
625            self.items,
626            self.codes,
627            self.slots,
628            self.counts,
629        ) else {
630            return None;
631        };
632        Some(Store {
633            indexes,
634            items,
635            codes,
636            slots,
637            used: counts.used,
638            live: counts.live,
639        })
640    }
641
642    /// The capacity a store not built yet is to be built at, if one was
643    /// kept.
644    pub(super) fn kept_capacity(self) -> Option<usize> {
645        self.slots.and_then(counted)
646    }
647}

649/// Both halves' entries and the unknown method, from one pass over the pool.
650pub(super) struct Parts {
651    pub(super) contents: Found,
652    pub(super) methods: Found,
653    pub(super) unknown: Option<ObjRef>,
654}
655
656pub(super) fn read_parts(interp: &mut Interp, receiver: ObjRef) -> Parts {
657    let scope = hash_scope(interp);
658    let mut parts = Parts {
659        contents: Found::default(),
660        methods: Found::default(),
661        unknown: None,
662    };
663    let Some(Body::Instance { pools, .. }) = interp.heap.get(receiver).map(|object| &object.body)
664    else {
665        return parts;
666    };
667    let names = &interp.hash_names;
668    let mut rest = pools.entries(scope);
669    while !rest.is_empty() {
670        // [`install_store`] binds a half's names together, in this order
671        // and as copies of [`PoolNames`]' own, so a run of them is recognised
672        // by identity before any name is compared byte by byte.
673        if let Some(found) = shared_run(rest, &names.contents) {
674            parts.contents = found;
675            rest = &rest[5..];
676            continue;
677        }
678        if let Some(found) = shared_run(rest, &names.methods) {
679            parts.methods = found;
680            rest = &rest[5..];
681            continue;
682        }
683        let PoolEntry { name, value, .. } = &rest[0];
684        rest = &rest[1..];
685        let value = *value;
686        // [`keep_capacity`] binds the contents' slot name alone, as a copy of
687        // [`PoolNames`]' own. Recognising it by identity only skips the byte
688        // comparisons below, which would read it as `HASH_BUCKETS` all the same.
689        if Rc::ptr_eq(name, &names.contents[3]) {
690            parts.contents.slots = Some(value);
691            continue;
692        }
693        match name.as_ref() {
694            HASH_INDEXES => parts.contents.indexes = Some(value),
695            HASH_ITEMS => parts.contents.items = Some(value),
696            HASH_NEXT => parts.contents.codes = Some(value),
697            HASH_BUCKETS => parts.contents.slots = Some(value),
698            HASH_FREE => parts.contents.counts = Counts::decode(value),
699            METHOD_INDEXES => parts.methods.indexes = Some(value),
700            METHOD_ITEMS => parts.methods.items = Some(value),
701            METHOD_NEXT => parts.methods.codes = Some(value),
702            METHOD_BUCKETS => parts.methods.slots = Some(value),
703            METHOD_FREE => parts.methods.counts = Counts::decode(value),
704            UNKNOWN_METHOD => parts.unknown = Some(value).filter(|held| *held != ObjRef::NIL),
705            _ => {}
706        }
707    }
708    parts
709}

711/// The contents' entries when they are the first five the hash scope binds,
712/// where [`install_store`] puts them in a store built before anything else
713/// is bound in that scope: [`read_parts`]'s first step, without the rest of its pass.
714#[inline]
715pub(super) fn leading_contents(interp: &mut Interp, receiver: ObjRef) -> Option<Found> {
716    let scope = hash_scope(interp);
717    let Some(Body::Instance { pools, .. }) = interp.heap.get(receiver).map(|object| &object.body)
718    else {
719        return None;
720    };
721    shared_run(pools.leading::<5>(scope)?, &interp.hash_names.contents)
722}
723
724/// One half's entries, when `entries` starts with that half's shared
725/// names in [`install_store`]'s order. A pool binds a name once, so what this
726/// finds is what the byte-by-byte reading would.
727#[inline]
728pub(super) fn shared_run(entries: &[PoolEntry], names: &[Rc<[u8]>; 5]) -> Option<Found> {
729    let [indexes, items, codes, slots, used, ..] = entries else {
730        return None;
731    };
732    let [indexes_at, items_at, codes_at, slots_at, used_at] = names;
733    (Rc::ptr_eq(&indexes.name, indexes_at)
734        && Rc::ptr_eq(&items.name, items_at)
735        && Rc::ptr_eq(&codes.name, codes_at)
736        && Rc::ptr_eq(&slots.name, slots_at)
737        && Rc::ptr_eq(&used.name, used_at))
738    .then(|| Found {
739        indexes: Some(indexes.value),
740        items: Some(items.value),
741        codes: Some(codes.value),
742        slots: Some(slots.value),
743        counts: Counts::decode(used.value),
744    })
745}

747/// The store `half` names, installing an empty one if it is not there yet.
748pub(super) fn store_in(
749    interp: &mut Interp,
750    receiver: ObjRef,
751    half: Half,
752) -> Result<Store, Failure> {
753    if half == CONTENTS {
754        return store_of(interp, receiver);
755    }
756    match read_store(interp, receiver, half)? {
757        Some(store) => Ok(store),
758        None => Ok(install_store(interp, receiver, half, MINIMUM_ENTRIES)),
759    }
760}
761
762pub(super) fn store_of(interp: &mut Interp, receiver: ObjRef) -> Result<Store, Failure> {
763    if let Some(found) = leading_contents(interp, receiver)
764        && let Some(store) = found.store()
765    {
766        return Ok(store);
767    }
768    let parts = read_parts(interp, receiver);
769    if let Some(store) = parts.contents.store() {
770        return Ok(store);
771    }
772    refuse_storeless(interp, receiver)?;
773    // The capacity [`initialize_store`] left pending, if it left one.
774    let capacity = parts.contents.kept_capacity().unwrap_or(MINIMUM_ENTRIES);
775    Ok(install_store(interp, receiver, CONTENTS, capacity))
776}

778/// The contents' store, or `None` for a receiver [`store_of`] would build an
779/// empty one for: what a read that finds nothing in an empty store answers
780/// the same without building it. Refuses as [`store_of`] does.
781pub(super) fn store_if_built(
782    interp: &mut Interp,
783    receiver: ObjRef,
784) -> Result<Option<Store>, Failure> {
785    if let Some(found) = leading_contents(interp, receiver)
786        && let Some(store) = found.store()
787    {
788        return Ok(Some(store));
789    }
790    let parts = read_parts(interp, receiver);
791    if let Some(store) = parts.contents.store() {
792        return Ok(Some(store));
793    }
794    refuse_storeless(interp, receiver)?;
795    if parts.contents.slots.is_none() {
796        keep_capacity(interp, receiver, MINIMUM_ENTRIES);
797    }
798    Ok(None)
799}

801/// Leaves `capacity` as the size [`store_of`] builds `receiver`'s store at,
802/// for a receiver that has neither a store nor a size kept for one. Taking
803/// the place of building the store, it is what makes a later `INIT` find the
804/// size already decided, as it would find a store built.
805pub(super) fn keep_capacity(interp: &mut Interp, receiver: ObjRef, capacity: usize) {
806    let scope = hash_scope(interp);
807    let capacity = interp.counted(capacity);
808    let [_, _, _, name, _] = &interp.hash_names.contents;
809    let name = Rc::clone(name);
810    interp.set_pool_variables(receiver, scope, &[(&name, capacity)]);
811}

813/// [`store_of`]'s refusal of a receiver that has no store and is not given
814/// one.
815pub(super) fn refuse_storeless(interp: &mut Interp, receiver: ObjRef) -> Result<(), Failure> {
816    // **A `Body::Native` receiver is not this store's**, and the check has to
817    // be here because the two are one method. `IdentityTable`'s `At`, `Put` and
818    // `[]` rows are donated to `StringTable` and then `StringTable`'s whole set
819    // to `Directory`, so `Table`, `IdentityTable`, `StringTable` and `Directory`
820    // share one method identity and one body -- which is why registering these
821    // names for `Table` alone reached a `StringTable` during the bootstrap. `Directory`, `StringTable` and
822    // `Properties` keep the entry map they already have until the task that
823    // owns them.
824    if matches!(
825        interp.heap.get(receiver).map(|object| &object.body),
826        Some(Body::Native(_))
827    ) || !holds_hash_store(interp, receiver)
828    {
829        return Err(crate::Raised::receiver_kind("a value that is not a hash collection").into());
830    }
831    Ok(())
832}

922/// Compacts the live entries of `store`, the receiver's, in their order, into
923/// fresh arrays of the requested capacity, and files them again from their
924/// cached codes. Sends nothing. The store stays linked or unlinked as it was.
925fn rebuild(
926    interp: &mut Interp,
927    receiver: ObjRef,
928    half: Half,
929    store: &Store,
930    capacity: usize,
931) -> Result<Store, Failure> {
932    let indexes = array_slots(interp, store.indexes)?;
933    let items = array_slots(interp, store.items)?;
934    let (_, codes) = codes_of(interp, store)?;
935    let linked = codes.linked();
936    let mut live = Vec::with_capacity(store.live);
937    for (position, index) in indexes.iter().enumerate().take(store.used) {
938        if let Some(index) = *index
939            && let Some(code) = codes.code(position)
940        {
941            live.push((index, items.get(position).copied().flatten(), code));
942        }
943    }
944    debug_assert_eq!(live.len(), store.live);
945    Ok(build(interp, receiver, half, &live, capacity, linked))
946}

948/// Adds an entry after every other, without looking for an index the store
949/// already holds -- the append half of `put`, and the whole of a multi-value
950/// `put`. `code` is the index's [`code_of`]. A full store is rebuilt first.
951///
952/// `known` is the receiver's store as it is now, as a probe answered it, and
953/// where that probe last found the chain of `code` ([`Probe::at`]).
954pub(super) fn append(
955    interp: &mut Interp,
956    receiver: ObjRef,
957    half: Half,
958    index: ObjRef,
959    item: Option<ObjRef>,
960    code: u64,
961    known: Option<(Store, Place)>,
962) -> Result<(), Failure> {
963    let (mut store, mut known) = match known {
964        Some((store, place)) => {
965            // `read_store`, not `store_in`: the check must not build a store.
966            debug_assert!(
967                read_store(interp, receiver, half).ok().flatten() == Some(store),
968                "a known store is the receiver's current one"
969            );
970            (store, Some(place))
971        }
972        None => (store_in(interp, receiver, half)?, None),
973    };
974    if store.used >= store.capacity(interp)? {
975        store = rebuild(
976            interp,
977            receiver,
978            half,
979            &store,
980            regrown(store.live, store.used),
981        )?;
982        known = None;
983    }
984    let (mut chain, mut stride) = chain_for_append(interp, &store, code, known)?;
985    if chain.is_ok() && stride != FIELDS {
986        // The first entry to join a chain: the store becomes linked. Its
987        // slots and positions stay, so the chain found stays where it is.
988        store = relink(interp, receiver, half, &store)?;
989        (chain, stride) = chain_for_append(interp, &store, code, None)?;
990    }
991    let position = store.used;
992    write_slot(interp, store.indexes, position, Some(index));
993    write_slot(interp, store.items, position, item);
994    // A code is wider than a position on a 32-bit target, so it is written
995    // as the number it is rather than through [`write_field`].
996    write_slot(
997        interp,
998        store.codes,
999        position * stride + CODE,
1000        Some(small(code.cast_signed())),
1001    );
1002    match chain {
1003        Ok((head, tail)) => {
1004            write_field(interp, &store, stride, tail, NEXT, Some(position));
1005            write_field(interp, &store, stride, head, TAIL, Some(position));
1006        }
1007        Err(slot) => write_slot(
1008            interp,
1009            store.slots,
1010            slot,
1011            Some(small(count_value(position) + 1)),
1012        ),
1013    }
1014    set_counts(
1015        interp,
1016        receiver,
1017        half,
1018        Counts {
1019            used: position + 1,
1020            live: store.live + 1,
1021        },
1022    );
1023    Ok(())
1024}

1026/// Gives an unlinked store a linked codes array holding the same codes, and
1027/// keeps its other arrays: every chain of an unlinked store is one entry
1028/// long, so no slot or position moves and there are no links to copy.
1029fn relink(
1030    interp: &mut Interp,
1031    receiver: ObjRef,
1032    half: Half,
1033    store: &Store,
1034) -> Result<Store, Failure> {
1035    let mut fields = vec![None; store.capacity(interp)? * FIELDS];
1036    {
1037        let (_, codes) = codes_of(interp, store)?;
1038        for position in 0..store.used {
1039            if let Some(code) = codes.code(position) {
1040                fields[position * FIELDS + CODE] = Some(small(code.cast_signed()));
1041            }
1042        }
1043    }
1044    let codes = interp.alloc_charged(BehaviourId::ARRAY, Body::array(fields));
1045    interp.roots.activity_mut().push_temp(codes);
1046    let scope = hash_scope(interp);
1047    interp.set_pool_variable(receiver, scope, half.codes, codes);
1048    bump_store_generation(interp);
1049    Ok(Store { codes, ..*store })
1050}
1051
1052/// What [`append`] joins for `code`: the oldest and newest entry of its
1053/// chain, or the slot a new chain takes; and the store's stride.
1054///
1055/// `known`'s slot is read again rather than trusted: the probe's
1056/// comparisons may have removed entries since it walked the slots, which
1057/// moves a chain's oldest entry or marks its slot deleted, though, without
1058/// an append, it neither fills a free slot nor starts a chain nor changes
1059/// the stride. So a free slot there is where the new chain goes, and an
1060/// entry of `code` there heads its chain; only another code's entry sends
1061/// the walk round again.
1062#[inline]
1063fn chain_for_append(
1064    interp: &Interp,
1065    store: &Store,
1066    code: u64,
1067    known: Option<Place>,
1068) -> Result<(Result<(usize, usize), usize>, usize), Failure> {
1069    let slots = array_slots(interp, store.slots)?;
1070    if let Some(place) = known
1071        && matches!(mark_in(slots, place.slot()), Mark::Empty | Mark::Deleted)
1072    {
1073        return Ok((Err(place.slot()), place.stride()));
1074    }
1075    let codes = Codes::new(array_slots(interp, store.codes)?, slots.len());
1076    let reread = known.and_then(|place| match mark_in(slots, place.slot()) {
1077        Mark::Entry(head) if codes.code(head) == Some(code) => Some(head),
1078        _ => None,
1079    });
1080    let chain = match reread {
1081        Some(head) => Ok(head),
1082        None => chain_in(slots, codes, code).map(|(_, head)| head),
1083    };
1084    Ok((chain.map(|head| (head, codes.tail(head))), codes.stride))
1085}

1087/// Writes one of the fields of the entry at `position`, in a codes array of
1088/// `stride` fields per entry.
1089fn write_field(
1090    interp: &mut Interp,
1091    store: &Store,
1092    stride: usize,
1093    position: usize,
1094    field: usize,
1095    value: Option<usize>,
1096) {
1097    debug_assert!(field < stride, "an unlinked store has no links to write");
1098    let value = value.map(|value| small(count_value(value)));
1099    write_slot(interp, store.codes, position * stride + field, value);
1100}

1102/// How taking the entry at a position out relinks its chain.
1103enum Unlink {
1104    /// It was the oldest: its slot names the next one, which takes over the
1105    /// chain's tail, or is marked deleted when it was the only one.
1106    Head {
1107        slot: usize,
1108        next: Option<usize>,
1109        tail: usize,
1110    },
1111    /// It followed `previous`, which now links past it, and the chain's
1112    /// oldest, at `head`, names `previous` as its tail if it was the tail.
1113    After {
1114        head: usize,
1115        previous: usize,
1116        next: Option<usize>,
1117        was_tail: bool,
1118    },
1119}
1120
1121/// Takes the entry at `position` out: a hole in the entry arrays, and its
1122/// chain relinked past it. Answers the item it held.
1123pub(super) fn remove_entry(
1124    interp: &mut Interp,
1125    receiver: ObjRef,
1126    half: Half,
1127    store: &mut Store,
1128    position: usize,
1129) -> Result<Option<ObjRef>, Failure> {
1130    let item = slot_at(interp, store.items, position)?;
1131    let (unlink, stride) = {
1132        let (slots, codes) = codes_of(interp, store)?;
1133        let unlink = codes
1134            .code(position)
1135            .and_then(|code| chain_in(slots, codes, code).ok())
1136            .and_then(|(slot, head)| {
1137                let next = codes.next(position);
1138                if head == position {
1139                    return Some(Unlink::Head {
1140                        slot,
1141                        next,
1142                        tail: codes.tail(head),
1143                    });
1144                }
1145                let previous = codes
1146                    .chain(head)
1147                    .find(|&at| codes.next(at) == Some(position))?;
1148                Some(Unlink::After {
1149                    head,
1150                    previous,
1151                    next,
1152                    was_tail: codes.tail(head) == position,
1153                })
1154            });
1155        (unlink, codes.stride)
1156    };
1157    match unlink {
1158        Some(Unlink::Head {
1159            slot,
1160            next: Some(next),
1161            tail,
1162        }) => {
1163            write_slot(
1164                interp,
1165                store.slots,
1166                slot,
1167                Some(small(count_value(next) + 1)),
1168            );
1169            write_field(interp, store, stride, next, TAIL, Some(tail));
1170        }
1171        Some(Unlink::Head {
1172            slot, next: None, ..
1173        }) => {
1174            write_slot(interp, store.slots, slot, Some(small(DELETED)));
1175        }
1176        Some(Unlink::After {
1177            head,
1178            previous,
1179            next,
1180            was_tail,
1181        }) => {
1182            write_field(interp, store, stride, previous, NEXT, next);
1183            if was_tail {
1184                write_field(interp, store, stride, head, TAIL, Some(previous));
1185            }
1186        }
1187        None => {}
1188    }
1189    write_slot(interp, store.indexes, position, None);
1190    write_slot(interp, store.items, position, None);
1191    for field in 0..stride {
1192        write_field(interp, store, stride, position, field, None);
1193    }
1194    store.live -= 1;
1195    set_counts(
1196        interp,
1197        receiver,
1198        half,
1199        Counts {
1200            used: store.used,
1201            live: store.live,
1202        },
1203    );
1204    Ok(item)
1205}

1207/// Rebuild a sparse span after its caller has finished using positions in
1208/// these arrays. Shrink without removing the room for appends that amortizes
1209/// the next rebuild.
1210pub(super) fn compact_after_removal(
1211    interp: &mut Interp,
1212    receiver: ObjRef,
1213    half: Half,
1214    store: &Store,
1215    removed: Option<ObjRef>,
1216) -> Result<(), Failure> {
1217    if store.used >= MINIMUM_COMPACT_SPAN && store.used - store.live > store.live {
1218        if let Some(item) = removed {
1219            interp.roots.activity_mut().push_temp(item);
1220        }
1221        rebuild(interp, receiver, half, store, store.live * 2)?;
1222    }
1223    Ok(())
1224}

1226/// Empties `half` in place, and then gives it fresh arrays.
1227pub(super) fn clear(interp: &mut Interp, receiver: ObjRef, half: Half) -> Result<(), Failure> {
1228    let store = store_in(interp, receiver, half)?;
1229    let stride = codes_of(interp, &store)?.1.stride;
1230    // Active lookups retain these contents through callbacks and growth. EMPTY
1231    // clears that contents in place, and then installs fresh arrays, so that
1232    // no position of these arrays is ever filled a second time.
1233    for position in 0..store.used.min(store.capacity(interp)?) {
1234        write_slot(interp, store.indexes, position, None);
1235        write_slot(interp, store.items, position, None);
1236        for field in 0..stride {
1237            write_field(interp, &store, stride, position, field, None);
1238        }
1239    }
1240    let slot_count = array_slots(interp, store.slots)?.len();
1241    for slot in 0..slot_count {
1242        write_slot(interp, store.slots, slot, Some(small(EMPTY)));
1243    }
1244    install_store(interp, receiver, half, MINIMUM_ENTRIES);
1245    Ok(())
1246}

1248/// Replaces the value under an index the table already holds, keeping its
1249/// place, and otherwise appends a new entry.
1250pub(super) fn insert(
1251    interp: &mut Interp,
1252    receiver: ObjRef,
1253    index: ObjRef,
1254    item: Option<ObjRef>,
1255) -> Result<(), Failure> {
1256    // A put replaces any method of the same name. Measured: `setMethod('M',
1257    // 'return 2')` then `d['M'] = 5` answers 5, and `unsetMethod('M')`
1258    // afterwards still answers 5.
1259    if method_store(interp, receiver)?.is_some() {
1260        take_in(interp, receiver, METHODS, index)?;
1261    }
1262    insert_in(interp, receiver, CONTENTS, index, item)
1263}
1264
1265/// [`insert`] into either half of a `Directory`.
1266pub(super) fn insert_in(
1267    interp: &mut Interp,
1268    receiver: ObjRef,
1269    half: Half,
1270    index: ObjRef,
1271    item: Option<ObjRef>,
1272) -> Result<(), Failure> {
1273    // `.local['OUTPUT'] = x` moves a route's far end.
1274    interp.bump_route_generation();
1275    let store = store_in(interp, receiver, half)?;
1276    let (store, found) = probe_in(interp, receiver, half, store, index)?;
1277    let place = match found.at {
1278        Ok(position) => {
1279            write_slot(interp, store.items, position, item);
1280            return Ok(());
1281        }
1282        Err(place) => place,
1283    };
1284    // The probe sent nothing after its last look at the store, so the store
1285    // it answered is the receiver's, the index is still absent and its chain
1286    // where the probe found it; the append sends nothing.
1287    append(
1288        interp,
1289        receiver,
1290        half,
1291        index,
1292        item,
1293        found.code,
1294        Some((store, place)),
1295    )
1296}

1298/// A multi-value `put`: another entry under `index`, after every other,
1299/// whatever the store already holds under it.
1300pub(super) fn insert_duplicate(
1301    interp: &mut Interp,
1302    receiver: ObjRef,
1303    index: ObjRef,
1304    item: Option<ObjRef>,
1305) -> Result<(), Failure> {
1306    let keys = keys_of(interp, receiver);
1307    let code = code_of(hash_of(interp, keys, index)?);
1308    append(interp, receiver, CONTENTS, index, item, code, None)
1309}
