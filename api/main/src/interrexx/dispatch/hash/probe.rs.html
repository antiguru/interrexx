3/// Where `index` is: the position of the oldest live entry under it, or,
4/// when there is none, where the chain of its code was when the probe last
5/// walked the slots of the store it answers with; and the index's code,
6/// which an append files a new entry under. An append that follows with
7/// nothing sent in between starts from that place rather than walking the
8/// slots again.
9pub(super) struct Probe {
10    pub(super) at: Result<usize, Place>,
11    pub(super) code: u64,
12}
13
14impl Probe {
15    /// The position of the oldest live entry under the index, if any.
16    #[inline]
17    pub(super) fn found(&self) -> Option<usize> {
18        self.at.ok()
19    }
20}

22pub(super) fn probe(
23    interp: &mut Interp,
24    receiver: ObjRef,
25    index: ObjRef,
26) -> Result<(Store, Probe), Failure> {
27    let store = store_of(interp, receiver)?;
28    probe_in(interp, receiver, CONTENTS, store, index)
29}

31/// [`probe`] against a store the caller has already read, so that a
32/// `Directory`'s method table can be searched with the same chain walk.
33#[inline]
34pub(super) fn probe_in(
35    interp: &mut Interp,
36    receiver: ObjRef,
37    half: Half,
38    store: Store,
39    index: ObjRef,
40) -> Result<(Store, Probe), Failure> {
41    let keys = keys_of(interp, receiver);
42    // String-value hash and index comparison never send, so that protocol
43    // has a traversal of its own that carries no callback restart state.
44    match keys {
45        Keys::StringValue => probe_string_value(interp, store, index),
46        _ => probe_by_protocol(interp, receiver, half, store, index, keys),
47    }
48}

50/// [`probe_in`] for the string-value protocol, which never sends: the index's
51/// string value is read once, for the hash and for every comparison, rather
52/// than copied out again for each.
53fn probe_string_value(
54    interp: &mut Interp,
55    store: Store,
56    index: ObjRef,
57) -> Result<(Store, Probe), Failure> {
58    let inline;
59    let plain: Option<&[u8]> = match index.decode() {
60        Decoded::Text(text) => {
61            inline = text;
62            Some(&inline)
63        }
64        Decoded::Heap { .. } => match interp.heap.get(index).map(|object| &object.body) {
65            Some(Body::Text { bytes, .. }) => Some(bytes.as_slice()),
66            _ => None,
67        },
68        _ => None,
69    };
70    // The plain index's code, kept for the general search below: that
71    // search reads the same bytes, so it need not hash them again.
72    let mut plain_code = None;
73    if let Some(wanted) = plain {
74        let code = code_of(super::super::string_hash(interp.hash_seed, wanted));
75        if let Some(at) = find_plain_string(interp, &store, code, wanted) {
76            return Ok((store, Probe { at, code }));
77        }
78        plain_code = Some(code);
79    }
80    let short;
81    let owned;
82    let wanted: &[u8] = match index.decode() {
83        Decoded::Text(text) => {
84            short = text;
85            &short
86        }
87        _ => {
88            owned = interp.to_text(index).into_owned();
89            &owned
90        }
91    };
92    let code =
93        plain_code.unwrap_or_else(|| code_of(super::super::string_hash(interp.hash_seed, wanted)));
94    debug_assert_eq!(
95        code,
96        code_of(super::super::string_hash(interp.hash_seed, wanted)),
97        "a plain index's string value is its bytes"
98    );
99    let at = find_text_in(interp, &store, code, wanted)?;
100    Ok((store, Probe { at, code }))
101}

103/// The position of the entry whose index's string value is `wanted`, in a
104/// string-keyed store, looked up by `code` -- `wanted`'s string hash's
105/// [`code_of`] -- without building an index object.
106pub(super) fn find_text(
107    interp: &mut Interp,
108    store: &Store,
109    code: u64,
110    wanted: &[u8],
111) -> Result<Option<usize>, Failure> {
112    Ok(find_text_in(interp, store, code, wanted)?.ok())
113}

115/// [`find_text`], or where the chain of `code` is, as [`Probe::at`].
116fn find_text_in(
117    interp: &mut Interp,
118    store: &Store,
119    code: u64,
120    wanted: &[u8],
121) -> Result<Result<usize, Place>, Failure> {
122    if let Some(found) = find_plain_string(interp, store, code, wanted) {
123        return Ok(found);
124    }
125    let (slots, codes) = codes_of(interp, store)?;
126    let chain = chain_in(slots, codes, code);
127    let place = Place::of(chain, codes);
128    let mut at = chain.ok().map(|(_, head)| head);
129    while let Some(position) = at {
130        if let Some(held) = slot_at(interp, store.indexes, position)?
131            && *interp.to_text(held) == *wanted
132        {
133            return Ok(Ok(position));
134        }
135        at = codes_of(interp, store)?.1.next(position);
136    }
137    Ok(Err(place))
138}

140/// What a lookup knows once one of its comparisons has written a store: the
141/// candidates the chain of the index's code held then, and the answer of each
142/// comparison sent since. From then on each of those candidates is compared at
143/// most once and an entry put after is not compared, so the lookup ends
144/// whatever the comparisons do.
145struct Guard {
146    candidates: Vec<ObjRef>,
147    compared: Vec<(ObjRef, bool)>,
148}
149
150/// What a [`Guard`] says of one candidate.
151enum Seen {
152    /// Compared since the guard began, with this answer.
153    Compared(bool),
154    /// In the chain when the guard began, and not compared since.
155    Pending,
156    /// Put after the guard began.
157    Unseen,
158}
159
160impl Guard {
161    /// The guard of a lookup whose comparison of `held` answered `matches`
162    /// and wrote a store, with `store` the receiver's store after it.
163    fn began(
164        interp: &mut Interp,
165        store: &Store,
166        code: u64,
167        held: ObjRef,
168        matches: bool,
169    ) -> Result<Guard, Failure> {
170        let mut candidates = Vec::new();
171        {
172            let (slots, codes) = codes_of(interp, store)?;
173            let indexes = array_slots(interp, store.indexes)?;
174            if let Ok((_, head)) = chain_in(slots, codes, code) {
175                for position in codes.chain(head) {
176                    if let Some(candidate) = indexes.get(position).copied().flatten() {
177                        candidates.push(candidate);
178                    }
179                }
180            }
181        }
182        // A candidate a later comparison removes must not be swept and its
183        // handle reused by an entry put after. The temporaries stay rooted
184        // until the clause that sent the lookup ends, whose frame
185        // (`Interp::enter_stepped_clause`) truncates them.
186        interp.roots.activity_mut().push_temp(held);
187        for &candidate in &candidates {
188            interp.roots.activity_mut().push_temp(candidate);
189        }
190        Ok(Guard {
191            candidates,
192            compared: vec![(held, matches)],
193        })
194    }
195
196    fn seen(&self, held: ObjRef) -> Seen {
197        if let Some(&(_, matches)) = self.compared.iter().find(|(object, _)| *object == held) {
198            Seen::Compared(matches)
199        } else if self.candidates.contains(&held) {
200            Seen::Pending
201        } else {
202            Seen::Unseen
203        }
204    }
205}

207fn probe_by_protocol(
208    interp: &mut Interp,
209    receiver: ObjRef,
210    half: Half,
211    mut store: Store,
212    index: ObjRef,
213    keys: Keys,
214) -> Result<(Store, Probe), Failure> {
215    let generation = interp.store_generation;
216    let code = code_of(hash_of(interp, keys, index)?);
217    if generation != interp.store_generation {
218        store = store_in(interp, receiver, half)?;
219    }
220    // None until a comparison writes a store, so a lookup whose comparisons
221    // write nothing keeps nothing beside its walk.
222    let mut guard: Option<Guard> = None;
223    'search: loop {
224        // A differing cached code is a different index for every key that
225        // keeps the hash contract, and nothing is sent to it: only the chain
226        // of the index's own code is compared.
227        let (slots, codes) = codes_of(interp, &store)?;
228        let chain = chain_in(slots, codes, code);
229        let place = Place::of(chain, codes);
230        let mut at = chain.ok().map(|(_, head)| head);
231        while let Some(position) = at {
232            let Some(held) = slot_at(interp, store.indexes, position)? else {
233                at = codes_of(interp, &store)?.1.next(position);
234                continue;
235            };
236            let known = match guard.as_ref().map(|guard| guard.seen(held)) {
237                None | Some(Seen::Pending) => None,
238                Some(Seen::Compared(matches)) => Some(matches),
239                Some(Seen::Unseen) => {
240                    at = codes_of(interp, &store)?.1.next(position);
241                    continue;
242                }
243            };
244            let matches = match known {
245                Some(matches) => matches,
246                None => {
247                    let generation = interp.store_generation;
248                    let writes = interp.store_writes;
249                    let matches = same_index(interp, &store, keys, position, index)?;
250                    if let Some(guard) = guard.as_mut() {
251                        guard.compared.push((held, matches));
252                    }
253                    if generation != interp.store_generation || writes != interp.store_writes {
254                        // The comparison ran code that wrote a store. New
255                        // arrays, or the compared entry gone, start the walk
256                        // again. In the same arrays a position keeps its entry
257                        // while it is live, so the answer stands.
258                        let current = store_in(interp, receiver, half)?;
259                        if guard.is_none() {
260                            guard = Some(Guard::began(interp, &current, code, held, matches)?);
261                        }
262                        let restart = !current.same_layout(&store)
263                            || slot_at(interp, current.indexes, position)? != Some(held);
264                        store = current;
265                        if restart {
266                            continue 'search;
267                        }
268                    }
269                    matches
270                }
271            };
272            if matches {
273                return Ok((
274                    store,
275                    Probe {
276                        at: Ok(position),
277                        code,
278                    },
279                ));
280            }
281            // Read after the comparison: a removal it made relinked the chain.
282            at = codes_of(interp, &store)?.1.next(position);
283        }
284        return Ok((
285            store,
286            Probe {
287                at: Err(place),
288                code,
289            },
290        ));
291    }
292}

294/// [`find_text`]'s chain walk with the arrays borrowed once, for a plain
295/// string `wanted` and candidates whose every index is one too: inline or a
296/// `Body::Text`, whose bytes are what `to_text` would answer. `None`, and the
297/// one-by-one walk instead, for anything else -- a candidate index of another
298/// kind, or an array that is not one, which that walk refuses.
299fn find_plain_string(
300    interp: &Interp,
301    store: &Store,
302    code: u64,
303    wanted: &[u8],
304) -> Option<Result<usize, Place>> {
305    let (slots, codes) = codes_of(interp, store).ok()?;
306    let indexes = array_slots(interp, store.indexes).ok()?;
307    let chain = chain_in(slots, codes, code);
308    let Ok((_, head)) = chain else {
309        return Some(Err(Place::of(chain, codes)));
310    };
311    for position in codes.chain(head) {
312        if let Some(held) = indexes.get(position).copied().flatten()
313            && plain_text_is(interp, held, wanted)?
314        {
315            return Some(Ok(position));
316        }
317    }
318    Some(Err(Place::of(chain, codes)))
319}

321/// Whether `held`'s string value is `wanted`, for a plain string only: the
322/// bytes in the handle, or a `Body::Text`'s own, which no redirect applies
323/// to. `None` for any other value.
324#[inline]
325fn plain_text_is(interp: &Interp, held: ObjRef, wanted: &[u8]) -> Option<bool> {
326    match held.decode() {
327        Decoded::Text(text) => Some(*text == *wanted),
328        Decoded::Heap { .. } => match &interp.heap.get(held)?.body {
329            Body::Text { bytes, .. } => Some(bytes.as_slice() == wanted),
330            _ => None,
331        },
332        _ => None,
333    }
