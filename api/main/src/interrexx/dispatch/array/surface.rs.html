17/// `Array~allItems`: every item, holes skipped, in index order.

81/// `Array~hasItem(item)`: whether any slot holds it.
82fn native_array_has_item(
83    interp: &mut Interp,
84    _cleared: Cleared,
85    receiver: ObjRef,
86    args: &[Option<ObjRef>],
87) -> Result<Option<ObjRef>, Failure> {
88    let wanted = item_argument(args)?;
89    let search = |interp: &mut Interp| {
90        for (_, item) in ordered_pairs(interp, receiver)? {
91            if same_item(interp, wanted, item)? {
92                return Ok(Some(crate::eval::logical(true)));
93            }
94        }
95        Ok(Some(crate::eval::logical(false)))
96    };
97    // An item not of a base class may compare through a Rexx `==`, and a
98    // string's `==` may run an item's Rexx `REQUEST`.
99    if interp.is_base_class(wanted) {
100        interp.deferred_level(receiver, b"HASITEM", args, search)
101    } else {
102        interp.primitive_level(receiver, b"HASITEM", args, search)
103    }
104}

122/// `Array~hasIndex(index...)`: whether that slot holds an item. It takes any
123/// number of arguments because a multi-dimensional subscript arrives as
124/// several.
125fn native_array_has_index(
126    interp: &mut Interp,
127    _cleared: Cleared,
128    receiver: ObjRef,
129    args: &[Option<ObjRef>],
130) -> Result<Option<ObjRef>, Failure> {
131    let Some(position) = position_in(interp, receiver, args, IndexUse::Get)? else {
132        return Ok(Some(crate::eval::logical(false)));
133    };
134    let held = matches!(
135        read_slots(interp, receiver, |slots| slots.get(position - 1).copied())?,
136        Some(Some(_))
137    );
138    Ok(Some(crate::eval::logical(held)))
139}

141/// `Array~remove(index...)`: takes the item out of that slot and answers it,
142/// or `.nil`.
143fn native_array_remove(
144    interp: &mut Interp,
145    _cleared: Cleared,
146    receiver: ObjRef,
147    args: &[Option<ObjRef>],
148) -> Result<Option<ObjRef>, Failure> {
149    let Some(position) = position_in(interp, receiver, args, IndexUse::Get)? else {
150        return Ok(Some(ObjRef::NIL));
151    };
152    let held = match read_slots(interp, receiver, |slots| slots.get(position - 1).copied())? {
153        Some(Some(item)) => item,
154        Some(None) | None => return Ok(Some(ObjRef::NIL)),
155    };
156    clear_array_slot(interp, receiver, position - 1)?;
157    Ok(Some(held))
158}

160/// `Array~removeItem(item)`: clears the first slot holding it and answers it,
161/// or `.nil`. On a `Queue` the slot is deleted and the items after it close
162/// up, because `removeItem` reaches the virtual `remove`, which `Queue`
163/// redefines as `deleteItem`.
164fn native_array_remove_item(
165    interp: &mut Interp,
166    _cleared: Cleared,
167    receiver: ObjRef,
168    args: &[Option<ObjRef>],
169) -> Result<Option<ObjRef>, Failure> {
170    let wanted = item_argument(args)?;
171    let slots = slots_of(interp, receiver)?;
172    for (offset, slot) in slots.iter().enumerate() {
173        let Some(item) = *slot else { continue };
174        if same_item(interp, wanted, item)? {
175            if is_queue(interp, receiver) {
176                array_splice(interp, receiver, offset, None)?;
177            } else {
178                clear_array_slot(interp, receiver, offset)?;
179            }
180            return Ok(Some(item));
181        }
182    }
183    Ok(Some(ObjRef::NIL))
184}

186/// `Array~supplier`: a `Supplier` over this array's items and indexes.
187fn native_array_supplier(
188    interp: &mut Interp,
189    _cleared: Cleared,
190    receiver: ObjRef,
191    _args: &[Option<ObjRef>],
192) -> Result<Option<ObjRef>, Failure> {
193    let pairs = ordered_pairs(interp, receiver)?;
194    let items: Vec<ObjRef> = pairs.iter().map(|(_, item)| *item).collect();
195    let indexes: Vec<ObjRef> = pairs.into_iter().map(|(index, _)| index).collect();
196    let items = array_of(interp, items);
197    interp.roots.activity_mut().push_temp(items);
198    let indexes = array_of(interp, indexes);
199    interp.roots.activity_mut().push_temp(indexes);
200    new_supplier(interp, items, indexes).map(Some)
201}

203/// Empties one slot of an array receiver.
204fn clear_array_slot(interp: &mut Interp, receiver: ObjRef, offset: usize) -> Result<(), Failure> {
205    let receiver = store_of(interp, receiver)?;
206    match interp.heap.get_mut(receiver).map(|object| &mut object.body) {
207        Some(Body::Array { slots, .. }) => {
208            slots.set_within(offset, None);
209            Ok(())
210        }
211        _ => Err(crate::Raised::receiver_kind("a value that is not an array").into()),
212    }
213}

230/// `Array~first` and `Array~last`: the INDEX of the outermost occupied slot,
231/// or `.nil`.
232fn array_end(
233    interp: &mut Interp,
234    receiver: ObjRef,
235    take_last: bool,
236    want_item: bool,
237) -> Result<Option<ObjRef>, Failure> {
238    let receiver = store_of(interp, receiver)?;
239    let found = read_slots(interp, receiver, |slots| {
240        if take_last {
241            slots.iter().rposition(Option::is_some)
242        } else {
243            slots.iter().position(Option::is_some)
244        }
245    })?;
246    let Some(offset) = found else {
247        return Ok(Some(ObjRef::NIL));
248    };
249    if want_item {
250        return Ok(Some(match array_slots(interp, receiver)?[offset] {
251            Some(item) => item,
252            None => ObjRef::NIL,
253        }));
254    }
255    let dimensions = dimensions_of(interp, receiver)?;
256    Ok(Some(subscript_object(
257        interp,
258        offset,
259        dimensions.as_deref(),
260    )))
261}

299/// `Array~next(index...)` and `Array~previous(index...)`: the nearest occupied
300/// index on that side, or `.nil`.
301fn array_step(
302    interp: &mut Interp,
303    receiver: ObjRef,
304    args: &[Option<ObjRef>],
305    forward: bool,
306) -> Result<Option<ObjRef>, Failure> {
307    // The index is read for the receiver itself, so a `Queue` reads an
308    // Array index by `queue_index` rather than spreading it.
309    let index;
310    let args = match args {
311        [Some(only)] => {
312            index = [Some(queue_index(interp, receiver, *only, 1)?)];
313            &index[..]
314        }
315        _ => args,
316    };
317    let store = store_of(interp, receiver)?;
318    // r[class.queue.next]: a queue's `next` and `previous` take one argument,
319    // and a second is 93.926 however many items the queue holds.
320    if args.len() > 1 && is_queue(interp, receiver) {
321        return Err(Raised::too_many_subscripts(1).into());
322    }
323    let receiver = store;
324    // **An index past the end still steps.** Measured:
325    // `.Array~of('a','b')~previous(5)` answers `2`, where bailing out on the
326    // out-of-range subscript answers `.nil`.
327    let position = match position_in(interp, receiver, args, IndexUse::Get)? {
328        Some(position) => position,
329        None => {
330            // An `Array` index stands for its items as subscripts.
331            let single = match args {
332                [Some(only)] => match super::index_list(interp, *only, 1)? {
333                    Some(list) if list.len() == 1 => list[0],
334                    Some(_) => None,
335                    None => Some(*only),
336                },
337                _ => None,
338            };
339            match single {
340                Some(only) => super::positive_index(interp, only, 1)?,
341                // **A subscript list with a component past its dimension is
342                // past the array** (Deviation 120): `next` answers `.nil` and
343                // `previous` the last item in `allItems` order.
344                None if forward => return Ok(Some(ObjRef::NIL)),
345                None => usize::MAX,
346            }
347        }
348    };
349    let offsets = occupied(interp, receiver)?;
350    let found = if forward {
351        offsets.into_iter().find(|offset| *offset + 1 > position)
352    } else {
353        offsets
354            .into_iter()
355            .rev()
356            .find(|offset| *offset + 1 < position)
357    };
358    let Some(offset) = found else {
359        return Ok(Some(ObjRef::NIL));
360    };
361    let dimensions = dimensions_of(interp, receiver)?;
362    Ok(Some(subscript_object(
363        interp,
364        offset,
365        dimensions.as_deref(),
366    )))
367}

400/// `Array~insert(item [, index])`: puts `item` **after** `index` and shifts
401/// the rest along, answering the index it landed on.
402fn native_array_insert(
403    interp: &mut Interp,
404    _cleared: Cleared,
405    receiver: ObjRef,
406    args: &[Option<ObjRef>],
407) -> Result<Option<ObjRef>, Failure> {
408    single_dimension_only(interp, receiver, "INSERT")?;
409    // **The item is optional**, unlike `append`'s: `insert` requires no
410    // argument for it. Measured, `.Array~of('x','y')~insert` answers `3` and
411    // leaves size 3 with items 2 -- an empty slot.
412    let item = args.first().copied().flatten();
413    let at = match args.get(1).copied() {
414        // `.nil` is the front, which is the one spelling that is not an index
415        // at all.
416        Some(Some(index)) if index == ObjRef::NIL => 0,
417        Some(Some(index)) => {
418            // An Array index holding more than one item is too many subscripts
419            // for a queue's one dimension (`r[class.queue.insert]`): the items
420            // are counted, not the slots up to the last one.
421            let held = interp.array_slots(index).map_or(0, |slots| {
422                slots.iter().filter(|slot| slot.is_some()).count()
423            });
424            if held > 1 && is_queue(interp, receiver) {
425                return Err(Raised::too_many_subscripts(1).into());
426            }
427            let position = single_subscript(interp, receiver, index, 2)?
428                .ok_or_else(|| Raised::too_many_subscripts(1))?;
429            // "the position must be location of an existing item within the
430            // bounds of the queue, unlike an array which can insert at empty
431            // slots or beyond the existing bounds". `insert` validates the
432            // index for an update, which extends rather than refusing, so a
433            // position past the queue's extent is 93.966 too and never `put`'s
434            // 93.918.
435            if is_queue(interp, receiver) && position > last_item(interp, receiver)? {
436                return Err(Raised::incorrect_queue_index(position).into());
437            }
438            position
439        }
440        // Omitted: after the last OCCUPIED slot, not the end of the array.
441        Some(None) | None => last_item(interp, receiver)?,
442    };
443    splice_absorbing_slack(interp, receiver, at, item)?;
444    Ok(Some(interp.counted(at + 1)))
445}

447/// `Array~delete(index...)`: takes the slot out, closes the gap and answers
448/// the item.
449pub(in crate::dispatch) fn native_array_delete(
450    interp: &mut Interp,
451    _cleared: Cleared,
452    receiver: ObjRef,
453    args: &[Option<ObjRef>],
454) -> Result<Option<ObjRef>, Failure> {
455    // A required-argument check on the index comes FIRST and is 93.903, where
456    // the index-validating family is 93.901.
457    let index = item_argument(args)?;
458    single_dimension_only(interp, receiver, "DELETE")?;
459    one_subscript_list(interp, receiver, index, 1)?;
460    let Some(position) = position_in(interp, receiver, args, IndexUse::Get)? else {
461        return Ok(Some(ObjRef::NIL));
462    };
463    let held = match read_slots(interp, receiver, |slots| slots.get(position - 1).copied())? {
464        Some(slot) => slot,
465        None => return Ok(Some(ObjRef::NIL)),
466    };
467    array_splice(interp, receiver, position - 1, None)?;
468    Ok(Some(held.unwrap_or(ObjRef::NIL)))
469}

492/// Refuses an `Array` index of more than one subscript to `insert` and `delete` with 93.926,
493/// also on an array with no dimension yet, which these methods take as single-dimensional
494/// (`r[class.array.insert]`, `r[class.array.delete]`). A list with a hole is 98.975 first, as
495/// [`super::index_list`] reads it; a queue reads its index by [`queue_index`] instead.
496fn one_subscript_list(
497    interp: &mut Interp,
498    receiver: ObjRef,
499    value: ObjRef,
500    position: usize,
501) -> Result<(), Failure> {
502    if is_queue(interp, receiver) {
503        return Ok(());
504    }
505    match super::index_list(interp, value, position)? {
506        Some(spread) if spread.len() > 1 => Err(Raised::too_many_subscripts(1).into()),
507        _ => Ok(()),
508    }
509}

511/// The one subscript `insert` and `section` take, as a single-dimensional
512/// receiver reads it: an `Array` argument stands for the subscripts it holds,
513/// and exactly one of them is a position. Multiple subscripts on a
514/// dimensionless array yield no position. `position` is the argument's
515/// place in the method's argument list.
516///
517/// A queue reads an `Array` index by [`queue_index`] instead.
518fn single_subscript(
519    interp: &mut Interp,
520    receiver: ObjRef,
521    value: ObjRef,
522    position: usize,
523) -> Result<Option<usize>, Failure> {
524    if interp.array_slots(value).is_none() {
525        return super::positive_index(interp, value, position).map(Some);
526    }
527    if is_queue(interp, receiver) {
528        let index = queue_index(interp, receiver, value, position)?;
529        return super::positive_index(interp, index, position).map(Some);
530    }
531    let spread = super::index_list(interp, value, position)?.unwrap_or_default();
532    match spread[..] {
533        [] => Err(Raised::not_enough_method_arguments(position).into()),
534        [Some(only)] => super::positive_index(interp, only, position).map(Some),
535        [None] => Err(Raised::missing_array_element(1).into()),
536        _ if super::array_is_fixed_dimension(interp, receiver)? => {
537            Err(Raised::too_many_subscripts(1).into())
538        }
539        _ => Ok(None),
540    }
541}

543/// `Array~section(start [, count])`: a new array over that run.
544fn native_array_section(
545    interp: &mut Interp,
546    _cleared: Cleared,
547    receiver: ObjRef,
548    args: &[Option<ObjRef>],
549) -> Result<Option<ObjRef>, Failure> {
550    // The dimension check comes FIRST here and the required argument second,
551    // which is the opposite order from `delete`.
552    single_dimension_only(interp, receiver, "SECTION")?;
553    let start = item_argument(args)?;
554    let start = single_subscript(interp, receiver, start, 1)?;
555    // The count is an optional length argument, so one that is not a
556    // non-negative whole number is 93.923. It reads an `Array` as the string
557    // request does, as its items' lines, and a refusal names the Array.
558    let count = match args.get(1).copied().flatten() {
559        Some(count) if interp.array_slots(count).is_some() => {
560            let lines = interp.required_string_value(count)?;
561            match super::super::optional_length_argument(interp, &[Some(lines)], 0) {
562                Ok(count) => count,
563                Err(_) => {
564                    let found = interp.string_value_text(count)?;
565                    return Err(Raised::invalid_length(&found).into());
566                }
567            }
568        }
569        _ => super::super::optional_length_argument(interp, args, 1)?,
570    };
571    let Some(start) = start else {
572        return Ok(Some(same_class_array(interp, receiver, Vec::new())?));
573    };
574    let slots = slots_of(interp, receiver)?;
575    let available = slots.len().saturating_sub(start.saturating_sub(1));
576    let count = count.map_or(available, |count| count.min(available));
577    let taken: Vec<Option<ObjRef>> = slots
578        .into_iter()
579        .skip(start.saturating_sub(1))
580        .take(count)
581        .collect();
582    Ok(Some(same_class_array(interp, receiver, taken)?))
583}

585/// `Array~dimensions`: an `Array` of the extents.
586fn native_array_dimensions(
587    interp: &mut Interp,
588    _cleared: Cleared,
589    receiver: ObjRef,
590    _args: &[Option<ObjRef>],
591) -> Result<Option<ObjRef>, Failure> {
592    // **A single-dimensional array answers its SIZE, not the extent it was
593    // fixed at.** For one, `dimensions` is `new_array(new_integer(size()))`,
594    // and only a multi-dimensional array reports the stored shape. Measured:
595    // `.Array~new(0)~append('q')` then `~dimensions` is `1` -- and
596    // `.Array~of()` the same -- where reading the fixed list answered the
597    // stale `0`.
598    let extents = match dimensions_of(interp, receiver)? {
599        Some(shape) if shape.len() > 1 => shape,
600        _ => vec![read_slots(interp, receiver, <[_]>::len)?],
601    };
602    let items = extents
603        .into_iter()
604        .map(|extent| interp.counted(extent))
605        .collect();
606    Ok(Some(array_of(interp, items)))
607}

609/// A new array of the receiver's own class over `slots`.
610fn same_class_array(
611    interp: &mut Interp,
612    receiver: ObjRef,
613    slots: Vec<Option<ObjRef>>,
614) -> Result<ObjRef, Failure> {
615    if interp.array_slots(receiver).is_some() {
616        let store = interp.alloc_charged(BehaviourId::ARRAY, Body::array(slots));
617        interp.roots.activity_mut().push_temp(store);
618        return Ok(store);
619    }
620    let class = interp.class_of_value(receiver).ok_or_else(|| {
621        Failure::from(crate::Raised::receiver_kind("a value that is not an array"))
622    })?;
623    // The receiver's class is sent `new` with the size, as `new` for an
624    // `Array` or a `Queue` makes the object and sends its `init`; the slots
625    // then go into the object's store.
626    let size = interp.counted(slots.len());
627    let caller = interp.caller();
628    let object = interp
629        .send_message(class, b"NEW", None, &[Some(size)], caller)?
630        .ok_or_else(|| Failure::from(Raised::no_result(b"NEW")))?;
631    interp.roots.activity_mut().push_temp(object);
632    let store = store_of(interp, object)?;
633    let Some(body) = interp.heap.get_mut(store).map(|object| &mut object.body) else {
634        return Err(crate::Raised::receiver_kind("a value that is not an array").into());
635    };
636    if !matches!(body, Body::Array { .. }) {
637        return Err(crate::Raised::receiver_kind("a value that is not an array").into());
638    }
639    let before = body.held_bytes();
640    *body = Body::array(slots);
641    let after = body.held_bytes();
642    interp.charge_growth(after.saturating_sub(before));
643    interp.heap.release_body_bytes(before.saturating_sub(after));
644    Ok(object)
645}

647/// [`array_splice_slot`], except that a trailing empty slot absorbs the shift
648/// instead of the array growing.
649fn splice_absorbing_slack(
650    interp: &mut Interp,
651    receiver: ObjRef,
652    at: usize,
653    item: Option<ObjRef>,
654) -> Result<(), Failure> {
655    // **Slack the array ALREADY had**, not slack the insert just created.
656    // Measured: `.Array~new(4)~insert('j')` leaves size 4, where
657    // `.Array~of('x','y')~insert` -- whose last slot is occupied -- leaves
658    // size 3.
659    let slack = read_slots(interp, receiver, |slots| slots.last() == Some(&None))?;
660    // Growing to `at` is not slack the array already had.
661    let length = read_slots(interp, receiver, <[_]>::len)?;
662    if at > length {
663        array_grow(interp, receiver, at)?;
664    }
665    array_splice_slot(interp, receiver, at, item)?;
666    if !slack {
667        return Ok(());
668    }
669    let store = store_of(interp, receiver)?;
670    match interp.heap.get_mut(store).map(|object| &mut object.body) {
671        Some(Body::Array { slots, .. }) => {
672            if slots.last() == Some(&None) {
673                slots.pop();
674            }
675            Ok(())
676        }
677        _ => Err(crate::Raised::receiver_kind("a value that is not an array").into()),
678    }
679}
