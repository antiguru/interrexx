3impl Interp {
4    /// Pushes one `::METHOD` body's activation and the [`CallTail`] its
5    /// sender's level state goes into, up to the point the body would run.
6    ///
7    /// [`CallTail`]: crate::run::CallTail
8    #[expect(
9        clippy::too_many_arguments,
10        reason = "the send's own parts, and the receiver's behaviour where the send has it"
11    )]
12    pub(super) fn begin_method(
13        &mut self,
14        _cleared: Cleared,
15        installed: crate::InstalledMethodBody,
16        resolution: Resolution,
17        super_scope: SuperScope,
18        receiver: ObjRef,
19        name: &[u8],
20        args: &[Option<ObjRef>],
21    ) -> Result<(), Failure> {
22        let Some(program) = self.programs.get(installed.program).map(Rc::clone) else {
23            return Err(Loud::missing_body().into());
24        };
25        // Every read is `get`, not an index: an `InstalledMethodBody`
26        // always names a real directive with a body, but this crate's rule for an internal
27        // inconsistency is a loud refusal rather than a panic, and
28        // `body_of`'s own `?` already follows it.
29        let Some(directive) = program.directives.get(installed.directive) else {
30            return Err(Loud::missing_body().into());
31        };
32        let Some(body) = body_of(&program, Some(installed.directive)) else {
33            return Err(Loud::missing_body().into());
34        };
35        let guarded = self.sends_guarded(
36            resolution.method,
37            crate::guards::directive_guarded(&directive.kind),
38        );
39        // The same guard `Interp::invoke_call` takes and for the same
40        // reason: a method that sends itself a message is an unbounded
41        // recursion, and it must become a reportable condition rather than a
42        // native abort.
43        let probe = 0u8;
44        if self.nesting_depth() >= MAX_ACTIVATION_DEPTH || self.stack_exhausted(&raw const probe) {
45            return Err(Raised::insufficient_stack().into());
46        }
47        let plan = self.plan_for(
48            BodyKey {
49                program: installed.program,
50                directive: Some(installed.directive),
51            },
52            body,
53            &program.symbols,
54            &program.source,
55        );
56        let frame = self.roots.push_slots(plan.len());
57        let (self_slot, super_slot) = (plan.self_slot, plan.super_slot);
58        let reserve_at_once =
59            guarded && !plan.traces_entry_early && !self.activities.guards.is_live();
60        let callee_id = self.next_activation_id();
61        let super_scope = match super_scope {
62            SuperScope::In(behaviour) => self.super_scope_in(behaviour, resolution.scope),
63            SuperScope::Known(super_scope) => super_scope,
64            SuperScope::OfReceiver => self.super_scope_for(receiver, resolution),
65        };
66        // **The calling convention, entered before anything reads it.** The
67        // receiver goes in here and is read back out below, so the identity
68        // the activation carries, the `SELF` the body reads and the caller a
69        // send inside the body resolves as all name one field rather than
70        // each taking its own copy of this function's argument.
71        let arguments = self.shared_arguments(args);
72        let name = self.activity.invocation_name(name);
73        let saved_context = std::mem::replace(
74            &mut self.activity.call_context,
75            crate::CallContext {
76                name: Rc::clone(&name),
77                arguments,
78                receiver: Some(receiver),
79            },
80        );
81        // **Shadows the parameter**, which is what makes the routing a
82        // property of scope rather than of a comment: past this line the name
83        // `receiver` is the convention's own field, so the bindings below
84        // cannot read the argument even by accident.
85        let receiver = self
86            .activity
87            .call_context
88            .receiver
89            .expect("the calling convention replaced directly above carries the receiver");
90        // The package the `::METHOD` was declared in, exactly as a
91        // `::ROUTINE` starts from its own. `running_activation` rather than
92        // `activation` because a send made outside any activation has no
93        // caller for `::OPTIONS NUMERIC INHERIT` to read.
94        let start = self.package_start(
95            installed.program,
96            self.running_activation().map(|caller| &caller.settings),
97        );
98        let identity = MethodIdentity {
99            name,
100            scope: resolution.scope,
101            receiver,
102        };
103        let mut callee = match self.spare_activation() {
104            Some(mut spare) => {
105                spare.reset_method(
106                    callee_id,
107                    program,
108                    installed.program,
109                    installed.directive,
110                    plan,
111                    frame,
112                    identity,
113                );
114                spare
115            }
116            None => Box::new(Activation::method(
117                callee_id,
118                program,
119                installed.program,
120                installed.directive,
121                plan,
122                frame,
123                identity,
124            )),
125        };
126        // Dictionary mutation must not replace the Method this invocation
127        // entered. Existing registered objects use the activation's GC root.
128        if let Some(executable) = self.executable_objects.get(resolution.method).copied() {
129            callee.cold_mut().executable = Some(executable);
130        }
131        if let Some(start) = start {
132            start.apply(&mut callee);
133        }
134        // With one activity the lock is the activation's flag alone, taken
135        // here unless a package-traced `>I>` has to come first.
136        let reserve_at_once = reserve_at_once && !callee.trace_mode.labels;
137        callee.flags = crate::activation::ActivationFlags::method(guarded, reserve_at_once);
138        self.push_boxed_activation(callee);
139        self.trace_package_invocation_entry();
140        if guarded && !reserve_at_once {
141            self.reserve_for_method(crate::guards::GuardKey {
142                object: receiver,
143                scope: resolution.scope,
144            });
145        }
146
147        // `SELF` and `SUPER`, the two locals set on a method activation before
148        // its first instruction. Measured in a class method of `::class K`:
149        // `say self` is `The K class` and `say super` is `The Class class`.
150        let self_slot = match self_slot {
151            Some(slot) => slot,
152            None => self.slot_of(b"SELF"),
153        };
154        self.set_variable(frame, self_slot, receiver);
155        let super_slot = match super_slot {
156            Some(slot) => slot,
157            None => self.slot_of(b"SUPER"),
158        };
159        // `.nil` for the topmost scope, which is what `superScope` answers
160        // there.
161        self.set_variable(frame, super_slot, super_scope.unwrap_or(ObjRef::NIL));
162
163        // The same level state `Interp::invoke_call` saves and restores, set
164        // to the same values its `::ROUTINE` arm uses -- that function's own
165        // comment enumerates the pieces and carries the measurement for
166        // each. A method's clause echoes are at indent 0 whatever the sending
167        // clause's own indent was: measured, a send from inside two nested
168        // `DO` blocks echoes the method's clauses at 0.
169        let saved_clause_state = self.save_clause_state();
170        let saved_base = std::mem::replace(&mut self.activity.activation_indent, 0);
171        let saved_line = std::mem::take(&mut self.activity.clause_line_override);
172        self.activity.call_tails.push(crate::run::CallTail::method(
173            saved_context,
174            saved_clause_state,
175            saved_base,
176            saved_line,
177        ));
178        Ok(())
179    }
180}
