32/// The bytes a small integer or an inline string renders as, for a reader
33/// that cannot allocate into the interpreter.
34fn rendered(value: ObjRef) -> Option<Vec<u8>> {
35    match value.decode() {
36        Decoded::SmallInt(number) => Some(number.to_string().into_bytes()),
37        Decoded::Text(inline) => Some(inline.to_vec()),
38        _ => None,
39    }
40}

42/// The pool name an extension's spelling addresses, a simple or a stem name,
43/// or `None` for one the variable lookup answers nothing for: a constant
44/// symbol or a compound name. The write then does nothing, which is all the
45/// API can see of the refusal.
46fn pool_variable_name(name: &[u8]) -> Option<Vec<u8>> {
47    let first = *name.first()?;
48    if first.is_ascii_digit()
49        || first == b'.'
50        || crate::run::shape_of(name) == crate::run::NameShape::Compound
51    {
52        return None;
53    }
54    Some(name.to_ascii_uppercase())
55}

58    /// Runs `binding`'s procedure against `args` with `receiver` as its self,
59    /// and answers what the extension returned, or `Entered` with the
60    /// activity parked for the call's driver exit
61    /// ([`Interp::leaves_driver`]).
62    ///
63    /// # Errors
64    /// The condition the extension raised, whatever converting an argument or
65    /// the result refuses, and [`Loud`] for a conversion this crate has not
66    /// written.
67    /// `reserved` says the send holds the method's guard lock, which the
68    /// call's frame then holds and releases where it still does.
69    pub(super) fn run_library_method(
70        &mut self,
71        binding: &LibraryBinding,
72        resolution: Resolution,
73        receiver: ObjRef,
74        name: &[u8],
75        args: &[Option<ObjRef>],
76        reserved: bool,
77    ) -> Result<Started<Option<ObjRef>>, Failure> {
78        let owner = self.pool_owner(receiver)?;
79        let Some(entry) = binding
80            .method
81            .and_then(|method| binding.library.method_at(method))
82        else {
83            return Err(Loud::library_procedure_gone().into());
84        };
85        if !entry.has_entry_point()
86            && let Some(body) = crate::extensions::method_body(&binding.library, &entry.name)
87        {
88            return self
89                .run_rust_method(body, owner, resolution, receiver, name, args, reserved)
90                .map(Started::Ran);
91        }
92        let held = HeldCall::Method(entry.held(), resolution.method);
93        let exits = self.leaves_driver();
94        self.push_native_frame(owner, resolution.scope, Some(receiver), name, args, None);
95        pin_enter!(self, crate::pinning::PinKind::NativeApiCallback);
96        let packaged = self.external_package_path(resolution.method).is_some();
97        self.native_frame_mut().packaged = packaged;
98        self.native_frame_mut().reserved = reserved;
99        let Some((answered, pending)) =
100            self.native_call(held, args, exits, receiver, resume_library_method)
101        else {
102            return Ok(Started::Entered);
103        };
104        self.end_library_method(answered, pending, receiver, resolution.scope, packaged)
105            .map(Started::Ran)
106    }

108    /// What a library method answers once its stub has returned `answered`,
109    /// with the call's frame still pushed and pinned: its frame popped, its
110    /// guard lock released where the frame still held it, and then its
111    /// condition or its value.
112    #[inline(always)]
113    fn end_library_method(
114        &mut self,
115        answered: Result<Option<ObjRef>, Refused>,
116        pending: Option<usize>,
117        receiver: ObjRef,
118        scope: ObjRef,
119        packaged: bool,
120    ) -> Result<Option<ObjRef>, Failure> {
121        self.root_native_answer(&answered);
122        let trapped = self.call_trapped_native_condition();
123        let popped = self.pop_native_frame();
124        pin_leave!(self);
125        self.collect_under_stress();
126        if popped.reserved {
127            self.release_guard(crate::guards::GuardKey {
128                object: receiver,
129                scope,
130            });
131        }
132        let trapped = trapped?;
133
134        // The condition first, because it is raised in the caller's frame once
135        // the call has returned and before anything is done with the value the
136        // extension wrote. For example, `.MyRe~new('[')` over `RegExp_Init` is
137        // error 38 at rc 218, and the extension returned zero on that path.
138        if let Some(number) = pending {
139            self.activity.native_reraise = true;
140            return Err(condition_of(number));
141        }
142        self.settle_native_call(answered, popped, packaged, trapped)
143    }

145    /// What a call by name to library code `code` ran: the copy the lookup
146    /// found, or for `.nil` the one object standing for `code`. The copy is
147    /// live: the call site that kept it pins it ([`Interp::repin`]), and one
148    /// just looked up is in the table the lookup read.
149    pub(crate) fn found_routine_object(&self, code: usize, copy: ObjRef) -> crate::RoutineObject {
150        if copy == ObjRef::NIL {
151            return crate::RoutineObject::OfCode;
152        }
153        debug_assert!(
154            self.heap.get(copy).is_some()
155                && matches!(
156                    self.executable_sources.get(&copy),
157                    Some(crate::ExecutableRecord {
158                        source: crate::ExecutableSource::Loaded { code: held },
159                        ..
160                    }) if *held == code
161                ),
162            "a call by name found a Routine copy that is gone or runs other code"
163        );
164        crate::RoutineObject::Given(copy)
165    }

167    /// Applies a call site's [`crate::run::Repin`]: a `Routine` copy stays
168    /// rooted while any call site keeps it, and a kept `::ROUTINE`'s program is
169    /// held by the package of the program owning the site, as the instruction
170    /// keeps its routine for the instruction's life.
171    #[inline]
172    pub(crate) fn repin(&mut self, repin: crate::run::Repin) {
173        if repin.pin == ObjRef::NIL && repin.unpin == ObjRef::NIL && repin.program.is_none() {
174            return;
175        }
176        self.repin_held(repin);
177    }

179    /// [`Interp::repin`] for a change that pins or unpins something.
180    #[cold]
181    #[inline(never)]
182    fn repin_held(&mut self, repin: crate::run::Repin) {
183        if repin.pin != ObjRef::NIL {
184            self.pin_copy(repin.pin);
185        }
186        if repin.unpin != ObjRef::NIL {
187            self.unpin_copy(repin.unpin);
188        }
189        if let Some(change) = repin.program {
190            if let Some(hold) = change.hold {
191                self.hold_site_program(hold, true);
192            }
193            if let Some(hold) = change.release {
194                self.hold_site_program(hold, false);
195            }
196        }
197    }

199    /// Takes (`take`) or drops one call site's hold: the holder's package
200    /// traces the held program's while any of its sites holds it, an edge
201    /// rather than a root, so packages whose sites keep each other's routines
202    /// are collected together.
203    #[cold]
204    pub(crate) fn hold_site_program(&mut self, hold: crate::run::SiteHold, take: bool) {
205        debug_assert!(
206            take || self.site_hold_count != 0,
207            "a site drops a hold no count has"
208        );
209        self.site_hold_count = match take {
210            true => self.site_hold_count + 1,
211            false => self.site_hold_count - 1,
212        };
213        let (Some(held), Some(package)) = (
214            self.program_package(hold.program),
215            self.program_package(hold.holder),
216        ) else {
217            return;
218        };
219        let Some(data) = self.package_data_mut(package) else {
220            return;
221        };
222        if take {
223            *data.site_held.entry(held).or_insert(0) += 1;
224        } else if let Some(count) = data.site_held.get_mut(&held) {
225            *count -= 1;
226            if *count == 0 {
227                data.site_held.remove(&held);
228            }
229        }
230    }

300    /// Runs the library routine [`Interp::library_codes`] row `code` is
301    /// against `args`, and answers what the extension returned, or `Entered`
302    /// as [`Interp::run_library_method`] does.
303    ///
304    /// `name` is the name the traceback's built-in routine line gives, and
305    /// `executable` what the routine's `StackFrame` answers.
306    ///
307    /// # Errors
308    /// The condition the extension raised, whatever converting an argument or
309    /// the result refuses, and [`Loud`] for a conversion or a routine style
310    /// this crate does not implement.
311    pub(crate) fn run_library_routine(
312        &mut self,
313        code: usize,
314        name: &[u8],
315        args: &[Option<ObjRef>],
316        executable: crate::RoutineObject,
317    ) -> Result<Started<Option<ObjRef>>, Failure> {
318        let Some((library, index)) = self.library_code_routine(code) else {
319            return Err(Loud::library_procedure_gone().into());
320        };
321        let Some(entry) = library.routine_at(index) else {
322            return Err(Loud::library_procedure_gone().into());
323        };
324        if !entry.has_entry_point()
325            && let Some(body) = crate::extensions::routine_body(library, &entry.name)
326        {
327            return self.run_rust_routine(body, code, name, args, executable);
328        }
329        if let Some(held) = entry.held_classic() {
330            return self
331                .call_classic_entry(name, args, |name, arguments, queue| {
332                    held.call_function(name, arguments, queue)
333                })
334                .map(Started::Ran);
335        }
336        let held = HeldCall::Routine(entry.held());
337        let exits = self.leaves_driver();
338        let root = std::mem::take(&mut self.leave_next_native);
339        self.push_native_frame(ObjRef::NIL, ObjRef::NIL, None, name, args, Some(code));
340        self.native_frame_mut().executable = executable;
341        pin_enter!(self, crate::pinning::PinKind::NativeApiCallback);
342        let program = self.library_code_program(code);
343        self.native_frame_mut().packaged = program.is_some();
344        let Some((answered, pending)) =
345            self.native_call(held, args, exits, ObjRef::NIL, resume_library_routine)
346        else {
347            // The call the root program makes first is the instance's root
348            // call, which `leave_next_native` sends off the baton.
349            if root && let Some(frame) = self.activity.native_call.as_ref().map(|call| call.frame())
350            {
351                self.note_root_call(frame);
352            }
353            return Ok(Started::Entered);
354        };
355        self.end_library_routine(answered, pending, name, program, args)
356            .map(Started::Ran)
357    }

359    /// [`Interp::run_library_method`] for a method of a library this crate
360    /// runs in Rust: the same frame, pin and guard as a native call, around
361    /// `body` in place of the stub.
362    #[expect(
363        clippy::too_many_arguments,
364        reason = "the native call's own inputs, which run_library_method passes through"
365    )]
366    fn run_rust_method(
367        &mut self,
368        body: crate::extensions::MethodBody,
369        owner: ObjRef,
370        resolution: Resolution,
371        receiver: ObjRef,
372        name: &[u8],
373        args: &[Option<ObjRef>],
374        reserved: bool,
375    ) -> Result<Option<ObjRef>, Failure> {
376        self.push_native_frame(owner, resolution.scope, Some(receiver), name, args, None);
377        pin_enter!(self, crate::pinning::PinKind::NativeApiCallback);
378        let packaged = self.external_package_path(resolution.method).is_some();
379        self.native_frame_mut().packaged = packaged;
380        self.native_frame_mut().reserved = reserved;
381        // A failure is held on the frame as a callback's condition is, which
382        // settles the sends it ended where the call returns.
383        let answered = match body(self, receiver, resolution.scope, args) {
384            Ok(value) => {
385                if let Some(value) = value {
386                    self.roots.activity_mut().push_temp(value);
387                }
388                value
389            }
390            Err(failure) => {
391                self.hold_native_condition(failure);
392                None
393            }
394        };
395        let popped = self.pop_native_frame();
396        pin_leave!(self);
397        self.collect_under_stress();
398        if popped.reserved {
399            self.release_guard(crate::guards::GuardKey {
400                object: receiver,
401                scope: resolution.scope,
402            });
403        }
404        match popped.raised {
405            Some(failure) => {
406                self.reraise_in_caller(&failure);
407                Err(failure)
408            }
409            None => Ok(answered),
410        }
411    }

413    /// Marks `failure`, a condition a native call ended with, as re-raised in the
414    /// call's caller when it is a `SYNTAX` condition, as
415    /// [`Interp::raise_held_condition`] does: its POSITION, PROGRAM, PACKAGE and
416    /// report line become the caller's clause's, PROPAGATED is 1, and its
417    /// traceback keeps the raising line. For example, over `RxCalcSqrt` (by name
418    /// and through `::routine ... external`), `RegExp_Match` and `RegExp_Parse`:

420    /// argument's MAKESTRING (40.1) alike (`lang/library_refusal_reraised.rex`,
421    /// `lang/library_string_argument.rex`).
422    fn reraise_in_caller(&mut self, failure: &Failure) {
423        if !matches!(failure, Failure::Raised(held) if held.condition != "SYNTAX") {
424            self.activity.native_reraise = true;
425        }
426    }

428    /// [`Interp::run_library_routine`] for a routine of a library this crate
429    /// runs in Rust, blamed as a native routine is. A body that parked its
430    /// activity ([`Interp::park_rust_routine`]) answers `Entered`, its frame
431    /// gone: its continuation answers the call once the activity wakes.
432    fn run_rust_routine(
433        &mut self,
434        body: crate::extensions::RoutineBody,
435        code: usize,
436        name: &[u8],
437        args: &[Option<ObjRef>],
438        executable: crate::RoutineObject,
439    ) -> Result<Started<Option<ObjRef>>, Failure> {
440        self.push_native_frame(ObjRef::NIL, ObjRef::NIL, None, name, args, Some(code));
441        self.native_frame_mut().executable = executable;
442        pin_enter!(self, crate::pinning::PinKind::NativeApiCallback);
443        let program = self.library_code_program(code);
444        self.native_frame_mut().packaged = program.is_some();
445        let answered = match body(self, args) {
446            Ok(value) => {
447                self.roots.activity_mut().push_temp(value);
448                Some(value)
449            }
450            Err(failure) => {
451                self.hold_native_condition(failure);
452                None
453            }
454        };
455        let (ran, code) = (self.native_frame().executable, self.native_frame().code);
456        let popped = self.pop_native_frame();
457        pin_leave!(self);
458        self.collect_under_stress();
459        match popped.raised {
460            None if self.activity.native_park.is_some() => Ok(Started::Entered),
461            None => Ok(Started::Ran(answered)),
462            Some(failure) => {
463                self.reraise_in_caller(&failure);
464                self.blame_native_routine(name, program, args, ran, code);
465                Err(failure)
466            }
467        }
468    }

470    /// What a library routine answers once its stub has returned `answered`,
471    /// as [`Interp::end_library_method`] does for a method, with the routine
472    /// blamed for a failure.
473    #[inline(always)]
474    fn end_library_routine(
475        &mut self,
476        answered: Result<Option<ObjRef>, Refused>,
477        pending: Option<usize>,
478        name: &[u8],
479        program: Option<crate::ProgramId>,
480        args: &[Option<ObjRef>],
481    ) -> Result<Option<ObjRef>, Failure> {
482        self.root_native_answer(&answered);
483        let trapped = self.call_trapped_native_condition();
484        let (ran, code) = (self.native_frame().executable, self.native_frame().code);
485        let popped = self.pop_native_frame();
486        pin_leave!(self);
487        self.collect_under_stress();
488        let outcome = match (pending, trapped) {
489            (Some(number), _) => {
490                self.activity.native_reraise = true;
491                Err(condition_of(number))
492            }
493            (None, Err(failure)) => Err(failure),
494            (None, Ok(trapped)) => {
495                self.settle_native_call(answered, popped, program.is_some(), trapped)
496            }
497        };
498        if outcome.is_err() {
499            self.blame_native_routine(name, program, args, ran, code);
500        }
501        outcome
502    }

504    /// The native call `held` against `args`, its frame pushed and pinned:
505    /// its answer and the condition it holds, run back to back on the baton;
506    /// or, where it `exits`, `None` with the call prepared on the activity's
507    /// record and the activity parked, its continuation `resume` on
508    /// `receiver`, and the frame unpinned.
509    #[inline(always)]
510    fn native_call(
511        &mut self,
512        held: HeldCall,
513        args: &[Option<ObjRef>],
514        exits: bool,
515        receiver: ObjRef,
516        resume: super::NativeResume,
517    ) -> Option<(Result<Option<ObjRef>, Refused>, Option<usize>)> {
518        let mut strings = CStringPool::new();
519        let thread = self.thread_context();
520        let frame = self.native_token();
521        if !exits {
522            // Registered ahead of the activation, which holds the interpreter: nothing reads
523            // the registration but `RexxSetHalt` from another thread.
524            let _reachable = crate::embedding::OtherRun::enter(self.timer.inbox_ref());
525            let stress = self.stress_collect;
526            let pool = self.opens_variable_pool(&held);
527            // A deterministic switch mode runs every native call on the baton.
528            if self.switch.is_none() {
529                return Some(self.vacated_native_call(
530                    &held,
531                    args,
532                    &thread,
533                    frame,
534                    (pool, stress),
535                    &mut strings,
536                ));
537            }
538            let activation = Activation::new(Conversion {
539                host: self.into(),
540                strings: &mut strings,
541            });
542            activation.set_frame(frame);
543            activation.set_between_halves(stress);
544            if pool {
545                activation.open_variable_pool();
546            }
547            let answered = thread.enter(&activation, |contexts| match &held {
548                HeldCall::Method(held, _) => {
549                    invoke::held_method(held, &contexts.method(), &activation, args)
550                }
551                HeldCall::Routine(held) => {
552                    invoke::held_routine(held, &contexts.call(), &activation, args)
553                }
554            });
555            return Some((answered, activation.pending()));
556        }
557        let mut native = NativeCall::empty();
558        let (prepared, pending) = {
559            let activation = Activation::new(Conversion {
560                host: self.into(),
561                strings: &mut strings,
562            });
563            activation.set_frame(frame);
564            let prepared = thread.enter(&activation, |contexts| {
565                let signature = match &held {
566                    HeldCall::Method(held, _) => invoke::held_signature(held, &contexts.method()),
567                    HeldCall::Routine(held) => {
568                        invoke::held_routine_signature(held, &contexts.call())
569                    }
570                };
571                signature.and_then(|signature| {
572                    invoke::prepare(&mut native, &signature, &activation, args)
573                })
574            });
575            (prepared, activation.pending())
576        };
577        if let Err(refused) = prepared {
578            return Some((Err(refused), pending));
579        }
580        pin_leave!(self);
581        let method = match &held {
582            HeldCall::Method(_, method) => Some(*method),
583            HeldCall::Routine(_) => None,
584        };
585        self.activity.native_call = Some(Box::new(NativeInFlight {
586            method,
587            call: Some(Box::new(OffBaton {
588                pool: self.opens_variable_pool(&held),
589                held,
590                native,
591                strings,
592            })),
593            frame,
594            pending,
595            stage: Stage::Prepared,
596            park: None,
597        }));
598        self.activity.native_park = Some(Box::new(super::NativePark {
599            reason: crate::scheduler::ParkReason::Native,
600            resume,
601            receiver,
602            thens: Vec::new(),
603        }));
604        None
605    }

607    /// [`Interp::native_call`]'s call `held` on the baton, in the native frame
608    /// `frame` of the running activity whose thread context is `thread`, with
609    /// the baton vacated while the stub runs (`r[native.threads.attach]`): a
610    /// thread the stub waits for may call back meanwhile. Its signature, its
611    /// arguments and its result are converted on the baton. `opens` says
612    /// whether the call opens the variable pool and whether it collects
613    /// between its halves.
614    fn vacated_native_call(
615        &mut self,
616        held: &HeldCall,
617        args: &[Option<ObjRef>],
618        thread: &interrexx_api::ffi::ThreadContext,
619        frame: u64,
620        opens: (bool, bool),
621        strings: &mut CStringPool,
622    ) -> (Result<Option<ObjRef>, Refused>, Option<usize>) {
623        let (pool, stress) = opens;
624        let vacated = self.vacated(frame);
625        let (answered, pending) = {
626            let activation = Activation::new(Conversion {
627                host: HostRef::through(&vacated),
628                strings,
629            });
630            activation.set_frame(frame);
631            activation.set_between_halves(stress);
632            if pool {
633                activation.open_variable_pool();
634            }
635            let answered = thread.enter(&activation, |contexts| {
636                let mut native = NativeCall::empty();
637                let signature = match held {
638                    HeldCall::Method(held, _) => invoke::held_signature(held, &contexts.method()),
639                    HeldCall::Routine(held) => {
640                        invoke::held_routine_signature(held, &contexts.call())
641                    }
642                }?;
643                invoke::prepare(&mut native, &signature, &activation, args)?;
644                if activation.between_halves() {
645                    activation.conversion().host.between_halves();
646                }
647                let completion = vacated.around(|| held.call(&mut native, contexts));
648                if activation.between_halves() {
649                    activation.conversion().host.between_halves();
650                }
651                invoke::finish(&activation, completion)
652            });
653            (answered, activation.pending())
654        };
655        drop(vacated);
656        self.pop_vacancy(frame);
657        (answered, pending)
658    }

695    /// record, with the call's frame pinned again and, under the collect
696    /// stress mode, a collection before `finish` reads it.
697    fn resume_native_call(&mut self) -> Result<(Box<NativeInFlight>, Completion), Failure> {
698        let mut call =
699            self.activity.native_call.take().ok_or_else(|| {
700                Loud::scheduler_inconsistency("a native call resumed with no record")
701            })?;
702        self.end_root_call(call.frame());
703        let Stage::Completed(completion) = std::mem::replace(&mut call.stage, Stage::Prepared)

706                Loud::scheduler_inconsistency("a native call resumed before it completed").into(),

708        };

710        if self.stress_collect {

712        }
713        Ok((call, completion))
714    }

737    /// `failure` in place of its completion: its frame popped and its guard
738    /// lock released. The native may still be running, so the frame keeps

743        let Some(call) = self.activity.native_call.take() else {
744            return;
745        };
746        self.end_root_call(call.frame());
747        let scope = self.native_frame().scope;

749            // No native body can still use kept objects or lent buffers.

757        if popped.reserved {
758            self.release_guard(crate::guards::GuardKey {
759                object: receiver,
760                scope,
761            });
762        }

765    /// Runs `library`'s `hook` in a native frame of its own, as
766    /// `Activity::run` runs a callback dispatcher, with the baton vacated
767    /// while the hook runs (`r[native.loading.reentrant]`), as an external
768    /// call runs: a thread the hook waits for may call into this interpreter
769    /// meanwhile. A deterministic switch mode keeps the baton, as it does for
770    /// every native call. `loading`, the guard of a loader's load, is
771    /// dropped as soon as the hook has returned or unwound.
772    ///
773    /// # Errors
774    /// The condition the hook raised, and [`Loud`] for an interface member it
775    /// reached that this crate has not written.
776    pub(crate) fn run_package_hook(
777        &mut self,
778        library: &Library,
779        hook: Hook,
780        loading: Option<crate::libraries::Loading>,
781    ) -> Result<(), Failure> {
782        if !library.has_hook(hook) {
783            return Ok(());
784        }
785        // Dropped before the baton is taken back: a load waiting for this
786        // loader may hold a lend above the vacancy, through a thread the
787        // loader called into this interpreter for, and the baton comes back
788        // only once that lend has ended (`r[native.loading.shared]`).
789        let returned = move |ran| {
790            drop(loading);
791            ran
792        };
793        self.push_native_frame(ObjRef::NIL, ObjRef::NIL, None, b"", &[], None);
794        pin_enter!(self, crate::pinning::PinKind::LibraryEntry);
795        let mut strings = CStringPool::new();
796        let thread = self.thread_context();
797        let inbox = self.timer.inbox();
798        let frame = self.native_token();
799        // A loader or unloader is external code, so it runs as
800        // `run_classic_handler` runs a classic handler. The vacancy's root
801        // pointer is derived from this borrow of the whole interpreter, as
802        // the last use of `self` until it is popped, so every callback the
803        // hook makes reaches the interpreter through it.
804        let (ran, pending) = if self.switch.is_none() {
805            let vacated = self.vacated(frame);
806            let ran = {
807                let activation = Activation::new(Conversion {
808                    host: HostRef::through(&vacated),
809                    strings: &mut strings,
810                });
811                activation.set_frame(frame);
812                let _reachable = crate::embedding::OtherRun::enter(&inbox);
813                let ran = thread.enter(&activation, |contexts| {
814                    vacated.around(|| returned(invoke::hook(library, hook, contexts, &activation)))
815                });
816                (ran, activation.pending())
817            };
818            drop(vacated);
819            self.pop_vacancy(frame);
820            ran
821        } else {
822            let activation = Activation::new(Conversion {
823                host: self.into(),
824                strings: &mut strings,
825            });
826            activation.set_frame(frame);
827            let _reachable = crate::embedding::OtherRun::enter(&inbox);
828            let ran = thread.enter(&activation, |contexts| {
829                returned(invoke::hook(library, hook, contexts, &activation))
830            });
831            (ran, activation.pending())
832        };
833        let popped = self.pop_native_frame();
834        pin_leave!(self);
835        if let Some(number) = pending {
836            return Err(condition_of(number));
837        }
838        let settled = self.settle_native_call(ran.map(|()| None), popped, true, None);
839        // A hook is no level of the failure's, so nothing re-raises it.
840        self.activity.native_reraise = false;
841        self.activity.native_reported_on_return = false;
842        settled.map(|_| ())
843    }

845    /// Runs the registered command handler `handler` for `command`, issued to
846    /// `environment`, in a native frame of its own, as `Activity::run` runs a
847    /// context command-handler dispatcher.
848    ///
849    /// # Errors
850    /// A condition a string conversion raised, and [`Loud`] for an interface
851    /// member the handler reached that this crate has not written. A
852    /// condition the handler raised is answered, not returned as an error.
853    pub(crate) fn run_command_handler(
854        &mut self,
855        handler: &CommandHandler,
856        environment: &[u8],
857        command: &[u8],
858        redirector: &Redirector,
859    ) -> Result<HandledCommand, Failure> {
860        let address = self.text(environment);
861        self.roots.activity_mut().push_temp(address);
862        let issued = self.text(command);
863        self.roots.activity_mut().push_temp(issued);
864        self.push_native_frame(ObjRef::NIL, ObjRef::NIL, None, b"", &[], None);
865        pin_enter!(self, crate::pinning::PinKind::NativeApiCallback);
866        let mut strings = CStringPool::new();
867        let thread = self.thread_context();
868        let inbox = self.timer.inbox();
869        let (answered, pending) = {
870            let frame = self.native_token();
871            let activation = Activation::new(Conversion {
872                host: self.into(),
873                strings: &mut strings,
874            });
875            activation.set_frame(frame);
876            let _reachable = crate::embedding::OtherRun::enter(&inbox);
877            let answered = thread.enter(&activation, |contexts| {
878                invoke::command(handler, contexts, &activation, address, issued, redirector)
879            });
880            (answered, activation.pending())
881        };
882        let popped = self.pop_native_frame();
883        pin_leave!(self);
884        if let Some(number) = pending {
885            return Err(condition_of(number));
886        }
887        let value = match answered {
888            Ok(value) => value,
889            Err(Refused::Raised) => {
890                return Err(popped
891                    .raised
892                    .expect("a host answering Raised holds the condition it raised"));
893            }
894            Err(refused) => return Err(self.refusal(refused, true, false)),
895        };
896        if let Some(value) = value {
897            self.roots.activity_mut().push_temp(value);
898        }
899        Ok(HandledCommand {
900            value,
901            raised: popped.raised,
902            additional: popped.additional,
903            result: popped.result,
904        })
905    }

907    /// `Interpreter::terminateInterpreter`'s two steps that run extension or
908    /// Rexx code: the last-chance `UNINIT`s, then the package unloaders.
909    /// Answers the refusals they met.
910    ///
911    /// Both run on a thread context of their own, as they run on the instance
912    /// created to terminate on: the running activity's context is retired
913    /// ([`interrexx_api::ffi::ThreadContext::retire`]) and set aside for the
914    /// length of the two steps, so the first native call or hook among them
915    /// makes a new one.
916    pub(crate) fn terminate(&mut self) -> Vec<Loud> {
917        // A thread that attaches or calls back from now on is not served: the
918        // attached activities end with the interpreter (`r[native.threads.end]`).
919        // A `Halt` or `SetTrace` through the instance still reaches the
920        // last-chance `UNINIT`s and the unloaders, and posts nothing once
921        // they have run.
922        self.timer.inbox().end();
923        let running = self.activity.thread.take();
924        if let Some(running) = &running {
925            running.retire();
926        }
927        let mut refused = self.run_termination_uninits();
928        refused.extend(self.run_package_unloaders());
929        let terminated = std::mem::replace(&mut self.activity.thread, running);
930        if terminated.is_some() {
931            self.termination_thread = terminated;
932        }
933        self.timer.inbox().close();
934        refused
935    }

937    /// each held library's unloader, in the order the library table walks them,
938    /// each library closed straight after its own, answering the refusal that
939    /// ended the walk if one did. For example, a condition raised inside an
940    /// unloader ends the walk, reports nothing, leaves the exit status alone
941    /// and leaves that library and every later one open. A refusal ends it too.
942    ///
943    /// The libraries are those of every instance of the process, and they are
944    /// walked only where this interpreter is the last live instance
945    /// (`r[native.loading.unload]`).
946    fn run_package_unloaders(&mut self) -> Option<Loud> {
947        // Held until the walk ends, however it ends, so that no instance is
948        // created while an unloader runs (`r[native.loading.again]`).
949        let mut unloading = self.libraries.end()?;
950        // A library an unloader loads is walked after the ones before it, in
951        // the same termination (`r[native.loading.unload]`).
952        let refused = 'walk: loop {
953            let mut batch = unloading.next();
954            if batch.is_empty() {
955                match unloading.finish() {
956                    crate::libraries::Finish::Done => break None,
957                    crate::libraries::Finish::More(more) => batch = more,
958                    // A loader another thread runs through this instance may
959                    // call into it, so the wait vacates the baton.
960                    crate::libraries::Finish::Wait(holders) => {
961                        self.wait_vacated(|| crate::libraries::await_holders(holders));
962                        continue;
963                    }
964                }
965            }
966            for (_, unload) in batch {
967                let library = unload.library();
968                match self.run_package_hook(&library, Hook::Unloader, None) {
969                    Ok(()) => {
970                        // A registered handler's code may be in this library, and
971                        // nothing ties the two together.
972                        if library.close() {
973                            self.command_handlers.clear();
974                            self.environments.close_registered();
975                            self.libraries.note_closed();
976                        }
977                    }
978                    Err(Failure::Loud(loud)) => break 'walk Some(*loud),
979                    Err(_) => break 'walk None,
980                }
981            }
982        };
983        self.libraries.unloaded(unloading);
984        refused
985    }

988    /// vacated (`r[native.threads.attach]`), in a native frame of its own, as
989    /// an external call runs: a thread that `wait` waits for may call into
990    /// this interpreter meanwhile. A deterministic switch mode keeps the
991    /// baton, as it does for every native call.
992    pub(crate) fn wait_vacated(&mut self, wait: impl FnOnce()) {
993        self.push_native_frame(ObjRef::NIL, ObjRef::NIL, None, b"", &[], None);
994        pin_enter!(self, crate::pinning::PinKind::NativeApiCallback);
995        if self.switch.is_none() {

997            let vacated = self.vacated(frame);
998            vacated.around(wait);
999            drop(vacated);

1001        } else {
1002            wait();
1003        }
1004        let _ = self.pop_native_frame();
1005        pin_leave!(self);
1006    }

1009    /// the condition its string conversion raised, or the boundary's own
1010    /// refusal. `packaged` is whether the code reports a package, which is
1011    /// what a refusal before the call is reported against with no line; one
1012    /// that reports none is reported against the caller's clause.
1013    ///
1014    /// `trapped` is the object [`Interp::call_trapped_native_condition`]
1015    /// built before the frame came off.
1016    fn settle_native_call(
1017        &mut self,
1018        answered: Result<Option<ObjRef>, Refused>,
1019        popped: Popped,
1020        packaged: bool,
1021        trapped: Option<ObjRef>,
1022    ) -> Result<Option<ObjRef>, Failure> {
1023        match answered {
1024            Err(Refused::Raised) => {
1025                let raised = popped
1026                    .raised
1027                    .expect("a host answering Raised holds the condition it raised");
1028                self.reraise_in_caller(&raised);
1029                Err(raised)
1030            }
1031            Ok(value) => match popped.raised {
1032                None => Ok(value),
1033                Some(raised) => {
1034                    let held = (raised, popped.additional, popped.result);
1035                    self.raise_held_condition(held, value, trapped)
1036                }
1037            },
1038            Err(refused) => {
1039                let failure = self.refusal(refused, packaged, popped.method);
1040                self.reraise_in_caller(&failure);
1041                Err(failure)
1042            }
1043        }
1044    }

1046    /// A condition a callback raised and the extension did not clear, raised
1047    /// once the call has returned: a `SYNTAX` condition as the call's failure;
1048    /// any other is offered to the caller's trap, and the call answers the
1049    /// condition's `RESULT` where no `SIGNAL ON` takes it, or else `returned`,
1050    /// what the call itself returned. As with `RAISE`, an untrapped `HALT` or
1051    /// `NOMETHOD` is an error. `held` is the condition with its `ADDITIONAL`
1052    /// and `RESULT` objects.
1053    fn raise_held_condition(
1054        &mut self,
1055        held: (Failure, Option<ObjRef>, Option<ObjRef>),
1056        returned: Option<ObjRef>,
1057        trapped: Option<ObjRef>,
1058    ) -> Result<Option<ObjRef>, Failure> {
1059        let (raised, additional, result) = held;
1060        let answer = result.or(returned);
1061        let condition = match &raised {
1062            Failure::Raised(held) if held.condition != "SYNTAX" => held.condition.to_string(),
1063            _ => {
1064                if additional.is_some() {
1065                    self.activity.pending_additional = additional;
1066                }
1067                self.activity.native_reraise = true;
1068                return Err(raised);
1069            }
1070        };
1071        let Failure::Raised(held) = &raised else {
1072            unreachable!("matched as a raise above")
1073        };
1074        match self.trap_for(condition.as_bytes()) {
1075            Some(trap) if trap.call => {
1076                let object = match trapped {
1077                    Some(object) => object,
1078                    None => {
1079                        self.activity.pending_additional = additional;
1080                        self.activity.pending_result = result;
1081                        self.build_condition_object(held, Some(true))?
1082                    }
1083                };
1084                self.activity.pending_traps.push_back(PendingTrap {
1085                    condition: condition.as_bytes().into(),
1086                    rc: None,
1087                    description: held.description.clone(),
1088                    object: Some(object),
1089                    activation: self.activation().id,
1090                    queued_during_delivery: false,
1091                    request: false,
1092                    fragment_depth: self.activity.fragment_depth,
1093                    trace_failure: None,
1094                });
1095                Ok(answer)
1096            }
1097            Some(_) => {
1098                self.activity.pending_additional = additional;
1099                self.activity.pending_result = result;
1100                Err(raised)
1101            }
1102            None if condition == "HALT" => {
1103                self.activity.native_reported_on_return = true;
1104                Err(Raised::halt().into())
1105            }
1106            None if condition == "NOMETHOD" => {
1107                let raised = match (additional, held.description.as_deref()) {
1108                    (Some(receiver), Some(description)) => {
1109                        // The "no method name" error's substitutions are the
1110                        // receiver object itself and the message name.
1111                        let name = self.text(description);
1112                        self.roots.activity_mut().push_temp(name);
1113                        let array = interrexx_api::callbacks::Surface::new_array(
1114                            self,
1115                            &[Some(receiver), Some(name)],
1116                        );
1117                        self.roots.activity_mut().push_temp(array);
1118                        let receiver = self.string_value_text(receiver)?;
1119                        self.activity.pending_additional = Some(array);
1120                        Raised::no_method(&receiver, description)
1121                    }
1122                    _ => Raised::syntax(97, 5, Vec::new()),
1123                };
1124                self.activity.native_reported_on_return = true;
1125                Err(raised.into())
1126            }
1127            None => Ok(answer),
1128        }
1129    }

1157    /// Pushes the frame of a native call: a method's when `receiver` is
1158    /// given, a routine's otherwise.
1159    fn push_native_frame(
1160        &mut self,
1161        owner: ObjRef,
1162        scope: ObjRef,
1163        receiver: Option<ObjRef>,
1164        name: &[u8],
1165        args: &[Option<ObjRef>],
1166        code: Option<usize>,
1167    ) {
1168        // A native call pushes no activation, and through `SetObjectVariable`
1169        // or `SetContextVariable` it can assign a tail variable all the same.
1170        self.shared_key_for = None;
1171        let mut frame = self.activity.native_spares.pop().unwrap_or_else(|| {
1172            Box::new(NativeFrame {
1173                owner: ObjRef::NIL,
1174                scope: ObjRef::NIL,
1175                method: false,
1176                receiver: ObjRef::NIL,
1177                name: Vec::new(),
1178                arguments: Vec::new(),
1179                argument_list: None,
1180                locals: Table::new(&self.handle_space),
1181                raised: None,
1182                additional: None,
1183                result: None,
1184                condition: None,
1185                callee: false,
1186                levels: None,
1187                callerless: false,
1188                code: None,
1189                executable: crate::RoutineObject::OfCode,
1190                kept: rustc_hash::FxHashSet::default(),
1191                lent: Vec::new(),
1192                caller: None,
1193                id: 0,
1194                packaged: false,
1195                reserved: false,
1196                temps_floor: 0,
1197            })
1198        });
1199        frame.owner = owner;
1200        frame.scope = scope;
1201        frame.method = receiver.is_some();
1202        frame.receiver = receiver.unwrap_or(ObjRef::NIL);
1203        frame.code = code;
1204        frame.executable = crate::RoutineObject::OfCode;
1205        frame.caller = self.running_activation().map(|activation| activation.id);
1206        // A native routine a `CallRoutine` call runs has no calling
1207        // activation, though it runs above the one the call was made from.

1276    /// Pops the innermost native frame and takes its held condition out of
1277    /// it, leaving the rest held.
1278    fn take_native_frame(&mut self) -> (Popped, Box<crate::NativeFrame>) {
1279        self.sync_api_changes();
1280        let mut frame = self
1281            .activity
1282            .native_handles
1283            .pop()
1284            .expect("the frame pushed for this call is still the innermost");
1285        // A condition `RAISE`d with no Rexx caller stops here.
1286        if std::mem::take(&mut frame.callee) {
1287            frame.raised = None;
1288            frame.additional = None;
1289            frame.result = None;
1290        }
1291        let answer = Popped {
1292            raised: frame.raised.take(),
1293            method: frame.method,
1294            reserved: std::mem::take(&mut frame.reserved),
1295            additional: frame.additional.take(),
1296            result: frame.result.take(),
1297        };
1298        frame.condition = None;
1299        // The levels of a condition the call raises again go back on the
1300        // activity for its report; a cleared one's are forgotten.
1301        if let Some(levels) = frame.levels.take()
1302            && answer.raised.is_some()
1303        {
1304            self.restore_failure_levels(levels);
1305        }
1306        for object in [answer.additional, answer.result].into_iter().flatten() {
1307            self.roots.activity_mut().push_temp(object);
1308        }
1309        (answer, frame)
1310    }

1312    /// Clears `frame` for reuse, ending its hold on its kept strings and lent
1313    /// buffers.
1314    fn end_native_frame(&mut self, frame: &mut crate::NativeFrame) {
1315        frame.name.clear();
1316        frame.arguments.clear();
1317        frame.argument_list = None;
1318        frame.locals.clear();
1319        // A drain walks every bucket of a table that once held a kept string, empty or not.
1320        if !frame.kept.is_empty() {
1321            for object in frame.kept.drain() {
1322                self.release_kept(object);
1323            }
1324        }
1325        for buffer in frame.lent.drain(..) {
1326            if let Some(state) = self.buffer_mut(buffer) {
1327                state.bytes.release();
1328            }
1329        }
1330    }

1332    /// Drops the kept copy of every handle-carried value no call in flight
1333    /// and no global reference holds, as a collection does. These copies
1334    /// grow without the heap growing, so a run of calls that allocates
1335    /// nothing would otherwise never meet a collection; this runs whenever
1336    /// their number passes twice what the last pass left, and 4096.
1337    pub(crate) fn drop_loose_kept_strings(&mut self) {
1338        let (holders, globals) = (&self.kept_holders, &self.global_references);
1339        self.kept_strings.retain(|object, _| {
1340            matches!(object.decode(), interrexx_core::Decoded::Heap { .. })
1341                || holders.contains_key(object)
1342                || globals.holds(*object)
1343        });
1344        self.kept_carried = self
1345            .kept_strings
1346            .keys()
1347            .filter(|object| !matches!(object.decode(), interrexx_core::Decoded::Heap { .. }))
1348            .count();
1349        self.kept_carried_limit = (self.kept_carried * 2).max(4096);
1350    }

1352    /// One native call's end for a handle-carried value's kept copy, which
1353    /// the next collection may then drop.
1354    fn release_kept(&mut self, object: ObjRef) {
1355        let std::collections::hash_map::Entry::Occupied(mut holders) =
1356            self.kept_holders.entry(object)
1357        else {
1358            return;
1359        };
1360        *holders.get_mut() -= 1;
1361        if *holders.get() == 0 {
1362            holders.remove();
1363        }
1364    }

1366    /// What the boundary's own refusal reports.
1367    fn refusal(&mut self, refused: Refused, packaged: bool, method: bool) -> Failure {
1368        let mut raised = match refused {
1369            Refused::MissingArgument { position } => Raised::missing_native_argument(position),
1370            Refused::NoStringValue { position } => {
1371                Raised::native_argument_needs_a_string_value(position)
1372            }
1373            // `found` is the argument's `stringValue()`, which is not its string
1374            // conversion: an array is `found "an Array".` for 88.921 and 88.905
1375            // where its string conversion joins its items. `NotLogical` below is
1376            // the exception: its `found` is the string the conversion answered,
1377            // `found "1\n2"` for that same array.
1378            Refused::InvalidDouble { position, argument } => {
1379                let found = self.native_found(argument);
1380                Raised::native_argument_not_a_double(position, &found)
1381            }
1382            Refused::NotPositive { position, argument } => {
1383                let found = self.native_found(argument);
1384                Raised::native_argument_not_positive(position, &found)
1385            }
1386            Refused::NotNonnegative { position, argument } => {
1387                let found = self.native_found(argument);
1388                Raised::native_argument_not_nonnegative(position, &found)
1389            }
1390            Refused::OutOfRange {
1391                position,
1392                min,
1393                max,
1394                argument,
1395            } => {
1396                let found = self.native_found(argument);
1397                Raised::native_argument_outside_range(position, min, max, &found)
1398            }
1399            Refused::NotLogical { found } => {
1400                let found = self.native_found(found);
1401                Raised::native_argument_not_logical(&found)
1402            }
1403            Refused::NotArray { argument } => {
1404                let found = self.native_found(argument);
1405                Raised::native_argument_not_an_array(&found)
1406            }
1407            Refused::NotInstance { position, class } => {
1408                Raised::native_argument_not_an_instance(position, class.id())
1409            }
1410            Refused::NotPointerString { position, argument } => {
1411                let found = self.native_found(argument);
1412                Raised::native_argument_not_a_pointer(position, &found)
1413            }
1414            Refused::NoStem { position, argument } => {
1415                let found = self.native_found(argument);
1416                Raised::native_argument_not_a_stem(method, position, &found)
1417            }
1418            Refused::TooManyArguments { expected } => Raised::too_many_external_arguments(expected),
1419            Refused::Signature | Refused::ResultSignature => {
1420                let number = refused
1421                    .error_number(method)
1422                    .expect("a signature refusal carries its number");
1423                Raised::incorrect_native_signature(number, refused == Refused::Signature)
1424            }
1425            Refused::ClassicStyle => return Loud::classic_style_routine().into(),
1426            Refused::UnfilledSlot { entry } => return Loud::unfilled_slot(entry).into(),
1427            // A handle naming no live object stays refused: a kept handle
1428            // whose object was collected, or an address no handle encodes,
1429            // is a use the API does not define (Deviation 95).
1430            Refused::StaleHandle => return Loud::stale_handle().into(),
1431            Refused::Raised => return Loud::conversion_raised().into(),
1432        };
1433        if !packaged {
1434            raised.delivery.lineless = false;
1435        }
1436        raised.into()
1437    }
1438}

1598/// A library method's continuation once its call has completed: `finish`,
1599/// then its end and its blame, as [`Interp::run_other`] blames.
1600fn resume_library_method(interp: &mut Interp, receiver: ObjRef) -> Result<Option<ObjRef>, Failure> {
1601    let (mut call, completion) = interp.resume_native_call()?;
1602    let frame = interp.native_frame();
1603    let (scope, name, args, packaged) = (
1604        frame.scope,
1605        frame.name.clone(),
1606        frame.arguments.clone(),
1607        frame.packaged,
1608    );
1609    let (answered, pending) = interp.finish_native_call(&call, completion);
1610    let Some(HeldCall::Method(_, method)) = call.call.take().map(|call| call.held) else {
1611        return Err(Loud::scheduler_inconsistency("a method resumed from a routine's call").into());
1612    };
1613    drop(call);
1614    let outcome = interp.end_library_method(answered, pending, receiver, scope, packaged);
1615    if outcome.is_err() {
1616        let scope_name = interp.scope_id(scope);
1617        let reraised = std::mem::take(&mut interp.activity.native_reraise);
1618        interp.blame_external_method(&name, &scope_name, method, receiver, &args, reraised);
1619    }
1620    outcome
1621}

1623/// A library routine's continuation once its call has completed, as
1624/// [`resume_library_method`] is a method's.
1625fn resume_library_routine(
1626    interp: &mut Interp,
1627    _receiver: ObjRef,
1628) -> Result<Option<ObjRef>, Failure> {
1629    let (call, completion) = interp.resume_native_call()?;
1630    let frame = interp.native_frame();
1631    let (name, args, code) = (frame.name.clone(), frame.arguments.clone(), frame.code);
1632    let program = code.and_then(|code| interp.library_code_program(code));
1633    let (answered, pending) = interp.finish_native_call(&call, completion);
1634    drop(call);
1635    interp.end_library_routine(answered, pending, &name, program, &args)
1636}

1692/// lends it the baton to enter the call, and it recalls the baton for each
1693/// callback and for the call's end, so that each runs as the call's activity.
1694pub(crate) struct PooledCall {
1695    work: Islanded<(Box<OffBaton>, interrexx_api::ffi::ThreadContext)>,
1696    frame: u64,
1697    pending: Option<usize>,
1698    activity: ActivityId,
1699    baton: Arc<InterpBaton>,
1700    inbox: Arc<Inbox<Posted>>,
1701}
1702
1703impl PooledCall {
1704    /// The call `call` in the native frame `frame` holding the condition
1705    /// `pending`, with `thread` as `activity`'s thread context.
1706    pub(crate) fn new(
1707        call: Box<OffBaton>,
1708        thread: interrexx_api::ffi::ThreadContext,
1709        frame: u64,
1710        pending: Option<usize>,
1711        activity: ActivityId,
1712        baton: &Arc<InterpBaton>,
1713        inbox: Arc<Inbox<Posted>>,
1714    ) -> PooledCall {
1715        PooledCall {
1716            work: Islanded::new((call, thread), baton),
1717            frame,
1718            pending,
1719            activity,
1720            baton: Arc::clone(baton),
1721            inbox,
1722        }
1723    }
1724
1725    /// On the pool thread: enters the call under the lend the holder makes,
1726    /// gives the baton back, calls the stub, and once it returns recalls the
1727    /// baton to leave the call, put its parts back on the activity's record
1728    /// and post its completion. A stub that panics recalls it to unwind.
1729    pub(crate) fn run(self, posts: &Inbox<Posted>, bound: Option<std::time::Duration>) {
1730        let probe = 0u8;
1731        let recalling = Recalling {
1732            baton: &self.baton,
1733            inbox: posts,
1734            activity: self.activity,
1735            frame: self.frame,
1736            base: (&raw const probe).addr(),
1737        };
1738        let entered = Lent::wait(&self.baton);
1739        let (mut call, thread) = self.work.take(&self.baton);
1740        let (completion, mut ended, pending, blocked) = {
1741            let OffBaton {
1742                held,
1743                native,
1744                strings,
1745                pool,
1746            } = &mut *call;
1747            let activation = Activation::new(Conversion {
1748                host: HostRef::through(&recalling),
1749                strings,
1750            });
1751            activation.set_frame(self.frame);
1752            if let Some(number) = self.pending {
1753                activation.raise(number);
1754            }
1755            if *pool {
1756                activation.open_variable_pool();
1757            }
1758            let _reachable = crate::embedding::OtherRun::enter(&self.inbox);
1759            let (completion, ended, blocked) = thread.enter(&activation, |contexts| {
1760                drop(entered);
1761                let watch = bound.map(crate::sim::native_watch::Watch::start);
1762                let completion = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
1763                    held.call(native, contexts)
1764                }));
1765                let blocked = watch.is_some_and(crate::sim::native_watch::Watch::finish);
1766                let ended = recalling.recall();
1767                match completion {
1768                    Ok(completion) => (completion, ended, blocked),
1769                    // Unwound under the lend: the call's state drops on the
1770                    // baton.
1771                    Err(payload) => std::panic::resume_unwind(payload),
1772                }
1773            });
1774            (completion, ended, activation.pending(), blocked)
1775        };
1776        let interp = ended.interp();
1777        if blocked {
1778            interp.sim_mark_blocked_native();
1779        }
1780        #[cfg(test)]
1781        if let Some(path) = &interp.panic_at_call_end {
1782            use std::io::Write;
1783            let mut file = std::fs::OpenOptions::new()
1784                .append(true)
1785                .create(true)
1786                .open(path)
1787                .expect("the test's file");
1788            writeln!(file, "end").expect("a line written");

1831/// The baton as a native call made on it reaches it while it runs with the
1832/// baton vacated ([`Interp::vacated`]): the call's own thread takes the
1833/// vacancy back for a callback where nobody holds it, as a kernel lock is
1834/// taken, and recalls the baton from the thread serving the vacancy where one
1835/// does, as a pool thread recalls it for a callback of its call.
1836pub(crate) struct Vacated {
1837    baton: Arc<InterpBaton>,
1838    inbox: Arc<Inbox<Posted>>,
1839    activity: ActivityId,
1840    frame: u64,
1841    /// The calling thread's stack base and room, for Rexx code a callback
1842    /// runs under the serving thread's lend.
1843    stack: (usize, usize),
1844    /// The vacancy's depth while the stub runs.
1845    depth: std::cell::Cell<Option<usize>>,
1846    /// The interpreter's root pointer, derived from the calling thread's own
1847    /// borrow, which reaches it until this drops.
1848    own: crate::island::Island,
1849}
1850
1851impl Vacated {
1852    /// Runs `call`, which touches nothing of the interpreter's but through
1853    /// callbacks, with the baton vacated, and takes it back: directly where
1854    /// it is free, else once the thread it was handed to has seen the call
1855    /// return and given it back.
1856    pub(crate) fn around<R>(&self, call: impl FnOnce() -> R) -> R {
1857        let depth = self.baton.vacate(self.own);
1858        self.depth.set(Some(depth));
1859        let answer = call();
1860        if !self.baton.reclaim_free(depth) {
1861            self.inbox.post(Posted::Returned {
1862                activity: self.activity,
1863                frame: self.frame,
1864            });
1865            self.baton.await_return(depth);
1866        }
1867        self.depth.set(None);
1868        answer
1869    }
1870}

1873    fn take_unless_held(&self) -> bool {
1874        if self.baton.held_here() {
1875            return false;
1876        }
1877        // A thread that claimed the vacancy may find no pool thread to hand it
1878        // to and free it again. Only a thread that serves the vacancy, or
1879        // holds a lend above it, serves a recall, so the recall is posted only
1880        // then, and this thread, whose call the vacancy is, never claims it.
1881        if let Some(depth) = self.depth.get()
1882            && self.baton.take_vacancy(depth)
1883        {
1884            return true;
1885        }
1886        self.inbox.post(Posted::Recall(Recall {
1887            by: Recaller::Call {
1888                activity: self.activity,
1889                frame: self.frame,
1890            },
1891            thread: std::thread::current().id(),
1892            base: Some(self.stack.0),
1893            room: Some(self.stack.1),
1894        }));
1895        std::mem::forget(crate::island::Lent::wait(&self.baton));
1896        true
1897    }

1908    /// callback, else the calling thread's own.

1910        let lent = (self.depth.get()).and_then(|depth| self.baton.lent_above(depth));

1912    }

2578    pub(crate) fn borrow_native_handle(&mut self, object: ObjRef) {
2579        let held = self.handle_space.peek(object);

2762/// `NumberString::newInstanceFromDouble(value, precision)` as its string
2763/// value: `%.*g` at two digits past the precision, capped at sixteen, then
2764/// rounded to the precision and formatted at it.
2765pub(crate) fn double_text(value: f64, precision: usize) -> Vec<u8> {
2766    if value.is_nan() {
2767        return b"nan".to_vec();
2768    }
2769    if value == f64::INFINITY {
2770        return b"+infinity".to_vec();
2771    }
2772    if value == f64::NEG_INFINITY {
2773        return b"-infinity".to_vec();
2774    }
2775    if precision == 0 {
2776        return no_digit_text(value);
2777    }
2778    let printed = percent_g(value, precision.min(16) + 2);
2779    let digits = u64::try_from(precision).unwrap_or(u64::MAX);
2780    Number::parse(&printed)
2781        .unwrap_or_else(|| unreachable!("%g prints a Rexx number: {printed}"))
2782        .into_round(digits)
2783        .format(digits)
2784        .into_bytes()
2785}

2787/// A new number at precision 0: truncation keeps no digit and moves only the
2788/// exponent, rounding adds one to it where the first digit dropped is 5 or
2789/// more, and `stringValue` renders the digitless number: the sign alone at
2790/// exponent 0, else `0` with any exponent past the first place.
2791fn no_digit_text(value: f64) -> Vec<u8> {
2792    if value == 0.0 {
2793        return b"0".to_vec();
2794    }
2795    let printed = percent_g(value, 2);
2796    let (sign, body) = match printed.strip_prefix('-') {
2797        Some(rest) => ("-", rest),
2798        None => ("", printed.as_str()),
2799    };
2800    let (mantissa, scale) = match body.split_once('e') {
2801        Some((mantissa, scale)) => (
2802            mantissa,
2803            scale
2804                .parse::<i64>()
2805                .unwrap_or_else(|_| unreachable!("%g prints a decimal exponent: {printed}")),
2806        ),
2807        None => (body, 0),
2808    };
2809    let (whole, fraction) = mantissa.split_once('.').unwrap_or((mantissa, ""));
2810    let digits = format!("{whole}{fraction}");
2811    let digits = digits.trim_start_matches('0');
2812    let width = |text: &str| i64::try_from(text.len()).unwrap_or(i64::MAX);
2813    let mut exponent = scale - width(fraction) + width(digits);
2814    if digits
2815        .as_bytes()
2816        .first()
2817        .is_some_and(|first| *first >= b'5')
2818    {
2819        exponent += 1;
2820    }
2821    if exponent == 0 {
2822        return sign.as_bytes().to_vec();
2823    }
2824    let shown = exponent - 1;
2825    let mut out = format!("{sign}0");
2826    if shown != 0 {
2827        let direction = if shown > 0 { '+' } else { '-' };
2828        out.push_str(&format!("E{direction}{}", shown.unsigned_abs()));
2829    }
2830    out.into_bytes()
2831}

2833/// C's `%.*g` with `significant` digits: the style `%e` would take where its
2834/// exponent is below -4 or not below `significant`, the style `%f` takes
2835/// otherwise, and in either no trailing zero after the point.
2836fn percent_g(value: f64, significant: usize) -> String {
2837    let scientific = format!("{:.*e}", significant - 1, value);
2838    let (mantissa, exponent) = scientific
2839        .split_once('e')
2840        .unwrap_or_else(|| unreachable!("{{:e}} prints an exponent: {scientific}"));
2841    let exponent: i64 = exponent
2842        .parse()
2843        .unwrap_or_else(|_| unreachable!("{{:e}} prints a decimal exponent: {scientific}"));
2844    let (sign, mantissa) = match mantissa.strip_prefix('-') {
2845        Some(rest) => ("-", rest),
2846        None => ("", mantissa),
2847    };
2848    let digits: String = mantissa.chars().filter(char::is_ascii_digit).collect();
2849    let width = i64::try_from(significant).unwrap_or(i64::MAX);
2850    if exponent < -4 || exponent >= width {
2851        let kept = digits.trim_end_matches('0');
2852        let kept = if kept.is_empty() { "0" } else { kept };
2853        let (first, rest) = kept.split_at(1);
2854        let point = if rest.is_empty() { "" } else { "." };
2855        return format!("{sign}{first}{point}{rest}e{exponent}");
2856    }
2857    if exponent >= 0 {
2858        let (whole, fraction) = digits.split_at(usize::try_from(exponent + 1).unwrap_or(0));
2859        let fraction = fraction.trim_end_matches('0');
2860        if fraction.is_empty() {
2861            return format!("{sign}{whole}");
2862        }
2863        return format!("{sign}{whole}.{fraction}");
2864    }
2865    let zeros = "0".repeat(usize::try_from(-exponent - 1).unwrap_or(0));
2866    format!("{sign}0.{zeros}{}", digits.trim_end_matches('0'))
2867}
