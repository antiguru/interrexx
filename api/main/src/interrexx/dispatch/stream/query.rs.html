7/// `query_exists`: the qualified name when it names a file, and the null string
8/// otherwise -- a directory answers nothing, because the check is for a regular
9/// file (`class.stream.query`).
10pub(in crate::dispatch) fn query_exists(
11    interp: &mut Interp,
12    _cleared: Cleared,
13    receiver: ObjRef,
14    _args: &[Option<ObjRef>],
15) -> Result<Option<ObjRef>, Failure> {
16    let state = require(interp, receiver)?;
17    // A `HANDLE:` stream names a descriptor, not a file: nothing exists by
18    // its name, open or not (`r[bif.stream.handle]`).
19    if state.handle.is_some() || state.descriptor {
20        return Ok(Some(interp.text(b"")));
21    }
22    // An open standard stream's qualified name is its name as written
23    // (`StreamInfo::openStd`), which it answers without a lookup.
24    if state.standard.is_some() && state.status != StreamStatus::Unknown {
25        let answer = state.name.clone();
26        return Ok(Some(interp.text_built(answer)));
27    }
28    // `StreamInfo::streamExists`: an open stream answers without a lookup, a
29    // device by the name as written and anything else by its qualified name.
30    if let Some(open) = state
31        .open
32        .as_ref()
33        .filter(|_| state.status != StreamStatus::Unknown)
34    {
35        let answer = if is_device(open.file()) {
36            state.name.clone()
37        } else {
38            state.qualified.to_vec()
39        };
40        return Ok(Some(interp.text_built(answer)));
41    }
42    // Closing an owned standard alias retains its descriptor and qualification
43    // for reopening, but EXISTS must still check the ordinary named file.
44    let retained_standard_name = state.standard.is_some() && state.open.is_some();
45    let path = qualified_path(interp, receiver)?;
46    let exists = std::fs::metadata(&path).is_ok_and(|meta| meta.is_file());
47    let answer = if exists {
48        if retained_standard_name {
49            require(interp, receiver)?.qualified.to_vec()
50        } else {
51            path.into_os_string().into_encoded_bytes()
52        }
53    } else {
54        Vec::new()
55    };
56    Ok(Some(interp.text_built(answer)))
57}

59/// Whether this stream uses the embedding's owned regular-file stdout descriptor.
60pub(super) fn owned_standard_output(interp: &Interp, receiver: ObjRef) -> Result<bool, Failure> {
61    Ok(
62        require(interp, receiver)?.standard == Some(StandardStream::Out)
63            && !interp.standard_transient[1]
64            && interp
65                .sinks
66                .as_ref()
67                .is_some_and(|sinks| sinks.own_descriptors),
68    )
69}

71/// `query_size`: the byte count for a regular file, `0` for anything else
72/// that stats, and the null string when nothing does. Measured: a directory
73/// answers `0` here where `query_exists` answers nothing. An open stream
74/// answers from its descriptor, once its held writes are in it
75/// (`StreamInfo::getStreamSize`).
76pub(in crate::dispatch) fn query_size(
77    interp: &mut Interp,
78    _cleared: Cleared,
79    receiver: ObjRef,
80    _args: &[Option<ObjRef>],
81) -> Result<Option<ObjRef>, Failure> {
82    let unopened = unopened_standard(interp, receiver)?;
83    if !unopened && let Some(handle) = require(interp, receiver)?.handle {
84        let size = standard_handle_size(interp, handle);
85        return Ok(Some(interp.text_built(size.to_string().into_bytes())));
86    }
87    if !unopened && owned_standard_output(interp, receiver)? {
88        interp.hand_over_stdout();
89        if owned_stdout_alias(interp, receiver)? {
90            if let Some(error) = ensure_owned_stdout_file(interp, receiver)? {
91                return open_failed(interp, receiver, &error);
92            }
93        } else {
94            #[cfg(unix)]
95            let size = rustix::fs::fstat(std::io::stdout()).ok().and_then(|stat| {
96                (rustix::fs::FileType::from_raw_mode(stat.st_mode)
97                    == rustix::fs::FileType::RegularFile)
98                    .then_some(stat.st_size)
99            });
100            #[cfg(windows)]
101            let size = crate::host::standard_file(1)
102                .ok()
103                .and_then(|file| file.metadata().ok())
104                .filter(std::fs::Metadata::is_file)
105                .map(|meta| meta.len());
106            return Ok(Some(match size {
107                Some(size) => interp.text_built(size.to_string().into_bytes()),
108                None => interp.text(b""),
109            }));
110        }
111    }
112    let open = if unopened {
113        None
114    } else if standard_input_file(interp, receiver)? {
115        interp.input_file()
116    } else {
117        require_mut(interp, receiver)?.open.as_mut()
118    };
119    if let Some(open) = open {
120        let size = open
121            .flushed()
122            .metadata()
123            .ok()
124            .filter(std::fs::Metadata::is_file)
125            .map_or(0, |meta| meta.len());
126        return Ok(Some(interp.text_built(size.to_string().into_bytes())));
127    }
128    let path = qualified_path(interp, receiver)?;
129    match std::fs::metadata(&path) {
130        Ok(meta) => {
131            let size = if meta.is_file() { meta.len() } else { 0 };
132            Ok(Some(interp.text_built(size.to_string().into_bytes())))
133        }
134        Err(_) => Ok(Some(interp.text(b""))),
135    }
136}

138/// Standard handles may refer to a regular file or to an embedding's
139/// in-memory streams, which have no file size.
140fn standard_handle_size(interp: &mut Interp, handle: StandardStream) -> u64 {
141    let metadata = match handle {
142        StandardStream::In => interp
143            .input_file()
144            .and_then(|open| open.file().metadata().ok()),
145        StandardStream::Out | StandardStream::Err => {
146            if !interp
147                .sinks
148                .as_ref()
149                .is_some_and(|sinks| sinks.own_descriptors)
150            {
151                return 0;
152            }
153            let descriptor = match handle {
154                StandardStream::Out => {
155                    interp.hand_over_stdout();
156                    1
157                }
158                _ => {
159                    interp.hand_over_stderr();
160                    2
161                }
162            };
163            descriptor_file(descriptor)
164                .and_then(|file| file.metadata())
165                .ok()
166        }
167    };
168    metadata
169        .filter(std::fs::Metadata::is_file)
170        .map_or(0, |meta| meta.len())
171}

173/// `query_time`: the modification time as `ctime` renders it, which the Rexx
174/// half re-parses into both the `DATETIME` and `TIMESTAMP` answers -- so the
175/// shape here is fixed by that parse (`. month day time year`) and not by
176/// taste.
177pub(in crate::dispatch) fn query_time(
178    interp: &mut Interp,
179    _cleared: Cleared,
180    receiver: ObjRef,
181    _args: &[Option<ObjRef>],
182) -> Result<Option<ObjRef>, Failure> {
183    let metadata = match standard_input_file(interp, receiver)? {
184        true => interp.input_file().map(|open| open.file().metadata()),
185        // An open `HANDLE:n` stream has the times of its descriptor.
186        false => {
187            let state = require(interp, receiver)?;
188            match (&state.open, state.descriptor) {
189                (Some(open), true) => Some(open.file().metadata()),
190                _ => None,
191            }
192        }
193    };
194    let metadata = match metadata {
195        Some(metadata) => metadata,
196        None => std::fs::metadata(qualified_path(interp, receiver)?),
197    };
198    let Ok(modified) = metadata.and_then(|meta| meta.modified()) else {
199        return Ok(Some(interp.text(b"")));
200    };
201    let seconds = match modified.duration_since(std::time::UNIX_EPOCH) {
202        Ok(since) => i128::from(since.as_secs()),
203        Err(before) => {
204            let gap = before.duration();
205            // A fractional time before the epoch belongs to the preceding second.
206            -i128::from(gap.as_secs()) - i128::from(gap.subsec_nanos() != 0)
207        }
208    };
209    let at = i64::try_from(seconds).ok().and_then(|seconds| {
210        let zone = program_zone(interp);
211        let offset = i64::from(zone.find_local_time_type(seconds).ok()?.ut_offset());
212        chrono::DateTime::from_timestamp(seconds.checked_add(offset)?, 0)
213    });
214    let Some(at) = at else {
215        return Ok(Some(interp.text(b"")));
216    };
217    let rendered = at.format("%a %b %e %H:%M:%S %Y").to_string();
218    Ok(Some(interp.text_built(rendered.into_bytes())))
219}

221/// `query_streamtype`: `UNKNOWN` until something opens the stream, which is
222/// what tells a persistent stream from a transient one. A standard stream
223/// answers from its descriptor instead, which it has without being opened.
224pub(in crate::dispatch) fn query_streamtype(
225    interp: &mut Interp,
226    _cleared: Cleared,
227    receiver: ObjRef,
228    _args: &[Option<ObjRef>],
229) -> Result<Option<ObjRef>, Failure> {
230    if unopened_standard(interp, receiver)? {
231        return Ok(Some(interp.text(b"UNKNOWN")));
232    }
233    if let Some(which) = standard_of(interp, receiver)? {
234        let answer: &[u8] = if standard_is_transient(interp, which) {
235            b"TRANSIENT"
236        } else {
237            b"PERSISTENT"
238        };
239        return Ok(Some(interp.text(answer)));
240    }
241    let answer: &[u8] = match require(interp, receiver)?.open.as_ref() {
242        None => b"UNKNOWN",
243        Some(open) if open.transient => b"TRANSIENT",
244        Some(_) => b"PERSISTENT",
245    };
246    Ok(Some(interp.text(answer)))
247}

249/// `query_handle`: the descriptor a standard stream stands for, the
250/// descriptor of an open stream, and the null string for an unopened one.
251/// Measured, the standard three answer `0`, `1` and `2` whatever the
252/// descriptors are redirected to.
253pub(in crate::dispatch) fn query_handle(
254    interp: &mut Interp,
255    _cleared: Cleared,
256    receiver: ObjRef,
257    _args: &[Option<ObjRef>],
258) -> Result<Option<ObjRef>, Failure> {
259    #[cfg(unix)]
260    use std::os::fd::AsRawFd;
261    if unopened_standard(interp, receiver)? {
262        return Ok(Some(interp.text(b"")));
263    }
264    let answer: &[u8] = match standard_of(interp, receiver)? {
265        Some(StandardStream::In) => b"0",
266        Some(StandardStream::Out) => b"1",
267        Some(StandardStream::Err) => b"2",
268        None => {
269            #[cfg(unix)]
270            let handle = require(interp, receiver)?
271                .open
272                .as_ref()
273                .map(|open| open.file().as_raw_fd().to_string().into_bytes());
274            #[cfg(windows)]
275            let handle = {
276                use std::os::windows::io::AsRawHandle;
277                require(interp, receiver)?.open.as_ref().map(|open| {
278                    (open.file().as_raw_handle().addr())
279                        .to_string()
280                        .into_bytes()
281                })
282            };
283            return Ok(Some(interp.text_built(handle.unwrap_or_default())));
284        }
285    };
286    Ok(Some(interp.text(answer)))
287}

289/// `interrexx_stream_mid_line(stream)`: whether the read position of `stream`
290/// lies inside a line and not at its start, as `1` or `0`. A supplier made
291/// there starts with the rest of that line (r[class.stream.deviation.140]),
292/// and one made at the end of a last line that has no line end is made inside
293/// that line, so it has nothing left to read (r[class.stream.supplier]). An
294/// unopened stream, and a transient one, answer `0`: neither has a position to
295/// be inside a line.
296pub(in crate::dispatch) fn supplier_mid_line(
297    interp: &mut Interp,
298    _cleared: Cleared,
299    _receiver: ObjRef,
300    args: &[Option<ObjRef>],
301) -> Result<Option<ObjRef>, Failure> {
302    let stream = args.first().copied().flatten().unwrap_or(ObjRef::NIL);
303    let mode = require(interp, stream)?.mode;
304    let record_length = if mode.record_based {
305        mode.record_length
306    } else {
307        0
308    };
309    let open = match standard_input_file(interp, stream)? {
310        true => interp.input_file(),
311        false => require_mut(interp, stream)?.open.as_mut(),
312    };
313    let mut inside = false;
314    if let Some(open) = open.filter(|open| !open.transient) {
315        let position = open.read_position;
316        if let Ok(line) = line_of(open, position, record_length) {
317            let (_, start) = seek_line(open, line, record_length);
318            inside = start != position;
319        }
320    }
321    Ok(Some(interp.text(if inside { b"1" } else { b"0" })))
322}

324/// `stream_query_position`: `SYS` answers the descriptor's own offset, and
325/// the rest answer a logical position. An unopened stream answers the null
326/// string, a transient one answers `1`.
327pub(in crate::dispatch) fn query_position(
328    interp: &mut Interp,
329    _cleared: Cleared,
330    receiver: ObjRef,
331    args: &[Option<ObjRef>],
332) -> Result<Option<ObjRef>, Failure> {
333    let options = match args.first().copied().flatten() {
334        Some(value) => interp.to_text(value).into_owned(),
335        None => Vec::new(),
336    };
337    let mut sys = false;
338    let mut read = false;
339    let mut write = false;
340    let mut by_char = false;
341    let mut by_line = false;
342    let mut tokens = Tokens::new(&options);
343    // Each token is mutually exclusive with a set of the others and the check
344    // runs before the flag is set, so a repeat is an error too: `SYS` combines
345    // with nothing and `READ WRITE` is a bare 93, while `READ LINE` is the pair
346    // `StreamSupplier~init` itself sends.
347    while let Some(token) = tokens.next() {
348        let clash = if matches(token, b"SYS") {
349            let clash = sys || read || write || by_char || by_line;
350            sys = true;
351            clash
352        } else if matches(token, b"READ") {
353            let clash = sys || read || write;
354            read = true;
355            clash
356        } else if matches(token, b"WRITE") {
357            let clash = sys || read || write;
358            write = true;
359            clash
360        } else if matches(token, b"CHAR") {
361            let clash = sys || by_char || by_line;
362            by_char = true;
363            clash
364        } else if matches(token, b"LINE") {
365            let clash = sys || by_char || by_line;
366            by_line = true;
367            clash
368        } else {
369            true
370        };
371        if clash {
372            return Err(Raised::syntax(93, 0, Vec::new()).into());
373        }
374    }
375    if unopened_standard(interp, receiver)? {
376        return Ok(Some(interp.text(b"")));
377    }
378    if owned_stdout_alias(interp, receiver)? {
379        interp.hand_over_stdout();
380        if let Some(error) = ensure_owned_stdout_file(interp, receiver)? {
381            return open_failed(interp, receiver, &error);
382        }
383    } else if !by_line && owned_standard_output(interp, receiver)? {
384        if sys {
385            interp.hand_over_stdout();
386        }
387        #[cfg(unix)]
388        let at = rustix::fs::seek(std::io::stdout(), rustix::fs::SeekFrom::Current(0)).ok();
389        #[cfg(windows)]
390        let at = crate::host::standard_file(1).ok().and_then(|mut file| {
391            use std::io::Seek as _;
392            file.stream_position().ok()
393        });
394        let answer = if sys {
395            at
396        } else if !write {
397            Some(1)
398        } else {
399            at.map(|at| {
400                at.saturating_add(u64::cast_from(interp.out.len()))
401                    .saturating_add(1)
402            })
403        };
404        return Ok(Some(match answer {
405            Some(at) => interp.text_built(at.to_string().into_bytes()),
406            None => interp.text(b""),
407        }));
408    }
409    let mode = require(interp, receiver)?.mode;
410    let mode = if standard_input_file(interp, receiver)? {
411        interrexx_core::OpenMode {
412            read_only: true,
413            ..interrexx_core::OpenMode::default()
414        }
415    } else {
416        mode
417    };
418    let open = match standard_input_file(interp, receiver)? {
419        true => interp.input_file(),
420        false => require_mut(interp, receiver)?.open.as_mut(),
421    };
422    let Some(open) = open else {
423        return Ok(Some(interp.text(b"")));
424    };
425    if open.transient {
426        return Ok(Some(interp.text(b"1")));
427    }
428    if sys {
429        let at = open.offset();
430        return Ok(Some(interp.text_built(at.to_string().into_bytes())));
431    }
432    // Neither given: the write position only for a write-only stream.
433    if !read && !write {
434        write = mode.write_only;
435    }
436    let record_length = if mode.record_based {
437        mode.record_length
438    } else {
439        0
440    };
441    // By lines, the line the position is in (rexxref funct.xml:4550-4555); a
442    // READ stream's write position is 0 in lines too.
443    let answer = match (write, by_line) {
444        (true, true) if open.write_position == 0 => Ok(0),
445        (true, true) => line_of(open, open.write_position, record_length),
446        (true, false) => Ok(open.write_position),
447        (false, true) => line_of(open, open.read_position, record_length),
448        (false, false) => Ok(open.read_position),
449    };
450    // A line count that cannot read the file raises with its errno: measured,
451    // `ERROR:9` on a stream opened write-only.
452    match answer {
453        Ok(answer) => Ok(Some(interp.text_built(answer.to_string().into_bytes()))),
454        Err(error) => {
455            let errno = error.raw_os_error().unwrap_or(0);
456            let mut answer = b"ERROR:".to_vec();
457            append_stream_errno(&mut answer, errno);
458            notready(interp, receiver, StreamStatus::Error(errno), answer)
459        }
460    }
