1//! `.File`'s `LIBRARY REXX` entry points.
2//!
3//! The class itself is Rexx (`interrexx-lib/rexx/file.rex`): it keeps `path`
4//! and `qualifiedPath`, and every native method here but
5//! `interrexx_file_string` is reached as `self~someImpl(qualifiedPath)`. So
6//! each one is a function of that path alone, and none of them touches the
7//! receiver.

19/// What a timestamp getter answers for a file it cannot stat. The Rexx half
20/// tests for this exact value and answers `.nil`, so it is a sentinel and not
21/// a time.
22const MISSING_TIMESTAMP: i64 = -999_999_999_999_999_999;

24/// The qualified path an entry point was handed.
25fn path_argument(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<PathBuf, Failure> {
26    let Some(value) = args.first().copied().flatten() else {
27        return Err(Raised::missing_method_argument(1).into());
28    };
29    Ok(os_path(&interp.to_text(value)).to_path_buf())
30}

32/// `1` or `0`, which is how every `.File` predicate answers.
33fn flag(interp: &mut Interp, yes: bool) -> Result<Option<ObjRef>, Failure> {
34    Ok(Some(interp.text(if yes { b"1" } else { b"0" })))
35}

37/// Whether the running user may reach `path` for `mode`.
38///
39/// **`stat` is not the question.** A file's mode bits do not say whether *this*
40/// user may read it, so this asks `access(2)`, which answers for the caller's
41/// own credentials -- measured at uid 1000, a mode-`000` file answers 0 for
42/// both where its owner's bits alone would say nothing useful. As root every
43/// one of these answers 1 instead, which is a property of the run and not of
44/// the code.
45fn reachable(path: &Path, write: bool) -> bool {
46    #[cfg(unix)]
47    {
48        use rustix::fs::Access;
49        let mode = if write {
50            Access::WRITE_OK
51        } else {
52            Access::READ_OK
53        };
54        rustix::fs::access(path, mode).is_ok()
55    }
56    #[cfg(windows)]
57    {
58        use std::os::windows::fs::OpenOptionsExt as _;
59        use windows_sys::Win32::Storage::FileSystem::{
60            FILE_FLAG_BACKUP_SEMANTICS, FILE_READ_DATA, FILE_WRITE_DATA,
61        };
62        std::fs::OpenOptions::new()
63            .access_mode(if write {
64                FILE_WRITE_DATA
65            } else {
66                FILE_READ_DATA
67            })
68            .custom_flags(FILE_FLAG_BACKUP_SEMANTICS)
69            .open(path)
70            .is_ok()
71    }
72}

177/// `file_list`: the names directly inside `path`, or `.nil` when it is not a
178/// directory.
179///
180/// **Raw `readdir` order, never sorted** (`class.file.list`): `.dot`,
181/// `two.txt`, `one.txt`, `nested` for a directory whose `os.listdir` gives
182/// exactly that. Dot files and subdirectories are in; `.` and `..` are not,
183/// which `read_dir` already handles.
184pub(super) fn list(
185    interp: &mut Interp,
186    _cleared: Cleared,
187    _receiver: ObjRef,
188    args: &[Option<ObjRef>],
189) -> Result<Option<ObjRef>, Failure> {
190    let path = path_argument(interp, args)?;
191    let Ok(entries) = std::fs::read_dir(&path) else {
192        return Ok(Some(ObjRef::NIL));
193    };
194    let mut slots: Vec<Option<ObjRef>> = Vec::new();
195    for entry in entries.flatten() {
196        let name = entry.file_name().into_encoded_bytes();
197        let held = interp.text_built(name);
198        // Rooted before the next allocation: the array below allocates, and
199        // the names built so far are reachable from nothing until it exists.
200        interp.roots.activity_mut().push_temp(held);
201        slots.push(Some(held));
202    }
203    Ok(Some(
204        interp.alloc_charged(BehaviourId::ARRAY, Body::array(slots)),
205    ))
206}

208/// One file timestamp as the Rexx half wants it: microseconds of **local**
209/// civil time since year 1, or [`MISSING_TIMESTAMP`].
210fn stamp(interp: &mut Interp, path: &Path, accessed: bool) -> Result<Option<ObjRef>, Failure> {
211    let read = std::fs::metadata(path).and_then(|meta| {
212        if accessed {
213            meta.accessed()
214        } else {
215            meta.modified()
216        }
217    });
218    let answer = match read {
219        Ok(time) => local_base_time(time),
220        Err(_) => MISSING_TIMESTAMP,
221    };
222    Ok(Some(interp.text_built(answer.to_string().into_bytes())))
223}

225/// A `SystemTime` as local civil microseconds since year 1.
226///
227/// **At the offset in force at that instant**, not the one in force now:
228/// measured, a March stamp and a November stamp both encode `+3600` here, and
229/// a summer reading of the same file would be an hour out if today's offset
230/// were used.
231fn local_base_time(time: std::time::SystemTime) -> i64 {
232    let unix_micros = match time.duration_since(std::time::UNIX_EPOCH) {
233        Ok(since) => i64::try_from(since.as_micros()).unwrap_or(i64::MAX),
234        Err(before) => -i64::try_from(before.duration().as_micros()).unwrap_or(i64::MAX),
235    };
236    let utc = UNIX_BASE_TIME + unix_micros;
237    utc + local_offset_micros(utc)
238}
239
240/// The `SystemTime` a local civil basetime names -- [`local_base_time`]
241/// inverted.
242///
243/// **One refinement, because the offset depends on the instant it is solving
244/// for.** The first guess reads the zone at the local reading itself, which is
245/// off by the offset; re-reading it at the resulting instant settles every
246/// case except the hour a DST jump repeats, where either reading is a real
247/// local time.
248fn system_time_of(base_time: i64) -> std::time::SystemTime {
249    let mut utc = base_time - local_offset_micros(base_time);
250    utc = base_time - local_offset_micros(utc);
251    let unix_micros = utc - UNIX_BASE_TIME;
252    let magnitude = std::time::Duration::from_micros(unix_micros.unsigned_abs());
253    if unix_micros >= 0 {
254        std::time::UNIX_EPOCH + magnitude
255    } else {
256        std::time::UNIX_EPOCH - magnitude
257    }
258}

291/// Which of a file's two stamps a write is setting.
292#[derive(Clone, Copy, PartialEq, Eq)]
293enum Stamp {
294    Modified,
295    Accessed,
296}
297
298/// Sets one stamp and leaves the other alone, through `utimensat` with
299/// `UTIME_OMIT`.
300///
301/// **The file is not opened.** A read-only file's stamp is still the owner's to
302/// set, answering `1` (`class.file.lastmodified-set`), which an implementation
303/// going through an open writable handle could not do.
304#[cfg(unix)]
305fn set_one_stamp(path: &Path, when: i64, which: Stamp) -> bool {
306    let target = system_time_of(when);
307    let spec = match target.duration_since(std::time::UNIX_EPOCH) {
308        Ok(since) => rustix::fs::Timespec {
309            tv_sec: i64::try_from(since.as_secs()).expect("from microseconds in an i64"),
310            tv_nsec: i64::from(since.subsec_nanos()),
311        },
312        // Before the epoch: one second earlier plus the remainder forward,
313        // because `tv_nsec` is never negative.
314        Err(before) => {
315            let gap = before.duration();
316            let (seconds, nanos) = match gap.subsec_nanos() {
317                0 => (
318                    -i64::try_from(gap.as_secs()).expect("from microseconds in an i64"),
319                    0,
320                ),
321                nanos => (
322                    -i64::try_from(gap.as_secs()).expect("from microseconds in an i64") - 1,
323                    1_000_000_000 - i64::from(nanos),
324                ),
325            };
326            rustix::fs::Timespec {
327                tv_sec: seconds,
328                tv_nsec: nanos,
329            }
330        }
331    };
332    let omit = rustix::fs::Timespec {
333        tv_sec: 0,
334        tv_nsec: rustix::fs::UTIME_OMIT,
335    };
336    let times = match which {
337        Stamp::Modified => rustix::fs::Timestamps {
338            last_access: omit,
339            last_modification: spec,
340        },
341        Stamp::Accessed => rustix::fs::Timestamps {
342            last_access: spec,
343            last_modification: omit,
344        },
345    };
346    rustix::fs::utimensat(rustix::fs::CWD, path, &times, rustix::fs::AtFlags::empty()).is_ok()
347}

349#[cfg(windows)]
350fn set_one_stamp(path: &Path, when: i64, which: Stamp) -> bool {
351    use std::os::windows::fs::OpenOptionsExt as _;
352    use windows_sys::Win32::Storage::FileSystem::{
353        FILE_FLAG_BACKUP_SEMANTICS, FILE_WRITE_ATTRIBUTES,
354    };
355    let times = std::fs::FileTimes::new();
356    let times = match which {
357        Stamp::Modified => times.set_modified(system_time_of(when)),
358        Stamp::Accessed => times.set_accessed(system_time_of(when)),
359    };
360    std::fs::OpenOptions::new()
361        .access_mode(FILE_WRITE_ATTRIBUTES)
362        .custom_flags(FILE_FLAG_BACKUP_SEMANTICS)
363        .open(path)
364        .and_then(|file| file.set_times(times))
365        .is_ok()
366}

432/// `file_delete_directory`: the directory itself, never its contents --
433/// measured, a non-empty one answers `0`. The Rexx half sends this when the

452/// `file_rename`: refuses a same-path rename and an existing target before it
453/// asks the file system, because `rename(2)` would answer success for the first
454/// and replace the second, where `~renameTo` answers `0` to both
455/// (`class.file.renameto`). The one target left to the file system is a
456/// directory taking the place of a directory, which `rename(2)` replaces when
457/// it is empty and refuses otherwise, as it does a target inside the receiver.
458pub(super) fn rename(
459    interp: &mut Interp,
460    _cleared: Cleared,
461    _receiver: ObjRef,
462    args: &[Option<ObjRef>],
463) -> Result<Option<ObjRef>, Failure> {
464    let from = path_argument(interp, args)?;
465    let Some(value) = args.get(1).copied().flatten() else {
466        return Err(Raised::missing_method_argument(2).into());
467    };
468    let to = os_path(&interp.to_text(value)).to_path_buf();
469    if from == to {
470        return flag(interp, false);
471    }
472    if let Ok(target) = std::fs::symlink_metadata(&to) {
473        let directory_onto_directory =
474            target.is_dir() && std::fs::symlink_metadata(&from).is_ok_and(|source| source.is_dir());
475        if !directory_onto_directory {
476            return flag(interp, false);
477        }
478    }
479    let moved = std::fs::rename(&from, &to).is_ok();
480    flag(interp, moved)
481}

483/// Rewrites `path`'s three write bits, for `file_set_read_only` and
484/// `file_set_writable`. Answers **nothing**: measured, `say f~setReadOnly` is
485/// 91.999 and the side effect happens anyway.
486fn write_bits(
487    interp: &mut Interp,
488    args: &[Option<ObjRef>],
489    writable: bool,
490) -> Result<Option<ObjRef>, Failure> {
491    let path = path_argument(interp, args)?;
492    if let Ok(meta) = std::fs::metadata(&path) {
493        #[cfg(unix)]
494        {
495            use std::os::unix::fs::PermissionsExt as _;
496            let mode = meta.permissions().mode();
497            // All three, not the owner's alone -- measured, `644` becomes `444`
498            // and then `666`, so the setter ignores the umask.
499            let rewritten = if writable {
500                mode | 0o222
501            } else {
502                mode & !0o222
503            };
504            let _ = std::fs::set_permissions(&path, std::fs::Permissions::from_mode(rewritten));
505        }
506        #[cfg(windows)]
507        {
508            let mut permissions = meta.permissions();
509            permissions.set_readonly(!writable);
510            let _ = std::fs::set_permissions(&path, permissions);
511        }
512    }
513    let _ = interp;
514    Ok(None)
515}

602    interp: &mut Interp,
603    _cleared: Cleared,
604    _receiver: ObjRef,
605    _args: &[Option<ObjRef>],
606) -> Result<Option<ObjRef>, Failure> {
607    let named = interp
608        .env_get(b"TMPDIR")
609        .filter(|value| !value.is_empty())
610        .map(<[u8]>::to_vec);
611    Ok(Some(match named {
612        Some(value) => interp.text_built(value),
613        None => interp.text(b"/tmp"),
614    }))
615}

617/// `file_search_path_impl`: the first directory of `pathList` holding `file`,
618/// joined, or `.nil`.
619///
620/// `file` is read as a string argument, so an object answers by its
621/// `makeString` and `.nil` is 88.909. A `file` that begins with a slash is the
622/// absolute path of the file and is answered whatever the directories. A
623/// relative directory is relative to the interpreter's current directory, and
624/// the answer is absolute.
625pub(super) fn search_path_impl(
626    interp: &mut Interp,
627    _cleared: Cleared,
628    _receiver: ObjRef,
629    args: &[Option<ObjRef>],
630) -> Result<Option<ObjRef>, Failure> {
631    let Some(wanted) = args.first().copied().flatten() else {
632        return Err(Raised::missing_method_argument(1).into());
633    };
634    let Some(wanted) = interp.blamed_string_conversion(wanted)? else {
635        return Err(Raised::native_argument_needs_a_string_value(1).into());
636    };
637    let wanted = interp.to_text(wanted).into_owned();
638    let Some(value) = args.get(1).copied().flatten() else {
639        return Err(Raised::missing_method_argument(2).into());
640    };
641    let list = interp.to_text(value).into_owned();
642    let is_file = |path: &[u8]| std::fs::metadata(os_path(path)).is_ok_and(|meta| meta.is_file());
643    if wanted.first() == Some(&b'/') {
644        return Ok(Some(if is_file(&wanted) {
645            interp.text_built(wanted)
646        } else {
647            ObjRef::NIL
648        }));
649    }
650    for directory in list
651        .split(|byte| *byte == b':')
652        .filter(|part| !part.is_empty())
653    {
654        let trimmed = directory.len()
655            - directory
656                .iter()
657                .rev()
658                .take_while(|byte| **byte == b'/')
659                .count();
660        let mut candidate = directory[..trimmed].to_vec();
661        candidate.push(b'/');
662        candidate.extend_from_slice(&wanted);
663        if candidate.first() != Some(&b'/') {
664            candidate = crate::builtin::platform::qualify_name(interp, &candidate);
665        }
666        if is_file(&candidate) {
667            return Ok(Some(interp.text_built(candidate)));
668        }
669    }
670    Ok(Some(ObjRef::NIL))
671}

673/// `file_qualify`: the absolute spelling of `path`, against the interpreter's
674/// own current directory, by [`crate::builtin::platform::qualify_name`].
675///
676/// **Purely lexical** (`class.file.absolute`): `a/b/../c.txt` answers
677/// `<cwd>/a/c.txt` although no such directory exists, so nothing here resolves
678/// a symlink or asks the file system anything.
679///
680/// This is the entry point the rest of the class waits on: `init` qualifies
681/// eagerly, so without it `.File~new` builds nothing at all.
682pub(super) fn qualify(
683    interp: &mut Interp,
684    _cleared: Cleared,
685    _receiver: ObjRef,
686    args: &[Option<ObjRef>],
687) -> Result<Option<ObjRef>, Failure> {
688    let Some(value) = args.first().copied().flatten() else {
689        return Err(Raised::missing_method_argument(1).into());
690    };
691    let path = interp.to_text(value).into_owned();
692    let qualified = crate::builtin::platform::qualify_name(interp, &path);
693    Ok(Some(interp.text_built(qualified)))
694}

696/// `interrexx_file_string`: argument 1 as a plain string when it is a string,
697/// and `.nil` when it is any other object. A number is a string, and an
698/// instance of a subclass of `String` is read by its own characters
699/// (r[class.index.string], r[class.file.deviation.109]). Nothing is sent to
700/// the argument: the Rexx half sends `request` with `STRING` itself where a
701/// rule reads an object as a string argument (r[class.file.string-args]), and
702/// asks this again of the answer.
703pub(super) fn string_or_nil(
704    interp: &mut Interp,
705    _cleared: Cleared,
706    _receiver: ObjRef,
707    args: &[Option<ObjRef>],
708) -> Result<Option<ObjRef>, Failure> {
709    let Some(value) = args.first().copied().flatten() else {
710        return Err(Raised::missing_method_argument(1).into());
711    };
712    if let Some(bytes) = super::string::subclass_text(interp, value) {
713        let bytes = bytes.to_vec();
714        return Ok(Some(interp.text_built(bytes)));
715    }
716    let plain = matches!(
717        interp.receiver_kind(value),
718        Ok(Primitive::String | Primitive::SmallInt)
719    );
720    Ok(Some(if plain { value } else { ObjRef::NIL }))
