3/// `Class~defineMethods(methods)`: install a whole table of instance methods
4/// in one mutation.
5pub(in crate::dispatch) fn native_define_methods(
6    interp: &mut Interp,
7    _cleared: Cleared,
8    receiver: ObjRef,
9    args: &[Option<ObjRef>],
10) -> Result<Option<ObjRef>, Failure> {
11    let class = class_receiver(interp, receiver)?;
12    rexx_defined_lock(interp, class)?;
13    let Some(Some(table)) = args.first().copied() else {
14        return Err(Raised::missing_named_argument("methods").into());
15    };
16    let pairs = method_table_pairs(interp, table)?;
17    let mut entries: Vec<(Box<[u8]>, Option<ObjRef>)> = Vec::with_capacity(pairs.len());
18    for (name, value) in pairs {
19        if value == ObjRef::NIL {
20            entries.push((name, None));
21            continue;
22        }
23        let value = if interp.receiver_kind(value) == Ok(Primitive::Method) {
24            value
25        } else {
26            // The entry's own index is the method's name, as stored: the
27            // method object is built under the index's string value and the
28            // dictionary is keyed under its upcase.
29            compile_method_source(interp, &name, value, "method source", SourceTaker::Method)?
30        };
31        entries.push((name, Some(value)));
32    }
33    interp
34        .define_method_table(class, &entries)
35        .ok_or_else(|| Failure::from(Loud::unrecorded_executable()))?;
36    Ok(None)
37}

39/// The (name, value) pairs `createMethodDictionary` obtains from `table`'s
40/// `SUPPLIER`, each index as its string value, in supplier order.
41pub(super) fn method_table_pairs(
42    interp: &mut Interp,
43    table: ObjRef,
44) -> Result<Vec<(Box<[u8]>, ObjRef)>, Failure> {
45    // A walk cannot see entries the interpreter answers by name rather than
46    // storing. A mutation over that empty walk would silently do nothing.
47    if hash::owns(interp, table)
48        && hash::string_keyed(interp, table)
49        && interp.holds_unbuilt_entry(table)
50    {
51        return Err(Loud::unreadable_collection().into());
52    }
53    // A user SUPPLIER can send protected methods or install methods
54    // recursively. Run that method under the normal dispatch context.
55    let caller = interp.caller();
56    let supplier = interp
57        .send_message(table, b"SUPPLIER", None, &[], caller)?
58        .unwrap_or(ObjRef::NIL);
59    interp.roots.activity_mut().push_temp(supplier);
60    // Only a `Supplier`, an instance of a subclass included, is read: any
61    // other answer, `.nil` or none among them, is 98.994 naming the object
62    // the methods were taken from (`r[class.class.methods-supplier]`).
63    if !is_supplier(interp, supplier) {
64        let shown = interp.string_value_text(table)?;
65        return Err(Raised::no_supplier(&shown).into());
66    }
67    let walked = crate::dispatch::collection::supplier::drain_supplier(interp, supplier)?;
68    let mut pairs = Vec::new();
69    for (index, item) in walked {
70        let name = interp.string_value_text(index)?.into_boxed_slice();
71        pairs.push((name, item));
72    }
73    Ok(pairs)
74}

76/// Whether `value` is a `Supplier`, an instance of a subclass included,
77/// which the walk reads directly.
78fn is_supplier(interp: &mut Interp, value: ObjRef) -> bool {
79    let class = interp.class_of_value(value);
80    let wanted = interp.object_model().supplier;
81    class.is_some_and(|class| interp.classes().is_a(class, wanted))
82}
