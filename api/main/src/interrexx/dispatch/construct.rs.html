32/// `StringTable~new` and `Directory~new`: an empty hash collection, each of
33/// which allocates and then sends `INIT` with the whole argument list.
34pub(super) fn native_hash_collection_new(
35    interp: &mut Interp,
36    cleared: Cleared,
37    receiver: ObjRef,
38    args: &[Option<ObjRef>],
39) -> Result<super::NativeStarted, Failure> {
40    // **A `Directory` a program makes is a collection, not the environment.**
41    // `NativeObject`'s map holds its keys already uppercased, by its callers,
42    // and a `.Directory~new` does not: measured, `d['lower'] = 1` leaves
43    // `allIndexes` reading `lower` and `d['LOWER']` answering `.nil`. So a
44    // collection gets the object-keyed store with a string-key protocol on
45    // top.
46    hash::native_hash_new(interp, cleared, receiver, args)
47}

78/// `Message~new(target, message, ...)`: a message object whose send has not
79/// been made.
80pub(super) fn native_message_new(
81    interp: &mut Interp,
82    _cleared: Cleared,
83    receiver: ObjRef,
84    args: &[Option<ObjRef>],
85) -> Result<super::NativeStarted, Failure> {
86    let class = class_receiver(interp, receiver)?;
87    if args.len() < 2 {
88        return Err(Raised::not_enough_method_arguments(2).into());
89    }
90    if args[0].is_none() {
91        return Err(Raised::missing_named_argument("message target").into());
92    }
93    let (name, scope) = decode_message_name(interp, args[1])?;
94    let arguments = match args.get(2) {
95        None => Vec::new(),
96        Some(None) => return Err(Raised::missing_named_argument("argument style").into()),
97        Some(Some(_)) => match option_method_argument(interp, args, 2, "AI")? {
98            Some(b'A') => {
99                if args.len() < 4 {
100                    return Err(Raised::not_enough_method_arguments(4).into());
101                }
102                if args.len() > 4 {
103                    return Err(Raised::too_many_method_arguments(4).into());
104                }
105                let Some(array) = args[3] else {
106                    return Err(Raised::missing_named_argument("message arguments").into());
107                };
108                array_argument(interp, array, ArrayArgument::Named("message arguments"))?
109            }
110            _ => args[3..].to_vec(),
111        },
112    };
113    let object = if class == interp.object_model().message {
114        interp.native_instance(class)
115    } else {
116        new_instance(interp, class)?
117    };
118    let target = args[0].expect("checked above");
119    interp.set_native_entry(object, MESSAGE_TARGET, target);
120    let name = interp.text(&name);
121    interp.set_native_entry(object, MESSAGE_NAME, name);
122    if let Some(scope) = scope {
123        interp.set_native_entry(object, MESSAGE_SCOPE, scope);
124    }
125    let arguments = interp.security_arguments_array(&arguments);
126    interp.set_native_entry(object, MESSAGE_ARGUMENTS, arguments);
127    super::begin_init(interp, object, &[])
128}

130/// `Method~new(name, source [, context] ...)` and `Routine~new(name, source [,
131/// context] ...)`: the first two are required, the context is resolved as
132/// `newFile`'s is, and `INIT` gets the arguments after it. The two
133/// constructors share one argument handling.
134pub(super) fn native_executable_new(
135    interp: &mut Interp,
136    _cleared: Cleared,
137    receiver: ObjRef,
138    args: &[Option<ObjRef>],
139) -> Result<super::NativeStarted, Failure> {
140    let class = class_receiver(interp, receiver)?;
141    let name = executable_name_argument(interp, args)?;
142    let Some(source) = args.get(1).copied().flatten() else {
143        return Err(Raised::missing_named_argument("source").into());
144    };
145    let (routine, base) = executable_base(interp, class);
146    let context = args.get(2).copied().flatten();
147    let parent = |interp: &mut Interp| match context {
148        Some(context) => executable_context(interp, context, b"NEW", 3),
149        None => Ok(None),
150    };
151    let name = interp.to_text(name).to_vec();
152    // Its directives install under the method's own level.
153    let object = interp.primitive_level(receiver, b"NEW", args, |interp| {
154        if routine {
155            compile_routine_source(interp, &name, source, "source", parent)
156        } else {
157            compile_method_source_in(
158                interp,
159                &name,
160                source,
161                "source",
162                SourceTaker::Executable,
163                parent,
164            )
165        }
166    })?;
167    if class != base {
168        native_as_instance(interp, object, class, base)?;
169    }
170    super::begin_init(interp, object, args.get(3..).unwrap_or_default())
171}

173/// The scope-pool entry, under `Method`'s, `Routine`'s or `Package`'s scope,
174/// that holds a subclass instance's annotation table.
175pub(super) const SUBCLASS_ANNOTATIONS: &[u8] = b"ANNOTATIONS";

177/// Completion of an executable or package `object` that `base`'s constructor
178/// built: it becomes an instance of `class`, a subclass of `base`, keeping its
179/// identity, so every record kept against the object still reaches it. An
180/// object an earlier call already made an instance takes `class`'s behaviour.
181pub(super) fn native_as_instance(
182    interp: &mut Interp,
183    object: ObjRef,
184    class: ObjRef,
185    base: ObjRef,
186) -> Result<(), Failure> {
187    if interp.classes().is_abstract(class) {
188        let id = interp.classes().id_string(class).as_bytes().to_vec();
189        return Err(Raised::abstract_class(&id).into());
190    }
191    let behaviour = interp.classes().instance_behaviour_handle(class);
192    let Some(held) = interp.heap.get_mut(object) else {
193        return Err(Loud::unknown_receiver("a value whose object is no longer live").into());
194    };
195    match &mut held.body {
196        Body::Native(native) => {
197            let mut pools = interrexx_core::ScopePools::new();
198            if let Some(table) = native.annotations() {
199                pools.set(base, SUBCLASS_ANNOTATIONS, table);
200            }
201            let state = match native.take_package() {
202                Some(state) => Some(interrexx_core::NativeState::Package(state)),
203                None => native
204                    .executable_package()
205                    .map(interrexx_core::NativeState::Executable),
206            };
207            held.body = Body::Instance {
208                class,
209                behaviour,
210                name: None,
211                pools,
212                own: None,
213                native: state.map(Box::new),
214            };
215        }
216        Body::Instance {
217            class: held_class,
218            behaviour: held_behaviour,
219            ..
220        } => {
221            *held_class = class;
222            *held_behaviour = behaviour;
223        }
224        _ => return Err(Loud::unknown_receiver("an object this crate did not build").into()),
225    }
226    // See `Interp::reqstr_armed`, which `new_instance` arms for the same
227    // reason.
228    interp.reqstr_armed = true;
229    if interp.classes().has_uninit(class) {
230        interp.heap.set_uninit(object);
231    }
232    Ok(())
233}

235/// `Method~newFile(name [, context])` and `Routine~newFile(name [, context])`:
236/// the executable a file's own text becomes, then `INIT` with no arguments.
237pub(super) fn native_new_file(
238    interp: &mut Interp,
239    _cleared: Cleared,
240    receiver: ObjRef,
241    args: &[Option<ObjRef>],
242) -> Result<super::NativeStarted, Failure> {
243    let class = class_receiver(interp, receiver)?;
244    let name = executable_name_argument(interp, args)?;
245    let (routine, base) = executable_base(interp, class);
246    let parent = match args.get(1).copied().flatten() {
247        Some(context) => executable_context(interp, context, b"NEWFILE", 2)?,
248        None => None,
249    };
250    let name = interp.to_text(name).to_vec();
251    let object = interp.new_file_executable(&name, routine, parent, false)?;
252    if class != base {
253        native_as_instance(interp, object, class, base)?;
254    }
255    super::begin_init(interp, object, &[])
256}

258/// Whether `class` is `Routine` or a subclass of it, and the base class,
259/// `Routine` or `Method`, whose executable it builds.
260fn executable_base(interp: &mut Interp, class: ObjRef) -> (bool, ObjRef) {
261    let routine_class = interp.routine_class();
262    if interp.classes().is_a(class, routine_class) {
263        (true, routine_class)
264    } else {
265        (false, interp.method_class())
266    }
267}

269/// `newFile`'s and `new`'s context argument, at `position` of `method`,
270/// resolved to the package the executable resolves names against. `None` is
271/// the caller's own package, which is what an absent argument and
272/// `"PROGRAMSCOPE"` both mean -- measured, the two answer the caller's
273/// `::ROUTINE` alike at rc 0.
274fn executable_context(
275    interp: &mut Interp,
276    context: ObjRef,
277    method: &[u8],
278    position: usize,
279) -> Result<Option<ObjRef>, Failure> {
280    let class = interp.class_of_value(context);
281    let resolved = if class == Some(interp.package_class()) {
282        interp.package_data(context).map(|_| context)
283    } else if class == Some(interp.routine_class()) || class == Some(interp.method_class()) {
284        interp.executable_context_package(context)?
285    } else if class == Some(interp.string_class())
286        && interp
287            .to_text(context)
288            .eq_ignore_ascii_case(b"PROGRAMSCOPE")
289    {
290        return Ok(None);
291    } else {
292        None
293    };
294    match resolved {
295        Some(package) => Ok(Some(package)),
296        // The found value is its string *conversion*, not its string value:
297        // measured, `.array~of(1)` reports `found "1"` where `.NIL` reports
298        // `found "The NIL object"`.
299        None => {
300            let found = interp.to_text(context).to_vec();
301            Err(Raised::argument_not_in_list(
302                method,
303                position,
304                "\"PROGRAMSCOPE\", Method, Routine, or Package object",
305                &found,
306            )
307            .into())
308        }
309    }
310}

312/// The external-method and external-routine loaders. Neither completes a
313/// subclass's object: sent to one, each answers the base class's.
314pub(super) fn native_load_external(
315    interp: &mut Interp,
316    _cleared: Cleared,
317    receiver: ObjRef,
318    args: &[Option<ObjRef>],
319) -> Result<Option<ObjRef>, Failure> {
320    let receiver_class = class_receiver(interp, receiver)?;
321    let name = executable_name_argument(interp, args)?;
322    let (routine, class) = executable_base(interp, receiver_class);
323    let Some(descriptor) = args.get(1).copied().flatten() else {
324        return Err(Raised::missing_named_argument("descriptor").into());
325    };
326    let descriptor = required_string_named_argument(interp, descriptor, "descriptor")?;
327    let descriptor = interp.to_text(descriptor).to_vec();
328    let name = interp.to_text(name).to_vec();
329    let Some((library, entry)) = external_specification(&descriptor, &name) else {
330        return Err(Raised::bad_external_specification(&descriptor).into());
331    };
332    // **The keyword is case-insensitive and the library name is not**, so
333    // `library REXX file_separator` and `LiBrArY REXX file_separator` each
334    // answer a `Method`, while `LIBRARY rexx file_separator` and `LIBRARY Rexx
335    // file_separator` answer `.nil`.
336    if let Some(package) = crate::internal_routines::Package::named(&library) {
337        // An internal package's routine table is not this registry.
338        if routine {
339            let Some(row) = crate::internal_routines::package_routine(package, &entry) else {
340                return Ok(Some(ObjRef::NIL));
341            };
342            let index = crate::internal_routines::index_of(row);
343            return Ok(Some(interp.internal_routine_object(index)));
344        }
345        let Some(external) = native::package_entry_point(package, &entry) else {
346            return Ok(Some(ObjRef::NIL));
347        };
348        let object = interp.native_instance(class);
349        interp.attach_empty_annotations(object);
350        interp.record_rexx_external_method(object, external);
351        return Ok(Some(object));
352    }
353    // The procedure the shared code is keyed by, as [`crate::LibraryCodeKey`]
354    // documents, or `None` where nothing answers the name. **Neither a library
355    // that is not there nor a procedure it does not export raises**:
356    // `.Method~loadExternalMethod('x', 'LIBRARY zorkolib RegExp_Parse')` and
357    // the same naming `rxregexp NoSuchEntry` both answer `The NIL object`, and
358    // `.Routine~loadExternalRoutine` answers it for the same two.
359    let procedure = match interp.resolve_library(&library) {
360        crate::LibraryLoad::Loaded(loaded) => {
361            if routine {
362                loaded.routine(&entry).map(|row| row.name.clone())
363            } else {
364                loaded.method(&entry).map(|_| entry.clone())
365            }
366        }
367        crate::LibraryLoad::Missing => None,
368        // For example, 98.982 on the first ask, as `Package~loadLibrary` raises
369        // it.
370        crate::LibraryLoad::Version => {
371            return Err(Raised::library_version(&library).into());
372        }
373        crate::LibraryLoad::Raised(failure) => return Err(failure),
374    };
375    let Some(procedure) = procedure else {
376        return Ok(Some(ObjRef::NIL));
377    };
378    let code = interp.library_code(crate::LibraryCodeKey {
379        library,
380        procedure,
381        routine,
382    });
383    // A routine is the library's own object for the row, the one an
384    // imported-routine table answers.
385    if routine {
386        return Ok(Some(interp.library_routine_object(code)));
387    }
388    let object = interp.native_instance(class);
389    interp.attach_empty_annotations(object);
390    interp.record_loaded_executable(object, code);
391    Ok(Some(object))
392}

394/// A `loadExternal*` descriptor split into its library and its entry point,
395/// or `None` for one that is not an external name specification.
396fn external_specification(descriptor: &[u8], name: &[u8]) -> Option<(Vec<u8>, Vec<u8>)> {
397    let mut words = descriptor
398        .split(|byte| byte.is_ascii_whitespace())
399        .filter(|word| !word.is_empty());
400    if !words.next()?.eq_ignore_ascii_case(b"LIBRARY") {
401        return None;
402    }
403    let library = words.next()?.to_vec();
404    let entry = words.next().map_or_else(|| name.to_vec(), <[u8]>::to_vec);
405    if words.next().is_some() {
406        return None;
407    }
408    Some((library, entry))
409}

423/// `String~new(value, ...)`: a string carrying `value`'s own bytes.
424pub(super) fn native_string_new(
425    interp: &mut Interp,
426    _cleared: Cleared,
427    receiver: ObjRef,
428    args: &[Option<ObjRef>],
429) -> Result<super::NativeStarted, Failure> {
430    let class = class_receiver(interp, receiver)?;
431    let Some(Some(value)) = args.first().copied() else {
432        return Err(Raised::missing_method_argument(1).into());
433    };
434    let text = required_string_argument(interp, value, 1)?;
435    if class != interp.object_model().string {
436        let bytes: Box<[u8]> = interp.to_text(text).into();
437        let held = bytes.len();
438        let object = new_instance(interp, class)?;
439        let Some(interrexx_core::Object {
440            body: Body::Instance { native, .. },
441            ..
442        }) = interp.heap.get_mut(object)
443        else {
444            unreachable!("new_instance allocates a Body::Instance")
445        };
446        *native = Some(Box::new(interrexx_core::NativeState::Text(bytes)));
447        interp.charge_unheld(held);
448        return super::begin_init(interp, object, &args[1..]);
449    }
450    let bytes = interp.to_text(text).to_vec();
451    let object = interp.text_built(bytes);
452    interp.roots.activity_mut().push_temp(object);
453    super::begin_init(interp, object, &args[1..])
454}

456/// `Stem~new(name, ...)`: a stem object whose name is the optional argument.
457pub(super) fn native_stem_new(
458    interp: &mut Interp,
459    _cleared: Cleared,
460    receiver: ObjRef,
461    args: &[Option<ObjRef>],
462) -> Result<super::NativeStarted, Failure> {
463    let class = class_receiver(interp, receiver)?;
464    let name = match args.first().copied().flatten() {
465        Some(value) => {
466            let text = required_string_argument(interp, value, 1)?;
467            interp.to_text(text).into_owned()
468        }
469        None => Vec::new(),
470    };
471    let stem_class = interp.object_model().stem;
472    let object = interp.alloc_with(
473        interrexx_core::BehaviourId::STEM,
474        Body::Stem {
475            name: name.into(),
476            default: None,
477            tails: interrexx_core::Tails::new(),
478            exposed: None,
479        },
480    );
481    interp.roots.activity_mut().push_temp(object);
482    let object = if class == stem_class {
483        object
484    } else {
485        let instance = new_instance(interp, class)?;
486        interp.set_pool_variable(instance, stem_class, super::hash::stem::STEM_STORE, object);
487        instance
488    };
489    let rest = args.get(1..).unwrap_or_default();
490    super::begin_init(interp, object, rest)
491}

516/// Anything that is not a `.Pointer` compares unequal, and two `.Pointer`s
517/// compare on their addresses.
518fn pointers_are_equal(
519    interp: &Interp,
520    receiver: ObjRef,
521    args: &[Option<ObjRef>],
522) -> Result<bool, Failure> {
523    let other = args
524        .first()
525        .copied()
526        .flatten()
527        .ok_or_else(|| Failure::from(Raised::missing_method_argument(1)))?;
528    let Some(other) = pointer_address(interp, other) else {
529        return Ok(false);
530    };
531    Ok(pointer_address(interp, receiver) == Some(other))
532}

580    receiver: ObjRef,
581    args: &[Option<ObjRef>],
582) -> Result<super::NativeStarted, Failure> {
583    let class = class_receiver(interp, receiver)?;
584    let Some(referent) = args.first().copied().flatten() else {
585        return Err(Raised::missing_method_argument(1).into());
586    };
587    let scope = interp.object_model().weak_reference;
588    let object = new_instance(interp, class)?;
589    let cell = weak_referent_cell(interp, referent);
590    interp.set_pool_variable(object, scope, WEAK_REFERENT, cell);
591    let rest: Vec<Option<ObjRef>> = args.iter().skip(1).copied().collect();
592    super::begin_init(interp, object, &rest)
593}
594
595/// `WeakReference~value`: the referent, or `.nil` once the collector has
596/// cleared it.
597fn native_weak_reference_value(
598    interp: &mut Interp,
599    _cleared: Cleared,
600    receiver: ObjRef,
601    _args: &[Option<ObjRef>],
602) -> Result<Option<ObjRef>, Failure> {
603    let scope = interp.object_model().weak_reference;
604    let cell = interp
605        .pools_of(receiver)
606        .and_then(|pools| pools.get(scope, WEAK_REFERENT));
607    let referent = match cell.and_then(|cell| interp.heap.get(cell).map(|object| &object.body)) {
608        Some(Body::WeakRef(referent)) => *referent,
609        _ => ObjRef::NIL,
610    };
611    Ok(Some(referent))
612}
