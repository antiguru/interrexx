28/// `Object~hasMethod(name)`: whether the receiver's behaviour answers `name`.
29pub(super) fn native_has_method(
30    interp: &mut Interp,
31    _cleared: Cleared,
32    receiver: ObjRef,
33    args: &[Option<ObjRef>],
34) -> Result<Option<ObjRef>, Failure> {
35    let Some(Some(argument)) = args.first().copied() else {
36        return Err(Raised::missing_method_argument(1).into());
37    };
38    let argument = required_string_argument(interp, argument, 1)?;
39    let name = String::from_utf8_lossy(&interp.to_text(argument).to_ascii_uppercase()).into_owned();
40    // **The object's own dictionary answers first**, so a `setMethod` name
41    // reports `.true` and a hidden one reports `.false` even where the class
42    // defines it -- the first answers `.true` and a send after the second raises.
43    if let Some(entry) = interp.own_method_entry(receiver, name.as_bytes()) {
44        return Ok(Some(crate::eval::logical(entry.is_some())));
45    }
46    // **A receiver with no class here is loud, not `.false`.** `send_message`
47    // screens for it before any method runs, so this arm is unreachable today;
48    // answering `.false` from it anyway would make this the one place in the module
49    // where a gap becomes an answer, and the answer would be wrong. **Which
50    // dictionary is asked follows the receiver**: with `::class K` plus
51    // `::method m class`: `.K~hasMethod('M')` is `.true` and
52    // `.K~hasMethod('LENGTH')` is `.false`, so a class object is asked about its
53    // class behaviour and not about `.Class`'s instances.
54    let behaviour = match interp.receiver_behaviour(receiver) {
55        Ok(behaviour) => behaviour,
56        Err(kind) => return Err(Loud::unknown_receiver(kind).into()),
57    };
58    let classes = &interp.object_model().classes;
59    let answers = match behaviour {
60        Behaviour::Instance { methods, .. } => classes.has_method_at(methods, &name),
61        Behaviour::ClassSide(class) => classes.class_has_method(class, &name),
62    };
63    Ok(Some(crate::eval::logical(answers)))
64}

66/// `Object~defaultName`: the receiver's class id with an article in front.
67pub(super) fn native_default_name(
68    interp: &mut Interp,
69    cleared: Cleared,
70    receiver: ObjRef,
71    _args: &[Option<ObjRef>],
72) -> Result<Option<ObjRef>, Failure> {
73    let Some(class) = native_class(interp, cleared, receiver, &[])? else {
74        return Err(Loud::unknown_receiver("a value with no class of its own").into());
75    };
76    // **An enhanced object takes no article**, returning before the `a`/`an`
77    // choice. For example, `.Object~subclass('K')~enhanced(t)~defaultName` is
78    // `enhanced K`.
79    let enhanced = is_enhanced_instance(interp, receiver);
80    let id = interp.classes().id_string(class);
81    let name = if enhanced {
82        crate::value::enhanced_object_name(id)
83    } else {
84        crate::environment::default_object_name(id)
85    };
86    Ok(Some(interp.text_built(name.into_bytes())))
87}

89/// `Object~class`: the class whose behaviour answers this receiver's messages
90/// -- the behaviour's owning class.
91pub(super) fn native_class(
92    interp: &mut Interp,
93    _cleared: Cleared,
94    receiver: ObjRef,
95    _args: &[Option<ObjRef>],
96) -> Result<Option<ObjRef>, Failure> {
97    let kind = match interp.receiver_kind(receiver) {
98        Ok(kind) => kind,
99        Err(gap) => return Err(Loud::unknown_receiver(gap).into()),
100    };
101    let model = interp.object_model();
102    Ok(Some(match kind {
103        Primitive::String | Primitive::SmallInt => model.string,
104        Primitive::Object => model.object,
105        Primitive::Array => model.array,
106        Primitive::Package => model.package,
107        Primitive::Method => model.method,
108        Primitive::Routine => model.routine,
109        Primitive::Context => model.context,
110        Primitive::StackFrame => model.stack_frame,
111        Primitive::RexxInfo => model.rexx_info,
112        Primitive::Message => model.message,
113        Primitive::VariableReference => model.variable_reference,
114        Primitive::Stem => model.stem,
115        Primitive::Class(class) => model.classes.class_of(class),
116        Primitive::Instance { class, .. } => class,
117    }))
118}

120/// `Object~isA(class)`: whether the receiver's **class** is `class` or a
121/// subclass of it. It asks the class and not the receiver.
122pub(super) fn native_is_a(
123    interp: &mut Interp,
124    cleared: Cleared,
125    receiver: ObjRef,
126    args: &[Option<ObjRef>],
127) -> Result<Option<ObjRef>, Failure> {
128    let other = class_argument(interp, args)?;
129    // `native_class` answers a class object for every receiver kind, so the
130    // `None` a `NativeMethod` may now answer is not one of its outcomes -- and
131    // a refusal rather than an `expect`, this crate's rule for an internal
132    // inconsistency.
133    let Some(own) = native_class(interp, cleared, receiver, &[])? else {
134        return Err(Loud::unknown_receiver("a value with no class of its own").into());
135    };
136    let answer = interp.classes().is_a(own, other);
137    Ok(Some(crate::eval::logical(answer)))
138}

151/// `RexxNilObject::getHashValue`, whose member is stamped with this sentinel
152/// rather than derived from anything.
153const NIL_HASH: u64 = 0xdead_beef;
154
155/// A string's hash value: the keyed hash of its characters under the
156/// interpreter's `seed` (`r[class.string.hashcode]`). Strings that collide
157/// under one seed do not in general collide under another, which is what
158/// keeps a program from choosing keys that share one probe path.
159#[inline]
160pub(super) fn string_hash(seed: HashSeed, bytes: &[u8]) -> u64 {
161    seed.hash(bytes)
162}
163
164/// A class's hash value: an unkeyed hash of its id, the same in every run.
165/// Only a string's hash code takes the seed.
166fn class_hash(id: &[u8]) -> u64 {
167    let mut hash: u64 = 0;
168    for byte in id {
169        hash = hash
170            .wrapping_mul(31)
171            .wrapping_add(i64::from(byte.cast_signed()).cast_unsigned());
172    }
173    hash
174}

188/// The hash value as a number, which is what [`native_hash_code`] renders and
189/// what the mapped collections' store hashes an index by. Split out for the
190/// second caller, not changed.
191pub(super) fn hash_value(interp: &mut Interp, receiver: ObjRef) -> u64 {
192    if receiver == ObjRef::NIL {
193        return NIL_HASH;
194    }
195    match interp.receiver_kind(receiver) {
196        Ok(Primitive::String | Primitive::SmallInt) => {
197            string_hash(interp.hash_seed, &interp.to_text(receiver))
198        }
199        Ok(Primitive::Class(class)) => class_hash(interp.classes().id_string(class).as_bytes()),
200        // A string's hash value is the bytes' hash for a subclass instance as
201        // well, and a copy of `.nil` keeps `.nil`'s member.
202        Ok(Primitive::Instance { behaviour, .. }) => {
203            let nil = interp.object_model().nil_behaviour;
204            match super::string::subclass_text(interp, receiver) {
205                Some(text) => string_hash(interp.hash_seed, text),
206                None if behaviour == nil => NIL_HASH,
207                None => receiver.bits(),
208            }
209        }
210        _ => receiver.bits(),
211    }
212}

267/// The shared body of the two above: the receiver's string value, then
268/// [`Interp::apply_binary`] on it, which concatenates the receiver's string
269/// with the other operand and converts the other operand exactly as every
270/// other concatenation does.
271fn concat_through_string_value(
272    interp: &mut Interp,
273    op: Operator,
274    receiver: ObjRef,
275    args: &[Option<ObjRef>],
276) -> Result<Option<ObjRef>, Failure> {
277    let other = operator_argument(args)?;
278    let frame = interp.roots.activity_mut().push_frame();
279    let joined = interp
280        .required_string_value(receiver)
281        .and_then(|alias| {
282            interp.roots.activity_mut().push_temp(alias);
283            interp.apply_binary(op, alias, other)
284        })
285        .map(Some);
286    interp.roots.activity_mut().pop_frame(frame);
287    joined
288}

290/// `Object~string`: the receiver's readable string representation, which is an
291/// `OBJECTNAME` send, except for a `String`, whose own override answers
292/// itself.
293pub(super) fn native_string(
294    interp: &mut Interp,
295    _cleared: Cleared,
296    receiver: ObjRef,
297    _args: &[Option<ObjRef>],
298) -> Result<Option<ObjRef>, Failure> {
299    // A `String` subclass instance answers its bytes, and a `Stem` subclass
300    // instance its stem's.
301    if super::string::subclass_text(interp, receiver).is_some() {
302        return Ok(Some(super::string::base_string(interp, receiver)));
303    }
304    if let Some(store) = interp.stem_subclass_store(receiver) {
305        let text = interp.stem_string_representation(store)?;
306        return Ok(Some(interp.text_built(text)));
307    }
308    // `Pointer` overrides the send below: `p~objectName = 'zzz'` leaves
309    // `p~string` answering the address.
310    if pointer_address(interp, receiver).is_some() {
311        let text = interp.default_string_value_text(receiver);
312        return Ok(Some(interp.text_built(text)));
313    }
314    // **Sent rather than shortcut for an instance**, whose `OBJECTNAME` a
315    // program can replace: `::METHOD defaultName` returning `overridden` makes
316    // `o~string` answer `overridden`.
317    let sent = match interp.receiver_kind(receiver) {
318        Ok(Primitive::Instance { .. }) => true,
319        Ok(Primitive::Class(class)) => declared_class(interp, class),
320        _ => false,
321    };
322    if sent {
323        let caller = interp.caller();
324        let send =
325            |interp: &mut Interp| interp.send_message(receiver, OBJECTNAME, None, &[], caller);
326        // A level of its own where the send reaches a Rexx method.
327        let own_name = interp.answers_by_method(receiver, OBJECTNAME);
328        let answered = if own_name || interp.answers_by_method(receiver, DEFAULTNAME) {
329            interp.primitive_level(receiver, b"STRING", &[], send)?
330        } else {
331            send(interp)?
332        };
333        if let Some(answered) = answered {
334            let text = match own_name {
335                true => interp.object_name_answer_text(receiver, answered)?,
336                false => interp.default_string_value_text(answered),
337            };
338            return Ok(Some(interp.text_built(text)));
339        }
340        let text = interp.default_string_value_text(receiver);
341        return Ok(Some(interp.text_built(text)));
342    }
343    let text = match interp.is_stem_object(receiver) {
344        true => interp.stem_string_representation(receiver)?,
345        false => interp.string_value_text(receiver)?,
346    };
347    Ok(Some(interp.text_built(text)))
348}

350/// Whether `class` is one a program made rather than one of the language's
351/// own. Only such a class can carry its own `DEFAULTNAME` on its class side
352/// (98.985 refuses additions to the language's classes), so only its rendering
353/// sends one, as `objectName` does for a class whose behaviour is not
354/// primitive.
355fn declared_class(interp: &mut Interp, class: ObjRef) -> bool {
356    let id = interp.classes().id_string(class).to_string();
357    interp.classes().system_lookup(&id) != Some(class)
358}

360/// `DEFAULTNAME` sent to `receiver` by the primitive `OBJECTNAME`, under its
361/// level where that reaches a Rexx method.
362fn default_name_sent(interp: &mut Interp, receiver: ObjRef) -> Result<Option<ObjRef>, Failure> {
363    let caller = interp.caller();
364    let send = |interp: &mut Interp| interp.send_message(receiver, DEFAULTNAME, None, &[], caller);
365    if interp.answers_by_method(receiver, DEFAULTNAME) {
366        interp.primitive_level(receiver, OBJECTNAME, &[], send)
367    } else {
368        send(interp)
369    }
370}

372/// `Object~objectName`: the name something gave this object, or the default
373/// name for one nothing has named.
374pub(super) fn native_object_name(
375    interp: &mut Interp,
376    _cleared: Cleared,
377    receiver: ObjRef,
378    _args: &[Option<ObjRef>],
379) -> Result<Option<ObjRef>, Failure> {
380    let kind = interp
381        .receiver_kind(receiver)
382        .map_err(|kind| Failure::from(Loud::unknown_receiver(kind)))?;
383    let name = match kind {
384        // `defaultName` from `.String`'s own id; `native_object_name_set`
385        // names no string.
386        Primitive::String | Primitive::SmallInt => b"a String".to_vec(),
387        // **A set name, or the answer to a `DEFAULTNAME` send.** The object's
388        // own `Object`-scope variable is read and, for anything that is not a
389        // base class, `DEFAULTNAME` is sent rather than derived -- so a class
390        // overriding `defaultName` decides what an unnamed instance answers,
391        // and a name set afterwards wins over it.
392        Primitive::Instance { .. } => match interp.instance_name(receiver) {
393            Some(held) => held,
394            None => {
395                let answered = default_name_sent(interp, receiver)?;
396                match answered {
397                    Some(answered) => interp.default_string_value_text(answered),
398                    None => interp.default_string_value_text(receiver),
399                }
400            }
401        },
402        // Derived from the class id rather than read through
403        // `string_value_text`, which for this receiver answers the *referenced*
404        // value: `~objectName` is the reference's own and is one of the names
405        // its `UNKNOWN` never sees. For example, `vr = 5; o = >vr; say
406        // o~objectName` is `a VariableReference` where `say o~string` is `5`.
407        Primitive::VariableReference => match interp.heap.object_name(receiver) {
408            Some(name) => name.to_vec(),
409            None => crate::environment::default_object_name("VariableReference").into_bytes(),
410        },
411        // The object's own `rendered` bytes rather than `string_value_text`,
412        // which for this receiver answers the **traceback line**:
413        // `StackFrameClass` overrides `stringValue()` and leaves
414        // `defaultName()` alone, the split `NativeObject::string_value`
415        // carries. For example, six renderings of one frame: `say f`,
416        // `~string`, `~makeString` and an `Array` join are the traceback line
417        // where `~objectName` and `~defaultName` are `a StackFrame`. Read
418        // rather than derived, so a name `~objectName=` has set still wins --
419        // measured, `f~objectName = 'tagged'` then `f~objectName` is `tagged`
420        // while `say f` is unchanged.
421        Primitive::StackFrame => match interp.heap.get(receiver).map(|held| &held.body) {
422            Some(Body::Native(native)) => native.rendered().to_vec(),
423            _ => return Err(Loud::receiver_class("a stack frame this crate did not build").into()),
424        },
425        // Derived for the same reason, and the two answers part here too: a
426        // stem's `string_value_text` is its default. For example, `s. = 'dflt';
427        // o = s.; say o~objectName` is `a Stem` where `say o~string` is `dflt`.
428        Primitive::Stem => match interp.heap.object_name(receiver) {
429            Some(name) => name.to_vec(),
430            None => crate::environment::default_object_name("Stem").into_bytes(),
431        },
432        Primitive::Class(class) if declared_class(interp, class) => {
433            match interp.classes().set_name(class) {
434                Some(held) => held.as_bytes().to_vec(),
435                None => match default_name_sent(interp, receiver)? {
436                    Some(answered) => interp.default_string_value_text(answered),
437                    None => interp.default_string_value_text(receiver),
438                },
439            }
440        }
441        Primitive::Object
442        | Primitive::Array
443        | Primitive::Class(_)
444        | Primitive::Package
445        | Primitive::Method
446        | Primitive::Routine
447        | Primitive::Context
448        | Primitive::RexxInfo
449        | Primitive::Message => interp.default_string_value_text(receiver),
450    };
451    Ok(Some(interp.text_built(name)))
452}

454/// `Object~objectName=`: replaces what `~objectName` answers, and with it
455/// every rendering of the object. It answers nothing.
456pub(super) fn native_object_name_set(
457    interp: &mut Interp,
458    _cleared: Cleared,
459    receiver: ObjRef,
460    args: &[Option<ObjRef>],
461) -> Result<Option<ObjRef>, Failure> {
462    let Some(Some(argument)) = args.first().copied() else {
463        return Err(Raised::missing_method_argument(1).into());
464    };
465    let argument = required_string_argument(interp, argument, 1)?;
466    let name = interp.to_text(argument).into_owned();
467    let kind = interp
468        .receiver_kind(receiver)
469        .map_err(|kind| Failure::from(Loud::unknown_receiver(kind)))?;
470    match kind {
471        Primitive::Class(class) => {
472            let name = String::from_utf8_lossy(&name).into_owned();
473            interp.classes().set_object_name(class, &name);
474        }
475        Primitive::Instance { .. } => {
476            let Some(object) = interp.heap.get_mut(receiver) else {
477                return Err(
478                    Loud::unknown_receiver("a value whose object is no longer live").into(),
479                );
480            };
481            match &mut object.body {
482                Body::Instance { name: held, .. } => *held = Some(name.as_slice().into()),
483                other => unreachable!("an instance receiver is Body::Instance, got {other:?}"),
484            }
485        }
486        Primitive::Package
487        | Primitive::Method
488        | Primitive::Routine
489        | Primitive::Context
490        | Primitive::StackFrame
491        | Primitive::RexxInfo
492        | Primitive::Message => {
493            let Some(object) = interp.heap.get_mut(receiver) else {
494                return Err(
495                    Loud::unknown_receiver("a value whose object is no longer live").into(),
496                );
497            };
498            match &mut object.body {
499                Body::Native(native) => native.set_rendered(&name),
500                other => {
501                    unreachable!("each of these receivers is Body::Native, got {other:?}")
502                }
503            }
504        }
505        Primitive::Array | Primitive::VariableReference | Primitive::Stem => {
506            interp.heap.set_object_name(receiver, &name);
507        }
508        // A String is value-based (Deviation 36).
509        Primitive::String | Primitive::SmallInt => {
510            return Err(Loud::object_name_of_a_value().into());
511        }
512        // `.nil`'s own behaviour hides `OBJECTNAME=`, so only a scope
513        // override reaches it here.
514        Primitive::Object => interp.set_nil_name(&name),
515    }
516    Ok(None)
517}

519/// `Object~setMethod(name, method, scope)`: attach one method to this object
520/// alone.
521pub(super) fn native_set_method(
522    interp: &mut Interp,
523    _cleared: Cleared,
524    receiver: ObjRef,
525    args: &[Option<ObjRef>],
526) -> Result<Option<ObjRef>, Failure> {
527    let name = method_name_argument(interp, args)?;
528    let source = match args.get(1).copied().flatten() {
529        None => None,
530        Some(source) if interp.receiver_kind(source) == Ok(Primitive::Method) => Some(source),
531        Some(source) => Some(compile_method_source(
532            interp,
533            &name,
534            source,
535            "method",
536            SourceTaker::Method,
537        )?),
538    };
539    // The method object is built ahead of the checks, then the instance method
540    // is defined over a fresh scope copy: the method's `.context~executable` is
541    // not the object `setMethod` was given.
542    let source = match source {
543        None => None,
544        Some(object) => {
545            let (floated, _) = scoped_method(interp, object, ObjRef::NIL)?;
546            interp.roots.activity_mut().push_temp(floated);
547            Some(floated)
548        }
549    };
550    let scope = set_method_scope(interp, receiver, args)?;
551    check_restricted_method(interp, receiver, b"SETMETHOD")?;
552    let entry = match source {
553        None => None,
554        Some(floated) => {
555            let (executable, method) = scoped_method(interp, floated, scope)?;
556            Some(ObjectMethod {

603/// The fourth access check: `run`, `setMethod` and `unsetMethod` may be sent
604/// only from a method of the receiving object itself or from a class method of
605/// a class it is an instance of.
606fn check_restricted_method(
607    interp: &mut Interp,
608    receiver: ObjRef,
609    name: &[u8],
610) -> Result<(), Failure> {
611    let caller = interp.caller();
612    // A routine or program context has no sender, which the private check
613    // refuses first: `run`, `setMethod` and `unsetMethod` are each private, so
614    // `check_private`'s own `caller.receiver()` arm has already answered for a
615    // caller with no receiver.
616    let Some(sender) = caller.receiver() else {
617        return Err(Raised::restricted_method(name).into());
618    };
619    if sender == receiver {
620        return Ok(());
621    }
622    // A class method of a class this object is an instance of.
623    if interp.is_class_object(sender)
624        && let Some(class) = interp.class_of_value(receiver)
625        && interp.classes().is_a(class, sender)
626    {
627        return Ok(());
628    }
629    Err(Raised::restricted_method(name).into())
630}

643/// [`native_copy`]'s copy of `receiver`, left on the temps.
644fn copy_object(interp: &mut Interp, receiver: ObjRef) -> Result<ObjRef, Failure> {
645    // **An `Array` and a `Stem` are copyable too**, and each keeps its own
646    // behaviour: `.Array~of('x','y')~copy~class~id` is `Array` and a stem's is
647    // `Stem`. The clone below is what makes both deep -- the slots and the
648    // tails travel with the body, so appending to the copy leaves the receiver
649    // alone (`a~items` 2 against `b~items` 3) and a tail written on the copy
650    // does not appear on the receiver.
651    let behaviour = match interp.receiver_kind(receiver) {
652        Ok(Primitive::Instance { .. } | Primitive::Message) => interrexx_core::BehaviourId::OBJECT,
653        Ok(Primitive::Array) => interrexx_core::BehaviourId::ARRAY,
654        Ok(Primitive::Stem) => interrexx_core::BehaviourId::STEM,
655        // A value carried in its handle is its own copy (D15).
656        Ok(Primitive::SmallInt) => return Ok(receiver),
657        Ok(Primitive::String) if interp.heap.get(receiver).is_none() => return Ok(receiver),
658        Ok(Primitive::String) => interrexx_core::BehaviourId::STRING,
659        // `StackFrameClass` has no `copy` of its own: a frame's copy is a
660        // different object answering the same `~type`, `~line` and `~context`.
661        Ok(Primitive::StackFrame) => interrexx_core::BehaviourId::OBJECT,
662        // For example, a copy reads and writes the variable the original names.
663        Ok(Primitive::VariableReference) => interrexx_core::BehaviourId::OBJECT,
664        Ok(Primitive::Method | Primitive::Routine) => {
665            interp.roots.activity_mut().push_temp(receiver);
666            let copy = interp
667                .copy_method_object(receiver)
668                .ok_or_else(|| Failure::from(Loud::unrecorded_executable()))?;
669            return Ok(copy);
670        }
671        // A field copy, then the deep copy's tables.
672        Ok(Primitive::Package) => interrexx_core::BehaviourId::OBJECT,
673        // Reached only through a scope override, `.nil`'s own behaviour
674        // hiding `COPY`.
675        Ok(Primitive::Object) => return Ok(copy_nil(interp)),
676        // rexxref forbids copying each of these on every route: `copy` raises
677        // 93.970 for a plain send, and a scope override reaches the generic
678        // copy, which copies.
679        Ok(Primitive::Class(_) | Primitive::Context | Primitive::RexxInfo) => {
680            let target = interp.string_value_text(receiver)?;
681            return Err(Raised::copy_not_supported(&target).into());
682        }
683        Err(kind) => return Err(Loud::unknown_receiver(kind).into()),
684    };
685    // The allocation below collects first, and while the cloned body is a
686    // local the collector does not walk, every value in it is reachable from
687    // the receiver and from nowhere else.
688    interp.roots.activity_mut().push_temp(receiver);
689    let package = interp.package_data(receiver).is_some();
690    if package {
691        interp.build_package_shares(receiver)?;
692    }
693    let Some(source) = interp.heap.get(receiver) else {
694        return Err(Loud::unknown_receiver("a value whose object is no longer live").into());
695    };
696    let mut body = source.body.clone();
697    // A stem copy copies each tail's own value, so an exposed tail comes over
698    // as one that holds none.
699    if let Body::Stem { exposed, .. } = &mut body {
700        *exposed = None;
701    }
702    // A `String` subclass instance's text is charged as `String~new` charges
703    // it, since `Body::held_bytes` does not report it.
704    let text = match &body {
705        Body::Instance {
706            native: Some(state),
707            ..
708        } => state.text().map_or(0, <[u8]>::len),
709        _ => 0,
710    };
711    let copy = interp.alloc_charged(behaviour, body);
712    interp.charge_unheld(text);
713    interp.roots.activity_mut().push_temp(copy);
714    duplicate_collection_stores(interp, copy);
715    duplicate_stem_store(interp, copy);
716    duplicate_executable(interp, receiver, copy);
717    if package {
718        duplicate_package_tables(interp, copy)?;
719    }
720    if let Some(name) = interp.heap.object_name(receiver).map(<[u8]>::to_vec) {
721        interp.heap.set_object_name(copy, &name);
722    }
723    if interp.is_message(copy) {
724        reset_copied_message(interp, copy);
725    }
726    if interp.answers_uninit(copy) {
727        interp.heap.set_uninit(copy);
728    }
729    // An `EventSemaphore` copy is posted where the receiver is; its state is
730    // not in the body. A `MutexSemaphore` copy starts free (Deviation 136).
731    if interp.event_posted(receiver) {
732        interp.set_event_posted(copy);
733    }
734    Ok(copy)
735}

737/// `.nil`'s copy, left on the temps: a distinct `Object` with `.nil`'s methods
738/// and its name, as a generic copy clones it. For example, the copy is not
739/// `.nil`, renders `The NIL object` and answers `an Object` for `~defaultName`.
740fn copy_nil(interp: &mut Interp) -> ObjRef {
741    let model = interp.object_model();

743    let body = Body::Instance {
744        class,
745        behaviour,
746        name: Some(interp.nil_name().into()),
747        pools: interrexx_core::ScopePools::new(),
748        own: None,
749        native: None,
750    };
751    let copy = interp.alloc_charged(BehaviourId::OBJECT, body);
752    interp.roots.activity_mut().push_temp(copy);
753    copy

774/// The pool entries a collection's CONTENTS live in, as (scope class, name).
775pub(super) const COLLECTION_STORES: &[(&str, &[u8])] = &[
776    ("Array", b"ITEMS"),
777    ("List", b"CHAIN"),
778    ("Table", b"HASHINDEXES"),
779    ("Table", b"HASHITEMS"),
780    ("Table", b"HASHNEXT"),
781    ("Table", b"HASHBUCKETS"),
782    // A `Directory`'s method table travels with the copy and is its own: the
783    // copy duplicates it rather than sharing it. Measured: `g = f~copy` then
784    // `g~unsetMethod('M')` leaves `f['M']` answering.
785    ("Table", b"METHODINDEXES"),
786    ("Table", b"METHODITEMS"),
787    ("Table", b"METHODNEXT"),
788    ("Table", b"METHODBUCKETS"),
789];

791/// Replaces each of [`COLLECTION_STORES`] on `copy` with an array of its own.
792fn duplicate_collection_stores(interp: &mut Interp, copy: ObjRef) {
793    for (scope_name, entry) in COLLECTION_STORES {
794        let Some(scope) = interp.classes().lookup(scope_name) else {
795            continue;
796        };
797        let held = match interp.heap.get(copy).map(|object| &object.body) {
798            Some(Body::Instance { pools, .. }) => pools.get(scope, entry),
799            _ => return,
800        };
801        let Some(held) = held else { continue };
802        let Some(slots) = interp.array_slots_of(held) else {
803            continue;
804        };
805        let fresh = interp.alloc_charged(BehaviourId::ARRAY, Body::array(slots));
806        interp.roots.activity_mut().push_temp(fresh);
807        interp.set_pool_variable(copy, scope, entry, fresh);
808    }
809}

813fn duplicate_stem_store(interp: &mut Interp, copy: ObjRef) {
814    let Some(store) = interp.stem_subclass_store(copy) else {

831/// Carries a `Method` or `Routine` subclass instance's executable records,
832/// and an annotation table of its own, over to `copy`, as
833/// [`Interp::copy_method_object`] does for a base one.
834fn duplicate_executable(interp: &mut Interp, original: ObjRef, copy: ObjRef) {
835    interp.copy_executable_records(original, copy);
836    let model = interp.object_model();
837    for scope in [model.method, model.routine] {
838        let held = interp
839            .pools_of(copy)
840            .and_then(|pools| pools.get(scope, super::construct::SUBCLASS_ANNOTATIONS));
841        let Some(body) = held.and_then(|table| interp.heap.get(table).map(|t| t.body.clone()))
842        else {
843            continue;
844        };
845        let fresh = interp.alloc_with(BehaviourId::OBJECT, body);
846        interp.roots.activity_mut().push_temp(fresh);
847        duplicate_collection_stores(interp, fresh);
848        interp.set_pool_variable(copy, scope, super::construct::SUBCLASS_ANNOTATIONS, fresh);
849    }
850}

852/// `copy`'s `.ROUTINES` table and annotation table become copies of their own.
853/// Every other table the copy holds it shares, and its routine, class and
854/// import fields are copies already, being the body's own.
855fn duplicate_package_tables(interp: &mut Interp, copy: ObjRef) -> Result<(), Failure> {
856    let routines = crate::environment::PackageTable::Routines.index();
857    if let Some(table) = interp
858        .package_data(copy)
859        .and_then(|data| data.tables[routines])
860    {
861        let fresh = copy_object(interp, table)?;
862        if let Some(data) = interp.package_data_mut(copy) {
863            data.tables[routines] = Some(fresh);
864        }
865    }
866    let scope = interp.object_model().package;
867    let annotations = match interp.heap.get(copy).map(|held| &held.body) {
868        Some(Body::Native(native)) => native.annotations(),
869        Some(Body::Instance { pools, .. }) => {
870            pools.get(scope, super::construct::SUBCLASS_ANNOTATIONS)
871        }
872        _ => None,
873    };
874    if let Some(table) = annotations {
875        let fresh = copy_object(interp, table)?;
876        match interp.heap.get_mut(copy).map(|held| &mut held.body) {
877            Some(Body::Native(native)) => native.set_annotations(fresh),
878            Some(Body::Instance { .. }) => {
879                interp.set_pool_variable(
880                    copy,
881                    scope,
882                    super::construct::SUBCLASS_ANNOTATIONS,
883                    fresh,
884                );
885            }
886            _ => {}
887        }
888    }
889    Ok(())
890}

892/// `Object~run(method [, option [, argument ...]])`: run a method on this
893/// object as if the object had defined it.
894pub(super) fn native_run(
895    interp: &mut Interp,
896    _cleared: Cleared,
897    receiver: ObjRef,
898    args: &[Option<ObjRef>],
899) -> Result<Option<ObjRef>, Failure> {
900    let Some(Some(source)) = args.first().copied() else {
901        return Err(Raised::missing_named_argument("method").into());
902    };
903    let (executable, method) = run_method(interp, source)?;
904    interp.roots.activity_mut().push_temp(executable);
905    let values = run_arguments(interp, args)?;
906    check_restricted_method(interp, receiver, b"RUN")?;
907    // For example, `.context~executable` in a method `run` ran is the object
908    // `run` was given the first time and a copy after.
909    interp.primitive_level(receiver, b"RUN", args, |interp| {
910        invoke_executable(
911            interp,
912            method,
913            executable,
914            receiver,
915            UNNAMED_METHOD,
916            &values,
917        )
918    })
919}

921/// Runs `method` on `receiver` under the `.nil` scope as the send of `name`,
922/// where no dictionary entry names the `Method` object `executable` it runs
923/// for: the activation carries that object for `.context~executable`.
924pub(super) fn invoke_executable(
925    interp: &mut Interp,
926    method: interrexx_core::MethodId,
927    executable: ObjRef,
928    receiver: ObjRef,
929    name: &[u8],
930    args: &[Option<ObjRef>],
931) -> Result<Option<ObjRef>, Failure> {
932    let resolution = Resolution {
933        scope: ObjRef::NIL,
934        method,
935    };
936    if interp.security_armed {
937        return invoke_armed_executable(interp, resolution, executable, receiver, name, args);
938    }
939    let started = interp.begin_invoke(resolution, None, receiver, name, args)?;
940    if let Started::Entered = started
941        && interp.method_bodies.contains_key(&method)
942    {
943        interp.activation_mut().cold_mut().executable = Some(executable);
944    }
945    interp.complete_send(started)
946}

948/// [`invoke_executable`] once a manager has been installed. `Method~run` is
949/// not a send, so this call's pass through the seam asks no manager. The seam
950/// consumes the mark on this call's own pass, so it reaches no other send.
951#[cold]
952#[inline(never)]
953fn invoke_armed_executable(
954    interp: &mut Interp,
955    resolution: Resolution,
956    executable: ObjRef,
957    receiver: ObjRef,
958    name: &[u8],

966        && interp.method_bodies.contains_key(&resolution.method)
967    {
968        interp.activation_mut().cold_mut().executable = Some(executable);
969    }
970    interp.complete_send(started)
971}

987/// `~run`'s `Individual`/`Array` option and the arguments behind it.
988fn run_arguments(
989    interp: &mut Interp,
990    args: &[Option<ObjRef>],
991) -> Result<Vec<Option<ObjRef>>, Failure> {
992    let Some(option) = args.get(1).copied() else {
993        return Ok(Vec::new());
994    };
995    let Some(option) = option else {
996        return Err(Raised::missing_named_argument("argument style").into());
997    };
998    let option = required_string_named_argument(interp, option, "argument style")?;
999    let text = interp.to_text(option).to_vec();
1000    match text.first().map(u8::to_ascii_uppercase) {
1001        Some(b'I') => Ok(args[2..].to_vec()),
1002        // `argCount < 3` and `argCount > 3` are separate raises with separate
1003        // numbers. For example, `self~run('return 1', 'A')` is `88.901 ...
1004        // argument argument array is required.` at rc 168, and a fourth
1005        // argument is `93.902 ... 3 expected.` at rc 163.
1006        Some(b'A') => {
1007            let Some(Some(array)) = args.get(2).copied() else {
1008                return Err(Raised::missing_named_argument("argument array").into());
1009            };
1010            if args.len() > 3 {
1011                return Err(Raised::too_many_method_arguments(3).into());
1012            }
1013            array_argument(interp, array, ArrayArgument::Named("argument array"))
1014        }
1015        _ => Err(Raised::method_option_not_recognised("AI", &text).into()),
1016    }
1017}

1094/// The `Message` object `~start` and `~startWith` answer, its send made by a
1095/// new activity.
1096fn started_message(
1097    interp: &mut Interp,
1098    receiver: ObjRef,
1099    message: ObjRef,
1100    args: &[Option<ObjRef>],
1101) -> Result<NativeStarted, Failure> {
1102    let (name, scope) = decode_message_name(interp, Some(message))?;
1103    interp.validate_scope_override(receiver, scope)?;
1104    let class = interp.object_model().message;
1105    let object = interp.native_instance(class);

1190/// The held send made, its answer or its condition recorded for `result`,
1191/// `completed` and `hasError`, and the condition raised here too. `then` is
1192/// the work left where the send enters a Rexx activation.
1193fn dispatch_held_message(
1194    interp: &mut Interp,
1195    message: ObjRef,
1196    then: Then,
1197) -> Result<NativeStarted, Failure> {
1198    if interp.started_messages.contains(&message) {
1199        return Err(Raised::message_reuse().into());
1200    }
1201    let (Some(target), Some(name), Some(arguments)) = (
1202        interp.native_entry(message, MESSAGE_TARGET),
1203        interp.native_entry(message, MESSAGE_NAME),
1204        interp.native_entry(message, MESSAGE_ARGUMENTS),
1205    ) else {
1206        return Err(crate::Raised::receiver_kind(NOT_A_MESSAGE).into());
1207    };
1208    let scope = interp.native_entry(message, MESSAGE_SCOPE);
1209    let name = interp.to_text(name).into_owned();
1210    let values = interp.array_slots_of(arguments).unwrap_or_default();
1211    clear_completion(interp, message);
1212    interp.validate_scope_override(target, scope)?;
1213    interp.set_notify_message(Some(message));
1214    interp.record_message_sender(message);
1215    let caller = interp.caller();
1216    let sent = match interp.begin_send(target, &name, scope, &values, caller) {
1217        Ok(Started::Entered) => return Ok(NativeStarted::Entered(then)),
1218        Ok(Started::Ran(value)) => Ok(value),
1219        Err(failure) => Err(failure),
1220    };
1221    record_held(interp, message, sent).map(NativeStarted::Ran)
1222}

1224/// A held send's outcome recorded on `message`, and the send's own answer.
1225pub(super) fn record_held(
1226    interp: &mut Interp,
1227    message: ObjRef,
1228    sent: Result<Option<ObjRef>, Failure>,
1229) -> Result<Option<ObjRef>, Failure> {
1230    match sent {
1231        Ok(answer) => {
1232            interp.set_notify_message(None);
1233            if let Some(answer) = answer {
1234                interp.set_native_entry(message, MESSAGE_RESULT, answer);
1235            }
1236            // An error a send of the same message made during this one stays:
1237            // the result-returned and raise-error flags are separate.
1238            interp.message_outcomes.entry(message).or_insert(None);
1239            interp.message_completed(message);
1240            interp.notify_parties(message)?;
1241            Ok(answer)
1242        }
1243        // Only a `SYNTAX` condition notifies the message, and only the one in
1244        // the activation's single slot: a send made later from the same
1245        // activation took it over. Any other leaves the message uncompleted.
1246        // The slot keeps it for the activation's trap search
1247        // ([`Interp::notify_slot_failure`]).
1248        Err(Failure::Raised(raised))
1249            if raised.condition == "SYNTAX" && interp.notify_slot() == Some(message) =>
1250        {
1251            interp.activity.failed_sends.push(message);
1252            interp
1253                .message_outcomes
1254                .insert(message, Some(raised.clone()));
1255            interp.message_completed(message);
1256            Err(Failure::Raised(raised))
1257        }
1258        Err(other) => Err(other),
1259    }
1260}

1452/// `Message~notify(target)`: `target` is sent `messageComplete` when the send
1453/// completes, and at once where it already has.
1454pub(super) fn native_message_notify(
1455    interp: &mut Interp,
1456    _cleared: Cleared,
1457    receiver: ObjRef,
1458    args: &[Option<ObjRef>],
1459) -> Result<Option<ObjRef>, Failure> {
1460    const ARGUMENT: &str = "notification target";
1461    let Some(target) = args.first().copied().flatten() else {
1462        return Err(Raised::missing_named_argument(ARGUMENT).into());
1463    };
1464    let Some(notification) = interp.rexx_package_class(b"MESSAGENOTIFICATION") else {
1465        return Err(Loud::environment_symbol(b".MESSAGENOTIFICATION").into());
1466    };
1467    let notifiable = interp
1468        .class_of_value(target)
1469        .is_some_and(|class| interp.classes().is_a(class, notification));
1470    if !notifiable {
1471        return Err(Raised::argument_not_an_instance(ARGUMENT, "MessageNotification").into());
1472    }
1473    let parties = match interp.native_entry(receiver, MESSAGE_PARTIES) {
1474        Some(parties) => parties,
1475        None => {
1476            let parties = interp.alloc_charged(
1477                BehaviourId::ARRAY,
1478                Body::Array {
1479                    dimensions: None,
1480                    slots: Vec::new().into(),
1481                },
1482            );
1483            interp.set_native_entry(receiver, MESSAGE_PARTIES, parties);
1484            parties
1485        }
1486    };
1487    if let Some(Body::Array { slots, .. }) = interp.heap.get_mut(parties).map(|held| &mut held.body)
1488    {
1489        let before = slots.capacity();
1490        slots.push(Some(target));
1491        let grown = slots.capacity() - before;
1492        interp.charge_growth(grown * interrexx_core::SLOT_BYTES);
1493    }
1494    let notified = interp.message_outcomes.contains_key(&receiver)
1495        && !interp.unnotified_messages.contains(&receiver);
1496    if notified {
1497        let caller = interp.caller();
1498        interp.send_message(target, MESSAGE_COMPLETE, None, &[Some(receiver)], caller)?;
1499    }
1500    Ok(None)
1501}

1626    /// The notification once the waiters are woken: `messageComplete` sent to
1627    /// each object `~notify` had named when it began, in order. `message`
1628    /// stays unnotified where one of them fails.
1629    pub(crate) fn notify_parties(&mut self, message: ObjRef) -> Result<(), Failure> {
1630        if let Some(parties) = self.native_entry(message, MESSAGE_PARTIES) {
1631            self.unnotified_messages.insert(message);
1632            let frame = self.roots.activity_mut().push_frame();
1633            self.roots.activity_mut().push_temp(message);
1634            self.roots.activity_mut().push_temp(parties);
1635            let count = self.array_slots(parties).map_or(0, <[_]>::len);
1636            let mut sent = Ok(None);
1637            for index in 0..count {
1638                let party = self
1639                    .array_slots(parties)
1640                    .and_then(|slots| slots.get(index).copied().flatten());
1641                let Some(party) = party else {
1642                    continue;
1643                };
1644                let caller = self.caller();
1645                sent = pinned!(
1646                    self,
1647                    crate::pinning::PinKind::Notification,
1648                    self.send_message(party, MESSAGE_COMPLETE, None, &[Some(message)], caller)
1649                );
1650                if sent.is_err() {
1651                    break;
1652                }
1653            }
1654            self.roots.activity_mut().pop_frame(frame);
1655            sent?;
1656            self.unnotified_messages.remove(&message);
1657        }
1658        Ok(())
1659    }

1680/// 98.905 where the activity running `message`'s send waits, through a chain
1681/// of waits, on the running one.
1682fn check_message_deadlock(interp: &Interp, message: ObjRef) -> Result<(), Failure> {
1683    match interp.message_runner(message) {
1684        // Where the running activity runs that send itself, the wait parks
1685        // like any other, so the other activities still run, and it is
1686        // refused once nothing else can run, whatever was loaded
1687        // ([`Interp::waits_on_its_own_message`], Deviation 20).
1688        Some(runner) if runner != interp.running_activity() && interp.deadlocks(runner) => {
1689            Err(crate::error::Raised::deadlock().into())
1690        }
1691        _ => Ok(()),
1692    }

1710impl Interp {
1711    /// The SYNTAX condition a kept condition object stands for once Rexx code
1712    /// may have changed it, made again from its `CODE` and `ADDITIONAL` as
1713    /// `RAISE SYNTAX` makes one: `RC`, `ERRORTEXT` and `MESSAGE` are written
1714    /// back from them, and a `CODE` that is not a string naming a valid error
1715    /// number, or an `ADDITIONAL` that is not an `Array`, is 93.900 here
1716    /// instead (`r[class.message.result.changed]`). A code with no message is
1717    /// raised as `RAISE SYNTAX` raises it, 98.941.
1718    pub(crate) fn rebuilt_kept_syntax(
1719        &mut self,
1720        raised: Raised,
1721        object: ObjRef,
1722    ) -> Result<Raised, Failure> {
1723        let code = self
1724            .condition_entry_held(object, b"CODE")
1725            .filter(|code| {
1726                matches!(
1727                    self.receiver_kind(*code),
1728                    Ok(Primitive::String | Primitive::SmallInt)
1729                )
1730            })
1731            .and_then(|code| crate::run::raised::syntax_code(&self.to_text(code)));
1732        let Some(code) = code else {
1733            return Err(Raised::method_user_defined(
1734                "the CODE of a SYNTAX condition object must be a valid error number",
1735            )
1736            .into());
1737        };
1738        let items = match self.condition_entry_held(object, b"ADDITIONAL") {
1739            // A SYNTAX condition always has the entry: one Rexx code removed
1740            // is an empty `Array` again.
1741            None => {
1742                let empty = self.additional_array(&[]);
1743                self.condition_put(object, b"ADDITIONAL", empty)?;
1744                Vec::new()
1745            }
1746            Some(additional) => match self.array_slots_of(additional) {
1747                Some(items) => items,
1748                None => {
1749                    return Err(Raised::method_user_defined(
1750                        "the ADDITIONAL of a SYNTAX condition object must be an array",
1751                    )
1752                    .into());
1753                }
1754            },
1755        };
1756        let mut substitutions = Vec::with_capacity(items.len());
1757        for item in items {
1758            substitutions.push(match item {
1759                Some(item) => self.string_value_text(item)?,
1760                None => Vec::new(),
1761            });
1762        }
1763        let made = crate::run::raised::raised_for_code(code, substitutions);
1764        if (made.number, made.sub) == (98, 941) {
1765            return Err(made.into());
1766        }
1767        let rc = self.text(made.number.to_string().as_bytes());
1768        self.condition_put(object, b"RC", rc)?;
1769        let errortext = self.text_built(made.message(made.number, 0));
1770        self.condition_put(object, b"ERRORTEXT", errortext)?;
1771        let message = match made.sub {
1772            0 => ObjRef::NIL,
1773            sub => self.text_built(made.message(made.number, sub)),
1774        };
1775        self.condition_put(object, b"MESSAGE", message)?;
1776        Ok(Raised {
1777            number: made.number,
1778            sub: made.sub,
1779            additional: made.additional,
1780            rc: made.rc,
1781            ..raised
1782        })
1783    }
1784}

1786/// [`native_message_result`] once the send has completed. A send made again
1787/// between the wake and this read has cleared the outcome, which answers
1788/// `.nil`.
1789fn message_result(interp: &mut Interp, message: ObjRef) -> Result<Option<ObjRef>, Failure> {
1790    match interp.message_outcomes.get(&message) {
1791        Some(Some(raised)) => {
1792            let raised = raised.clone();
1793            let Some(object) = interp.native_entry(message, MESSAGE_CONDITION) else {
1794                return Err(Failure::Raised(raised));
1795            };
1796            let raised = if raised.condition == "SYNTAX" {
1797                interp.rebuilt_kept_syntax(*raised, object)?
1798            } else {
1799                *raised
1800            };
1801            let lines = interp.traceback_lines(object)?;
1802            interp.reraise_kept_condition(raised, object, lines)
1803        }
1804        Some(None) => {
1805            let held = interp.native_entry(message, MESSAGE_RESULT);
1806            Ok(Some(held.unwrap_or(ObjRef::NIL)))
1807        }
1808        None => Ok(Some(ObjRef::NIL)),
1809    }
1810}

1924/// A message name, or a two-item array of a name and the class to start the
1925/// method search from.
1926pub(super) fn decode_message_name(
1927    interp: &mut Interp,
1928    message: Option<ObjRef>,
1929) -> Result<(Vec<u8>, Option<ObjRef>), Failure> {
1930    let Some(message) = message else {
1931        return Err(Raised::missing_named_argument("message name").into());
1932    };
1933    // A string names the message directly and gets no conversion messages.
1934    if matches!(
1935        interp.receiver_kind(message),
1936        Ok(Primitive::String | Primitive::SmallInt)
1937    ) {
1938        return Ok((interp.to_text(message).to_ascii_uppercase(), None));
1939    }
1940    // An instance of a `String` subclass is a name made of its characters
1941    // (Deviation 109).
1942    if interp.is_string_valued(message) {
1943        let text = interp.required_string_value(message)?;
1944        return Ok((interp.to_text(text).to_ascii_uppercase(), None));
1945    }
1946    // A plain `Array` is read as it stands, holes and all. Any other object
1947    // is sent `request('ARRAY')`, so an `Array` subclass and an object that
1948    // answers an array name a message too.
1949    let plain_array = interp.object_model().array;
1950    let answered = if interp.array_body(message).is_some()
1951        && interp.class_of_value(message) == Some(plain_array)
1952    {
1953        Some(message)
1954    } else {
1955        interp.request_array_value(message)?
1956    };
1957    let slots = answered.and_then(|answered| {
1958        interp.roots.activity_mut().push_temp(answered);
1959        interp.array_slots_of(answered)
1960    });
1961    let Some(slots) = slots else {
1962        // Preserve array precedence, including for objects with both
1963        // conversions. A non-array can still supply an unscoped name.
1964        let is_array = interp
1965            .class_of_value(message)
1966            .is_some_and(|class| interp.classes().is_a(class, plain_array));
1967        if !is_array && let Some(text) = interp.blamed_string_conversion(message)? {
1968            return Ok((interp.to_text(text).to_ascii_uppercase(), None));
1969        }
1970        let rendered = interp.string_value_text(message)?;
1971        return Err(Raised::message_name_shape(&rendered).into());
1972    };
1973    // `messageArgCount() != 2`, which is `lastItem` and not the slot count: a
1974    // trailing empty slot is not an element. For example, `o~send(('M',))` is
1975    // 93.946 where the same array is `~size` 2.
1976    let slots = &slots[..message_argument_count(&slots)];
1977    if slots.len() != 2 {
1978        return Err(Raised::message_array_shape().into());
1979    }
1980    // `stringArgument` distinguishes an empty slot from a value with no string
1981    // value. For example, `o~send((, .K))` is `88.901`, naming `message name`,
1982    // and `o~send((.nil, .K))` is `88.909`, naming the same, both rc 168.
1983    let Some(name) = slots[0] else {
1984        return Err(Raised::missing_named_argument("message name").into());
1985    };
1986    let name = required_string_named_argument(interp, name, "message name")?;
1987    let name = interp.to_text(name).to_ascii_uppercase();
1988    let scope = slots[1].filter(|scope| interp.is_class_object(*scope));
1989    if scope.is_none() {
1990        return Err(Raised::argument_not_a_class("SCOPE").into());
1991    }
1992    Ok((name, scope))
1993}

1995/// The message's argument count, which is the array's `lastItem`: the position
1996/// of the last filled slot, so a trailing empty slot is not an argument and an
1997/// interior one is an omitted argument.
1998fn message_argument_count(slots: &[Option<ObjRef>]) -> usize {
1999    slots
2000        .iter()
2001        .rposition(Option::is_some)
2002        .map_or(0, |at| at + 1)
2003}

2066/// [`array_argument_object`] for a `value` that is not a single-dimensional
2067/// array.
2068#[inline(never)]
2069fn converted_array_argument(
2070    interp: &mut Interp,
2071    value: ObjRef,
2072    overload: ArrayArgument,
2073) -> Result<ObjRef, Failure> {
2074    let array = match interp.array_body(value) {
2075        Some(_) => Some(value),
2076        None => interp.request_array_value(value)?,
2077    };
2078    if let Some(array) = array {
2079        interp.roots.activity_mut().push_temp(array);
2080        // An answer that is an `Array` subclass instance holds its slots in
2081        // the store behind it.
2082        let plain_array = interp.object_model().array;
2083        let is_array = interp
2084            .class_of_value(array)
2085            .is_some_and(|held| interp.classes().is_a(held, plain_array));
2086        let store = if is_array {
2087            super::array::collection_store(interp, array)
2088        } else {
2089            array
2090        };
2091        if let Some((_, dimensions)) = interp.array_body(store)
2092            && single_dimensional(dimensions)
2093        {
2094            return Ok(store);
2095        }
2096    }
2097    Err(match overload {
2098        ArrayArgument::Named(argument) => Raised::argument_not_single_dimensional(argument).into(),
2099        ArrayArgument::Positional => {
2100            let found = interp.string_value_text(value)?;
2101            Raised::object_not_single_dimensional(&found).into()
2102        }
2103    })
2104}

2106/// `Object~request(class)`: the receiver converted to `class`, or `.nil`.
2107pub(super) fn native_request(
2108    interp: &mut Interp,
2109    _cleared: Cleared,
2110    receiver: ObjRef,
2111    args: &[Option<ObjRef>],
2112) -> Result<Option<ObjRef>, Failure> {
2113    let Some(Some(argument)) = args.first().copied() else {
2114        return Err(Raised::missing_method_argument(1).into());
2115    };
2116    // The class name is read as an upcased required string, which is the
2117    // request's own first step -- so a class name with no string value is
2118    // 88.909 here and not `.nil`.
2119    let argument = required_string_argument(interp, argument, 1)?;
2120    let wanted = interp.to_text(argument).to_ascii_uppercase();
2121    let mut make = b"MAKE".to_vec();
2122    make.extend_from_slice(&wanted);
2123    if interp.lookup(receiver, &make, None).is_some() {
2124        // `resultOrNil`: a `MAKE` method that returns nothing answers `.nil`
2125        // rather than leaving the send with no value.
2126        let sent = interp.primitive_level(receiver, b"REQUEST", args, |interp| {
2127            let caller = interp.caller();
2128            interp.send_message(receiver, &make, None, &[], caller)
2129        })?;
2130        return Ok(Some(sent.unwrap_or(ObjRef::NIL)));
2131    }
2132    Ok(Some(match interp.receiver_class_id(receiver) {
2133        Some(id) if id.as_bytes().to_ascii_uppercase() == wanted => receiver,
2134        Some(_) | None => ObjRef::NIL,
2135    }))
2136}
