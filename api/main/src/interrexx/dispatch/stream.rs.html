30/// The state block, or 48.1 when `~uninit` has already dropped it, or 98.901
31/// where `init` never ran (r[class.stream.deviation.135]). The reference run
32/// segfaults for the unguarded half of its entry points here (oracle defect,
33/// `corpus/oracle-crashes.txt` entry 12b); answering 48.1 from every one of
34/// them is this crate's licensed divergence.
35fn require(interp: &Interp, receiver: ObjRef) -> Result<&StreamState, Failure> {
36    match interp.stream(receiver) {
37        Some(state) => Ok(state),
38        None => Err(no_state(interp, receiver)),
39    }
40}

42/// [`require`] for a caller that changes the state.
43fn require_mut(interp: &mut Interp, receiver: ObjRef) -> Result<&mut StreamState, Failure> {
44    if interp.stream(receiver).is_none() {
45        return Err(no_state(interp, receiver));
46    }
47    Ok(interp
48        .stream_mut(receiver)
49        .expect("the state was just found"))
50}

52/// The refusal of a receiver with no stream state: 98.901 when `init` never
53/// ran, which is when the name it keeps was never set, and otherwise 48.1,
54/// `uninit` having dropped the state.
55fn no_state(interp: &Interp, receiver: ObjRef) -> Failure {
56    let set_up = interp
57        .pools_of(receiver)
58        .is_some_and(|pools| pools.find(b"STREAM_NAME").is_some());
59    if set_up {
60        Raised::syntax(48, 1, vec![b"Stream not initialized".to_vec()]).into()
61    } else {
62        Raised::syntax(98, 901, vec![b"a Stream".to_vec()]).into()
63    }
64}

66/// A write argument's bytes. `LINEOUT` and `CHAROUT` require a *string
67/// value*, not a string rendering: an object answering `makeString` writes
68/// what that answers, and one answering none refuses. Measured,
69/// `.stdout~charout(.K~new)` is `Argument 1 must have a string value` at
70/// rc 168 where `~string` would have answered `a K`, and an instance whose
71/// class defines `makeString` writes what the method returns.
72/// The refusal's own built-in method traceback line comes from the dispatch
73/// layer, not from here -- measured, blaming it at this site as well puts the
74/// line in twice. It is a marshalling raise.
75fn write_text(interp: &mut Interp, value: ObjRef) -> Result<Vec<u8>, Failure> {
76    let Some(text) = interp.blamed_string_conversion(value)? else {
77        let mut raised = Raised::argument_needs_a_string_value(1);
78        marshalling(&mut raised);
79        return Err(raised.into());
80    };
81    Ok(interp.to_text(text).into_owned())
82}

96/// One of the three standard streams, built for `.local` rather than by a
97/// program. **Not `paths::normalize`d**: measured, `.stdout~qualify` answers
98/// the bare `STDOUT`, where every other stream answers a path, so the
99/// qualified name is the name. Ready from the start (`~state` `READY`,
100/// `~description` `READY:`) and marked with `standard`, which is what parts it
101/// from the ordinary `Stream` a program builds for the same name --
102/// `.Stream~new('STDOUT')` is a different object.
103///
104/// No `INIT` runs, so the `stream_name` object variable `::METHOD string`
105/// exposes is assigned here; an unset exposed variable would otherwise render
106/// as its own derived name.
107pub(crate) fn standard_stream(
108    interp: &mut Interp,
109    class: ObjRef,
110    which: StandardStream,
111) -> Result<ObjRef, Failure> {
112    let name: &[u8] = match which {
113        StandardStream::In => b"STDIN",
114        StandardStream::Out => b"STDOUT",
115        StandardStream::Err => b"STDERR",
116    };
117    let object = new_instance(interp, class)?;
118    // Takes no part in finalization: it lives in `.local` for the run and
119    // wraps a descriptor this crate did not open, and the sweep would
120    // otherwise drop its state before a program's own finalizer writes
121    // through it.
122    interp.heap.clear_uninit_all(&[object]);
123    let named = interp.text(name);
124    interp.set_pool_variable(object, class, b"STREAM_NAME", named);
125    let mut state = StreamState::new(name.to_vec(), name.into());
126    state.standard = Some(which);
127    state.status = StreamStatus::Ready;
128    // `.STDOUT` and `.STDERR` are opened `NOBUFFER` in the class library, where
129    // a program's own `.Stream~new('STDOUT')` is buffered.
130    state.mode.nobuffer = true;
131    state.local = true;
132    let Some(held) = interp.heap.get_mut(object) else {
133        unreachable!("new_instance just allocated and rooted it")
134    };
135    let Body::Instance { native, .. } = &mut held.body else {
136        unreachable!("new_instance allocates Body::Instance")
137    };
138    *native = Some(Box::new(NativeState::Stream(Box::new(state))));
139    Ok(object)
140}

142/// `interrexx_standard_stream(name)`: a new standard stream as
143/// [`standard_stream`] builds it for `.local`, where argument 1 is `STDIN`,
144/// `STDOUT` or `STDERR`. The class library's `LocalServer` stores what it
145/// answers in `.local` when it is created. A library without a `Stream`
146/// class answers `.nil`.
147pub(super) fn local_standard_stream(
148    interp: &mut Interp,
149    _cleared: Cleared,
150    _receiver: ObjRef,
151    args: &[Option<ObjRef>],
152) -> Result<Option<ObjRef>, Failure> {
153    let Some(value) = args.first().copied().flatten() else {
154        return Err(Raised::missing_method_argument(1).into());
155    };
156    let name = interp.to_text(value).into_owned();
157    let which = match name.as_slice() {
158        b"STDIN" => StandardStream::In,
159        b"STDOUT" => StandardStream::Out,
160        b"STDERR" => StandardStream::Err,
161        _ => {
162            return Err(
163                Raised::method_argument_not_in_list(1, "STDIN, STDOUT, STDERR", &name).into(),
164            );
165        }
166    };
167    let Some(class) = interp.rexx_package_class(b"STREAM") else {
168        return Ok(Some(ObjRef::NIL));
169    };
170    standard_stream(interp, class, which).map(Some)
171}

173/// `stream_init`: records the name as written and the path it resolves to.
174/// The resolution happens **here**, so a later `DIRECTORY()` cannot move a
175/// stream that has already been named.
176pub(super) fn init(
177    interp: &mut Interp,
178    _cleared: Cleared,
179    receiver: ObjRef,
180    args: &[Option<ObjRef>],
181) -> Result<Option<ObjRef>, Failure> {
182    let name = match args.first().copied().flatten() {
183        Some(value) => interp.to_text(value).into_owned(),
184        None => Vec::new(),
185    };
186    let qualified = crate::builtin::platform::qualify_name(interp, &name);
187    let state = StreamState::new(name, qualified.into_boxed_slice());
188    let Some(object) = interp.heap.get_mut(receiver) else {
189        unreachable!("a live receiver")
190    };
191    // The receiver may hold a state of another class's, a buffer's among
192    // them, where this method was run against it.
193    let before = object.body.held_bytes();
194    let Body::Instance { native, .. } = &mut object.body else {
195        unreachable!("Stream is allocated as Body::Instance")
196    };
197    *native = Some(Box::new(NativeState::Stream(Box::new(state))));
198    interp.heap.release_body_bytes(before);
199    Ok(None)
200}

202/// `std_set`: marks a stream named `STDIN`, `STDOUT` or `STDERR`, with or
203/// without a trailing colon. The Rexx half decides which name qualifies and
204/// only then sends this, so the marker is taken from the name it recorded.
205pub(super) fn std_set(
206    interp: &mut Interp,
207    _cleared: Cleared,
208    receiver: ObjRef,
209    _args: &[Option<ObjRef>],
210) -> Result<Option<ObjRef>, Failure> {
211    let state = require_mut(interp, receiver)?;
212    let mut upper = state.name.to_ascii_uppercase();
213    if upper.last() == Some(&b':') {
214        upper.pop();
215    }
216    state.standard = match upper.as_slice() {
217        b"STDIN" => Some(StandardStream::In),
218        b"STDOUT" => Some(StandardStream::Out),
219        b"STDERR" => Some(StandardStream::Err),
220        _ => None,
221    };
222    Ok(None)
223}

225/// `handle_set`: marks a `HANDLE:n` stream, `n` an `int`. `HANDLE:0`,
226/// `HANDLE:1` and `HANDLE:2` read and write through the standard stream of
227/// that number, and any other `n` through a duplicate of descriptor `n`
228/// that an open makes (`r[bif.stream.handle]`).
229pub(super) fn handle_set(
230    interp: &mut Interp,
231    _cleared: Cleared,
232    receiver: ObjRef,
233    args: &[Option<ObjRef>],
234) -> Result<Option<ObjRef>, Failure> {
235    let Some(Some(value)) = args.first().copied() else {
236        return Err(Raised::missing_native_argument(1).into());
237    };
238    let number = match value.decode() {
239        Decoded::SmallInt(number) => Some(number),
240        _ => {
241            let text = interp.to_text(value);
242            Number::parse_bytes(&text).and_then(|number| number.int64_value(DIGITS64))
243        }
244    };
245    let (min, max) = (i64::from(i32::MIN), i64::from(i32::MAX));
246    let Some(descriptor) = number.filter(|number| (min..=max).contains(number)) else {
247        // A marshalling raise, reported at the sending clause.
248        let found = interp.string_value_text(value)?;
249        let mut raised =
250            Raised::native_argument_outside_range(1, i128::from(min), i128::from(max), &found);
251        raised.delivery.lineless = true;
252        raised.delivery.internal_package = true;
253        return Err(raised.into());
254    };
255    let which = match descriptor {
256        0 => StandardStream::In,
257        1 => StandardStream::Out,
258        2 => StandardStream::Err,
259        _ => {
260            require_mut(interp, receiver)?.descriptor = true;
261            return Ok(None);
262        }
263    };
264    let state = require_mut(interp, receiver)?;
265    state.handle = Some(which);
266    state.standard = Some(which);
267    Ok(None)
268}

301/// `stream_description`: the state word, a colon, and what more the state
302/// knows -- the errno and its text for an error, `EOF` for a stream that read
303/// past the end, nothing for the other two.
304pub(super) fn description(
305    interp: &mut Interp,
306    _cleared: Cleared,
307    receiver: ObjRef,
308    _args: &[Option<ObjRef>],
309) -> Result<Option<ObjRef>, Failure> {
310    let state = require(interp, receiver)?;
311    let mut out = state.state_word().to_vec();
312    out.push(b':');
313    match state.status {
314        StreamStatus::NotReady => out.extend_from_slice(b"EOF"),
315        StreamStatus::Error(errno) => append_stream_errno(&mut out, errno),
316        StreamStatus::Unknown | StreamStatus::Ready => {}
317    }
318    Ok(Some(interp.text_built(out)))
319}
320
321fn append_stream_errno(out: &mut Vec<u8>, errno: i32) {
322    out.extend_from_slice(errno.to_string().as_bytes());
323    if errno == 0 {
324        return;
325    }
326    let text = std::io::Error::from_raw_os_error(errno).to_string();
327    // Strip the suffix Rust adds to the system's error text.
328    let text = text
329        .split_once(" (os error")
330        .map_or(text.as_str(), |(text, _)| text);
331    if !text.is_empty() {
332        out.push(b' ');
333        out.extend_from_slice(text.as_bytes());
334    }
335}

353/// Puts the stream in `status` and raises `NOTREADY` for it, answering
354/// `answer`: what the entry point returns when no trap transfers control,
355/// and the condition's `RESULT` (`StreamInfo::notreadyError`).
356#[cold]

358fn notready(
359    interp: &mut Interp,
360    receiver: ObjRef,
361    status: StreamStatus,
362    answer: Vec<u8>,
363) -> Result<Option<ObjRef>, Failure> {

365    state.status = status;
366    let name = state.name.clone();
367    let answer = interp.text_built(answer);

369    interp.raise_notready(&name, receiver, answer)?;
370    Ok(Some(answer))
371}

373/// One token of an options string: runs of non-whitespace (blank and tab,
374/// r[conv.string.whitespace]), except that `=`, `+`, `-` and `<` are
375/// single-character tokens and also end the run before them.
376struct Tokens<'a> {
377    source: &'a [u8],
378    offset: usize,
379    length: usize,
380}
381
382/// The bytes that end a token and, alone, form one.
383const SPECIAL: &[u8] = b"=+-<";
384
385impl<'a> Tokens<'a> {
386    fn new(source: &'a [u8]) -> Tokens<'a> {
387        Tokens {
388            source,
389            offset: 0,
390            length: 0,
391        }
392    }
393
394    fn next(&mut self) -> Option<&'a [u8]> {
395        self.offset += self.length;
396        while matches!(self.source.get(self.offset), Some(b' ' | b'\t')) {
397            self.offset += 1;
398        }
399        let Some(&first) = self.source.get(self.offset) else {
400            self.length = 0;
401            return None;
402        };
403        self.length = if SPECIAL.contains(&first) {
404            1
405        } else {
406            self.source[self.offset..]
407                .iter()
408                .take_while(|byte| !matches!(**byte, b' ' | b'\t') && !SPECIAL.contains(byte))
409                .count()
410        };
411        Some(&self.source[self.offset..self.offset + self.length])
412    }
413}

415/// `StreamToken::toNumber`: digits only -- no sign, no exponent -- and no
416/// overflow. An empty token converts to zero, which every caller rejects.
417fn to_number(token: &[u8]) -> Option<u64> {
418    let mut value: u64 = 0;
419    for byte in token {
420        let digit = byte.checked_sub(b'0').filter(|digit| *digit <= 9)?;
421        value = value.checked_mul(10)?.checked_add(u64::from(digit))?;
422    }
423    Some(value)
424}

426/// An option the open table admits.
427#[derive(Copy, Clone, PartialEq, Eq)]
428enum OpenOption {
429    Read,
430    Write,
431    Both,
432    Append,
433    Replace,
434    NoBuffer,
435    Binary,
436    RecLength,
437    Shared,
438}

440/// The table in its own order with each entry's minimum abbreviation: the
441/// first entry whose leading bytes match caselessly wins, so `re` reaches READ
442/// rather than REPLACE and is then too short for it. The three SHARE options
443/// share a row because every share flag is `0` on unix -- they are parsed for
444/// their conflicts and change no open.
445const OPEN_TABLE: &[(&[u8], usize, OpenOption)] = &[
446    (b"READ", 3, OpenOption::Read),
447    (b"WRITE", 1, OpenOption::Write),
448    (b"BOTH", 2, OpenOption::Both),
449    (b"APPEND", 2, OpenOption::Append),
450    (b"REPLACE", 3, OpenOption::Replace),
451    (b"NOBUFFER", 3, OpenOption::NoBuffer),
452    (b"BINARY", 2, OpenOption::Binary),
453    (b"RECLENGTH", 3, OpenOption::RecLength),
454    (b"SHARED", 6, OpenOption::Shared),
455    (b"SHAREREAD", 6, OpenOption::Shared),
456    (b"SHAREWRITE", 6, OpenOption::Shared),
457];

459/// What the options parsed to, in the terms of the native entry: the mode
460/// booleans and the two `oflag` bits the table's mutual-exclusion actions test.
461#[derive(Default)]
462struct Parsed {
463    read_only: bool,
464    write_only: bool,
465    read_write: bool,
466    append: bool,
467    truncate: bool,
468    nobuffer: bool,
469    record_based: bool,
470    shared_set: bool,
471    record_length: u64,
472}

474/// `parser()` over the open table. `Err` is the bare 93 every conflicting,
475/// repeated, too-short or unknown option answers.
476fn parse_open_options(options: &[u8]) -> Result<Parsed, ()> {
477    let mut parsed = Parsed::default();
478    let mut tokens = Tokens::new(options);
479    while let Some(token) = tokens.next() {
480        let found = OPEN_TABLE.iter().find(|(name, _, _)| {
481            name.len() >= token.len() && name[..token.len()].eq_ignore_ascii_case(token)
482        });
483        let Some((_, minimum, option)) = found else {
484            return Err(());
485        };
486        if token.len() < *minimum {
487            return Err(());
488        }
489        match option {
490            OpenOption::Read => {
491                if parsed.read_only || parsed.read_write || parsed.write_only || parsed.append {
492                    return Err(());
493                }
494                if parsed.truncate {
495                    return Err(());
496                }
497                parsed.read_only = true;
498            }
499            OpenOption::Write => {
500                if parsed.read_only || parsed.read_write || parsed.write_only {
501                    return Err(());
502                }
503                parsed.write_only = true;
504            }
505            OpenOption::Both => {
506                if parsed.read_only || parsed.read_write || parsed.write_only {
507                    return Err(());
508                }
509                parsed.read_write = true;
510            }
511            OpenOption::Append => {
512                if parsed.read_only || parsed.append || parsed.truncate {
513                    return Err(());
514                }
515                parsed.append = true;
516            }
517            OpenOption::Replace => {
518                if parsed.read_only || parsed.append || parsed.truncate {
519                    return Err(());
520                }
521                parsed.truncate = true;
522            }
523            OpenOption::NoBuffer => {
524                if parsed.nobuffer {
525                    return Err(());
526                }
527                parsed.nobuffer = true;
528            }
529            OpenOption::Binary => {
530                if parsed.record_based {
531                    return Err(());
532                }
533                parsed.record_based = true;
534            }
535            OpenOption::RecLength => {
536                // `MIB`: BINARY has to have come first.
537                if !parsed.record_based {
538                    return Err(());
539                }
540                // The length is required (r[bif.stream.open.binary]), and a
541                // second RECLENGTH is a repeat whether or not it has one.
542                if parsed.record_length != 0 {
543                    return Err(());
544                }
545                match tokens
546                    .next()
547                    .and_then(to_number)
548                    .filter(|length| *length != 0)
549                {
550                    Some(length) => parsed.record_length = length,
551                    None => return Err(()),
552                }
553            }
554            OpenOption::Shared => {
555                if parsed.shared_set {
556                    return Err(());
557                }
558                parsed.shared_set = true;
559            }
560        }
561    }
562    Ok(parsed)
563}

565/// `stream_open`: `READY:`, or `ERROR:n` for an open the file system refused.
566/// A conflicting, repeated, too-short or unknown option is a bare 93, as is a
567/// `BINARY` whose record length ends up zero.
568pub(super) fn open(
569    interp: &mut Interp,
570    _cleared: Cleared,
571    receiver: ObjRef,
572    args: &[Option<ObjRef>],
573) -> Result<Option<ObjRef>, Failure> {
574    let options = match args.first().copied().flatten() {
575        Some(value) => write_text(interp, value)?,
576        None => Vec::new(),
577    };
578    declared_only(args, 1)?;
579    let owned_alias = owned_stdout_alias(interp, receiver)?;
580    release_held(interp, receiver)?;
581    if owned_alias
582        && let Some(Err(error)) = require_mut(interp, receiver)?
583            .open
584            .as_mut()
585            .map(interrexx_core::OpenFile::flush)
586    {
587        return open_failed(interp, receiver, &error);
588    }
589    let state = require_mut(interp, receiver)?;
590    // A second open closes and reopens. Standard output retains its positions.
591    if !owned_alias {
592        state.open = None;
593    }
594    state.mode = interrexx_core::OpenMode::default();
595    state.status = StreamStatus::Unknown;
596    // A `HANDLE:n` stream takes the options a file does and reports its name
597    // as written from here on.
598    if state.handle.is_some() {
599        state.qualified = state.name.as_slice().into();
600        if !parse_open_options(&options).is_ok_and(|parsed| handle_options(&parsed)) {
601            return Err(Raised::syntax(93, 0, Vec::new()).into());
602        }
603        state.status = StreamStatus::Ready;
604        return Ok(Some(interp.text(b"READY:")));
605    }
606    // The three standard streams take only NOBUFFER, found anywhere in the
607    // options, and are ready without touching the file system. `.local`'s stay
608    // unbuffered: they write where `SAY` and trace output go.
609    if state.standard.is_some() {
610        state.mode.nobuffer = state.local
611            || options
612                .to_ascii_uppercase()
613                .windows(8)
614                .any(|word| word == b"NOBUFFER");
615        state.qualified = state.name.as_slice().into();
616        state.status = StreamStatus::Ready;
617        if owned_alias {
618            if let Some(error) = ensure_owned_stdout_file(interp, receiver)? {
619                return open_failed(interp, receiver, &error);
620            }
621            let state = require_mut(interp, receiver)?;
622            state
623                .open
624                .as_mut()
625                .unwrap()
626                .reopen_standard_output(!state.mode.nobuffer);
627        }
628        return Ok(Some(interp.text(b"READY:")));
629    }
630    let path = crate::paths::os_path(&state.qualified).to_path_buf();
631
632    let Ok(mut parsed) = parse_open_options(&options) else {
633        return Err(Raised::syntax(93, 0, Vec::new()).into());
634    };
635    // A truncating BINARY open has nothing to take a record length from.
636    if parsed.record_based && parsed.truncate && parsed.record_length == 0 {
637        return Err(Raised::syntax(93, 0, Vec::new()).into());
638    }
639    // No explicit WRITE or BOTH, and not READ: the default is BOTH.
640    if !parsed.write_only && !parsed.read_write && !parsed.read_only {
641        parsed.read_write = true;
642    }
643    // WRITE opens the stream for writing only (rexxref funct.xml:4302). The
644    // descriptor is still opened for reading where it can be, so a line
645    // number given to LINEOUT can be found; the reads refuse by the mode.
646
647    let descriptor = stream_descriptor(interp, receiver)?;
648    if descriptor.is_some() {
649        if !handle_options(&parsed) {
650            return Err(Raised::syntax(93, 0, Vec::new()).into());
651        }
652        // Named by its name as written once it is open, which `qualify`
653        // answers.
654        let state = require_mut(interp, receiver)?;
655        state.qualified = state.name.as_slice().into();
656    }
657    if descriptor.is_none()
658        && parsed.read_only
659        && !std::fs::metadata(&path).is_ok_and(|meta| meta.is_file())
660    {
661        let answer = format!("ERROR:{ENOENT}").into_bytes();
662        return notready(interp, receiver, StreamStatus::Error(ENOENT), answer);
663    }
664
665    // APPEND is `O_APPEND` (`SysFile::open`), so a held write lands at the
666    // end the file has when it is written, after what a command or another
667    // stream appended meanwhile.
668    let mut opening = std::fs::OpenOptions::new();
669    if parsed.read_only {
670        opening.read(true);
671    } else {
672        opening
673            .read(true)
674            .write(true)
675            .create(true)
676            .append(parsed.append);
677        if parsed.truncate {
678            opening.truncate(true);
679        }
680    }
681    let opened = match descriptor.map(descriptor_file) {
682        Some(Ok(file)) => file,
683        Some(Err(error)) => return open_failed(interp, receiver, &error),
684        None => match opening.open(&path) {
685            Ok(file) => file,
686            // An explicit WRITE retries write-only, so a file that grants write and
687            // not read opens. The retry is a create-for-write open, which does not
688            // truncate even under REPLACE.
689            Err(_) if parsed.write_only => {
690                match std::fs::OpenOptions::new()
691                    .write(true)
692                    .create(true)
693                    .truncate(false)
694                    .append(parsed.append)
695                    .open(&path)
696                {
697                    Ok(file) => file,
698                    Err(error) => return open_failed(interp, receiver, &error),
699                }
700            }
701            Err(error) => return open_failed(interp, receiver, &error),
702        },
703    };
704    // A directory opens on unix and is then rejected as if it had not, which
705    // is why a directory answers `ERROR:2` and not `ERROR:21`.
706    let metadata = opened.metadata().ok();
707    if metadata.as_ref().is_none_or(std::fs::Metadata::is_dir) {
708        let answer = format!("ERROR:{ENOENT}").into_bytes();
709        return notready(interp, receiver, StreamStatus::Error(ENOENT), answer);
710    }
711    let size = metadata.map_or(0, |meta| meta.len());
712    let transient = is_transient(&opened);
713    // The stream-type check: a BINARY open with no RECLENGTH takes the file's
714    // own size as the record, and a size of zero is the bare 93. A transient
715    // stream's record is one byte.
716    if parsed.record_based && parsed.record_length == 0 {
717        if transient {
718            parsed.record_length = 1;
719        } else if size == 0 {
720            return Err(Raised::syntax(93, 0, Vec::new()).into());
721        } else {
722            parsed.record_length = size;
723        }
724    }
725    // A persistent writeable stream writes at the end; a read-only one has no
726    // write position at all, which is the `0` `QUERY POSITION WRITE` answers.
727    // A `HANDLE:n` stream's read-only write position is character 1.
728    let write_position: i64 = match (parsed.read_only, descriptor) {
729        (true, None) => 0,
730        (true, Some(_)) => 1,
731        (false, _) => as_position(size) + 1,
732    };
733
734    let state = require_mut(interp, receiver)?;
735    state.mode = interrexx_core::OpenMode {
736        read_only: parsed.read_only,
737        write_only: parsed.write_only,
738        read_write: parsed.read_write,
739        append: parsed.append,
740        nobuffer: parsed.nobuffer,
741        record_based: parsed.record_based,
742        record_length: parsed.record_length,
743    };
744    let buffered = !parsed.nobuffer && !std::io::IsTerminal::is_terminal(&opened);
745    state.open = Some(interrexx_core::OpenFile::new(
746        opened,
747        write_position,
748        transient,
749        buffered,
750    ));
751    state.append_extent = match parsed.append && !transient {
752        true => size,
753        false => 0,
754    };
755    state.status = StreamStatus::Ready;
756    Ok(Some(interp.text(b"READY:")))
757}

792/// Opens the stream if nothing has, the way an implicit open does: read-write
793/// first, then write-only or read-only depending on what the caller wants
794/// (`StreamInfo::implicitOpen`). A read never creates the file. `answer` is
795/// what the caller returns when the open fails.
796fn ensure_open(
797    interp: &mut Interp,
798    receiver: ObjRef,
799    for_write: bool,
800    answer: &[u8],
801) -> Result<bool, Failure> {
802    if require(interp, receiver)?.open.is_some() {
803        return Ok(true);
804    }
805    if let Some(descriptor) = stream_descriptor(interp, receiver)? {
806        let state = require_mut(interp, receiver)?;
807        state.qualified = state.name.as_slice().into();
808        let (opened, mode) = match descriptor_file(descriptor) {
809            Ok(file) => {
810                let mode = descriptor_mode(&file);
811                (file, mode)
812            }
813            Err(error) => return fail_open(interp, receiver, &error, answer),
814        };
815        return settle_implicit_open(interp, receiver, opened, mode);
816    }
817    let path = qualified_path(interp, receiver)?;
818    let mut opening = std::fs::OpenOptions::new();
819    // **Never truncates.** An implicit open creates a missing file but keeps
820    // what an existing one holds -- it opens read-write with create and no
821    // truncate -- which is what lets a `lineout` to an existing file append at
822    // the end instead of emptying it.
823    opening.read(true).write(true).truncate(false);
824    if for_write {
825        opening.create(true);
826    }
827    let (opened, mode) = match opening.open(&path) {
828        Ok(file) => (
829            file,
830            interrexx_core::OpenMode {
831                read_write: true,
832                ..interrexx_core::OpenMode::default()
833            },
834        ),
835        // `O_WRONLY` alone, as the implicit open's own fallback is: it creates
836        // nothing, because the read-write attempt above already carried
837        // `O_CREAT` for a write, so reaching here means the open failed for
838        // some reason other than the file being absent.
839        Err(_) if for_write => match std::fs::OpenOptions::new().write(true).open(&path) {
840            Ok(file) => (
841                file,
842                interrexx_core::OpenMode {
843                    write_only: true,
844                    ..interrexx_core::OpenMode::default()
845                },
846            ),
847            Err(error) => return fail_open(interp, receiver, &error, answer),
848        },
849        Err(_) => match std::fs::OpenOptions::new().read(true).open(&path) {
850            Ok(file) => (
851                file,
852                interrexx_core::OpenMode {
853                    read_only: true,
854                    ..interrexx_core::OpenMode::default()
855                },
856            ),
857            Err(error) => return fail_open(interp, receiver, &error, answer),
858        },
859    };
860    // `SysFile::open` closes a directory it opened and fails with ENOENT.
861    if opened.metadata().is_ok_and(|meta| meta.is_dir()) {
862        let error = std::io::Error::from_raw_os_error(ENOENT);
863        return fail_open(interp, receiver, &error, answer);
864    }
865    settle_implicit_open(interp, receiver, opened, mode)
866}

868/// The end of an implicit open, once `opened` is the stream's file in `mode`.
869fn settle_implicit_open(
870    interp: &mut Interp,
871    receiver: ObjRef,
872    opened: std::fs::File,
873    mode: interrexx_core::OpenMode,
874) -> Result<bool, Failure> {
875    let size = size_of(&opened);
876    let transient = is_transient(&opened);
877    // A stream that can write writes at the end, whichever operation opened
878    // it. A read-only or transient one keeps the implicit open's character 1.
879    let write_position = if mode.read_only || transient {
880        1
881    } else {
882        as_position(size) + 1
883    };
884    let state = require_mut(interp, receiver)?;
885    state.mode = mode;
886    state.status = StreamStatus::Ready;
887    let buffered = !std::io::IsTerminal::is_terminal(&opened);
888    state.open = Some(interrexx_core::OpenFile::new(
889        opened,
890        write_position,
891        transient,
892        buffered,
893    ));
894    state.append_extent = 0;
895    Ok(true)
896}

905/// The descriptor a `HANDLE:n` stream for some `n` other than 0, 1 and 2
906/// reads and writes, from its name, and `None` for any other stream. A number
907/// no descriptor can have is answered as -1, which an open refuses as
908/// `EBADF`.
909fn stream_descriptor(interp: &Interp, receiver: ObjRef) -> Result<Option<i32>, Failure> {
910    let state = require(interp, receiver)?;
911    if !state.descriptor {
912        return Ok(None);
913    }
914    let digits = state.name.get(b"HANDLE:".len()..).unwrap_or_default();
915    let number = Number::parse_bytes(digits).and_then(|number| number.int64_value(DIGITS64));
916    Ok(Some(
917        number.and_then(|n| i32::try_from(n).ok()).unwrap_or(-1),
918    ))
919}

930/// The mode an implicit open of a `HANDLE:n` stream takes: the access mode the
931/// descriptor itself was opened with.
932#[cfg(unix)]
933fn descriptor_mode(file: &std::fs::File) -> interrexx_core::OpenMode {
934    use rustix::fs::OFlags;
935    // `O_RDONLY` is no bit at all, on Linux and macOS alike: the access mode is
936    // whichever of `O_WRONLY` and `O_RDWR` is set, or neither.
937    let access = rustix::fs::fcntl_getfl(file).map_or(OFlags::RDWR, |flags| {
938        flags & (OFlags::WRONLY | OFlags::RDWR)
939    });
940    interrexx_core::OpenMode {
941        read_only: access.is_empty(),
942        write_only: access == OFlags::WRONLY,
943        read_write: access == OFlags::RDWR,
944        ..interrexx_core::OpenMode::default()
945    }
946}

1034/// Installs the owned stdout duplicate if absent, returning an OS error to the
1035/// entry point that determines the NOTREADY result.
1036fn ensure_owned_stdout_file(
1037    interp: &mut Interp,
1038    receiver: ObjRef,
1039) -> Result<Option<std::io::Error>, Failure> {
1040    let state = require(interp, receiver)?;
1041    if state.open.is_some() {
1042        return Ok(None);
1043    }
1044    let buffered = !state.mode.nobuffer;
1045    #[cfg(unix)]
1046    let file = match rustix::io::dup(std::io::stdout()) {
1047        Ok(descriptor) => std::fs::File::from(descriptor),
1048        Err(error) => return Ok(Some(error.into())),
1049    };
1050    #[cfg(windows)]
1051    let file = match crate::host::standard_file(1) {
1052        Ok(file) => file,
1053        Err(error) => return Ok(Some(error)),
1054    };
1055    require_mut(interp, receiver)?.open =
1056        Some(interrexx_core::OpenFile::standard_output(file, buffered));
1057    Ok(None)
1058}

1060/// A write to a standard stream: to the sink for the two output ones, and a
1061/// refusal for `.STDIN`. Answers what the entry point answers -- `0` written,
1062/// or the residual count. Measured: `.stdin~lineout('x')` raises `NOTREADY`
1063/// and answers `1`, and the raise happens whether or not a trap is armed.
1064fn standard_write(
1065    interp: &mut Interp,
1066    receiver: ObjRef,
1067    which: StandardStream,
1068    bytes: &[u8],
1069    residual: &[u8],
1070) -> Result<Option<ObjRef>, Failure> {
1071    if which == StandardStream::In {
1072        return notready(
1073            interp,
1074            receiver,
1075            StreamStatus::Error(EACCES),
1076            residual.to_vec(),
1077        );
1078    }
1079    if owned_stdout_alias(interp, receiver)? {
1080        // Preserve writes issued to the local sink before this alias touches
1081        // the shared descriptor, including its first buffered-write seek.
1082        interp.hand_over_stdout();
1083        let error = match ensure_owned_stdout_file(interp, receiver)? {
1084            Some(error) => Some(error),
1085            None => require_mut(interp, receiver)?
1086                .open
1087                .as_mut()
1088                .unwrap()
1089                .write_standard_output(bytes)
1090                .err(),
1091        };
1092        if let Some(error) = error {
1093            return notready(
1094                interp,
1095                receiver,
1096                StreamStatus::Error(error.raw_os_error().unwrap_or(0)),
1097                residual.to_vec(),
1098            );
1099        }
1100        require_mut(interp, receiver)?.status = StreamStatus::Ready;
1101        return Ok(Some(interp.text(b"0")));
1102    }
1103    let state = require_mut(interp, receiver)?;
1104    state.status = StreamStatus::Ready;
1105    if state.handle.is_some() || state.mode.nobuffer {
1106        // A descriptor the sinks could not write is the stream's error from
1107        // then on, and so is one the hand-over this write makes finds.
1108        let sink = usize::from(which == StandardStream::Err);
1109        if interp.sink_errors[sink].is_none() {
1110            emit(interp, which, bytes);
1111        }
1112        if let Some(errno) = interp.sink_errors[sink] {
1113            return notready(
1114                interp,
1115                receiver,
1116                StreamStatus::Error(errno),
1117                residual.to_vec(),
1118            );
1119        }
1120        return Ok(Some(interp.text(b"0")));
1121    }
1122    // Held as `SysFile::write` holds a file's writes.
1123    if bytes.len() > STREAM_BUFFER {
1124        let held = std::mem::take(&mut state.held);
1125        emit(interp, which, &held);
1126        emit(interp, which, bytes);
1127        return Ok(Some(interp.text(b"0")));
1128    }
1129    let mut rest = bytes;
1130    while !rest.is_empty() {
1131        let state = require_mut(interp, receiver)?;
1132        if state.held.len() == STREAM_BUFFER {
1133            let held = std::mem::take(&mut state.held);
1134            emit(interp, which, &held);
1135            continue;
1136        }
1137        let take = rest.len().min(STREAM_BUFFER - state.held.len());
1138        state.held.extend_from_slice(&rest[..take]);
1139        rest = &rest[take..];
1140    }
1141    Ok(Some(interp.text(b"0")))
1142}

1144/// Writes `bytes` to the descriptor standard stream `which` stands for.
1145fn emit(interp: &mut Interp, which: StandardStream, bytes: &[u8]) {
1146    match which {
1147        StandardStream::Out => interp.write_out(bytes),
1148        StandardStream::Err => interp.write_err(bytes),
1149        StandardStream::In => {}
1150    }

1153/// Hands what a buffered standard stream holds to its descriptor, as a
1154/// flush, a close, a reopen and an `UNINIT` do.
1155fn release_held(interp: &mut Interp, receiver: ObjRef) -> Result<(), Failure> {
1156    if owned_stdout_alias(interp, receiver)? {
1157        // The OpenFile flush or drop follows pending local output in order.
1158        interp.hand_over_stdout();
1159    }
1160    let state = require_mut(interp, receiver)?;
1161    let Some(which) = state.standard else {
1162        return Ok(());
1163    };
1164    if state.held.is_empty() {
1165        return Ok(());
1166    }
1167    let held = std::mem::take(&mut state.held);
1168    emit(interp, which, &held);
1169    Ok(())
1170}

1185/// Positioning a transient stream is 93.958, and a standard stream reaches that

1191    interp: &Interp,
1192    which: StandardStream,
1193    positioned: bool,
1194) -> Result<(), Failure> {
1195    if positioned && standard_is_transient(interp, which) {
1196        return Err(Raised::transient_positioning().into());
1197    }

1201/// `stream_charin`: `length` characters from `start`, or from the read
1202/// position. A short read raises `NOTREADY` and answers what it got, and
1203/// leaves `NOTREADY:EOF` whether it got some or nothing. A `start` past the
1204/// end is `NOTREADY:EOF` before anything is read.
1205pub(super) fn charin(
1206    interp: &mut Interp,
1207    _cleared: Cleared,
1208    receiver: ObjRef,
1209    args: &[Option<ObjRef>],
1210) -> Result<Option<ObjRef>, Failure> {
1211    let start = optional_position(interp, args.first().copied().flatten(), 1)?;
1212    let length = match args.get(1).copied().flatten() {
1213        Some(value) => whole_number(interp, value, 2)?,
1214        None => 1,
1215    };
1216    declared_only(args, 2)?;
1217    if let Some(which) = standard_opened(interp, receiver)? {
1218        refuse_transient_position(interp, which, start.is_some())?;
1219        let start = start.map(positive).transpose()?;
1220        let StandardStream::In = which else {
1221            // Nothing is read from an output stream: a zero-length ask
1222            // answers the null string and any other raises NOTREADY.
1223            if length == 0 {
1224                return Ok(Some(interp.text(b"")));
1225            }
1226            return unreadable(interp, receiver);
1227        };
1228        if let Some(start) = start {
1229            match interp.input_seek_char(start) {
1230                None => return Err(Raised::transient_positioning().into()),
1231                Some(false) => {
1232                    return notready(interp, receiver, StreamStatus::NotReady, Vec::new());
1233                }
1234                Some(true) => {}
1235            }
1236        }
1237        let wanted = usize::try_from(length).unwrap_or(usize::MAX);
1238        // A zero-length ask reads nothing and leaves the state alone, as it
1239        // does on the file path: measured, `.stdin~charin( , 0)` answers the
1240        // null string and `~description` stays `READY:`, where treating it as
1241        // a read of nothing would leave `NOTREADY:EOF`.
1242        if wanted == 0 {
1243            return Ok(Some(interp.text(b"")));
1244        }
1245        let read = interp.input_bytes(wanted)?;
1246        // Asking for 50 of six bytes answers all six and leaves
1247        // `NOTREADY:EOF`, asking for exactly six stays `READY:`, and a further
1248        // ask reads past the end, which is `NOTREADY:EOF` too
1249        // (rexxref streamclasses.xml:1638-1652).
1250        let status = match read.len() {
1251            0 => StreamStatus::NotReady,
1252            got if got < wanted => StreamStatus::NotReady,
1253            _ => StreamStatus::Ready,
1254        };
1255        if status != StreamStatus::Ready {
1256            return notready(interp, receiver, status, read);
1257        }
1258        require_mut(interp, receiver)?.status = status;
1259        return Ok(Some(interp.text_built(read)));
1260    }
1261    if !read_setup(interp, receiver, b"")? {
1262        return Ok(Some(interp.text(b"")));
1263    }
1264    let state = require_mut(interp, receiver)?;
1265    let write_only = state.mode.write_only;
1266    let Some(open) = state.open.as_mut() else {
1267        return Ok(Some(interp.text(b"")));
1268    };
1269    if let Some(start) = start {
1270        if open.transient {
1271            return Err(Raised::transient_positioning().into());
1272        }
1273        let start = as_position(positive(start)?);
1274        if start > as_position(size_of(open.flushed())) {
1275            return notready(interp, receiver, StreamStatus::NotReady, Vec::new());
1276        }
1277        open.read_position = start;
1278    }
1279    let at = open.read_position;
1280    let wanted = usize::try_from(length).unwrap_or(usize::MAX);
1281    if wanted == 0 {
1282        return Ok(Some(interp.text(b"")));
1283    }
1284    if write_only {
1285        return unreadable(interp, receiver);
1286    }
1287    // A read that fails reports its errno: measured, `EBADF` on a stream
1288    // opened write-only is `ERROR:9`.
1289    let read = match read_at(open, offset_of(at), wanted) {
1290        Ok(read) => read,
1291        Err(error) => {
1292            let errno = error.raw_os_error().unwrap_or(0);
1293            return notready(interp, receiver, StreamStatus::Error(errno), Vec::new());
1294        }
1295    };
1296    if read.is_empty() {
1297        // A read past the end is `NOTREADY:EOF`, as LINEIN's is
1298        // (rexxref streamclasses.xml:1638-1652).
1299        return notready(interp, receiver, StreamStatus::NotReady, Vec::new());
1300    }
1301    open.read_position = at + memory_count(read.len());
1302    if read.len() < wanted {
1303        return notready(interp, receiver, StreamStatus::NotReady, read);
1304    }
1305    Ok(Some(interp.text_built(read)))
1306}

1320/// `stream_linein`: one line from `line`, or from the read position. At the
1321/// end of the file the answer is the null string and the state is
1322/// `NOTREADY:EOF`.
1323pub(super) fn linein(
1324    interp: &mut Interp,
1325    _cleared: Cleared,
1326    receiver: ObjRef,
1327    args: &[Option<ObjRef>],
1328) -> Result<Option<ObjRef>, Failure> {
1329    let line = optional_position(interp, args.first().copied().flatten(), 1)?;
1330    let count = match args.get(1).copied().flatten() {
1331        Some(value) => whole_number(interp, value, 2)?,
1332        None => 1,
1333    };
1334    declared_only(args, 2)?;
1335    if !matches!(count, 0 | 1) {
1336        return Err(Raised::syntax(93, 0, Vec::new()).into());
1337    }
1338    if let Some(which) = standard_opened(interp, receiver)? {
1339        refuse_transient_position(interp, which, line.is_some())?;
1340        let line = line.map(positive).transpose()?;
1341        let StandardStream::In = which else {
1342            if count == 0 {
1343                return Ok(Some(interp.text(b"")));
1344            }
1345            return unreadable(interp, receiver);
1346        };
1347        if let Some(line) = line
1348            && !interp.input_seek_line(line)
1349        {
1350            return Err(Raised::transient_positioning().into());
1351        }
1352    } else {
1353        if !read_setup(interp, receiver, b"")? {
1354            return Ok(Some(interp.text(b"")));
1355        }
1356        if let Some(line) = line
1357            && !write_only(interp, receiver)?
1358        {
1359            seek_to_line(interp, receiver, line)?;
1360        }
1361    }
1362    if count == 0 {
1363        return Ok(Some(interp.text(b"")));
1364    }
1365    if write_only(interp, receiver)? {
1366        return unreadable(interp, receiver);
1367    }
1368    if let Some(record_length) = record_length(interp, receiver)? {
1369        // The rest of the current record: a short one, or nothing at all, is
1370        // `NOTREADY:EOF` answering what it got.
1371        let read = read_record(interp, receiver, record_length)?;
1372        let wanted = record_remainder(read.from, record_length);
1373        if read.bytes.is_empty() {
1374            return notready(interp, receiver, past_the_end(read.errno), Vec::new());
1375        }
1376        if read.bytes.len() < wanted {
1377            return notready(interp, receiver, StreamStatus::NotReady, read.bytes);
1378        }
1379        return Ok(Some(interp.text_built(read.bytes)));
1380    }
1381    Ok(Some(match next_line(interp, receiver)? {
1382        Some(text) => interp.text_built(text),
1383        None => interp.text(b""),
1384    }))
1385}

1410/// What is left of the record holding `position`: a whole record unless a
1411/// character operation stopped inside one.
1412fn record_remainder(position: i64, record_length: u64) -> usize {
1413    let length = as_position(record_length);
1414    let left = length - (position - 1).rem_euclid(length);
1415    usize::try_from(left).unwrap_or(usize::MAX)
1416}

1427/// Reads the rest of the current record and moves the read position past
1428/// what it got. A failed read reads nothing and carries its errno.
1429fn read_record(
1430    interp: &mut Interp,
1431    receiver: ObjRef,
1432    record_length: u64,
1433) -> Result<RecordRead, Failure> {
1434    let state = require_mut(interp, receiver)?;
1435    let Some(open) = state.open.as_mut() else {
1436        return Ok(RecordRead {
1437            from: 1,
1438            bytes: Vec::new(),
1439            errno: 0,
1440        });
1441    };
1442    let from = open.read_position;
1443    let wanted = record_remainder(from, record_length);
1444    let (bytes, errno) = match read_at(open, offset_of(from), wanted) {
1445        Ok(bytes) => (bytes, 0),
1446        Err(error) => (Vec::new(), error.raw_os_error().unwrap_or(0)),
1447    };
1448    open.read_position = from + memory_count(bytes.len());
1449    Ok(RecordRead { from, bytes, errno })
1450}

1452/// One line from the read position, or `None` when there is none: the reading
1453/// half of `stream_linein`, shared with `stream_arrayin` so the two cannot
1454/// grow separate answers at the end of input.
1455///
1456/// **The end of input raises here.** Whether the caller sees that as an error
1457/// is [`Interp::raise_notready`]'s decision, and a `SIGNAL ON NOTREADY` makes
1458/// it one -- which is how the orx `arrayIn` ends its fill and answers the
1459/// array it passed in.
1460fn next_line(interp: &mut Interp, receiver: ObjRef) -> Result<Option<Vec<u8>>, Failure> {
1461    if let Some(which) = standard_opened(interp, receiver)? {
1462        let StandardStream::In = which else {
1463            return Ok(None);
1464        };
1465        return match interp.input_line()? {
1466            Some(text) => {
1467                // A read that succeeds clears whatever a refused write left:
1468                // measured, `.stdin~lineout` leaves `ERROR:13` and the next
1469                // `.stdin~linein` reads its line and answers `READY:` again.
1470                require_mut(interp, receiver)?.status = StreamStatus::Ready;
1471                Ok(Some(text))
1472            }
1473            None => {
1474                notready(interp, receiver, StreamStatus::NotReady, Vec::new())?;
1475                Ok(None)
1476            }
1477        };
1478    }
1479    if !ensure_open(interp, receiver, false, b"")? {
1480        return Ok(None);
1481    }
1482    let state = require_mut(interp, receiver)?;
1483    let Some(open) = state.open.as_mut() else {
1484        return Ok(None);
1485    };
1486    let at = open.read_position;
1487    match read_line_from(open, offset_of(at)).unwrap_or(None) {
1488        Some((text, next)) => {
1489            open.read_position = next;
1490            Ok(Some(text))
1491        }
1492        None => {
1493            notready(interp, receiver, StreamStatus::NotReady, Vec::new())?;
1494            Ok(None)
1495        }
1496    }
1497}

1499/// `stream_arrayin`: fills the caller's array with the lines that are left.
1500///
1501/// **The array is the orx `arrayIn`'s own**, built as `.array~new(self~lines)`
1502/// and filled in place, and this answers nothing: the end of input raises, and
1503/// that method's `SIGNAL ON NOTREADY` is what returns the array. Appending is
1504/// right despite the array arriving pre-sized -- measured on both engines,
1505/// `.array~new(3)` holds no items and the first append lands on slot 1.
1506pub(super) fn arrayin(
1507    interp: &mut Interp,
1508    _cleared: Cleared,
1509    receiver: ObjRef,
1510    args: &[Option<ObjRef>],
1511) -> Result<Option<ObjRef>, Failure> {
1512    let Some(array) = args.first().copied().flatten() else {
1513        return Err(Raised::missing_method_argument(1).into());
1514    };
1515    if standard_opened(interp, receiver)?.is_none() && !read_setup(interp, receiver, b"")? {
1516        return Ok(None);
1517    }
1518    if write_only(interp, receiver)? {
1519        unreadable(interp, receiver)?;
1520        return Ok(None);
1521    }
1522    // A record stream appends a record at a time, a short last one included,
1523    // and ends on the read that gets nothing.
1524    if let Some(record_length) = record_length(interp, receiver)? {
1525        loop {
1526            let read = read_record(interp, receiver, record_length)?;
1527            if read.bytes.is_empty() {
1528                notready(interp, receiver, past_the_end(read.errno), Vec::new())?;
1529                return Ok(None);
1530            }
1531            let item = interp.text_built(read.bytes);
1532            interp.roots.activity_mut().push_temp(item);
1533            super::collection::append_slot(interp, array, item)?;
1534        }
1535    }
1536    while let Some(text) = next_line(interp, receiver)? {
1537        let item = interp.text_built(text);
1538        // Rooted before the append, which allocates when the array grows.
1539        interp.roots.activity_mut().push_temp(item);
1540        super::collection::append_slot(interp, array, item)?;
1541    }
1542    Ok(None)
1543}

1545/// Moves the read position to the start of `line`.
1546fn seek_to_line(interp: &mut Interp, receiver: ObjRef, line: i64) -> Result<(), Failure> {
1547    let state = require_mut(interp, receiver)?;
1548    let record_length = if state.mode.record_based {
1549        state.mode.record_length
1550    } else {
1551        0
1552    };
1553    let Some(open) = state.open.as_mut() else {
1554        return Ok(());
1555    };
1556    // The same rule `SEEK` follows, reached by naming a line instead:
1557    // measured, `.Stream~new('/dev/null')~supplier` raises this, because
1558    // `StreamSupplier~init` reads line 1 of a stream it has decided is not
1559    // transient before opening it.
1560    if open.transient {
1561        return Err(Raised::transient_positioning().into());
1562    }
1563    let target = as_position(positive(line)?);
1564    let (_, at) = seek_line(open, target, record_length);
1565    open.read_position = at;
1566    Ok(())

1569/// `stream_charout`: writes `data` at `start` or the write position and
1570/// answers the count it could not write, which is `0` for every write that
1571/// completes. With neither argument it closes the stream.
1572pub(super) fn charout(
1573    interp: &mut Interp,
1574    _cleared: Cleared,
1575    receiver: ObjRef,
1576    args: &[Option<ObjRef>],
1577) -> Result<Option<ObjRef>, Failure> {
1578    let data = match args.first().copied().flatten() {
1579        Some(value) => Some(write_text(interp, value)?),
1580        None => None,
1581    };
1582    let start = optional_position(interp, args.get(1).copied().flatten(), 2)?;
1583    declared_only(args, 2)?;
1584    if data.is_none() && start.is_none() {
1585        return close_for_write(interp, _cleared, receiver, b"0");
1586    }
1587    if let Some(which) = standard_opened(interp, receiver)? {
1588        // The standard input refuses every write, so a start is never looked
1589        // at: it fails with NOTREADY, not with a positioning error.
1590        if which != StandardStream::In {
1591            refuse_transient_position(interp, which, start.is_some())?;
1592            start.map(positive).transpose()?;
1593        }
1594        let data = data.unwrap_or_default();
1595        let residual = data.len().to_string().into_bytes();
1596        return standard_write(interp, receiver, which, &data, &residual);
1597    }
1598    let residual = data.as_ref().map_or(0, Vec::len).to_string().into_bytes();
1599    if !write_setup(interp, receiver, &residual)? {
1600        return Ok(Some(interp.text_built(residual)));
1601    }
1602    let state = require_mut(interp, receiver)?;
1603    let append = state.mode.append;
1604    let extent = state.append_extent;
1605    let Some(open) = state.open.as_mut() else {
1606        return Ok(Some(interp.text(b"0")));
1607    };
1608    if let Some(start) = start {
1609        if open.transient {
1610            return Err(Raised::transient_positioning().into());
1611        }
1612        let start = as_position(positive(start)?);
1613        if within_append_extent(extent, start) {
1614            return notready(interp, receiver, StreamStatus::NotReady, residual);
1615        }
1616        open.write_position = start;
1617    }
1618    // A null string writes nothing, so it moves nothing either.
1619    let Some(data) = data.filter(|data| !data.is_empty()) else {
1620        return Ok(Some(interp.text(b"0")));
1621    };
1622    let at = landing(open, append, open.write_position);
1623    match write_open(open, at, &data) {
1624        Ok(()) => {
1625            open.write_position = at + memory_count(data.len());
1626            Ok(Some(interp.text(b"0")))
1627        }
1628        Err(error) => {
1629            let errno = error.raw_os_error().unwrap_or(0);
1630            notready(interp, receiver, StreamStatus::Error(errno), residual)
1631        }
1632    }
1633}

1635/// `stream_lineout`: writes `data` and a newline, answering `0` on success and
1636/// `1` on failure. With neither argument it closes the stream.
1637pub(super) fn lineout(
1638    interp: &mut Interp,
1639    _cleared: Cleared,
1640    receiver: ObjRef,
1641    args: &[Option<ObjRef>],
1642) -> Result<Option<ObjRef>, Failure> {
1643    let data = match args.first().copied().flatten() {
1644        Some(value) => Some(write_text(interp, value)?),
1645        None => None,
1646    };
1647    let line = optional_position(interp, args.get(1).copied().flatten(), 2)?;
1648    declared_only(args, 2)?;
1649    if data.is_none() && line.is_none() {
1650        // A record the close could not complete is reported, and the stream
1651        // is closed all the same, so a later LINEOUT does not meet it again.
1652        if let Some(errno) = complete_record(interp, receiver)? {
1653            close(interp, _cleared, receiver, &[])?;
1654            return notready(interp, receiver, StreamStatus::Error(errno), b"1".to_vec());
1655        }
1656        return close_for_write(interp, _cleared, receiver, b"1");
1657    }
1658    if let Some(which) = standard_opened(interp, receiver)? {
1659        refuse_transient_position(interp, which, line.is_some())?;
1660        line.map(positive).transpose()?;
1661        let mut bytes = data.unwrap_or_default();
1662        bytes.push(b'\n');
1663        return standard_write(interp, receiver, which, &bytes, b"1");
1664    }
1665    if !write_setup(interp, receiver, b"1")? {
1666        return Ok(Some(interp.text(b"1")));
1667    }
1668    // Line positioning on a text stream open only for writing is an error
1669    // (rexxref funct.xml:4454-4455): nothing is written or moved. A record
1670    // stream's lines are character positions, which a binary stream may name
1671    // past its end (funct.xml:3203-3205).
1672    if line.is_some() && write_only(interp, receiver)? && record_length(interp, receiver)?.is_none()
1673    {
1674        return notready(interp, receiver, StreamStatus::NotReady, b"1".to_vec());
1675    }
1676    // Positioning by line with no string first completes the record the
1677    // write position is inside of, and only then is a line that is not a
1678    // positive whole number refused, as ooRexx's method does; one with a
1679    // string pads only its own record (`r[bif.lineout.record]`). A transient
1680    // stream has no record to complete.
1681    if line.is_some()
1682        && data.is_none()
1683        && let Some(errno) = complete_record(interp, receiver)?
1684    {
1685        return notready(interp, receiver, StreamStatus::Error(errno), b"1".to_vec());
1686    }
1687    let record = record_length(interp, receiver)?;
1688    let mode = require(interp, receiver)?.mode;
1689    let state = require_mut(interp, receiver)?;
1690    let extent = state.append_extent;
1691    let Some(open) = state.open.as_mut() else {
1692        return Ok(Some(interp.text(b"1")));
1693    };
1694    // The line write position: a line number names a line, not a character, so
1695    // it goes through the same walk `SEEK ... LINE` uses. Writing at the
1696    // number itself overwrites whatever happens to sit at that offset.
1697    if let Some(line) = line {
1698        let record_length = if mode.record_based {
1699            mode.record_length
1700        } else {
1701            0
1702        };
1703        if open.transient {
1704            return Err(Raised::transient_positioning().into());
1705        }
1706        let line = as_position(positive(line)?);
1707        let (_, at) = seek_line(open, line, record_length);
1708        if within_append_extent(extent, at) {
1709            return notready(interp, receiver, StreamStatus::NotReady, b"1".to_vec());
1710        }
1711        open.write_position = at;
1712    }
1713    let Some(data) = data else {
1714        return Ok(Some(interp.text(b"0")));
1715    };
1716    if let Some(record_length) = record {
1717        return write_record(interp, receiver, record_length, data);
1718    }
1719    // A ctrl-Z is data like any other byte: Unix-like systems use no
1720    // end-of-file character in text files (rexxref intro.xml:3322-3330).
1721    let at = landing(open, mode.append, open.write_position);

1900/// `stream_lines`: `Count` -- the method's default -- answers the lines left
1901/// after the read position, `Normal` answers 1 or 0.
1902pub(super) fn lines(
1903    interp: &mut Interp,
1904    _cleared: Cleared,
1905    receiver: ObjRef,
1906    args: &[Option<ObjRef>],
1907) -> Result<Option<ObjRef>, Failure> {
1908    // `stream_lines`: the first letter only, and anything but `N` or `C` is
1909    // 93.915.
1910    let option = match args.first().copied().flatten() {
1911        Some(value) => Some(write_text(interp, value)?),
1912        None => None,
1913    };
1914    declared_only(args, 1)?;
1915    let quick = match option {
1916        Some(text) => match text.first().map(u8::to_ascii_uppercase) {
1917            Some(b'N') => true,
1918            Some(b'C') => false,
1919            _ => return Err(Raised::method_option_not_recognised("CN", &text).into()),
1920        },
1921        None => false,
1922    };
1923    if let Some(which) = standard_opened(interp, receiver)? {
1924        let StandardStream::In = which else {
1925            return Ok(Some(interp.text(b"0")));
1926        };
1927        let left = interp.input_remaining_lines(!quick).unwrap_or(1);
1928        let answer = if quick { u64::from(left > 0) } else { left };
1929        return Ok(Some(interp.text_built(answer.to_string().into_bytes())));
1930    }
1931    if !ensure_open(interp, receiver, false, b"0")? {
1932        return Ok(Some(interp.text(b"0")));
1933    }
1934    let record_length = record_length(interp, receiver)?;
1935    let state = require_mut(interp, receiver)?;
1936    let readable = state.mode.read_only || state.mode.read_write;
1937    let Some(open) = state.open.as_mut() else {
1938        return Ok(Some(interp.text(b"0")));
1939    };
1940    // A stream that cannot read has no lines (`StreamInfo::lines`), and a
1941    // transient stream answers whether data waits, as `chars` does.
1942    if !readable {
1943        return Ok(Some(interp.text(b"0")));
1944    }
1945    if open.transient {
1946        let waiting = transient_has_data(open);
1947        return Ok(Some(interp.text(if waiting { b"1" } else { b"0" })));
1948    }
1949    let size = as_position(size_of(open.flushed()));
1950    if open.read_position > size {
1951        return Ok(Some(interp.text(b"0")));
1952    }
1953    // A record stream counts the records left, a partial last one included,
1954    // and answers that count for `Normal` too.
1955    if let Some(record_length) = record_length {
1956        let length = as_position(record_length);
1957        let records = size / length + i64::from(size % length != 0);
1958        let answer = records - (open.read_position - 1) / length;
1959        return Ok(Some(interp.text_built(answer.to_string().into_bytes())));
1960    }
1961    if quick {
1962        return Ok(Some(interp.text(b"1")));
1963    }
1964    // The lines that remain, counted each time (rexxref funct.xml:3306-3307).
1965    // The count is a read, so it leaves the stream `READY`
1966    // (`StreamInfo::lines` calls `readSetup` before it scans).
1967    let answer = lines_after(open, open.read_position).unwrap_or(0);
1968    state.status = StreamStatus::Ready;
1969    Ok(Some(interp.text_built(answer.to_string().into_bytes())))
1970}

2092/// The position option table. **Every word takes one letter**, unlike OPEN's
2093/// per-option minimums, so `r`, `w`, `c` and `l` all match; a bare number is
2094/// the offset, and anything else fails.
2095fn parse_positioning(options: &[u8]) -> Result<Positioning, ()> {
2096    let mut parsed = Positioning {
2097        style: SeekFrom::Start,
2098        style_set: false,
2099        read: false,
2100        write: false,
2101        by_char: false,
2102        by_line: false,
2103        offset: None,
2104    };
2105    let mut tokens = Tokens::new(options);
2106    while let Some(token) = tokens.next() {
2107        let style = match token {
2108            b"=" => Some(SeekFrom::Start),
2109            b"<" => Some(SeekFrom::End),
2110            b"+" => Some(SeekFrom::Forward),
2111            b"-" => Some(SeekFrom::Backward),
2112            _ => None,
2113        };
2114        if let Some(style) = style {
2115            if parsed.style_set {
2116                return Err(());
2117            }
2118            parsed.style = style;
2119            parsed.style_set = true;
2120            continue;
2121        }
2122        if matches(token, b"READ") {
2123            if parsed.read || parsed.write {
2124                return Err(());
2125            }
2126            parsed.read = true;
2127        } else if matches(token, b"WRITE") {
2128            if parsed.read || parsed.write {
2129                return Err(());
2130            }
2131            parsed.write = true;
2132        } else if matches(token, b"CHAR") {
2133            if parsed.by_char || parsed.by_line {
2134                return Err(());
2135            }
2136            parsed.by_char = true;
2137        } else if matches(token, b"LINE") {
2138            if parsed.by_char || parsed.by_line {
2139                return Err(());
2140            }
2141            parsed.by_line = true;
2142        } else if parsed.offset.is_none() {
2143            // The fallback token: an offset, once, digits only and within
2144            // `int64_t` -- which is why `seek 1e1`, `seek =1.5` and
2145            // `seek 9223372036854775808` are the bare 93.
2146            match to_number(token).and_then(|number| i64::try_from(number).ok()) {

2152        }

2319/// [`position`]'s move of `open`'s positions, once the stream is open and
2320/// the pointers and unit are chosen.
2321fn seek_open(
2322    open: &mut interrexx_core::OpenFile,
2323    parsed: &Positioning,
2324    offset: i64,
2325    mode: interrexx_core::OpenMode,
2326    record_length: u64,
2327    extent: u64,
2328) -> Seeked {
2329    let from = if parsed.read {
2330        open.read_position
2331    } else {
2332        open.write_position
2333    };
2334    let size = as_position(size_of(open.flushed()));
2335    if parsed.by_char {
2336        // `+` and `-` move from the position the command names
2337        // (Deviation 49).
2338        let landed = match parsed.style {
2339            SeekFrom::Start => offset,
2340            SeekFrom::End => size - offset + 1,
2341            SeekFrom::Forward | SeekFrom::Backward => from + offset,
2342        };
2343        if landed < 1 || (parsed.write && within_append_extent(extent, landed)) {
2344            return Seeked::Invalid;
2345        }
2346        if parsed.read {
2347            open.read_position = landed;
2348        }
2349        if parsed.write {
2350            open.write_position = landed;
2351        }
2352        return Seeked::Landed(landed.cast_unsigned());
2353    }
2354    // Line positioning on a text stream open only for writing is an error
2355    // (rexxref funct.xml:4454-4455); a record stream's lines are character
2356    // positions.
2357    if !(mode.read_write || mode.read_only) && record_length == 0 {
2358        return Seeked::Invalid;
2359    }
2360    let lines = match parsed.style {
2361        SeekFrom::Start => Ok(offset),
2362        SeekFrom::End => line_count(open, record_length).map(|count| as_position(count) - offset),
2363        SeekFrom::Forward | SeekFrom::Backward => {
2364            line_of(open, from, record_length).map(|current| current + offset)
2365        }
2366    };
2367    let target = match lines {
2368        Ok(target) => target,
2369        Err(error) => return Seeked::Failed(error.raw_os_error().unwrap_or(0)),
2370    };
2371    if target < 1 {
2372        return Seeked::Invalid;
2373    }
2374    let (line, at) = seek_line(open, target, record_length);
2375    if parsed.write && within_append_extent(extent, at) {
2376        return Seeked::Invalid;
2377    }
2378    if parsed.read {
2379        open.read_position = at;
2380    }
2381    if parsed.write {
2382        open.write_position = at;
2383    }
2384    Seeked::Landed(line)
2385}

2397/// `stream_close`: `READY:` for a stream that was open, and the null string
2398/// for one that never was. Either way the state goes back to `UNKNOWN`.
2399///
2400/// A standard stream answers from its state instead of taking its
2401/// [`interrexx_core::OpenFile`], which an owned persistent stdout alias retains. Measured, and the three rows agree on one rule --
2402/// anything but `UNKNOWN` closes as `READY:`: the first close answers
2403/// `READY:` and the second the null string, a write after a close makes the
2404/// next close `READY:` again, and an exhausted `.STDIN` sitting at `NOTREADY`
2405/// also closes as `READY:`.
2406pub(super) fn close(
2407    interp: &mut Interp,
2408    _cleared: Cleared,
2409    receiver: ObjRef,
2410    _args: &[Option<ObjRef>],
2411) -> Result<Option<ObjRef>, Failure> {
2412    // A stream `uninit` has dropped the state of closes as nothing and
2413    // answers no value, as a second `uninit` does (`r[class.stream.uninit]`);
2414    // one `init` never set up is refused as the other methods refuse it.
2415    if interp.stream(receiver).is_none() {
2416        return match no_state(interp, receiver) {
2417            Failure::Raised(raised) if (raised.number, raised.sub) == (48, 1) => Ok(None),
2418            refused => Err(refused),
2419        };
2420    }
2421    release_held(interp, receiver)?;
2422    let state = require_mut(interp, receiver)?;
2423    let was_open = if state.standard.is_some() {
2424        if let Some(open) = state.open.as_mut() {
2425            let _ = open.flush();
2426        }
2427        state.status != StreamStatus::Unknown
2428    } else {
2429        state.open.take().is_some()
2430    };
2431    state.status = StreamStatus::Unknown;
2432    let answer: &[u8] = if was_open { b"READY:" } else { b"" };
2433    Ok(Some(interp.text(answer)))
2434}

2436/// `stream_flush`: writes what the stream holds and answers `READY:`,
2437/// including for a stream nothing has opened -- measured, and it leaves the
2438/// state alone. A write that fails is `NOTREADY` answering `ERROR:n`
2439/// (`StreamInfo::streamFlush`).
2440///
2441/// A standard output or error stream hands the interpreter's buffer for its
2442/// descriptor over. A buffered one whose held writes that hand-over refused
2443/// answers the descriptor's error, and so does every flush after it until the
2444/// stream writes again: the refused bytes are kept and each retry fails.
2445pub(super) fn flush(
2446    interp: &mut Interp,
2447    _cleared: Cleared,
2448    receiver: ObjRef,
2449    _args: &[Option<ObjRef>],
2450) -> Result<Option<ObjRef>, Failure> {
2451    let state = require(interp, receiver)?;
2452    let buffered = state.handle.is_none() && !state.mode.nobuffer;
2453    let releasing =
2454        buffered && (!state.held.is_empty() || matches!(state.status, StreamStatus::Error(_)));
2455    let sink = match state.standard {
2456        Some(StandardStream::Out) => Some(0),
2457        Some(StandardStream::Err) => Some(1),
2458        Some(StandardStream::In) | None => None,
2459    };
2460    release_held(interp, receiver)?;
2461    if let Some(sink) = sink {
2462        match sink {
2463            0 => interp.hand_over_stdout(),
2464            _ => interp.hand_over_stderr(),
2465        }
2466        if releasing && let Some(errno) = interp.sink_errors[sink] {
2467            let answer = format!("ERROR:{errno}").into_bytes();
2468            return notready(interp, receiver, StreamStatus::Error(errno), answer);
2469        }
2470    }
2471    let state = require_mut(interp, receiver)?;
2472    if let Some(Err(error)) = state.open.as_mut().map(interrexx_core::OpenFile::flush) {
2473        let errno = error.raw_os_error().unwrap_or(0);
2474        let answer = format!("ERROR:{errno}").into_bytes();
2475        return notready(interp, receiver, StreamStatus::Error(errno), answer);
2476    }
2477    Ok(Some(interp.text(b"READY:")))
2478}

2480/// `stream_uninit`: drops the state block, after which every entry point here
2481/// raises 48.1. **Answers nothing**, so a value context is 91.999: which
2482/// returns no result either.
2483pub(super) fn uninit(
2484    interp: &mut Interp,
2485    _cleared: Cleared,
2486    receiver: ObjRef,
2487    _args: &[Option<ObjRef>],
2488) -> Result<Option<ObjRef>, Failure> {

2490        release_held(interp, receiver)?;

2492    if let Some(object) = interp.heap.get_mut(receiver) {
2493        let before = object.body.held_bytes();
2494        if let Body::Instance { native, .. } = &mut object.body {
2495            *native = None;
2496        }
2497        let after = object.body.held_bytes();
2498        interp.heap.rehold_body_bytes(before, after);
2499    }
2500    Ok(None)
