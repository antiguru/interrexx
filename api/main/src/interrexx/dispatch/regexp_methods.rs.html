1//! The five native methods the `rxregexp` library exports, which
2//! `rxregexp.cls` binds to `RegularExpression`
3//! (`docs/design/extensions.md`). The
4//! compiled pattern and the match mode live in the instance's own native
5//! state; `!POS` is the receiver's variable in the method's scope.

17/// What an instance's pattern is.
18#[derive(Clone, Debug)]
19enum Compiled {
20    /// None was given: `new` with no pattern.
21    Absent,
22    Pattern(Rc<Regex>),
23    /// The last `parse` failed, after which everything matches.
24    Failed,
25    /// `uninit` has run (Deviation 88).
26    Freed,
27}
28
29#[derive(Clone, Debug)]
30struct RegExpState {
31    compiled: Compiled,
32    minimal: bool,
33}
34
35impl ExtensionFields for RegExpState {
36    fn clone_fields(&self) -> Box<dyn ExtensionFields> {
37        Box::new(self.clone())
38    }
39}

41/// The receiver's state, or `None` where no `init` has stored one.
42fn state(interp: &Interp, receiver: ObjRef) -> Option<&RegExpState> {
43    match &interp.heap.get(receiver)?.body {
44        Body::Instance {
45            native: Some(native),
46            ..
47        } => match &**native {
48            NativeState::Extension(state) => state.downcast_ref(),
49            _ => None,
50        },
51        _ => None,
52    }
53}
54
55/// Stores `state` as the receiver's. A receiver that cannot hold one raises
56/// 93.900, as the methods used without one do.
57fn store(interp: &mut Interp, receiver: ObjRef, state: RegExpState) -> Result<(), Failure> {
58    let Some(object) = interp.heap.get_mut(receiver) else {
59        unreachable!("a live receiver")
60    };
61    let Body::Instance { native, .. } = &mut object.body else {
62        return Err(Raised::method_user_defined(UNUSABLE).into());
63    };
64    if native
65        .as_deref()
66        .is_some_and(|held| !matches!(held, NativeState::Extension(_)))
67    {
68        return Err(Raised::method_user_defined(UNUSABLE).into());
69    }
70    *native = Some(Box::new(NativeState::Extension(ExtensionState::new(
71        Box::new(state),
72    ))));
73    Ok(())
74}

76/// Deviation 88's message for an instance no pattern was ever compiled for.
77const NO_PATTERN: &str = "RegularExpression has no pattern; give one to new or parse";
78/// Deviation 88's message for an instance after `uninit`.
79const UNINITIALIZED: &str = "RegularExpression has been uninitialized";
80/// The message for a receiver whose body has another native state.
81const UNUSABLE: &str = "RegularExpression methods cannot run on this receiver";

83/// Sets the receiver's `!POS` in `scope`.
84fn set_position(
85    interp: &mut Interp,
86    receiver: ObjRef,
87    scope: ObjRef,
88    position: usize,
89) -> Result<(), Failure> {
90    let value = interp.counted(position);
91    let owner = interp.pool_owner(receiver)?;
92    interp.set_pool_variable(owner, scope, b"!POS", value);
93    Ok(())
94}

96/// A string argument's bytes, 88.909 for an object with no string value, as
97/// the native boundary converts one.
98fn text_argument(interp: &mut Interp, value: ObjRef, position: usize) -> Result<Vec<u8>, Failure> {
99    let Some(text) = interp.blamed_string_conversion(value)? else {
100        return Err(Raised::native_argument_needs_a_string_value(position).into());
101    };
102    Ok(interp.to_text(text).into_owned())
103}

105/// 88.922 where more than `arity` arguments were passed, as the native
106/// boundary checks a method's signature.
107fn check_arity(args: &[Option<ObjRef>], arity: usize) -> Result<(), Failure> {
108    if args.len() > arity {
109        return Err(Raised::too_many_external_arguments(arity).into());
110    }
111    Ok(())
112}

114/// The arguments a method takes, read from the left as the native boundary
115/// reads them: a required one that is missing is 88.901 and one with no
116/// string value 88.909, and only then are too many arguments 88.922, so an
117/// argument beyond the last the method takes is not read. `required` leading
118/// arguments must be present.
119fn read_arguments(
120    interp: &mut Interp,
121    args: &[Option<ObjRef>],
122    arity: usize,
123    required: usize,
124) -> Result<Vec<Option<Vec<u8>>>, Failure> {
125    let mut texts = Vec::with_capacity(arity);
126    for at in 0..arity {
127        match args.get(at).copied().flatten() {
128            Some(value) => texts.push(Some(text_argument(interp, value, at + 1)?)),
129            None if at < required => return Err(Raised::missing_native_argument(at + 1).into()),
130            None => texts.push(None),
131        }
132    }
133    check_arity(args, arity)?;
134    Ok(texts)
135}

137/// A match type: `Some(minimal)` for `MINIMAL` or `MAXIMAL`, `Some(None)`
138/// for `CURRENT` where `current` allows it, `None` for anything else.
139fn match_type(text: &[u8], current: bool) -> Option<Option<bool>> {
140    let upper = text.to_ascii_uppercase();
141    match &upper[..] {
142        b"MINIMAL" => Some(Some(true)),
143        b"MAXIMAL" => Some(Some(false)),
144        b"CURRENT" if current => Some(None),
145        _ => None,
146    }
147}

150fn incorrect_call() -> Failure {
151    Raised::syntax(93, 0, Vec::new()).into()

154/// `RegExp_Init([pattern [, matchtype]])`: compiles the pattern, if any, and
155/// sets the mode. A compile error raises 38, ahead of a bad match type's 93.
156pub(crate) fn init(
157    interp: &mut Interp,
158    receiver: ObjRef,
159    _scope: ObjRef,
160    args: &[Option<ObjRef>],
161) -> Result<Option<ObjRef>, Failure> {
162    let texts = read_arguments(interp, args, 2, 0)?;
163    let mut minimal = Some(false);
164    if let Some(text) = &texts[1] {
165        minimal = match_type(text, false).map(|kind| kind.unwrap_or(false));
166    }
167    let mut compiled = Compiled::Absent;
168    if let Some(pattern) = &texts[0] {
169        match regexp::compile(pattern) {
170            Ok((regex, _)) => compiled = Compiled::Pattern(Rc::new(regex)),
171            Err(_) => return Err(Raised::syntax(38, 0, Vec::new()).into()),
172        }
173    }
174    let Some(minimal) = minimal else {
175        return Err(incorrect_call());
176    };
177    store(interp, receiver, RegExpState { compiled, minimal })?;
178    // The library keeps its automaton behind `CSELF`, which a class binding
179    // these methods itself can read: a `.Pointer` until `uninit` drops it.
180    let cself = Host::new_pointer(interp, std::ptr::null_mut());
181    Host::set_object_variable(interp, b"CSELF", Some(cself));
182    Ok(None)
183}

185/// `RegExp_Uninit()`: frees the pattern. Once freed, a second `uninit`, as
186/// the collection runs after an explicit one, does nothing.
187pub(crate) fn uninit(
188    interp: &mut Interp,
189    receiver: ObjRef,
190    _scope: ObjRef,
191    args: &[Option<ObjRef>],
192) -> Result<Option<ObjRef>, Failure> {
193    check_arity(args, 0)?;
194    Host::drop_object_variable(interp, b"CSELF");
195    if let Some(held) = state(interp, receiver) {
196        let minimal = held.minimal;
197        store(
198            interp,
199            receiver,
200            RegExpState {
201                compiled: Compiled::Freed,
202                minimal,
203            },
204        )?;
205    }
206    Ok(None)
207}

209/// `RegExp_Parse(pattern [, matchtype])`: 0 or a compile error code, with
210/// `!POS` the bytes consumed or the error's position. A failed parse leaves
211/// an instance that matches everything.
212pub(crate) fn parse(
213    interp: &mut Interp,
214    receiver: ObjRef,
215    scope: ObjRef,
216    args: &[Option<ObjRef>],
217) -> Result<Option<ObjRef>, Failure> {
218    let texts = read_arguments(interp, args, 2, 1)?;
219    let held = state(interp, receiver).cloned();
220    if matches!(
221        held.as_ref().map(|held| &held.compiled),
222        Some(Compiled::Freed)
223    ) {
224        return Err(Raised::method_user_defined(UNINITIALIZED).into());
225    }
226    let mut minimal = held.as_ref().is_some_and(|held| held.minimal);
227    if let Some(text) = &texts[1] {
228        match match_type(text, true) {
229            Some(Some(kind)) => minimal = kind,
230            Some(None) => {}
231            None => return Err(incorrect_call()),
232        }
233    }
234    let Some(pattern) = &texts[0] else {
235        unreachable!("the pattern is required")
236    };
237    let (compiled, code, position) = match regexp::compile(pattern) {
238        Ok((regex, consumed)) => (Compiled::Pattern(Rc::new(regex)), 0, consumed),
239        Err(error) => (Compiled::Failed, usize::from(error.code), error.position),
240    };
241    store(interp, receiver, RegExpState { compiled, minimal })?;
242    set_position(interp, receiver, scope, position)?;
243    Ok(Some(interp.counted(code)))
244}

246/// The receiver's pattern for `match` and `pos`, 93.900 where there is none.
247fn pattern_of(interp: &Interp, receiver: ObjRef) -> Result<(Option<Rc<Regex>>, bool), Failure> {
248    let Some(held) = state(interp, receiver) else {
249        return Err(Raised::method_user_defined(NO_PATTERN).into());
250    };
251    match &held.compiled {
252        Compiled::Pattern(regex) => Ok((Some(Rc::clone(regex)), held.minimal)),
253        Compiled::Failed => Ok((None, held.minimal)),
254        Compiled::Absent => Err(Raised::method_user_defined(NO_PATTERN).into()),
255        Compiled::Freed => Err(Raised::method_user_defined(UNINITIALIZED).into()),
256    }
257}

259/// The subject argument of `match` and `pos`, 88.901 where it is omitted.
260fn subject(interp: &mut Interp, args: &[Option<ObjRef>]) -> Result<Vec<u8>, Failure> {
261    let mut texts = read_arguments(interp, args, 1, 1)?;
262    Ok(texts.swap_remove(0).expect("the subject is required"))
263}

265/// `RegExp_Match(string)`: 1 or 0, with `!POS` set.
266pub(crate) fn match_string(
267    interp: &mut Interp,
268    receiver: ObjRef,
269    scope: ObjRef,
270    args: &[Option<ObjRef>],
271) -> Result<Option<ObjRef>, Failure> {
272    let text = subject(interp, args)?;
273    let (regex, minimal) = pattern_of(interp, receiver)?;
274    let (matched, position) = match regex {
275        None => (true, 0),
276        Some(regex) if minimal => regex.match_minimal(&text),
277        Some(regex) => regex.match_maximal(&text),
278    };
279    set_position(interp, receiver, scope, position)?;
280    Ok(Some(interp.counted(usize::from(matched))))
281}

283/// `RegExp_Pos(haystack)`: the 1-based start of the first match or 0, with
284/// `!POS` set.
285pub(crate) fn pos(
286    interp: &mut Interp,
287    receiver: ObjRef,
288    scope: ObjRef,
289    args: &[Option<ObjRef>],
290) -> Result<Option<ObjRef>, Failure> {
291    let text = subject(interp, args)?;
292    let (regex, minimal) = pattern_of(interp, receiver)?;
293    let (start, position) = match regex {
294        None => (1, 0),
295        Some(regex) => regex.pos(&text, minimal),
296    };
297    set_position(interp, receiver, scope, position)?;
298    Ok(Some(interp.counted(start)))
