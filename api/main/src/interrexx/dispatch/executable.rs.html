54/// What the four setters have changed on one `Method` object, over what its
55/// directive declared.
56#[derive(Copy, Clone, Default)]
57pub(crate) struct MethodFlagWrites {
58    pub(crate) unguarded: Option<bool>,
59    pub(crate) private: bool,
60    pub(crate) protected: bool,
61}
62
63impl Interp {
64    /// Whether a send of `method`, `declared` guarded by its directive,
65    /// reserves its guard lock: `setGuarded` and `setUnguarded` on its
66    /// `Method` object overrule the directive, as `isGuarded` reads them.
67    #[inline]
68    pub(crate) fn sends_guarded(
69        &self,
70        method: interrexx_classes::MethodId,
71        declared: bool,
72    ) -> bool {
73        if self.method_flag_writes.is_empty() {
74            return declared;
75        }
76        self.written_guard(method, declared)
77    }
78
79    #[cold]
80    #[inline(never)]
81    fn written_guard(&self, method: interrexx_classes::MethodId, declared: bool) -> bool {
82        self.method_flag_writes
83            .iter()
84            .find(|(object, _)| {
85                self.executable_sources
86                    .get(object)
87                    .is_some_and(|record| record.installed == Some(method))
88            })
89            .and_then(|(_, writes)| writes.unguarded)
90            .map_or(declared, |unguarded| !unguarded)
91    }
92}

103/// One flag's value for this receiver: what the directive declared, with any
104/// write the setters have made over it.
105fn flag(interp: &Interp, receiver: ObjRef, which: Flag) -> Result<bool, Failure> {
106    let declared = declared_flag(interp, source_of(interp, receiver)?, which)
107        || (which == Flag::Protected && interp.installed_is_protected(receiver))
108        || (which == Flag::Private && interp.installed_is_private(receiver));
109    let written = interp.method_flag_writes.get(&receiver).copied();
110    Ok(match (which, written) {
111        (Flag::Unguarded, Some(writes)) => writes.unguarded.unwrap_or(declared),
112        (Flag::Private, Some(writes)) => declared || writes.private,
113        (Flag::Protected, Some(writes)) => declared || writes.protected,
114        _ => declared,
115    })
116}

118/// One flag as the declaring directive set it.
119fn declared_flag(interp: &Interp, source: ExecutableSource, which: Flag) -> bool {
120    let ExecutableSource::Directive { program, directive } = source else {
121        // A primitive carries the flag word `new MethodClass` left it: every
122        // bit clear, which `isGuarded` reads as guarded. For example,
123        // `.Object~method('objectName')` answers `0 0 0 1 0 0 0` for the seven
124        // readers in the order they are declared here.
125        return false;
126    };
127    let Some(directive) = interp
128        .programs
129        .get(program)
130        .and_then(|program| program.directives.get(directive))
131    else {
132        return false;
133    };
134    match &directive.kind {
135        DirectiveKind::Method(method) => match which {
136            Flag::Abstract => method.abstract_,
137            Flag::Attribute => method.attribute,
138            Flag::Constant => false,
139            Flag::Unguarded => method.guard == GuardOption::Unguarded,
140            Flag::Package => method.access == Access::Package,
141            Flag::Private => method.access == Access::Private,
142            Flag::Protected => method.protection == Protection::Protected,
143        },
144        DirectiveKind::Attribute(attribute) => match which {
145            Flag::Abstract => attribute.abstract_,
146            // The attribute getter and the method an `::ATTRIBUTE ... GET` with
147            // a body takes both mark the attribute state. For example,
148            // `::attribute ATT` and `::attribute ATT get` with a body each
149            // answer `isAttribute` `.true`.
150            Flag::Attribute => true,
151            Flag::Constant => false,
152            Flag::Unguarded => attribute.guard == GuardOption::Unguarded,
153            Flag::Package => attribute.access == Access::Package,
154            Flag::Private => attribute.access == Access::Private,
155            Flag::Protected => attribute.protection == Protection::Protected,
156        },
157        // A constant getter is marked constant and unguarded outright, taking
158        // no access or protection option. For example, `::constant CON 5`
159        // answers `0 0 1 0 0 0 0`, and it is the one directive whose
160        // `isGuarded` is `.false` without an `UNGUARDED` keyword.
161        DirectiveKind::Constant(_) => matches!(which, Flag::Constant | Flag::Unguarded),
162        // A `::ROUTINE` is not a `Method` and answers none of these; the rows
163        // are not bound at `Routine`.
164        _ => false,
165    }
166}

313/// `Method~setSecurityManager` and `Routine~setSecurityManager`, which set the
314/// security manager on the executable's code.
315fn set_security_manager(
316    interp: &mut Interp,
317    _cleared: Cleared,
318    receiver: ObjRef,
319    args: &[Option<ObjRef>],
320) -> Result<Option<ObjRef>, Failure> {
321    let source = source_of(interp, receiver)?;
322    let rexx_code = is_rexx_code(interp, source);
323    // **The manager lands on the package, not on the executable.** Both
324    // setters write the manager onto the package of a Rexx-source body, so a
325    // manager set through one method of a package is in force for every
326    // activation of that package's code.
327    if rexx_code {
328        let program = match source {
329            ExecutableSource::Directive { program, .. }
330            | ExecutableSource::ConstantExpression { program, .. }
331            | ExecutableSource::Main { program } => program,
332            ExecutableSource::Native
333            | ExecutableSource::External { .. }
334            | ExecutableSource::Loaded { .. } => unreachable!("native code is not Rexx code"),
335        };
336        let package = interp.package_object(crate::plan::Package::Program(program))?;
337        interp.install_security_manager(package, args.first().copied().flatten());
338    }
339    Ok(Some(crate::eval::logical(rexx_code)))
340}

342/// Whether this executable's body is Rexx source -- the distinction
343/// `setSecurityManager` answers and `~source` reads.
344fn is_rexx_code(interp: &Interp, source: ExecutableSource) -> bool {
345    let (program, directive) = match source {
346        ExecutableSource::Main { .. } | ExecutableSource::ConstantExpression { .. } => return true,
347        ExecutableSource::Native
348        | ExecutableSource::External { .. }
349        | ExecutableSource::Loaded { .. } => return false,
350        ExecutableSource::Directive { program, directive } => (program, directive),
351    };
352    let Some(directive) = interp
353        .programs
354        .get(program)
355        .and_then(|program| program.directives.get(directive))
356    else {
357        return false;
358    };
359    match &directive.kind {
360        DirectiveKind::Method(method) => method.body.is_some() && method.external.is_none(),
361        DirectiveKind::Attribute(attribute) => {
362            attribute.body.is_some() && attribute.external.is_none()
363        }
364        DirectiveKind::Routine(routine) => routine.body.is_some() && routine.external.is_none(),
365        _ => false,
366    }
367}

369/// The package the declaring directive belongs to, `REXX` for a primitive, and
370/// `.nil` for a loaded library procedure no directive has bound.
371fn package(
372    interp: &mut Interp,
373    _cleared: Cleared,
374    receiver: ObjRef,
375    _args: &[Option<ObjRef>],
376) -> Result<Option<ObjRef>, Failure> {
377    let source = source_of(interp, receiver)?;
378    Ok(Some(match interp.source_package(source) {
379        // The interpreter's own library programs are the `REXX` package:
380        // `.Stream~method('INIT')~package~name` is `REXX` and its `source` has
381        // no lines.
382        Some(crate::plan::Package::Program(program))
383            if interp.library_programs.contains(&program) =>
384        {
385            interp.package_object(crate::plan::Package::Rexx)?
386        }
387        Some(package) => interp.package_object(package)?,
388        None => ObjRef::NIL,
389    }))
390}

392/// The body's own source lines as an `Array`.
393fn source(
394    interp: &mut Interp,
395    _cleared: Cleared,
396    receiver: ObjRef,
397    _args: &[Option<ObjRef>],
398) -> Result<Option<ObjRef>, Failure> {
399    let lines = match source_of(interp, receiver)? {
400        // The REXX package's methods are restored from the image with no
401        // source: every method of `.OrderedCollection`, `.DateTime` and
402        // `.Stream` answers no lines.
403        ExecutableSource::Directive { program, .. } | ExecutableSource::Main { program }
404            if interp.library_programs.contains(&program) =>
405        {
406            Vec::new()
407        }
408        ExecutableSource::Directive { program, directive } => {
409            block_source_lines(interp, program, Some(directive))?
410        }
411        ExecutableSource::Main { program } => block_source_lines(interp, program, None)?,
412        ExecutableSource::ConstantExpression { program, directive } => {
413            directive_clause_lines(interp, program, directive)
414        }
415        ExecutableSource::Native
416        | ExecutableSource::External { .. }
417        | ExecutableSource::Loaded { .. } => Vec::new(),
418    };
419    array_of_texts(interp, lines).map(Some)
420}

422/// The physical lines a `::CONSTANT` expression's method answers as its source:
423/// its own directive clause's, and not the last `::CONSTANT` clause's for each.
424fn directive_clause_lines(interp: &Interp, program: ProgramId, directive: usize) -> Vec<Vec<u8>> {
425    let Some(source) = interp.programs.get(program) else {
426        return Vec::new();
427    };
428    let text = &source.source;
429    let Some(span) = source
430        .directives
431        .get(directive)
432        .map(|d| d.clause_span.clone())
433    else {
434        return Vec::new();
435    };
436    if !text.is_traceable() || span.is_empty() {
437        return Vec::new();
438    }
439    let (first, last) = (text.line_of(span.start), text.line_of(span.end - 1));
440    (first..=last)
441        .filter_map(|line| text.line(line).map(<[u8]>::to_vec))
442        .collect()
443}

445/// The physical lines one code block's source covers: `Some(i)` is directive
446/// `i`'s own body and `None` is the program's main section. Refuses a
447/// `program` that was freed, which only a holder the collector was not shown
448/// reaches.
449fn block_source_lines(
450    interp: &Interp,
451    program: ProgramId,
452    directive: Option<usize>,
453) -> Result<Vec<Vec<u8>>, Failure> {
454    let source = interp.programs.get(program);
455    debug_assert!(
456        source.is_some(),
457        "{program:?} was freed while its executable was held"
458    );
459    let Some(source) = source else {
460        return Err(Loud::unknown_receiver("an executable whose program was freed").into());
461    };
462    let text = &source.source;
463    if !text.is_traceable() {
464        return Ok(Vec::new());
465    }
466    let (first, from) = match directive {
467        None => (text.main_section_first_line(), 0),
468        Some(directive) => match block_start(source, directive) {
469            Some(start) => block_first_line(text, start),
470            None => return Ok(Vec::new()),
471        },
472    };
473    let next = directive.map_or(0, |directive| directive + 1);
474    let (last, upto) = match source.directives.get(next) {
475        // A synthetic directive covers no source: it is the one
476        // `Interp::record_compiled_body` files over a text that had none, and
477        // a written directive's clause always spans at least `::method x`.
478        Some(next) if !next.clause_span.is_empty() => block_last_line(text, next.clause_span.start),
479        _ => {
480            let last = text.line_count();
481            (last, text.line_span(last).map_or(0, |span| span.end))
482        }
483    };
484    // `extractSourceLines`' own step back over an end that sits at the start of
485    // a line: the block's end offset is that line's length, so an **empty**
486    // last line drops out. For example, counting the items of a body followed
487    // by `n` blank lines: one blank answers the body alone and two answer the
488    // body and one blank, at the end of the file and before a following
489    // directive alike; and a `::method` with nothing after it but one blank
490    // line answers `Array(0)`.
491    let (last, upto, stepped_back) = match text.line_span(last) {
492        Some(span) if span.is_empty() => {
493            let stepped = last.saturating_sub(1);
494            (
495                stepped,
496                text.line_span(stepped).map_or(0, |span| span.end),
497                true,
498            )
499        }
500        _ => (last, upto, false),
501    };
502    // A body that starts after a `;` and ends on the same line, with nothing
503    // between them, holds no line (`r[class.method.source]`): `::method f ;`
504    // at the end of the text, or before the next directive, is a source of no
505    // lines. A blank line after it that stepped back leaves its one empty
506    // line.
507    if directive.is_some()
508        && !stepped_back
509        && last == first
510        && from > 0
511        && text
512            .line_span(first)
513            .is_some_and(|span| span.start.max(from) >= upto.min(span.end))
514    {
515        return Ok(Vec::new());
516    }
517    Ok((first..=last)
518        .map_while(|line| {
519            let span = text.line_span(line)?;
520            let cut = span.start.max(from)..span.end.min(upto);
521            text.span_bytes(cut.clone())
522                .map(<[u8]>::to_vec)
523                .or_else(|| Some(Vec::new()))
524        })
525        .collect())
526}

528/// The line a block begins on, and the byte within the text it begins at.
529fn block_first_line(text: &interrexx_parse::ProgramSource, start: usize) -> (usize, usize) {
530    // The `;` may be inside the clause's own span or just past it, which is a
531    // difference between a directive clause and a body clause and not one the
532    // answer turns on.
533    let after = match text.span_bytes(start..start + 1) {
534        Some(b";") => start + 1,
535        _ => start,
536    };
537    match text.span_bytes(after.saturating_sub(1)..after) {
538        Some(b";") => (text.line_of(after.saturating_sub(1)), after),
539        _ => (text.line_of(start) + 1, 0),
540    }
541}

543/// The line a block ends on, and the byte it ends at, for a block terminated
544/// by the directive clause beginning at `next`.
545fn block_last_line(text: &interrexx_parse::ProgramSource, next: usize) -> (usize, usize) {
546    let line = text.line_of(next);
547    match text.line_span(line) {
548        Some(span) if span.start < next => (line, next),
549        _ => {
550            let previous = line.saturating_sub(1);
551            (
552                previous,
553                text.line_span(previous).map_or(0, |span| span.end),
554            )
555        }
556    }
557}

559/// The byte the block whose source is being extracted begins after, or `None`
560/// for a directive that owns no Rexx source at all.
561fn block_start(program: &Program, directive: usize) -> Option<usize> {
562    let clause_end = program.directives.get(directive)?.clause_span.end;
563    match &program.directives[directive].kind {
564        DirectiveKind::Method(method) => method.body.as_ref().map(|_| clause_end),
565        DirectiveKind::Routine(routine) => routine.body.as_ref().map(|_| clause_end),
566        DirectiveKind::Attribute(attribute) => attribute
567            .body
568            .as_ref()
569            .and_then(|body| body.instructions.first())
570            .map(|first| first.clause_span.end),
571        _ => None,
572    }
573}

600    receiver: ObjRef,
601    args: &[Option<ObjRef>],
602) -> Result<NativeStarted, Failure> {
603    let Some(Some(array)) = args.first().copied() else {
604        return Err(Raised::missing_method_argument(1).into());
605    };
606    let values = super::array_argument(interp, array, super::ArrayArgument::Positional)?;
607    begin_routine(interp, receiver, values, None, (b"CALLWITH", args))
608}

610/// the routine run directly, under the null string for a name, over the
611/// items of `arguments` or over none.
612/// # Errors
613/// 98.913 for `arguments` that is not an array, and what the routine raised.
614pub(crate) fn call_routine_directly(
615    interp: &mut Interp,
616    routine: ObjRef,
617    arguments: Option<ObjRef>,
618) -> Result<Option<ObjRef>, Failure> {
619    let values = match arguments {
620        Some(array) => super::array_argument(interp, array, super::ArrayArgument::Positional)?,
621        None => Vec::new(),
622    };
623    let started = begin_routine(interp, routine, values, Some(b""), (b"", &[]))?;
624    interp.complete_native(started)
625}

627/// The half [`call`] and [`call_with`] share, once the arguments are values,
628/// `name` being what a Rexx routine's frame is called, or `None` for the
629/// routine's own name: a directive's name, the name a compiled image gives it,
630/// or the name its package was created under. Up to the point a Rexx routine's
631/// body would run. `sent` is the message that runs it and that message's own
632/// arguments, which a routine run to its end here lists as a frame of its own;
633/// an empty name for none.
634fn begin_routine(
635    interp: &mut Interp,
636    receiver: ObjRef,
637    mut values: Vec<Option<ObjRef>>,
638    name: Option<&[u8]>,
639    sent: (&[u8], &[Option<ObjRef>]),
640) -> Result<NativeStarted, Failure> {
641    let Some(record) = interp.executable_sources.get(&receiver).copied() else {
642        return Err(Loud::unrecorded_executable().into());
643    };
644    // A library routine runs under its own table spelling, which is the name
645    // `Routine~call` passes: measured, `r~call(1, 2, 3)` over `RxCalcSqrt` is
646    // reported under a built-in routine line for RxCalcSqrt whether a directive or
647    // an external-routine load answered `r`.
648    let code = match (record.source, record.routine) {
649        (crate::ExecutableSource::Loaded { code }, _) => Some(code),
650        (_, Some((program, directive))) => {
651            interp.library_routine_code(crate::InstalledRoutine { program, directive })
652        }
653        _ => None,
654    };
655    if let Some(code) = code {
656        let name = interp.library_code_key(code).procedure.clone();
657        return interp
658            .run_library_routine(code, &name, &values, crate::RoutineObject::Given(receiver))
659            .map(|started| match started {
660                crate::run::Started::Ran(value) => NativeStarted::Ran(value),
661                crate::run::Started::Entered => NativeStarted::Entered(super::Then::Pass),
662            });
663    }
664    let row = match record.routine {
665        Some((program, directive)) => {
666            interp.rexx_routine_row(crate::InstalledRoutine { program, directive })
667        }
668        None => interp.rexx_routine_object(receiver),
669    };
670    if let Some(row) = row {
671        return interp
672            .run_internal_given(row, crate::run::InternalName::Row, &mut values, 0, receiver)
673            .map(|started| match started {
674                Started::Ran(value) => NativeStarted::Ran(Some(value)),
675                Started::Entered => NativeStarted::Entered(Then::Pass),
676            });
677    }
678    let Some((program, directive)) = record.routine else {
679        let crate::ExecutableSource::Main { program } = record.source else {
680            return Err(Loud::unrecorded_executable().into());
681        };
682        let name = match name {
683            Some(name) => name.to_vec(),
684            None => match interp
685                .programs
686                .get(program)
687                .and_then(|program| program.source.image_name())
688            {
689                Some(name) => name.to_vec(),
690                None => interp.program_display_name(program).to_vec(),
691            },
692        };
693        let (message, arguments) = sent;
694        let level = (!message.is_empty())
695            .then(|| interp.lookup(receiver, message, None))
696            .flatten();
697        if let Some(resolution) = level {
698            // A level held aside goes on below this one
699            // ([`Interp::deferred_level`]).
700            if interp.activity.deferred_level.is_some() {
701                interp.raise_deferred_level();
702            }
703            let caller = interp.running_activation().map(|activation| activation.id);
704            interp
705                .activity
706                .method_levels
707                .push(crate::activity::MethodLevel {
708                    caller,
709                    name: message.into(),
710                    resolution: Some(resolution),
711                    receiver,
712                    arguments: arguments.into(),
713                });
714        }
715        let ran = interp.call_program_main(program, values, &name);
716        if level.is_some() {
717            interp.activity.method_levels.pop();
718        }
719        return ran.map(NativeStarted::Ran);
720    };
721    let name = match name {
722        Some(name) => name.to_vec(),
723        None => interp.routine_directive_name(program, directive),
724    };
725    let started = interp.begin_installed_routine(program, directive, &mut values, &name)?;
726    // `.context~executable` inside answers the object the call came through:
727    // `r~call == r` and `r~copy~call == r~copy`.
728    if let NativeStarted::Entered(_) = started {
729        interp.activation_mut().cold_mut().executable = Some(receiver);
730    }
731    Ok(started)
732}
