8/// What the protocol's conversion limbs answered, before anything is built.
9enum RequiredString {
10    Object(ObjRef),
11    Bytes(Vec<u8>),
12}
13
14/// The message `Object~request("STRING")` looks for, and the one limb of the
15/// required-string protocol a program can write.
16pub(crate) const MAKESTRING: &[u8] = b"MAKESTRING";
17
18/// The message that asks an object for a value of a class.
19const REQUEST: &[u8] = b"REQUEST";
20
21/// What the default string value sends.
22pub(crate) const OBJECTNAME: &[u8] = b"OBJECTNAME";
23
24/// What the default object name sends for an object nothing has named and
25/// whose class is not a base class.
26pub(crate) const DEFAULTNAME: &[u8] = b"DEFAULTNAME";
27
28/// The required-string protocol's last limb, past every conversion.
29pub(crate) const STRING: &[u8] = b"STRING";

31/// What a value's own string value is, before the protocol's fallbacks: the
32/// value itself, the object a `makeString` answered, bytes no object holds, or
33/// nothing.
34enum StringConversion {
35    /// The object that *is* the string value: the value itself, or a stem's
36    /// own default value, or what a `makeString` answered.
37    Object(ObjRef),
38    /// The string value as bytes no object holds.
39    Bytes(Vec<u8>),
40    /// An array, whose string value is its items joined.
41    Array(ObjRef),
42    /// The receiver's behaviour has a `makeString` to send.
43    MakeString,
44    /// None of those, which is `.nil` from the string request's point of view.
45    None,
46}

48impl Interp {
49    /// Requests a string value, falling back to the receiver's readable
50    /// string representation and offering NOSTRING when conversion fails.
51    #[inline]
52    pub(crate) fn required_string_value(&mut self, value: ObjRef) -> Result<ObjRef, Failure> {
53        // **A string and a small integer are their own string value**, and
54        // that is the answer whether the latch is armed or clear:
55        // `classify_string_conversion`'s first arm returns
56        // `StringConversion::Object(value)` for both, before it touches
57        // `self`, allocates, or looks at the heap. Answering here rather than
58        // three calls deeper is what keeps an armed interpreter from paying
59        // `required_string_dispatch` + `required_string_answer` +
60        // `classify_string_conversion` for every operand of every comparison.
61        if matches!(value.decode(), Decoded::SmallInt(_) | Decoded::Text(_)) {
62            return Ok(value);
63        }
64        if self.reqstr_armed {
65            // **A heap string or number is its own string value as much as an
66            // inline one is** -- the same `classify_string_conversion` arm
67            // answers `Object(value)` for `Body::Text` and `Body::Num`. It
68            // costs the heap lookup that arm makes anyway, and it keeps the
69            // `push_temp`, because unlike the two decodings above this value
70            // does live in the arena and the caller's rooting of it is not
71            // this function's to assume.
72            if matches!(
73                self.heap.get(value).map(|object| &object.body),
74                Some(Body::Text { .. } | Body::Num { .. })
75            ) {
76                self.roots.activity_mut().push_temp(value);
77                return Ok(value);
78            }
79            return self.required_string_dispatch(value);
80        }
81        debug_assert!(
82            unshared!(self, self.required_string_latch_holds(value)),
83            "the required-string latch is off where the protocol would answer differently \
84             or raise"
85        );
86        Ok(value)
87    }

89    /// Whether the fast path is the right answer for `value` with the latch
90    /// clear -- one test per limb the latch claims cannot fire.
91    fn required_string_latch_holds(&mut self, value: ObjRef) -> bool {
92        // Limb 3's route, and it is not about `value` at all: a trap that
93        // could take NOSTRING while the latch is clear is wrong for every
94        // rendered object, because `Interp::exec_condition_trap` is what arms
95        // both and a clear latch here means it did not. The gate is the same
96        // one `required_string_dispatch` applies, so this fires exactly where
97        // that would have raised.
98        if self.trap_for(b"NOSTRING").is_some_and(|trap| !trap.call)
99            || self.condition_raises_syntax(b"NOSTRING")
100        {
101            return false;
102        }
103        // Limb 1's route: the conversion limbs, run in full, against the
104        // rendering the fast path hands back instead.
105        let expected = self.to_text(value).into_owned();
106        let answered = match self.required_string_answer(value) {
107            Ok(answered) => answered,
108            Err(_) => return false,
109        };
110        let bytes = match answered {
111            Some(RequiredString::Object(text)) => self.to_text(text).into_owned(),
112            Some(RequiredString::Bytes(bytes)) => bytes,
113            // `stringValue()`, which is what the fast path's own rendering of
114            // an object with no string value comes to.
115            None => match self.string_value_text(value) {
116                Ok(bytes) => bytes,
117                Err(_) => return false,
118            },
119        };
120        bytes == expected
121    }

123    /// **The answer is rooted here and not at the call sites.** A `makeString`
124    /// can allocate and can collect, and both fallbacks below build a string
125    /// of their own, so every value this returns is one the caller did not
126    /// have a root for. Rooting once, on the arm that can produce a fresh
127    /// object, is what keeps the gate above free of a temp push per rendered
128    /// value.
129    #[cold]
130    #[inline(never)]
131    fn required_string_dispatch(&mut self, value: ObjRef) -> Result<ObjRef, Failure> {
132        match self.required_string_answer(value) {
133            Ok(Some(RequiredString::Object(text))) => {
134                self.roots.activity_mut().push_temp(text);
135                return Ok(text);
136            }
137            Ok(Some(RequiredString::Bytes(bytes))) => {
138                let text = self.text_built(bytes);
139                self.roots.activity_mut().push_temp(text);
140                return Ok(text);
141            }
142            Ok(None) => {}
143            Err(failure) => {
144                self.blame_request(value);
145                return Err(failure);
146            }
147        }
148        // A reference whose referenced value has no string value stands for
149        // that value from here on: it is the object causing NOSTRING, and the
150        // object sent `STRING` (rexxref condtra.xml, NOSTRING's DESCRIPTION
151        // and ADDITIONAL).
152        let value = self.conversion_subject(value)?;
153        self.roots.activity_mut().push_temp(value);
154        // `sendMessage(STRING)`. An instance's or a class's `STRING` can be a
155        // Rexx method, so it is sent; the shortcut below is what the send
156        // answers wherever `STRING` resolves to `native_string`. For example,
157        // with `::METHOD string` returning `from-string`, `say o`, `'x' o` and
158        // `length(o)` all follow it, and with `::METHOD string CLASS` returning
159        // `KS`, `say .k` prints `KS`.
160        let readable = if matches!(
161            self.receiver_kind(value),
162            Ok(Primitive::Instance { .. } | Primitive::Class(_))
163        ) {
164            let caller = self.caller();
165            match pinned!(
166                self,
167                crate::pinning::PinKind::Conversion,
168                self.send_message(value, STRING, None, &[], caller)
169            )? {
170                // An answer with no string value counts as no answer, and the
171                // default string value stands in for it
172                // (`r[conv.string.answer]`).
173                Some(answered) => match self.string_answer_conversion(answered)? {
174                    Some(text) => text,
175                    None => self.string_value_text(value)?,
176                },
177                None => self.string_value_text(value)?,
178            }
179        } else if self.is_stem_object(value) {
180            self.stem_string_representation(value)?
181        } else {
182            self.string_value_text(value)?
183        };
184        // Gated on the trap rather than raised unconditionally, the shape
185        // `Interp::novalue_raised` describes: an untrapped NOSTRING resumes
186        // with the readable rendering, so a raise nothing can take would
187        // build a condition per rendered object and then throw it away. A
188        // `CALL ON` trap is excluded because `CALL ON NOSTRING` is a parse
189        // error and `CALL ON ANY` is measured not to catch a condition with
190        // no resumption point.
191        if self.trap_for(b"NOSTRING").is_some_and(|trap| !trap.call) {
192            // The object is the condition's `ADDITIONAL`.
193            self.activity.pending_additional = Some(value);
194            return Err(Raised::nostring(&readable).into());
195        }
196        // `::OPTIONS NOSTRING SYNTAX`, and only where nothing trapped: the
197        // trap wins, measured -- `signal on nostring` over `say .array` in
198        // such a file runs the handler, and `signal on syntax` takes 98.973.
199        if self.condition_raises_syntax(b"NOSTRING") {
200            return Err(Raised::nostring_syntax(&readable).into());
201        }
202        let readable = self.text_built(readable);
203        self.roots.activity_mut().push_temp(readable);
204        Ok(readable)
205    }

207    /// What a user `STRING` method's answer reads as: its text under the
208    /// primitive conversion; for a `Stem` subclass instance, its default
209    /// value read the same way, or its name; for any other instance of a
210    /// class defined in Rexx, other than a `Queue`, and for a class object,
211    /// what its `REQUEST('STRING')` answers; else `None`, an answer with no
212    /// string value (`r[conv.string.answer]`).
213    ///
214    /// The default values are followed in a loop rather than by recursion,
215    /// so a chain of any length does not grow the native stack, and a chain
216    /// that comes back to an instance it already passed is an answer with no
217    /// string value (`r[conv.string.answer]`).
218    pub(crate) fn string_answer_conversion(
219        &mut self,
220        mut answered: ObjRef,
221    ) -> Result<Option<Vec<u8>>, Failure> {
222        // Only a default value that is itself a `Stem` subclass instance
223        // needs the set, so the common answer allocates none.
224        let mut passed: Option<FxHashSet<ObjRef>> = None;
225        loop {
226            if let Some(text) = self.primitive_answer_text(answered)? {
227                return Ok(Some(text));
228            }
229            let Some(store) = self.stem_subclass_store(answered) else {
230                break;
231            };
232            let default = match self.heap.get(store).map(|object| &object.body) {
233                Some(Body::Stem { default, .. }) => *default,
234                _ => None,
235            };
236            let Some(default) = default else {
237                return Ok(Some(self.to_text(store).into_owned()));
238            };
239            if !passed
240                .get_or_insert_with(FxHashSet::default)
241                .insert(answered)
242            {
243                return Ok(None);
244            }
245            answered = default;
246        }
247        // A `Stem`, a `VariableReference` or a `StackFrame` has no string
248        // value here, whatever its `REQUEST` answers, and neither has a
249        // `Queue` nor an instance of a subclass of it.
250        let asked = match self.heap.get(answered).map(|object| &object.body) {
251            Some(Body::Instance { .. }) => !super::collection::is_queue(self, answered),
252            _ => self.heap.is_class(answered),
253        };
254        if !asked {
255            return Ok(None);
256        }
257        Ok(match self.required_string_answer(answered)? {
258            Some(RequiredString::Object(text)) => Some(self.to_text(text).into_owned()),
259            Some(RequiredString::Bytes(bytes)) => Some(bytes),
260            None => None,
261        })
262    }

264    /// What a user method's answer reads as under the primitive string
265    /// conversion. A string, a number or an instance of a `String` subclass is
266    /// itself, an array (not a queue) is its items joined by a newline, and a
267    /// buffer is its text. Any other object is `None`: it has no string value
268    /// here, and nothing is sent to it.
269    pub(crate) fn primitive_answer_text(
270        &mut self,
271        answered: ObjRef,
272    ) -> Result<Option<Vec<u8>>, Failure> {
273        let joined = match answered.decode() {
274            Decoded::SmallInt(_) | Decoded::Text(_) => None,
275            Decoded::Heap { .. } if !self.heap.is_class(answered) => {
276                if self.is_string_valued(answered) || self.buffer(answered).is_some() {
277                    None
278                } else {
279                    // An `Array` subclass instance's items live in a store
280                    // its pool holds; a `Queue` has no string value.
281                    let store = super::array::collection_store(self, answered);
282                    if self.array_slots(store).is_none()
283                        || super::collection::is_queue(self, answered)
284                    {
285                        return Ok(None);
286                    }
287                    Some(store)
288                }
289            }
290            _ => return Ok(None),
291        };
292        Ok(Some(match joined {
293            Some(store) => self.string_conversion_array_text(store)?,
294            None => self.to_text(answered).into_owned(),
295        }))
296    }

298    /// Every supplied argument of one builtin call **except the positions
299    /// `crate::builtin::raw_argument_positions` exempts for `name`**, in
300    /// position order, through the required-string protocol -- `None` when the
301    /// protocol cannot change any of them, so the caller passes its own list
302    /// on.
303    pub(crate) fn required_string_arguments(
304        &mut self,
305        name: &'static [u8],
306        args: &[Option<ObjRef>],
307    ) -> Result<Option<Vec<Option<ObjRef>>>, Failure> {
308        if !self.reqstr_armed {
309            return Ok(None);
310        }
311        // **Scanned before anything is built.** `None` means "the arguments
312        // stand as they are", so a call whose every argument is already its
313        // own string value can answer that instead of allocating a vector to
314        // hold copies of what the caller already has. An armed interpreter
315        // otherwise allocated one `Vec` per builtin call for the whole run,
316        // and being armed is not rare: any instance of a Rexx class arms it.
317        //
318        // A heap string or number is its own string value too, which is
319        // [`Interp::required_string_value`]'s answer for it; that answer roots
320        // it, so it is rooted here as well. Only a position whose conversion
321        // could answer another object makes the list worth copying, and only
322        // then is the exemption table asked: an exempt position copied
323        // unconverted is the list the caller had.
324        let stands = args.iter().flatten().all(|&value| match value.decode() {
325            Decoded::SmallInt(_) | Decoded::Text(_) => true,
326            _ if matches!(
327                self.heap.get(value).map(|object| &object.body),
328                Some(Body::Text { .. } | Body::Num { .. })
329            ) =>
330            {
331                self.roots.activity_mut().push_temp(value);
332                true
333            }
334            _ => false,
335        });
336        if stands {
337            return Ok(None);
338        }
339        let raw = crate::builtin::raw_argument_positions(name);
340        let mut converted = Vec::with_capacity(args.len());
341        for (index, argument) in args.iter().enumerate() {
342            let position = index + 1;
343            converted.push(match argument {
344                None => None,
345                Some(value) if raw.contains(&position) => Some(*value),
346                Some(value) => Some(self.required_string_value(*value)?),
347            });
348        }
349        Ok(Some(converted))
350    }

352    /// The protocol's conversion limbs, answered without building anything --
353    /// [`Interp::string_conversion`]'s own body with the materialisation left
354    /// to the caller.
355    fn required_string_answer(&mut self, value: ObjRef) -> Result<Option<RequiredString>, Failure> {
356        Ok(match self.classify_string_conversion(value) {
357            StringConversion::Object(text) => Some(RequiredString::Object(text)),
358            StringConversion::Bytes(bytes) => Some(RequiredString::Bytes(bytes)),
359            StringConversion::Array(store) => Some(RequiredString::Bytes(
360                self.string_conversion_array_text(store)?,
361            )),
362            StringConversion::None => None,
363            StringConversion::MakeString => {
364                // What `makeString` answered has to be a real string, and an
365                // object that is not one converts here or counts as no answer
366                // at all. With `say .K` and a class-side `makeString`, `return
367                // 5` prints `5`, `return .Object~superClasses` prints the empty
368                // line an empty array joins to, and `return .array` prints `The
369                // K class` -- the third fell back.
370                // A string or a number is itself; any other answer is read by
371                // the primitive conversion, and one that has no string value
372                // there, a `Stem` or a `VariableReference` included, is no
373                // answer (`r[conv.string.answer]`).
374                match self.send_make_string(value)? {
375                    Some(answered) if self.is_plain_string(answered) => {
376                        Some(RequiredString::Object(answered))
377                    }
378                    Some(answered) => self
379                        .primitive_answer_text(answered)?
380                        .map(RequiredString::Bytes),
381                    None => None,
382                }
383            }
384        })
385    }

387    /// Whether `value` is a string or a number, which is its own string value.
388    fn is_plain_string(&self, value: ObjRef) -> bool {
389        match value.decode() {
390            Decoded::SmallInt(_) | Decoded::Text(_) => true,
391            Decoded::Heap { .. } => matches!(
392                self.heap.get(value).map(|object| &object.body),
393                Some(Body::Text { .. } | Body::Num { .. })
394            ),
395            _ => false,
396        }
397    }

399    /// The protocol's conversion limbs alone, with **no** `~string` fallback
400    /// and **no** NOSTRING condition. `None` is `.nil`.
401    pub(crate) fn string_conversion(&mut self, value: ObjRef) -> Result<Option<ObjRef>, Failure> {
402        Ok(match self.required_string_answer(value)? {
403            Some(RequiredString::Object(text)) => Some(text),
404            Some(RequiredString::Bytes(bytes)) => Some(self.text_built(bytes)),
405            None => {
406                self.conversion_subject(value)?;
407                None
408            }
409        })
410    }

412    /// [`Interp::string_conversion`] with the `REQUEST` traceback line on it,
413    /// for a caller inside a native method's own activation -- see
414    /// [`Interp::blame_request`] for why the two frames stack there.
415    pub(crate) fn blamed_string_conversion(
416        &mut self,
417        value: ObjRef,
418    ) -> Result<Option<ObjRef>, Failure> {
419        let converted = self.string_conversion(value);
420        if converted.is_err() {

422        }
423        converted

426    /// Which limb of the protocol answers for `value`.
427    fn classify_string_conversion(&mut self, value: ObjRef) -> StringConversion {
428        let redirect = match value.decode() {
429            Decoded::Nil => return self.make_string_or_none(value),
430            // A string and a number are their own string value.
431            Decoded::SmallInt(_) | Decoded::Text(_) => {
432                return StringConversion::Object(value);
433            }
434            Decoded::Heap { .. } if self.heap.is_class(value) => {
435                return self.make_string_or_none(value);
436            }
437            Decoded::Heap { .. } => match self.heap.get(value) {
438                // A handle whose slot is gone, which `Interp::to_text` turns
439                // into its own tripwire. Answering the value keeps that the
440                // one report rather than adding a second.
441                None => return StringConversion::Object(value),
442                Some(object) => match &object.body {
443                    Body::Text { .. } | Body::Num { .. } => {
444                        return StringConversion::Object(value);
445                    }
446                    // A stem's string value forwards to the default value, and
447                    // a stem holding none is its own derived name, which is
448                    // what `to_text` renders for it.
449                    Body::Stem { default: None, .. } => {
450                        return StringConversion::Object(value);
451                    }
452                    Body::Stem {
453                        default: Some(default),
454                        ..
455                    } => Some(*default),
456                    // An array's string value, the items joined by a newline
457                    // -- **not** its `string`, which is `an Array`.
458                    Body::Array { .. } => None,
459                    // **The referenced value's string value**: a string
460                    // value is what `request('STRING')` answers (rexxref
461                    // provide.xml, "Required String Values"), and a
462                    // reference's `request` forwards to the referenced
463                    // variable (utilityclasses.xml, VariableReference
464                    // `request`). An unset variable is its own name.
465                    Body::VarRef(_) => return self.reference_conversion(value),
466                    Body::Instance { .. } => return self.instance_conversion(value),
467                    _ => return self.make_string_or_none(value),
468                },
469            },
470        };
471        match redirect {
472            Some(default) => self.classify_string_conversion(default),
473            None => StringConversion::Array(value),
474        }
475    }

477    /// [`Interp::classify_string_conversion`]'s array arm, split out so the
478    /// borrow of the heap object above ends before the join runs.
479    fn string_conversion_array_text(&mut self, value: ObjRef) -> Result<Vec<u8>, Failure> {
480        let items = self.array_slots_of(value).unwrap_or_default();
481        self.array_string(&items, b"\n")
482    }

484    /// [`Interp::classify_string_conversion`]'s instance arm, out of line so
485    /// that the arms a string takes stay as small as they were: a `Stem`
486    /// subclass instance answers as its stem, which is where `request`
487    /// forwards, so as the value its stem's chain ends at; any other instance
488    /// as [`Interp::make_string_or_none`] finds it. A chain that leads back
489    /// has no string value here; [`Interp::conversion_subject`] refuses it.
490    #[inline(never)]
491    fn instance_conversion(&mut self, value: ObjRef) -> StringConversion {
492        // An instance of a `String` subclass is read by its own characters,
493        // and neither `request`, `makeString` nor `string` is sent to it
494        // (Deviation 109).
495        if let Some(Body::Instance {
496            native: Some(state),
497            ..
498        }) = self.heap.get(value).map(|object| &object.body)
499            && let interrexx_core::NativeState::Text(bytes) = &**state
500        {
501            return StringConversion::Bytes(bytes.to_vec());
502        }
503        match self.stem_subclass_store(value) {
504            Some(store) => self.chain_conversion(store),
505            None => self.make_string_or_none(value),
506        }
507    }

509    /// [`Interp::classify_string_conversion`] of the value the chain of links
510    /// starting at `value` ends at, or for a chain ending at an unset
511    /// variable that variable's name. A chain that leads back has no string
512    /// value here.
513    fn chain_conversion(&mut self, value: ObjRef) -> StringConversion {
514        match self.chain_end(value) {
515            Referent::Value(end) => self.classify_string_conversion(end),
516            Referent::Unset(last) => {
517                let name = self
518                    .as_variable_reference(last.unwrap_or(value))
519                    .map(|reference| reference.name.to_vec())
520                    .unwrap_or_default();
521                StringConversion::Bytes(name)
522            }
523            Referent::Cycle => StringConversion::None,
524        }
525    }

527    /// Whether the receiver's own behaviour answers `MAKESTRING`.
528    fn make_string_or_none(&mut self, value: ObjRef) -> StringConversion {
529        match self.lookup(value, MAKESTRING, None) {
530            Some(_) => StringConversion::MakeString,
531            None if self.has_request_override(value) => StringConversion::MakeString,
532            None => StringConversion::None,
533        }
534    }

536    /// Whether the receiver has a `request` method that is not the primitive
537    /// one, which sends `makeString` itself: such a method is what a string
538    /// value is asked of (r[conv.string.object]).
539    fn has_request_override(&mut self, value: ObjRef) -> bool {
540        let Some(resolution) = self.lookup(value, REQUEST, None) else {
541            return false;
542        };
543        self.object_model().natives.get(resolution.method).is_none()
544    }

546    /// [`Interp::classify_string_conversion`] of a variable reference:
547    /// [`Interp::chain_conversion`], the limb of the value its chain of
548    /// references and stem defaults ends at. A cyclic reference has no string
549    /// value here; [`Interp::conversion_subject`] refuses it.
550    #[cold]
551    #[inline(never)]
552    fn reference_conversion(&mut self, value: ObjRef) -> StringConversion {
553        self.chain_conversion(value)
554    }

556    /// The object a conversion of `value` falls back on: `value`, or for a
557    /// variable reference the value its chain of references ends at. A chain
558    /// that leads back to itself anywhere, through references or stem
559    /// defaults, never ends at a value the conversion could use, and is 11.1,
560    /// the error unbounded recursion answers.
561    fn conversion_subject(&self, value: ObjRef) -> Result<ObjRef, Failure> {
562        if self.chain_is_cyclic(value) {
563            return Err(Raised::insufficient_stack().into());
564        }
565        match self.referent(value) {
566            Referent::Value(referenced) => Ok(referenced),
567            Referent::Unset(_) | Referent::Cycle => Ok(value),
568        }
569    }

571    /// Sends `makeString` on the receiver's behalf.
572    fn send_make_string(&mut self, receiver: ObjRef) -> Result<Option<ObjRef>, Failure> {
573        if self.has_request_override(receiver) {
574            // The receiver's own `request` is asked for the string value, as
575            // the primitive `request` would send `makeString`.
576            let wanted = self.text(STRING);
577            self.roots.activity_mut().push_temp(wanted);
578            let caller = self.caller();
579            return pinned!(
580                self,
581                crate::pinning::PinKind::Conversion,
582                self.send_message(receiver, REQUEST, None, &[Some(wanted)], caller)
583            );
584        }
585        self.send_for_conversion(receiver, MAKESTRING)
586    }
587
588    /// Sends `name`, with no arguments, on the receiver's behalf.
589    pub(crate) fn send_for_conversion(
590        &mut self,
591        receiver: ObjRef,
592        name: &[u8],
593    ) -> Result<Option<ObjRef>, Failure> {
594        // The sending side is the frame the conversion happens in, not the
595        // conversion itself: the private check asks the top frame's receiver,
596        // and the string request runs no frame of its own that could answer
597        // that question differently.
598        let caller = self.caller();
599        pinned!(
600            self,
601            crate::pinning::PinKind::Conversion,
602            self.send_message(receiver, name, None, &[], caller)
603        )
604    }
605
606    /// The traceback line the `REQUEST` activation contributes when a
607    /// `makeString` reached through the protocol fails.
608    fn blame_request(&mut self, value: ObjRef) {
609        let class = self.text(b"STRING");
610        self.roots.activity_mut().push_temp(class);
611        self.blame_native_method(b"REQUEST", "Object", value, &[Some(class)]);
612    }
613}

615/// An argument of a method that the method needs as text, converted through the
616/// required-string protocol, or 88.909 for a value that has no string value at
617/// all.
618/// ```text
619/// 'abc'~hasMethod(5)                oracle `0` rc 0    a number has one
620/// 'abc'~hasMethod(a.)               oracle `0` rc 0    an unset stem is its own name
621/// 'abc'~hasMethod(.nil)             oracle 88.909 rc 168
622/// 'abc'~hasMethod(.String)          oracle 88.909 rc 168
623/// 'abc'~hasMethod(.environment)     oracle 88.909 rc 168
624/// a. = .array; 'abc'~hasMethod(a.)  oracle 88.909 rc 168
625/// a. = .nil;   'abc'~hasMethod(a.)  oracle 88.909 rc 168
626/// ```
627pub(super) fn required_string_argument(
628    interp: &mut Interp,
629    value: ObjRef,
630    position: usize,
631) -> Result<ObjRef, Failure> {
632    match interp.blamed_string_conversion(value)? {
633        Some(text) => Ok(text),
634        None => Err(Raised::argument_needs_a_string_value(position).into()),
635    }
636}
637
638/// [`required_string_argument`] for an argument 88.909 names rather than
639/// numbers.
640pub(super) fn required_string_named_argument(
641    interp: &mut Interp,
642    value: ObjRef,
643    argument: &'static str,
644) -> Result<ObjRef, Failure> {
645    match interp.blamed_string_conversion(value)? {
646        Some(text) => Ok(text),
647        None => Err(Raised::named_argument_needs_a_string_value(argument).into()),
648    }
