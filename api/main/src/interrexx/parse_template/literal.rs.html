1use super::*;
2use crate::plan::Plan;
3use interrexx_parse::{CodeBody, ExprView, SymbolTable};
4
5/// The slot writes for a literal source and a template whose positions and
6/// target values are known independently of the running activation.
7pub(crate) fn literal_writes(
8    body: &CodeBody,
9    parse: &Parse,
10    plan: &Plan,
11    symbols: &SymbolTable,
12) -> Option<Vec<(u32, ObjRef)>> {
13    let ParseSource::Value(Some(expression)) = &parse.source else {
14        return None;
15    };
16    let ExprView::Literal(bytes) = body.expr(*expression).kind() else {
17        return None;
18    };
19    if parse.upper || parse.lower || crate::value::canonical_small_int(bytes).is_some() {
20        return None;
21    }
22    let mut cursor = Cursor::new(bytes.len());
23    let mut writes = Vec::new();
24    for trigger in &parse.template {
25        let trigger = trigger.as_ref()?;
26        if trigger.kind == TriggerKind::End {
27            cursor.move_to_end();
28        } else {
29            let operand = match body.expr(trigger.value?).kind() {
30                ExprView::Literal(operand) => operand,
31                ExprView::Constant(id) => symbols.name(id).as_bytes(),
32                _ => return None,
33            };
34            match trigger.kind {
35                TriggerKind::String => cursor.search(bytes, operand),
36                TriggerKind::Mixed => cursor.caseless_search(bytes, operand),
37                kind => {
38                    // Every one-digit position fits even NUMERIC DIGITS 1
39                    // (r[parse.patterns.number]). Wider operands are checked
40                    // against the settings in force when the clause runs.
41                    let &[digit] = operand else { return None };
42                    if !digit.is_ascii_digit() {
43                        return None;
44                    }
45                    let offset = usize::from(digit - b'0');
46                    match kind {
47                        TriggerKind::Plus => cursor.forward(offset),
48                        TriggerKind::Minus => cursor.backward(offset),
49                        TriggerKind::Absolute => cursor.absolute(offset),
50                        TriggerKind::PlusLength => cursor.forward_length(offset),
51                        TriggerKind::MinusLength => cursor.backward_length(offset),
52                        _ => return None,
53                    }
54                }
55            }
56        }
57        for (index, target) in trigger.targets.iter().enumerate() {
58            let piece = if index + 1 == trigger.targets.len() {
59                cursor.remainder()
60            } else {
61                cursor.next_word(bytes)
62            };
63            let Some(target) = target else { continue };
64            let id = body.expr(*target).variable()?;
65            let slot = u32::try_from(plan.slot_for_symbol(id)?).ok()?;
66            // Inline pieces can live in the plan without adding GC roots.
67            writes.push((slot, ObjRef::inline_text(&bytes[piece])?));
68        }
69    }
70    (!writes.is_empty()).then_some(writes)
71}
