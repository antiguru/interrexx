5    /// Evaluates one trigger's operand, if it has one, and moves the cursor.
6    #[inline(always)]
7    pub(super) fn apply_trigger(
8        &mut self,
9        code: &Code<'_>,
10        trigger: &ParseTrigger,
11        source: &SourceText,
12        cursor: &mut Cursor,
13        indent: usize,
14    ) -> Result<(), Failure> {
15        // `End` is the only kind with no operand, so every other arm below
16        // can ask for one. A missing operand on a kind that needs one is not
17        // representable from a program that parsed -- `parse_template`
18        // (`interrexx-parse`) fills `value` on every trigger it builds except
19        // `End`, and refuses a trigger with nothing after it (38.901) -- and
20        // it is reported as the gap it would be rather than defaulted to a
21        // position.
22        let operand = match trigger.kind {
23            TriggerKind::End => {
24                cursor.move_to_end();
25                return Ok(());
26            }
27            _ => match &trigger.value {
28                Some(operand) => code.body.expr(*operand),
29                None => return Err(Loud::parse_trigger_operand().into()),
30            },
31        };
32        // Read before the operand only to gate the literal search, which
33        // evaluates nothing; the `>>>` line below reads it again, after an
34        // operand that may have changed it (`(trace('R'))`).
35        let results = self.trace_mode().results;
36        // **A literal pattern is searched for as the bytes it spells** when
37        // no line will show it: its value would be a string holding exactly
38        // those bytes, which is its own string value, so building it changes
39        // nothing a search can see.
40        if let (TriggerKind::String | TriggerKind::Mixed, ExprView::Literal(needle)) =
41            (trigger.kind, operand.kind())
42            && !results
43            && !self.tracing_intermediates()
44        {
45            let string = source.bytes(self);
46            if trigger.kind == TriggerKind::String {
47                cursor.search(string, needle);
48            } else {
49                cursor.caseless_search(string, needle);
50            }
51            return Ok(());
52        }
53        let value = self.trigger_operand(code, operand)?;
54        self.apply_trigger_value(trigger, value, source, cursor, indent)
55    }

57    /// [`Interp::apply_trigger`] once the operand of a trigger that has one is
58    /// evaluated: its `>>>` line, its conversion and the cursor's movement.
59    #[inline(always)]
60    pub(super) fn apply_trigger_value(
61        &mut self,
62        trigger: &ParseTrigger,
63        value: ObjRef,
64        source: &SourceText,
65        cursor: &mut Cursor,
66        indent: usize,
67    ) -> Result<(), Failure> {
68        self.roots.activity_mut().push_temp(value);
69        // `>>>` for the operand, after `eval`'s own `>L>`/`>V>` and before
70        // the conversion that can fail: `integerTrigger` traces and only then
71        // converts, so the operand's own value line is emitted even on the
72        // 26.4 path. **Copied only where a copy is needed**, which is neither
73        // of the two arms below. `trace_result` returns at once unless
74        // `results` is on, and a search reads the bytes where they already
75        // are: `Cursor` is a local of `exec_parse` rather than a field, so a
76        // shared borrow of the value can be live while it is written to. The
77        // unconditional `to_vec` here was 5.2% of every allocation
78        // `samples/rexxcps.rex` made -- one per trigger, on a program whose
79        // templates are mostly numeric triggers that never look at the bytes
80        // at all.
81        if self.trace_mode().results {
82            let rendered = self.traced_string_value_text(value)?;
83            self.trace_result(indent, &rendered);
84        }
85        match trigger.kind {
86            // Convert before borrowing the source because conversion can allocate.
87            // Both source and needle bytes are borrowed only for the search.
88            TriggerKind::String => {
89                let needle = self.required_string_value(value)?;
90                // The operand's earlier root still protects an unchanged answer.
91                if needle != value {
92                    self.roots.activity_mut().push_temp(needle);
93                }
94                let needle = self.render(needle);
95                cursor.search(source.bytes(self), needle.text(self));
96                Ok(())
97            }
98            TriggerKind::Mixed => {
99                let needle = self.required_string_value(value)?;
100                // A different answer needs its own root before rendering.
101                if needle != value {
102                    self.roots.activity_mut().push_temp(needle);
103                }
104                let needle = self.render(needle);
105                cursor.caseless_search(source.bytes(self), needle.text(self));
106                Ok(())
107            }
108            _ => {
109                // 26.4, and the conversion is bounded by the **active**
110                // `NUMERIC DIGITS` rather than by a fixed width. For example
111                // `numeric digits 2; parse value d with p +(100) q` is 26.4, as
112                // are a fractional operand, a non-numeric one and a negative
113                // one. The substitution is the operand's own rendering --
114                // `found "1E2"` for `+(1e2)`, which is D15's rule showing
115                // through rather than a re-rendering here.
116                //
117                // **An object that is not a string is read through its string
118                // value** (`requested_number`, `numberValue`'s own
119                // protocol), so a `MutableBuffer` of `3` is the position 3.
120                let offset = match self.whole_nonneg(value) {
121                    Some(offset) => Some(offset),
122                    None => self.whole_nonneg_requested(value)?,
123                };
124                let Some(offset) = offset else {
125                    // The one place the operand's own rendering is still
126                    // owned, and it is the raise's substitution.
127                    return Err(Raised::syntax(26, 4, vec![self.to_text(value).to_vec()]).into());
128                };
129                // Clamped rather than checked: an offset past `usize` is
130                // indistinguishable from one past the string's own end, and
131                // every operation below treats those the same way.
132                let offset = usize::try_from(offset).unwrap_or(usize::MAX);
133                match trigger.kind {
134                    TriggerKind::Plus => cursor.forward(offset),
135                    TriggerKind::Minus => cursor.backward(offset),
136                    TriggerKind::Absolute => cursor.absolute(offset),
137                    TriggerKind::PlusLength => cursor.forward_length(offset),
138                    TriggerKind::MinusLength => cursor.backward_length(offset),
139                    // Handled above; the operand kinds and the movement
140                    // kinds partition `TriggerKind`, and this arm is what
141                    // keeps that a compile-time exhaustive match rather than
142                    // an assumption.
143                    TriggerKind::End | TriggerKind::String | TriggerKind::Mixed => {}
144                }
145                Ok(())
146            }
147        }
148    }

150    /// A trigger's operand value: [`Interp::eval_in_parse`], with a leaf
151    /// read directly where `TRACE I` will not show it. A leaf evaluates no
152    /// sub-expression, so the evaluation-depth bookkeeping `eval_in_parse`
153    /// wraps around it has nothing to guard.
154    #[inline(always)]
155    fn trigger_operand(
156        &mut self,
157        code: &Code<'_>,
158        operand: ExprRef<'_>,
159    ) -> Result<ObjRef, Failure> {
160        if !self.tracing_intermediates() {
161            match operand.kind() {
162                ExprView::Literal(bytes) => return Ok(self.literal(bytes)),
163                ExprView::Constant(id) => {
164                    return Ok(self.literal(code.symbols.name(id).as_bytes()));
165                }
166                ExprView::Variable(id) => {
167                    return self.read_symbol(code, None, crate::eval::SymbolRead::Simple, id, None);
168                }
169                _ => {}
170            }
171        }
172        self.eval_in_parse(code, operand)
173    }
