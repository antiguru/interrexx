1//! The template walk as the compiled stream drives it. [`Op::ParseBegin`]
2//! takes the first parse string and walks the entries up to the first whose
3//! operand the stream evaluates, and each [`Op::ParseRun`] walks one such
4//! entry, with its operand from a register, and the entries after it up to
5//! the next. Inside one op the walk is the template engine's own loop, with
6//! the cursor in locals; between two ops it lives in registers of the
7//! clause's own frame, so it is rooted, it is the frame's to move and to drop,
8//! and a nested activation that parses has a walk of its own.
9//!
10//! [`Op::ParseBegin`]: crate::ir::Op::ParseBegin
11//! [`Op::ParseRun`]: crate::ir::Op::ParseRun
12
13use super::*;
14use crate::ir::ParseOpening;
15use interrexx_core::RegFrame;
16use interrexx_parse::Instruction;
17use interrexx_parse::InstructionKind;

19/// How many registers one compiled walk keeps, from its base.
20pub(crate) const WALK_REGISTERS: u16 = 5;
21
22/// The parse string: a string that is its own string value.
23const SOURCE: u16 = 0;
24/// The number the parse string renders, which a piece covering the whole
25/// string assigns, or `.nil`.
26const ORIGIN: u16 = 1;
27/// Where the last match began and one past where it ended, as small
28/// integers: all of [`Cursor`] an op hands the next. The section and the word
29/// position are not filed, because the entry a run starts at is a trigger,
30/// and every movement and search sets them afresh from these two.
31const PATTERN_START: u16 = 2;
32const PATTERN_END: u16 = 3;
33/// `PARSE ARG`: the argument the next template reads.
34const NEXT_ARGUMENT: u16 = 4;
35
36/// A position or count as the small integer a register holds. A string is
37/// far shorter than the range a small integer covers.
38fn position(at: usize) -> ObjRef {
39    i64::try_from(at)
40        .ok()
41        .and_then(ObjRef::small_int)
42        .unwrap_or(ObjRef::NIL)
43}
44
45/// The position a register filed by [`position`] holds.
46fn at(registers: RegFrame<'_>, base: u16, offset: u16) -> usize {
47    match registers.get(base + offset).decode() {
48        Decoded::SmallInt(at) => usize::try_from(at).unwrap_or(0),
49        _ => 0,
50    }
51}
52
53/// The `PARSE` a walk op's clause is.
54fn parse_of(clause: &Instruction) -> Result<&Parse, Failure> {
55    match &clause.kind {
56        InstructionKind::Parse(parse)
57        | InstructionKind::Arg(parse)
58        | InstructionKind::Pull(parse) => Ok(parse),
59        kind => Err(Loud::instruction(kind).into()),
60    }
61}

63/// What the opening found: the walk's parse string, in place, with the
64/// number a whole-string piece assigns, or the clause for the template
65/// engine with the source value already in hand.
66enum Opened {
67    Walk(ObjRef, SourceText, Option<ObjRef>),
68    Engine(Option<ObjRef>),
69}
70
71/// One walk in hand: the parse string, the cursor over it, and for `PARSE
72/// ARG` the argument the next template reads.
73struct Walk {
74    value: ObjRef,
75    source: SourceText,
76    cursor: Cursor,
77    next_argument: usize,
78}
79
80impl Interp {
81    /// `value` as a parse string read in place, and the number a whole-string
82    /// piece assigns: what [`Interp::source_text`] answers for a string that
83    /// is its own string value and needs no rendering, or `None` for any
84    /// other value. Asking sends nothing and changes nothing, so a `None` can
85    /// still hand the clause to [`Interp::exec_parse`].
86    fn plain_parse_source(&self, value: ObjRef) -> Option<(SourceText, Option<ObjRef>)> {
87        match value.decode() {
88            Decoded::Text(inline) => Some((SourceText::Inline(inline), None)),
89            Decoded::SmallInt(number) => crate::value::small_int_inline_text(number)
90                .map(|inline| (SourceText::Inline(inline), Some(value))),
91            Decoded::Heap { .. } => self.heap.body_text(value).map(|bytes| {
92                (
93                    SourceText::Text {
94                        value,
95                        length: bytes.len(),
96                    },
97                    None,
98                )
99            }),
100            _ => None,
101        }
102    }
103
104    /// A walk over `value`, which [`Interp::plain_parse_source`] took, from
105    /// its origin.
106    fn walk_over(&self, value: ObjRef, next_argument: usize) -> Walk {
107        let (source, origin) = self
108            .plain_parse_source(value)
109            .unwrap_or((SourceText::Owned(Vec::new()), None));
110        let mut cursor = Cursor::new(source.len());
111        cursor.origin = origin;
112        Walk {
113            value,
114            source,
115            cursor,
116            next_argument,
117        }
118    }
119
120    /// The argument `PARSE ARG` reads for template `at`, as a parse string:
121    /// an omitted or absent one is the null string.
122    fn argument_parse_value(&self, at: usize) -> ObjRef {
123        match self.activity.call_context.arguments.get(at) {
124            Some(Some(argument)) => *argument,
125            Some(None) | None => ObjRef::inline_text(b"").unwrap_or(ObjRef::NIL),
126        }
127    }
128
129    /// Files `walk` in the registers from `base`, for the op that goes on
130    /// with it.
131    fn file_walk(registers: RegFrame<'_>, base: u16, walk: &Walk) {
132        let cursor = &walk.cursor;
133        registers.set(base + SOURCE, walk.value);
134        registers.set(base + ORIGIN, cursor.origin.unwrap_or(ObjRef::NIL));
135        registers.set(base + PATTERN_START, position(cursor.pattern_start));
136        registers.set(base + PATTERN_END, position(cursor.pattern_end));
137        registers.set(base + NEXT_ARGUMENT, position(walk.next_argument));
138    }
139
140    /// The walk [`Interp::file_walk`] filed from `base`.
141    fn filed_walk(&self, registers: RegFrame<'_>, base: u16) -> Walk {
142        let value = registers.get(base + SOURCE);
143        let mut walk = self.walk_over(value, at(registers, base, NEXT_ARGUMENT));
144        let origin = registers.get(base + ORIGIN);
145        walk.cursor.origin = (origin != ObjRef::NIL).then_some(origin);
146        walk.cursor.pattern_start = at(registers, base, PATTERN_START);
147        walk.cursor.pattern_end = at(registers, base, PATTERN_END);
148        walk
149    }
150
151    /// Template entries `from..to` of `parse` applied to `walk`, the first
152    /// with `operand` where the stream evaluated it: a trigger and then its
153    /// targets, or a comma, which steps to the next parse string
154    /// (`r[parse.multiple.comma]`).
155    fn walk_entries(
156        &mut self,
157        code: &Code<'_>,
158        parse: &Parse,
159        from: usize,
160        to: usize,
161        mut operand: Option<ObjRef>,
162        walk: &mut Walk,
163    ) -> Result<(), Failure> {
164        let indent = self.activity.clause_state.current_value_indent;
165        for entry in parse.template.get(from..to).unwrap_or(&[]) {
166            let Some(trigger) = entry else {
167                // The next argument for `PARSE ARG`, and the null string for
168                // every other source, parsed from its first character.
169                let next = match parse.source {
170                    ParseSource::Arg => {
171                        walk.next_argument += 1;
172                        self.argument_parse_value(walk.next_argument - 1)
173                    }
174                    _ => ObjRef::inline_text(b"").unwrap_or(ObjRef::NIL),
175                };
176                *walk = self.walk_over(next, walk.next_argument);
177                continue;
178            };
179            match operand.take() {
180                Some(value) => self.apply_trigger_value(
181                    trigger,
182                    value,
183                    &walk.source,
184                    &mut walk.cursor,
185                    indent,
186                )?,
187                None => {
188                    self.apply_trigger(code, trigger, &walk.source, &mut walk.cursor, indent)?;
189                }
190            }
191            self.assign_targets(code, trigger, &walk.source, &mut walk.cursor, indent)?;
192        }
193        Ok(())
194    }

196    /// [`crate::ir::Op::ParseBegin`]: takes the first parse string of the
197    /// clause and walks its entries up to `to`, and answers whether the
198    /// compiled walk took the clause. It does not where a `>K>`, `>>>` or
199    /// `>=>` line will be traced, under a fold, for a source read as a line
200    /// or built by the interpreter, for `PARSE VAR` into words, or where a
201    /// parse string is not a string that is its own string value -- a number
202    /// to render, an object to ask, a stem or a compound to read -- and
203    /// [`Interp::exec_parse`] then runs the clause whole, with nothing done
204    /// that it would do again.
205    #[inline(never)]
206    #[expect(
207        clippy::too_many_arguments,
208        reason = "the op's own fields, beside the clause and the frame they index"
209    )]
210    pub(crate) fn parse_begin(
211        &mut self,
212        code: &Code<'_>,
213        clause: &Instruction,
214        src: u16,
215        base: u16,
216        to: u16,
217        opening: ParseOpening,
218        strings: u8,
219        registers: RegFrame<'_>,
220    ) -> Result<bool, Failure> {
221        let parse = parse_of(clause)?;
222        let evaluated = (src != crate::ir::Op::ARG_OMITTED).then(|| registers.get(src));
223        if opening == ParseOpening::Literal {
224            let traced = self.traced_mode();
225            if traced.results || traced.intermediates {
226                self.exec_parse(code, parse, evaluated)?;
227                return Ok(false);
228            }
229            return Ok(true);
230        }
231        let (first, source, origin) =
232            match self.walked_source(code, parse, evaluated, opening, strings)? {
233                Opened::Walk(first, source, origin) => (first, source, origin),
234                Opened::Engine(evaluated) => {
235                    self.exec_parse(code, parse, evaluated)?;
236                    return Ok(false);
237                }
238            };
239        let mut cursor = Cursor::new(source.len());
240        cursor.origin = origin;
241        let mut walk = Walk {
242            value: first,
243            source,
244            cursor,
245            next_argument: 1,
246        };
247        let to = usize::from(to);
248        self.walk_entries(code, parse, 0, to, None, &mut walk)?;
249        if to < parse.template.len() && base != crate::ir::Op::ARG_OMITTED {
250            Self::file_walk(registers, base, &walk);
251        }
252        Ok(true)
253    }

255    /// The first parse string the compiled walk takes, read in place, and
256    /// the number a whole-string piece assigns; or, for a clause it hands to
257    /// the template engine, the source value already in hand, which the engine
258    /// takes rather than evaluating or reading it again. The engine has the
259    /// clause where the compiler handed it over, where a line will be traced,
260    /// and where the parse string is not its own string value.
261    fn walked_source(
262        &mut self,
263        code: &Code<'_>,
264        parse: &Parse,
265        evaluated: Option<ObjRef>,
266        opening: ParseOpening,
267        strings: u8,
268    ) -> Result<Opened, Failure> {
269        if opening == ParseOpening::Engine {
270            return Ok(Opened::Engine(evaluated));
271        }
272        let traced = self.traced_mode();
273        if traced.results || traced.intermediates {
274            return Ok(Opened::Engine(evaluated));
275        }
276        let (first, hand_over) = match (opening, &parse.source) {
277            // Held by the register the stream evaluated it into, for the
278            // whole clause.
279            (ParseOpening::Value, _) => (
280                evaluated.unwrap_or_else(|| ObjRef::inline_text(b"").unwrap_or(ObjRef::NIL)),
281                evaluated,
282            ),
283            // A simple name, which the compiler judged: its read is the
284            // variable's value and `NOVALUE` for an unset one, whose
285            // `.local~novalue` hook the read has then sent. Read once, here,
286            // before any target is assigned, so a target that is the source
287            // variable does not change what is parsed (`r[parse.index.order]`);
288            // and rooted for the clause, because that target is then all that
289            // held it, and the pieces built after it are allocations.
290            (ParseOpening::Var, ParseSource::Var(id)) => {
291                let (value, novalue) = self.read(code, *id);
292                self.novalue_check(novalue, value)?;
293                self.roots.activity_mut().push_temp(value);
294                (value, Some(value))
295            }
296            // Every argument a template of this clause will read, judged
297            // before the first is taken, since a later one that needs
298            // rendering could no longer hand the clause back. Held by the
299            // call's own argument list.
300            (ParseOpening::Arg, _) => {
301                if (0..usize::from(strings)).any(|at| {
302                    self.plain_parse_source(self.argument_parse_value(at))
303                        .is_none()
304                }) {
305                    return Ok(Opened::Engine(evaluated));
306                }
307                (self.argument_parse_value(0), evaluated)
308            }
309            _ => return Ok(Opened::Engine(evaluated)),
310        };
311        Ok(match self.plain_parse_source(first) {
312            Some((source, origin)) => Opened::Walk(first, source, origin),
313            None => Opened::Engine(hand_over),
314        })
315    }
316
317    /// [`crate::ir::Op::ParseRun`]: template entries `from..to` of the
318    /// clause applied to the walk filed from `base`, entry `from` with its
319    /// operand in register `operand`.
320    #[inline(never)]
321    #[expect(
322        clippy::too_many_arguments,
323        reason = "the op's own fields, beside the clause and the frame they index"
324    )]
325    pub(crate) fn parse_run(
326        &mut self,
327        code: &Code<'_>,
328        clause: &Instruction,
329        base: u16,
330        from: u16,
331        to: u16,
332        operand: u16,
333        registers: RegFrame<'_>,
334    ) -> Result<(), Failure> {
335        let parse = parse_of(clause)?;
336        let mut walk = self.filed_walk(registers, base);
337        let value = registers.get(operand);
338        let to = usize::from(to);
339        self.walk_entries(code, parse, usize::from(from), to, Some(value), &mut walk)?;
340        if to < parse.template.len() {
341            Self::file_walk(registers, base, &walk);
342        }
343        Ok(())
344    }
345}
