1//! `ClassRegistry`: class objects, and the name-to-class registry, over
2//! [`ClassGraph`](crate::ClassGraph).
3
4use crate::class_graph::{ClassGraph, ClassKind, Epoch, InheritRefusal};
5use crate::method_dict::MethodSlot;
6use interrexx_core::MethodId;
7use interrexx_core::NameMap;
8use interrexx_core::{BehaviourHandle, ObjRef};

10/// Every map here is a [`NameMap`], for the reason
11/// [`crate::MethodDict`] states: they are all written once per class while
12/// the library is being built, before a program's first clause runs.
13pub struct ClassRegistry {
14    graph: ClassGraph,
15    next_method: u32,
16    /// Id string as declared (`~id`), keyed by identity.
17    names: NameMap<ObjRef, String>,
18    /// `~defaultName`, keyed by identity -- see [`ClassRegistry::default_name`].
19    default_names: NameMap<ObjRef, String>,
20    /// `~objectName=`'s store for a class object, keyed by identity, holding
21    /// only the classes something has renamed.
22    object_names: NameMap<ObjRef, String>,
23    /// Uppercased name -> identity, the registry's own lookup direction -- each
24    /// class is put into `.environment` under its uppercased name.
25    by_name: NameMap<String, ObjRef>,
26    /// The same direction for the kernel directory, a second table `.NAME`
27    /// never reaches. See [`ClassRegistry::define_system_class`].
28    by_system_name: NameMap<String, ObjRef>,
29    /// The bootstrap's private-method rows, in mint order -- see
30    /// [`ClassRegistry::private_native_methods`].
31    private_native_methods: Vec<MethodId>,
32    /// The bootstrap's protected instance- and class-method rows, in mint order
33    /// -- see [`ClassRegistry::protected_native_methods`].
34    protected_native_methods: Vec<MethodId>,
35}
36

38    fn default() -> Self {

64    /// Register `id` (already allocated by the caller) as a class
65    /// named `name`, and build its graph entry -- `~subclass`/bootstrap
66    /// construction plus the environment registration, fused into one call
67    /// since this crate models no separate install step.
68    pub fn define_reserved(
69        &mut self,
70        id: ObjRef,
71        name: &str,
72        superclass: Option<ObjRef>,
73        kind: ClassKind,
74        metaclass: ObjRef,
75    ) {
76        self.graph.define_class(id, superclass, kind, metaclass);
77        self.names.insert(id, name.to_string());
78        self.default_names.insert(id, format!("The {name} class"));
79        self.by_name.insert(name.to_ascii_uppercase(), id);
80    }

330    /// Drops every row keyed by a class the collector took.
331    pub fn expunge(&mut self, dead: &[ObjRef]) {
332        let ClassRegistry {
333            graph,
334            next_method: _,
335            names,
336            default_names,
337            object_names,
338            by_name,
339            by_system_name,
340            private_native_methods: _,
341            protected_native_methods: _,
342        } = self;
343        graph.expunge(dead);
344        for class in dead {
345            names.remove(class);
346            default_names.remove(class);
347            object_names.remove(class);
348        }
349        by_name.retain(|_, class| !dead.contains(class));
350        by_system_name.retain(|_, class| !dead.contains(class));
351    }

417    /// `~isA`/`~isSubclassOf` -- the same test as `~isA`: `class == other`, or
418    /// any of `class`'s ancestors is compatible with `other`, recursively over
419    /// the same `superClasses` list `~superClasses` reads (not the flattened
420    /// behaviour -- a mixin donated only through `inheritInstanceMethods` does
421    /// **not** make `~isA` true).
422    pub fn is_a(&self, class: ObjRef, other: ObjRef) -> bool {
423        class == other
424            || self
425                .graph
426                .ancestors(class)
427                .iter()
428                .any(|&sup| self.is_a(sup, other))
429    }

811/// Which bucket of the uninit table a class object with id `id` lands in.
812fn uninit_bucket(id: &[u8]) -> usize {
813    /// The default hash-table size, which `new_identity_table` builds the
814    /// uninit table at.
815    const BUCKETS: u64 = 17;
816    let hash = id.iter().fold(0u64, |hash, &byte| {
817        hash.wrapping_mul(31)
818            .wrapping_add(i64::from(byte.cast_signed()).cast_unsigned())
819    });
820    usize::try_from(hash % BUCKETS).expect("below BUCKETS")
821}
