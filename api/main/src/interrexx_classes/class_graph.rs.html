10/// The source of every [`ClassGraph::epoch`] value in the process, so that two
11/// graphs never share one: a stamp taken from one graph cannot match another.
12static NEXT_EPOCH: AtomicU64 = AtomicU64::new(1);
13
14/// A stamp of a graph's dispatch-visible state: see [`ClassGraph::epoch`].
15#[derive(Copy, Clone, PartialEq, Eq, Debug)]
16pub struct Epoch(u64);
17
18impl Epoch {
19    fn fresh() -> Epoch {
20        Epoch(NEXT_EPOCH.fetch_add(1, Ordering::Relaxed))
21    }
22}

25    fn default() -> Epoch {
26        Epoch::fresh()

41/// Why [`ClassGraph::inherit`] refused to build the edge, each variant named
42/// for the `SYNTAX` condition `~inherit` reports in its place.
43#[derive(Copy, Clone, PartialEq, Eq, Debug)]
44pub enum InheritRefusal {
45    /// `SYNTAX` 98.942: the target is not a
46    /// `MIXINCLASS`, so nothing may inherit it.
47    NotAMixin,
48    /// `SYNTAX` 98.944: one of
49    /// the two classes is already an ancestor of the other, in whichever
50    /// direction.
51    Recursive,
52    /// `SYNTAX` 98.943: the inheriting class does not already have the mixin's
53    /// base class in scope. The base class travels with the refusal because the
54    /// message names it.
55    BaseClass(ObjRef),
56    /// `SYNTAX` 98.945: a class named as a position or as an `~uninherit`
57    /// target is not one this class inherits. The class travels with the
58    /// refusal because the message names it, and it is the *named* class rather
59    /// than the mixin: `.K~inherit(.M1, .M2)` reports 98.945 with `.M2` as its
60    /// second insert.
61    NotInherited(ObjRef),
62}

72/// One flattened behaviour, plus a monotonic version bumped by every write to
73/// its dictionary, a handle's reuse included: the dictionary of a freed
74/// handle is cleared, which is a write.
75struct Behaviour {
76    dict: MethodDict,
77    version: u64,
78    /// Set by [`ClassGraph::set_primitive_instances`] on the instance
79    /// behaviour of a class whose instances answer from primitives. An object
80    /// that holds it is an instance of that class itself: a subclass and
81    /// `enhanced` give their objects a behaviour that
82    /// [`ClassGraph::alloc_behaviour`] allocates, which starts unset.
83    primitive: bool,
84}

86struct ClassDef {
87    /// `Regular` or `Mixin`. `inherit`'s mixin check reads this: only a `Mixin`
88    /// may be inherited.
89    kind: ClassKind,
90    /// Cascade order: the explicit superclass first (if any), then every mixin
91    /// `inherit` has folded in, in the order it was called.
92    superclasses: Vec<ObjRef>,
93    /// Every class that must be rebuilt when this one's own dictionaries
94    /// change: populated by both [`ClassGraph::define_class`] (the explicit
95    /// superclass) and [`ClassGraph::inherit`] (a mixin), and walked
96    /// identically either way, populated uniformly from both `~subclass` and
97    /// `~inherit`.
98    subclasses: Vec<ObjRef>,
99    /// This class's own instance methods, always at its own scope.
100    own_instance_methods: MethodDict,
101    /// This class's own class (static) methods.
102    own_class_methods: MethodDict,
103    /// Rebuilt in place by every change, never replaced: every instance
104    /// holds this handle, so each sees every change (`r[class.methods.late]`).
105    instance_behaviour: BehaviourHandle,
106    class_behaviour: BehaviourHandle,
107    /// What `inherit`'s base-class compatibility check reads.
108    base_class: ObjRef,
109    /// The class whose *instance* behaviour gets merged into this class's own
110    /// class-behaviour by [`ClassGraph::cascade_build`] (a class's class
111    /// behaviour includes its metaclass's instance behaviour). Every class the
112    /// bootstrap builds has `.Class` here, including `.Class` itself,
113    /// self-referentially: `.class~metaclass~id` is `"Class"`.
114    metaclass: ObjRef,
115    /// The class this class object's own behaviour belongs to, set from the
116    /// metaclass `~subclass` was given or defaulted to, and what `~class`
117    /// answers for a class object.
118    /// ```text
119    /// ::CLASS T  SUBCLASS MC METACLASS M1   ~metaClass MC     ~class M1      part
120    /// ::CLASS T2 SUBCLASS MC                ~metaClass MC     ~class Class   part
121    /// ::CLASS M3 SUBCLASS MC METACLASS MC   ~metaClass MC     ~class MC      same
122    /// ::class MC MIXINCLASS Class           ~metaClass Class  ~class Class   same
123    /// ::class Z  SUBCLASS Class             ~metaClass Class  ~class Class   same
124    /// ::CLASS K  METACLASS M1               ~metaClass M1     ~class M1      same
125    /// ```
126    owning_class: ObjRef,
127    /// This class may be named as another class's metaclass, which `~subclass`
128    /// tests before it will build anything from the metaclass it was handed.
129    is_metaclass: bool,
130    /// This class's instances need `UNINIT` run when they are collected, which
131    /// is what instance creation reads to register each new instance.
132    has_uninit: bool,
133    /// Some class this one inherits from carries `UNINIT`, set by each
134    /// constructor that builds an inheritance edge: `~subclass`, `~mixinClass`
135    /// and `~inherit`.
136    parent_has_uninit: bool,
137    /// `~new` on this class refuses, set by marking the class abstract and read
138    /// as the first step of instance creation.
139    is_abstract: bool,
140    /// This class may not be altered from Rexx. Image preparation sets it on every
141    /// class in the image, which is every class the image holds: the ones the native
142    /// bootstrap builds and the ones the interpreter's own Rexx-written library
143    /// declares alike. A `::CLASS` a program declares does not carry it.
144    rexx_defined: bool,
145    /// Built by [`ClassGraph::define_cloned_class`] and not yet given a
146    /// superclass by [`ClassGraph::reparent`].
147    cloned: bool,
148    /// For a class [`ClassGraph::define_cloned_class`] built, the class whose
149    /// instance behaviour its class behaviour copies, which
150    /// [`ClassGraph::rebuild_class_side`] copies again. Kept apart from
151    /// `owning_class`, which `~enhanced` repoints from the dummy subclass
152    /// the copy came from to the class it enhanced. Every other class has
153    /// its own id here.
154    cloned_from: ObjRef,
155}

157/// The class graph and its behaviour storage.
158#[derive(Default)]
159pub struct ClassGraph {
160    /// [`NameMap`] for `MethodDict`'s reason: the cascade reads this map
161    /// once per superclass per class while the library is being built.
162    classes: NameMap<ObjRef, ClassDef>,
163    behaviours: Vec<Behaviour>,
164    /// The behaviours an expunged class emptied, which a later allocation
165    /// takes before it grows [`ClassGraph::behaviours`]. Only an instance of
166    /// a class holds one of its handles, and an instance keeps its class
167    /// alive.
168    free_behaviours: Vec<BehaviourHandle>,
169    /// The methods a `define`, `defineMethods` or `delete` took out of a
170    /// class's own instance dictionary, a `define` with no method or with
171    /// `.nil` included, since [`ClassGraph::take_unnamed_methods`],
172    /// each with that class: a method no dictionary names any more can be let
173    /// go of. The image's classes record none.
174    dropped: HashSet<(ObjRef, MethodId)>,
175    /// The uninit table restricted to class objects: the entries the uninit
176    /// check makes, in the order it makes them.
177    uninit_classes: Vec<ObjRef>,
178    /// See [`ClassGraph::epoch`].
179    epoch: Epoch,
180    /// When each instance behaviour's inputs last changed and when each was
181    /// built, which says whether [`ClassGraph::define`] may recompute only
182    /// the names it changed.
183    inputs: InputTicks,
184    /// Bumped by every change to which classes exist or to any class's
185    /// `superclasses`: a define, a reparent, an `inherit`, an `uninherit` and
186    /// an expunge. An answer derived from the ancestry ([`Self::ancestors`]
187    /// read transitively, as `is_a` does) holds while this stays put. A define
188    /// counts because an identity an expunge freed can be minted again.
189    ancestry: u64,
190}

192/// A clock over the cascade's instance-side inputs -- each class's own
193/// instance dictionary and superclass list -- read by the mutators that
194/// change one name ([`ClassGraph::define`] and its siblings): a behaviour
195/// built after every input it was built from last changed is the cascade of
196/// those inputs, so the change can be applied to that one name of it rather
197/// than rebuilding it whole. A behaviour built before is rebuilt whole, as a
198/// class an update never reached (a subclass a failed `subclass` left out
199/// of its superclass's list) needs.
200#[derive(Default)]
201struct InputTicks {
202    /// Raised by every change of an input.
203    now: u64,
204    /// When each class's inputs last changed; a class not in the map has
205    /// not changed since the clock started.
206    changed: NameMap<ObjRef, u64>,
207    /// When each behaviour, by index, was last built by the cascade.
208    built: Vec<u64>,
209}
210
211impl InputTicks {
212    /// Records a change of `class`'s own instance dictionary or superclasses.
213    fn touch(&mut self, class: ObjRef) {
214        self.now += 1;
215        self.changed.insert(class, self.now);
216    }
217
218    fn changed(&self, class: ObjRef) -> u64 {
219        self.changed.get(&class).copied().unwrap_or(0)
220    }
221
222    fn set_built(&mut self, handle: BehaviourHandle) {
223        let index = handle.index();
224        if self.built.len() <= index {
225            self.built.resize(index + 1, 0);
226        }
227        self.built[index] = self.now;
228    }
229
230    fn built(&self, handle: BehaviourHandle) -> u64 {
231        self.built.get(handle.index()).copied().unwrap_or(0)
232    }
233}

247    /// still answers it.

266    fn alloc_behaviour(&mut self) -> BehaviourHandle {
267        if let Some(handle) = self.free_behaviours.pop() {
268            self.behaviours[handle.index()].primitive = false;
269            return handle;
270        }
271        let handle = BehaviourHandle::new(self.behaviours.len());
272        self.behaviours.push(Behaviour {
273            dict: MethodDict::new(),
274            version: 0,
275            primitive: false,
276        });
277        handle
278    }

280    /// Register a new class and build its two initial behaviours by cascading
281    /// from `superclass`. `superclass` is `None` only for a root class with no
282    /// ancestor (a test's own stand-in for `.Object`).
283    pub fn define_class(
284        &mut self,
285        id: ObjRef,
286        superclass: Option<ObjRef>,
287        kind: ClassKind,
288        metaclass: ObjRef,
289    ) {
290        self.bump_epoch();
291        let derived_from_metaclass = superclass.filter(|sup| self.classes[sup].is_metaclass);
292        let owning_class = metaclass;
293        let metaclass = derived_from_metaclass.unwrap_or(metaclass);
294        let (base_class, parent_has_uninit) = match kind {
295            ClassKind::Regular => (id, superclass.is_some_and(|sup| self.uninit_reaches(sup))),
296            ClassKind::Mixin => {
297                let target = superclass.expect("a mixin class names its MIXINCLASS target");
298                (
299                    self.classes[&target].base_class,
300                    self.uninit_reaches(target),
301                )
302            }
303        };
304        self.ancestry_changed();
305        let instance_behaviour = self.alloc_behaviour();
306        let class_behaviour = self.alloc_behaviour();
307        self.classes.insert(
308            id,
309            ClassDef {
310                kind,
311                superclasses: superclass.into_iter().collect(),
312                subclasses: Vec::new(),
313                own_instance_methods: MethodDict::new(),
314                own_class_methods: MethodDict::new(),
315                instance_behaviour,
316                class_behaviour,
317                base_class,
318                metaclass,
319                owning_class,
320                is_metaclass: derived_from_metaclass.is_some(),
321                has_uninit: false,
322                parent_has_uninit,
323                is_abstract: false,
324                rexx_defined: false,
325                cloned: false,
326                cloned_from: id,
327            },
328        );
329        if let Some(sup) = superclass {
330            self.classes.get_mut(&sup).unwrap().subclasses.push(id);
331        }
332        self.inputs.touch(id);
333        self.rebuild_behaviour(id, Side::Instance);
334        self.rebuild_behaviour(id, Side::Class);
335    }

337    /// A class `~new` makes: a copy of
338    /// `receiver` whose class behaviour is a copy of `receiver`'s instance
339    /// behaviour, whose instance side derives from `root` alone, and which no
340    /// subclass list holds. `metaclass` is `receiver` where it is not one of
341    /// the image's own classes, and `.Class` where it is.
342    pub fn define_cloned_class(
343        &mut self,
344        id: ObjRef,
345        receiver: ObjRef,
346        root: ObjRef,
347        metaclass: ObjRef,
348    ) {
349        self.bump_epoch();
350        self.ancestry_changed();
351        let instance_behaviour = self.alloc_behaviour();
352        let class_behaviour = self.alloc_behaviour();
353        self.classes.insert(
354            id,
355            ClassDef {
356                kind: ClassKind::Regular,
357                superclasses: vec![root],
358                subclasses: Vec::new(),
359                own_instance_methods: MethodDict::new(),
360                own_class_methods: MethodDict::new(),
361                instance_behaviour,
362                class_behaviour,
363                base_class: id,
364                metaclass,
365                owning_class: receiver,
366                is_metaclass: false,
367                has_uninit: false,
368                parent_has_uninit: false,
369                is_abstract: false,
370                rexx_defined: false,
371                cloned: true,
372                cloned_from: receiver,
373            },
374        );
375        self.inputs.touch(id);
376        self.rebuild_behaviour(id, Side::Instance);
377        let source = self.classes[&receiver].instance_behaviour;
378        let dict = self.behaviours[source.index()].dict.clone();
379        let behaviour = &mut self.behaviours[class_behaviour.index()];
380        behaviour.dict = dict;
381        behaviour.version += 1;
382    }

384    /// A behaviour of its own holding `class`'s current instance methods with
385    /// each of `hidden` hidden in front, as `defineInstanceMethod` with a
386    /// `.nil` method does to one object's behaviour copy.
387    pub fn hiding_behaviour(&mut self, class: ObjRef, hidden: &[&str]) -> BehaviourHandle {
388        self.bump_epoch();
389        let source = self.classes[&class].instance_behaviour;
390        let mut dict = self.behaviours[source.index()].dict.clone();
391        for name in hidden {
392            dict.add_hidden(name);
393        }
394        let handle = self.alloc_behaviour();
395        let behaviour = &mut self.behaviours[handle.index()];
396        behaviour.dict = dict;
397        behaviour.version += 1;
398        handle
399    }

421    /// `~subclass`'s rewiring of the class its metaclass's `NEW`
422    /// answered: `superclass` becomes its
423    /// only superclass, it becomes a metaclass when `superclass` is one,
424    /// `metaclass` owns its class behaviour, and both behaviours are rebuilt
425    /// in place. Its own method dictionaries are kept.
426    pub fn reparent(&mut self, id: ObjRef, superclass: ObjRef, kind: ClassKind, metaclass: ObjRef) {
427        self.bump_epoch();
428        let derived_from_metaclass = self.classes[&superclass].is_metaclass;
429        let base_class = match kind {
430            ClassKind::Regular => id,
431            ClassKind::Mixin => self.classes[&superclass].base_class,
432        };
433        let parent_has_uninit = self.uninit_reaches(superclass);
434        self.ancestry_changed();
435        let old = std::mem::take(&mut self.classes.get_mut(&id).unwrap().superclasses);
436        for sup in old {
437            if let Some(def) = self.classes.get_mut(&sup) {
438                def.subclasses.retain(|sub| *sub != id);
439            }
440        }
441        let def = self.classes.get_mut(&id).unwrap();
442        def.kind = kind;
443        def.superclasses = vec![superclass];
444        def.base_class = base_class;
445        if derived_from_metaclass {
446            def.is_metaclass = true;
447            def.metaclass = superclass;
448        }
449        def.owning_class = metaclass;
450        def.parent_has_uninit = parent_has_uninit;
451        def.cloned = false;
452        def.cloned_from = id;
453        self.classes
454            .get_mut(&superclass)
455            .unwrap()
456            .subclasses
457            .push(id);
458        self.inputs.touch(id);
459        self.rebuild_behaviour(id, Side::Instance);
460        self.rebuild_behaviour(id, Side::Class);
461    }

488    /// `~enhanced`'s merge of its methods into the dummy subclass's
489    /// own instance dictionary, each defined at `scope`, `None` hiding the

491    pub fn enhance_instance_methods(
492        &mut self,
493        class: ObjRef,
494        scope: ObjRef,

536    /// The uninit check, both halves of it: set [`Self::has_uninit`] when the
537    /// class's **flattened** instance behaviour answers `UNINIT`, whether the
538    /// class defines it or inherits it, and enter the class object itself in
539    /// the `uninit_classes` list when its **class** behaviour answers `UNINIT`
540    /// -- which is where a `::METHOD uninit CLASS` lands.
541    pub fn check_uninit(&mut self, class: ObjRef) {
542        let instance = self.behaviour_handle(class, Side::Instance);
543        if self.behaviours[instance.index()].dict.has_method("UNINIT") {
544            self.classes.get_mut(&class).unwrap().has_uninit = true;
545        }
546        let class_side = self.behaviour_handle(class, Side::Class);
547        if self.behaviours[class_side.index()]
548            .dict
549            .has_method("UNINIT")
550            && !self.uninit_classes.contains(&class)
551        {
552            self.uninit_classes.push(class);
553        }
554    }

556    /// Drops every trace of `dead` from the graph, and scrubs the subclass
557    /// lists of the classes that survive.
558    pub fn expunge(&mut self, dead: &[ObjRef]) {
559        self.bump_epoch();
560        self.ancestry_changed();
561        let ClassGraph {
562            classes,
563            behaviours,
564            free_behaviours,
565            dropped,
566            uninit_classes,
567            epoch: _,
568            inputs,
569            ancestry: _,
570        } = self;
571        for class in dead {
572            // The dictionaries are the weight: a flattened behaviour holds
573            // every method the class answers, inherited ones included.
574            // Emptied rather than removed, because a behaviour is named by
575            // its index and removing one would move every later index.
576            if let Some(def) = classes.get(class) {
577                for handle in [def.instance_behaviour, def.class_behaviour] {
578                    let behaviour = &mut behaviours[handle.index()];
579                    behaviour.dict.clear();
580                    behaviour.version += 1;
581                    debug_assert!(!free_behaviours.contains(&handle), "{handle:?} freed twice");
582                    free_behaviours.push(handle);
583                }
584            }
585            classes.remove(class);
586            inputs.changed.remove(class);
587            uninit_classes.retain(|held| held != class);
588        }
589        // A subclass list holds classes that may have died: the list is of weak
590        // references and pruned as it is read.
591        for def in classes.values_mut() {

621    /// Answers each dropped method that no behaviour and no class's own
622    /// dictionary names any more, with the class that dropped it, and
623    /// forgets every dropped method.
624    pub fn take_unnamed_methods(&mut self) -> Vec<(ObjRef, MethodId)> {
625        let ClassGraph {
626            classes,
627            behaviours,
628            dropped,
629            ..
630        } = self;
631        let mut unnamed: Vec<(ObjRef, MethodId)> = dropped.drain().collect();
632        if !unnamed.is_empty() {
633            // Only the dropped methods are in question, a few
634            // hundred where the dictionaries hold tens of thousands of
635            // entries: probe the entries against those, oldest first, and pass
636            // by every dictionary whose newest method is older than the oldest
637            // still open -- which, once the inherited methods are found, is
638            // every dictionary built before the dropped methods were minted.
639            let mut open: Vec<u32> = unnamed.iter().map(|(_, method)| method.0).collect();
640            open.sort_unstable();
641            open.dedup();
642            let mut named = vec![false; open.len()];
643            let mut oldest = 0;
644            let dicts = behaviours.iter().map(|behaviour| &behaviour.dict).chain(
645                classes
646                    .values()
647                    .flat_map(|def| [&def.own_instance_methods, &def.own_class_methods]),
648            );
649            'scan: for dict in dicts {
650                if dict.newest().is_none_or(|newest| newest.0 < open[oldest]) {
651                    continue;
652                }
653                for method in dict.methods() {
654                    let Ok(at) = open.binary_search(&method.0) else {
655                        continue;
656                    };
657                    named[at] = true;
658                    while named[oldest] {
659                        oldest += 1;
660                        if oldest == open.len() {
661                            break 'scan;
662                        }
663                    }
664                }
665            }
666            unnamed.retain(|(_, method)| open.binary_search(&method.0).is_ok_and(|at| !named[at]));
667            unnamed.sort_unstable_by_key(|(class, method)| (class.bits(), method.0));
668            unnamed.dedup();
669        }
670        unnamed
671    }

738    /// The instance- and class-behaviour construction, unified over which
739    /// per-class dictionary side is folded in. Ancestors are processed
740    /// furthest-inherited-first (`superclasses.iter().rev()`), skipping a
741    /// superclass already present -- the diamond guard -- then this class's own
742    /// dictionary is folded in last. The most recently folded-in ancestor
743    /// always wins a name conflict, which is the merge-order rule the mixin and
744    /// diamond tests check: the first-listed `INHERIT` mixin is folded in
745    /// *last* among the mixins (closest to `class`'s own scope in cascade
746    /// order), and an explicit `SUBCLASS` target is folded in last of all the
747    /// ancestors, so it outranks every mixin.
748    fn cascade_build(
749        classes: &NameMap<ObjRef, ClassDef>,
750        behaviours: &[Behaviour],
751        class: ObjRef,
752        target: &mut MethodDict,
753        side: Side,
754    ) {
755        let superclasses = classes[&class].superclasses.clone();
756        for sup in superclasses.into_iter().rev() {
757            if !target.has_scope(sup) {
758                Self::cascade_build(classes, behaviours, sup, target, side);
759            }
760        }
761        if !target.has_scope(class) {
762            if side == Side::Class && !classes[&class].superclasses.is_empty() {
763                let metaclass = classes[&class].metaclass;
764                if !target.has_scope(metaclass) {
765                    let meta_instance = classes[&metaclass].instance_behaviour;
766                    target.merge(&behaviours[meta_instance.index()].dict);
767                }
768            }
769            let own = match side {
770                Side::Instance => &classes[&class].own_instance_methods,
771                Side::Class => &classes[&class].own_class_methods,
772            };
773            target.merge_methods(own);
774            target.add_scope(class);
775        }
776    }

778    /// The rebuild of a behaviour: clear the *existing* dictionary in place, then cascade into
779    /// it, rather than building a separate dictionary and swapping it in. The handle's
780    /// identity is unaffected either way.
781    fn rebuild_behaviour(&mut self, class: ObjRef, side: Side) {
782        let handle = self.behaviour_handle(class, side);
783        let mut dict = std::mem::take(&mut self.behaviours[handle.index()].dict);
784        dict.clear();
785        dict.reserve(self.rebuilt_names(class, side));
786        Self::cascade_build(&self.classes, &self.behaviours, class, &mut dict, side);
787        let behaviour = &mut self.behaviours[handle.index()];
788        behaviour.dict = dict;
789        behaviour.version += 1;
790        if side == Side::Instance {
791            self.inputs.set_built(handle);
792        }
793    }

827    /// `~define`. The instance behaviours of `class` and of every subclass
828    /// are updated in place, so the instances made before the call see the
829    /// change as the ones made after it do (`r[class.methods.late]`), and
830    /// so do the class objects whose metaclass is among them.
831    pub fn define(&mut self, class: ObjRef, name: &str, method: MethodId) {
832        self.bump_epoch();

835        self.note_dropped(class, name);
836        let def = self.classes.get_mut(&class).expect("define: unknown class");
837        def.own_instance_methods.replace_method(name, class, method);
838        if name.eq_ignore_ascii_case("UNINIT") {
839            def.has_uninit = true;
840        }
841        self.redefine(class, prior, kept, &[name], true);
842    }

844    /// [`ClassGraph::define`] of each of `methods` in turn, with one cascade instead of
845    /// one each.
846    ///
847    /// The two differ only in what nothing can see: the behaviours the calls in between
848    /// would have built, which nothing can read because nothing runs between the calls,
849    /// and how often a rebuilt behaviour's version moves. Where that equivalence needs
850    /// more than this checks -- a name given twice, a name `class` already defines,
851    /// `UNINIT` -- the calls are made one by one.
852    pub fn define_all(&mut self, class: ObjRef, methods: &[(String, MethodId)]) {
853        self.bump_epoch();
854        let prior = self.inputs.changed(class);
855        let def = &self.classes[&class];
856        let mut names: Vec<String> = methods
857            .iter()
858            .map(|(name, _)| name.to_ascii_uppercase())
859            .collect();
860        let together = methods.len() > 1
861            && names
862                .iter()
863                .all(|name| name != "UNINIT" && !def.own_instance_methods.contains(name))
864            && {
865                names.sort_unstable();
866                names.windows(2).all(|pair| pair[0] != pair[1])
867            };
868        if !together {
869            for (name, method) in methods {
870                self.define(class, name, *method);
871            }
872            return;
873        }
874        let def = self
875            .classes
876            .get_mut(&class)
877            .expect("define_all: unknown class");
878        for (name, method) in methods {
879            def.own_instance_methods
880                .replace_method(name, class, *method);
881        }
882        // Every name is new to the class, so the behaviours are rebuilt whole.
883        self.redefine(class, prior, false, &[], true);
884    }

970    /// What follows a change to the entries under `names` in `class`'s own
971    /// instance dictionary, whose inputs had last changed at `prior`: the
972    /// update in place of its instance behaviour and of every subclass's
973    /// (only the instance side), each followed by the uninit check where one
974    /// of `names` is `UNINIT`, the one change that can make it resolve, and
975    /// then the class behaviours that follow them
976    /// ([`Self::follow_metaclasses`]). Without `cascade` nothing changed, and
977    /// nothing is updated.
978    ///
979    /// A behaviour [`Self::built_from_current`] clears has only `names`
980    /// recomputed, which is all a rebuild would change of it, where `kept`
981    /// says each of `names` was in the class's own dictionary before the
982    /// change and still is: then every dictionary the cascade reads lists its
983    /// names in the order it did, a rebuild would enter the names in the
984    /// order it did, and the behaviour lists them, as `Class~methods` does,
985    /// in the order a rebuild would. Any other is rebuilt whole.
986    fn redefine(&mut self, class: ObjRef, prior: u64, kept: bool, names: &[&str], cascade: bool) {
987        if !cascade {
988            return;
989        }
990        self.inputs.touch(class);
991        let uninit = names.iter().any(|name| name.eq_ignore_ascii_case("UNINIT"));
992        // A table may name one method twice; each name is recomputed once.
993        let mut keys: Vec<String> = Vec::new();
994        if kept {
995            keys = names.iter().map(|name| name.to_ascii_uppercase()).collect();
996            keys.sort_unstable();
997            keys.dedup();
998        }
999        let handle = self.classes[&class].instance_behaviour;
1000        if kept && self.built_from_current(handle, class, class, prior) {
1001            self.recompute_names(class, &keys);
1002        } else {
1003            self.rebuild_behaviour(class, Side::Instance);
1004        }
1005        if uninit {
1006            self.check_uninit(class);
1007        }
1008        let subclasses = self.classes[&class].subclasses.clone();
1009        for sub in subclasses {
1010            self.redefine_below(sub, class, prior, kept, &keys, uninit);
1011        }
1012        self.follow_metaclasses(class);
1013    }

1015    /// The class behaviours that merged an instance behaviour a change to
1016    /// `changed` rebuilt -- its own or a subclass's, where that class is a
1017    /// class object's metaclass -- rebuilt in their turn, so a class object
1018    /// follows every change to its metaclass and the classes behind it
1019    /// (`r[class.methods.late]`).
1020    fn follow_metaclasses(&mut self, changed: ObjRef) {
1021        let mut reached = vec![changed];
1022        let mut at = 0;
1023        while at < reached.len() {
1024            for &sub in &self.classes[&reached[at]].subclasses {
1025                if !reached.contains(&sub) {
1026                    reached.push(sub);
1027                }
1028            }
1029            at += 1;
1030        }
1031        let merged: Vec<ObjRef> = reached
1032            .into_iter()
1033            .filter(|&class| {
1034                self.classes
1035                    .values()
1036                    .any(|def| def.metaclass == class || (def.cloned && def.cloned_from == class))
1037            })
1038            .collect();
1039        if merged.is_empty() {
1040            return;
1041        }
1042        let followers: Vec<ObjRef> = self
1043            .classes
1044            .iter()
1045            .filter(|(_, def)| {
1046                let dict = &self.behaviours[def.class_behaviour.index()].dict;
1047                merged.iter().any(|&class| dict.has_scope(class))
1048            })
1049            .map(|(&class, _)| class)
1050            .collect();
1051        for class in followers {
1052            self.rebuild_class_side(class);
1053            self.check_uninit(class);
1054        }
1055    }

1058    /// class `~new` made copies its metaclass's instance behaviour, the root
1059    /// class merges its metaclass's after its own cascade, as the bootstrap
1060    /// builds it ([`Self::bootstrap_root_class_behaviour`]), and every other

1121    /// Whether `handle`, `class`'s instance behaviour, was built after every
1122    /// input of its cascade last changed, taking `changed`'s last change to
1123    /// be at `prior`: then it is the cascade of the inputs as they stood
1124    /// before `changed`'s latest change.
1125    fn built_from_current(
1126        &self,
1127        handle: BehaviourHandle,
1128        class: ObjRef,
1129        changed: ObjRef,
1130        prior: u64,
1131    ) -> bool {
1132        let built = self.inputs.built(handle);
1133        let mut pending = vec![class];
1134        while let Some(at) = pending.pop() {
1135            let last = if at == changed {
1136                prior
1137            } else {
1138                self.inputs.changed(at)
1139            };
1140            if last > built {
1141                return false;
1142            }
1143            pending.extend_from_slice(&self.classes[&at].superclasses);
1144        }
1145        true
1146    }

1148    /// The entries under `keys` of `class`'s instance behaviour made again
1149    /// by the cascade, every other name left as it stands, as a rebuild of a
1150    /// behaviour that differs from its inputs' cascade only under `keys`
1151    /// would make it.
1152    fn recompute_names(&mut self, class: ObjRef, keys: &[String]) {
1153        let mut made = MethodDict::new();
1154        Self::cascade_names(&self.classes, class, &mut made, keys);
1155        let handle = self.classes[&class].instance_behaviour;
1156        let behaviour = &mut self.behaviours[handle.index()];
1157        for key in keys {
1158            behaviour.dict.adopt_entry(&mut made, key);
1159        }
1160        behaviour.version += 1;
1161        self.inputs.set_built(handle);
1162        #[cfg(debug_assertions)]
1163        {
1164            let mut whole = MethodDict::new();
1165            whole.reserve(self.rebuilt_names(class, Side::Instance));
1166            Self::cascade_build(
1167                &self.classes,
1168                &self.behaviours,
1169                class,

1239    /// `~inherit`'s own two-argument form: the same operation as
1240    /// [`Self::inherit`], with the optional `position` the `INHERIT` keyword
1241    /// of a `::CLASS` directive cannot supply.
1242    pub fn inherit_at(
1243        &mut self,
1244        class: ObjRef,
1245        mixin: ObjRef,
1246        position: Option<ObjRef>,
1247    ) -> Result<(), InheritRefusal> {
1248        self.bump_epoch();
1249        if !matches!(self.classes[&mixin].kind, ClassKind::Mixin) {
1250            return Err(InheritRefusal::NotAMixin);
1251        }
1252        if self.behaviour_has_scope(class, Side::Class, mixin) {
1253            return Err(InheritRefusal::Recursive);
1254        }
1255        if self.behaviour_has_scope(mixin, Side::Class, class) {
1256            return Err(InheritRefusal::Recursive);
1257        }
1258        let mixin_base = self.classes[&mixin].base_class;
1259        if !self.behaviour_has_scope(class, Side::Class, mixin_base) {
1260            return Err(InheritRefusal::BaseClass(mixin_base));
1261        }
1262        if !self.behaviour_has_scope(class, Side::Instance, mixin_base) {
1263            return Err(InheritRefusal::BaseClass(mixin_base));
1264        }
1265        let at = match position {
1266            None => self.classes[&class].superclasses.len(),
1267            Some(position) => {
1268                match self.classes[&class]
1269                    .superclasses
1270                    .iter()
1271                    .position(|&sup| sup == position)
1272                {
1273                    Some(at) => at,
1274                    None => return Err(InheritRefusal::NotInherited(position)),
1275                }
1276            }
1277        };
1278        self.classes
1279            .get_mut(&class)
1280            .unwrap()
1281            .superclasses
1282            .insert(at, mixin);
1283        self.inputs.touch(class);
1284        self.ancestry_changed();
1285        self.classes.get_mut(&mixin).unwrap().subclasses.push(class);
1286        self.update_sub_classes(class);
1287        self.follow_metaclasses(class);
1288        if self.uninit_reaches(mixin) {
1289            self.classes.get_mut(&class).unwrap().parent_has_uninit = true;
1290        }
1291        Ok(())
1292    }

1294    /// `~uninherit`: take a mixin back out of the superclass list --
1295    /// `~uninherit`.
1296    pub fn uninherit(&mut self, class: ObjRef, mixin: ObjRef) -> Result<(), InheritRefusal> {
1297        self.bump_epoch();
1298        if !matches!(self.classes[&mixin].kind, ClassKind::Mixin) {
1299            return Err(InheritRefusal::NotAMixin);
1300        }
1301        let at = self.classes[&class]
1302            .superclasses
1303            .iter()
1304            .position(|&sup| sup == mixin)
1305            .filter(|at| *at > 0);
1306        let Some(at) = at else {
1307            return Err(InheritRefusal::NotInherited(mixin));
1308        };
1309        self.classes
1310            .get_mut(&class)
1311            .unwrap()
1312            .superclasses
1313            .remove(at);
1314        self.inputs.touch(class);
1315        self.ancestry_changed();
1316        // The subclass-list removal takes out the one entry it finds, and a
1317        // class reaches a mixin's subclass list once: a second `~inherit` of
1318        // the same mixin is the recursive refusal above.
1319        if let Some(sub) = self.classes[&mixin]
1320            .subclasses
1321            .iter()
1322            .position(|&sub| sub == class)
1323        {
1324            self.classes.get_mut(&mixin).unwrap().subclasses.remove(sub);
1325        }
1326        self.update_sub_classes(class);
1327        self.follow_metaclasses(class);
1328        Ok(())
1329    }

1354    /// dictionary, then rebuilds `class`'s instance behaviour in place -- **no

1361            &mut self
1362                .classes
